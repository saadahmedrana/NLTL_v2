ice:MinimumEnginePowerShape a sh:NodeShape ;
    sh:targetNode ssp:iceShip ;
    sh:message "Ship does not meet TRAFICOM minimum engine power requirements." ;
    
    sh:sparql [
        sh:message "Engine power ({?actualPower} kW) is below the minimum 2800 kW required for IA Super ice class." ;
        sh:select """
            PREFIX vsso: <https://github.com/w3c/vsso#>
            PREFIX ssp: <https://w3id.org/mtl-requirements/ssp#>
            PREFIX fmu: <https://w3id.org/mtl-requirements/fmu#>
            
            SELECT $this ?actualPower ?iceClass
            WHERE {
                $this ssp:iceClass ?iceClass .
                
                $this ssp:hasComponent ?engine .
                
                ?engine ssp:hasVariable ssp:EngineMinPower .
                ssp:EngineMinPower ssp:hasVariableValue ?actualPower .
                
                FILTER(?iceClass = 'IA Super' && ?actualPower < 2800.0)
            }
        """
    ] ;
    
    sh:sparql [
        sh:message "Engine power ({?actualPower} kW) is below the minimum 1000 kW required for IA, IB, or IC ice class." ;
        sh:select """
            PREFIX vsso: <https://github.com/w3c/vsso#>
            PREFIX ssp: <https://w3id.org/mtl-requirements/ssp#>
            PREFIX fmu: <https://w3id.org/mtl-requirements/fmu#>
            
            SELECT $this ?actualPower ?iceClass
            WHERE {
                # Get the ship's ice class
                $this ssp:iceClass ?iceClass .
                
                # Get the engine component
                $this ssp:hasComponent ?engine .
                
                # Get the engine power through the variable
                ?engine ssp:hasVariable ssp:EngineMinPower .
                ssp:EngineMinPower ssp:hasVariableValue ?actualPower .
                
                # Check if power is below requirement for other ice classes
                FILTER((?iceClass = 'IA' || ?iceClass = 'IB' || ?iceClass = 'IC') && ?actualPower < 1000.0)
            }
        """
    ] .

ice:IceLoadHeightShape a sh:NodeShape ;
    sh:targetNode ssp:iceLoadMaxHeight ;
    sh:message "Ice load height does not meet requirements." ;
    
    sh:sparql [
        sh:message "Ice load height h_max ({?height} m) does not match the required value of {?requiredHeight} m for {?iceClass} ice class." ;
        sh:select """
            PREFIX ssp: <https://w3id.org/mtl-requirements/ssp#> 
            PREFIX fmu: <https://w3id.org/mtl-requirements/fmu#> 
            PREFIX ssn: <http://www.w3.org/ns/ssn/> 
            
            SELECT $this ?height ?requiredHeight ?iceClass
            WHERE {
                # height value 
                ssp:iceLoadMaxHeight ssp:hasVariableValue ?height .
                
                ssp:iceLoadMaxHeight ssn:isPropertyOf ?hull .
                ?hull ssn:isPropertyOf ?ship .
                ?ship ssp:iceClass ?iceClass .
                
                # Use conditional binding to determine required height
                BIND(
                    IF(?iceClass = "IA Super", 0.35,
                    IF(?iceClass = "IA", 0.30,
                    IF(?iceClass = "IB", 0.25,
                    IF(?iceClass = "IC", 0.22, 0))))
                    AS ?requiredHeight
                )
                
                # Direct comparison
                FILTER(?height != ?requiredHeight)
            }
        """
    ] .

ice:LoadDistributionFactorShape a sh:NodeShape ;
    sh:targetNode ssp:Hull001 ;
    sh:message "Load distribution factor (f4) validation failed." ;
    
    sh:sparql [
        sh:message "Load distribution factor ({?actualFactor}) does not match the calculated value ({?calculatedF4}) based on load area and frame spacing." ;
        sh:select """
            PREFIX ssp: <https://w3id.org/mtl-requirements/ssp#>
            PREFIX ssn: <http://www.w3.org/ns/ssn/> 
            
            SELECT $this ?actualFactor ?calculatedF4
            WHERE {
                # Get the stored load distribution factor
                ssp:loadDistribution ssn:isPropertyOf $this .
                ssp:loadDistribution ssp:hasVariableValue ?actualFactor .
                
                # Get relevant ship parameters
                $this ssn:isPropertyOf ?ship .
                ssp:loadArea ssn:isPropertyOf ?ship .
                ssp:loadArea ssp:hasVariableValue ?h .
                ssp:frameSpacing ssn:isPropertyOf ?ship .
                ssp:frameSpacing ssp:hasVariableValue ?s .
                                
                # Calculate f4 according to regulations: f4 = (1 - 0.2 h/s)
                BIND((1 - (0.2 * (?h / ?s))) AS ?calculatedF4)
                
                # Check if the stored value matches calculated value (with small tolerance)
                FILTER(ABS(?actualFactor - ?calculatedF4) > 0.01)
            }
        """
    ] .
