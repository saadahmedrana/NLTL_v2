<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloTON_Metric
            ]
        ] ;
        sh:message "The ship must provide exactly one upper ice waterline displacement in kilotonnes."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:KiloTON_Metric
                    ]
                ]
            ]
        ] ;
        sh:message "At least one upper ice waterline case with a displacement in kilotonnes must be provided."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline displacement must equal the greatest displacement among the linked upper ice waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiNode .
                ?duiNode qudt:numericValue ?duiValue .

                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?case .
                    ?case nltl:waterlineCaseDisplacement ?caseDisplacement .
                    ?caseDisplacement qudt:numericValue ?caseValue .
                    FILTER (?caseValue > ?duiValue)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?matchingCase .
                    ?matchingCase nltl:waterlineCaseDisplacement ?matchingDisplacement .
                    ?matchingDisplacement qudt:numericValue ?matchingValue .
                    FILTER (?matchingValue = ?duiValue)
                }
            }
        """
    ] .
<END_SHACL>
