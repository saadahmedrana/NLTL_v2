<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_006_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:asternIceOperationIntent ;
        sh:datatype xsd:boolean ;
        sh:message "Astern ice operation intent, when recorded, must be a boolean."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship is intended to operate astern in ice regions, its aft section design basis must be \"Bow and Bow Intermediate hull area requirements\"." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:asternIceOperationIntent true .
                FILTER NOT EXISTS {
                    $this nltl:aftSectionDesignBasis "Bow and Bow Intermediate hull area requirements"^^xsd:string .
                }
            }
        """
    ] .
<END_SHACL>
