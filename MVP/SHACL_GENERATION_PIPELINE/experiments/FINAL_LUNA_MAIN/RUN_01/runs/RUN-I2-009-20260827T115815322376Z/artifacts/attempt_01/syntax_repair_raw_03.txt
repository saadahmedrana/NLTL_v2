        sh:maxCount 1
    ] .

        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-009-HullAreaAssignment>
    a sh:NodeShape ;
    sh:class nltl:hullAreaAssignment ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:hasValue nltl:otherIceStrengthenedArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The hull area assignment must identify an other ice-strengthened area."
    ] .

<urn:nltl:generated-shape:I2-009-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:node <urn:nltl:generated-shape:I2-009-HullAreaAssignment> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide the hull area assignment for this requirement."
    ] ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node <urn:nltl:generated-shape:I2-009-MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Average ice pressure must be supplied in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node <urn:nltl:generated-shape:I2-009-Unitless> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load patch aspect ratio must be supplied as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node <urn:nltl:generated-shape:I2-009-Metre> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The non-bow load patch height must be supplied in metres."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node <urn:nltl:generated-shape:I2-009-Metre> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The non-bow load patch length must be supplied in metres."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node <urn:nltl:generated-shape:I2-009-Degree> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The buttock angle must be supplied in degrees."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node <urn:nltl:generated-shape:I2-009-Degree> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The normal frame angle must be supplied in degrees."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "3.6"^^xsd:decimal ;
                sh:maxInclusive "3.6"^^xsd:decimal ;
                sh:message "The fixed load patch aspect ratio must be 3.6."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow form applicability must be true exactly when the buttock angle is positive and below 80 degrees and the normal frame angle exceeds 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:buttockAngle ?gamma ;
                      nltl:normalFrameAngle ?beta ;
                      nltl:bowFormApplicability ?app .
                ?gamma qudt:numericValue ?gammaValue .
                ?beta qudt:numericValue ?betaValue .
                FILTER (
                    (
                        ?gammaValue > 0 &&
                        ?gammaValue < 80 &&
                        ?betaValue > 10 &&
                        ?app != true
                    ) ||
                    (
                        (
                            ?gammaValue <= 0 ||
                            ?gammaValue >= 80 ||
                            ?betaValue <= 10
                        ) &&
                        ?app != false
                    )
                )
            }
        """
    ] .
