<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:I2_011_ApplicablePc6Shape
        gen:I2_011_ApplicablePc7Shape
        gen:I2_011_NonApplicableBulbousShape
        gen:I2_011_NonApplicablePolarShape
    ) ;
    sh:message "The PC6/PC7 bulbous-bow applicability condition must be explicitly established, and applicable ships must satisfy the bow-force constraints." .

gen:I2_011_ApplicablePc6Shape
    a sh:NodeShape ;
    sh:and (
        gen:I2_011_ApplicableBaseShape
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc6 ;
                sh:minCount 1 ;
                sh:message "An applicable ship must be assigned Polar Class PC6 or PC7." 
            ]
        ]
    ) .

gen:I2_011_ApplicablePc7Shape
    a sh:NodeShape ;
    sh:and (
        gen:I2_011_ApplicableBaseShape
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc7 ;
                sh:minCount 1 ;
                sh:message "An applicable ship must be assigned Polar Class PC6 or PC7." 
            ]
        ]
    ) .

gen:I2_011_ApplicableBaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "An applicable ship must record that a bulbous bow is present."
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:node gen:I2_011_ShapeCoefficientValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "For an applicable ship, shapeCoefficient must equal 0.6 and use the dimensionless unit."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node gen:I2_011_LoadPatchAspectRatioValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "For an applicable ship, loadPatchAspectRatio must equal 1.3 and use the dimensionless unit."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:node gen:I2_011_MethodIiiForceValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The method (iii) bow design force must be supplied in MN."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:node gen:I2_011_MethodIvForceValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The method (iv) bow design force must be supplied in MN."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:node gen:I2_011_BowDesignForceValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bow design force must be supplied in kN."
    ] ;
    sh:sparql [
        sh:message "The bow design force must equal the method (iv) force converted from MN to kN within a relative tolerance of 1e-6, and must not be less than the method (iii) force converted to kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode ;
                      nltl:bowDesignForce ?resultNode .
                ?iiiNode qudt:numericValue ?iii ;
                         qudt:unit qudt:MegaN .
                ?ivNode qudt:numericValue ?iv ;
                        qudt:unit qudt:MegaN .
                ?resultNode qudt:numericValue ?result ;
                            qudt:unit qudt:KiloN .
                BIND((?iv * 1000) AS ?expected)
                BIND(IF(ABS(?expected) < 1, 0.000001, ABS(?expected) * 0.000001) AS ?tolerance)
                FILTER (
                    ?result < (?expected - ?tolerance) ||
                    ?result > (?expected + ?tolerance) ||
                    ?result < (?iii * 1000)
                )
            }
        """
    ] .

gen:I2_011_NonApplicableBulbousShape
