<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_019> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:hullAreaValue ;
                            sh:property [
                                sh:path nltl:hullAreaFactor ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:class qudt:QuantityValue ;
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue unit:UNITLESS ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:sparql [
                        sh:message "selectedHullAreaFactor must equal the largest hullAreaFactor among the structural member's spanned hull areas." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                {
                                    SELECT $this (MAX(?factor) AS ?expectedMax)
                                    WHERE {
                                        $this nltl:hasSpannedHullArea ?area .
                                        ?area nltl:hullAreaFactor ?factorNode .
                                        ?factorNode qudt:numericValue ?factor .
                                    }
                                    GROUP BY $this
                                }
                                $this nltl:selectedHullAreaFactor ?selectedNode .
                                ?selectedNode qudt:numericValue ?selected .
                                BIND(IF(?expectedMax >= 0,
                                        ?expectedMax,
                                        -?expectedMax) AS ?scale)
                                BIND((0.000001 * ?scale + 0.000000000001) AS ?tolerance)
                                FILTER (
                                    ?selected > (?expectedMax + ?tolerance) ||
                                    ?selected < (?expectedMax - ?tolerance)
                                )
                            }
                        """
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:maxCount 0
                    ]
                ]
            )
        ]
    ] .
<END_SHACL>
