<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gs:CalculationCaseShape ;
        sh:message "A calculation case with the declared interpolation inputs is required."
    ] ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:node gs:ObliquePlatingInterpolationCaseShape ;
        sh:message "Each represented oblique plating interpolation case must contain the applicable angle, endpoints, operands, and result."
    ] .

gs:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:longitudinalFramingNetPlateThickness ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantityShape ;
        sh:message "The longitudinal framing net plate thickness is required in the calculation case."
    ] ;
    sh:property [
        sh:path nltl:transverseFramingNetPlateThickness ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantityShape ;
        sh:message "The transverse framing net plate thickness is required in the calculation case."
    ] .

gs:ObliquePlatingInterpolationCaseShape
    a sh:NodeShape ;
    sh:class nltl:obliquePlatingInterpolationCase ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:FramingAngleQuantityShape ;
        sh:message "The framing angle must be present and strictly between 20 and 70 degrees."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:node gs:InterpolationPointShape ;
        sh:message "The lower interpolation endpoint is required."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:node gs:InterpolationPointShape ;
        sh:message "The upper interpolation endpoint is required."
    ] ;
    sh:property [
        sh:path nltl:interpolatedNetPlateThickness ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantityShape ;
        sh:message "The externally calculated interpolated net plate thickness is required."
    ] .

gs:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:minCount 1 ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:message "Each interpolation point must provide its coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:node gs:DecimalQuantityShape ;
        sh:message "Each interpolation point must provide its externally calculated result."
    ] .

gs:FramingAngleQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 20 ;
        sh:maxExclusive 70
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG>
    ] .

gs:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gs:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .

gs:DecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] .
<END_SHACL>
