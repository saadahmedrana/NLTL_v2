<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2-029-ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2-029-CalculationCaseShape
    ] .

gen:I2-029-CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The calculation case must include attached shell plating in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The calculation case must exclude attached shell plating from the shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The calculation case must explicitly represent whether flange material is included in the shear area."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:class nltl:structuralMember
    ] ;
    sh:sparql [
        sh:message "When flange material is included in the shear area, the linked structural member must have a fitted flange." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:flangeMaterialIncludedInShearArea ?included .
                FILTER (datatype(?included) = xsd:boolean)
                FILTER (?included = true)

                FILTER NOT EXISTS {
                    $this nltl:sectionCalculationCaseStructuralMember ?member .
                    ?member nltl:flangeFitted ?fitted .
                    FILTER (datatype(?fitted) = xsd:boolean)
                    FILTER (?fitted = true)
                }
            }
        """
    ] .
<END_SHACL>
