<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_031_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:hasValue nltl:flatBarSection ]
            [ sh:hasValue nltl:profileSection ]
        )
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] .

gen:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:in (
            nltl:bottomStructureLocation
            nltl:sideStructureLocation
        )
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node gen:I2_031_LoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bottom structural member must have a load case whose patch-load dimension is parallel with the frame direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:structuralLocation nltl:bottomStructureLocation .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    ?loadCase nltl:loadPatchOrientation
                        nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed the plastic strength of the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:plasticStrength ?strength .
                ?strength qudt:numericValue ?strengthValue .
                FILTER (?demandValue > ?strengthValue)
            }
        """
    ] .

gen:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_031_QuantityValueShape
    ] .

gen:I2_031_QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_031_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
