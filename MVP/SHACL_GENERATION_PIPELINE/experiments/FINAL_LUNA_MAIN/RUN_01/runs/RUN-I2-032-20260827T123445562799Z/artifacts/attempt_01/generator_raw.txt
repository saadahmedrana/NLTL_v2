            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure/qudt:numericValue ?pavg ;
                      nltl:loadPatchHeight/qudt:numericValue ?b ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:shearArea/qudt:numericValue ?aw ;
                      nltl:span/qudt:numericValue ?a ;
                      nltl:yieldStrength/qudt:numericValue ?sy ;
                      nltl:hasStructuralMember ?member .

                ?member nltl:frameSpacing/qudt:numericValue ?s ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredTransverseFrameShearArea/qudt:numericValue ?at .

                BIND(IF(?a < ?b, ?a, ?b) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?s * (?af * ?ppf * ?pavg)) / (0.577 * ?sy) AS ?expectedAt)

                FILTER(
                    (?aw < ?at) ||
                    (ABS(?at - ?expectedAt) >
                     (0.000001 * ABS(?expectedAt) + 0.000000001))
                )
            }
        """
    ] .
