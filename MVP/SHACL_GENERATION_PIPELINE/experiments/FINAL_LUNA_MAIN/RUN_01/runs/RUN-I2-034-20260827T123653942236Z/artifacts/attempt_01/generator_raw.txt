<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:generatedShape_I2_034> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each represented longitudinal frame in a side structure and each represented load case, combined shear-and-bending demand must not exceed both the plastic strength and the midspan plastic collapse load, with required quantity values and units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:longitudinalFrameOrientation true ;
                        nltl:structuralLocation nltl:sideStructureLocation ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .

                FILTER NOT EXISTS {
                    ?member nltl:plasticStrength ?strength ;
                            nltl:midspanPlasticCollapseLoad ?collapseLoad .

                    ?strength qudt:numericValue ?strengthValue ;
                              qudt:unit qudt:KiloN .
                    ?collapseLoad qudt:numericValue ?collapseValue ;
                                  qudt:unit qudt:KiloN .

                    ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                    ?demand qudt:numericValue ?demandValue .

                    FILTER (
                        datatype(?strengthValue) = xsd:decimal &&
                        datatype(?collapseValue) = xsd:decimal &&
                        datatype(?demandValue) = xsd:decimal &&
                        ?demandValue <= ?strengthValue &&
                        ?demandValue <= ?collapseValue
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
