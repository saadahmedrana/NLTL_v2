@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_042_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_042_StructuralMemberShape
    ] .

gen:I2_042_StructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:flangeWidth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Flange width must be at least five times the net web thickness." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#netWebThickness> ?netThicknessNode ;
                      <https://w3id.org/nltl/vocab#flangeWidth> ?flangeWidthNode .
                ?netThicknessNode <http://qudt.org/schema/qudt/numericValue> ?netThickness ;
                                  <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/MilliM> .
                ?flangeWidthNode <http://qudt.org/schema/qudt/numericValue> ?flangeWidth ;
                                  <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/MilliM> .
                FILTER (?flangeWidth < (5 * ?netThickness))
            }
        """
    ] .
<END_SHACL>
