gen:I2_046_QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.0"^^xsd:decimal ;
        sh:message "Each linked internal ice-strengthened structure must have a corrosion/abrasion addition of at least 1.0 mm."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each internal structure corrosion/abrasion addition must use the millimetre unit."
    ] .

gen:I2_046_InternalStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:internalStructureCorrosionAbrasionAddition ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:class qudt:QuantityValue ;
        sh:node gen:I2_046_QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked internal ice-strengthened structure must specify exactly one corrosion/abrasion addition."
    ] .

gen:I2_046_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "When Polar Class requirements apply, the ship must provide one valid Polar Class value."
    ] ;
    sh:property [
        sh:path nltl:hasInternalIceStrengthenedStructure ;
        sh:node gen:I2_046_InternalStructureShape ;
        sh:message "Every linked internal ice-strengthened structure must satisfy the corrosion/abrasion addition requirement."
    ] .

gen:I2_046_NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:maxCount 0 ;
        sh:message "A ship for which the Polar Class requirement is non-applicable must not provide a Polar Class value for this requirement."
    ] .

gen:I2_046_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClassRequirementsApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            gen:I2_046_ApplicableShape
            gen:I2_046_NonApplicableShape
        ) ;
        sh:message "The ship must state exactly one Polar Class applicability boolean and satisfy the corresponding applicability branch."
    ] .
