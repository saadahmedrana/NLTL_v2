<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-ThicknessShape>
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeE
            nltl:steelGradeAh
            nltl:steelGradeDh
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-ActualGradeShape>
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The steel-grade table selection is missing, not applicable incorrectly, or the actual grade rank is below the required rank for its steel-strength category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:asBuiltPlateThickness ?q ;
        nltl:steelMaterialClass ?mc ;
        nltl:steelStrengthCategory ?sc ;
        nltl:actualHullStructuralSteelGrade ?actual .
  ?q qudt:numericValue ?t ;
     qudt:unit <http://qudt.org/vocab/unit/MilliM> .
  ?actual nltl:steelGradeOrderRank ?actualRank .

  BIND(
    IF(?t > 0 && ?t <= 10,
       IF(?mc = nltl:steelMaterialClassThree,
          IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
             IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 1, 0)),
          IF(?mc IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo) &&
             ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6, nltl:polarClassPc7), 1, 0)),
    IF(?t > 10 && ?t <= 15,
       IF(?mc = nltl:steelMaterialClassOne, 1,
          IF(?mc = nltl:steelMaterialClassTwo,
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 2,
                IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 1, 0)),
             IF(?mc = nltl:steelMaterialClassThree,
                IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
                   IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)), 0))),
    IF(?t > 15 && ?t <= 25,
       IF(?mc IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo),
          IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 2,
             IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 1, 0)),
          IF(?mc = nltl:steelMaterialClassThree,
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
                IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)), 0)),
    IF(?t > 25 && ?t <= 35,
       IF(?mc = nltl:steelMaterialClassOne,
          IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 2,
             IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 1, 0)),
          IF(?mc IN (nltl:steelMaterialClassTwo, nltl:steelMaterialClassThree),
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
                IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)), 0)),
    IF(?t > 35 && ?t <= 40,
       IF(?mc = nltl:steelMaterialClassOne, 2,
          IF(?mc = nltl:steelMaterialClassTwo,
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
                IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)),
             IF(?mc = nltl:steelMaterialClassThree,
                IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3), 0,
                   IF(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6, nltl:polarClassPc7), 3, 0)), 0))),
    IF(?t > 40 && ?t <= 45,
       IF(?mc IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo),
          IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
             IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)),
          IF(?mc = nltl:steelMaterialClassThree,
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3), 0,
                IF(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5), 4,
                   IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 3, 0))), 0)),
    IF(?t > 45 && ?t <= 50,
       IF(?mc IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo),
          IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 3,
             IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 2, 0)),
          IF(?mc = nltl:steelMaterialClassThree,
             IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), 0,
                IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 3, 0)), 0)), 0)))))))
    AS ?requiredRank)

  BIND(
    IF(?sc = nltl:normalStrengthSteelCategory,
       IF(?requiredRank = 1, nltl:steelGradeB,
          IF(?requiredRank = 2, nltl:steelGradeD,
             IF(?requiredRank = 3, nltl:steelGradeE, nltl:steelGradeValue))),
       IF(?requiredRank = 1, nltl:steelGradeAh,
          IF(?requiredRank = 2, nltl:steelGradeDh,
             IF(?requiredRank = 3, nltl:steelGradeEh,
                IF(?requiredRank = 4, nltl:steelGradeFh, nltl:steelGradeValue)))))
    AS ?requiredGrade)

  BIND(
    (?t > 0 && ?t <= 50 &&
     ?mc IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo, nltl:steelMaterialClassThree) &&
     ?sc IN (nltl:normalStrengthSteelCategory, nltl:highTensileSteelCategory) &&
     ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6, nltl:polarClassPc7))
    AS ?basicValid)

  BIND(
    (?mc = nltl:steelMaterialClassThree &&
     ?requiredRank = 0 &&
     ((?t > 35 && ?t <= 45 && ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)) ||
      (?t > 45 && ?t <= 50 && ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5))))
    AS ?notApplicable)

  FILTER(
    !?basicValid ||
    (?requiredRank = 0 && !?notApplicable) ||
    (?notApplicable && EXISTS {
      $this nltl:requiredHullStructuralSteelGrade ?unexpected
    }) ||
    (?requiredRank > 0 && !EXISTS {
      $this nltl:requiredHullStructuralSteelGrade ?required
      FILTER(?required = ?requiredGrade)
    }) ||
    (?requiredRank > 0 && EXISTS {
      $this nltl:requiredHullStructuralSteelGrade ?required
      ?required nltl:steelGradeOrderRank ?requiredActualRank
      FILTER(?actualRank < ?requiredActualRank)
    })
  )
}
"""
    ] .

<urn:nltl:generated-shape:I2-048-ThicknessShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-048-ActualGradeShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each actual hull structural steel grade must have exactly one integer order rank."
    ] .
<END_SHACL>
