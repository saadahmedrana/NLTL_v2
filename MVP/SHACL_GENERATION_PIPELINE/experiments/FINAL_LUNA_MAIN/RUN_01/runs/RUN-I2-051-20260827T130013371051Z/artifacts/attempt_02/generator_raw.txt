nltlg:I2_051_CalculationCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:positiveShearForce
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For positive shear, the longitudinal distribution factor and position must satisfy the prescribed envelope, and the design vertical ice shear force must equal Cf times FIB within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                       nltl:designVerticalIceForceAtBow ?fibQuantity ;
                       nltl:designVerticalIceShearForce ?fiQuantity ;
                       nltl:longitudinalDistributionFactor ?cfQuantity ;
                       nltl:upperIceWaterlineLengthLUI ?luiQuantity .

                $this nltl:shearForceDirection nltl:positiveShearForce ;
                       nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity .

                ?fibQuantity qudt:numericValue ?fib .
                ?fiQuantity qudt:numericValue ?fi .
                ?cfQuantity qudt:numericValue ?cf .
                ?luiQuantity qudt:numericValue ?lui .
                ?positionQuantity qudt:numericValue ?position .

                FILTER (
                    (
                        ?cf = 0 &&
                        ?position >= 0 &&
                        ?position <= (0.6 * ?lui)
                    )
                    ||
                    (
                        ?cf = 1 &&
                        ?position >= (0.9 * ?lui) &&
                        ?position <= ?lui
                    )
                )

                BIND((?cf * ?fib * 1000) AS ?expected)
                BIND(IF(?expected < 0.001, 0.000001, 0.000001 * ?expected) AS ?tolerance)

                FILTER (
                    ?fi < (?expected - ?tolerance) ||
                    ?fi > (?expected + ?tolerance)
                )
            }
        """
    ] .
