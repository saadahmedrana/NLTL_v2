    sh:property [

                    ]

    sh:sparql [
        sh:message "For positive shear, the longitudinal distribution factor, position envelope, and calculated design vertical ice shear force must conform to the prescribed rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalDistributionFactor ?cfNode ;
                      nltl:designVerticalIceForceAtBow ?fibNode ;
                      nltl:designVerticalIceShearForce ?fiNode ;
                      nltl:upperIceWaterlineLengthLUI ?luiNode ;
                      nltl:hasCalculationCase ?case .

                ?cfNode qudt:numericValue ?cf .
                ?fibNode qudt:numericValue ?fib .
                ?fiNode qudt:numericValue ?fi .

                ?positionNode qudt:numericValue ?position .

                FILTER (
                    !(
                        (?cf = 0 &&
                         ?position >= 0 &&
                         ?position <= (0.6 * ?lui))
                        ||
                        (?cf = 1 &&
                         ?position >= (0.9 * ?lui) &&
                         ?position <= ?lui)
                    )
                    ||
                    (
                        (?expected >= 0 &&
                         (
                            (?expected <= 1 &&
                             (?fi < (?expected - 0.000001) ||
                              ?fi > (?expected + 0.000001)))
                            ||
                            (?expected > 1 &&
                             (?fi < (?expected - (0.000001 * ?expected)) ||
                              ?fi > (?expected + (0.000001 * ?expected))))
                         ))
                        ||
                        (?expected < 0 &&
                         (
                            (?expected >= -1 &&
                             (?fi < (?expected - 0.000001) ||
                              ?fi > (?expected + 0.000001)))
                            ||
                            (?expected < -1 &&
                             (?fi < (?expected - (0.000001 * (-?expected))) ||
                              ?fi > (?expected + (0.000001 * (-?expected)))))
                         ))
                    )
                )
            }
        """
