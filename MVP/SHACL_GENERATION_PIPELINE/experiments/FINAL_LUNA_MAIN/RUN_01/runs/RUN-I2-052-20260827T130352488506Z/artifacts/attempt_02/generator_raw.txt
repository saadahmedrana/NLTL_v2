<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-052-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-052-CalculationCaseShape>
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must contain the externally determined interpolated value."
    ] .

<urn:nltl:generated-shape:I2-052-CalculationCaseShape>
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-052-InterpolationPointShape>
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-052-InterpolationPointShape>
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-052-InterpolationPointShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each interpolation lower and upper endpoint must be a member of the calculation case's interpolation-point set." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:interpolationLowerEndpoint ?lower .
                $this nltl:interpolationUpperEndpoint ?upper .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasInterpolationPoint ?lower .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasInterpolationPoint ?upper .
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-052-InterpolationPointShape>
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ]
    ] .
<END_SHACL>
