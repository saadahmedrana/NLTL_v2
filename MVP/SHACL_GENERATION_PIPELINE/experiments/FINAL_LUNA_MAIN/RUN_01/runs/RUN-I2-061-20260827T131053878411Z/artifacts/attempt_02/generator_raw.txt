@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-061-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-CalculationCaseShape> ;
        sh:message "The ship must have at least one calculation case for the prescribed shell-plating or local-frame analytical procedure."
    ] .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:class nltl:hullStructure ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-HullStructureShape> ;
        sh:message "Each calculation case must identify the assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "The calculation case must use exactly one prescribed analytical procedure; a direct calculation method must not replace it."
    ] .

<urn:nltl:generated-shape:I2-061-HullStructureShape>
    a sh:NodeShape ;
    sh:class nltl:hullStructure ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The assessed hull structure must provide a single thickness quantity in metres."
    ] .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:maxCount 1
    ] .
