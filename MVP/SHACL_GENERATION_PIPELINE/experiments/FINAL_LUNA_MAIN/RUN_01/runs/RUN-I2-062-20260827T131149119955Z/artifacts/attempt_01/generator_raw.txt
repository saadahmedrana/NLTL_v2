<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-062-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked load carrying stringer or web frame forming part of a grillage system must have a grillage relationship and a recorded direct calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER (
                    NOT EXISTS {
                        ?member nltl:partOfGrillageSystem ?grillage .
                        ?grillage a nltl:grillageSystem .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:directCalculationUsed ?used .
                        FILTER (?used = true)
                    }
                )
            }
        """
    ] .
<END_SHACL>
