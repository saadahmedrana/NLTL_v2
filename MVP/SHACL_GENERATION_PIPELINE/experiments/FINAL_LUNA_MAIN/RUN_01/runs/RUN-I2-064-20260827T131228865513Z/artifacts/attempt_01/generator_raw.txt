@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_064_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gs:I2_064_CalculationCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_064_StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal shear stress on a linked web must be less than yield strength divided by sqrt(3) for the linear calculation method." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .
                FILTER (?stress >= (?yield / 1.7320508075688772))
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal flange von Mises stress on a linked flange must be less than 1.15 times yield strength for the linear calculation method." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .
                FILTER (?stress >= (1.15 * ?yield))
            }
        """
    ] .
