<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-064-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:or ( _:linearCalculationConstraint _:nonlinearCalculationConstraint )
        ]
    ] ;
    sh:sparql [
        sh:message "For a linear calculation case, yield strength, member stress results, and buckling acceptance evidence must be present with MPa quantities and satisfy the declared direct checks." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:hasCalculationCase ?linearCase .
                    ?linearCase nltl:calculationMethod
                        nltl:linearCalculationMethodValue .
                }

                FILTER NOT EXISTS {
                    $this nltl:yieldStrength ?yieldQuantity .
                    ?yieldQuantity qudt:numericValue ?yieldValue ;
                                   qudt:unit unit:MegaPA .
                    FILTER ( datatype(?yieldValue) = xsd:decimal )
                }

                FILTER NOT EXISTS {
                    $this nltl:hasStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:hasStructuralMemberWeb ?web .
                    ?web a nltl:structuralMemberWeb .
                    ?web nltl:nominalShearStress ?shearQuantity .
                    ?shearQuantity qudt:numericValue ?shearValue ;
                                   qudt:unit unit:MegaPA .
                    FILTER ( datatype(?shearValue) = xsd:decimal )
                }

                FILTER NOT EXISTS {
                    $this nltl:hasStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:hasStructuralMemberFlange ?flange .
                    ?flange a nltl:structuralMemberFlange .
                    ?flange nltl:nominalFlangeVonMisesStress ?flangeQuantity .
                    ?flangeQuantity qudt:numericValue ?flangeValue ;
                                    qudt:unit unit:MegaPA .
                    FILTER ( datatype(?flangeValue) = xsd:decimal )
                }

                FILTER NOT EXISTS {
                    $this nltl:yieldStrength ?yieldQuantity .
                    ?yieldQuantity qudt:numericValue ?yieldValue ;
                                   qudt:unit unit:MegaPA .
                    FILTER ( datatype(?yieldValue) != xsd:decimal )

                    $this nltl:hasStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:hasStructuralMemberWeb ?web .
                    ?web a nltl:structuralMemberWeb .
                    ?web nltl:nominalShearStress ?shearQuantity .
                    ?shearQuantity qudt:numericValue ?shearValue ;
                                   qudt:unit unit:MegaPA .
                    FILTER ( datatype(?shearValue) = xsd:decimal )
                    FILTER ( ?shearValue >= (?yieldValue / 1.7320508075688772) )
                }

                FILTER NOT EXISTS {
                    $this nltl:yieldStrength ?yieldQuantity .
                    ?yieldQuantity qudt:numericValue ?yieldValue ;
                                   qudt:unit unit:MegaPA .
                    FILTER ( datatype(?yieldValue) != xsd:decimal )

                    $this nltl:hasStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:hasStructuralMemberFlange ?flange .
                    ?flange a nltl:structuralMemberFlange .
                    ?flange nltl:nominalFlangeVonMisesStress ?flangeQuantity .
                    ?flangeQuantity qudt:numericValue ?flangeValue ;
                                    qudt:unit unit:MegaPA .
                    FILTER ( datatype(?flangeValue) = xsd:decimal )
                    FILTER ( ?flangeValue >= (1.15 * ?yieldValue) )
                }

                FILTER NOT EXISTS {
                    $this nltl:hasStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:bucklingCriteriaSatisfied true .
                }
            }
        """
    ] .

_:linearCalculationConstraint
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:hasValue nltl:linearCalculationMethodValue ;
        sh:nodeKind sh:IRI
    ] .

_:nonlinearCalculationConstraint
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:hasValue nltl:nonlinearCalculationMethodValue ;
        sh:nodeKind sh:IRI
    ] .
<END_SHACL>
