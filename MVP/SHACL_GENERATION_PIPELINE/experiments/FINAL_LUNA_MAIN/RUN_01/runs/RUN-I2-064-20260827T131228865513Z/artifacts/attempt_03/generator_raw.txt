<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:CalculationCaseShape ;
        sh:message "The ship must provide a calculation case using the linear calculation method."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gen:StressQuantityShape ;
        sh:message "The applicable ship must provide yield strength as a decimal MegaPascal quantity."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:hasValue nltl:linearCalculationMethodValue ;
        sh:message "The calculation case must explicitly identify the linear calculation method."
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasStructuralMemberWeb ;
        sh:node gen:StructuralMemberWebShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberFlange ;
        sh:node gen:StructuralMemberFlangeShape
    ] .

gen:StructuralMemberWebShape
    a sh:NodeShape ;
    sh:class nltl:structuralMemberWeb ;
    sh:property [
        sh:path nltl:nominalShearStress ;
        sh:minCount 1 ;
        sh:node gen:StressQuantityShape ;
        sh:message "Each applicable structural member web must provide nominal shear stress in MegaPascals."
    ] ;
    sh:property [
        sh:path nltl:bucklingCriteriaSatisfied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The applicable structural member web must satisfy the buckling criteria."
    ] ;
    sh:sparql [
        sh:message "Nominal shear stress in an applicable web must be strictly less than yield strength divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberWeb $this .
                ?ship nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                ?ship nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                $this nltl:nominalShearStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stressValue .
                FILTER (datatype(?yieldValue) = xsd:decimal) .
                FILTER (datatype(?stressValue) = xsd:decimal) .
                FILTER (?stressValue >= (?yieldValue / 1.7320508075688772)) .
            }
        """
    ] .

gen:StructuralMemberFlangeShape
    a sh:NodeShape ;
    sh:class nltl:structuralMemberFlange ;
    sh:property [
        sh:path nltl:nominalFlangeVonMisesStress ;
        sh:minCount 1 ;
        sh:node gen:StressQuantityShape ;
        sh:message "Each applicable structural member flange must provide nominal von Mises stress in MegaPascals."
    ] ;
    sh:property [
        sh:path nltl:bucklingCriteriaSatisfied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The applicable structural member flange must satisfy the buckling criteria."
    ] ;
    sh:sparql [
        sh:message "Nominal von Mises stress in an applicable flange must be strictly less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberFlange $this .
                ?ship nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                ?ship nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                $this nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stressValue .
                FILTER (datatype(?yieldValue) = xsd:decimal) .
                FILTER (datatype(?stressValue) = xsd:decimal) .
                FILTER (?stressValue >= (1.15 * ?yieldValue)) .
            }
        """
    ] .

gen:StressQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must contain exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaPA ;
        sh:message "The stress quantity must use the canonical MegaPascal unit."
    ] .
<END_SHACL>
