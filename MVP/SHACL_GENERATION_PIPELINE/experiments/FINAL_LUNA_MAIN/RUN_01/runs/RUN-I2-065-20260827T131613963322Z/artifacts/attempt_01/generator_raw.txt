<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_065_CalculationCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A calculation case using the nonlinear method must provide fracture, major buckling, and yielding margins on the ship, together with valid quantity values and units for the required structural inputs." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER (
                    NOT EXISTS {
                        $this nltl:fractureMargin ?fracture .
                        ?fracture qudt:numericValue ?fractureNumber ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER ( datatype(?fractureNumber) = xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:majorBucklingMargin ?buckling .
                        ?buckling qudt:numericValue ?bucklingNumber ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER ( datatype(?bucklingNumber) = xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldingMargin ?yielding .
                        ?yielding qudt:numericValue ?yieldingNumber ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER ( datatype(?yieldingNumber) = xsd:decimal )
                    }
                )
            }
        """
    ] .

gen:I2_065_CalculationCaseShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            sh:property [
                sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:permanentLateralDeformation ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:permanentOutOfPlaneDeformation ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:relevantStructuralDimension ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:or (
                    [ sh:hasValue nltl:directCalculationMethodValue ]
                    [ sh:hasValue nltl:linearCalculationMethodValue ]
                    [ sh:hasValue nltl:prescribedAnalyticalProcedure ]
                )
            ]
        ]
    ) .
<END_SHACL>
