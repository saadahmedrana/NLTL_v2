<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO003Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:openWaterIceCondition
            nltl:iceConditionLessSevereThanCategoryAAndB
        ) ;
        sh:message "A ship must declare an allowed design ice condition: open water or ice conditions less severe than categories A and B."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:message "A ship with an applicable design ice condition must have Polar Code category C."
    ] .
<END_SHACL>
