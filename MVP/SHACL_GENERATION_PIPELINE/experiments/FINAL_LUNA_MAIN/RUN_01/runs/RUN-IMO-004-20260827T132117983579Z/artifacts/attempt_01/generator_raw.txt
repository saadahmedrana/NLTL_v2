<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships meeting the sea-ice, winter-growth, and thickness conditions must be classified as first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                       nltl:iceThickness ?thickness ;
                       nltl:winterGrowthCount ?growth .

                ?thickness qudt:numericValue ?thicknessValue ;
                           qudt:unit unit:M .

                FILTER (
                    datatype(?thicknessValue) = xsd:decimal &&
                    ?thicknessValue >= "0.3"^^xsd:decimal &&
                    ?thicknessValue <= "2.0"^^xsd:decimal
                )

                FILTER (
                    datatype(?growth) = xsd:integer &&
                    ?growth <= 1
                )

                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
<END_SHACL>
