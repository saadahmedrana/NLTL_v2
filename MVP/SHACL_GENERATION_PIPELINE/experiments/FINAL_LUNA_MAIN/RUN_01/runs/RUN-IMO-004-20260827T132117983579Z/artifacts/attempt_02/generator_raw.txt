<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:iceOriginValue
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:integer ;
        sh:maxInclusive 1
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "0.3"^^xsd:decimal ;
                sh:maxInclusive "2.0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the origin is sea ice, winter growth is at most one, and thickness is between 0.3 m and 2.0 m, the classification must contain first-year ice and no other classification." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
           nltl:winterGrowthCount ?growth ;
           nltl:iceThickness ?thickness .
    ?thickness qudt:numericValue ?thickValue ;
               qudt:unit unit:M .
    FILTER(
        ?growth <= "1"^^xsd:integer &&
        ?thickValue >= "0.3"^^xsd:decimal &&
        ?thickValue <= "2.0"^^xsd:decimal
    )
    FILTER(
        NOT EXISTS {
            $this nltl:iceTypeClassification nltl:firstYearIce
        }
        ||
        EXISTS {
            $this nltl:iceTypeClassification ?classification .
            FILTER(?classification != nltl:firstYearIce)
        }
    )
}
"""
    ] .
<END_SHACL>
