<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_005_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship must declare its ice condition."
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "A ship must declare whether ice is present."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice condition may be 'ice free waters' only when total ice present is false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition ;
                      nltl:totalIcePresent ?icePresent .
                FILTER (
                    STR(?condition) = "ice free waters" &&
                    ?icePresent != false
                )
            }
        """
    ] .
<END_SHACL>
