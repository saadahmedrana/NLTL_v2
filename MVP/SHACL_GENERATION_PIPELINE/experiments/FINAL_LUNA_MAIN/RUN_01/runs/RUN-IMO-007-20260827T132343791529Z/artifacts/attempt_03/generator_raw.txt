<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an open-water classification, sea ice concentration must be less than 1 and use the unitless unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:seaIceConcentration ?quantity .
                    ?quantity qudt:unit qudt:UNITLESS ;
                              qudt:numericValue ?value .
                    FILTER (?value < 1)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an open-water classification, no represented sea ice concentration may be 1 or greater." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                       nltl:seaIceConcentration ?quantity .
                ?quantity qudt:unit qudt:UNITLESS ;
                          qudt:numericValue ?value .
                FILTER (?value >= 1)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an open-water classification, ice of land origin must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:iceOfLandOriginPresent false .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an open-water classification, ice of land origin must not be recorded as true or with a non-boolean value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                       nltl:iceOfLandOriginPresent ?value .
                FILTER (
                    datatype(?value) != xsd:boolean ||
                    ?value != false
                )
            }
        """
    ] .
<END_SHACL>
