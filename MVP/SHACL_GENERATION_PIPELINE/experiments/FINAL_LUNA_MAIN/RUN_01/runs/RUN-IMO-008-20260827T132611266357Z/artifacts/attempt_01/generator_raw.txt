<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-008_ThinFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        sh:message "A ship classified as thin first-year ice must provide ice thickness from 0.3 m through 0.7 m using a decimal value and the metre unit." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceTypeClassification nltl:thinFirstYearIce .
    FILTER (
        NOT EXISTS {
            $this nltl:iceThickness ?anyThickness .
        }
        ||
        EXISTS {
            $this nltl:iceThickness ?thickness .
            FILTER NOT EXISTS {
                ?thickness qudt:numericValue ?value .
                ?thickness qudt:unit unit:M .
                FILTER (
                    datatype(?value) = xsd:decimal &&
                    ?value >= 0.3 &&
                    ?value <= 0.7
                )
            }
        }
    )
}
"""
    ] .
<END_SHACL>
