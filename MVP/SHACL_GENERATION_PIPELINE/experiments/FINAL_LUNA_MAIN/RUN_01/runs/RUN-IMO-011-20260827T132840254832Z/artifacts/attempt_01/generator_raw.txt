<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDailyLowTemperatureObservation ;
        sh:class nltl:dailyLowTemperatureObservation ;
        sh:minCount 1 ;
        sh:node nltlgen:IMO_011_DailyLowTemperatureObservationShape ;
        sh:message "Each ship must provide dated daily low-temperature observations with daily low-temperature values."
    ] ;
    sh:property [
        sh:path nltl:dataPeriod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:IMO_011_DataPeriodQuantityShape ;
        sh:message "The ship must provide one data period in years."
    ] ;
    sh:property [
        sh:path nltl:meanDailyLowTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:IMO_011_MeanDailyLowTemperatureQuantityShape ;
        sh:message "The ship must provide one externally computed mean daily low temperature in degrees Celsius."
    ] ;
    sh:or (
        nltlgen:IMO_011_DataPeriodAtLeastTenYearsShape
        nltlgen:IMO_011_DataPeriodBelowTenYearsShape
    ) .

nltlgen:IMO_011_DailyLowTemperatureObservationShape
    a sh:NodeShape ;
    sh:class nltl:dailyLowTemperatureObservation ;
    sh:property [
        sh:path nltl:dailyTemperatureObservationDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each daily low-temperature observation must have one calendar date."
    ] ;
    sh:property [
        sh:path nltl:dailyLowTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:IMO_011_DailyLowTemperatureQuantityShape ;
        sh:message "Each daily low-temperature observation must carry one daily low-temperature value in degrees Celsius."
    ] .

nltlgen:IMO_011_DailyLowTemperatureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IMO_011_DataPeriodQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/YR> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IMO_011_MeanDailyLowTemperatureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IMO_011_DataPeriodAtLeastTenYearsShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:dataPeriod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 10
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/YR>
            ]
        ]
    ] .

nltlgen:IMO_011_DataPeriodBelowTenYearsShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:dataPeriod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive 10
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/YR>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:administrationDatasetApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A data period shorter than ten years requires an approved administration dataset."
    ] .
<END_SHACL>
