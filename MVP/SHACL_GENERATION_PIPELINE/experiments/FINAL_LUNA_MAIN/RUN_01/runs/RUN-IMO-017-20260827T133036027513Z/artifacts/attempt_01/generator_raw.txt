<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO017ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "A ship shall have a Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gs:IMO017SolasScheduleShape ;
        sh:message "A ship shall have the applicable SOLAS certificate schedule."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gs:IMO017SupplementShape ;
        sh:message "The Polar Ship Certificate shall include a supplement."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement shall be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:sparql [
        sh:message "Each Polar Ship Certificate validity, survey, and endorsement date shall match a correspondingly categorized SOLAS certificate schedule date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSolasCertificateSchedule ?schedule .
                FILTER (
                    EXISTS {
                        $this nltl:polarCertificateValidityDates ?date .
                        FILTER NOT EXISTS {
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                    nltl:certificateScheduleDate ?date .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:surveyDates ?date .
                        FILTER NOT EXISTS {
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                    nltl:certificateScheduleDate ?date .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:endorsementDates ?date .
                        FILTER NOT EXISTS {
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                    nltl:certificateScheduleDate ?date .
                        }
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each categorized SOLAS certificate schedule date shall match the corresponding Polar Ship Certificate date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSolasCertificateSchedule ?schedule .
                ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                ?record nltl:certificateScheduleDateType ?category ;
                        nltl:certificateScheduleDate ?date .
                FILTER (
                    (?category = nltl:validityDateCategory &&
                     NOT EXISTS { $this nltl:polarCertificateValidityDates ?date })
                    ||
                    (?category = nltl:surveyDateCategory &&
                     NOT EXISTS { $this nltl:surveyDates ?date })
                    ||
                    (?category = nltl:endorsementDateCategory &&
                     NOT EXISTS { $this nltl:endorsementDates ?date })
                )
            }
        """
    ] .

gs:IMO017SolasScheduleShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gs:IMO017DateRecordShape ;
        sh:message "The SOLAS certificate schedule shall contain certificate schedule date records."
    ] .

gs:IMO017DateRecordShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record shall contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall have a permitted date category."
    ] .

gs:IMO017SupplementShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Certificate supplement shall record the equipment required by the Code."
    ] .
<END_SHACL>
