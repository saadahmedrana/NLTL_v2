<urn:nltl:generated-shape:IMO-019-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:IMO-019-LowTemperatureApplicableShape>
        <urn:nltl:generated-shape:IMO-019-LowTemperatureNotApplicableShape>
    ) ;
    sh:sparql [
        sh:message "When the ship operates in low air temperature, polarServiceTemperature must be at least 10 degrees Celsius below lowestMdlt." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true ;
                      nltl:lowestMdlt ?mdltNode ;
                      nltl:polarServiceTemperature ?pstNode .
                ?mdltNode qudt:numericValue ?mdlt ;
                          qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                ?pstNode qudt:numericValue ?pst ;
                         qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                FILTER (?pst > (?mdlt - 10))
            }
        """
    ] .
