<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCapabilityMethodologyApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ice capability methodology applicability flag must be supplied as a boolean."
    ] ;
    sh:sparql [
        sh:message "When the ice capability methodology is applicable, the Polar Water Operational Manual must contain it." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceCapabilityMethodologyApplicable ?applicable .
                FILTER (datatype(?applicable) = xsd:boolean && ?applicable = true)
                FILTER NOT EXISTS {
                    $this nltl:polarWaterOperationalManualIceCapabilityMethodologyPresent ?present .
                    FILTER (datatype(?present) = xsd:boolean && ?present = true)
                }
            }
        """
    ] .
<END_SHACL>
