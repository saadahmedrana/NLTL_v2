@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO037CaseShape
    a sh:NodeShape ;
    sh:class nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each loading-condition case must identify the loading condition."
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Each loading-condition case must provide externally computed result evidence."
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Each loading-condition case must state whether the alternative instrument applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:residualStabilityFactorSI ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minInclusive "1.0"^^xsd:decimal ;
                        sh:maxInclusive "1.0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The residual stability factor must equal one when the alternative instrument does not apply."
            ]
        ]
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:message "The permitted alternative instrument must have a compliant residual-stability outcome."
            ]
        ]
    ) ;
    sh:message "Each applicable loading condition must satisfy the residual stability factor or the permitted alternative-instrument outcome." .

gen:IMO037ApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionCase ;
        sh:class nltl:loadingConditionCase ;
        sh:node gen:IMO037CaseShape
    ] .

gen:IMO037CategoryCShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] .

gen:IMO037Pre2017ShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxInclusive "2016-12-31"^^xsd:date
    ] .

gen:IMO037ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO037ApplicableShipBranch
        gen:IMO037CategoryCShipBranch
        gen:IMO037Pre2017ShipBranch
    ) ;
    sh:message "For a category A or B ship constructed on or after 1 January 2017, every loading-condition case must provide the required residual-stability outcome." .
