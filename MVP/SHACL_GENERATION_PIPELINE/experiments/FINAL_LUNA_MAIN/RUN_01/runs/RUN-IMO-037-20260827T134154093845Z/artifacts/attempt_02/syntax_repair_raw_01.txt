<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO037ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:polarShipCategoryC
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2017-01-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:date ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:hasLoadingConditionCase ;
                sh:node gen:IMO037LoadingConditionCaseShape ;
            ]
        ]
    ) ;
    sh:message "Category A or B ships constructed on or after 1 January 2017 require conforming results for each linked loading condition case." .

gen:IMO037LoadingConditionCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:residualStabilityFactorSI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:IMO037ResidualStabilityQuantityShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When the alternative instrument is not applicable, the externally calculated residual stability factor s_i must be within the documented tolerance of the expected value 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:alternativeInstrumentApplicable ?applicable .
                FILTER(?applicable = false)
                $this nltl:residualStabilityFactorSI ?quantity .
                ?quantity qudt:numericValue ?value .
                FILTER(
                    ?value < "0.999999"^^xsd:decimal ||
                    ?value > "1.000001"^^xsd:decimal
                )
            }
        """
    ] .

gen:IMO037ResidualStabilityQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .
<END_SHACL>
