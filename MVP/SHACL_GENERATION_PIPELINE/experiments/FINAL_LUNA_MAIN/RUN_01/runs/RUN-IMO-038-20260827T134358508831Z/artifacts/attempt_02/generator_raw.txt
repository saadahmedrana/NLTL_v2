<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_038_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Damage center position relative to maximum breadth is required." ;
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Upper ice waterline length must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Upper ice waterline draught must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Longitudinal damage extent must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "760"^^xsd:decimal ;
                sh:maxInclusive "760"^^xsd:decimal ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Transverse penetration must equal 760 mm." ;
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Vertical damage extent must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Selected vertical damage position must be a single metre quantity." ;
    ] ;
    sh:sparql [
        sh:message "Longitudinal and vertical damage extents do not satisfy the required piecewise formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longNode ;
                      nltl:verticalDamageExtent ?verticalNode .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .
                ?longNode qudt:numericValue ?long .
                ?verticalNode qudt:numericValue ?vertical .
                BIND(IF(?forward = true, 0.045 * ?length, 0.015 * ?length) AS ?expectedLong)
                BIND(IF(0.20 * ?draught < ?expectedLong,
                        0.20 * ?draught,
                        ?expectedLong) AS ?expectedVertical)
                FILTER (
                    ABS(?long - ?expectedLong) >
                    IF(0.000001 * ABS(?expectedLong) > 0.000000001,
                       0.000001 * ABS(?expectedLong),
                       0.000000001)
                    ||
                    ABS(?vertical - ?expectedVertical) >
                    IF(0.000001 * ABS(?expectedVertical) > 0.000000001,
                       0.000001 * ABS(?expectedVertical),
                       0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected vertical damage position must be between the keel and 1.20 times the upper ice waterline draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedVerticalDamagePosition ?positionNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode .
                ?positionNode qudt:numericValue ?position .
                ?draughtNode qudt:numericValue ?draught .
                FILTER (?position < 0 || ?position > 1.20 * ?draught)
            }
        """
    ] .
<END_SHACL>
