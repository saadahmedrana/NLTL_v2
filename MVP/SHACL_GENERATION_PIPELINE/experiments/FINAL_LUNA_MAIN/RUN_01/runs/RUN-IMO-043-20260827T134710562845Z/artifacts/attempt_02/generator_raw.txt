<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_043_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:node gen:IMO_043_MachineryInstallationShape
    ] .

gen:IMO_043_MachineryInstallationShape
    a sh:NodeShape ;
    sh:targetClass nltl:machineryInstallationItem ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:iceAccretionHazard ;
        sh:message "The machinery installation shall cover the ice accretion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowAccumulationHazard ;
        sh:message "The machinery installation shall cover the snow accumulation hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIceIngestionHazard ;
        sh:message "The machinery installation shall cover the seawater ice ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:liquidFreezingOrViscosityIncreaseHazard ;
        sh:message "The machinery installation shall cover the liquid freezing or viscosity increase hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIntakeTemperatureHazard ;
        sh:message "The machinery installation shall cover the seawater intake temperature hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowIngestionHazard ;
        sh:message "The machinery installation shall cover the snow ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:minCount 1 ;
        sh:message "The machinery installation shall have hazard protection evidence."
    ] ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:node gen:IMO_043_AssociatedEquipmentShape
    ] .

gen:IMO_043_AssociatedEquipmentShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each associated equipment item shall be linked to a machinery installation that covers all six specified hazards and has hazard protection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?installation nltl:hasAssociatedEquipment $this .
                FILTER NOT EXISTS {
                    ?installation nltl:hasApplicableHazard nltl:iceAccretionHazard .
                    ?installation nltl:hasApplicableHazard nltl:snowAccumulationHazard .
                    ?installation nltl:hasApplicableHazard nltl:seawaterIceIngestionHazard .
                    ?installation nltl:hasApplicableHazard nltl:liquidFreezingOrViscosityIncreaseHazard .
                    ?installation nltl:hasApplicableHazard nltl:seawaterIntakeTemperatureHazard .
                    ?installation nltl:hasApplicableHazard nltl:snowIngestionHazard .
                    ?installation nltl:hasHazardProtectionEvidence ?evidence .
                }
            }
        """
    ] .
<END_SHACL>
