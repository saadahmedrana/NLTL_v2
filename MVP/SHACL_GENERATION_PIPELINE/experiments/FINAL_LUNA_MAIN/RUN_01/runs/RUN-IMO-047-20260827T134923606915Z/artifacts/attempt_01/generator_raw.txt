<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each ship linked to an essential engine, combustion-air temperature control must be present and the combustion-air temperature must be within the manufacturer's inclusive minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?engine .
                FILTER NOT EXISTS {
                    $this nltl:combustionAirTemperatureControlPresent true .
                    $this nltl:combustionAirTemperature ?temperatureQuantity .
                    ?temperatureQuantity qudt:numericValue ?temperature ;
                                         qudt:unit qudt:DEG_C .
                    $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                    ?minimumQuantity qudt:numericValue ?minimum ;
                                     qudt:unit qudt:DEG_C .
                    $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .
                    ?maximumQuantity qudt:numericValue ?maximum ;
                                     qudt:unit qudt:DEG_C .
                    FILTER(datatype(?temperature) = xsd:decimal)
                    FILTER(datatype(?minimum) = xsd:decimal)
                    FILTER(datatype(?maximum) = xsd:decimal)
                    FILTER(?temperature >= ?minimum)
                    FILTER(?temperature <= ?maximum)
                }
            }
        """
    ] .
<END_SHACL>
