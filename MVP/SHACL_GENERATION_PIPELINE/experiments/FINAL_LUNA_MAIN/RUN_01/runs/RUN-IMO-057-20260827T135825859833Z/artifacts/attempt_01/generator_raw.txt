<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO057Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked fire pump, emergency fire pump, water-mist pump, and water-spray pump must have a containing compartment maintained above 0 deg C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component rdf:type ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER NOT EXISTS {
                    ?component nltl:hasContainingCompartment ?compartment .
                    ?compartment rdf:type nltl:compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature qudt:numericValue ?value ;
                                 qudt:unit unit:DEG_C .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value > 0
                    )
                }
            }
        """
    ] .
<END_SHACL>
