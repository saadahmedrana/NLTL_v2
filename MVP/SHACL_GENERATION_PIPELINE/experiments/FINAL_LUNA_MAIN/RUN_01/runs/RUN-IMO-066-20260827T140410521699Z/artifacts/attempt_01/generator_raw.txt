<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-066-ShipIceAccretion>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship's ice-accretion exposure assessment is required and must be boolean."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship exposed to ice accretion, every protected item must have one permitted category and associated ice or snow removal or prevention means, and all required protected-item categories must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipExposedToIceAccretion ?exposed .
                FILTER (?exposed = true)

                {
                    $this nltl:hasProtectedItem ?item .
                    FILTER NOT EXISTS {
                        ?item nltl:protectedItemCategory ?category .
                    }
                }
                UNION
                {
                    $this nltl:hasProtectedItem ?item .
                    ?item nltl:protectedItemCategory ?category .
                    FILTER (
                        ?category NOT IN (
                            nltl:escapeRouteProtectedItem,
                            nltl:musterStationProtectedItem,
                            nltl:embarkationAreaProtectedItem,
                            nltl:survivalCraftProtectedItem,
                            nltl:launchingApplianceProtectedItem,
                            nltl:survivalCraftAccessProtectedItem
                        )
                    )
                }
                UNION
                {
                    $this nltl:hasProtectedItem ?item .
                    FILTER NOT EXISTS {
                        ?item nltl:hasIceSnowRemovalOrPreventionMeans ?means .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:escapeRouteProtectedItem .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:musterStationProtectedItem .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:embarkationAreaProtectedItem .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:survivalCraftProtectedItem .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:launchingApplianceProtectedItem .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory nltl:survivalCraftAccessProtectedItem .
                    }
                }
            }
        """
    ] .
<END_SHACL>
