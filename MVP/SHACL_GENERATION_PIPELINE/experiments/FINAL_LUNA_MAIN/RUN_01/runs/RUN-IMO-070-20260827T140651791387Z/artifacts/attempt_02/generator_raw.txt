<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_070_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable passenger ship, each person on board must have an immersion suit or thermal protective aid assignment, and size compatibility must be true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER (
        EXISTS { $this a nltl:passengerShip }
        ||
        EXISTS {
            $this nltl:shipType "passenger_ship"^^xsd:string
        }
    )
    FILTER (
        NOT EXISTS {
            $this nltl:sizeCompatibilityStatus true
        }
        ||
        EXISTS {
            $this nltl:hasPersonOnBoardMember ?person
            FILTER NOT EXISTS {
                ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment .
                ?equipment a ?equipmentKind .
                FILTER (
                    ?equipmentKind IN (
                        nltl:immersionSuitEquipment,
                        nltl:thermalProtectiveAidEquipment
                    )
                )
            }
        }
    )
}
""" ;
    ] .
<END_SHACL>
