<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-077> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "When additional equipment is carried in survival craft, the required capacity inputs and results must be present and satisfy both capacity comparisons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:additionalEquipmentCarriedInSurvivalCraft ?applicability .
    FILTER (
        datatype(?applicability) = xsd:boolean
        && ?applicability = true
    )
    FILTER NOT EXISTS {
        $this nltl:personsCapacityRequirement ?persons ;
              nltl:additionalEquipmentCapacityRequirement ?additional ;
              nltl:survivalCraftAvailableCapacity ?available ;
              nltl:launchingApplianceCapacity ?launchingNode ;
              nltl:loadedSurvivalCraftRequirement ?loadedNode .

        ?launchingNode a qudt:QuantityValue ;
                       qudt:numericValue ?launching ;
                       qudt:unit unit:KiloGM .

        ?loadedNode a qudt:QuantityValue ;
                    qudt:numericValue ?loaded ;
                    qudt:unit unit:KiloGM .

        FILTER (
            datatype(?persons) = xsd:integer
            && datatype(?additional) = xsd:integer
            && datatype(?available) = xsd:integer
            && datatype(?launching) = xsd:decimal
            && datatype(?loaded) = xsd:decimal
        )
        FILTER (
            ?available >= (?persons + ?additional)
            && ?launching >= ?loaded
        )
    }
}
""" ;
    ] .
<END_SHACL>
