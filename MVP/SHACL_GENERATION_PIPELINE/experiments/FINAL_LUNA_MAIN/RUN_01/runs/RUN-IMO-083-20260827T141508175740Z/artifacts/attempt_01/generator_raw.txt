<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_083_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_083_ApplicableBranch
        gen:IMO_083_NotApplicableBranch
    ) .

gen:IMO_083_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Ice-accretion applicability must be recorded as true for the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:hasRequiredNavigationOrCommunicationAntenna ;
        sh:minCount 1 ;
        sh:node gen:IMO_083_AntennaShape ;
        sh:message "When ice accretion is likely, each required navigation or communication antenna must be identified and have ice-accumulation prevention."
    ] .

gen:IMO_083_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "Ice-accretion applicability must be recorded as true or false."
    ] .

gen:IMO_083_AntennaShape
    a sh:NodeShape ;
    sh:class nltl:navigationOrCommunicationAntenna ;
    sh:property [
        sh:path nltl:antennaIceAccumulationPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each required navigation or communication antenna must have ice-accumulation prevention means."
    ] .
<END_SHACL>
