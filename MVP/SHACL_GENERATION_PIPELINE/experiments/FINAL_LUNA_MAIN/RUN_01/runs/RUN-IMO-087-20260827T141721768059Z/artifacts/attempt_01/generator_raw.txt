<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-087-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DEG
            ]
        ] ;
        sh:message "A planned or actual latitude in degrees is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:gnssCompassOrEquivalentCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:sparql [
        sh:message "For ships proceeding to latitudes over 80 degrees, at least one GNSS compass or equivalent and connections to both main and emergency power are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plannedOrActualLatitude ?latitudeNode .
                ?latitudeNode qudt:numericValue ?latitude .
                FILTER (?latitude > "80"^^xsd:decimal)

                FILTER NOT EXISTS {
                    $this nltl:gnssCompassOrEquivalentCount ?count .
                    FILTER (?count >= 1)
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToMainPower ?mainPower .
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToEmergencyPower ?emergencyPower .
                }
            }
        """
    ] .
<END_SHACL>
