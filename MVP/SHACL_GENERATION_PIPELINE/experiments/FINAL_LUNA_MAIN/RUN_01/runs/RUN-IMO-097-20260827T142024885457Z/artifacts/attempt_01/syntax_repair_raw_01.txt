<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-097> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating status is required." ;
    ] ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's released rescue boat or lifeboat status is required." ;
    ] ;
    sh:sparql [
        sh:message "When the ship operates in low air temperature and a rescue boat or lifeboat is released, each linked survival craft must carry the required communication devices, and each required rescue boat or lifeboat must carry a distress-alert device." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true ;
                      nltl:releasedRescueBoatOrLifeboat true .
                {
                    $this nltl:hasSurvivalCraft ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                    }
                }
                UNION
                {
                    $this nltl:hasSurvivalCraft ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                    }
                }
                UNION
                {
                    $this nltl:hasSurvivalCraft ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?distressDevice .
                    }
                }
                UNION
                {
                    $this nltl:hasRequiredRescueBoatOrLifeboat ?lifeboat .
                    FILTER NOT EXISTS {
                        ?lifeboat nltl:hasDistressAlertDevice ?distressAlertDevice .
                    }
                }
            }
        """ ;
    ] .
<END_SHACL>
