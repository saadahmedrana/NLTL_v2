<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:LowAirTemperatureApplicableShape
        gen:LowAirTemperatureNotApplicableShape
    ) .

gen:LowAirTemperatureApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The ship must indicate that it operates in low air temperature for this requirement to apply."
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft ;
        sh:node gen:OtherSurvivalCraftShape
    ] .

gen:LowAirTemperatureNotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( false )
    ] .

gen:OtherSurvivalCraftShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one location-signal device."
    ] ;
    sh:property [
        sh:path nltl:hasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one on-scene transmit/receive communication device."
    ] .
<END_SHACL>
