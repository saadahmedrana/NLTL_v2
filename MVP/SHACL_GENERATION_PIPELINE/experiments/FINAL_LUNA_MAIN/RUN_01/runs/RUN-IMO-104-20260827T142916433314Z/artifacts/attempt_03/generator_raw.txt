<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO104ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:class nltl:crewMember ;
        sh:minCount 1 ;
        sh:message "The ship shall identify its crew members."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:minCount 1 ;
        sh:node gen:IMO104CrewMemberShape ;
        sh:message "The ship shall provide the crew-member inventory and applicable familiarization records."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each procedure or equipment item relevant to a crew member's assigned duties shall be represented by that crew member's familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this (nltl:hasCrewMemberInventory|nltl:hasCrewMember) ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gen:IMO104CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty ;
        sh:minCount 1 ;
        sh:node gen:IMO104AssignedDutyShape ;
        sh:message "Each crew member shall have an assigned duty linked to relevant PWOM items."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO104FamiliarizationRecordShape ;
        sh:message "Each crew member shall have a PWOM familiarization record."
    ] .

gen:IMO104AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:message "Each assigned duty shall identify its relevant PWOM procedure or equipment items."
    ] .

gen:IMO104FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:message "Each familiarization record shall identify the PWOM items covered."
    ] .
<END_SHACL>
