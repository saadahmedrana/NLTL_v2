<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO106ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:class nltl:dischargedMaterialTypeValue ;
        sh:message "A ship must provide its discharged material classification."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "The applicability of Polar Code paragraph 1.1.1 must be recorded."
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast, Polar Code paragraph 1.1.1 must be recorded as not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material ;
                      nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (?applicable != false)
            }
        """
    ] .
<END_SHACL>
