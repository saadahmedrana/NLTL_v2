<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, each oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have at least 0.76 m separation from the outer shell." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:tankType ?tankType .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )
                FILTER (
                    ?tankType IN (
                        "oil_residue_tank"^^xsd:string,
                        "oily_bilge_water_holding_tank"^^xsd:string
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:individualTankCapacity ?smallCapacityNode .
                    ?smallCapacityNode qudt:numericValue ?smallCapacity ;
                                         qudt:unit unit:M3 .
                    FILTER (datatype(?smallCapacity) = xsd:decimal)
                    FILTER (?smallCapacity <= "30"^^xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:individualTankCapacity ?largeCapacityNode .
                    ?largeCapacityNode qudt:numericValue ?largeCapacity ;
                                         qudt:unit unit:M3 .
                    FILTER (datatype(?largeCapacity) = xsd:decimal)
                    FILTER (?largeCapacity > "30"^^xsd:decimal)

                    $this nltl:tankOuterShellSeparation ?separationNode .
                    ?separationNode qudt:numericValue ?separation ;
                                     qudt:unit unit:M .
                    FILTER (datatype(?separation) = xsd:decimal)
                    FILTER (?separation >= "0.76"^^xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
