<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO111Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryC ) ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:maxInclusive "2016-12-31"^^xsd:date ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For every applicable oil residue or oily bilge water holding tank, a capacity not greater than 30 m3 is exempt; every capacity greater than 30 m3 requires outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:hasComponent ?tank .
                ?tank rdf:type ?tankType .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER (?tankType IN (
                    nltl:oilResidueTank,
                    nltl:oilyBilgeWaterHoldingTank
                ))
                FILTER (
                    NOT EXISTS {
                        $this nltl:individualTankCapacity ?capacityQuantity .
                        ?capacityQuantity a qudt:QuantityValue ;
                                           qudt:numericValue ?capacity ;
                                           qudt:unit <http://qudt.org/vocab/unit/M3> .
                        FILTER (datatype(?capacity) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:individualTankCapacity ?largeCapacityQuantity .
                        ?largeCapacityQuantity a qudt:QuantityValue ;
                                                qudt:numericValue ?largeCapacity ;
                                                qudt:unit <http://qudt.org/vocab/unit/M3> .
                        FILTER (
                            datatype(?largeCapacity) = xsd:decimal &&
                            ?largeCapacity > "30"^^xsd:decimal
                        )
                        FILTER NOT EXISTS {
                            $this nltl:tankOuterShellSeparation ?separationQuantity .
                            ?separationQuantity a qudt:QuantityValue ;
                                                 qudt:numericValue ?separation ;
                                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (
                                datatype(?separation) = xsd:decimal &&
                                ?separation >= "0.76"^^xsd:decimal
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
