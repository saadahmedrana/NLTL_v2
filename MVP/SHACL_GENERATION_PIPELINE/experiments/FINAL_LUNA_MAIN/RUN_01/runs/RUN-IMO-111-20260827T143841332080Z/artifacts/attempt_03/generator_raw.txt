<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_111_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one Category A or Category B classification."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable ship must have a construction date on or after 1 January 2017."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every applicable oil-residue or oily-bilge-water holding tank with capacity greater than 30 m3 must have a corresponding outer-shell separation of at least 0.76 m; tank type, capacity, and required separation data must be valid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasComponent ?tank .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                ?tank nltl:tankType ?tankType .
                FILTER (
                    STR(?tankType) IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )

                FILTER (
                    NOT EXISTS {
                        ?tank nltl:individualTankCapacity ?capacityNode .
                        ?capacityNode qudt:numericValue ?capacityValue ;
                                      qudt:unit unit:M3 .
                        FILTER (datatype(?capacityValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?tank nltl:individualTankCapacity ?capacityNode2 .
                        ?capacityNode2 qudt:numericValue ?capacityValue2 ;
                                       qudt:unit unit:M3 .
                        FILTER (datatype(?capacityValue2) = xsd:decimal)
                        FILTER (?capacityValue2 > 30)
                        FILTER NOT EXISTS {
                            ?tank nltl:tankOuterShellSeparation ?separationNode .
                            ?separationNode qudt:numericValue ?separationValue ;
                                             qudt:unit unit:M .
                            FILTER (datatype(?separationValue) = xsd:decimal)
                            FILTER (?separationValue >= 0.76)
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
