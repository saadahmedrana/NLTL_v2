<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_112_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An operating area is required to determine whether the Arctic waters prohibition applies."
    ] ;
    sh:property [
        sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of noxious liquid substances or mixtures containing them into the sea shall be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (
                    datatype(?area) = xsd:string &&
                    str(?area) = "Arctic_waters"
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?v0 .
                        FILTER (
                            datatype(?v0) = xsd:boolean &&
                            ?v0 = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?v1 .
                        FILTER (
                            !(datatype(?v1) = xsd:boolean &&
                            ?v1 = false)
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
