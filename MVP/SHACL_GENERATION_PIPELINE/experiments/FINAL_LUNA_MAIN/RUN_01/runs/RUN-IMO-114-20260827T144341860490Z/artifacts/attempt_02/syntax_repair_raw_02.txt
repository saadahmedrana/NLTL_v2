<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_114_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, the Administration approval status must be approved and the applicable certificate polar operation entry must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 ;
                      nltl:cargoTankType ?tankType .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                    &&
                    ?requiresType3 = true
                    &&
                    ?tankType = "3"
                    &&
                    (
                        NOT EXISTS {
                            $this nltl:administrationCarriageApprovalStatus
                                  nltl:evidenceStateApproved
                        }
                        ||
                        NOT EXISTS {
                            $this nltl:applicableCertificatePolarOperationEntryPresent
                                  true
                        }
                    )
                )
            }
        """
    ] .
</END_SHACL>
