<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-114> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, the Administration approval must be approved and the applicable polar operation certificate entry must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )

                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )

                $this nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 .
                FILTER (
                    datatype(?requiresType3) = xsd:boolean &&
                    ?requiresType3 = true
                )

                $this nltl:cargoTankType ?tankType .
                FILTER (
                    datatype(?tankType) = xsd:string &&
                    ?tankType = "3"
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:administrationCarriageApprovalStatus
                            nltl:evidenceStateApproved
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:applicableCertificatePolarOperationEntryPresent
                            ?entryPresent .
                        FILTER (
                            datatype(?entryPresent) = xsd:boolean &&
                            ?entryPresent = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
