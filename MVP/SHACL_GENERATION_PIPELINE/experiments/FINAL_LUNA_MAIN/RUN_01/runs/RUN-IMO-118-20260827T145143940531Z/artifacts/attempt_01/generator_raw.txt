<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-118> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction date on or after 1 January 2017 is required to establish this rule's applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:class nltl:passengerShip
        ]
    ) ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a qualifying category A or B ship, or any passenger ship, constructed on or after 1 January 2017, sewage discharge to sea must be false unless compliance with paragraph 4.2.1.3 is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:constructionDate ?date .
                FILTER (
                    YEAR(?date) > 2017 ||
                    (
                        YEAR(?date) = 2017 &&
                        (
                            MONTH(?date) > 1 ||
                            (MONTH(?date) = 1 && DAY(?date) >= 1)
                        )
                    )
                )

                {
                    ?this nltl:shipCategory ?category .
                    FILTER (
                        ?category IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        )
                    )
                }
                UNION
                {
                    ?this a nltl:passengerShip .
                }

                FILTER NOT EXISTS {
                    ?this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ?status .
                    FILTER ( ?status = true )
                }

                FILTER NOT EXISTS {
                    ?this nltl:sewageDischargeToSea ?discharge .
                    FILTER ( ?discharge = false )
                }
            }
        """
    ] .
<END_SHACL>
