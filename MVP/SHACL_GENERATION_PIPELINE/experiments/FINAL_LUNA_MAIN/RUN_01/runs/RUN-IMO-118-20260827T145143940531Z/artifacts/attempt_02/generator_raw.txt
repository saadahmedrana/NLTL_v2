<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_118_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Applicability evidence is missing or incomplete: a construction date and a category or passenger-ship classification are required to determine this rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    (
                        NOT EXISTS {
                            $this nltl:constructionDate ?date .
                        }
                        &&
                        NOT EXISTS {
                            $this rdf:type nltl:passengerShip .
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryC .
                        }
                    )
                    ||
                    (
                        EXISTS {
                            $this nltl:constructionDate ?date .
                            FILTER (
                                ?date >= "2017-01-01"^^xsd:date
                            )
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryA .
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryB .
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryC .
                        }
                        &&
                        NOT EXISTS {
                            $this rdf:type nltl:passengerShip .
                        }
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, sewage discharge to sea must be false unless compliance with Polar Code paragraph 4.2.1.3 is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    EXISTS {
                        $this nltl:constructionDate ?date .
                        FILTER (
                            ?date >= "2017-01-01"^^xsd:date
                        )
                    }
                    &&
                    (
                        EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryA .
                        }
                        ||
                        EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryB .
                        }
                        ||
                        EXISTS {
                            $this rdf:type nltl:passengerShip .
                        }
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:sewageDischargeToSea ?discharge .
                    }
                    ||
                    (
                        EXISTS {
                            $this nltl:sewageDischargeToSea ?discharge .
                            FILTER (?discharge = true)
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ?status .
                            FILTER (?status = true)
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
