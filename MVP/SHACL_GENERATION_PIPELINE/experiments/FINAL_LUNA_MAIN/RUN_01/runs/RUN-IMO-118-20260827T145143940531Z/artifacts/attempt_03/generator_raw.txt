<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_118_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:message "Any supplied construction date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:message "Any supplied ship category must be Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:message "Any supplied sewage-discharge value must be an xsd:boolean."
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:message "Any supplied paragraph 4.2.1.3 compliance status must be an xsd:boolean."
    ] ;
    sh:sparql [
        sh:message "Applicability cannot be determined: provide a valid construction date, and when the date is on or after 1 January 2017 provide ship-category or passenger-ship evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    (
                        NOT EXISTS {
                            $this nltl:constructionDate ?dateAfter .
                            FILTER (
                                datatype(?dateAfter) = xsd:date &&
                                ?dateAfter >= "2017-01-01"^^xsd:date
                            )
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:constructionDate ?dateBefore .
                            FILTER (
                                datatype(?dateBefore) = xsd:date &&
                                ?dateBefore < "2017-01-01"^^xsd:date
                            )
                        }
                    )
                    ||
                    (
                        EXISTS {
                            $this nltl:constructionDate ?applicableDate .
                            FILTER (
                                datatype(?applicableDate) = xsd:date &&
                                ?applicableDate >= "2017-01-01"^^xsd:date
                            )
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:shipCategory ?categoryEvidence
                        }
                        &&
                        NOT EXISTS {
                            $this a nltl:passengerShip
                        }
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Sewage discharge to sea is prohibited for an applicable ship unless paragraph 4.2.1.3 compliance is true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (
                    datatype(?date) = xsd:date &&
                    ?date >= "2017-01-01"^^xsd:date
                )

                FILTER (
                    EXISTS {
                        $this nltl:shipCategory ?category .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this a nltl:passengerShip
                    }
                )

                $this nltl:sewageDischargeToSea ?discharge .
                FILTER (?discharge = true)

                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ?status .
                    FILTER (?status = true)
                }
            }
        """
    ] .
<END_SHACL>
