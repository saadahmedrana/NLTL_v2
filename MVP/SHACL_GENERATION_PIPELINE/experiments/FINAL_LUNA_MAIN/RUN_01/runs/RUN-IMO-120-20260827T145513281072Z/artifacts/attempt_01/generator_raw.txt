<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120-DirectStatic> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall identify its operating area."
    ] ;
    sh:property [
        sh:path nltl:arcticWaters ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the Arctic-waters applicability information."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall record whether food waste is discharged to sea."
    ] ;
    sh:sparql [
        sh:message "For a ship operating in Arctic_waters and discharging food waste to sea, the nearest land, ice-shelf, or fast-ice distance shall be at least 12 nautical miles, the distance from an area with ice concentration above one tenth shall be recorded, and the applicable ice data shall be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .

                FILTER (
                    STR(?area) = "Arctic_waters" &&
                    ?discharge = true
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                        ?nearest qudt:numericValue ?nearestValue ;
                                 qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal &&
                            ?nearestValue >= "12"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recordedDistance .
                        ?recordedDistance qudt:numericValue ?recordedValue ;
                                           qudt:unit unit:MI_N .
                        FILTER (datatype(?recordedValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceConcentration ?ice .
                        ?ice qudt:numericValue ?iceValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (datatype(?iceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearbyIce .
                        ?nearbyIce qudt:numericValue ?nearbyValue ;
                                   qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?nearbyValue) = xsd:decimal &&
                            ?nearbyValue > "0.1"^^xsd:decimal
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
