<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must specify its operating area so the conditional discharge rule can be evaluated."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship must specify whether food waste is discharged to sea so the conditional rule can be evaluated."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship operates in Arctic_waters and discharges food waste to sea, it must provide nearby ice concentration above 1/10, a distance to the nearest land ice shelf or fast ice of at least 12 nautical miles, and the recorded distance from an area with ice concentration above 1/10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .

                FILTER (
                    STR(?area) = "Arctic_waters" &&
                    ?discharge = true
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearby .
                        ?nearby a qudt:QuantityValue ;
                                qudt:numericValue ?nearValue ;
                                qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?nearValue) = xsd:decimal &&
                            ?nearValue > "0.1"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                        ?nearest a qudt:QuantityValue ;
                                qudt:numericValue ?nearestValue ;
                                qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal &&
                            ?nearestValue >= "12"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recorded .
                        ?recorded a qudt:QuantityValue ;
                                 qudt:numericValue ?recordedValue ;
                                 qudt:unit unit:MI_N .
                        FILTER (datatype(?recordedValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
