        sh:minCount 1 ;

        sh:message "The ship's arctic waters applicability must be recorded as Arctic_waters."
    ] ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:hasValue "Arctic_waters"^^xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's operating area must be recorded as Arctic_waters."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Food waste discharge to sea must be recorded as true for this applicability case."
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;

        sh:node [

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

                sh:maxCount 1

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When operating in Arctic_waters and discharging food waste to sea, the ship must be at least 12 nautical miles from the nearest land, ice shelf, or fast ice, and must record the distance from an area whose nearby ice concentration exceeds one tenth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:arcticWaters ?arctic ;
                      nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea true ;
                      nltl:nearbyIceConcentration ?nearby .

                ?nearby qudt:numericValue ?nearbyValue ;
                        qudt:unit unit:UNITLESS .

                FILTER (
                    ?arctic = "Arctic_waters"^^xsd:string &&
                    ?area = "Arctic_waters"^^xsd:string &&
                    datatype(?nearbyValue) = xsd:decimal &&
                    ?nearbyValue > "0.1"^^xsd:decimal
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                        ?nearest qudt:numericValue ?nearestValue ;
                                 qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal &&
                            ?nearestValue >= "12"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recorded .
                        ?recorded qudt:numericValue ?recordedValue ;
                                 qudt:unit unit:MI_N .
                        FILTER (datatype(?recordedValue) = xsd:decimal)
                    }
                )
            }
        """
