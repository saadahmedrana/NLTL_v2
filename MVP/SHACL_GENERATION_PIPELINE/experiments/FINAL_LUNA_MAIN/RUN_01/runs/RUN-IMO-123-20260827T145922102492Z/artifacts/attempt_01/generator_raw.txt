<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-123-ship> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:residueNotRecoverableByCommonMethods ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:shipEnRoute ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:harmfulSubstancePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:departureAndDestinationWithinArctic ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:routeRemainsWithinArctic ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:adequateReceptionFacilitiesAvailable ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When cargo residue is not recoverable by common methods, all discharge conditions and a paired discharge-distance record must be present; the nearest land, ice shelf, or fast-ice distance must be at least 12 nautical miles." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:residueNotRecoverableByCommonMethods true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipEnRoute true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:harmfulSubstancePresent false
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:departureAndDestinationWithinArctic true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:routeRemainsWithinArctic true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:adequateReceptionFacilitiesAvailable false
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?iceConcentration .
                        ?iceConcentration qudt:numericValue ?iceValue ;
                                           qudt:unit unit:UNITLESS .
                        FILTER(datatype(?iceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDischargeDistanceRecord ?record
                    }
                    ||
                    EXISTS {
                        $this nltl:hasDischargeDistanceRecord ?record .
                        FILTER NOT EXISTS {
                            ?record nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ?areaDistance .
                            ?areaDistance qudt:numericValue ?areaValue ;
                                          qudt:unit unit:MI_N .
                            FILTER(datatype(?areaValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasDischargeDistanceRecord ?record .
                        FILTER NOT EXISTS {
                            ?record nltl:distanceToNearestLandIceShelfOrFastIce ?nearestDistance .
                            ?nearestDistance qudt:numericValue ?nearestValue ;
                                              qudt:unit unit:MI_N .
                            FILTER(
                                datatype(?nearestValue) = xsd:decimal
                                && ?nearestValue >= 12
                            )
                        }
                    }
                )
            }
        """
    ] .

</END_SHACL>
