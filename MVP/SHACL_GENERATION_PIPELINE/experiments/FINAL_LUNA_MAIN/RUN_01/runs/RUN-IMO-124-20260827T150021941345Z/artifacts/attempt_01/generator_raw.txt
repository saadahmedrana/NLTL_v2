<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-124> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must record its operating area for applicability determination."
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must record whether a MARPOL Annex V regulation 6.1 discharge occurs."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an Antarctic-area ship making a MARPOL Annex V regulation 6.1 discharge, the nearest-fast-ice distance must be at least 12 nautical miles, the applicable ice-area distance must be recorded, nearby ice concentration must exceed 0.1, and food waste must not be discharged onto ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:operatingArea ?area ;
          nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .
    FILTER (STR(?area) = "Antarctic_area")
    FILTER NOT EXISTS {
        $this nltl:distanceToNearestFastIce ?fastIceDistance ;
              nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?iceAreaDistance ;
              nltl:nearbyIceConcentration ?nearbyConcentration ;
              nltl:foodWasteDischargeOntoIce false .

        ?fastIceDistance qudt:numericValue ?fastValue ;
                         qudt:unit unit:MI_N .
        ?iceAreaDistance qudt:numericValue ?iceAreaValue ;
                         qudt:unit unit:MI_N .
        ?nearbyConcentration qudt:numericValue ?nearbyValue ;
                              qudt:unit unit:UNITLESS .

        FILTER (datatype(?fastValue) = xsd:decimal)
        FILTER (datatype(?iceAreaValue) = xsd:decimal)
        FILTER (datatype(?nearbyValue) = xsd:decimal)
        FILTER (?fastValue >= 12)
        FILTER (?nearbyValue > 0.1)
        FILTER NOT EXISTS {
            $this nltl:foodWasteDischargeOntoIce true .
        }
    }
}
"""
    ] .
</END_SHACL>
