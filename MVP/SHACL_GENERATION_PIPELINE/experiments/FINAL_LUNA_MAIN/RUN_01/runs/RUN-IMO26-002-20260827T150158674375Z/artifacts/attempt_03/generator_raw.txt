<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthOverallQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:message "Length overall must provide a numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:message "Length overall must use metres."
    ] .

nltlgen:GrossTonnageQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:message "Gross tonnage must provide a numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:message "Gross tonnage must use the unitless QUDT unit."
    ] .

nltlgen:Chapter9Dash1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ship type is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthOverallQuantityShape ;
        sh:message "Length overall is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:GrossTonnageQuantityShape ;
        sh:message "Gross tonnage is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Trade status is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Chapter 9-1 applicability must be recorded."
    ] ;
    sh:sparql [
        sh:message "Chapter 9-1 applicability must be true exactly for qualifying fishing vessels, pleasure yachts, or cargo ships, and false otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:shipType ?type ;
                      nltl:lengthOverall ?lengthQuantity ;
                      nltl:grossTonnage ?tonnageQuantity ;
                      nltl:engagedInTrade ?trade ;
                      nltl:polarCodeChapter9Dash1Applicable ?app .

                ?lengthQuantity qudt:numericValue ?lengthValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .

                ?tonnageQuantity qudt:numericValue ?tonnageValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .

                FILTER (
                    ?app !=
                    (
                        (?type = "fishing_vessel" && ?lengthValue >= 24) ||
                        (?type = "pleasure_yacht" && ?tonnageValue >= 300 && ?trade = false) ||
                        (?type = "cargo_ship" && ?tonnageValue >= 300 && ?tonnageValue < 500)
                    )
                )
            }
        """
    ] .
<END_SHACL>
