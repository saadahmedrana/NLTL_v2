<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:headingOrPositionSystem ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        sh:message "Each represented heading-reference or position-fixing system must have an intended-area suitability approval status of Approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:headingOrPositionSystem ?system .
                FILTER (
                    NOT EXISTS {
                        $this nltl:intendedAreaSuitabilityApprovalStatus
                              nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:intendedAreaSuitabilityApprovalStatus ?status .
                        FILTER (?status != nltl:evidenceStateApproved)
                    }
                )
            }
        """
    ] .
<END_SHACL>
