<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:requiredSensorProjectsBelowHull ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship must state whether required sensors project below the hull."
    ] ;
    sh:property [
        sh:path nltl:sensorIceProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Sensors that project below the hull must be protected against ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:requiredSensorProjectsBelowHull ?required .
                FILTER (?required = true)
                FILTER NOT EXISTS {
                    $this nltl:sensorIceProtectionPresent ?protected .
                    FILTER (?protected = true)
                }
            }
        """
    ] .
<END_SHACL>
