<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_016_Chapter11_1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship operating in polar waters that meets a Chapter 11-1 ship-type and threshold condition must have polarCodeChapter11Dash1Applicable true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters ?polar .
                FILTER(?polar = true)

                FILTER(
                    EXISTS {
                        $this nltl:shipType ?type .
                        FILTER(datatype(?type) = xsd:string)
                        FILTER(?type = "fishing_vessel")
                        $this nltl:lengthOverall ?lengthQuantity .
                        ?lengthQuantity qudt:numericValue ?length .
                        ?lengthQuantity qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER(datatype(?length) = xsd:decimal)
                        FILTER(?length >= 24)
                    }
                    ||
                    EXISTS {
                        $this nltl:shipType ?type .
                        FILTER(datatype(?type) = xsd:string)
                        FILTER(?type = "pleasure_yacht")
                        $this nltl:grossTonnage ?tonnageQuantity .
                        ?tonnageQuantity qudt:numericValue ?tonnage .
                        ?tonnageQuantity qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER(datatype(?tonnage) = xsd:decimal)
                        FILTER(?tonnage >= 300)
                        $this nltl:engagedInTrade ?trade .
                        FILTER(?trade = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:shipType ?type .
                        FILTER(datatype(?type) = xsd:string)
                        FILTER(?type = "cargo_ship")
                        $this nltl:grossTonnage ?tonnageQuantity .
                        ?tonnageQuantity qudt:numericValue ?tonnage .
                        ?tonnageQuantity qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER(datatype(?tonnage) = xsd:decimal)
                        FILTER(?tonnage >= 300)
                        FILTER(?tonnage < 500)
                    }
                )

                FILTER NOT EXISTS {
                    $this nltl:polarCodeChapter11Dash1Applicable true .
                }
            }
        """
    ] .
<END_SHACL>
