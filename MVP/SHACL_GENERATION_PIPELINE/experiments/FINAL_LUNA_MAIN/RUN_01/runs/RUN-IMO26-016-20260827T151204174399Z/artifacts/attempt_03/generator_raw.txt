<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-016> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When a ship operates in polar waters and meets a Chapter 11-1 applicability branch, polarCodeChapter11Dash1Applicable must be true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipOperatesInPolarWaters true .
    FILTER NOT EXISTS {
        {
            $this rdf:type nltl:fishingVessel .
            $this nltl:lengthOverall ?length .
            ?length qudt:numericValue ?lengthValue .
            ?length qudt:unit <http://qudt.org/vocab/unit/M> .
            FILTER(datatype(?lengthValue) = xsd:decimal && ?lengthValue >= 24)
            $this nltl:polarCodeChapter11Dash1Applicable true .
        }
        UNION
        {
            $this rdf:type nltl:pleasureYacht .
            $this nltl:grossTonnage ?tonnage .
            ?tonnage qudt:numericValue ?tonnageValue .
            ?tonnage qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
            FILTER(datatype(?tonnageValue) = xsd:decimal && ?tonnageValue >= 300)
            $this nltl:engagedInTrade false .
            $this nltl:polarCodeChapter11Dash1Applicable true .
        }
        UNION
        {
            $this rdf:type nltl:cargoShip .
            $this nltl:grossTonnage ?tonnage .
            ?tonnage qudt:numericValue ?tonnageValue .
            ?tonnage qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
            FILTER(datatype(?tonnageValue) = xsd:decimal && ?tonnageValue >= 300 && ?tonnageValue < 500)
            $this nltl:polarCodeChapter11Dash1Applicable true .
        }
    }
}
"""
    ] .
<END_SHACL>
