<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-009-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        ) ;
        sh:message "A ship must provide an allowed ice class for TRF-009 applicability evaluation."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to evaluate TRF-009 applicability."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A delivery date is required to evaluate the TRF-009 deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An assessment date is required to evaluate the TRF-009 deadline."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:message "TRF-009 applies only to an IA Super or IA ship whose keel was laid or construction reached a similar stage before 1 September 2003." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionStageDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            YEAR(?constructionStageDate) < 2003
                            ||
                            (
                                YEAR(?constructionStageDate) = 2003
                                && MONTH(?constructionStageDate) < 9
                            )
                            ||
                            (
                                YEAR(?constructionStageDate) = 2003
                                && MONTH(?constructionStageDate) = 9
                                && DAY(?constructionStageDate) < 1
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "For an applicable ship, the required 2017 Ice Class Regulations compliance method and deadline must be satisfied." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionStageDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            YEAR(?constructionStageDate) < 2003
                            ||
                            (
                                YEAR(?constructionStageDate) = 2003
                                && MONTH(?constructionStageDate) < 9
                            )
                            ||
                            (
                                YEAR(?constructionStageDate) = 2003
                                && MONTH(?constructionStageDate) = 9
                                && DAY(?constructionStageDate) < 1
                            )
                        )
                        FILTER (
                            NOT EXISTS {
                                $this nltl:engineOutputComplianceMethod ?method .
                                FILTER (
                                    ?method IN (
                                        nltl:traficom2017Section3Point2Point2Method,
                                        nltl:traficom2017Section3Point2Point4Method
                                    )
                                )
                            }
                            ||
                            !(
                                YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
                                ||
                                (
                                    YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                    && MONTH(?assessmentDate) = 1
                                    && DAY(?assessmentDate) = 1
                                )
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
