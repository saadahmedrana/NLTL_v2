<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF009ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice class." ;
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "Construction stage date, when provided, must be an xsd:date." ;
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:message "Delivery date, when provided, must be an xsd:date." ;
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:message "Assessment date, when provided, must be an xsd:date." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An IA or IA Super ship must provide a valid construction stage date before the conditional compliance rule can be assessed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionStageDate .
                    FILTER (datatype(?constructionStageDate) = xsd:date)
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA or IA Super ship constructed before 1 September 2003 must have an allowed 2017 compliance method and an assessment no later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))

                $this nltl:constructionStageDate ?constructionStageDate .
                FILTER (datatype(?constructionStageDate) = xsd:date)
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        (
                            MONTH(?constructionStageDate) < 9 ||
                            (
                                MONTH(?constructionStageDate) = 9 &&
                                DAY(?constructionStageDate) < 1
                            )
                        )
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                    $this nltl:assessmentDate ?assessmentDate .
                    $this nltl:engineOutputComplianceMethod ?method .

                    FILTER (datatype(?deliveryDate) = xsd:date)
                    FILTER (datatype(?assessmentDate) = xsd:date)
                    FILTER (?method IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    ))

                    FILTER (
                        YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                        (
                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                            (
                                MONTH(?assessmentDate) < 1 ||
                                (
                                    MONTH(?assessmentDate) = 1 &&
                                    DAY(?assessmentDate) <= 1
                                )
                            )
                        )
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
