<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF012ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF012IceWaterlineShape ;
        sh:message "The ship must represent each intended ice-operating waterline with profile points." ;
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF012IceWaterlineShape ;
        sh:message "The ship must represent the externally determined lower ice waterline with profile points." ;
    ] .

nltlgen:TRF012IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF012ProfilePointShape ;
        sh:message "Each represented ice waterline must have at least one waterline profile point." ;
    ] .

nltlgen:TRF012ProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:node nltlgen:TRF012LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must carry one longitudinal position in metres." ;
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:node nltlgen:TRF012LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must carry one vertical coordinate in metres." ;
    ] .

nltlgen:TRF012LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
