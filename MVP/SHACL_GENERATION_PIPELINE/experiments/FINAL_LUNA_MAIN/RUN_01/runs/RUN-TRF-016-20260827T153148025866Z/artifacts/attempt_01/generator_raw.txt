                $this nltl:navigatingInIce true ;

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

                $this nltl:navigatingInIce true ;
                      nltl:forwardDraught ?forwardNode ;
                      nltl:displacementAtUpperIceWaterline ?displacementNode ;

                ?displacementNode qudt:numericValue ?displacement .
                ?thicknessNode qudt:numericValue ?thickness .

                FILTER (ABS(?forward - ?expected) >
                        ((0.000001 * ABS(?expected)) + 0.000001))
