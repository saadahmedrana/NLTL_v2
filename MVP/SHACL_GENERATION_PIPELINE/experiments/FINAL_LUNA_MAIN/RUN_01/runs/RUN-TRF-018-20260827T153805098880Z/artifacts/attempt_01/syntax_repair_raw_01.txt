<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-018> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceClass is required to determine the applicable engine-output threshold."
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "calculatedRequiredPower must be a single kilowatt quantity with a decimal numeric value."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "maximumContinuousRatingPower must be a single kilowatt quantity with a decimal numeric value."
    ] ;
    sh:sparql [
        sh:message "maximumContinuousRatingPower must be at least the calculated required power and at least 1000 kW for ice classes IA, IB, or IC, or at least 2800 kW for ice class IA Super." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedQuantity ;
                      nltl:maximumContinuousRatingPower ?maximumQuantity .
                ?calculatedQuantity qudt:numericValue ?calculatedPower ;
                                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?maximumQuantity qudt:numericValue ?maximumPower ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                FILTER (
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        )
                        && (
                            ?maximumPower < ?calculatedPower
                            || ?maximumPower < 1000
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper
                        && (
                            ?maximumPower < ?calculatedPower
                            || ?maximumPower < 2800
                        )
                    )
                )
            }
        """
    ] .
</BEGIN_SHACL>
