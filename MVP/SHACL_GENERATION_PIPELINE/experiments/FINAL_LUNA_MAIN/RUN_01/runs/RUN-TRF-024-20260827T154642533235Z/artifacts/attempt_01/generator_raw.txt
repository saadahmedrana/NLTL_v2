<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-024-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:message "A ship must provide one of the canonical Finnish-Swedish ice-class values."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The construction stage date is required to determine applicability of the twenty-year ice-class deadline."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?ice ;
          nltl:constructionStageDate ?construction .
    FILTER (
        ?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa)
        &&
        (
            YEAR(?construction) < 2003
            ||
            (
                YEAR(?construction) = 2003
                && MONTH(?construction) < 9
            )
            ||
            (
                YEAR(?construction) = 2003
                && MONTH(?construction) = 9
                && DAY(?construction) < 1
            )
        )
    )
}
""" ;
                sh:message "The non-applicable branch is valid only when the ship is not ice class IA Super or IA, or its construction stage was not before 1 September 2003."
            ]
        ] ;
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?ice ;
          nltl:constructionStageDate ?construction .
    FILTER (
        NOT (
            ?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa)
            &&
            (
                YEAR(?construction) < 2003
                ||
                (
                    YEAR(?construction) = 2003
                    && MONTH(?construction) < 9
                )
                ||
                (
                    YEAR(?construction) = 2003
                    && MONTH(?construction) = 9
                    && DAY(?construction) < 1
                )
            )
        )
    )
}
""" ;
                sh:message "The applicable branch requires an ice class IA Super or IA ship whose construction stage was before 1 September 2003."
            ] ;
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?ice ;
          nltl:constructionStageDate ?construction .
    FILTER (
        ?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa)
        &&
        (
            YEAR(?construction) < 2003
            ||
            (
                YEAR(?construction) = 2003
                && MONTH(?construction) < 9
            )
            ||
            (
                YEAR(?construction) = 2003
                && MONTH(?construction) = 9
                && DAY(?construction) < 1
            )
        )
    )
    FILTER (
        NOT EXISTS {
            $this nltl:deliveryDate ?delivery .
        }
        ||
        NOT EXISTS {
            $this nltl:assessmentDate ?assessment .
        }
        ||
        NOT EXISTS {
            $this nltl:deliveryDate ?delivery ;
                  nltl:assessmentDate ?assessment .
            FILTER (
                YEAR(?assessment) < YEAR(?delivery) + 20
                ||
                (
                    YEAR(?assessment) = YEAR(?delivery) + 20
                    && MONTH(?assessment) = 1
                    && DAY(?assessment) = 1
                )
            )
        }
    )
}
""" ;
                sh:message "For an applicable IA Super or IA ship, the assessment date must be no later than 1 January of the year twenty years after delivery."
            ]
        ]
    ) .
<END_SHACL>
