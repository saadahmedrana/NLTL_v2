nltl:TRF025PositiveLengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0 ;
        sh:message "The length operand must be a strictly positive decimal quantity."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length operand must use metres."
    ] .

nltl:TRF025NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF025Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one controlled ice class."
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one boolean bulbous-bow indication."
    ] ;
    sh:or (
        [ sh:node nltl:TRF025ApplicableShape ]
        [ sh:node nltl:TRF025NonApplicableShape ]
    ) .
