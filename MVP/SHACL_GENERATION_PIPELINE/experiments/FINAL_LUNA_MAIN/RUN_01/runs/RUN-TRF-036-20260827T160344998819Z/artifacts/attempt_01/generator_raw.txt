<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF036ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:node <urn:nltl:generated-shape:TRF036ParameterSetShape>
    ] .

<urn:nltl:generated-shape:TRF036ParameterSetShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The ice class, level ice thickness, and design ice load height must match Table 4-1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:iceClass ?iceClass .
                    $this nltl:levelIceThickness ?levelNode .
                    $this nltl:designIceLoadHeight ?heightNode .
                    ?levelNode qudt:numericValue ?level .
                    ?heightNode qudt:numericValue ?height .
                    FILTER (
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?level <= 1.0 &&
                            ?height >= 0.35 &&
                            ?height <= 0.35
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIa &&
                            ?level <= 0.8 &&
                            ?height >= 0.30 &&
                            ?height <= 0.30
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIb &&
                            ?level <= 0.6 &&
                            ?height >= 0.25 &&
                            ?height <= 0.25
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIc &&
                            ?level <= 0.4 &&
                            ?height >= 0.22 &&
                            ?height <= 0.22
                        )
                    )
                }
            }
        """
    ] .
<END_SHACL>
