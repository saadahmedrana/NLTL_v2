<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF036IceClassDesignParameterSetShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The parameter set must provide one level ice thickness in metres."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The parameter set must provide one design ice load height in metres."
    ] ;
    sh:property [
        sh:path (
            [
                sh:inversePath nltl:hasIceClassDesignParameterSet
            ]
            nltl:iceClass
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:message "The parameter set must be linked from one ship with one permitted Finnish-Swedish ice class."
    ] ;
    sh:sparql [
        sh:message "The level ice thickness and design ice load height must satisfy the Table 4-1 values for the linked ship's ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                      nltl:iceClass ?class .

                $this nltl:levelIceThickness ?hiQuantity ;
                       nltl:designIceLoadHeight ?hQuantity .

                ?hiQuantity qudt:numericValue ?hi .
                ?hQuantity qudt:numericValue ?h .

                FILTER (
                    !(
                        (
                            ?class = nltl:iceClassIaSuper &&
                            ?hi <= "1.0"^^xsd:decimal &&
                            ?h >= "0.35"^^xsd:decimal &&
                            ?h <= "0.35"^^xsd:decimal
                        ) ||
                        (
                            ?class = nltl:iceClassIa &&
                            ?hi <= "0.8"^^xsd:decimal &&
                            ?h >= "0.30"^^xsd:decimal &&
                            ?h <= "0.30"^^xsd:decimal
                        ) ||
                        (
                            ?class = nltl:iceClassIb &&
                            ?hi <= "0.6"^^xsd:decimal &&
                            ?h >= "0.25"^^xsd:decimal &&
                            ?h <= "0.25"^^xsd:decimal
                        ) ||
                        (
                            ?class = nltl:iceClassIc &&
                            ?hi <= "0.4"^^xsd:decimal &&
                            ?h >= "0.22"^^xsd:decimal &&
                            ?h <= "0.22"^^xsd:decimal
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
