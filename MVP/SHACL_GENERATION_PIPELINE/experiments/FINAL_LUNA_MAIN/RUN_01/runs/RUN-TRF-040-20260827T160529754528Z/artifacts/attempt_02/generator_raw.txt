<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF040Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted Finnish-Swedish ice class." ;
    ] ;
    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sidescuttles shall not be situated in the ice belt." ;
    ] ;
    sh:property [
        sh:path nltl:openWaterServiceSpeed ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bowRegionEquivalent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forefootShellStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:midbodyEquivalent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulwarkStrengthStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freeingPortStrengthStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The conditional ice-belt strengthening requirements, including the applicable spatial extent, are not satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    $this nltl:iceClass nltl:iceClassIaSuper .
                    FILTER NOT EXISTS {
                        $this nltl:spacing ?spacing .
                    }
                }
                UNION
                {
                    $this nltl:iceClass nltl:iceClassIaSuper .
                    FILTER NOT EXISTS {
                        $this nltl:forefootShellStrengtheningStatus ?forefoot .
                    }
                }
                UNION
                {
                    $this nltl:iceClass nltl:iceClassIaSuper .
                    FILTER NOT EXISTS {
                        $this nltl:bowRegionEquivalent ?bow .
                    }
                }
                UNION
                {
                    $this nltl:iceClass nltl:iceClassIaSuper ;
                          nltl:forefootShellStrengtheningStatus ?forefoot ;
                          nltl:bowRegionEquivalent ?bow .
                    FILTER (?forefoot != ?bow)
                }
                UNION
                {
                    $this nltl:iceClass ?iceClass ;
                          nltl:openWaterServiceSpeed ?speedNode .
                    FILTER (
                        ?iceClass = nltl:iceClassIaSuper ||
                        ?iceClass = nltl:iceClassIa
                    )
                    ?speedNode <http://qudt.org/schema/qudt/numericValue> ?speed .
                    FILTER (?speed >= 18)
                    FILTER NOT EXISTS {
                        $this nltl:upperBowIceBeltStrengtheningStatus ?upper .
                    }
                }
                UNION
                {
                    $this nltl:iceClass ?iceClass ;
                          nltl:openWaterServiceSpeed ?speedNode .
                    FILTER (
                        ?iceClass = nltl:iceClassIaSuper ||
                        ?iceClass = nltl:iceClassIa
                    )
                    ?speedNode <http://qudt.org/schema/qudt/numericValue> ?speed .
                    FILTER (?speed >= 18)
                    FILTER NOT EXISTS {
                        $this nltl:midbodyEquivalent ?midbody .
                    }
                }
                UNION
                {
                    $this nltl:iceClass ?iceClass ;
                          nltl:openWaterServiceSpeed ?speedNode ;
                          nltl:upperBowIceBeltStrengtheningStatus ?upper ;
                          nltl:midbodyEquivalent ?midbody .
                    FILTER (
                        ?iceClass = nltl:iceClassIaSuper ||
                        ?iceClass = nltl:iceClassIa
                    )
                    ?speedNode <http://qudt.org/schema/qudt/numericValue> ?speed .
                    FILTER (?speed >= 18)
                    FILTER (?upper != ?midbody)
                }
                UNION
                {
                    $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                    FILTER NOT EXISTS {
                        $this nltl:bulwarkStrengthStatus true .
                    }
                }
                UNION
                {
                    $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                    FILTER NOT EXISTS {
                        $this nltl:freeingPortStrengthStatus true .
                    }
                }
            }
        """
    ] .
<END_SHACL>
