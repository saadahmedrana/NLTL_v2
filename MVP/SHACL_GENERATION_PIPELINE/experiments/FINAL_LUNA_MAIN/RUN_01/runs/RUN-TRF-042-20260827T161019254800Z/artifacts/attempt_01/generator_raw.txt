            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                      nltl:spacing ?sQuantity ;

                BIND(

                        (0.000001 * ABS(?expectedF2)) > 0.000000000001,
                        0.000001 * ABS(?expectedF2),

                    AS ?tolerance

                    ?ratio > 1.8 ||
                    ABS(?actualF2 - ?expectedF2) > ?tolerance
                )
            }
        """
