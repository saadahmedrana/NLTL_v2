<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node gen:TRF043FrameShape
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:tankTop ]
            )
        ]
    ] ;
    sh:sparql [
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .
                ?extension qudt:numericValue ?value .
                FILTER (datatype(?value) = xsd:decimal)
                FILTER (?value <= 250.0)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true
                }
            }
        """
    ] .

gen:TRF043FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Framing vertical extensions must satisfy the applicable Table 4-6 ice-class and hull-region values, including the upper bow ice-belt condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQuantity ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowQuantity ;
                      nltl:upperBowIceBeltRequired ?upperRequired ;
                      nltl:reachesUpperBowIceBeltTop ?reachesUpper ;
                      nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .
                ?aboveQuantity qudt:numericValue ?above .
                ?belowQuantity qudt:numericValue ?below .
                FILTER (datatype(?above) = xsd:decimal)
                FILTER (datatype(?below) = xsd:decimal)
                FILTER (
                    (!?upperRequired || ?reachesUpper = true)
                    &&
                    (
                        (
                            ?iceClass = nltl:iceClassIaSuper
                            &&
                            (
                                (
                                    ?region = nltl:bowRegion
                                    &&
                                    ?reachesTankTop = true
                                    &&
                                    ?below >= 2.0
                                )
                                ||
                                (
                                    ?region = nltl:midbodyRegion
                                    &&
                                    ?below >= 2.0
                                )
                                ||
                                (
                                    ?region = nltl:sternRegion
                                    &&
                                    ?below >= 1.6
                                )
                            )
                            &&
                            ?above >= 1.2
                        )
                        ||
                        (
                            (
                                ?iceClass = nltl:iceClassIa
                                ||
                                ?iceClass = nltl:iceClassIb
                                ||
                                ?iceClass = nltl:iceClassIc
                            )
                            &&
                            (
                                (
                                    ?region = nltl:bowRegion
                                    &&
                                    ?below >= 1.6
                                )
                                ||
                                (
                                    ?region = nltl:midbodyRegion
                                    &&
                                    ?below >= 1.3
                                )
                                ||
                                (
                                    ?region = nltl:sternRegion
                                    &&
                                    ?below >= 1.0
                                )
                            )
                            &&
                            ?above >= 1.0
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
