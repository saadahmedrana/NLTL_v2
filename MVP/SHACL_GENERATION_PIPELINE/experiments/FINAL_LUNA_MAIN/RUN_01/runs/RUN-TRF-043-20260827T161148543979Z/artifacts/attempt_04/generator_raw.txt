<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] .

gen:TRF043FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing vertical extensions do not satisfy the applicable Table 4-6 values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQ ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowQ .
                ?aboveQ qudt:numericValue ?above .
                ?belowQ qudt:numericValue ?below .
                FILTER (
                    !(
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?region = nltl:bowRegion &&
                            EXISTS {
                                $this nltl:reachesTankTopOrBelowFloorTop true
                            } &&
                            ?above >= 1.2
                        )
                        ||
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?region = nltl:midbodyRegion &&
                            ?above >= 1.2 &&
                            ?below >= 2.0
                        )
                        ||
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?region = nltl:sternRegion &&
                            ?above >= 1.2 &&
                            ?below >= 1.6
                        )
                        ||
                        (
                            ?iceClass IN (
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            ?region = nltl:bowRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.6
                        )
                        ||
                        (
                            ?iceClass IN (
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            ?region = nltl:midbodyRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.3
                        )
                        ||
                        (
                            ?iceClass IN (
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            ?region = nltl:sternRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.0
                        )
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an upper bow ice belt is required, framing ice strengthening must reach at least the recorded upper bow ice belt top." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperBowIceBeltRequired true ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQ .
                ?aboveQ qudt:numericValue ?above .
                FILTER NOT EXISTS {
                    ?ship nltl:hasComponent $this ;
                          nltl:upperBowIceBeltTop ?top .
                    FILTER (xsd:decimal(?top) <= ?above)
                }
            }
        """
    ] .

gen:TRF043MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extensionQ .
                ?extensionQ qudt:numericValue ?extension .
                FILTER (?extension <= 250)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true
                }
            }
        """
    ] .

gen:TRF043MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
