<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A frame shall have a strengthened part."
    ] ;
    sh:property [
        sh:path ( nltl:hasStrengthenedPart nltl:hasLowerEnd ) ;
        sh:minCount 1 ;
        sh:message "Each frame shall have a strengthened part with a lower end."
    ] ;
    sh:sparql [
        sh:message "Each strengthened part of a main frame or intermediate ice frame shall be a transverse frame with a lower end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:frame ;
                      nltl:hasStrengthenedPart ?part .
                FILTER (
                    EXISTS { $this a nltl:mainFrame }
                    || EXISTS { $this a nltl:intermediateIceFrame }
                )
                FILTER NOT EXISTS {
                    ?part a nltl:transverseFrame ;
                          nltl:hasLowerEnd ?end .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The lower end shall be attached to a supporting structure or satisfy the permitted horizontal-member alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:frame ;
                      nltl:hasStrengthenedPart ?part .
                FILTER (
                    EXISTS { $this a nltl:mainFrame }
                    || EXISTS { $this a nltl:intermediateIceFrame }
                )
                ?part a nltl:transverseFrame ;
                      nltl:hasLowerEnd ?end .
                FILTER NOT EXISTS {
                    ?end nltl:hasAttachedSupportingStructure ?support .
                }
                FILTER NOT EXISTS {
                    ?end nltl:hasTerminationBelowSupportingStructure ?support ;
                         nltl:hasHorizontalConnectionMember ?member .
                    ?member nltl:connectsToAdjacentMainFrame ?main .
                    $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true ;
                          nltl:sameScantlingsAsMainFrame true .
                    ?main nltl:mainFrameBelowIceBeltStrengthened true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A main frame shall be ice strengthened below the lower edge of the ice belt." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:frame ,
                      nltl:mainFrame .
                FILTER NOT EXISTS {
                    $this nltl:mainFrameBelowIceBeltStrengthened true .
                }
            }
        """
    ] .
<END_SHACL>
