<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each strengthened main frame or intermediate ice frame must have a lower end attached to a deck, tank top, tank-boundary plating, or ice stringer." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStrengthenedPart ?part .
                ?part rdf:type ?partType .
                FILTER (?partType IN (
                    nltl:mainFrame,
                    nltl:intermediateIceFrame
                ))
                FILTER NOT EXISTS {
                    ?part nltl:hasLowerEnd ?lowerEnd .
                    ?lowerEnd nltl:hasAttachedSupportingStructure ?support .
                    ?support rdf:type ?supportType .
                    FILTER (?supportType IN (
                        nltl:deckStructure,
                        nltl:tankTop,
                        nltl:tankBoundaryPlating,
                        nltl:iceStringer
                    ))
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A represented horizontal connection member must be conditional on an intermediate ice frame terminating below a permitted supporting structure at or below the ice-belt lower limit, and must connect to an adjacent main frame with the same scantlings." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStrengthenedPart ?intermediate .
                ?intermediate rdf:type nltl:intermediateIceFrame .
                ?intermediate nltl:hasLowerEnd ?lowerEnd .
                ?lowerEnd nltl:hasHorizontalConnectionMember ?member .
                FILTER NOT EXISTS {
                    ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?support .
                    ?support rdf:type ?supportType .
                    FILTER (?supportType IN (
                        nltl:deckStructure,
                        nltl:tankTop,
                        nltl:tankBoundaryPlating,
                        nltl:iceStringer
                    ))
                    ?intermediate nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
                    ?member rdf:type nltl:horizontalConnectionMember .
                    ?member nltl:connectsToAdjacentMainFrame ?mainFrame .
                    ?mainFrame rdf:type nltl:mainFrame .
                    ?intermediate nltl:sameScantlingsAsMainFrame true .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each strengthened main frame below the ice belt must be ice strengthened." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStrengthenedPart ?mainPart .
                ?mainPart rdf:type nltl:mainFrame .
                FILTER NOT EXISTS {
                    ?mainPart nltl:mainFrameBelowIceBeltStrengthened true
                }
            }
        """
    ] .

</END_SHACL>
