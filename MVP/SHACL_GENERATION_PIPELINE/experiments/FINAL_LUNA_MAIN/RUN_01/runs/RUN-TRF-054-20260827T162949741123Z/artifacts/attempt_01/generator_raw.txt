nltlg:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltlg:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltlg:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one dimensionless frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltlg:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one distance to the ice belt in metres."
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one stringer span in metres."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltlg:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design line load must be present in kN/m and equal the regulated calculation."
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node nltlg:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The outside-ice-belt factor must be present and equal 1 minus the belt distance divided by the adjacent stringer distance."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltlg:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The section modulus must be present in cm3 and equal the regulated calculation."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltlg:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The required shear area must be present in cm2 and equal the regulated calculation."
    ] ;
    sh:sparql [
        sh:message "The design line load, outside-ice-belt factor, section modulus, and required shear area do not satisfy the TRF-054 formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?loadNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?loadNode qudt:numericValue ?reportedLoad .
                ?factorNode qudt:numericValue ?reportedFactor .
                ?zNode qudt:numericValue ?reportedZ .
                ?aNode qudt:numericValue ?reportedA .

                BIND(IF((?p * ?h) > 0.15, (?p * ?h), 0.15) AS ?loadMN)
                BIND((1000 * ?loadMN) AS ?expectedLoad)
                BIND((1 - (?hs / ?ls)) AS ?expectedFactor)
                BIND((0.80 * 1.8 * ?loadMN * ?l * ?l / (?m * ?sigma) * ?expectedFactor * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?loadMN * ?l / (2 * ?sigma) * ?expectedFactor * 10000) AS ?expectedA)

                BIND(IF((ABS(?expectedLoad) * 0.000001) > 0.000000001,
                    (ABS(?expectedLoad) * 0.000001), 0.000000001) AS ?tolLoad)
                BIND(IF((ABS(?expectedFactor) * 0.000001) > 0.000000001,
                    (ABS(?expectedFactor) * 0.000001), 0.000000001) AS ?tolFactor)
                BIND(IF((ABS(?expectedZ) * 0.000001) > 0.000000001,
                    (ABS(?expectedZ) * 0.000001), 0.000000001) AS ?tolZ)
                BIND(IF((ABS(?expectedA) * 0.000001) > 0.000000001,
                    (ABS(?expectedA) * 0.000001), 0.000000001) AS ?tolA)

                FILTER(
                    ABS(?reportedLoad - ?expectedLoad) > ?tolLoad ||
                    ABS(?reportedFactor - ?expectedFactor) > ?tolFactor ||
                    ABS(?reportedZ - ?expectedZ) > ?tolZ ||
                    ABS(?reportedA - ?expectedA) > ?tolA
                )
            }
        """
    ] .
<END_SHACL>
