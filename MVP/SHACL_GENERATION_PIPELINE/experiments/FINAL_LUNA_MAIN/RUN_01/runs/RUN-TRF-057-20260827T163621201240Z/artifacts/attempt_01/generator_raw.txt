<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF057WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "webFrameIceLoad must equal 1800 times webFrameSpacing times the greater of icePressure multiplied by iceLoadHeight and 0.15, within a relative tolerance of 1e-6 with an absolute fallback of 1e-6 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:iceLoadHeight ?heightQuantity ;
                      nltl:webFrameSpacing ?spacingQuantity ;
                      nltl:webFrameIceLoad ?loadQuantity .

                ?pressureQuantity qudt:numericValue ?pressure ;
                                  qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spacingQuantity qudt:numericValue ?spacing ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                ?loadQuantity qudt:numericValue ?actualLoad ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                BIND ((?pressure * ?height) AS ?pressureHeight)
                BIND (IF(?pressureHeight < 0.15, 0.15, ?pressureHeight) AS ?effectiveLineLoad)
                BIND ((1800 * ?effectiveLineLoad * ?spacing) AS ?expectedLoad)
                BIND (ABS(?actualLoad - ?expectedLoad) AS ?difference)

                FILTER (
                    ?difference > 0.000001 &&
                    ?difference > (ABS(?expectedLoad) * 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
