<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF058QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF058NotSupportedCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The supported stringer applicability state must be recorded as false when the stringer is not outside the ice belt."
    ] .

nltlgen:TRF058SupportedCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The supported stringer applicability state must be recorded as true for the outside-ice-belt calculation."
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:node nltlgen:TRF058QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:node nltlgen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node nltlgen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:node nltlgen:TRF058QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadForce ?iceQuantity ;
                      nltl:stringerOutsideIceBeltHeight ?heightQuantity ;
                      nltl:stringerSpan ?spanQuantity ;
                      nltl:adjustedIceLoadForce ?adjustedQuantity .
                ?iceQuantity qudt:numericValue ?ice ;
                             qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spanQuantity qudt:numericValue ?span ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .
                ?adjustedQuantity qudt:numericValue ?adjusted ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                BIND (?ice * (1 - (?height / ?span)) AS ?expected)
                FILTER (
                    ABS(?adjusted - ?expected) > (ABS(?expected) * 0.000001)
                    &&
                    ABS(?adjusted - ?expected) > 0.000000001
                )
            }
        """
    ] .

nltlgen:TRF058LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:or (
        nltlgen:TRF058NotSupportedCase
        nltlgen:TRF058SupportedCase
    ) .

</END_SHACL>
