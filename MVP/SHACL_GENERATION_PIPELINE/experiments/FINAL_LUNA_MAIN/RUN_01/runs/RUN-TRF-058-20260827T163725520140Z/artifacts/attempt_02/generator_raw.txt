        sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within numerical tolerance." ;

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true ;
                      nltl:iceLoadForce ?forceQuantity ;
                      nltl:stringerOutsideIceBeltHeight ?heightQuantity ;
                      nltl:stringerSpan ?spanQuantity ;
                      nltl:adjustedIceLoadForce ?adjustedQuantity .

                ?forceQuantity qudt:numericValue ?force .
                ?heightQuantity qudt:numericValue ?height .
                ?spanQuantity qudt:numericValue ?span .
                ?adjustedQuantity qudt:numericValue ?adjusted .

                BIND (?force * (1 - (?height / ?span)) AS ?expected)
                BIND (IF(?expected < 0, 0 - ?expected, ?expected) AS ?expectedMagnitude)
                BIND (IF((1e-6 * ?expectedMagnitude) > 1e-9,
                         1e-6 * ?expectedMagnitude,
                         1e-9) AS ?tolerance)

                FILTER (
                    (?adjusted < (?expected - ?tolerance)) ||
                    (?adjusted > (?expected + ?tolerance))
                )
            }
        """
    ] .
