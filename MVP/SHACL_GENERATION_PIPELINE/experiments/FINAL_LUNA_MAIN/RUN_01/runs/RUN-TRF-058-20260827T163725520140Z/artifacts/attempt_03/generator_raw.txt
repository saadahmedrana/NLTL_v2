@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "supportedStringerOutsideIceBelt must be explicitly provided as true or false."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:stringerOutsideIceBeltHeight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/M>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:stringerSpan ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/M>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:adjustedIceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan) within the stated tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true ;
                      nltl:iceLoadForce ?forceQuantity ;
                      nltl:stringerOutsideIceBeltHeight ?heightQuantity ;
                      nltl:stringerSpan ?spanQuantity ;
                      nltl:adjustedIceLoadForce ?adjustedQuantity .

                ?forceQuantity qudt:numericValue ?force ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spanQuantity qudt:numericValue ?span ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .
                ?adjustedQuantity qudt:numericValue ?adjusted ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                BIND (?force * (1 - (?height / ?span)) AS ?expected)
                BIND (IF(?expected >= 0, ?expected, 0 - ?expected) AS ?magnitude)
                BIND (IF(?magnitude > 0.001, ?magnitude * 0.000001, 0.000001) AS ?tolerance)

                FILTER (
                    ?adjusted < (?expected - ?tolerance) ||
                    ?adjusted > (?expected + ?tolerance)
                )
            }
        """
    ] .
