<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF063ApplicableShipBranch
        gen:TRF063NonApplicableShipBranch
    ) ;
    sh:message "Twin- and triple-screw ships shall have the required side-propeller strengthening envelope; other ships are not subject to this conditional envelope constraint." .

gen:TRF063ApplicableShipBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:maxInclusive 3
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:node gen:TRF063EnvelopeShape
    ] .

gen:TRF063NonApplicableShipBranch a sh:NodeShape ;
    sh:or (
        gen:TRF063SingleOrZeroPropellerBranch
        gen:TRF063FourOrMorePropellersBranch
    ) .

gen:TRF063SingleOrZeroPropellerBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:maxInclusive 1
    ] .

gen:TRF063FourOrMorePropellersBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:minInclusive 4
    ] .

gen:TRF063EnvelopeShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:message "The side-propeller strengthening envelope shall identify the tank top as its boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF063LengthQuantityShape ;
        sh:message "The forward strengthening extent shall be at least 1.5 metres."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF063LengthQuantityShape ;
        sh:message "The aft strengthening extent shall be at least 1.5 metres."
    ] .

gen:TRF063LengthQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.5
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

<END_SHACL>
