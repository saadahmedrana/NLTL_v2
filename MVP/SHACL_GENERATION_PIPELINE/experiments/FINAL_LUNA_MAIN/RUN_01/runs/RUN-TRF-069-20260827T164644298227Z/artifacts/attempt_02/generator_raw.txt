<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_069_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:message "The ship must specify an applicable ice class."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:message "The ship must specify an open or ducted propeller type."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:message "The ship must specify a controllable or fixed pitch type."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The maximum ice torque must be provided as a quantity in kN*m."
    ] ;
    sh:sparql [
        sh:message "For an applicable propulsion-machinery configuration, one linked design condition must affirm normal operational conditions, normal service-life propeller ice loads, fixed-pitch reversal loads, and the exclusion of stopped-propeller dragging and radial ice-entry loads from the prescribed model." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?condition .
                    ?condition nltl:normalOperationalConditions true ;
                               nltl:normalServiceLifePropellerIceLoadCondition true ;
                               nltl:fixedPitchPropellerReversalLoadIncluded true ;
                               nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                               nltl:radialIceEntryOutsideLoadModel true .
                }
            }
        """
    ] .
<END_SHACL>
