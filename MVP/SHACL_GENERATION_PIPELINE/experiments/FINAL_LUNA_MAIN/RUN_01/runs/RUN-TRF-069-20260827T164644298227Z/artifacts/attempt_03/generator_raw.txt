<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        [
            sh:node <urn:nltl:generated-shape:TRF-069-ApplicableShape>
        ]
        [
            sh:node <urn:nltl:generated-shape:TRF-069-NonApplicableShape>
        ]
    ) .

<urn:nltl:generated-shape:TRF-069-ApplicableShape>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "An applicable propulsion-machinery design condition must provide all required ice-load scope assertions as typed boolean true values on one linked design condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?condition .
                    ?condition nltl:normalOperationalConditions ?normalOperational ;
                               nltl:normalServiceLifePropellerIceLoadCondition ?serviceLife ;
                               nltl:fixedPitchPropellerReversalLoadIncluded ?reversal ;
                               nltl:stoppedPropellerDraggingOutsideLoadModel ?dragging ;
                               nltl:radialIceEntryOutsideLoadModel ?radialEntry .

                    FILTER (
                        datatype(?normalOperational) = xsd:boolean
                        && ?normalOperational = true
                        && datatype(?serviceLife) = xsd:boolean
                        && ?serviceLife = true
                        && datatype(?reversal) = xsd:boolean
                        && ?reversal = true
                        && datatype(?dragging) = xsd:boolean
                        && ?dragging = true
                        && datatype(?radialEntry) = xsd:boolean
                        && ?radialEntry = true
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-069-NonApplicableShape>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This alternative is only valid when the ship is outside the specified ice-class, propeller-type, or pitch-type applicability combination." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )
            }
        """
    ] .
<END_SHACL>
