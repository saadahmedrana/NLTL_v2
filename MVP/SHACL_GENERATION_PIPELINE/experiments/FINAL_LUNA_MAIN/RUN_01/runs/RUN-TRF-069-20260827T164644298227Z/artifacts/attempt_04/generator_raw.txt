<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:designCondition
    ] ;
    sh:sparql [
        sh:message "For an applicable propulsion-machinery configuration, one linked design condition must assert all five required boolean conditions as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ic ;
                      nltl:propellerType ?pt ;
                      nltl:pitchType ?pitch .

                FILTER (
                    ?ic IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
                FILTER (?pt IN ("open", "ducted"))
                FILTER (?pitch IN ("controllable", "fixed"))

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?dc .
                    ?dc rdf:type nltl:designCondition ;
                        nltl:normalOperationalConditions ?normal ;
                        nltl:normalServiceLifePropellerIceLoadCondition ?service ;
                        nltl:fixedPitchPropellerReversalLoadIncluded ?reversal ;
                        nltl:stoppedPropellerDraggingOutsideLoadModel ?dragging ;
                        nltl:radialIceEntryOutsideLoadModel ?radial .

                    FILTER (
                        datatype(?normal) = xsd:boolean &&
                        ?normal = true &&
                        datatype(?service) = xsd:boolean &&
                        ?service = true &&
                        datatype(?reversal) = xsd:boolean &&
                        ?reversal = true &&
                        datatype(?dragging) = xsd:boolean &&
                        ?dragging = true &&
                        datatype(?radial) = xsd:boolean &&
                        ?radial = true
                    )
                }
            }
        """
    ] .
<END_SHACL>
