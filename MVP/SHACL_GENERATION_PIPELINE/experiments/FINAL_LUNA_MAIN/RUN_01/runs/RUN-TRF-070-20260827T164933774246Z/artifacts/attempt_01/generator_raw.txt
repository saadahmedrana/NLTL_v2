gs:TRF070_LocalStrengthResidualShape
    a sh:NodeShape ;
    sh:targetClass nltl:designCondition ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable main-propulsion azimuthing or fixed thruster, local-strength capacity shall be at least the local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasDesignCondition $this ;
                      nltl:mainPropulsionThruster true ;
                      nltl:thrusterType ?type ;
                      nltl:localIcePressure ?pressure .
                FILTER (STR(?type) IN ("azimuthing", "fixed"))
                ?pressure qudt:numericValue ?pressureValue .
                $this nltl:designConditionLocalStrengthCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .
