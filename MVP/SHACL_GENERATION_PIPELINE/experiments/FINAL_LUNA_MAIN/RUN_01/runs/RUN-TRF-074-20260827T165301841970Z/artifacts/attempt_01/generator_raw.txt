<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF074Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:node nltlgen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:node nltlgen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For ice class IB or IC, when ballast-condition propeller top depth is below hi, the propulsion system shall be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                ?depthNode qudt:numericValue ?depth .

                $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                ?hiNode qudt:numericValue ?hi .

                FILTER (?depth < ?hi)

                FILTER NOT EXISTS {
                    $this nltl:propulsionSystemDesignedForIceClass
                        nltl:iceClassIa
                }
            }
        """
    ] .
<END_SHACL>
