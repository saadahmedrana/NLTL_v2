<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-077-ForceQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-077-ShapeParameterQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-077-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the ice load cycle count for the spectrum."
    ] ;
    sh:property [
        sh:path nltl:bladeIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-077-ForceQuantityShape> ;
        sh:message "The ship must provide the blade ice load in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:maximumBladeIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-077-ForceQuantityShape> ;
        sh:message "The ship must provide the maximum blade ice load in kilonewtons."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "open" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:loadSpectrumShapeParameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:node <urn:nltl:generated-shape:TRF-077-ShapeParameterQuantityShape> ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "0.75"^^xsd:decimal ;
                        sh:maxInclusive "0.75"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "ducted" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:loadSpectrumShapeParameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:node <urn:nltl:generated-shape:TRF-077-ShapeParameterQuantityShape> ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "1.0"^^xsd:decimal ;
                        sh:maxInclusive "1.0"^^xsd:decimal
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The blade ice load must satisfy 0 <= blade ice load <= maximum blade ice load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeIceLoad ?load .
                ?load qudt:numericValue ?loadValue .
                $this nltl:maximumBladeIceLoad ?maximum .
                ?maximum qudt:numericValue ?maximumValue .
                FILTER (
                    ?loadValue < "0"^^xsd:decimal ||
                    ?loadValue > ?maximumValue ||
                    ?maximumValue < "0"^^xsd:decimal
                )
            }
        """
    ] .
<END_SHACL>
