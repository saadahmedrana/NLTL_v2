<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_079_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerImmersionFunction ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerShaftCentrelineDepth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The propeller immersion function and ice load cycle coefficient k2 must satisfy the specified formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?iceQuantity ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Quantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:propellerImmersionFunction ?fQuantity ;
                      nltl:propellerShaftCentrelineDepth ?depthQuantity .

                ?iceQuantity qudt:numericValue ?ice .
                ?k2Quantity qudt:numericValue ?reportedK2 .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?fQuantity qudt:numericValue ?reportedF .
                ?depthQuantity qudt:numericValue ?depth .

                FILTER (?diameter != 0)

                BIND (((?depth - ?ice) / (?diameter / 2)) - 1 AS ?expectedF)

                BIND (
                    IF(
                        ?expectedF < 0,
                        0 - ?expectedF,
                        ?expectedF
                    )
                    AS ?fMagnitude
                )

                BIND (
                    IF(
                        ?expectedF < 0,
                        0.8 - ?expectedF,
                        IF(
                            ?expectedF <= 1,
                            0.8 - (0.4 * ?expectedF),
                            IF(
                                ?expectedF <= 2.5,
                                0.6 - (0.2 * ?expectedF),
                                0.1
                            )
                        )
                    )
                    AS ?expectedK2
                )

                BIND (
                    IF(
                        ?fMagnitude > 1,
                        0.000001 * ?fMagnitude,
                        0.000001
                    )
                    AS ?fTolerance
                )

                BIND (
                    IF(
                        ?expectedK2 > 1,
                        0.000001 * ?expectedK2,
                        0.000001
                    )
                    AS ?k2Tolerance
                )

                FILTER (
                    (?reportedF < (?expectedF - ?fTolerance)) ||
                    (?reportedF > (?expectedF + ?fTolerance)) ||
                    (?reportedK2 < (?expectedK2 - ?k2Tolerance)) ||
                    (?reportedK2 > (?expectedK2 + ?k2Tolerance))
                )
            }
        """
    ] .
<END_SHACL>
