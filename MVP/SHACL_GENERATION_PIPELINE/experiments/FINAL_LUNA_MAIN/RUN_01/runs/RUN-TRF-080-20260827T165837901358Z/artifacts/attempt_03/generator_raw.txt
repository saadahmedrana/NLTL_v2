<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080PropellerIceInteractionShape
    a sh:NodeShape ;
    sh:targetClass nltl:propellerIceInteractionComponent ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasComponent ]
            nltl:iceLoadCycleCount
        ) ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The linked ship must provide exactly one integer ice load cycle count."
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasComponent ]
            nltl:propellerBladeCount
        ) ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The linked ship must provide exactly one integer propeller blade count."
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasComponent ]
            nltl:effectiveIceLoadCycleCount
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The linked ship must provide exactly one effective ice load cycle count quantity."
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasComponent ]
            nltl:effectiveIceLoadCycleCount
            qudt:numericValue
        ) ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The effective ice load cycle count numeric value must be exactly one decimal."
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasComponent ]
            nltl:effectiveIceLoadCycleCount
            qudt:unit
        ) ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The effective ice load cycle count must use the unitless QUDT unit."
    ] ;
    sh:sparql [
        sh:message "The effective ice load cycle count must equal iceLoadCycleCount multiplied by propellerBladeCount within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this ;
                      nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?quantity .
                ?quantity <http://qudt.org/schema/qudt/numericValue> ?reported .

                BIND((?nice * ?blades) AS ?expected)
                BIND(
                    IF(
                        ?expected = 0,
                        "0.000001"^^xsd:decimal,
                        IF(
                            ?expected > 0,
                            (?expected * "0.000001"^^xsd:decimal),
                            ((0 - ?expected) * "0.000001"^^xsd:decimal)
                        )
                    ) AS ?tolerance
                )

                FILTER(
                    ?reported < (?expected - ?tolerance) ||
                    ?reported > (?expected + ?tolerance)
                )
            }
        """
    ] .
<END_SHACL>
