<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one integer ice load cycle count."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one integer propeller blade count."
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF080EffectiveQuantityShape ;
        sh:message "The ship must have exactly one effective ice load cycle count quantity."
    ] ;
    sh:sparql [
        sh:message "For a ship linked to a propeller/ice interaction component, effectiveIceLoadCycleCount must equal iceLoadCycleCount multiplied by propellerBladeCount within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component ;
                      nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?result .

                ?component rdf:type nltl:propellerIceInteractionComponent .
                ?result qudt:numericValue ?actual .

                BIND ((?nice * ?blades) AS ?expected)
                BIND ((?expected * 0.000001 + 0.000000001) AS ?tolerance)

                FILTER (
                    (?actual - ?expected > ?tolerance) ||
                    (?expected - ?actual > ?tolerance)
                )
            }
        """
    ] .

gen:TRF080EffectiveQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The effective ice load cycle count must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The effective ice load cycle count must use the unitless QUDT unit."
    ] .

gen:TRF080ApplicableComponentShape
    a sh:NodeShape ;
    sh:targetClass nltl:propellerIceInteractionComponent ;
    sh:class nltl:machineryComponent .

</SHACL>
