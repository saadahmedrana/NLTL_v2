            PREFIX qudt: <http://qudt.org/schema/qudt/>

                      nltl:hydrodynamicBollardThrust ?bollard ;
                      nltl:forwardIceThrust ?reportedTf ;

                ?reportedBackwardDesign qudt:numericValue ?backwardDesign .

                BIND (IF(?expectedForward >= ?expectedBackward,
                         ?expectedForward,
                         ?expectedBackward) AS ?expectedDesign)

                BIND (IF((IF(?expectedTf >= 0, ?expectedTf, 0 - ?expectedTf) * 1e-6) >= 1e-9,
                         IF(?expectedTf >= 0, ?expectedTf, 0 - ?expectedTf) * 1e-6,
                         1e-9) AS ?tolTf)
                BIND (IF((IF(?expectedTb >= 0, ?expectedTb, 0 - ?expectedTb) * 1e-6) >= 1e-9,
                         IF(?expectedTb >= 0, ?expectedTb, 0 - ?expectedTb) * 1e-6,
                         1e-9) AS ?tolTb)
                BIND (IF((IF(?expectedDesign >= 0, ?expectedDesign, 0 - ?expectedDesign) * 1e-6) >= 1e-9,
                         IF(?expectedDesign >= 0, ?expectedDesign, 0 - ?expectedDesign) * 1e-6,
                         1e-9) AS ?tolDesign)

                FILTER (
                    ?tf < (?expectedTf - ?tolTf) ||
                    ?tf > (?expectedTf + ?tolTf) ||
                    ?tb < (?expectedTb - ?tolTb) ||
                    ?tb > (?expectedTb + ?tolTb) ||
                    ?forwardDesign < (?expectedDesign - ?tolDesign) ||
                    ?forwardDesign > (?expectedDesign + ?tolDesign) ||
                    ?backwardDesign < (?expectedDesign - ?tolDesign) ||
                    ?backwardDesign > (?expectedDesign + ?tolDesign) ||
                    ?design < (?expectedDesign - ?tolDesign) ||
                    ?design > (?expectedDesign + ?tolDesign)
                )
            }
        """
