        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single pitch-value-known indicator is required to select the applicable pitch rule."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:pattern "^(?!CP$).+$"
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:pattern "^CP$"
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;

            sh:property [
                sh:path nltl:propellerType ;

            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;

                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;

                    ] ;

                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "For a CP propeller with unknown pitch value, propellerPitchAtMcrBollard must equal 0.7 times propellerPitchAtMcrFreeRunning within a relative tolerance of 1e-6, with a 1e-9 absolute fallback." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>

                    WHERE {
                        $this nltl:propellerPitchAtMcrBollard ?bollardNode ;
                              nltl:propellerPitchAtMcrFreeRunning ?freeNode .
                        ?bollardNode qudt:numericValue ?bollardValue ;
                                      qudt:unit unit:M .
                        ?freeNode qudt:numericValue ?freeValue ;
                                   qudt:unit unit:M .
                        BIND(0.7 * ?freeValue AS ?expected)
                        BIND(IF(ABS(?expected) * 0.000001 > 0.000000001,
                                ABS(?expected) * 0.000001,
                                0.000000001) AS ?tolerance)
                        FILTER(ABS(?bollardValue - ?expected) > ?tolerance)
                    }
                """
            ]
        ]
