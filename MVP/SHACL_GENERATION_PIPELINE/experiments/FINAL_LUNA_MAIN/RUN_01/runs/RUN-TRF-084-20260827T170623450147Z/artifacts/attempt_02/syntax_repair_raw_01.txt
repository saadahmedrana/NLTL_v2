<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "CP" ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "CP" ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "For a CP propeller with unknown pitch value, propeller pitch at MCR in bollard condition must equal 0.7 times the free-running value within relative tolerance 1e-6 and absolute fallback 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:propellerType ?type .
                        FILTER (STR(?type) = "CP") .
                        $this nltl:pitchValueKnown false .
                        $this nltl:propellerPitchAtMcrBollard ?bollard .
                        ?bollard qudt:numericValue ?bollardValue ;
                                 qudt:unit unit:M .
                        $this nltl:propellerPitchAtMcrFreeRunning ?freeRunning .
                        ?freeRunning qudt:numericValue ?freeRunningValue ;
                                     qudt:unit unit:M .
                        FILTER (
                            isNumeric(?bollardValue) &&
                            isNumeric(?freeRunningValue) &&
                            ABS(?bollardValue - (0.7 * ?freeRunningValue)) >
                            (0.000001 * ABS(0.7 * ?freeRunningValue) + 0.000000001)
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "The CP-specific pitch constraint is not applicable to a non-CP propeller." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:propellerType ?type .
                            FILTER (STR(?type) != "CP")
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
