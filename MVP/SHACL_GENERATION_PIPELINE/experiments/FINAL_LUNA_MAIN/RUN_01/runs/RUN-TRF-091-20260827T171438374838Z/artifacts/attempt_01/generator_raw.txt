<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-091-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:message "The ship shall identify an ice milling sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-091-PropulsionLoadCaseShape> ;
        sh:message "The ship shall identify a propulsion load case for the propulsion-line strength evaluation."
    ] .

<urn:nltl:generated-shape:TRF-091-PropulsionLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCasePurposeValue ;
        sh:minCount 1 ;
        sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
        sh:not [
            sh:hasValue nltl:stallingAnalysisPurposeValue
        ] ;
        sh:message "The propulsion load case shall be designated for propulsion-line strength evaluation and not for propulsion system stalling analysis."
    ] .
<END_SHACL>
