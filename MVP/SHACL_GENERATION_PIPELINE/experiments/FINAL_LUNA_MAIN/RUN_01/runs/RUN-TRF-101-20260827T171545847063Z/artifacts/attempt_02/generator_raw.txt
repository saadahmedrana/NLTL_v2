<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_101_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:LengthMetresQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node gen:LengthMetresQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:node gen:BladeFailureLoadCaseShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:ApprovedAnalysisBranch
        gen:CalculatedLoadBranch
    ) ;
    sh:sparql [
        sh:message "bladeFailureStatus must be true when blade tip offset exceeds 10% of propeller diameter, and false otherwise" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset ?offsetValue ;
                      nltl:propellerDiameter ?diameterValue ;
                      nltl:bladeFailureStatus ?status .
                ?offsetValue qudt:numericValue ?offset .
                ?diameterValue qudt:numericValue ?diameter .
                FILTER (
                    (?offset > (0.10 * ?diameter) && ?status != true) ||
                    (?offset <= (0.10 * ?diameter) && ?status != false)
                )
            }
        """
    ] .

gen:CalculatedLoadBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:node gen:LengthMetresQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:node gen:LengthMetresQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:node gen:LengthMetresQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:node gen:PressureMegaPascalQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:node gen:PressureMegaPascalQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:node gen:PressureMegaPascalQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:node gen:ForceKiloNewtonsQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "bladeReferenceStrength1 must equal 0.6 times bladeMinimumYieldStrength plus 0.4 times bladeUltimateTensileStrength" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeMinimumYieldStrength ?yieldValue ;
                      nltl:bladeUltimateTensileStrength ?ultimateValue ;
                      nltl:bladeReferenceStrength1 ?referenceValue .
                ?yieldValue qudt:numericValue ?yield .
                ?ultimateValue qudt:numericValue ?ultimate .
                ?referenceValue qudt:numericValue ?reference .
                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expected)
                BIND ((0.000001 * ABS(?expected)) + 0.000001 AS ?tolerance)
                FILTER (ABS(?reference - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "bladeFailureUltimateLoad must equal the declared blade failure load formula within a relative tolerance of 1e-6 with a small absolute fallback" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?chordValue ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?thicknessValue ;
                      nltl:bladeCylindricalRootRadius ?radiusValue ;
                      nltl:bladeReferenceStrength1 ?referenceValue ;
                      nltl:propellerDiameter ?diameterValue ;
                      nltl:bladeFailureUltimateLoad ?loadValue .
                ?chordValue qudt:numericValue ?chord .
                ?thicknessValue qudt:numericValue ?thickness .
                ?radiusValue qudt:numericValue ?radius .
                ?referenceValue qudt:numericValue ?reference .
                ?diameterValue qudt:numericValue ?diameter .
                ?loadValue qudt:numericValue ?load .
                BIND ((300 * ?chord * ?thickness * ?thickness * ?reference) /
                      ((0.8 * ?diameter) - (2 * ?radius)) AS ?expected)
                BIND ((0.000001 * ABS(?expected)) + 0.000001 AS ?tolerance)
                FILTER (ABS(?load - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
