nltlg:CalculatedLoadAlternativeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:PressureQuantityShape
    ] ;
    sh:sparql [
        sh:message "The calculated blade failure ultimate load and reference strength must satisfy the declared formulas within a relative tolerance of 1e-6, with an absolute fallback near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeMinimumYieldStrength ?yieldNode ;
                      nltl:bladeReferenceStrength1 ?refNode ;
                      nltl:bladeUltimateTensileStrength ?ultimateNode ;
                      nltl:propellerDiameter ?dNode ;
                      nltl:bladeFailureUltimateLoad ?fNode ;
                      nltl:hasLoadCase ?case .

                ?cNode qudt:numericValue ?c .
                ?rNode qudt:numericValue ?r .
                ?tNode qudt:numericValue ?t .
                ?yieldNode qudt:numericValue ?yield .
                ?refNode qudt:numericValue ?ref .
                ?ultimateNode qudt:numericValue ?ultimate .
                ?dNode qudt:numericValue ?d .
                ?fNode qudt:numericValue ?f .
                ?case nltl:bladeFailureLoadApplicationRadiusRatio ?ratioNode .
                ?ratioNode qudt:numericValue ?ratio .

                BIND((0.6 * ?yield + 0.4 * ?ultimate) AS ?expectedRef)
                BIND((300 * ?c * ?t * ?t * ?expectedRef / (0.8 * ?d - 2 * ?r)) AS ?expectedF)
                BIND((?expectedRef * 0.000001) AS ?refRelativeTolerance)
                BIND((?expectedF * 0.000001) AS ?fRelativeTolerance)
                BIND(IF(?refRelativeTolerance > 0.000001, ?refRelativeTolerance, 0.000001) AS ?refTolerance)
                BIND(IF(?fRelativeTolerance > 0.000001, ?fRelativeTolerance, 0.000001) AS ?fTolerance)
                BIND((?expectedRef + ?refTolerance) AS ?refUpper)
                BIND((?expectedRef - ?refTolerance) AS ?refLower)
                BIND((?expectedF + ?fTolerance) AS ?fUpper)
                BIND((?expectedF - ?fTolerance) AS ?fLower)

                FILTER (
                    (?ref > ?refUpper) ||
                    (?ref < ?refLower) ||
                    (?f > ?fUpper) ||
                    (?f < ?fLower) ||
                    (?ratio != 0.8)
                )
            }
        """
    ] .

nltlg:BladeFailureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node nltlg:BladeFailureLoadCaseShape
    ] ;
    sh:or (
        nltlg:ApprovedAnalysisAlternativeShape
        nltlg:CalculatedLoadAlternativeShape
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset ?offsetNode ;
                      nltl:propellerDiameter ?diameterNode ;
                      nltl:bladeFailureStatus ?status .

                ?offsetNode qudt:numericValue ?offset .
                ?diameterNode qudt:numericValue ?diameter .

                FILTER (
                    (?offset > (0.10 * ?diameter)) &&
                    (?status != true)
                )
            }
        """
    ] .
