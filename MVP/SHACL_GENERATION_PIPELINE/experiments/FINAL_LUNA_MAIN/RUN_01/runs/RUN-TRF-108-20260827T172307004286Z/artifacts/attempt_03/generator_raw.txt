<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_108_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single stress-life curve type applicability value is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:message "For a known stress-life curve, the selected curve must be recorded."
            ] ;
            sh:property [
                sh:path nltl:propellerBladeMaterialProperty ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "For a known stress-life curve, the blade material properties must be recorded."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Correspondence evidence for the selected stress-life curve and blade material properties is required."
            ] ;
            sh:sparql [
                sh:message "The known stress-life curve selection, blade material properties, and correspondence evidence must be represented together on the ship." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:stressLifeCurveSelection ?curve ;
                                  nltl:propellerBladeMaterialProperty ?material ;
                                  nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                            ?curve a nltl:stressLifeCurveTypeValue .
                            ?evidence a nltl:evidenceArtifact .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The known stress-life curve branch cannot be used when the stress-life curve type is unknown." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this nltl:stressLifeCurveType ?curveType .
                        FILTER (LCASE(STR(?curveType)) = "unknown")
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:hasValue nltl:twoSlopeStressLifeCurve ;
                sh:minCount 1 ;
                sh:message "When the stress-life curve is unknown, the two-slope stress-life curve must be selected."
            ] ;
            sh:sparql [
                sh:message "The unknown stress-life curve branch requires an applicability value of unknown." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:stressLifeCurveType ?curveType .
                            FILTER (LCASE(STR(?curveType)) = "unknown")
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
