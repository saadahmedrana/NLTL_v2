<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-108-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A stress-life curve type is required to select the applicable rule branch."
    ] ;
    sh:or (
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:propellerBladeMaterialProperty ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "The blade material properties must be provided for a known stress-life curve."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A stress-life curve must be selected."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Evidence must document correspondence between the selected stress-life curve and the blade material properties."
            ] ;
            sh:sparql [
                sh:message "The selected stress-life curve, blade material properties, and correspondence evidence must be explicitly tied together." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:stressLifeCurveType ?curveType .
                        FILTER(LCASE(STR(?curveType)) != "unknown")
                        $this nltl:stressLifeCurveSelection ?curve .
                        $this nltl:propellerBladeMaterialProperty ?material .
                        $this nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                        ?evidence rdf:subject ?curve ;
                                  rdf:predicate nltl:propellerBladeMaterialProperty ;
                                  rdf:object ?material .
                    }
                """
            ]
        ]
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:twoSlopeStressLifeCurve ;
                sh:message "When the stress-life curve type is unknown, the two-slope stress-life curve must be selected."
            ] ;
            sh:sparql [
                sh:message "The two-slope alternative applies only when the stress-life curve type is unknown." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:stressLifeCurveType ?curveType .
                        FILTER(LCASE(STR(?curveType)) = "unknown")
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
