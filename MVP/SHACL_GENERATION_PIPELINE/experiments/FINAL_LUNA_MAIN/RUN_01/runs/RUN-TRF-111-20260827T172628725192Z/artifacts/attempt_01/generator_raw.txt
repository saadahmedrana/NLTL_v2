<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-111-ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one extreme ice force quantity in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:in (
                    nltl:extremeOperationalLoadCase
                    nltl:fatigueLoadCase
                    nltl:bladeFailureLoadCase
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:class qudt:QuantityValue ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:fatigueSafetyFactor ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The ship must contain extreme operational, fatigue, and blade failure load cases, with yield safety factors of at least 1.3 and 1.0 for the applicable yield cases and fatigue safety factors of at least 1.5 for every linked shaft-line component." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?extremeCase .
                    ?extremeCase nltl:shaftLineLoadCaseType nltl:extremeOperationalLoadCase .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?fatigueCase .
                    ?fatigueCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?bladeCase .
                    ?bladeCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?badExtremeCase .
                    ?badExtremeCase nltl:shaftLineLoadCaseType nltl:extremeOperationalLoadCase .
                    FILTER NOT EXISTS {
                        ?badExtremeCase nltl:componentYieldSafetyFactor ?extremeFactor .
                        ?extremeFactor qudt:numericValue ?extremeValue .
                        FILTER (?extremeValue >= 1.3)
                    }
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?badBladeCase .
                    ?badBladeCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                    FILTER NOT EXISTS {
                        ?badBladeCase nltl:componentYieldSafetyFactor ?bladeFactor .
                        ?bladeFactor qudt:numericValue ?bladeValue .
                        FILTER (?bladeValue >= 1.0)
                    }
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerShaftLineComponent ?component .
                    FILTER NOT EXISTS {
                        ?component nltl:fatigueSafetyFactor ?fatigueFactor .
                        ?fatigueFactor qudt:numericValue ?fatigueValue .
                        FILTER (?fatigueValue >= 1.5)
                    }
                }
            }
        """
    ] .
<END_SHACL>
