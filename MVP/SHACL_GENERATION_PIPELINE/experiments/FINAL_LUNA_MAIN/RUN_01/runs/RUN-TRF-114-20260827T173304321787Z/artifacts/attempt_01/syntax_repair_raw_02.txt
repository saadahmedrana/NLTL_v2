@prefix qudt: <http://qudt.org/schema/qudt/> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-114-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide a typical ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node [
            sh:class nltl:thrusterIceImpactLoadCase ;
            sh:property [
                sh:path nltl:designIceBlockTableReference ;
                sh:class nltl:tableReferenceValue ;
                sh:hasValue nltl:traficomTable6Dash3Reference ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must reference TRAFICOM Table 6-3 for the design ice block."
            ] ;
            sh:property [
                sh:path nltl:impactLoadCaseTableReference ;
                sh:class nltl:tableReferenceValue ;
                sh:hasValue nltl:traficomTable6Dash16Reference ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must reference TRAFICOM Table 6-16 for its impact load case."
            ] ;
            sh:property [
                sh:path nltl:thrusterImpactCaseAppliesToBody ;
                sh:class nltl:thrusterBody ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:class nltl:contactGeometryValue ;
                sh:in (
                    nltl:hemisphericalContactGeometry
                    nltl:nonHemisphericalContactGeometry
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must classify its contact geometry as hemispherical or non-hemispherical."
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:contactGeometryClassification ;
                        sh:hasValue nltl:hemisphericalContactGeometry ;
                        sh:minCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:contactGeometryClassification ;
                        sh:hasValue nltl:nonHemisphericalContactGeometry ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:contactGeometryCorrespondenceEvidence ;
                        sh:class nltl:evidenceArtifact ;
                        sh:minCount 1 ;
                        sh:message "Non-hemispherical contact geometry requires correspondence evidence."
                    ]
                ]
            ) ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "The impact case must provide evidence for the loaded impact area."
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactDemand ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterImpactCaseAppliesToBody ;
                sh:node [
                    sh:class nltl:thrusterBody ;
                    sh:property [
                        sh:path nltl:thrusterResistanceCapacity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:message "The thruster body must provide its externally assessed resistance capacity."
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each non-hemispherical impact case must have an equivalent impact sphere radius on the ship." ;
        sh:select """PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasThrusterIceImpactLoadCase ?case .
    ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
    FILTER NOT EXISTS {
        $this nltl:equivalentImpactSphereRadius ?radius .
        ?radius qudt:numericValue ?value ;
                qudt:unit <http://qudt.org/vocab/unit/M> .
    }
}"""
    ] ;
    sh:sparql [
        sh:message "The thruster resistance capacity must be at least the externally assessed impact demand for every impact case." ;
        sh:select """PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasThrusterIceImpactLoadCase ?case .
    ?case nltl:thrusterIceImpactDemand ?demand ;
           nltl:thrusterImpactCaseAppliesToBody ?body .
    ?demand qudt:numericValue ?demandValue ;
            qudt:unit <http://qudt.org/vocab/unit/KiloN> .
    ?body nltl:thrusterResistanceCapacity ?capacity .
    ?capacity qudt:numericValue ?capacityValue ;
             qudt:unit <http://qudt.org/vocab/unit/KiloN> .
    FILTER ( ?capacityValue < ?demandValue )
}"""
    ] .
