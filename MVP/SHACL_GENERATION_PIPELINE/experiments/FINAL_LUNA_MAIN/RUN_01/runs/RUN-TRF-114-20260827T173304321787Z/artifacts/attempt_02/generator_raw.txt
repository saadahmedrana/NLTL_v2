@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF114ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall declare a single ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:thrusterIceImpactLoadCase ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoadCaseIdentifier ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designIceBlockTableReference ;
                sh:class nltl:tableReferenceValue ;
                sh:hasValue nltl:traficomTable6Dash3Reference ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:impactLoadCaseTableReference ;
                sh:class nltl:tableReferenceValue ;
                sh:hasValue nltl:traficomTable6Dash16Reference ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterImpactCaseAppliesToBody ;
                sh:class nltl:thrusterBody ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:thrusterResistanceCapacity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactDemand ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Each impact case shall include evidence for the loaded impact area."
            ] ;
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:or (
                    [
                        sh:hasValue nltl:hemisphericalContactGeometry
                    ]
                    [
                        sh:hasValue nltl:nonHemisphericalContactGeometry ;
                        sh:property [
                            sh:path nltl:contactGeometryCorrespondenceEvidence ;
                            sh:class nltl:evidenceArtifact ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ) ;
                sh:message "Each impact case shall classify its contact geometry; non-hemispherical geometry shall include correspondence evidence."
            ] ;
            sh:sparql [
                sh:message "Thruster resistance capacity shall be at least the externally assessed impact demand." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:thrusterIceImpactDemand ?demand ;
                              nltl:thrusterImpactCaseAppliesToBody ?body .
                        ?demand qudt:numericValue ?demandValue .
                        ?body nltl:thrusterResistanceCapacity ?capacity .
                        ?capacity qudt:numericValue ?capacityValue .
                        FILTER (datatype(?demandValue) = xsd:decimal)
                        FILTER (datatype(?capacityValue) = xsd:decimal)
                        FILTER (?capacityValue < ?demandValue)
                    }
                """
            ]
        ] ;
        sh:message "The ship shall provide at least one complete thruster ice impact load case."
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical impact case shall have a valid equivalent sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?radiusValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?radiusValue) = xsd:decimal)
                }
            }
        """
    ] .
