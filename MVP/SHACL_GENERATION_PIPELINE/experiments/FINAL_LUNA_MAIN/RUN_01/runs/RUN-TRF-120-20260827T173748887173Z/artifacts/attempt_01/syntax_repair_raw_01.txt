<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record the damping taken into account for thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass as a decimal quantity in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the thruster global natural frequency as a decimal quantity in hertz."
    ] ;
    sh:sparql [
        sh:message "Each linked thruster body used for global natural-frequency estimation must include ship attachment stiffness and a calculation case with a longitudinal or transverse vibration direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasComponent ?thrusterBody .
                ?thrusterBody a nltl:thrusterBody .
                FILTER (
                    NOT EXISTS {
                        ?thrusterBody nltl:shipAttachmentStiffness ?stiffness .
                        ?stiffness a qudt:QuantityValue ;
                                   qudt:numericValue ?stiffnessValue ;
                                   qudt:unit unit:KiloN-PER-M .
                        FILTER (datatype(?stiffnessValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?thrusterBody nltl:shipAttachmentStiffness ?stiffnessNode .
                        ?stiffnessNode qudt:unit ?stiffnessUnit .
                        FILTER (?stiffnessUnit != unit:KiloN-PER-M)
                    }
                    ||
                    NOT EXISTS {
                        ?thrusterBody nltl:hasNaturalFrequencyCalculationCase ?calculationCase .
                        ?calculationCase a nltl:calculationCase .
                    }
                    ||
                    EXISTS {
                        ?thrusterBody nltl:hasNaturalFrequencyCalculationCase ?calculationCase .
                        FILTER ( NOT EXISTS {
                            ?calculationCase a nltl:calculationCase ;
                                              nltl:vibrationDirection ?direction .
                            FILTER (
                                ?direction IN (
                                    nltl:longitudinalDirection,
                                    nltl:transverseDirection
                                )
                            )
                        } )
                    }
                )
            }
        """
    ] .
<END_SHACL>
