<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF120Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record damping used for thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass as a single kilogram quantity."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the estimated thruster natural frequency as a single hertz quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each existing thruster natural-frequency estimation case must cover both longitudinal and transverse directions." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasComponent ?body .
  ?body nltl:hasNaturalFrequencyCalculationCase ?case .
  ?case nltl:vibrationDirection ?direction .
  FILTER (?direction IN (nltl:longitudinalDirection, nltl:transverseDirection))
  FILTER NOT EXISTS {
    ?body nltl:hasNaturalFrequencyCalculationCase ?otherCase .
    ?otherCase nltl:vibrationDirection ?otherDirection .
    FILTER (?otherDirection != ?direction) .
    FILTER (?otherDirection IN (nltl:longitudinalDirection, nltl:transverseDirection))
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each existing thruster natural-frequency estimation case must be associated with a thruster body having ship attachment stiffness in kN/m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasComponent ?body .
  ?body nltl:hasNaturalFrequencyCalculationCase ?case .
  FILTER NOT EXISTS {
    ?body nltl:shipAttachmentStiffness ?stiffness .
    ?stiffness a qudt:QuantityValue ;
               qudt:numericValue ?value ;
               qudt:unit unit:KiloN-PER-M .
    FILTER (datatype(?value) = xsd:decimal)
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A thruster body used in natural-frequency estimation must not have multiple ship attachment stiffness values." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasComponent ?body .
  ?body nltl:hasNaturalFrequencyCalculationCase ?case .
  ?body nltl:shipAttachmentStiffness ?first .
  ?body nltl:shipAttachmentStiffness ?second .
  FILTER (?first != ?second)
}
"""
    ] .
</END_SHACL>
