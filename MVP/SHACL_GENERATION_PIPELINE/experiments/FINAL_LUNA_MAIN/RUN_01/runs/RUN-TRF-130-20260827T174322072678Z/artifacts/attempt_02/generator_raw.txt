<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:VolumeQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AreaQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:RatioQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive "4"^^xsd:decimal
    ] .

gen:InletChestShape a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each inlet chest must provide its recommended chest volume in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:node gen:RatioQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each inlet chest must have a strainer open area ratio of at least four."
    ] .

gen:StandardInletChestShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:hasValue "near_centreline_and_well_aft_if_possible" ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A standard inlet chest must be situated near the centreline and well aft if possible."
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AlternativeInletChestShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each alternative inlet chest must support alternating cooling-water intake and discharge."
    ] ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:hasValue "near_centreline_and_well_aft_if_possible" ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF130Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the cooling-water chest volume and height requirements cannot be met."
    ] ;
    sh:property [
        sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:node gen:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Total relevant engine power is required for the inlet-chest volume calculation."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:inletChest ;
        sh:node gen:InletChestShape ;
        sh:minCount 1 ;
        sh:message "At least one cooling-water inlet chest is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape [ sh:node gen:StandardInletChestShape ] ;
                sh:qualifiedMinCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape [ sh:node gen:AlternativeInletChestShape ] ;
                sh:qualifiedMinCount 2
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each inlet chest must have a recommended volume approximately equal to total relevant engine power divided by 750, and its strainer open area must be at least four times its inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:totalRelevantEnginePower ?tpNode ;
                      nltl:hasComponent ?ch .
                ?tpNode qudt:numericValue ?tp .
                ?ch nltl:recommendedChestVolume ?rvNode ;
                    nltl:strainerOpenArea ?saNode ;
                    nltl:inletPipeArea ?ipaNode .
                ?rvNode qudt:numericValue ?rv .
                ?saNode qudt:numericValue ?sa .
                ?ipaNode qudt:numericValue ?ipa .
                BIND((?tp / 750) AS ?expected)
                FILTER (
                    ?sa < (4 * ?ipa)
                    ||
                    (
                        ?expected >= 1 &&
                        (
                            ?rv < (?expected - (?expected * 0.000001))
                            ||
                            ?rv > (?expected + (?expected * 0.000001))
                        )
                    )
                    ||
                    (
                        ?expected < 1 &&
                        (
                            ?rv < (?expected - 0.000001)
                            ||
                            ?rv > (?expected + 0.000001)
                        )
                    )
                )
            }
        """
    ] .
</END_SHACL>
