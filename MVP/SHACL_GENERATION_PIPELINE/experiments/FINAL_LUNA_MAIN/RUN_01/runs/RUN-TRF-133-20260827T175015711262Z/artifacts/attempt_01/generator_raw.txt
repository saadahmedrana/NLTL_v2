gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice-class value."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF133WarningTriangleShape ;
        sh:message "The ship must have one conforming warning triangle marking."
    ] ;
    sh:or (
        gen:TRF133TimberApplicableShape
        gen:TRF133TimberNotApplicableShape
    ) .
