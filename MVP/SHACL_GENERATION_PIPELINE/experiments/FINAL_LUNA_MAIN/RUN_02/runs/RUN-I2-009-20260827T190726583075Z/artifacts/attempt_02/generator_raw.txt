<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:class nltl:hullAreaAssignment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:assignedHullArea ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:otherIceStrengthenedArea
                    nltl:bowIntermediateBottomHullArea
                    nltl:midbodyBottomHullArea
                    nltl:sternBottomHullArea
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The hull area assignment must identify a permitted hullAreaValue."
            ]
        ] ;
        sh:message "A ship must have one hull area assignment."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Average ice pressure must be a single MPa quantity."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 3.6 ;
                sh:maxInclusive 3.6 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The load patch aspect ratio must be exactly 3.6 and unitless."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The non-bow load patch height must be a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The non-bow load patch length must be a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The buttock angle must be a single degree quantity."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The normal frame angle must be a single degree quantity."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability must be recorded as one boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow-form applicability must be true exactly when buttock angle is positive and below 80 degrees and normal frame angle is above 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability ?app ;
                      nltl:buttockAngle/qudt:numericValue ?gamma ;
                      nltl:normalFrameAngle/qudt:numericValue ?beta .
                FILTER (
                    (?app = true &&
                     NOT EXISTS {
                         $this nltl:buttockAngle/qudt:numericValue ?g1 ;
                              nltl:normalFrameAngle/qudt:numericValue ?b1 .
                         FILTER (?g1 > 0 && ?g1 < 80 && ?b1 > 10)
                     })
                    ||
                    (?app = false &&
                     EXISTS {
                         $this nltl:buttockAngle/qudt:numericValue ?g2 ;
                              nltl:normalFrameAngle/qudt:numericValue ?b2 .
                         FILTER (?g2 > 0 && ?g2 < 80 && ?b2 > 10)
                     })
                )
            }
        """
    ] .
