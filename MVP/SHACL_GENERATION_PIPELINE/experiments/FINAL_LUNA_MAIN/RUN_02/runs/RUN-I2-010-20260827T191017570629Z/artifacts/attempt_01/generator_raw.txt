<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-010> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must record its polar class." ;
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must record whether it has a vertical-sided bow form." ;
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must record its bow vertical-side status." ;
    ] ;
    sh:sparql [
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:verticalSidedBowForm ?verticalForm ;
        nltl:bowVerticalSideStatus ?status .
  FILTER (?verticalForm != ?status)
}
""" ;
        sh:message "The vertical-sided bow indicators must agree." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
            ] ;
            sh:property [
                sh:path nltl:verticalSidedBowForm ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:bowVerticalSideStatus ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethod ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must record the design ice-force calculation method." ;
            ] ;
            sh:property [
                sh:path nltl:normalFrameAngle ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 0 ;
                        sh:maxInclusive 10 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                ] ;
                sh:message "For an applicable ship, every represented normal frame angle must be between 0 and 10 degrees." ;
            ] ;
        ]
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  FILTER (
    NOT EXISTS {
      $this nltl:polarClass ?pc .
      FILTER (?pc IN (
        nltl:polarClassPc6,
        nltl:polarClassPc7
      ))
    }
    ||
    (
      EXISTS {
        $this nltl:verticalSidedBowForm false
      }
      &&
      EXISTS {
        $this nltl:bowVerticalSideStatus false
      }
    )
  )
}
""" ;
                sh:message "The ship is outside the PC6/PC7 vertical-sided-bow applicability condition." ;
            ] ;
        ]
    ) .
<END_SHACL>
