<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMilliMQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_021_ShellPlateThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthMilliMQuantityShape
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthMilliMQuantityShape
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthMQuantityShape
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For each plating item, thickness in metres must equal iceLoadRequiredNetPlateThickness plus corrosionAbrasionAllowance in millimetres converted to metres, within a 1e-6 relative tolerance and a 1e-6 metre absolute fallback near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:iceLoadRequiredNetPlateThickness ?netNode ;
                         nltl:corrosionAbrasionAllowance ?allowanceNode ;
                         nltl:thickness ?thicknessNode .
                ?netNode qudt:numericValue ?net .
                ?allowanceNode qudt:numericValue ?allowance .
                ?thicknessNode qudt:numericValue ?reported .

                BIND ((?net + ?allowance) / 1000 AS ?expected)
                BIND (?reported - ?expected AS ?difference)
                BIND (IF(?expected < 0, 0 - ?expected, ?expected) AS ?expectedAbsolute)
                BIND (
                    IF(
                        ?expectedAbsolute < 0.000001,
                        0.000001,
                        ?expectedAbsolute * 0.000001
                    ) AS ?tolerance
                )
                BIND (
                    IF(?difference < 0, 0 - ?difference, ?difference)
                    AS ?absoluteDifference
                )

                FILTER (?absoluteDifference > ?tolerance)
            }
        """
    ] .
<END_SHACL>
