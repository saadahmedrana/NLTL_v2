gen:I2_026_ShipShape
    a sh:NodeShape ;

    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026_StructuralMemberShape ;
        sh:message "Each represented structural member must satisfy the framing-member support-fixity rule."
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must state whether it is continuous through its support."
    ] ;
    sh:or (
        gen:I2_026_ContinuousFixedShape

        gen:I2_026_RestraintFixedShape
