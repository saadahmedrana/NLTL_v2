gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:CalculationCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each calculation case must provide interpolation endpoints when the linked framing angle is oblique (20 < Omega < 70 degrees)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleQuantity .
                ?angleQuantity qudt:numericValue ?omega ;
                    qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (
                    xsd:decimal(?omega) > 20 &&
                    xsd:decimal(?omega) < 70
                )
                $this nltl:hasCalculationCase ?calculationCase .
                FILTER NOT EXISTS {
                    ?calculationCase nltl:interpolationLowerEndpoint ?lower .
                    ?calculationCase nltl:interpolationUpperEndpoint ?upper .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the local-frame area exceeds the attached-plate-flange area, each calculation case must provide the externally computed plastic neutral axis height." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaQuantity ;
                    nltl:netAttachedShellPlateThickness ?thicknessQuantity ;
                    nltl:frameSpacing ?spacingQuantity .
                ?areaQuantity qudt:numericValue ?area ;
                    qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                    qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?spacingQuantity qudt:numericValue ?spacing ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    xsd:decimal(?area) >
                    1000 * xsd:decimal(?thickness) * xsd:decimal(?spacing)
                )
                $this nltl:hasCalculationCase ?calculationCase .
                FILTER NOT EXISTS {
                    ?calculationCase nltl:plasticNeutralAxisHeight ?neutralAxisQuantity .
                }
            }
        """
    ] .
