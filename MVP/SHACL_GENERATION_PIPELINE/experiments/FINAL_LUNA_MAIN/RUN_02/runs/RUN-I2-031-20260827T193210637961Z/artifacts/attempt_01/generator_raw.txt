<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-031-QuantityDecimalShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity must contain one decimal numeric value."
    ] .

<urn:nltl:generated-shape:I2-031-QuantityKiloNShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A force quantity must contain one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The force quantity must use kN."
    ] .

<urn:nltl:generated-shape:I2-031-LoadCaseShape>
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityDecimalShape> ;
        sh:message "Each load case must provide one combined shear and bending demand."
    ] .

<urn:nltl:generated-shape:I2-031-BottomLoadCaseShape>
    a sh:NodeShape ;
    sh:node <urn:nltl:generated-shape:I2-031-LoadCaseShape> ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:message "A bottom-structure load case must apply the patch load with dimension b parallel to the frame direction."
    ] .

<urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must identify its structural location."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-031-LoadCaseShape> ;
        sh:message "Each applicable structural member must have a load case."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityKiloNShape> ;
        sh:message "Each applicable structural member must provide its plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityKiloNShape> ;
        sh:message "Each applicable structural member must provide the midspan load causing plastic collapse in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must identify its plastic collapse mechanism."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node <urn:nltl:generated-shape:I2-031-BottomLoadCaseShape> ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                $this nltl:plasticStrength ?strength .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?strength qudt:numericValue ?strengthValue .
                ?demand qudt:numericValue ?demandValue .
                FILTER (?demandValue > ?strengthValue)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must equal the midspan load that causes the plastic collapse mechanism." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strength .
                $this nltl:midspanPlasticCollapseLoad ?midspanLoad .
                ?strength qudt:numericValue ?strengthValue .
                ?midspanLoad qudt:numericValue ?midspanValue .
                FILTER (?strengthValue != ?midspanValue)
            }
        """
    ] .
