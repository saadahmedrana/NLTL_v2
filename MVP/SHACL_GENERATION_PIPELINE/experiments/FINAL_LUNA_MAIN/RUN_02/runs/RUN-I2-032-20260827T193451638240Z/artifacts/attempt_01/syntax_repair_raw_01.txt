gen:I2_032_StructuralMemberShape
    a sh:NodeShape ;

            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredTransverseFrameShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:CentiM2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The actual net effective shear area must be at least the calculated required transverse frame shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasStructuralMember $this ;
                      nltl:averageIcePressure ?pressureQ ;
                      nltl:frameSpan ?frameSpanQ ;
                      nltl:loadPatchHeight ?heightQ ;
                      nltl:peakPressureFactor ?peakFactorQ ;
                      nltl:shearArea ?actualAreaQ ;
                      nltl:yieldStrength ?yieldQ .

                $this nltl:frameSpacing ?spacingQ ;
                      nltl:selectedHullAreaFactor ?areaFactorQ ;
                      nltl:requiredTransverseFrameShearArea ?requiredAreaQ .

                ?hullAreaValue a nltl:hullAreaValue ;
                                nltl:hullAreaFactor ?hullFactorQ .

                ?pressureQ qudt:numericValue ?pressure .
                ?frameSpanQ qudt:numericValue ?frameSpan .
                ?heightQ qudt:numericValue ?height .
                ?peakFactorQ qudt:numericValue ?peakFactor .
                ?actualAreaQ qudt:numericValue ?actualArea .
                ?yieldQ qudt:numericValue ?yield .
                ?spacingQ qudt:numericValue ?spacing .
                ?areaFactorQ qudt:numericValue ?areaFactor .
                ?requiredAreaQ qudt:numericValue ?requiredArea .
                ?hullFactorQ qudt:numericValue ?hullFactor .

                BIND(IF(?frameSpan < ?height, ?frameSpan, ?height) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?spacing *
                      (?areaFactor * ?peakFactor * ?pressure) /
                      (0.577 * ?yield)) AS ?calculatedRequiredArea)

                FILTER (?requiredArea < ?calculatedRequiredArea)
                FILTER (?actualArea < ?requiredArea)
            }
        """
    ] .
