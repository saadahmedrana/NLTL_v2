<urn:nltl:generated-shape:I2-032-StructuralMemberShape> a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityLength> ;

<urn:nltl:generated-shape:I2-032-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityPressure> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one average ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityLength> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one frame span in metres."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityLength> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one load-patch height in metres."
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityDimensionless> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one peak pressure factor."
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityAreaM2> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one actual net effective shear area in square metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityLength> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one span in metres."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:I2-032-QuantityPressure> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-032-StructuralMemberShape>
    ] ;
    sh:sparql [
        sh:message "The required transverse frame shear area must equal 100^2*0.5*min(frameSpan,loadPatchHeight)*frameSpacing*selectedHullAreaFactor*peakPressureFactor*averageIcePressure/(0.577*yieldStrength), within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9; actual shear area must be at least that result after conversion from m2 to cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressure ;
                      nltl:frameSpan ?frameSpan ;
                      nltl:loadPatchHeight ?loadHeight ;
                      nltl:peakPressureFactor ?peakFactor ;
                      nltl:shearArea ?shearArea ;
                      nltl:span ?span ;
                      nltl:yieldStrength ?yieldStrength ;
                      nltl:hasStructuralMember ?member .

                ?pressure qudt:numericValue ?pressureValue .
                ?frameSpan qudt:numericValue ?frameSpanValue .
                ?loadHeight qudt:numericValue ?loadHeightValue .
                ?peakFactor qudt:numericValue ?peakFactorValue .
                ?shearArea qudt:numericValue ?shearAreaValue .
                ?span qudt:numericValue ?spanValue .
                ?yieldStrength qudt:numericValue ?yieldStrengthValue .

                ?member nltl:frameSpacing ?spacing ;
                        nltl:selectedHullAreaFactor ?areaFactor ;
                        nltl:requiredTransverseFrameShearArea ?requiredArea .

                ?spacing qudt:numericValue ?spacingValue .
                ?areaFactor qudt:numericValue ?areaFactorValue .
                ?requiredArea qudt:numericValue ?requiredAreaValue .

                BIND(IF(?frameSpanValue < ?loadHeightValue,
                        ?frameSpanValue,
                        ?loadHeightValue) AS ?lesserLength)
                BIND((10000 * 0.5 * ?lesserLength * ?spacingValue *
                      ?areaFactorValue * ?peakFactorValue * ?pressureValue) /
                     (0.577 * ?yieldStrengthValue) AS ?expectedArea)
                BIND((ABS(?expectedArea) * 0.000001) + 0.000000001 AS ?tolerance)

                FILTER(
                    ABS(?requiredAreaValue - ?expectedArea) > ?tolerance ||
                    (?shearAreaValue * 10000) < ?requiredAreaValue
                )
            }
        """
    ] .
