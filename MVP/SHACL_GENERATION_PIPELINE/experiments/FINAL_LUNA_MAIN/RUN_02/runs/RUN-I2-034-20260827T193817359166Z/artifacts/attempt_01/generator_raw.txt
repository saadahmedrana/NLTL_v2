<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-034-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must identify the applicable frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node nltl:StructuralMemberApplicabilityShape
    ] .

nltl:StructuralMemberApplicabilityShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:structuralLocation ?location .
                        FILTER (?location != nltl:sideStructureLocation)
                    }
                """
            ]
        ]
        nltl:ApplicableLongitudinalSideFrameShape
    ) .

nltl:ApplicableLongitudinalSideFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable side-structure frame must be identified as longitudinal."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:message "An applicable longitudinal frame must have sideStructureLocation."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node nltl:LoadCaseDemandShape ;
        sh:message "An applicable longitudinal side frame must have a represented load case with combined shear-and-bending demand."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:KiloNewtonQuantityShape ;
        sh:message "An applicable member must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:KiloNewtonQuantityShape ;
        sh:message "An applicable member must provide its midspan plastic-collapse load."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:plasticStrength ?plasticStrength ;
                      nltl:midspanPlasticCollapseLoad ?collapseLoad ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .

                ?plasticStrength qudt:numericValue ?plasticStrengthValue .
                ?collapseLoad qudt:numericValue ?collapseLoadValue .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .

                ?demand qudt:numericValue ?demandValue .

                FILTER (
                    ?demandValue > ?plasticStrengthValue ||
                    ?demandValue > ?collapseLoadValue
                )
            }
        """ ;
        sh:message "The combined shear-and-bending demand must not exceed the member plastic strength or the midspan plastic-collapse load."
    ] .

nltl:LoadCaseDemandShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:UnspecifiedQuantityShape ;
        sh:message "Each represented load case must provide combined shear-and-bending demand."
    ] .

nltl:UnspecifiedQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .
<END_SHACL>
