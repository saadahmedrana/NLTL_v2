<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade requirement case must identify one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The steel-grade case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-ThicknessQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each case must identify one supported steel material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each case must identify one supported steel strength category."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeValueShape> ;
        sh:message "Each case must identify exactly one actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeValueShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required grade must be selected from Table 8 for the Polar Class, thickness, material class and strength category; non-applicable combinations must not have a required grade, and the actual grade must be at least the required grade within its strength family." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:asBuiltPlateThickness ?tv ;
        nltl:steelMaterialClass ?mat ;
        nltl:steelStrengthCategory ?cat ;
        nltl:actualHullStructuralSteelGrade ?actual .
  ?tv qudt:numericValue ?t .
  ?ship nltl:hasSteelGradeRequirementCase $this ;
        nltl:polarClass ?pc .

  OPTIONAL { $this nltl:requiredHullStructuralSteelGrade ?required . }
  OPTIONAL { ?actual nltl:steelGradeOrderRank ?arank . }
  OPTIONAL { ?required nltl:steelGradeOrderRank ?rrank . }

  BIND(IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3), 1,
       IF(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5), 2,
       IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7), 3, 0))) AS ?pcgroup)

  BIND(IF(?t <= 10, 1,
       IF(?t <= 15, 2,
       IF(?t <= 20, 3,
       IF(?t <= 25, 4,
       IF(?t <= 30, 5,
       IF(?t <= 35, 6,
       IF(?t <= 40, 7,
       IF(?t <= 45, 8,
       IF(?t <= 50, 9, 0))))))))) AS ?band)

  BIND((?mat = nltl:steelMaterialClassThree &&
        ?cat = nltl:normalStrengthSteelCategory &&
        (?band = 7 || ?band = 8 || ?band = 9) &&
        ?pcgroup = 1) AS ?notApplicable)

  BIND((?band > 0 && !?notApplicable) AS ?applicable)

  BIND(
    IF(?band = 1,
       IF(?mat = nltl:steelMaterialClassThree,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)),
          IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB)),
    IF(?band <= 4,
       IF(?mat = nltl:steelMaterialClassOne,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD)),
       IF(?mat = nltl:steelMaterialClassTwo,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD)),
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)))),
    IF(?band <= 6,
       IF(?mat = nltl:steelMaterialClassOne,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeAh, nltl:steelGradeB),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD)),
       IF(?mat = nltl:steelMaterialClassTwo,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)),
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)))),
    IF(?band <= 8,
       IF(?mat = nltl:steelMaterialClassOne,
          IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD),
       IF(?mat = nltl:steelMaterialClassTwo,
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)),
          IF(?pcgroup = 1,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeFh, ?actual),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)))),
       IF(?mat IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo),
          IF(?pcgroup = 3,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeDh, nltl:steelGradeD),
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)),
          IF(?pcgroup = 1,
             IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeFh, ?actual),
             IF(?pcgroup = 2,
                IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeFh, ?actual),
                IF(?cat = nltl:highTensileSteelCategory, nltl:steelGradeEh, nltl:steelGradeE)))))
  ) AS ?expected)

  FILTER(
    ?band = 0 ||
    (!?applicable && BOUND(?required)) ||
    (?applicable &&
      (!BOUND(?required) ||
       ?required != ?expected ||
       !BOUND(?arank) ||
       !BOUND(?rrank) ||
       ?arank < ?rrank ||
       (?cat = nltl:normalStrengthSteelCategory &&
        ?actual NOT IN (nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE)) ||
       (?cat = nltl:highTensileSteelCategory &&
        ?actual NOT IN (nltl:steelGradeAh, nltl:steelGradeDh, nltl:steelGradeEh, nltl:steelGradeFh))))
  )
}
"""
    ] .

<urn:nltl:generated-shape:I2-048-ThicknessQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-048-GradeValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
