<urn:nltl:generated-shape:I2-048-CaseShape>

    ] ;

        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The Table 8 selection is missing, incorrect, or present for a non-applicable combination." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:asBuiltPlateThickness ?q ;
        nltl:steelMaterialClass ?m ;
        nltl:steelStrengthCategory ?s .
  ?q qudt:numericValue ?t ;
     qudt:unit <http://qudt.org/vocab/unit/MilliM> .
  ?ship nltl:polarClass ?pc ;
        nltl:hasSteelGradeRequirementCase $this .

  BIND((?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)) AS ?pc13)
  BIND((?pc IN (nltl:polarClassPc4, nltl:polarClassPc5)) AS ?pc45)
  BIND((?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5)) AS ?pc15)
  BIND((?pc IN (nltl:polarClassPc6, nltl:polarClassPc7)) AS ?pc67)
  BIND((?s = nltl:highTensileSteelCategory) AS ?high)

  BIND(
    (?m = nltl:steelMaterialClassThree &&
     (
       (?t > 35 && ?t <= 45 && ?pc13) ||
       (?t > 45 && ?t <= 50 && ?pc15)
     ))
    AS ?notApplicable
  )
  BIND((! ?notApplicable) AS ?applicable)

  BIND(
    IF(
      ?m = nltl:steelMaterialClassOne,
      IF(
        ?t <= 10,
        nltl:steelGradeB,
        IF(
          ?t <= 30,
          IF(?pc15, nltl:steelGradeD, nltl:steelGradeB),
          IF(?pc15, nltl:steelGradeE, nltl:steelGradeD)
        )
      ),
      IF(
        ?m = nltl:steelMaterialClassTwo,
        IF(
          ?t <= 10,
          nltl:steelGradeB,
          IF(
            ?t <= 25,
            IF(?pc15, nltl:steelGradeD, nltl:steelGradeB),
            IF(?pc15, nltl:steelGradeE, nltl:steelGradeD)
          )
        ),
        IF(
          ?t <= 10,
          nltl:steelGradeB,
          IF(
            ?t <= 35,
            IF(?pc15, nltl:steelGradeE, nltl:steelGradeB),
            IF(
              ?t <= 45,
              IF(?pc15, nltl:steelGradeE, nltl:steelGradeD),
              nltl:steelGradeE
            )
          )
        )
      )
    )
    AS ?normalGrade
  )

  BIND(
    IF(
      ?m = nltl:steelMaterialClassOne,
      IF(
        ?t <= 10,
        nltl:steelGradeAh,
        IF(
          ?t <= 30,
          IF(?pc15, nltl:steelGradeDh, nltl:steelGradeAh),
          IF(?pc15, nltl:steelGradeEh, nltl:steelGradeDh)
        )
      ),
      IF(
        ?m = nltl:steelMaterialClassTwo,
        IF(
          ?t <= 10,
          nltl:steelGradeAh,
          IF(
            ?t <= 25,
            IF(?pc15, nltl:steelGradeDh, nltl:steelGradeAh),
            IF(?pc15, nltl:steelGradeEh, nltl:steelGradeDh)
          )
        ),
        IF(
          ?t <= 10,
          nltl:steelGradeAh,
          IF(
            ?t <= 35,
            IF(?pc15, nltl:steelGradeEh, nltl:steelGradeAh),
            IF(
              ?t <= 45,
              IF(?pc15, nltl:steelGradeEh, nltl:steelGradeDh),
              nltl:steelGradeEh
            )
          )
        )
      )
    )
    AS ?highGrade
  )

  BIND(IF(?high, ?highGrade, ?normalGrade) AS ?expectedGrade)

  FILTER(
    (?applicable &&
      NOT EXISTS {
        $this nltl:requiredHullStructuralSteelGrade ?required .
        FILTER(?required = ?expectedGrade)
      })
    ||
    (! ?applicable &&
      EXISTS {
        $this nltl:requiredHullStructuralSteelGrade ?nonApplicableResult
      })
  )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual steel grade does not meet the required Table 8 rank within its steel-strength category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:actualHullStructuralSteelGrade ?actual ;
        nltl:requiredHullStructuralSteelGrade ?required ;
        nltl:steelStrengthCategory ?category .
  ?actual nltl:steelGradeOrderRank ?actualRank .
  ?required nltl:steelGradeOrderRank ?requiredRank .
  FILTER(?actualRank < ?requiredRank)
  FILTER(
    (?category = nltl:normalStrengthSteelCategory &&
      ?actual IN (
        nltl:steelGradeB,
        nltl:steelGradeD,
        nltl:steelGradeE
      ) &&
      ?required IN (
        nltl:steelGradeB,
        nltl:steelGradeD,
        nltl:steelGradeE
      ))
    ||
    (?category = nltl:highTensileSteelCategory &&
      ?actual IN (
        nltl:steelGradeAh,
        nltl:steelGradeDh,
        nltl:steelGradeEh,
        nltl:steelGradeFh
      ) &&
      ?required IN (
        nltl:steelGradeAh,
        nltl:steelGradeDh,
        nltl:steelGradeEh,
        nltl:steelGradeFh
      ))
  )
}
"""
    ] .
