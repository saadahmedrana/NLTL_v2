<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_061_CalculationCaseShape ;
        sh:message "Each linked calculation case must follow the prescribed analytical procedure model."
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gen:I2_061_HullStructureShape ;
        sh:message "A calculation case must identify its assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "The calculation case must use the prescribed analytical procedure and not a direct calculation as an alternative."
    ] .

gen:I2_061_HullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "The assessed hull structure must provide thickness as a metre-valued decimal quantity."
    ] .
<END_SHACL>
