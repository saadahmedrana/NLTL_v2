<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_064_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:hasValue nltl:linearCalculationMethodValue ;
                sh:message "A calculation case must identify the linear calculation method."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one yield-strength quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasStructuralMemberWeb ;
                sh:node [
                    sh:class nltl:structuralMemberWeb ;
                    sh:property [
                        sh:path nltl:nominalShearStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:message "The applicable web hull structure must satisfy the buckling criteria."
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberFlange ;
                sh:node [
                    sh:class nltl:structuralMemberFlange ;
                    sh:property [
                        sh:path nltl:nominalFlangeVonMisesStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:message "The applicable flange hull structure must satisfy the buckling criteria."
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For the linear calculation method, nominal web shear stress must be less than yield strength divided by sqrt(3)." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
    $this nltl:yieldStrength ?yieldQuantity .
    ?yieldQuantity qudt:numericValue ?yield .
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasStructuralMemberWeb ?web .
    ?web nltl:nominalShearStress ?stressQuantity .
    ?stressQuantity qudt:numericValue ?stress .
    FILTER (?stress >= (?yield / 1.7320508075688772))
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For the linear calculation method, nominal flange von Mises stress must be less than 1.15 times yield strength." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
    $this nltl:yieldStrength ?yieldQuantity .
    ?yieldQuantity qudt:numericValue ?yield .
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasStructuralMemberFlange ?flange .
    ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
    ?stressQuantity qudt:numericValue ?stress .
    FILTER (?stress >= (1.15 * ?yield))
}
"""
    ] .
<END_SHACL>
