<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-067> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship has a structural connection, strength continuity status must be recorded as true on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:structuralConnection ?connection .
                FILTER NOT EXISTS {
                    $this nltl:strengthContinuityStatus ?status .
                    FILTER (
                        datatype(?status) = xsd:boolean &&
                        ?status = true
                    )
                }
            }
        """
    ] .
<END_SHACL>
