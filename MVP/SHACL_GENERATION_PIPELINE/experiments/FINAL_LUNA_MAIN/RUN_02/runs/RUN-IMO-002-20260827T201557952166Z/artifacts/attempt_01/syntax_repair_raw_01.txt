<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:ShipDesignIceCategoryShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition to determine the category rule branch."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its Polar Code category to determine the category rule branch."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryA
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:thinFirstYearIceWithPossibleOldIceInclusions
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryB
            ]
        ]
        [
            sh:sparql [
                sh:message "A ship with a design ice condition other than thin first-year ice with possible old-ice inclusions is outside this category-B condition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:designIceCondition ?ice .
                        FILTER NOT EXISTS {
                            $this nltl:designIceCondition
                                nltl:thinFirstYearIceWithPossibleOldIceInclusions .
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
