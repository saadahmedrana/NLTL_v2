<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-002-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryA ;
            ] ;
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions
                ) ;
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryB ;
            ] ;
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:iceConditionLessSevereThanCategoryAAndB
                    nltl:iceFreeIceCondition
                    nltl:openWaterIceCondition
                    nltl:otherWatersIceCondition
                ) ;
            ] ;
        ]
    ) ;
    sh:message "A ship designed for at least thin first-year ice must have Polar Ship Category B unless it is Category A." .
<END_SHACL>
