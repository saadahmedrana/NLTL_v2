<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_005_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship described as being in ice free waters must have totalIcePresent false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER (?condition = "ice free waters"^^xsd:string)
    FILTER (
        NOT EXISTS {
            $this nltl:totalIcePresent ?presentFalse .
            FILTER (?presentFalse = false)
        }
        ||
        EXISTS {
            $this nltl:totalIcePresent ?presentTrue .
            FILTER (?presentTrue = true)
        }
    )
}
"""
    ] .
<END_SHACL>
