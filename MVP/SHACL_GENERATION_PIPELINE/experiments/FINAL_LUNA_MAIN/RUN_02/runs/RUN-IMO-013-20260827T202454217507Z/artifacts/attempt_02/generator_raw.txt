@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_013_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:lowestMdlt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxExclusive "-10"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A ship with a lowest MDLT below -10 deg C must declare shipOperatesInLowAirTemperature as true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:lowestMdlt ?quantity .
    ?quantity qudt:numericValue ?value ;
              qudt:unit unit:DEG_C .
    FILTER (?value < "-10"^^xsd:decimal)
    FILTER NOT EXISTS {
        $this nltl:shipOperatesInLowAirTemperature true .
    }
}
"""
    ] .
