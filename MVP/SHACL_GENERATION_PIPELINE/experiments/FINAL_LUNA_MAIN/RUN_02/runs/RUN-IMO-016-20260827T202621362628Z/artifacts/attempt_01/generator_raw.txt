<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO016ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:node gen:IMO016CertificateShape ;
        sh:message "The ship shall have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate language shall be recorded."
    ] ;
    sh:sparql [
        sh:message "If the certificate language is neither English, French, nor Spanish, an approved translation into one of those languages shall be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (STR(?language) NOT IN ("English", "French", "Spanish"))
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent ?translated .
                    FILTER (?translated = true)
                }
            }
        """
    ] .

gen:IMO016CertificateShape a sh:NodeShape ;
    sh:class nltl:polarShipCertificateForm ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate shall use a form corresponding to the Appendix I model."
    ] .
<END_SHACL>
