<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:node [
            sh:property [
                sh:path nltl:combustionAirTemperatureControlPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Each essential engine must have combustion-air temperature control present."
            ] ;
            sh:property [
                sh:path nltl:combustionAirTemperature ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each essential engine must provide one combustion-air temperature in degrees Celsius."
            ] ;
            sh:property [
                sh:path nltl:manufacturerMinAirTemperature ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each essential engine must provide one manufacturer minimum combustion-air temperature in degrees Celsius."
            ] ;
            sh:property [
                sh:path nltl:manufacturerMaxAirTemperature ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each essential engine must provide one manufacturer maximum combustion-air temperature in degrees Celsius."
            ] ;
            sh:sparql [
                sh:message "The combustion-air temperature must be within the engine manufacturer's minimum and maximum criteria." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:combustionAirTemperature ?temperatureNode .
                        $this nltl:manufacturerMinAirTemperature ?minimumNode .
                        $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                        ?temperatureNode qudt:numericValue ?temperature .
                        ?minimumNode qudt:numericValue ?minimum .
                        ?maximumNode qudt:numericValue ?maximum .
                        FILTER (
                            ?temperature < ?minimum ||
                            ?temperature > ?maximum
                        )
                    }
                """
            ]
        ]
    ] .
