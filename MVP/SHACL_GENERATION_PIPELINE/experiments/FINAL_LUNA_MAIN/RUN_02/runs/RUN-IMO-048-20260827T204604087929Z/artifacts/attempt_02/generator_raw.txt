<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-048-ExposedMaterialApprovalShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:exposedMachineryOrFoundationMaterial ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each exposed machinery or foundation material requires approved material evidence, an authorized approving authority, an authorized approval standard, and a decimal polar service temperature in degrees Celsius." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:exposedMachineryOrFoundationMaterial ?material .

                FILTER (
                    NOT EXISTS {
                        $this nltl:materialApprovalStatus nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:materialApprovalStatus ?status
                        FILTER (?status != nltl:evidenceStateApproved)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:approvingAuthority ?authority
                        FILTER (
                            ?authority = "Administration"
                            ||
                            ?authority = "recognized organization accepted by the Administration"
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:approvingAuthority ?invalidAuthority
                        FILTER (
                            ?invalidAuthority != "Administration"
                            &&
                            ?invalidAuthority != "recognized organization accepted by the Administration"
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:approvalStandard ?standard
                        FILTER (
                            ?standard = "standard acceptable to the Organization"
                            ||
                            ?standard = "another standard offering an equivalent level of safety"
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:approvalStandard ?invalidStandard
                        FILTER (
                            ?invalidStandard != "standard acceptable to the Organization"
                            &&
                            ?invalidStandard != "another standard offering an equivalent level of safety"
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:polarServiceTemperature ?temperatureNode .
                        ?temperatureNode a qudt:QuantityValue ;
                            qudt:numericValue ?temperatureValue ;
                            qudt:unit unit:DEG_C .
                        FILTER (datatype(?temperatureValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:polarServiceTemperature ?invalidTemperatureNode .
                        FILTER NOT EXISTS {
                            ?invalidTemperatureNode a qudt:QuantityValue ;
                                qudt:numericValue ?validTemperatureValue ;
                                qudt:unit unit:DEG_C .
                            FILTER (datatype(?validTemperatureValue) = xsd:decimal)
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
