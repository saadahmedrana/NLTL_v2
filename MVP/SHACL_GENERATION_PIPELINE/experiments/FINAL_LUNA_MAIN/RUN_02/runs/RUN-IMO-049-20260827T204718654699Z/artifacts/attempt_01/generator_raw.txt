<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO049_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltl:IMO049_CategoryABBranchShape
        nltl:IMO049_CategoryCBranchShape
    ) .

nltl:IMO049_CategoryABBranchShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1 ;
        sh:message "The ship shall be Category A or Category B for this approval branch."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node nltl:IMO049_CategoryABComponentShape
    ] .

nltl:IMO049_CategoryABComponentShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each applicable machinery component shall identify its component type."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node nltl:IMO049_CategoryABApprovalShape ;
        sh:message "Each applicable machinery component shall have an approval record."
    ] .

nltl:IMO049_CategoryABApprovalShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval evidence shall be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval shall be issued by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval shall use an Organization-acceptable standard or an equivalent-safety standard."
    ] .

nltl:IMO049_CategoryCBranchShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
        sh:message "The ship shall be Category C for this approval branch."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The Category C ship shall be ice-strengthened for this approval branch."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node nltl:IMO049_CategoryCComponentShape
    ] .

nltl:IMO049_CategoryCComponentShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each applicable machinery component shall identify its component type."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node nltl:IMO049_CategoryCApprovalShape ;
        sh:message "Each applicable machinery component shall have an approval record."
    ] .

nltl:IMO049_CategoryCApprovalShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval evidence shall be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval shall be issued by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:hasValue "acceptable standard adequate for operating ice type and concentration" ;
        sh:minCount 1 ;
        sh:message "The Category C approval shall use an acceptable standard adequate for the operating ice type and concentration."
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Category C approval evidence shall specify the operating ice type."
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:node nltl:IMO049_IceConcentrationShape ;
        sh:message "Category C approval evidence shall specify the operating ice concentration."
    ] .

nltl:IMO049_IceConcentrationShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1
    ] .
<END_SHACL>
