<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO049_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:machineryComponentType ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "Each applicable machinery component for a Category A or B ship must have approved scantling evidence with an authorized authority and an acceptable or equivalent-safety standard." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipCategory ?category .
                        FILTER (?category IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        ))
                        $this nltl:machineryComponentType ?componentType .
                        FILTER (?componentType IN (
                            "propeller blade",
                            "propeller blades",
                            "propulsion line",
                            "steering equipment",
                            "appendage",
                            "appendages",
                            "other appendages"
                        ))
                        $this nltl:hasComponent ?component .
                        ?component a nltl:machineryComponent .
                        FILTER NOT EXISTS {
                            ?component nltl:hasApprovalRecord ?approval .
                            ?approval a nltl:approvalRecord ;
                                nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                                nltl:approvingAuthority ?authority ;
                                nltl:approvalStandard ?standard .
                            FILTER (?authority IN (
                                "Administration",
                                "recognized organization accepted by the Administration"
                            ))
                            FILTER (?standard IN (
                                "standard acceptable to the Organization",
                                "another standard offering an equivalent level of safety"
                            ))
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:machineryComponentType ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "Each applicable machinery component for an ice-strengthened Category C ship must have approved scantling evidence with an authorized authority, an adequate acceptable standard, and operating-ice context." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipCategory nltl:polarShipCategoryC ;
                            nltl:shipIceStrengthened true .
                        $this nltl:machineryComponentType ?componentType .
                        FILTER (?componentType IN (
                            "propeller blade",
                            "propeller blades",
                            "propulsion line",
                            "steering equipment",
                            "appendage",
                            "appendages",
                            "other appendages"
                        ))
                        $this nltl:hasComponent ?component .
                        ?component a nltl:machineryComponent .
                        FILTER NOT EXISTS {
                            ?component nltl:hasApprovalRecord ?approval .
                            ?approval a nltl:approvalRecord ;
                                nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                                nltl:approvingAuthority ?authority ;
                                nltl:approvalStandard ?standard ;
                                nltl:operatingIceType ?iceType ;
                                nltl:operatingIceConcentration ?concentration .
                            FILTER (?authority IN (
                                "Administration",
                                "recognized organization accepted by the Administration"
                            ))
                            FILTER (?standard =
                                "acceptable standard adequate for operating ice type and concentration"
                            )
                            ?concentration qudt:numericValue ?numericValue ;
                                qudt:unit qudt:UNITLESS .
                            FILTER (datatype(?numericValue) = xsd:decimal)
                        }
                    }
                """
            ]
        ]
    ] .
<END_SHACL>
