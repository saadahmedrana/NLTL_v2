<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_055_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Valve exposure must be provided as a boolean."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Valve type must be provided as a string."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:valveExposed ;
                sh:hasValue false
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:valveExposed ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:valveType ;
                        sh:in ( "isolating" "pressure" "vacuum" )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:iceAccretionProtectionPresent ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue true ;
                        sh:message "An exposed isolating, pressure, or vacuum valve must have ice accretion protection."
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:continuouslyAccessible ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue true ;
                        sh:message "An exposed isolating, pressure, or vacuum valve must remain continuously accessible."
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed isolating, pressure, or vacuum valve must not have a false ice-protection or accessibility value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:valveExposed true ;
                      nltl:valveType ?type .
                FILTER (?type IN ("isolating", "pressure", "vacuum"))
                FILTER (
                    EXISTS {
                        $this nltl:iceAccretionProtectionPresent ?protection .
                        FILTER (!sameTerm(?protection, true))
                    }
                    ||
                    EXISTS {
                        $this nltl:continuouslyAccessible ?accessible .
                        FILTER (!sameTerm(?accessible, true))
                    }
                )
            }
        """
    ] .
<END_SHACL>
