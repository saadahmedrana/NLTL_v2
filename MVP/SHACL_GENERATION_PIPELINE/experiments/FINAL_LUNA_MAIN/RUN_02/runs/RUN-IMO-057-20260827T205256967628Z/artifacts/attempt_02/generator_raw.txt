<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

g:IMO057FirePumpCompartmentTemperatureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each fire pump, emergency fire pump, water-mist pump, and water-spray pump must be linked to a compartment having a maintained temperature above 0 deg C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX qudtunit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpClass .
                FILTER (
                    ?pumpClass IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER NOT EXISTS {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment a nltl:compartment ;
                        nltl:maintainedTemperature ?temperature .
                    ?temperature a qudt:QuantityValue ;
                        qudt:unit qudtunit:DEG_C ;
                        qudt:numericValue ?value .
                    FILTER (datatype(?value) = xsd:decimal)
                    FILTER (?value > "0"^^xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
