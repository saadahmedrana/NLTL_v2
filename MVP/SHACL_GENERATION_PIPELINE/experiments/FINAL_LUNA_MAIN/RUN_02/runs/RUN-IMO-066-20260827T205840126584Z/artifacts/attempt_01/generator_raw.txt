<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO066ProtectedItemShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:protectedItemCategory ;
        sh:in (
            nltl:escapeRouteProtectedItem
            nltl:musterStationProtectedItem
            nltl:embarkationAreaProtectedItem
            nltl:survivalCraftProtectedItem
            nltl:launchingApplianceProtectedItem
            nltl:survivalCraftAccessProtectedItem
        ) ;
        sh:minCount 1 ;
        sh:message "Each protected item must identify one permitted protected-item category."
    ] ;
    sh:property [
        sh:path nltl:hasIceSnowRemovalOrPreventionMeans ;
        sh:class nltl:iceSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:message "Each protected item must have associated ice or snow removal or prevention means."
    ] .

nltlgen:IMO066ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's ice-accretion exposure assessment must be recorded as one boolean value."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:class nltl:protectedItem ;
                sh:minCount 1 ;
                sh:node nltlgen:IMO066ProtectedItemShape
            ] ;
            sh:sparql [
                sh:message "When the ship is exposed to ice accretion, every required protected-item category must have associated ice or snow removal or prevention means." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT $this
                    WHERE {
                        $this nltl:shipExposedToIceAccretion ?exposed .
                        FILTER (?exposed = true)
                        FILTER (
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item1 .
                                ?item1 nltl:protectedItemCategory nltl:escapeRouteProtectedItem .
                                ?item1 nltl:hasIceSnowRemovalOrPreventionMeans ?means1 .
                                ?means1 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item2 .
                                ?item2 nltl:protectedItemCategory nltl:musterStationProtectedItem .
                                ?item2 nltl:hasIceSnowRemovalOrPreventionMeans ?means2 .
                                ?means2 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item3 .
                                ?item3 nltl:protectedItemCategory nltl:embarkationAreaProtectedItem .
                                ?item3 nltl:hasIceSnowRemovalOrPreventionMeans ?means3 .
                                ?means3 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item4 .
                                ?item4 nltl:protectedItemCategory nltl:survivalCraftProtectedItem .
                                ?item4 nltl:hasIceSnowRemovalOrPreventionMeans ?means4 .
                                ?means4 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item5 .
                                ?item5 nltl:protectedItemCategory nltl:launchingApplianceProtectedItem .
                                ?item5 nltl:hasIceSnowRemovalOrPreventionMeans ?means5 .
                                ?means5 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasProtectedItem ?item6 .
                                ?item6 nltl:protectedItemCategory nltl:survivalCraftAccessProtectedItem .
                                ?item6 nltl:hasIceSnowRemovalOrPreventionMeans ?means6 .
                                ?means6 a nltl:iceSnowRemovalOrPreventionMeans .
                            }
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
