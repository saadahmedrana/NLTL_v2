<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO070ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one ship type."
    ] ;
    sh:or (
        gen:IMO070PassengerBranch
        gen:IMO070NonPassengerBranch
    ) .

gen:IMO070PassengerBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "passenger_ship"
    ] ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A passenger ship must have exactly one true size-compatibility status."
    ] ;
    sh:property [
        sh:path nltl:hasPersonOnBoardMember ;
        sh:node gen:IMO070PersonShape ;
        sh:message "Each person on board a passenger ship must have one qualifying survival-equipment assignment."
    ] .

gen:IMO070NonPassengerBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The passenger-ship branch must not be bypassed for a ship whose type is passenger_ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipType ?shipType .
                FILTER (STR(?shipType) = "passenger_ship")
            }
        """
    ] .

gen:IMO070PersonShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasPersonalSurvivalEquipmentAssignment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO070AssignmentShape ;
        sh:message "Each person on board must have exactly one personal survival-equipment assignment."
    ] .

gen:IMO070AssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:assignedPersonalSurvivalEquipment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO070EquipmentChoiceShape ;
        sh:message "Each assignment must contain exactly one immersion suit or thermal protective aid equipment item."
    ] .

gen:IMO070EquipmentChoiceShape
    a sh:NodeShape ;
    sh:or (
        gen:IMO070ImmersionSuitShape
        gen:IMO070ThermalAidShape
    ) .

gen:IMO070ImmersionSuitShape
    a sh:NodeShape ;
    sh:class nltl:immersionSuitEquipment .

gen:IMO070ThermalAidShape
    a sh:NodeShape ;
    sh:class nltl:thermalProtectiveAidEquipment .
<END_SHACL>
