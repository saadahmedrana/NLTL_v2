<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-074> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:abandonmentOntoIceOrLandPotential ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:groupSurvivalEquipmentCarried ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When abandonment onto ice or land potential is true, group survival equipment must be carried or approved equivalent functionality must be provided by normal life-saving appliances." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:abandonmentOntoIceOrLandPotential true .
                FILTER NOT EXISTS {
                    $this nltl:groupSurvivalEquipmentCarried true .
                }
                FILTER NOT EXISTS {
                    $this nltl:normalLifeSavingEquivalentFunctionalityApprovalStatus
                        nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
