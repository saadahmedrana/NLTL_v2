<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_077_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_077_ApplicableShape
        gen:IMO_077_NotApplicableShape
    ) .

gen:IMO_077_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether additional equipment is carried in the survival craft."
    ] ;
    sh:property [
        sh:path nltl:personsCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The persons capacity requirement is required when additional equipment is carried."
    ] ;
    sh:property [
        sh:path nltl:additionalEquipmentCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The additional equipment capacity requirement is required when additional equipment is carried."
    ] ;
    sh:property [
        sh:path nltl:survivalCraftAvailableCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The available survival-craft capacity is required when additional equipment is carried."
    ] ;
    sh:property [
        sh:path nltl:launchingApplianceCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The launching appliance capacity must be supplied as a kilogram quantity."
    ] ;
    sh:property [
        sh:path nltl:loadedSurvivalCraftRequirement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The loaded survival-craft requirement must be supplied as a kilogram quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When additional equipment is carried in the survival craft, available survival-craft capacity must be at least the persons requirement plus the additional equipment requirement, and launching-appliance capacity must be at least the loaded survival-craft requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:additionalEquipmentCarriedInSurvivalCraft true ;
                      nltl:personsCapacityRequirement ?persons ;
                      nltl:additionalEquipmentCapacityRequirement ?additional ;
                      nltl:survivalCraftAvailableCapacity ?available ;
                      nltl:launchingApplianceCapacity ?launchingNode ;
                      nltl:loadedSurvivalCraftRequirement ?loadedNode .

                ?launchingNode qudt:numericValue ?launching ;
                               qudt:unit unit:KiloGM .
                ?loadedNode qudt:numericValue ?loaded ;
                            qudt:unit unit:KiloGM .

                FILTER (
                    ?available < (?persons + ?additional) ||
                    ?launching < ?loaded
                )
            }
        """
    ] .

gen:IMO_077_NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether additional equipment is carried in the survival craft."
    ] .
<END_SHACL>
