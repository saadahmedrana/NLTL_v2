<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_087_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DEG
            ]
        ]
    ] ;
    sh:or (
        [
            sh:node gen:IMO_087_ApplicableBranch
        ]
        [
            sh:node gen:IMO_087_NonApplicableBranch
        ]
    ) .

gen:IMO_087_ApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:gnssCompassOrEquivalentCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:minInclusive 1 ;
        sh:message "A ship proceeding over 80 degrees latitude must have at least one GNSS compass or equivalent."
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:benchmarkEntity
        ] ;
        sh:message "The GNSS compass or equivalent must be connected to the ship's main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:benchmarkEntity
        ] ;
        sh:message "The GNSS compass or equivalent must be connected to the ship's emergency power."
    ] ;
    sh:sparql [
        sh:message "This branch applies only when the planned or actual latitude is over 80 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plannedOrActualLatitude ?latitudeNode .
                ?latitudeNode qudt:numericValue ?latitudeValue .
                FILTER (?latitudeValue <= 80)
            }
        """
    ] .

gen:IMO_087_NonApplicableBranch a sh:NodeShape ;
    sh:sparql [
        sh:message "The conditional equipment requirements apply when the planned or actual latitude is over 80 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plannedOrActualLatitude ?latitudeNode .
                ?latitudeNode qudt:numericValue ?latitudeValue .
                FILTER (?latitudeValue > 80)
            }
        """
    ] .
<END_SHACL>
