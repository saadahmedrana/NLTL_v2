<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO102ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled 12.3.1 ice condition."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:class nltl:polarTrainingShipTypeValue ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled 12.3.1 ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gen:IMO102CrewMemberShape
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:class nltl:polarTrainingCrewRoleValue ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented crew member must have exactly one controlled polar training role."
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:class nltl:polarTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] ;
    sh:property [
        sh:path nltl:hasPolarTrainingRecord ;
        sh:class nltl:polarTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable crew member must have both declared polar training record relationships." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                       nltl:iceConditionClassification ?ice ;
                       nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role .
                FILTER (
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType
                        )
                    ) ||
                    ?ice = nltl:otherWatersIceCondition
                )
                FILTER NOT EXISTS {
                    $this nltl:hasCrewTrainingRecord ?crewRecord .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPolarTrainingRecord ?polarRecord .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The crew member's training record must contain the matrix-required level and a valid STCW qualification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                       nltl:iceConditionClassification ?ice ;
                       nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role ;
                      nltl:hasCrewTrainingRecord ?record .
                ?record nltl:requiredTrainingLevel ?required ;
                        nltl:trainingRecordLevel ?recorded ;
                        nltl:stcwQualificationValid ?valid .
                FILTER (
                    (?ice = nltl:openWaterIceCondition &&
                     ?shipType IN (
                         nltl:tankerPolarTrainingShipType,
                         nltl:passengerShipPolarTrainingShipType
                     ) &&
                     (?required != nltl:basicPolarTrainingLevel ||
                      ?recorded != nltl:basicPolarTrainingLevel)) ||
                    (?ice = nltl:otherWatersIceCondition &&
                     ?role IN (
                         nltl:masterPolarTrainingRole,
                         nltl:chiefMatePolarTrainingRole
                     ) &&
                     (?required != nltl:advancedPolarTrainingLevel ||
                      ?recorded != nltl:advancedPolarTrainingLevel)) ||
                    (?ice = nltl:otherWatersIceCondition &&
                     ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                     (?required != nltl:basicPolarTrainingLevel ||
                      ?recorded != nltl:basicPolarTrainingLevel)) ||
                    ?valid != true
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The linked polar training record must contain the matrix-required level and a valid STCW qualification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                       nltl:iceConditionClassification ?ice ;
                       nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role ;
                      nltl:hasPolarTrainingRecord ?record .
                ?record nltl:requiredTrainingLevel ?required ;
                        nltl:trainingRecordLevel ?recorded ;
                        nltl:stcwQualificationValid ?valid .
                FILTER (
                    (?ice = nltl:openWaterIceCondition &&
                     ?shipType IN (
                         nltl:tankerPolarTrainingShipType,
                         nltl:passengerShipPolarTrainingShipType
                     ) &&
                     (?required != nltl:basicPolarTrainingLevel ||
                      ?recorded != nltl:basicPolarTrainingLevel)) ||
                    (?ice = nltl:otherWatersIceCondition &&
                     ?role IN (
                         nltl:masterPolarTrainingRole,
                         nltl:chiefMatePolarTrainingRole
                     ) &&
                     (?required != nltl:advancedPolarTrainingLevel ||
                      ?recorded != nltl:advancedPolarTrainingLevel)) ||
                    (?ice = nltl:otherWatersIceCondition &&
                     ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                     (?required != nltl:basicPolarTrainingLevel ||
                      ?recorded != nltl:basicPolarTrainingLevel)) ||
                    ?valid != true
                )
            }
        """
    ] .

gen:IMO102TrainingRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:class nltl:trainingLevelValue ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:class nltl:trainingLevelValue ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#boolean> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
