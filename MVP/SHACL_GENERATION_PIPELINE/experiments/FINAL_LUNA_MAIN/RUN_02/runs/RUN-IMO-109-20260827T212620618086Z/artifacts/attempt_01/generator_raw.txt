<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_109_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship category of A, B, or C is required."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship type is required."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date is required."
    ] ;
    sh:property [
        sh:path nltl:cargoTankCarriesOil ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether the cargo tank carries oil is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a category A or B ship constructed on or after 1 January 2017 that is not an oil tanker and whose cargo tanks carry oil, cargo tank outer shell separation must be at least 0.76 m and use metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil ?carriesOil .

                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (?carriesOil = true)
                FILTER (?shipType != "oil_tanker")
                FILTER NOT EXISTS {
                    $this a nltl:oilTanker .
                }

                FILTER NOT EXISTS {
                    $this nltl:cargoTankOuterShellSeparation ?separation .
                    ?separation a qudt:QuantityValue ;
                                qudt:numericValue ?value ;
                                qudt:unit unit:M .
                    FILTER (datatype(?value) = xsd:decimal)
                    FILTER (?value >= "0.76"^^xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Cargo tank outer shell separation must be a decimal quantity in metres and must be at least 0.76 m when the rule applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil ?carriesOil .

                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (?carriesOil = true)
                FILTER (?shipType != "oil_tanker")
                FILTER NOT EXISTS {
                    $this a nltl:oilTanker .
                }

                $this nltl:cargoTankOuterShellSeparation ?separation .
                ?separation a qudt:QuantityValue ;
                            qudt:numericValue ?value .
                FILTER (
                    datatype(?value) != xsd:decimal
                    || NOT EXISTS {
                        ?separation qudt:unit unit:M .
                    }
                    || ?value < "0.76"^^xsd:decimal
                )
            }
        """
    ] .
<END_SHACL>
