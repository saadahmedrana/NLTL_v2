<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_112_Arctic_NLS_Discharge_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the operating area is Arctic_waters, discharge of noxious liquid substances or mixtures must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters")
                FILTER (
                    NOT EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                        FILTER (?discharge = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                        FILTER (?discharge != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
