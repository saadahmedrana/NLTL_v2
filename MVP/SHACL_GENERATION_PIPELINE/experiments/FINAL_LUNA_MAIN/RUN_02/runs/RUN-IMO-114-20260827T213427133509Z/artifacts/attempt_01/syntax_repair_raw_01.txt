<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-114> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, Administration approval and the applicable polar operation certificate entry are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                $this nltl:constructionDate ?date .
                $this nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 .
                $this nltl:cargoTankType ?tankType .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?date >= "2017-01-01"^^xsd:date
                    && ?requiresType3 = true
                    && ?tankType = "3"
                )

                FILTER ( NOT EXISTS {
                    $this nltl:administrationCarriageApprovalStatus nltl:evidenceStateApproved .
                } )
                FILTER ( NOT EXISTS {
                    $this nltl:applicableCertificatePolarOperationEntryPresent "true"^^xsd:boolean .
                } )
                FILTER ( NOT EXISTS {
                    $this nltl:administrationCarriageApprovalStatus ?otherStatus .
                    FILTER ( ?otherStatus != nltl:evidenceStateApproved )
                } )
                FILTER ( NOT EXISTS {
                    $this nltl:applicableCertificatePolarOperationEntryPresent ?otherEntry .
                    FILTER ( ?otherEntry != true^^xsd:boolean )
                } )
            }
        """
    ] .
<END_SHACL>
