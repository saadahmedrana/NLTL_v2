<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-119> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship category must be recorded as Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice concentration must be recorded as a single dimensionless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:extendedPeriodOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a Category A or B ship operating for an extended period in ice concentration greater than 1/10 while sewage discharge occurs, an approved sewage treatment plant and approved MARPOL certification and administration approval are required, with no conflicting outcome values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )

                $this nltl:iceConcentration ?iceQuantity .
                ?iceQuantity qudt:numericValue ?iceValue .
                ?iceQuantity qudt:unit unit:UNITLESS .
                FILTER ( ?iceValue > "1"^^xsd:decimal )

                $this nltl:extendedPeriodOperation true .
                $this nltl:sewageDischargeOccurs true .

                FILTER (
                    NOT EXISTS {
                        $this nltl:approvedSewageTreatmentPlantUsed true
                    }
                    ||
                    EXISTS {
                        $this nltl:approvedSewageTreatmentPlantUsed ?plantValue .
                        FILTER ( ?plantValue != true )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?annexStatus .
                        FILTER ( ?annexStatus != nltl:evidenceStateApproved )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolCertificationStatus nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:marpolCertificationStatus ?marpolStatus .
                        FILTER ( ?marpolStatus != nltl:evidenceStateApproved )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:administrationDischargeApprovalStatus nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:administrationDischargeApprovalStatus ?approvalStatus .
                        FILTER ( ?approvalStatus != nltl:evidenceStateApproved )
                    }
                )
            }
        """
    ] .
<END_SHACL>
