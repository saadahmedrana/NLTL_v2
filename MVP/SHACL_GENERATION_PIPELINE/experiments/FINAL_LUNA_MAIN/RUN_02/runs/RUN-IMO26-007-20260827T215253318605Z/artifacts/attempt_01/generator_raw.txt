<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:NotApplicableIceNotStrengthened
        gen:NotApplicableConstructedBeforeDate
        gen:ApplicableAndCompliant
    ) ;
    sh:message "For ice-strengthened ships constructed on or after 1 January 2026, the required depth-sounding equipment alternative or approved equivalent must be present." .

gen:NotApplicableIceNotStrengthened a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] .

gen:NotApplicableConstructedBeforeDate a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxExclusive "2026-01-01"^^xsd:date
    ] .

gen:ApplicableAndCompliant a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:minInclusive "2026-01-01"^^xsd:date
    ] ;
    sh:or (
        gen:TwoIndependentEchoSoundingDevices
        gen:OneEchoSoundingDeviceWithTwoIndependentTransducers
        gen:ApprovedEquivalentDepthSoundingDevice
    ) .

gen:TwoIndependentEchoSoundingDevices a sh:NodeShape ;
    sh:property [
        sh:path nltl:independentEchoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2
    ] .

gen:OneEchoSoundingDeviceWithTwoIndependentTransducers a sh:NodeShape ;
    sh:property [
        sh:path nltl:echoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 1
    ] ;
    sh:property [
        sh:path nltl:independentTransducerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2
    ] .

gen:ApprovedEquivalentDepthSoundingDevice a sh:NodeShape ;
    sh:property [
        sh:path nltl:administrationApprovedEquivalentDepthSoundingDevicePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] .
<END_SHACL>
