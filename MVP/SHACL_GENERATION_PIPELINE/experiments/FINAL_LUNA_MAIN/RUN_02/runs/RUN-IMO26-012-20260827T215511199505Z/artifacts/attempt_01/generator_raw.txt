<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO26_012_HeadingMeansTopologyShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When at least two non-magnetic heading means are represented, each must be connected to main and emergency power and be independent from another distinct represented means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?m1 .
                $this nltl:hasNonMagneticHeadingMeans ?m2 .
                FILTER (?m1 != ?m2)

                {
                    FILTER NOT EXISTS {
                        ?m1 nltl:connectedToMainPower ?mainPower .
                        FILTER (isIRI(?mainPower))
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?m1 nltl:connectedToEmergencyPower ?emergencyPower .
                        FILTER (isIRI(?emergencyPower))
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?m1 nltl:independentFromHeadingMeans ?otherMeans .
                        $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                        FILTER (?otherMeans != ?m1)
                    }
                }
            }
        """
    ] .
<END_SHACL>
