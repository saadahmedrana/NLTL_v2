<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-012-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:node [
            sh:class nltl:nonMagneticHeadingMeans
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of at least 500 gross tonnage shall have at least two distinct typed non-magnetic heading means; each shall be connected to main and emergency power and be independent from another distinct heading means on the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:grossTonnage ?tonnageNode .
                ?tonnageNode qudt:numericValue ?tonnageValue ;
                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (?tonnageValue >= 500)

                FILTER (
                    NOT EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?firstMeans ;
                              nltl:hasNonMagneticHeadingMeans ?secondMeans .
                        ?firstMeans a nltl:nonMagneticHeadingMeans .
                        ?secondMeans a nltl:nonMagneticHeadingMeans .
                        FILTER (?firstMeans != ?secondMeans)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?means .
                        ?means a nltl:nonMagneticHeadingMeans .
                        FILTER NOT EXISTS {
                            ?means nltl:connectedToMainPower ?mainPower .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?means .
                        ?means a nltl:nonMagneticHeadingMeans .
                        FILTER NOT EXISTS {
                            ?means nltl:connectedToEmergencyPower ?emergencyPower .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?means .
                        ?means a nltl:nonMagneticHeadingMeans .
                        FILTER NOT EXISTS {
                            $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                            ?otherMeans a nltl:nonMagneticHeadingMeans .
                            ?means nltl:independentFromHeadingMeans ?otherMeans .
                            FILTER (?otherMeans != ?means)
                        }
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:IMO26-012-HeadingMeansShape> a sh:NodeShape ;
    sh:targetClass nltl:nonMagneticHeadingMeans ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans
    ] .

<END_SHACL>
