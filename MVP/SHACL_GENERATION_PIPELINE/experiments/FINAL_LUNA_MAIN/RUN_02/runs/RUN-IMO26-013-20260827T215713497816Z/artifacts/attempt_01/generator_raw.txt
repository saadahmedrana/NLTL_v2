<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_013_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:node gen:LatitudeQuantityShape
    ] ;
    sh:or (
        gen:IMO26_013_ApplicableShape
        gen:IMO26_013_NonApplicableShape
    ) .

gen:LatitudeQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LatitudeOver80Shape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minExclusive 80
    ] .

gen:LatitudeAtMost80Shape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:maxInclusive 80
    ] .

gen:IMO26_013_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:node gen:LatitudeOver80Shape
    ] ;
    sh:property [
        sh:path nltl:gnssCompassOrEquivalentCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:minInclusive 1 ;
        sh:message "A ship proceeding to a latitude over 80 degrees must have at least one GNSS compass or equivalent."
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:message "The required GNSS compass or equivalent must be connected to the ship's main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:message "The required GNSS compass or equivalent must be connected to the ship's emergency power."
    ] .

gen:IMO26_013_NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:node gen:LatitudeAtMost80Shape
    ] .
<END_SHACL>
