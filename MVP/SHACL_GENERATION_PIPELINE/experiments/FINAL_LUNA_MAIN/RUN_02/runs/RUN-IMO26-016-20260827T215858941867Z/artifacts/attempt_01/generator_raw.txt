<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "A ship must provide whether it operates in polar waters."
    ] ;
    sh:sparql [
        sh:message "For a ship operating in polar waters, Chapter 11-1 applicability must be true and the ship particulars must satisfy one permitted scope case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters ?operates ;
                      nltl:polarCodeChapter11Dash1Applicable ?applicable .
                FILTER (datatype(?operates) = xsd:boolean)
                FILTER (?operates = true)
                FILTER (datatype(?applicable) = xsd:boolean)
                FILTER (?applicable = true)

                FILTER NOT EXISTS {
                    {
                        $this nltl:shipType ?shipType ;
                              nltl:lengthOverall ?lengthQuantity .
                        FILTER (datatype(?shipType) = xsd:string)
                        FILTER (?shipType = "fishing_vessel")
                        ?lengthQuantity qudt:numericValue ?lengthValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?lengthValue) = xsd:decimal)
                        FILTER (?lengthValue >= 24)
                    }
                    UNION
                    {
                        $this nltl:shipType ?shipType ;
                              nltl:grossTonnage ?grossQuantity ;
                              nltl:engagedInTrade ?trade .
                        FILTER (datatype(?shipType) = xsd:string)
                        FILTER (?shipType = "pleasure_yacht")
                        FILTER (datatype(?trade) = xsd:boolean)
                        FILTER (?trade = false)
                        ?grossQuantity qudt:numericValue ?grossValue ;
                                        qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?grossValue) = xsd:decimal)
                        FILTER (?grossValue >= 300)
                    }
                    UNION
                    {
                        $this nltl:shipType ?shipType ;
                              nltl:grossTonnage ?grossQuantity .
                        FILTER (datatype(?shipType) = xsd:string)
                        FILTER (?shipType = "cargo_ship")
                        ?grossQuantity qudt:numericValue ?grossValue ;
                                        qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?grossValue) = xsd:decimal)
                        FILTER (?grossValue >= 300)
                        FILTER (?grossValue < 500)
                    }
                }
            }
        """
    ] .
<END_SHACL>
