<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have one construction contract date."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulatoryProvisionValue ;
        sh:in (
            nltl:iceClassDraughtChapter2Provision2021
            nltl:iceClassSection1Point8Provision2021
        ) ;
        sh:minCount 2 ;
        sh:message "Both the 2021 chapter 2 Ice Class Draught provision and section 1.8 Ice Classes provision must apply."
    ] ;
    sh:sparql [
        sh:message "The 2021 Ice Class Regulations must be recorded for ships contracted on or after 5 July 2021 and must not be recorded as applicable for earlier contract dates." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    (
                        (
                            YEAR(?date) > 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) > 7) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) >= 5)
                        )
                        &&
                        NOT EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                  nltl:iceClassRegulationEdition2021 .
                        }
                    )
                    ||
                    (
                        (
                            YEAR(?date) < 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) < 7) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) < 5)
                        )
                        &&
                        EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                  nltl:iceClassRegulationEdition2021 .
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
