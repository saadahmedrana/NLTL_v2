<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-003> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2010 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        sh:message "For ships contracted on or after 1 January 2012 and before 1 January 2019, the applicable Ice Class regulation edition must be the 2010 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    (
                        YEAR(?date) > 2012 ||
                        (
                            YEAR(?date) = 2012 &&
                            (
                                MONTH(?date) > 1 ||
                                (MONTH(?date) = 1 && DAY(?date) >= 1)
                            )
                        )
                    )
                    &&
                    (
                        YEAR(?date) < 2019 ||
                        (
                            YEAR(?date) = 2019 &&
                            (
                                MONTH(?date) < 1 ||
                                (MONTH(?date) = 1 && DAY(?date) < 1)
                            )
                        )
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                            nltl:iceClassRegulationEdition2010 .
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?edition .
                        FILTER (?edition != nltl:iceClassRegulationEdition2010)
                    }
                )
            }
        """
    ] .
</END_SHACL>
