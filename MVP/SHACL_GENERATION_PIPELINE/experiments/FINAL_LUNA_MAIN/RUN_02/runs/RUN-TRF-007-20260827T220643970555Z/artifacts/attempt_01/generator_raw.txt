<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of TRF-007."
    ] ;
    sh:or (
        gen:TRF007ApplicableShape
        gen:TRF007NonApplicableClassShape
        gen:TRF007NonApplicableDateShape
    ) .

gen:TRF007ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide its construction-stage date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide its delivery date."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide an assessment date."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "TRF-007 applies only to IA or IA Super ships constructed before 1 September 2003 and requires compliance by 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIa,
                        nltl:iceClassIaSuper
                    )
                )
                FILTER (
                    YEAR(?constructionDate) < 2003
                    || (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) < 9
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
                    || (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        && MONTH(?assessmentDate) = 1
                        && DAY(?assessmentDate) = 1
                    )
                )
            }
        """
    ] .

gen:TRF007NonApplicableClassShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "TRF-007 is not applicable because the ship is neither ice class IA nor ice class IA Super." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass NOT IN (
                        nltl:iceClassIa,
                        nltl:iceClassIaSuper
                    )
                )
            }
        """
    ] .

gen:TRF007NonApplicableDateShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An IA or IA Super ship must provide its construction-stage date to determine whether TRF-007 applies."
    ] ;
    sh:sparql [
        sh:message "TRF-007 is not applicable because the IA or IA Super ship reached the relevant construction stage on or after 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIa,
                        nltl:iceClassIaSuper
                    )
                )
                FILTER (
                    YEAR(?constructionDate) > 2003
                    || (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) >= 9
                    )
                )
            }
        """
    ] .
<END_SHACL>
