<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must have exactly one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "Construction stage date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:message "Delivery date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:message "Assessment date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:message "The engine output compliance method must be one of the permitted 2017 regulation methods."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA or IA Super ship, construction stage date is required to determine applicability of the rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?stage .
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA or IA Super ship must have the required delivery date, assessment date, permitted compliance method, and deadline assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:constructionStageDate ?stage .

                FILTER (?ice IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))

                FILTER (?stage < "2003-09-01"^^xsd:date)

                FILTER (
                    NOT EXISTS {
                        $this nltl:deliveryDate ?delivery .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:assessmentDate ?assessment .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (?method IN (
                            nltl:traficom2017Section3Point2Point2Method,
                            nltl:traficom2017Section3Point2Point4Method
                        ))
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:deliveryDate ?delivery ;
                              nltl:assessmentDate ?assessment .
                        FILTER (
                            YEAR(?assessment) = YEAR(?delivery) + 20
                            &&
                            MONTH(?assessment) = 1
                            &&
                            DAY(?assessment) = 1
                        )
                    }
                )
            }
        """ ;
    ] .
</BEGIN_SHACL>
