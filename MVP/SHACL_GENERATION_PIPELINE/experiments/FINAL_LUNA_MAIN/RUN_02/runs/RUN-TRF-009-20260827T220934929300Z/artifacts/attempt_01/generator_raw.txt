<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-009-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have ice class IA Super or IA for this rule."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction stage date is required to determine applicability of this rule."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The delivery date is required to determine the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessment date is required to determine whether the compliance deadline has passed."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:TRF-009-ApplicableCase>
        <urn:nltl:generated-shape:TRF-009-NonApplicableCase>
    ) .

<urn:nltl:generated-shape:TRF-009-ApplicableCase>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, compliance must be established no later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .

                BIND(YEAR(?constructionStageDate) AS ?constructionYear)
                BIND(MONTH(?constructionStageDate) AS ?constructionMonth)
                BIND(DAY(?constructionStageDate) AS ?constructionDay)

                BIND(
                    (?constructionYear < 2003 ||
                     (?constructionYear = 2003 && ?constructionMonth < 9) ||
                     (?constructionYear = 2003 && ?constructionMonth = 9 && ?constructionDay < 1))
                    AS ?laidBeforeCutoff
                )

                BIND(
                    (?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    ) && ?laidBeforeCutoff)
                    AS ?applicable
                )

                BIND(YEAR(?deliveryDate) + 20 AS ?deadlineYear)
                BIND(YEAR(?assessmentDate) AS ?assessmentYear)
                BIND(MONTH(?assessmentDate) AS ?assessmentMonth)
                BIND(DAY(?assessmentDate) AS ?assessmentDay)

                FILTER(
                    !?applicable ||
                    ?assessmentYear > ?deadlineYear ||
                    (?assessmentYear = ?deadlineYear &&
                     (?assessmentMonth > 1 ||
                      (?assessmentMonth = 1 && ?assessmentDay > 1)))
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-009-NonApplicableCase>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This branch applies only when the ship is not an IA Super or IA ship with construction begun before 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate .

                BIND(YEAR(?constructionStageDate) AS ?constructionYear)
                BIND(MONTH(?constructionStageDate) AS ?constructionMonth)
                BIND(DAY(?constructionStageDate) AS ?constructionDay)

                FILTER(
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    ) &&
                    (
                        ?constructionYear < 2003 ||
                        (?constructionYear = 2003 && ?constructionMonth < 9) ||
                        (?constructionYear = 2003 && ?constructionMonth = 9 && ?constructionDay < 1)
                    )
                )
            }
        """
    ] .
<END_SHACL>
