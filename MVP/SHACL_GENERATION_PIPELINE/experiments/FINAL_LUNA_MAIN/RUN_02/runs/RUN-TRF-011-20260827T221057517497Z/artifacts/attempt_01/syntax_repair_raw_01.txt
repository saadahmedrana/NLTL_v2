<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF011IceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF011IceWaterlineShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must contain exactly the pointwise envelope of the intended ice-operating waterlines." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT *
            WHERE {
                {
                    SELECT ?x
                           (MAX(?iv) AS ?imax)
                           (MIN(?uv) AS ?umin)
                           (MAX(?uv) AS ?umax)
                           (COUNT(?up) AS ?ucount)
                    WHERE {
                        $this nltl:hasIntendedIceOperatingWaterline ?iw .
                        ?iw nltl:hasWaterlineProfilePoint ?ip .
                        ?ip nltl:longitudinalPosition ?ixn ;
                            nltl:verticalCoordinate ?iyn .
                        ?ixn qudt:numericValue ?x ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?iyn qudt:numericValue ?iv ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .

                        OPTIONAL {
                            $this nltl:hasUpperIceWaterline ?uw .
                            ?uw nltl:hasWaterlineProfilePoint ?up .
                            ?up nltl:longitudinalPosition ?uxn ;
                                nltl:verticalCoordinate ?uyn .
                            ?uxn qudt:numericValue ?ux ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?uyn qudt:numericValue ?uv ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (?ux = ?x)
                        }
                    }
                    GROUP BY ?x
                    HAVING (?ucount = 0 || ?umin != ?umax || ?umax != ?imax)
                }
                UNION
                {
                    SELECT ?ux
                    WHERE {
                        {
                            SELECT ?ux
                                   (MAX(?uv) AS ?umax)
                                   (COUNT(?ip) AS ?icount)
                            WHERE {
                                $this nltl:hasUpperIceWaterline ?uw .
                                ?uw nltl:hasWaterlineProfilePoint ?up .
                                ?up nltl:longitudinalPosition ?uxn ;
                                    nltl:verticalCoordinate ?uyn .
                                ?uxn qudt:numericValue ?ux ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .
                                ?uyn qudt:numericValue ?uv ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .

                                OPTIONAL {
                                    $this nltl:hasIntendedIceOperatingWaterline ?iw .
                                    ?iw nltl:hasWaterlineProfilePoint ?ip .
                                    ?ip nltl:longitudinalPosition ?ixn .
                                    ?ixn qudt:numericValue ?ix ;
                                         qudt:unit <http://qudt.org/vocab/unit/M> .
                                    FILTER (?ix = ?ux)
                                }
                            }
                            GROUP BY ?ux
                            HAVING (?icount = 0)
                        }
                    }
                }
            }
        """
    ] .

gen:TRF011IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gen:TRF011ProfilePointShape
    ] .

gen:TRF011ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gen:TRF011LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gen:TRF011LengthQuantityShape
    ] .

gen:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
