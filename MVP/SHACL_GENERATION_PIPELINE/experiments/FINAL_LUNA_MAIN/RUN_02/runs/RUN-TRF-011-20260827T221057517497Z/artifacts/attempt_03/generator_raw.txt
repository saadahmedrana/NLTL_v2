<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The UIWL ordinates shall equal the maximum intended ice-operating waterline ordinate at every longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this ?x ?expectedMax ?actualMin ?actualMax
WHERE {
  {
    SELECT $this ?x
           (MAX(?expectedY) AS ?expectedMax)
           (MIN(?actualY) AS ?actualMin)
           (MAX(?actualY) AS ?actualMax)
    WHERE {
      {
        $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
        ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
        ?intendedPoint nltl:longitudinalPosition ?intendedLongitudinal .
        ?intendedLongitudinal qudt:numericValue ?x ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .
        ?intendedPoint nltl:verticalCoordinate ?intendedVertical .
        ?intendedVertical qudt:numericValue ?expectedY ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
      }
      UNION
      {
        $this nltl:hasUpperIceWaterline ?upperWaterline .
        ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
        ?upperPoint nltl:longitudinalPosition ?upperLongitudinal .
        ?upperLongitudinal qudt:numericValue ?x ;
                           qudt:unit <http://qudt.org/vocab/unit/M> .
        ?upperPoint nltl:verticalCoordinate ?upperVertical .
        ?upperVertical qudt:numericValue ?actualY ;
                       qudt:unit <http://qudt.org/vocab/unit/M> .
      }
    }
    GROUP BY $this ?x
  }
  FILTER (
    !BOUND(?expectedMax) ||
    !BOUND(?actualMin) ||
    !BOUND(?actualMax) ||
    ABS(?actualMin - ?expectedMax) >
      (0.000001 * ABS(?expectedMax) + 0.000000001) ||
    ABS(?actualMax - ?expectedMax) >
      (0.000001 * ABS(?expectedMax) + 0.000000001)
  )
}
""" ;
    ] .

gen:TRF_011_IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_ProfilePointShape
    ] .

gen:TRF_011_ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF_011_LongitudinalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF_011_VerticalQuantityShape
    ] .

gen:TRF_011_LongitudinalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF_011_VerticalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
