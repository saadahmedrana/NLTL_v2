<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF012ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF012IceWaterlineShape ;
        sh:message "An intended ice-operating waterline with profile points is required."
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF012IceWaterlineShape ;
        sh:message "A lower ice waterline with profile points is required."
    ] .

gs:TRF012IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gs:TRF012ProfilePointShape ;
        sh:message "The ice waterline must be represented by profile points."
    ] .

gs:TRF012ProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF012LengthQuantityShape ;
        sh:message "Each waterline profile point must have one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF012LengthQuantityShape ;
        sh:message "Each waterline profile point must have one vertical coordinate in metres."
    ] .

gs:TRF012LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qudt:M ;
        sh:message "The quantity must use metres as its unit."
    ] .
<END_SHACL>
