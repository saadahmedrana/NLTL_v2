<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF014ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:firstScheduledDryDockingDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasClassCertificate ;
        sh:class nltl:classCertificate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014ClassCertificateShape
    ] ;
    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:class nltl:iceDraughtRestrictionDocument ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape
    ] ;
    sh:sparql [
        sh:message "The required ice-draught marking condition, including the directional draught comparison and applicable dry-docking date, is not satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:assessmentDate ?assessmentDate ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summerNode ;
                      nltl:upperIceWaterlineDraught ?upperNode ;
                      nltl:hasClassCertificate ?certificate .

                ?summerNode qudt:numericValue ?summerValue ;
                            qudt:unit unit:M .
                ?upperNode qudt:numericValue ?upperValue ;
                           qudt:unit unit:M .
                ?certificate nltl:maximumIceClassDraughtAmidships ?maximumNode .
                ?maximumNode qudt:numericValue ?maximumValue ;
                             qudt:unit unit:M .

                FILTER (
                    (
                        ?constructionDate >= "2007-07-01"^^xsd:date &&
                        ?summerValue > ?upperValue &&
                        NOT EXISTS {
                            $this nltl:warningTrianglePresent true ;
                                  nltl:iceClassDraughtMarkPresent true ;
                                  nltl:iceClassDraughtMarkDraughtAmidships ?markNode .
                            ?markNode qudt:numericValue ?markValue ;
                                     qudt:unit unit:M .
                            FILTER (?markValue = ?maximumValue)
                        }
                    )
                    ||
                    (
                        ?constructionDate < "2007-07-01"^^xsd:date &&
                        ?upperValue < ?summerValue &&
                        NOT EXISTS {
                            $this nltl:warningTrianglePresent true ;
                                  nltl:iceClassDraughtMarkPresent true ;
                                  nltl:iceClassDraughtMarkDraughtAmidships ?markNode .
                            ?markNode qudt:numericValue ?markValue ;
                                     qudt:unit unit:M .
                            FILTER (?markValue = ?maximumValue)
                        } &&
                        (
                            NOT EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
                            }
                            ||
                            EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
                                FILTER (?dryDockingDate <= ?assessmentDate)
                            }
                        )
                    )
                )
            }
        """
    ] .
