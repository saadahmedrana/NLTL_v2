<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF015Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:intendedRouteSeaWaterSalinity ;
        sh:node gen:SalinityQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadingCalculationSeaWaterSalinity ;
        sh:node gen:SalinityQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF015NotNavigatingInIceShape
        gen:TRF015NavigatingInIceShape
    ) .

gen:SalinityQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:gramPerKilogramSalinityUnit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF015NotNavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF015NavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingTrim ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumPermittedIceTrim ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When navigating in ice, operating draught must not exceed the upper ice waterline draught and operating trim must not exceed the maximum permitted ice trim." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true .
                $this nltl:operatingDraught ?operatingDraughtQuantity .
                $this nltl:upperIceWaterlineDraught ?upperWaterlineQuantity .
                $this nltl:operatingTrim ?operatingTrimQuantity .
                $this nltl:maximumPermittedIceTrim ?maximumTrimQuantity .

                ?operatingDraughtQuantity qudt:numericValue ?operatingDraught .
                ?upperWaterlineQuantity qudt:numericValue ?upperWaterlineDraught .
                ?operatingTrimQuantity qudt:numericValue ?operatingTrim .
                ?maximumTrimQuantity qudt:numericValue ?maximumPermittedTrim .

                FILTER (
                    ?operatingDraught > ?upperWaterlineDraught ||
                    ?operatingTrim > ?maximumPermittedTrim
                )
            }
        """
    ] .
<END_SHACL>
