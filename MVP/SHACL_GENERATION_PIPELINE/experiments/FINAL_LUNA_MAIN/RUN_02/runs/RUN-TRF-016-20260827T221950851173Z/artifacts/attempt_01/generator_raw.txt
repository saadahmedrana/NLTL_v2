                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    WHERE {
                        $this nltl:draughtAmidships/qudt:numericValue ?draught ;
                              nltl:liwlDraughtAmidships/qudt:numericValue ?liwl .

                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:displacementAtUpperIceWaterline/qudt:numericValue ?selected .
                        FILTER NOT EXISTS {

                            FILTER (?candidate = ?selected)
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Forward draught must satisfy the UIWL displacement and level-ice-thickness formula." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:forwardDraught/qudt:numericValue ?forward ;
                              nltl:displacementAtUpperIceWaterline/qudt:numericValue ?displacement ;
                              nltl:levelIceThickness/qudt:numericValue ?thickness .
                        BIND (IF(
                            ((2.0 + (0.00025 * ?displacement)) * ?thickness) < (4.0 * ?thickness),
                            ((2.0 + (0.00025 * ?displacement)) * ?thickness),
                            (4.0 * ?thickness)
                        ) AS ?minimumForward)
                        FILTER (?forward < ?minimumForward)
                    }
                """
            ] ;

                              nltl:usedToReachLowerIceWaterline true .

                        }
                    }
                """
