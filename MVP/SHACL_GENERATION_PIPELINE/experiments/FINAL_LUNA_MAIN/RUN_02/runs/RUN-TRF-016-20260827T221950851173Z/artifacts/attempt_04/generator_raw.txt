<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-016-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "navigatingInIce must be true for the ice-navigation branch."
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:node [
                    sh:class nltl:iceWaterline
                ] ;
                sh:minCount 1 ;
                sh:message "A lower ice waterline is required when navigating in ice."
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:node [
                    sh:class nltl:iceWaterline
                ] ;
                sh:minCount 1 ;
                sh:message "An upper ice waterline is required when navigating in ice."
            ] ;
            sh:property [
                sh:path nltl:draughtAmidships ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "draughtAmidships must be supplied as a metre quantity."
            ] ;
            sh:property [
                sh:path nltl:liwlDraughtAmidships ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "liwlDraughtAmidships must be supplied as a metre quantity."
            ] ;
            sh:property [
                sh:path nltl:propellerHighestPointSubmerged ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The propeller highest point must be submerged when navigating in ice."
            ] ;
            sh:property [
                sh:path nltl:displacementAtUpperIceWaterline ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Displacement at the upper ice waterline must be supplied in metric tons."
            ] ;
            sh:property [
                sh:path nltl:levelIceThickness ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "levelIceThickness must be supplied as a metre quantity."
            ] ;
            sh:property [
                sh:path nltl:forwardDraught ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "forwardDraught must be supplied as a metre quantity."
            ] ;
            sh:sparql [
                sh:message "draughtAmidships must be at least the LIWL draught amidships." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#draughtAmidships> ?draughtNode ;
                              <https://w3id.org/nltl/vocab#liwlDraughtAmidships> ?liwlNode .
                        ?draughtNode <http://qudt.org/schema/qudt/numericValue> ?draught .
                        ?liwlNode <http://qudt.org/schema/qudt/numericValue> ?liwl .
                        FILTER (?draught < ?liwl)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "forwardDraught must be at least the applicable level-ice-thickness formula threshold." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#forwardDraught> ?forwardNode ;
                              <https://w3id.org/nltl/vocab#displacementAtUpperIceWaterline> ?displacementNode ;
                              <https://w3id.org/nltl/vocab#levelIceThickness> ?thicknessNode .
                        ?forwardNode <http://qudt.org/schema/qudt/numericValue> ?forward .
                        ?displacementNode <http://qudt.org/schema/qudt/numericValue> ?displacement .
                        ?thicknessNode <http://qudt.org/schema/qudt/numericValue> ?thickness .
                        FILTER (
                            ?forward <
                            IF(
                                ((2 + (0.00025 * ?displacement)) * ?thickness) <
                                (4 * ?thickness),
                                ((2 + (0.00025 * ?displacement)) * ?thickness),
                                (4 * ?thickness)
                            )
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Every ballast tank situated above the LIWL and needed to reach it must have freezing prevention." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#hasBallastTank> ?tank .
                        ?tank <https://w3id.org/nltl/vocab#situatedAboveLowerIceWaterline> ?above ;
                              <https://w3id.org/nltl/vocab#usedToReachLowerIceWaterline> ?needed .
                        FILTER (?above = true && ?needed = true)
                        FILTER NOT EXISTS {
                            ?tank <https://w3id.org/nltl/vocab#freezingPreventionPresent> true .
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
