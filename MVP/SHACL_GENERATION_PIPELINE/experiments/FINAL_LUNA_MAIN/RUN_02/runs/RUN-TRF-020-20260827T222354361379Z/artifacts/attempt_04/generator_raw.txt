        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;

    ] ;
    sh:property [

        sh:node [

                sh:path qudt:numericValue ;

    sh:property [

            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;

            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;

    sh:property [

        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;

            nltl:conventionalPropulsionSystem

        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;

    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;

        sh:node [

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date .

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?maximumPower .

                $this nltl:minimumRequiredPowerAtUpperIceWaterline ?upperPower .

                    ?maximumValue < ?upperValue

gen:TRF020NonApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "The non-applicable branch is invalid because the construction date and ice class satisfy the applicability condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date .
                $this nltl:iceClass ?class .
                FILTER (
                    ?date >= "2003-09-01"^^xsd:date
                    &&
                    ?class IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
            }
        """
    ] .
