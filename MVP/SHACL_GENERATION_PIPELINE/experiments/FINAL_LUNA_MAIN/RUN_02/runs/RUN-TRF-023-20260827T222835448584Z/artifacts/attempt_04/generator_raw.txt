<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in ( nltl:iceClassIb nltl:iceClassIc ) ;
                sh:minCount 1 ;
                sh:message "The ship must have ice class IB or IC for this applicability branch."
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRuleEdition1985 ;
                sh:minCount 1 ;
                sh:message "The 1985 ice class regulation edition must apply to retain ice class IB or IC."
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloW ;
                        sh:maxCount 1 ;
                        sh:minCount 1 ;
                        sh:message "Maximum continuous rating power must be expressed in kW."
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:requiredPowerUnder1985Rules ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloW ;
                        sh:maxCount 1 ;
                        sh:minCount 1 ;
                        sh:message "Required power under the 1985 rules must be expressed in kW."
                    ]
                ]
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:maximumContinuousRatingPower ?mcrNode .
                        ?mcrNode qudt:numericValue ?mcrValue .
                        $this nltl:requiredPowerUnder1985Rules ?requiredNode .
                        ?requiredNode qudt:numericValue ?requiredValue .
                        FILTER (?mcrValue >= ?requiredValue)
                    }
                """ ;
                sh:message "For applicable ice class IB or IC under the 1985 rules, maximum continuous rating power must be at least the required power."
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:iceClass ?iceClass .
                            FILTER (?iceClass IN (
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ))
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRuleEdition1985 .
                        }
                    }
                """ ;
                sh:message "The 1985 IB/IC power requirement is not applicable to this ship."
            ]
        ]
    ) .
<END_SHACL>
