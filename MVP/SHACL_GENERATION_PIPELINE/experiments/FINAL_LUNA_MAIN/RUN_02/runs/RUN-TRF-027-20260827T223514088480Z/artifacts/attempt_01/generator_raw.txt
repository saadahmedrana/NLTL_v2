        sh:message "For an applicable IA or IA Super ship, clampedHullGeometryTerm must equal ((L*T)/B^2)^3 clamped to the inclusive range 5 through 20, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date ;
                      nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:clampedHullGeometryTerm ?resultNode .

                ?lengthNode qudt:numericValue ?length ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?breadthNode qudt:numericValue ?breadth ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?resultNode qudt:numericValue ?reported ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .

                FILTER (?date < "2003-09-01"^^xsd:date)
                FILTER (?breadth != 0)

                BIND (((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth)) AS ?raw)
                BIND (
                    IF (?raw < 5, 5,
                        IF (?raw > 20, 20, ?raw)
                    ) AS ?expected
                )
                FILTER (
                    ABS(?reported - ?expected) >
                    ((0.000001 * ABS(?expected)) + 0.000000001)
                )
            }
        """
    ] .
