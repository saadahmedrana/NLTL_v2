gen:TRF027Shape

    sh:property [
        sh:path nltl:iceClass ;

        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
        ) ;
        sh:message "iceClass must contain exactly one permitted controlled ice-class value."
    ] ;
    sh:or (
        gen:TRF027ApplicableShape
        gen:TRF027IaAfterCutoffShape
        gen:TRF027IaSuperAfterCutoffShape
        gen:TRF027OtherIceClassShape
    ) .

    sh:property [
        sh:path nltl:coefficientC3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:coefficientC4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:coefficientC5 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:C5QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F1QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F4QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G1QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:clampedHullGeometryTerm ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ClampedGeometryQuantityShape
    ] ;
    sh:sparql [
        sh:message "clampedHullGeometryTerm must equal the clamped value of ((shipLength * iceClassDraught) / shipBreadth^2)^3 within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:clampedHullGeometryTerm ?resultNode .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .
                ?breadthNode qudt:numericValue ?breadth .
                ?resultNode qudt:numericValue ?result .
                FILTER (?breadth != 0)
                BIND (((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth))) AS ?raw)
                BIND (IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw)) AS ?expected)
                FILTER (ABS(?result - ?expected) >
                        ((0.000001 * ABS(?expected)) + 0.000001))
            }
        """
    ] .
