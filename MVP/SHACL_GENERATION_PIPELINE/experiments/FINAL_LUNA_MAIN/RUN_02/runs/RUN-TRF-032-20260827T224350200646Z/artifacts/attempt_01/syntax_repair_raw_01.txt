<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_032_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice strengthening status is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or ( _:branch1 _:branch2 _:branch3 ) .

_:branch1 sh:sparql [
    sh:message "The ship must be identified as ice strengthened for this non-applicable branch." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
        SELECT $this
        WHERE {
            $this nltl:iceStrengtheningStatus ?iceStatus .
            FILTER (?iceStatus != true)
        }
    """
] .

_:branch2 sh:sparql [
    sh:message "The regulation-derived scantling must not be less than the classification society required scantling when this branch is selected." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX qudt: <http://qudt.org/schema/qudt/>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
        SELECT $this
        WHERE {
            $this nltl:regulationRequiredScantling ?regulationQuantity .
            ?regulationQuantity qudt:numericValue ?regulationValue .
            $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
            ?classificationQuantity qudt:numericValue ?classificationValue .
            FILTER (?regulationValue < ?classificationValue)
        }
    """
] .

_:branch3
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When the regulation-derived scantling is less than the classification society requirement for a non-ice-strengthened ship, the selected design scantling must equal the classification society requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus ?iceStatus .
                $this nltl:regulationRequiredScantling ?regulationQuantity .
                ?regulationQuantity qudt:numericValue ?regulationValue .
                $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                ?classificationQuantity qudt:numericValue ?classificationValue .
                $this nltl:selectedDesignScantling ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                FILTER (
                    !(
                        ?iceStatus = false &&
                        ?regulationValue < ?classificationValue &&
                        ?selectedValue = ?classificationValue
                    )
                )
            }
        """
    ] .
<END_SHACL>
