<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF036ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] .

gs:TRF036ParameterSetShape a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The linked ship ice class must select the corresponding Table 4-1 level ice thickness limit and design ice load height." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                      nltl:iceClass ?iceClass .
                $this nltl:levelIceThickness ?levelQuantity ;
                       nltl:designIceLoadHeight ?heightQuantity .
                ?levelQuantity qudt:numericValue ?level .
                ?heightQuantity qudt:numericValue ?height .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper &&
                     ?level <= 1.0 &&
                     ?height = 0.35)
                    ||
                    (?iceClass = nltl:iceClassIa &&
                     ?level <= 0.8 &&
                     ?height = 0.30)
                    ||
                    (?iceClass = nltl:iceClassIb &&
                     ?level <= 0.6 &&
                     ?height = 0.25)
                    ||
                    (?iceClass = nltl:iceClassIc &&
                     ?level <= 0.4 &&
                     ?height = 0.22)
                )
            }
        """
    ] .
</END_SHACL>
