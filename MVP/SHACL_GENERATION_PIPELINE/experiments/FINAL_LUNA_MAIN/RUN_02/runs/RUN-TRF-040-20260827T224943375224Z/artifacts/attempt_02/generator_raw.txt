<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF040Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted ice-class value."
    ] ;

    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Spacing shall be supplied as a single metre-valued quantity."
    ] ;

    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Thickness shall be supplied as a single metre-valued quantity."
    ] ;

    sh:property [
        sh:path nltl:designOperationSpeedInIce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design operation speed in ice shall be supplied as a single knot-valued quantity."
    ] ;

    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sidescuttles shall not be situated in the ice belt."
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA Super, forefoot shell strengthening shall equal the bow-region equivalent status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER NOT EXISTS {
                    $this nltl:forefootShellStrengtheningStatus ?forefoot ;
                          nltl:bowRegionEquivalent ?bow .
                    FILTER (?forefoot = ?bow)
                }
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA Super or IA with open-water service speed of at least 18 knots, upper-bow strengthening shall equal the midbody equivalent status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:openWaterServiceSpeed ?speedValue .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )
                ?speedValue qudt:numericValue ?speed ;
                            qudt:unit unit:KN .
                FILTER (?speed >= 18)
                FILTER NOT EXISTS {
                    $this nltl:upperBowIceBeltStrengtheningStatus ?upperBow ;
                          nltl:midbodyEquivalent ?midbody .
                    FILTER (?upperBow = ?midbody)
                }
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Open-water service speed is required for IA Super or IA ships to determine the upper-bow strengthening condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )
                FILTER NOT EXISTS {
                    $this nltl:openWaterServiceSpeed ?speedValue .
                }
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the weather deck is below the upper limit of the ice belt, bulwark and freeing-port strength shall meet the ice-belt shell-strength requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                FILTER NOT EXISTS {
                    $this nltl:bulwarkStrengthStatus true .
                }
                FILTER NOT EXISTS {
                    $this nltl:freeingPortStrengthStatus true .
                }
            }
        """
    ] .

</BEGIN_SHACL>
