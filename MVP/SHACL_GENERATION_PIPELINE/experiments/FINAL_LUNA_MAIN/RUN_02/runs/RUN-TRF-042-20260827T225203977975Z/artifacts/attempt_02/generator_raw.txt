<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF042PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "designIceLoadHeight must be a single metre quantity." 
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "spacing must be a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "longitudinalPlatingFactorF2 must be a single dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "corrosionAbrasionAddition must be a single millimetre quantity."
    ] ;
    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The h/s ratio must have positive designIceLoadHeight and spacing, and must be no greater than 1.8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                FILTER (?s <= 0 || ?h <= 0 || ?h > (1.8 * ?s))
            }
        """
    ] ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable piecewise formula within the specified tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?f2Node .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?f2Node qudt:numericValue ?observed .
                FILTER (?s > 0 && ?h > 0 && ?h <= (1.8 * ?s))
                BIND (?h / ?s AS ?ratio)
                BIND (
                    IF(
                        ?ratio <= 1,
                        (0.4 / ?ratio) + 0.6,
                        1.4 - (0.4 * ?ratio)
                    ) AS ?expected
                )
                BIND (
                    IF(
                        (0.000001 * ABS(?expected)) > 0.000000000001,
                        (0.000001 * ABS(?expected)),
                        0.000000000001
                    ) AS ?tolerance
                )
                FILTER (ABS(?observed - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "corrosionAbrasionAddition must not exceed the normal 2 mm increment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc .
                FILTER (?tc > 2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A corrosionAbrasionAddition below 2 mm requires a maintained special surface coating and approved material status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc .
                FILTER (?tc < 2)
                FILTER NOT EXISTS {
                    $this nltl:specialSurfaceCoatingMaintained true ;
                          nltl:materialApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
