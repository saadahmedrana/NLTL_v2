<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gs:TRF043TerminationCaseShape
    ] .

gs:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:class nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MilliM
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the extension is no more than 250 mm, termination at the adjacent boundary must be permitted." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
    ?quantity qudt:numericValue ?value .
    FILTER (?value <= 250)
    FILTER NOT EXISTS {
        $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true^^xsd:boolean .
    }
}
"""
    ] .
<END_SHACL>
