<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF047MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A main frame covered by this requirement must be ice strengthened below the ice belt."
    ] ;
    sh:property g:TRF047StrengthenedPartPropertyShape ;
    sh:sparql g:TRF047LowerEndConstraint .

g:TRF047IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:property g:TRF047StrengthenedPartPropertyShape ;
    sh:sparql g:TRF047LowerEndConstraint .

g:TRF047StrengthenedPartPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:hasStrengthenedPart ;
    sh:minCount 1 ;
    sh:node g:TRF047StrengthenedPartShape .

g:TRF047StrengthenedPartShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "The strengthened part must identify its lower end."
    ] .

g:TRF047LowerEndConstraint
    a sh:SPARQLConstraint ;
    sh:message "Each lower end must be attached to a permitted deck, tank boundary plating, tank top, or ice stringer, or an intermediate ice frame must satisfy the permitted termination and horizontal-member alternative with the same scantlings as the main frames." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

        SELECT $this
        WHERE {
            $this nltl:hasStrengthenedPart ?strengthenedPart .
            ?strengthenedPart nltl:hasLowerEnd ?lowerEnd .

            FILTER NOT EXISTS {
                ?lowerEnd nltl:hasAttachedSupportingStructure ?attachedStructure .
                ?attachedStructure rdf:type ?attachedType .
                FILTER (
                    ?attachedType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    )
                )
            }

            FILTER NOT EXISTS {
                $this rdf:type nltl:intermediateIceFrame .
                $this nltl:sameScantlingsAsMainFrame true .

                ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?supportingStructure .
                ?supportingStructure rdf:type nltl:supportingStructure .
                ?supportingStructure nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .

                ?lowerEnd nltl:hasHorizontalConnectionMember ?horizontalMember .
                ?horizontalMember rdf:type nltl:horizontalConnectionMember .
                ?horizontalMember nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                ?adjacentMainFrame rdf:type nltl:mainFrame .
            }
        }
    """ .
<END_SHACL>
