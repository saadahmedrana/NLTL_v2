nltlg:TRF049_BracketThicknessComparison
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each installed bracket must have at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                {
                    FILTER NOT EXISTS {
                        $this nltl:frameWebThickness ?frameThickness .
                        ?frameThickness qudt:numericValue ?frameValue .
                        ?frameThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    }
                }
                UNION
                {
                    ?bracket nltl:bracketThickness ?bracketThickness .
                    ?bracketThickness qudt:numericValue ?bracketValue .
                    ?bracketThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    $this nltl:frameWebThickness ?frameThickness .
                    ?frameThickness qudt:numericValue ?frameValue .
                    ?frameThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (?bracketValue < ?frameValue)
                }
            }
        """
    ] .
