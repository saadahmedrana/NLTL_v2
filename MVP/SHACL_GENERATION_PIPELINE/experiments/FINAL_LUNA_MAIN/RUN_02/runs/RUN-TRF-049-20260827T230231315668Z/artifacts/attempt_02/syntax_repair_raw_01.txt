gs:TRF049BracketThicknessComparison
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachment ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bracket thickness must be at least the web thickness of its frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?frame nltl:hasFrameAttachment $this .
                ?frame nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                $this nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

gs:TRF049LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting web frame of a longitudinal frame must be linked to an attachment of that frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this nltl:hasSupportingWebFrame ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting bulkhead of a longitudinal frame must be linked to an attachment of that frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this nltl:hasSupportingBulkhead ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting web frame must point back to an attachment of the longitudinal frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this nltl:hasSupportingWebFrame ?support .
