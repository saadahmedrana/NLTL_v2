gs:TRF049FrameInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gs:TRF049AttachmentShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:minInclusive 2
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A frame with an installed connection bracket must provide frame web thickness in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?thickness .
                    ?thickness qudt:numericValue ?value ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (datatype(?frameValue) = xsd:decimal)
                FILTER (datatype(?bracketValue) = xsd:decimal)
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this

                $this nltl:hasAttachedSupportingStructure ?support .
                FILTER NOT EXISTS {
                    ?support nltl:hasSupportingStructureAttachment $this .
                }
            }
        """
    ] .
