gs:TRF049ConnectionBracketShape
    a sh:NodeShape ;
    sh:targetClass nltl:connectionBracket ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MilliM
            ]
        ] ;
        sh:message "An installed bracket must have a millimetre bracket thickness."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket must be stiffened against buckling."
    ] ;
    sh:sparql [
        sh:message "A bracket must have at least the same thickness as the frame web plate, and the frame web thickness must be present in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?frame nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket $this .
                FILTER NOT EXISTS {
                    ?frame nltl:frameWebThickness ?frameThickness .
                    ?frameThickness qudt:numericValue ?frameValue ;
                                     qudt:unit qudt:MilliM .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must not be less than frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?frame nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket $this .
                ?frame nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                                 qudt:unit qudt:MilliM .
                $this nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                  qudt:unit qudt:MilliM .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

gs:TRF049LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:sparql [
        sh:message "Within the ice-strengthened area, every listed supporting web frame or bulkhead must be connected through a bracketed attachment in both relationship directions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                $this nltl:withinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:hasConnectionBracket ?bracket .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] .
