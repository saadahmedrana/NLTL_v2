<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF052FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasAntitrippingSupportRegion ;
        sh:in (
            nltl:allRegions
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
            nltl:otherIceStrengthenedAreaValue
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame does not satisfy the applicable antitripping support spacing or scope requirement, and no demonstrated equivalent support calculation is recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:frameAsymmetrical ?asym ;
                      nltl:frameWebAngleToShell ?angleNode ;
                      nltl:frameSpan ?spanNode ;
                      nltl:iceClass ?iceClass .

                ?angleNode qudt:numericValue ?angle .
                ?spanNode qudt:numericValue ?span .

                FILTER (
                    NOT EXISTS {
                        $this nltl:equivalentSupportStatusByDirectCalculation true
                    }
                    &&
                    (
                        (
                            (?asym = true || ?angle < 90)
                            &&
                            (
                                NOT EXISTS {
                                    $this nltl:antitrippingSupportSpacing ?spacingNode .
                                    ?spacingNode qudt:numericValue ?spacing
                                }
                                ||
                                EXISTS {
                                    $this nltl:antitrippingSupportSpacing ?spacingNode2 .
                                    ?spacingNode2 qudt:numericValue ?spacing2 .
                                    FILTER (?spacing2 > 1300)
                                }
                            )
                        )
                        ||
                        (
                            ?span > 4
                            &&
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:allRegions
                            }
                        )
                        ||
                        (
                            ?span <= 4
                            &&
                            (
                                (
                                    ?iceClass = nltl:iceClassIaSuper
                                    &&
                                    (
                                        NOT EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion nltl:allRegions
                                        }
                                        ||
                                        EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion ?unexpectedSuperRegion .
                                            FILTER (
                                                ?unexpectedSuperRegion != nltl:allRegions
                                            )
                                        }
                                    )
                                )
                                ||
                                (
                                    ?iceClass = nltl:iceClassIa
                                    &&
                                    (
                                        NOT EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                        }
                                        ||
                                        NOT EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                                        }
                                        ||
                                        EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion ?unexpectedIaRegion .
                                            FILTER (
                                                ?unexpectedIaRegion != nltl:bowRegion
                                                &&
                                                ?unexpectedIaRegion != nltl:midbodyRegion
                                            )
                                        }
                                    )
                                )
                                ||
                                (
                                    (
                                        ?iceClass = nltl:iceClassIb
                                        ||
                                        ?iceClass = nltl:iceClassIc
                                    )
                                    &&
                                    (
                                        NOT EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                        }
                                        ||
                                        EXISTS {
                                            $this nltl:hasAntitrippingSupportRegion ?unexpectedIbIcRegion .
                                            FILTER (
                                                ?unexpectedIbIcRegion != nltl:bowRegion
                                            )
                                        }
                                    )
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
