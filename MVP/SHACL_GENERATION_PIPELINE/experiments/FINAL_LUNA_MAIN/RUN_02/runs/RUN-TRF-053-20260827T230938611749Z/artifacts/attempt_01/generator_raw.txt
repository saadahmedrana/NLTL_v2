<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF-053-QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must have one unit."
    ] .

gen:TRF-053
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:TRF-053-IcePressureShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice pressure is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:TRF-053-DesignIceLoadHeightShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design ice load height is required in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:TRF-053-SpanShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stringer span is required in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:TRF-053-FrameMomentFactorShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame moment factor m is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:TRF-053-YieldStrengthShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Yield strength is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:TRF-053-DesignLineLoadShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculated design line load is required in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:TRF-053-SectionModulusShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculated section modulus is required in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:TRF-053-RequiredShearAreaShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculated required shear area is required in cm2."
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, or required shear area does not satisfy the prescribed calculation; derived decimal results use a relative tolerance of 1e-6 with an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureValue ;
                      nltl:designIceLoadHeight ?heightValue ;
                      nltl:span ?spanValue ;
                      nltl:frameMomentFactorM ?momentFactorValue ;
                      nltl:yieldStrength ?yieldStrengthValue ;
                      nltl:designLineLoad ?lineLoadValue ;
                      nltl:sectionModulus ?sectionModulusValue ;
                      nltl:requiredShearArea ?shearAreaValue .

                ?pressureValue qudt:numericValue ?pressure .
                ?heightValue qudt:numericValue ?height .
                ?spanValue qudt:numericValue ?span .
                ?momentFactorValue qudt:numericValue ?momentFactor .
                ?yieldStrengthValue qudt:numericValue ?yieldStrength .
                ?lineLoadValue qudt:numericValue ?lineLoad .
                ?sectionModulusValue qudt:numericValue ?sectionModulus .
                ?shearAreaValue qudt:numericValue ?shearArea .

                BIND(IF((?pressure * ?height) > 0.15,
                        (?pressure * ?height),
                        0.15) AS ?expectedLineLoad)
                BIND((0.9 * 1.8 * ?expectedLineLoad * ?span * ?span /
                      (?momentFactor * ?yieldStrength) * 1000000)
                     AS ?expectedSectionModulus)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 *
                      ?expectedLineLoad * ?span /
                      (2 * ?yieldStrength) * 10000)
                     AS ?expectedShearArea)

                FILTER(
                    ABS(?lineLoad - ?expectedLineLoad) >
                        (0.000001 * ABS(?expectedLineLoad) + 0.000000001)
                    ||
                    ABS(?sectionModulus - ?expectedSectionModulus) >
                        (0.000001 * ABS(?expectedSectionModulus) + 0.000000001)
                    ||
                    ABS(?shearArea - ?expectedShearArea) >
                        (0.000001 * ABS(?expectedShearArea) + 0.000000001)
                )
            }
        """
    ] .
