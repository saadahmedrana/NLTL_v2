@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "Each ice stringer must provide one ice pressure quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each ice stringer must provide one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each ice stringer must provide one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:message "Each ice stringer must provide one dimensionless frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "Each ice stringer must provide one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForcePerLengthQuantityShape ;
        sh:message "The design line load must be reported in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:message "The section modulus must be reported in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape ;
        sh:message "The required shear area must be reported in cm2."
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area do not satisfy the TRF-053 formulas; the line-load result is converted from MN/m to kN/m for designLineLoad." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?lineNode qudt:numericValue ?line .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND((?p * ?h) AS ?product)
                BIND(IF(?product < 0.15, 0.15, ?product) AS ?lineMN)
                BIND((?lineMN * 1000) AS ?expectedLine)
                BIND((0.9 * 1.8 * ?lineMN * ?l * ?l /
                      (?m * ?sigma) * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?lineMN * ?l /
                      (2 * ?sigma) * 10000) AS ?expectedA)

                FILTER (
                    ABS(?line - ?expectedLine) >
                        (ABS(?expectedLine) * 0.000001 + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                        (ABS(?expectedZ) * 0.000001 + 0.000000001)
                    ||
                    ABS(?a - ?expectedA) >
                        (ABS(?expectedA) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
