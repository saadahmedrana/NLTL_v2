g:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node g:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node g:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node g:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node g:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node g:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node g:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node g:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node g:ForcePerLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node g:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node g:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node g:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy the TRF-054 formulas within a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?loadNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?spanNode qudt:numericValue ?span .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?loadNode qudt:numericValue ?reportedLoad .
                ?factorNode qudt:numericValue ?reportedFactor .
                ?zNode qudt:numericValue ?reportedZ .
                ?aNode qudt:numericValue ?reportedA .

                BIND((?p * ?h) AS ?rawLoadMn)
                BIND(IF(?rawLoadMn < 0.15, 0.15, ?rawLoadMn) AS ?expectedLoadMn)
                BIND(?expectedLoadMn * 1000 AS ?expectedLoadKn)
                BIND(1 - (?hs / ?ls) AS ?expectedFactor)
                BIND((0.80 * 1.8 * ?expectedLoadMn * ?ls * ?ls /
                      (?m * ?sigma) * ?expectedFactor * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?expectedLoadMn * ?ls /
                      (2 * ?sigma) * ?expectedFactor * 10000) AS ?expectedA)

                BIND(IF(?expectedLoadKn < 0, -?expectedLoadKn, ?expectedLoadKn) AS ?absLoad)
                BIND(IF(?expectedFactor < 0, -?expectedFactor, ?expectedFactor) AS ?absFactor)
                BIND(IF(?expectedZ < 0, -?expectedZ, ?expectedZ) AS ?absZ)
                BIND(IF(?expectedA < 0, -?expectedA, ?expectedA) AS ?absA)

                BIND(IF(?absLoad > 0.000000001, ?absLoad * 0.000001, 0.000000001) AS ?tolLoad)
                BIND(IF(?absFactor > 0.000000001, ?absFactor * 0.000001, 0.000000001) AS ?tolFactor)
                BIND(IF(?absZ > 0.000000001, ?absZ * 0.000001, 0.000000001) AS ?tolZ)
                BIND(IF(?absA > 0.000000001, ?absA * 0.000001, 0.000000001) AS ?tolA)

                FILTER(
                    ?reportedLoad < (?expectedLoadKn - ?tolLoad) ||
                    ?reportedLoad > (?expectedLoadKn + ?tolLoad) ||
                    ?reportedFactor < (?expectedFactor - ?tolFactor) ||
                    ?reportedFactor > (?expectedFactor + ?tolFactor) ||
                    ?reportedZ < (?expectedZ - ?tolZ) ||
                    ?reportedZ > (?expectedZ + ?tolZ) ||
                    ?reportedA < (?expectedA - ?tolA) ||
                    ?reportedA > (?expectedA + ?tolA)
                )
            }
        """
    ] .
