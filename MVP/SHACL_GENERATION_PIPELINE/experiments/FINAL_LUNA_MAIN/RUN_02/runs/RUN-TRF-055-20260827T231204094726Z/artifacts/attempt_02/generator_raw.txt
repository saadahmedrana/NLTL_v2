        sh:maxCount 1

gen:TRF055NarrowDeckStripShape
    a sh:NodeShape ;

        sh:path nltl:abreastOfHatch ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The narrow deck strip must be abreast of a hatch."
    ] ;

    sh:property [
        sh:path nltl:servesAsIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The narrow deck strip must serve as an ice stringer."
    ] ;

    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;

    ] ;

    sh:property [
        sh:path nltl:actualShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055ShearAreaQuantityShape
    ] ;

    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055ShearAreaQuantityShape
    ] ;

    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055SectionModulusQuantityShape
    ] ;

    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:evidenceState
    ] ;

    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055LineLoadQuantityShape
    ] ;

    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055LengthQuantityShape
    ] ;

    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055LengthQuantityShape
    ] ;

    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:or (
            [
                sh:hasValue true ;
                sh:property [
                    sh:path nltl:permittedReducedLineLoad ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node [
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minCount 1 ;
                            sh:maxCount 1 ;
                            sh:minInclusive 0.10 ;
                            sh:maxExclusive 0.15
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M> ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ] ;
                sh:property [
                    sh:path nltl:reducedLineLoadApprovedByClassificationSociety ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
            ]
            [
                sh:hasValue false
            ]
        )
    ] ;

    sh:sparql [
        sh:message "The actual section modulus and shear area must meet their respective required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus/qudt:numericValue ?actualZ .
                $this nltl:requiredSectionModulus/qudt:numericValue ?requiredZ .
                $this nltl:actualShearArea/qudt:numericValue ?actualA .
                $this nltl:requiredShearArea/qudt:numericValue ?requiredA .
                FILTER (?actualZ < ?requiredZ || ?actualA < ?requiredA)
            }
        """
    ] .

gen:TRF055VeryLongHatchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reducedLineLoadApprovedByClassificationSociety ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:permittedReducedLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a very long hatch, the permitted reduced line load must represent the design line load multiplied by the hatch opening length within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designLineLoad/qudt:numericValue ?p .
                $this nltl:hatchOpeningLength/qudt:numericValue ?h .
                $this nltl:permittedReducedLineLoad/qudt:numericValue ?reported .
                FILTER (?reported >= 0.10 && ?reported < 0.15)
                BIND (?p * ?h AS ?expected)
                FILTER (
                    ABS(?reported - ?expected) >
                    (ABS(?expected) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .
