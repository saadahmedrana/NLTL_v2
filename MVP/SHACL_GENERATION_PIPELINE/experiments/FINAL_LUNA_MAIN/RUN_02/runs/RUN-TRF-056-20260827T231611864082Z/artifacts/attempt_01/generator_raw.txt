<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF056ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node [
            sh:class nltl:weatherdeckHatch ;
            sh:property [
                sh:path nltl:hatchOpeningLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:or (
                [
                    sh:sparql [
                        sh:message "A hatch opening longer than half the ship breadth requires deflection representation and both hatch design evidence links." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>

                            SELECT $this
                            WHERE {
                                ?ship nltl:hasWeatherdeckHatch $this .
                                $this nltl:hatchOpeningLength/qudt:numericValue ?openingLength .
                                ?ship nltl:shipBreadth/qudt:numericValue ?breadth .
                                FILTER (?openingLength > (?breadth / 2))
                            }
                        """
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shipSideDeflection ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:hasHatchCoverDesignEvidence ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:hatchCoverDesignEvidence
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:hasHatchFittingDesignEvidence ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:hatchFittingDesignEvidence
                        ]
                    ] ;
                    sh:sparql [
                        sh:message "The conditional hatch requirement applies only when the opening length is greater than half the ship breadth." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>

                            SELECT $this
                            WHERE {
                                ?ship nltl:hasWeatherdeckHatch $this .
                                $this nltl:hatchOpeningLength/qudt:numericValue ?openingLength .
                                ?ship nltl:shipBreadth/qudt:numericValue ?breadth .
                                FILTER (!(?openingLength > (?breadth / 2)))
                            }
                        """
                    ]
                ]
            )
        ]
    ] ;
    sh:sparql [
        sh:message "A ship linked to a weatherdeck hatch must provide ship breadth to establish hatch-opening applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                FILTER NOT EXISTS {
                    $this nltl:shipBreadth ?breadthQuantity .
                    ?breadthQuantity qudt:numericValue ?breadthValue .
                }
            }
        """
    ] .
<END_SHACL>
