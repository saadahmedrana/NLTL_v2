<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF056ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node gen:TRF056WeatherdeckHatchShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a hatch opening longer than half the ship breadth, ship-side deflection and both hatch-cover and hatch-fitting design evidence must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth .
                $this nltl:hasWeatherdeckHatch ?hatch .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?opening .
                FILTER (
                    ?opening > (?breadth / 2)
                    &&
                    (
                        NOT EXISTS {
                            ?hatch nltl:shipSideDeflection ?deflection
                        }
                        ||
                        NOT EXISTS {
                            ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence
                        }
                        ||
                        NOT EXISTS {
                            ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence
                        }
                    )
                )
            }
        """
    ] .

gen:TRF056WeatherdeckHatchShape a sh:NodeShape ;
    sh:class nltl:weatherdeckHatch ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shipSideDeflection ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchCoverDesignEvidence ;
        sh:node [
            sh:class nltl:hatchCoverDesignEvidence
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchFittingDesignEvidence ;
        sh:node [
            sh:class nltl:hatchFittingDesignEvidence
        ]
    ] .

gen:TRF056HatchCoverEvidenceShape a sh:NodeShape ;
    sh:targetClass nltl:hatchCoverDesignEvidence ;
    sh:class nltl:documentEvidence .

gen:TRF056HatchFittingEvidenceShape a sh:NodeShape ;
    sh:targetClass nltl:hatchFittingDesignEvidence ;
    sh:class nltl:documentEvidence .
<END_SHACL>
