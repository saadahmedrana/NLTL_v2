@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF058Shape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:or (
        gen:TRF058NotApplicable
        gen:TRF058Applicable
    ) .

gen:TRF058NotApplicable
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The supportedStringerOutsideIceBelt applicability value must be false when the adjusted ice-load calculation is not applicable."
    ] .

gen:TRF058Applicable
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The supportedStringerOutsideIceBelt applicability value must be true for this calculation branch."
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF058ForceQuantity
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF058HeightQuantity
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF058SpanQuantity
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF058ForceQuantity
    ] ;
    sh:sparql [
        sh:message "adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true ;
                      nltl:iceLoadForce ?forceQuantity ;
                      nltl:stringerOutsideIceBeltHeight ?heightQuantity ;
                      nltl:stringerSpan ?spanQuantity ;
                      nltl:adjustedIceLoadForce ?adjustedQuantity .

                ?forceQuantity <http://qudt.org/schema/qudt/numericValue> ?force .
                ?heightQuantity <http://qudt.org/schema/qudt/numericValue> ?height .
                ?spanQuantity <http://qudt.org/schema/qudt/numericValue> ?span .
                ?adjustedQuantity <http://qudt.org/schema/qudt/numericValue> ?adjusted .

                FILTER (?span != 0)
                BIND (?force * (1 - (?height / ?span)) AS ?expected)
                BIND (ABS(?expected) * 0.000001 AS ?relativeTolerance)
                BIND (IF(?relativeTolerance > 0.000000001, ?relativeTolerance, 0.000000001) AS ?tolerance)
                FILTER (ABS(?adjusted - ?expected) > ?tolerance)
            }
        """
    ] .
