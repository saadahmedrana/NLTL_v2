<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix gen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one propeller count to establish applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minInclusive 2 ;
                sh:maxInclusive 3
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:sidePropellerStrengtheningEnvelope ;
                sh:minCount 1 ;
                sh:node gen:TRF063StrengtheningEnvelopeShape ;
                sh:message "A twin- or triple-screw ship must provide a side-propeller strengthening envelope."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:maxInclusive 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minInclusive 4
            ]
        ]
    ) .

gen:TRF063StrengtheningEnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The strengthening envelope boundary must be the tank top."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:node gen:TRF063LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The forward strengthening extent must be at least 1.5 metres."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:node gen:TRF063LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The aft strengthening extent must be at least 1.5 metres."
    ] .

gen:TRF063LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
