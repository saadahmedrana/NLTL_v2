<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable propulsion-machinery configuration, the required design-condition record and maximum ice torque input must be present with the prescribed values and unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    &&
                    ?propellerType IN ("open"^^xsd:string, "ducted"^^xsd:string)
                    &&
                    ?pitchType IN ("controllable"^^xsd:string, "fixed"^^xsd:string)
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:hasDesignCondition ?designCondition .
                        ?designCondition a nltl:designCondition ;
                            nltl:normalOperationalConditions true ;
                            nltl:normalServiceLifePropellerIceLoadCondition true ;
                            nltl:fixedPitchPropellerReversalLoadIncluded true ;
                            nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                            nltl:radialIceEntryOutsideLoadModel true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:maximumIceTorque ?torque .
                        ?torque a qudt:QuantityValue ;
                            qudt:numericValue ?numericValue ;
                            qudt:unit unit:KiloN-M .
                        FILTER (datatype(?numericValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumIceTorque ?torqueWithBadUnit .
                        ?torqueWithBadUnit qudt:unit ?badUnit .
                        FILTER (?badUnit != unit:KiloN-M)
                    }
                )
            }
        """
    ] .
<END_SHACL>
