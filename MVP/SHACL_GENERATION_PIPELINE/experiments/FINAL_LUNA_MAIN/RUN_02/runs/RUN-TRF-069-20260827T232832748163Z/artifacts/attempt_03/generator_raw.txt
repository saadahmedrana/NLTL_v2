<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

<urn:nltl:generated-shape:TRF_069_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:node [
            sh:class nltl:designCondition
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable propulsion-machinery configuration, the maximum ice torque and required design-condition scope must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumIceTorque ?torque .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDesignCondition ?condition .
                        ?condition nltl:normalOperationalConditions true ;
                                   nltl:normalServiceLifePropellerIceLoadCondition true ;
                                   nltl:fixedPitchPropellerReversalLoadIncluded true ;
                                   nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                                   nltl:radialIceEntryOutsideLoadModel true .
                    }
                )
            }
        """
    ] .
<END_SHACL>
