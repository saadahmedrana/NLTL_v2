    a sh:NodeShape ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:hasValue "azimuthing" ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF070KiloNewtonQuantity
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF070KiloNewtonQuantity
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node gs:TRF070MegaPascalQuantity
    ] ;
    sh:property [
        sh:path nltl:serviceLife ;
        sh:minCount 1 ;
        sh:node gs:TRF070YearQuantity
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node gs:TRF070AzimuthingDesignCondition
    ] ;
    sh:sparql [
        sh:message "The azimuthing thruster-body ice-interaction load must equal the expected load for the ship's service life." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:thrusterType "azimuthing" .
                FILTER NOT EXISTS {
                    ?this nltl:thrusterBodyIceInteractionLoad ?bodyLoad .
                    ?bodyLoad qudt:numericValue ?bodyValue ;
                              qudt:unit ?bodyUnit .
                    ?this nltl:hasDesignCondition ?condition .
                    ?condition nltl:designConditionExpectedLoad ?expectedLoad .
                    ?expectedLoad qudt:numericValue ?expectedValue ;
                                  qudt:unit ?expectedUnit .
                    FILTER (?bodyUnit = ?expectedUnit)
                    FILTER (?bodyValue = ?expectedValue)
                }
            }
        """
    ] .
