<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice class to determine applicability of the propeller-submersion rule."
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ballast-condition propeller top depth must be provided as a metre quantity."
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The required propeller submersion depth hi must be provided as a metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IB or IC, when the ballast-condition propeller top depth is less than hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
                ?depthQuantity qudt:numericValue ?depth .

                $this nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
                ?hiQuantity qudt:numericValue ?hi .

                FILTER (xsd:decimal(?depth) < xsd:decimal(?hi))

                FILTER NOT EXISTS {
                    $this nltl:propulsionSystemDesignedForIceClass ?designedClass .
                    FILTER (?designedClass = nltl:iceClassIa)
                }
            }
        """
    ] .
<END_SHACL>
