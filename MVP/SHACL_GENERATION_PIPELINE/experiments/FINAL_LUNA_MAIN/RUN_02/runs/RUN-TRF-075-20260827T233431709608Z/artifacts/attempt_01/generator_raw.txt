<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:ForceQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

nltlgen:TRF075Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node [
            sh:class nltl:propellerBladeLoadCase ;
            sh:property [
                sh:path nltl:propellerBladeLoadCaseBlade ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:propellerBlade
            ] ;
            sh:property [
                sh:path nltl:propellerBladeLoadCaseIceBlock ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:iceBlock
            ] ;
            sh:xone (
                [
                    sh:path nltl:backwardBladeForce ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node nltlgen:ForceQuantityShape
                ]
                [
                    sh:path nltl:forwardBladeForce ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node nltlgen:ForceQuantityShape
                ]
            ) ;
            sh:message "Each propeller blade load case must link to one blade and one ice block and specify exactly one blade force direction."
        ]
    ] .

<END_SHACL>
