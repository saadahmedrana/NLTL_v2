        sh:node g:TableLookupCaseShape ;
        sh:minCount 2
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:node g:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;

    sh:property [

        sh:maxCount 1

    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The ship must provide both the Table 6-6 and Table 6-7 lookup cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?table6 .
                        ?table6 nltl:tableReference nltl:traficomTable6Dash6 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?table7 .
                        ?table7 nltl:tableReference nltl:traficomTable6Dash7 .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 * k2 * k3 * N_class * n_n within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qv: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?table6 ;
                      nltl:hasTableLookupCase ?table7 ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?reported .

                ?table6 nltl:tableReference nltl:traficomTable6Dash6 ;
                         nltl:selectedIceClassCycleCount ?nClassNode .
                ?table7 nltl:tableReference nltl:traficomTable6Dash7 ;

                ?k2Node qv:numericValue ?k2 .
                ?k3Node qv:numericValue ?k3 .
                ?nClassNode qv:numericValue ?nClass .
                ?nnNode qv:numericValue ?nn .
                BIND ((?k1 * ?k2 * ?k3 * ?nClass * ?nn) AS ?expected)
                BIND ((?reported - ?expected) AS ?diff)
                FILTER (
                    IF(?diff >= 0, ?diff, -?diff)
                    >

                        0.000001 *
                        IF(?expected >= 0, ?expected, -?expected)
                        + 0.000000001
                    )
                )
            }
        """
