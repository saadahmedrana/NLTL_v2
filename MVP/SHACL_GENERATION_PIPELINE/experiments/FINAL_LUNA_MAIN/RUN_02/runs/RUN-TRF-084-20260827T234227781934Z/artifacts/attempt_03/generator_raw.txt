@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:pitchValueKnown ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrBollard ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:pitchValueKnown ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue false ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrBollard ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrFreeRunning ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:sparql [
                        a sh:SPARQLConstraint ;
                        sh:message "For a CP propeller with unknown pitch value, the bollard pitch must equal 0.7 times the free-running pitch within a relative tolerance of 1e-6 and an absolute fallback of 1e-6." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                            SELECT $this
                            WHERE {
                                $this nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
                                       nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                                ?freeQuantity qudt:numericValue ?free ;
                                              qudt:unit <http://qudt.org/vocab/unit/M> .
                                ?bollardQuantity qudt:numericValue ?observed ;
                                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                                BIND((0.7 * ?free) AS ?expected)
                                BIND(
                                    IF(
                                        ?expected >= 0,
                                        (0.000001 * ?expected) + 0.000001,
                                        (0 - (0.000001 * ?expected)) + 0.000001
                                    ) AS ?tolerance
                                )
                                FILTER(
                                    ?observed < (?expected - ?tolerance) ||
                                    ?observed > (?expected + ?tolerance)
                                )
                            }
                        """
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The conditional CP-propeller requirement is not applicable when propeller type is not CP." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT $this
                    WHERE {
                        $this nltl:propellerType ?type .
                        FILTER(?type = "CP")
                    }
                """
            ]
        ]
    ] .
