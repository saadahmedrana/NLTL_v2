<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF088Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumEngineTorqueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue true
            ]
            [
                sh:hasValue false
            ]
        ]
    ] ;
    sh:or (
        gen:TRF088KnownBranch
        gen:TRF088UnknownBranch
    ) .

gen:TRF088KnownBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:maximumEngineTorqueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:TRF088UnknownBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:maximumEngineTorqueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:maximumEngineTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF088TorqueQuantity
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:TRF088TableLookupCase
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When maximum engine torque is unknown, the reported torque must equal the numeric value selected from TRAFICOM Table 6-10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumEngineTorque ?torqueNode .
                ?torqueNode qudt:numericValue ?reportedValue .
                $this nltl:hasTableLookupCase ?lookupCase .
                ?lookupCase nltl:tableReference nltl:traficomTable6Dash10 .
                nltl:traficomTable6Dash10 qudt:numericValue ?selectedValue .
                FILTER (?reportedValue >= ?selectedValue &&
                        ?reportedValue <= ?selectedValue)
            }
        """
    ] .

gen:TRF088TorqueQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN-M>
    ] .

gen:TRF088TableLookupCase
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash10
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] .
<END_SHACL>
