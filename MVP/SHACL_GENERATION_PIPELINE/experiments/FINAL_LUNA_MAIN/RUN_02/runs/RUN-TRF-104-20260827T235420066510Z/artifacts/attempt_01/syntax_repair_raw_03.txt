<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF104ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:pyramidStrengthPrinciple ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship shall record that its propulsion line is designed according to the pyramid strength principle."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship shall provide the propeller-blade-loss state needed to verify the pyramid strength principle."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:message "The ship shall identify its propeller shaft line components."
    ] ;
    sh:sparql [
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:propellerBladeLoss true .
  $this nltl:hasPropellerShaftLineComponent ?component .
  FILTER NOT EXISTS {
    ?component nltl:significantDamageFromBladeLoss false .
  }
}
""" ;
        sh:message "When propeller blade loss occurs, no linked propeller shaft line component may have significant damage from that blade loss."
    ] .
</END_SHACL>
