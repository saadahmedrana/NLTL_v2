<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF108Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF108KnownCurveCase
        gen:TRF108UnknownCurveCase
    ) .

gen:TRF108KnownCurveCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A known SN-curve type must be recorded."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The propeller-blade material property is required to establish SN-curve correspondence."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:stressLifeCurveTypeValue
        ] ;
        sh:message "The selected SN-curve type must be linked through stressLifeCurveSelection."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "Evidence establishing correspondence between the selected SN-curve type and blade material properties is required."
    ] .

gen:TRF108UnknownCurveCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:maxCount 0 ;
        sh:message "The unknown SN-curve alternative applies when no SN-curve type is recorded."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:twoSlopeStressLifeCurve ;
        sh:message "If the SN-curve type is unknown, the two-slope SN curve must be used."
    ] .

<END_SHACL>
