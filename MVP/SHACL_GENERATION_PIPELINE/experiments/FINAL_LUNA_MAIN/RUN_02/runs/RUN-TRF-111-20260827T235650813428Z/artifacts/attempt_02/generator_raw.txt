<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide the extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:message "The ship shall identify a propeller shaft line component."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:message "The ship shall identify the applicable ship component."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minInclusive 1.3
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minInclusive 1.0
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase ;
                        sh:minCount 1
                    ]
                ]
            )
        ] ;
        sh:message "Each propulsion load case shall identify an applicable load-case type and satisfy its required direct safety-factor check."
    ] ;
    sh:sparql [
        sh:message "For an applicable fatigue load case, every linked ship component shall provide a fatigue safety factor of at least 1.5 and use the unitless unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?factor qudt:numericValue ?value .
                    FILTER (?value >= 1.5)
                }
            }
        """
    ] .

</END_SHACL>
