<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ExtremeIceForceQuantity ;
        sh:message "The ship must provide the extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ] ;
        sh:message "The ship must identify the applicable propeller shaft line component."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node gen:FatigueShipComponent ;
        sh:message "Each applicable ship component must provide a fatigue safety factor."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF111PropulsionLoadCase ;
        sh:message "The ship must provide applicable propulsion load cases with the required safety factors."
    ] .

gen:ExtremeIceForceQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .

gen:FatigueShipComponent
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:fatigueSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FatigueSafetyFactorQuantity ;
        sh:message "The ship component fatigue safety factor must be at least 1.5."
    ] .

gen:FatigueSafetyFactorQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:minInclusive 1.5
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .

gen:TRF111PropulsionLoadCase
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:shaftLineLoadCaseTypeValue ;
                sh:hasValue nltl:extremeOperationalLoadCase
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:YieldSafetyFactor130Quantity ;
                sh:message "The extreme operational load case yield safety factor must be at least 1.3."
            ]
        ]
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:shaftLineLoadCaseTypeValue ;
                sh:hasValue nltl:fatigueLoadCase
            ]
        ]
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:shaftLineLoadCaseTypeValue ;
                sh:hasValue nltl:bladeFailureLoadCase
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:YieldSafetyFactor100Quantity ;
                sh:message "The blade failure load case yield safety factor must be at least 1.0."
            ]
        ]
    ) .

gen:YieldSafetyFactor130Quantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:minInclusive 1.3
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .

gen:YieldSafetyFactor100Quantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:minInclusive 1.0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .
<END_SHACL>
