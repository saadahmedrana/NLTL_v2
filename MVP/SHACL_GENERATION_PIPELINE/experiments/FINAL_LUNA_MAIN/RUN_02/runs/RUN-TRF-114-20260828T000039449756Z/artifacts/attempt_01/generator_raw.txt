gen:TRF114CaseResistanceCheck
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterIceImpactLoadCase ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed thruster resistance capacity must be at least the thruster ice impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .
                $this nltl:thrusterImpactCaseAppliesToBody ?body .
                ?body nltl:thrusterResistanceCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacityValue .
                FILTER (?capacityValue < ?demandValue)
            }
        """
    ] .
