<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each thruster-body natural-frequency calculation case, damping, water added mass, and thruster natural frequency must be supplied on the ship with their canonical datatypes and units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody ;
                      nltl:hasNaturalFrequencyCalculationCase ?case .

                FILTER (
                    NOT EXISTS {
                        $this nltl:damping ?damping .
                        FILTER (datatype(?damping) = xsd:string)
                    }
                    ||
                    EXISTS {
                        $this nltl:damping ?damping1 .
                        $this nltl:damping ?damping2 .
                        FILTER (?damping1 != ?damping2)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:waterAddedMass ?waterMass .
                        ?waterMass a qudt:QuantityValue ;
                                   qudt:numericValue ?waterMassValue ;
                                   qudt:unit unit:KiloGM .
                        FILTER (datatype(?waterMassValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:waterAddedMass ?waterMass .
                        FILTER NOT EXISTS {
                            ?waterMass a qudt:QuantityValue ;
                                       qudt:numericValue ?waterMassValue ;
                                       qudt:unit unit:KiloGM .
                            FILTER (datatype(?waterMassValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:waterAddedMass ?waterMass1 .
                        $this nltl:waterAddedMass ?waterMass2 .
                        FILTER (?waterMass1 != ?waterMass2)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency .
                        ?frequency a qudt:QuantityValue ;
                                   qudt:numericValue ?frequencyValue ;
                                   qudt:unit unit:HZ .
                        FILTER (datatype(?frequencyValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency .
                        FILTER NOT EXISTS {
                            ?frequency a qudt:QuantityValue ;
                                       qudt:numericValue ?frequencyValue ;
                                       qudt:unit unit:HZ .
                            FILTER (datatype(?frequencyValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency1 .
                        $this nltl:thrusterNaturalFrequency ?frequency2 .
                        FILTER (?frequency1 != ?frequency2)
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-120-ThrusterBodyShape>
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:calculationCase ;
        sh:node [
            sh:property [
                sh:path nltl:vibrationDirection ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:longitudinalDirection
                    nltl:transverseDirection
                )
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloN-PER-M
            ]
        ]
    ] .
<END_SHACL>
