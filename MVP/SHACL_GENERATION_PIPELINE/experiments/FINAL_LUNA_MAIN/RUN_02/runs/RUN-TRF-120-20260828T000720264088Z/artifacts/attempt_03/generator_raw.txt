<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF120ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF120NoEstimationCase
        gen:TRF120ApplicableEstimation
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked natural-frequency calculation case must specify longitudinal or transverse vibration direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                FILTER NOT EXISTS {
                    ?case nltl:vibrationDirection ?direction .
                    FILTER (?direction IN (
                        nltl:longitudinalDirection,
                        nltl:transverseDirection
                    ))
                }
            }
        """ ;
    ] .

gen:TRF120NoEstimationCase
    a sh:NodeShape ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:hasNaturalFrequencyCalculationCase ) ;
        sh:maxCount 0 ;
    ] .

gen:TRF120ApplicableEstimation
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Damping must be recorded when a thruster natural-frequency estimation case is declared." ;
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF120WaterAddedMassQuantity ;
        sh:message "Water added mass must be recorded with the canonical kilogram unit when a thruster natural-frequency estimation case is declared." ;
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF120NaturalFrequencyQuantity ;
        sh:message "Thruster natural frequency must be recorded with the canonical hertz unit when a thruster natural-frequency estimation case is declared." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked natural-frequency estimation case must model ship attachment stiffness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                FILTER NOT EXISTS {
                    ?body nltl:shipAttachmentStiffness ?stiffness .
                    ?stiffness a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:KiloN-PER-M .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """ ;
    ] .

gen:TRF120WaterAddedMassQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloGM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gen:TRF120NaturalFrequencyQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:HZ ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
