<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF123ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:node [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The occasional-force case must explicitly confirm exclusion of propeller blades."
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:node [
                    sh:property [
                        sh:path q:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path q:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:class nltl:shipComponent ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:occasionalForceComponentStress ;
                        sh:node [
                            sh:property [
                                sh:path q:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path q:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:hasMaterialProperties ;
                        sh:class nltl:materialProperties ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path nltl:componentMaterialYieldStrength ;
                                sh:node [
                                    sh:property [
                                        sh:path q:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path q:unit ;
                                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ] ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each applicable occasional-force case and assessed component, stress multiplied by the safety factor must not exceed the material yield strength, allowing a relative tolerance of 1e-6 and an absolute fallback of 1e-9 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasOccasionalForceLoadCase ?case .
                ?case nltl:propellerBladeExcludedFromOccasionalForceScope true .
                ?case nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?case nltl:componentYieldSafetyFactor ?factorQuantity .

                ?stressQuantity q:numericValue ?stress .
                ?stressQuantity q:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?yieldQuantity q:numericValue ?yield .
                ?yieldQuantity q:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?factorQuantity q:numericValue ?factor .
                ?factorQuantity q:unit <http://qudt.org/vocab/unit/UNITLESS> .

                BIND((?stress * ?factor) AS ?adjustedStress)
                FILTER(
                    ?adjustedStress >
                    (
                        ?yield +
                        IF(
                            (ABS(?yield) * 0.000001) > 0.000000001,
                            ABS(?yield) * 0.000001,
                            0.000000001
                        )
                    )
                )
            }
        """
    ] .

</END_SHACL>
