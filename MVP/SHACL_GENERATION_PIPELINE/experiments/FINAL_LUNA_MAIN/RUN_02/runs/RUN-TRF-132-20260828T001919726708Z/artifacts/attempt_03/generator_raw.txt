<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerInKiloWShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnspecifiedQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PowerAtLeast740Shape
    a sh:PropertyShape ;
    sh:path nltl:maximumContinuousRatingPower ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        sh:class qudt:QuantityValue ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:minInclusive "740"^^xsd:decimal
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue unit:KiloW ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] .

gen:TRF132ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerInKiloWShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnspecifiedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnspecifiedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerInKiloWShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ] ;
            sh:property gen:PowerAtLeast740Shape
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ] ;
            sh:property gen:PowerAtLeast740Shape
        ]
        [
            sh:sparql [
                sh:message "For ice classes other than IB and IC, the ice class must be explicitly represented." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER (?iceClass NOT IN (nltl:iceClassIb, nltl:iceClassIc))
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be no less than f1*f2*f3*(f4*Delta + P0), within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:engineOutputFormulaBasePowerP0 ?p0Quantity ;
                       nltl:engineOutputFormulaDisplacement ?deltaQuantity ;
                       nltl:engineOutputFormulaFactorF1 ?f1Quantity ;
                       nltl:engineOutputFormulaFactorF2 ?f2Quantity ;
                       nltl:engineOutputFormulaFactorF3 ?f3Quantity ;
                       nltl:engineOutputFormulaFactorF4 ?f4Quantity ;
                       nltl:maximumContinuousRatingPower ?actualQuantity .

                ?p0Quantity qudt:numericValue ?p0 .
                ?deltaQuantity qudt:numericValue ?delta .
                ?f1Quantity qudt:numericValue ?f1 .
                ?f2Quantity qudt:numericValue ?f2 .
                ?f3Quantity qudt:numericValue ?f3 .
                ?f4Quantity qudt:numericValue ?f4 .
                ?actualQuantity qudt:numericValue ?actual .

                BIND ((?f1 * ?f2 * ?f3) * ((?f4 * ?delta) + ?p0) AS ?expected)
                FILTER (
                    ?actual +
                    (ABS(?expected) * "0.000001"^^xsd:decimal) +
                    "0.000001"^^xsd:decimal
                    < ?expected
                )
            }
        """
    ] .
<END_SHACL>
