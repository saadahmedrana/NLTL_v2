<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-133-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:message "The ship must have one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:node <urn:nltl:generated-shape:TRF-133-IceClassDraughtMarkingShape> ;
        sh:minCount 1 ;
        sh:message "An ice-class draught marking is required."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:node <urn:nltl:generated-shape:TRF-133-WarningTriangleMarkingShape> ;
        sh:minCount 1 ;
        sh:message "A warning triangle marking is required."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Timber load-line applicability must be recorded."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the timber load-line mark applies, the ice-class draught mark must reference the timber load-line vertical reference." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:timberLoadLineMarkApplicable ?app .
                FILTER (?app = true)
                $this nltl:hasIceClassDraughtMarking ?mark .
                FILTER NOT EXISTS {
                    ?mark nltl:draughtMarkAftReferencePoint
                          nltl:timberLoadLineVerticalReference .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the timber load-line mark does not apply, the ice-class draught mark must reference the load-line ring centre." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:timberLoadLineMarkApplicable ?app .
                FILTER (?app = false)
                $this nltl:hasIceClassDraughtMarking ?mark .
                FILTER NOT EXISTS {
                    ?mark nltl:draughtMarkAftReferencePoint
                          nltl:loadLineRingCentreReference .
                }
            }
        """
    ] .
