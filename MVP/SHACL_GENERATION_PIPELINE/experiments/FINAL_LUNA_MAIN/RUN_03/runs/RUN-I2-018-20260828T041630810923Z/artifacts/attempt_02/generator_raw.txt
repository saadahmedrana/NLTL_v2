    sh:property [
        sh:path nltl:hasIceLoadPatch ;

    sh:property [

        sh:maxCount 1 ;

    sh:property [

    sh:property [

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;

            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;

    sh:property [

            sh:class nltl:peakPressureFactorLookupCase ;

            sh:property [
                sh:path nltl:localizedStructuralMemberCategory ;

                sh:maxCount 1 ;
                sh:in (
                    nltl:bottomStructureFrameCategory
                    nltl:longitudinallyFramedPlatingCategory

            ] ;

    sh:sparql [
        sh:message "averageIcePressure must equal iceForce divided by loadPatchHeight and loadPatchLength, with kN converted to MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceForce/qudt:numericValue ?force ;
                      nltl:loadPatchHeight/qudt:numericValue ?height ;
                      nltl:loadPatchLength/qudt:numericValue ?length ;
                      nltl:averageIcePressure/qudt:numericValue ?reported .
                FILTER (?height != 0)
                FILTER (?length != 0)
                BIND ((?force * 0.001) / (?height * ?length) AS ?expected)
                FILTER (
                    ?reported > (?expected + (?expected * 0.000001) + 0.000001)
                    ||
                    ?reported < (?expected - (?expected * 0.000001) - 0.000001)
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "peakPressureFactor must equal the selected factor from the applicable Table 3 lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor/qudt:numericValue ?peak ;
                      nltl:hasPeakPressureFactorLookupCase ?case .
                ?case nltl:localizedStructuralMemberCategory ?category ;
                      nltl:selectedPeakPressureFactor/qudt:numericValue ?selected ;
                      nltl:tableReference nltl:iacsUrI2Table3 .
                FILTER (?peak != ?selected)
            }
        """
    ] .
