<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_019_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_019_StructuralMemberShape
    ] .

gen:I2_019_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:node gen:I2_019_HullAreaValueShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_019_SelectedFactorShape
    ] ;
    sh:sparql [
        sh:message "A structural member spanning a hull area must have a selected hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSpannedHullArea ?area .
                FILTER NOT EXISTS {
                    $this nltl:selectedHullAreaFactor ?selected .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected hull area factor must equal the maximum spanned hull area factor within the permitted decimal tolerance and must not be below any operand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedHullAreaFactor ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .

                {
                    SELECT ?member (MAX(?operandValue) AS ?expectedMaximum)
                    WHERE {
                        ?member nltl:hasSpannedHullArea ?area .
                        ?area nltl:hullAreaFactor ?factorQuantity .
                        ?factorQuantity qudt:numericValue ?operandValue .
                    }
                    GROUP BY ?member
                }

                FILTER (?member = $this)
                FILTER (
                    EXISTS {
                        $this nltl:hasSpannedHullArea ?greaterArea .
                        ?greaterArea nltl:hullAreaFactor ?greaterFactorQuantity .
                        ?greaterFactorQuantity qudt:numericValue ?greaterValue .
                        FILTER (?greaterValue > ?selectedValue)
                    }
                    ||
                    (
                        ABS(?selectedValue - ?expectedMaximum) >
                        IF(
                            ABS(?expectedMaximum) > 0,
                            ABS(?expectedMaximum) * "0.000001"^^xsd:decimal,
                            "0.000001"^^xsd:decimal
                        )
                    )
                )
            }
        """
    ] .

gen:I2_019_HullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_019_OperandFactorShape
    ] .

gen:I2_019_OperandFactorShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_019_SelectedFactorShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
