<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-019-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-019-StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2-019-StructuralMemberShape>
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:node <urn:nltl:generated-shape:I2-019-HullAreaValueShape>
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-019-SelectedFactorShape>
    ] ;
    sh:sparql [
        sh:message "selectedHullAreaFactor must equal the maximum spanned hullAreaFactor within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:selectedHullAreaFactor ?resultQuantity .
                ?resultQuantity qudt:numericValue ?reported .

                {
                    SELECT ?member (MAX(?operandValue) AS ?expected)
                    WHERE {
                        ?member nltl:hasSpannedHullArea ?area .
                        ?area nltl:hullAreaFactor ?operandQuantity .
                        ?operandQuantity qudt:numericValue ?operandValue .
                    }
                    GROUP BY ?member
                }

                FILTER (?member = $this)

                BIND(
                    IF(
                        ?expected > 0,
                        ?expected * 0.000001,
                        IF(
                            ?expected < 0,
                            (-?expected) * 0.000001,
                            0.000001
                        )
                    ) AS ?tol
                )

                FILTER (
                    ?reported < (?expected - ?tol)
                    || ?reported > (?expected + ?tol)
                    || EXISTS {
                        $this nltl:hasSpannedHullArea ?otherArea .
                        ?otherArea nltl:hullAreaFactor ?otherQuantity .
                        ?otherQuantity qudt:numericValue ?otherValue .
                        FILTER (?otherValue > ?reported)
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-019-HullAreaValueShape>
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-019-OperandFactorShape>
    ] .

<urn:nltl:generated-shape:I2-019-OperandFactorShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-019-SelectedFactorShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
