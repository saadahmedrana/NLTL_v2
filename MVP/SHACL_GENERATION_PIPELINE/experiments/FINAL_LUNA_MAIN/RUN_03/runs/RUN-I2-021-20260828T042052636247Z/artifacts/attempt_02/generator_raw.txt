<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_021_ShellPlateThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For each plating item, thickness must equal ice-load required net plate thickness plus corrosion and abrasion allowance, converted from millimetres to metres within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:iceLoadRequiredNetPlateThickness ?tnetQuantity ;
                         nltl:corrosionAbrasionAllowance ?allowanceQuantity ;
                         nltl:thickness ?thicknessQuantity .
                ?tnetQuantity qudt:numericValue ?tnet ;
                              qudt:unit unit:MilliM .
                ?allowanceQuantity qudt:numericValue ?allowance ;
                                   qudt:unit unit:MilliM .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                                  qudt:unit unit:M .

                BIND((?tnet + ?allowance) / 1000 AS ?expected)
                BIND(?thickness - ?expected AS ?difference)
                BIND(IF(?difference < 0, 0 - ?difference, ?difference) AS ?absoluteDifference)
                BIND(IF(?expected < 0, 0 - ?expected, ?expected) AS ?absoluteExpected)
                BIND(IF(?absoluteExpected > 0,
                        ?absoluteExpected * 0.000001,
                        0.000001) AS ?tolerance)
                FILTER(?absoluteDifference > ?tolerance)
            }
        """
    ] .
<END_SHACL>
