<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_022_ComplexReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked plating must provide a valid framing angle and hull-area classification so applicability can be established." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating a nltl:plating .
                FILTER (
                    NOT EXISTS {
                        ?plating nltl:framingAngleOmega ?angleQuantity .
                        ?angleQuantity qudt:numericValue ?angleValue ;
                                        qudt:unit unit:DEG .
                        FILTER (datatype(?angleValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?plating nltl:platingHullAreaClassification ?area .
                        ?area a nltl:hullAreaValue .
                    }
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Applicable plating must be linked to a structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating a nltl:plating ;
                         nltl:framingAngleOmega ?angleQuantity ;
                         nltl:platingHullAreaClassification ?area .
                ?angleQuantity qudt:numericValue ?angleValue ;
                                qudt:unit unit:DEG .
                FILTER (datatype(?angleValue) = xsd:decimal)
                FILTER (?angleValue >= 70)
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
                FILTER NOT EXISTS {
                    ?plating nltl:platingSupportedByStructuralMember ?member .
                    ?member a nltl:structuralMember .
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Applicable plating must contain the declared readiness inputs and externally computed net thickness result with the specified units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating a nltl:plating ;
                         nltl:framingAngleOmega ?angleQuantity ;
                         nltl:platingHullAreaClassification ?area ;
                         nltl:platingSupportedByStructuralMember ?member .
                ?angleQuantity qudt:numericValue ?angleValue ;
                                qudt:unit unit:DEG .
                ?member a nltl:structuralMember .
                FILTER (datatype(?angleValue) = xsd:decimal)
                FILTER (?angleValue >= 70)
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
                FILTER (
                    NOT EXISTS {
                        $this nltl:loadPatchHeight ?loadPatchHeightQuantity .
                        ?loadPatchHeightQuantity qudt:numericValue ?loadPatchHeightValue ;
                                                  qudt:unit unit:M .
                        FILTER (datatype(?loadPatchHeightValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:peakPressureFactor ?peakFactorQuantity .
                        ?peakFactorQuantity qudt:numericValue ?peakFactorValue ;
                                             qudt:unit unit:UNITLESS .
                        FILTER (datatype(?peakFactorValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:averageIcePressure ?averagePressureQuantity .
                        ?averagePressureQuantity qudt:numericValue ?averagePressureValue ;
                                                   qudt:unit unit:MegaPA .
                        FILTER (datatype(?averagePressureValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yieldStrengthQuantity .
                        ?yieldStrengthQuantity qudt:numericValue ?yieldStrengthValue ;
                                               qudt:unit unit:MegaPA .
                        FILTER (datatype(?yieldStrengthValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:frameSpacing ?spacingQuantity .
                        ?spacingQuantity qudt:numericValue ?spacingValue ;
                                          qudt:unit unit:M .
                        FILTER (datatype(?spacingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:selectedHullAreaFactor ?factorQuantity .
                        ?factorQuantity qudt:numericValue ?factorValue ;
                                         qudt:unit unit:UNITLESS .
                        FILTER (datatype(?factorValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?plating nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
                        ?thicknessQuantity qudt:numericValue ?thicknessValue ;
                                            qudt:unit unit:MilliM .
                        FILTER (datatype(?thicknessValue) = xsd:decimal)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
