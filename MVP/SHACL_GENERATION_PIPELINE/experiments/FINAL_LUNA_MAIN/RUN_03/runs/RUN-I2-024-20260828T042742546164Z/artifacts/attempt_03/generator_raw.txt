    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:node [
            sh:class nltl:obliquePlatingInterpolationCase ;
            sh:property [
                sh:path nltl:framingAngle ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:longitudinalFramingNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:transverseFramingNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;

                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable oblique-plating case must have exactly one lower and upper interpolation endpoint, complete endpoint inputs and results, one interpolated net plate thickness result, and a calculation case with both framing thickness operands." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasObliquePlatingInterpolationCase ?case .
                ?case nltl:framingAngle ?angle .
                ?angle qudt:numericValue ?angleValue .
                FILTER (
                    datatype(?angleValue) = xsd:decimal &&
                    ?angleValue > 20 &&
                    ?angleValue < 70
                )

                FILTER NOT EXISTS {
                    ?case nltl:interpolationLowerEndpoint ?lower ;
                          nltl:interpolationUpperEndpoint ?upper .

                    FILTER NOT EXISTS {
                        ?case nltl:interpolationLowerEndpoint ?lowerOther .
                        FILTER (?lowerOther != ?lower)
                    }
                    FILTER NOT EXISTS {
                        ?case nltl:interpolationUpperEndpoint ?upperOther .
                        FILTER (?upperOther != ?upper)
                    }

                    ?lower nltl:interpolationPointCoordinate ?lowerCoordinate ;
                           nltl:interpolationPointResult ?lowerResult .
                    ?upper nltl:interpolationPointCoordinate ?upperCoordinate ;
                           nltl:interpolationPointResult ?upperResult .

                    FILTER NOT EXISTS {
                        ?lower nltl:interpolationPointResult ?lowerResultOther .
                        FILTER (?lowerResultOther != ?lowerResult)
                    }
                    FILTER NOT EXISTS {
                        ?upper nltl:interpolationPointResult ?upperResultOther .
                        FILTER (?upperResultOther != ?upperResult)
                    }

                    ?lowerCoordinate qudt:numericValue ?lowerCoordinateValue ;
                                     qudt:unit unit:UNITLESS .
                    ?upperCoordinate qudt:numericValue ?upperCoordinateValue ;
                                     qudt:unit unit:UNITLESS .
                    ?lowerResult qudt:numericValue ?lowerResultValue .
                    ?upperResult qudt:numericValue ?upperResultValue .

                    FILTER (
                        datatype(?lowerCoordinateValue) = xsd:decimal &&
                        datatype(?upperCoordinateValue) = xsd:decimal &&
                        datatype(?lowerResultValue) = xsd:decimal &&
                        datatype(?upperResultValue) = xsd:decimal
                    )

                    ?case nltl:interpolatedNetPlateThickness ?netThickness .
                    FILTER NOT EXISTS {
                        ?case nltl:interpolatedNetPlateThickness ?netThicknessOther .
                        FILTER (?netThicknessOther != ?netThickness)
                    }
                    ?netThickness qudt:numericValue ?netThicknessValue ;
                                  qudt:unit unit:MilliM .
                    FILTER (datatype(?netThicknessValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?calculationCase .
                    ?calculationCase
                        nltl:longitudinalFramingNetPlateThickness ?longitudinal ;
                        nltl:transverseFramingNetPlateThickness ?transverse .
                    ?longitudinal qudt:numericValue ?longitudinalValue ;
                                  qudt:unit unit:MilliM .
                    ?transverse qudt:numericValue ?transverseValue ;
                                qudt:unit unit:MilliM .
                    FILTER (
                        datatype(?longitudinalValue) = xsd:decimal &&
                        datatype(?transverseValue) = xsd:decimal
                    )
                }
            }
        """
    ] .
