    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable member load case must not exceed plastic strength, and plastic strength must match the expected midspan collapse load within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity ;
                      nltl:midspanPlasticCollapseLoad ?midspanQuantity ;
                      nltl:hasStructuralMemberLoadCase ?case .

                ?strengthQuantity qudt:numericValue ?strength .
                ?midspanQuantity qudt:numericValue ?midspan .
                ?case nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand .

                FILTER (
                    ?demand > ?strength ||
                    ABS(?strength - ?midspan) >
                    IF(
                        ABS(?midspan) * 0.000001 > 0.000000001,
                        ABS(?midspan) * 0.000001,
                        0.000000001
                    )
                )
            }
        """
    ] .
