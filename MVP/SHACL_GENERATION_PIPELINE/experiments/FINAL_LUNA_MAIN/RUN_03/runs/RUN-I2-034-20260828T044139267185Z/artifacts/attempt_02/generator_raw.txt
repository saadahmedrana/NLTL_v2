<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gs:I2_034_StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented applicable side longitudinal frame load case must provide decimal shear-and-bending demand, plastic strength, and midspan plastic-collapse load values with the required quantity structure and unit." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:longitudinalFrameOrientation true ;
            nltl:structuralLocation nltl:sideStructureLocation ;
            nltl:hasStructuralMemberLoadCase ?case .
    FILTER (
        NOT EXISTS {
            ?case nltl:combinedShearAndBendingDemand ?demand .
            ?demand qudt:numericValue ?demandValue .
            FILTER (datatype(?demandValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?member nltl:plasticStrength ?strength .
            ?strength qudt:numericValue ?strengthValue ;
                     qudt:unit qudt:KiloN .
            FILTER (datatype(?strengthValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?member nltl:midspanPlasticCollapseLoad ?collapse .
            ?collapse qudt:numericValue ?collapseValue ;
                      qudt:unit qudt:KiloN .
            FILTER (datatype(?collapseValue) = xsd:decimal)
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear-and-bending demand must not exceed plastic strength, and plastic strength must agree with the midspan plastic-collapse load within a relative tolerance of 1e-6 with a 0.000001 absolute fallback." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:longitudinalFrameOrientation true ;
            nltl:structuralLocation nltl:sideStructureLocation ;
            nltl:hasStructuralMemberLoadCase ?case ;
            nltl:plasticStrength ?strength ;
            nltl:midspanPlasticCollapseLoad ?collapse .
    ?case nltl:combinedShearAndBendingDemand ?demand .
    ?demand qudt:numericValue ?demandValue .
    ?strength qudt:numericValue ?strengthValue ;
             qudt:unit qudt:KiloN .
    ?collapse qudt:numericValue ?collapseValue ;
              qudt:unit qudt:KiloN .
    FILTER (
        datatype(?demandValue) = xsd:decimal &&
        datatype(?strengthValue) = xsd:decimal &&
        datatype(?collapseValue) = xsd:decimal
    )
    BIND(max(?collapseValue, -?collapseValue) AS ?collapseAbsolute)
    BIND(max(0.000001, 0.000001 * ?collapseAbsolute) AS ?tolerance)
    FILTER (
        ?demandValue > ?strengthValue ||
        ?strengthValue < (?collapseValue - ?tolerance) ||
        ?strengthValue > (?collapseValue + ?tolerance)
    )
}
"""
    ] .

gs:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

</END_SHACL>
