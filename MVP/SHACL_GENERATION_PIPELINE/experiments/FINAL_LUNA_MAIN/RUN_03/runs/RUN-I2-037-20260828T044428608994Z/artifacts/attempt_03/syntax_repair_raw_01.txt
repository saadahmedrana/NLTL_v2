<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-037-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:node [
            sh:class nltl:iceLoadPatch ;
        ] ;
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:node [
            sh:class nltl:iceLoadPatchDesignCase ;
        ] ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each web frame or load-carrying stringer load case must be linked to an ice load patch and design case with confirmed capacity minimization, combined bending/shear effect, application location, and member capacity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?loadCase
            WHERE {
                $this a nltl:ship .
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberClass .
                FILTER (?memberClass IN (nltl:webFrame, nltl:loadCarryingStringer))
                ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    $this nltl:hasIceLoadPatch ?patch .
                    $this nltl:hasIceLoadPatchDesignCase ?designCase .
                    ?designCase nltl:memberCapacityMinimizationConfirmed true .
                    ?loadCase nltl:combinedShearBendingEffect ?combinedEffect .
                    ?loadCase nltl:loadPatchApplicationLocation ?location .
                    ?member nltl:memberCapacity ?capacity .
                    ?capacity qudt:numericValue ?numericValue ;
                             qudt:unit unit:KiloN .
                }
            }
        """ ;
    ] .

<urn:nltl:generated-shape:I2-037-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:targetClass nltl:loadCarryingStringer ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-037-LoadCaseShape> ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
        ] ;
    ] .

<urn:nltl:generated-shape:I2-037-LoadCaseShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMemberLoadCase ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
    ] .

<urn:nltl:generated-shape:I2-037-DesignCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceLoadPatchDesignCase ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
    ] .
<END_SHACL>
