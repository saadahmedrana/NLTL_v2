            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;

        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;

            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;

            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_051_CalculationCaseShape
    ] ;
    sh:sparql [
        sh:message "The positive-shear design vertical ice shear force must equal the longitudinal distribution factor multiplied by the bow ice force, converted from MN to kN, within the required tolerance; the prescribed end intervals require Cf equal to 0 or 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:hasCalculationCase ?case .

                ?fibQuantity qudt:numericValue ?fib .
                ?fiQuantity qudt:numericValue ?fi .
                ?cfQuantity qudt:numericValue ?cf .
                ?luiQuantity qudt:numericValue ?lui .

                ?case nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:shearForceDirection nltl:positiveShearForce .
                ?positionQuantity qudt:numericValue ?position .

                BIND((?cf * ?fib * 1000) AS ?expected)
                BIND((?expected * 0.000001 + 0.000001) AS ?tolerance)

                FILTER (
                    (
                        ?position <= (?lui * 0.6) &&
                        (?cf < 0 || ?cf > 0)
                    ) ||
                    (
                        ?position >= (?lui * 0.9) &&
                        (?cf < 1 || ?cf > 1)
                    ) ||
                    ?fi > (?expected + ?tolerance) ||
                    ?fi < (?expected - ?tolerance)
                )
            }
        """
    ] .
