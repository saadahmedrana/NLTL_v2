<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member in way of a cut-out must have a cut-out relationship and the ship must record that the member does not cause instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true .
                FILTER (
                    NOT EXISTS {
                        ?member nltl:hasCutout ?cutout .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:memberInstability false .
                    }
                    ||
                    EXISTS {
                        $this nltl:memberInstability ?instability .
                        FILTER (?instability != false)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where stiffening is necessary for a member in way of a cut-out, structural stiffening must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true .
                ?member nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member recorded as being in way of a cut-out must link to a cut-out." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:memberInWayOfCutout true .
                FILTER NOT EXISTS {
                    $this nltl:hasCutout ?cutout .
                }
            }
        """
    ] .

</END_SHACL>
