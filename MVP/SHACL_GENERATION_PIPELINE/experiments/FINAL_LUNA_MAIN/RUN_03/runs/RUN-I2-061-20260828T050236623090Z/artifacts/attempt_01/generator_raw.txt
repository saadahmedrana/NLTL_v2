<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify a calculation case for the prescribed hull-structure procedure."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:node gen:AssessedHullStructureShape ;
        sh:minCount 1 ;
        sh:message "Each calculation case must identify its assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must use the prescribed analytical procedure rather than a direct calculation."
    ] .

gen:AssessedHullStructureShape
    a sh:NodeShape ;
    sh:class nltl:hullStructure ;
    sh:property [
        sh:path nltl:thickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessed hull structure must provide its thickness as a metre quantity with a decimal numeric value."
    ] .
<END_SHACL>
