<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO002ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare one design ice condition required to determine the category rule."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare one Polar Code category."
    ] ;
    sh:sparql [
        sh:message "A ship designed for at least thin first-year ice with possible old-ice inclusions and not assigned Category A must be assigned Category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions .
                $this nltl:shipCategory ?category .
                FILTER (?category != nltl:polarShipCategoryA)
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryB .
                }
            }
        """
    ] .
<END_SHACL>
