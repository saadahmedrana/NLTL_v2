<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO_003_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare a design ice condition."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for open water or for ice conditions less severe than categories A and B must have ship category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition = nltl:openWaterIceCondition ||
                    ?condition = nltl:iceConditionLessSevereThanCategoryAAndB
                )
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryC .
                }
            }
        """
    ] .
<END_SHACL>
