<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO_006_MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:class nltl:iceTypeValue ;
        sh:message "Each ship must provide an ice type classification."
    ] ;
    sh:sparql [
        sh:message "For medium first-year ice, ice thickness must be a decimal quantity in metres between 0.7 m and 1.2 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?quantity
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?quantity .
                        FILTER (
                            NOT EXISTS {
                                ?quantity rdf:type qudt:QuantityValue
                            }
                            ||
                            NOT EXISTS {
                                ?quantity qudt:numericValue ?value
                            }
                            ||
                            EXISTS {
                                ?quantity qudt:numericValue ?value .
                                FILTER (
                                    datatype(?value) != xsd:decimal
                                    || ?value < "0.7"^^xsd:decimal
                                    || ?value > "1.2"^^xsd:decimal
                                )
                            }
                            ||
                            NOT EXISTS {
                                ?quantity qudt:unit unit:M
                            }
                            ||
                            EXISTS {
                                ?quantity qudt:unit ?otherUnit .
                                FILTER (?otherUnit != unit:M)
                            }
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
