@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice condition classification is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceConditionClassification ;
                sh:hasValue nltl:openWaterIceCondition
            ] ;
            sh:property [
                sh:path nltl:seaIceConcentration ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:maxExclusive "1"^^xsd:decimal ;
                        sh:message "For open water, sea ice concentration must be a decimal strictly less than 1."
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:UNITLESS ;
                        sh:message "Sea ice concentration must use the unitless unit."
                    ]
                ] ;
                sh:message "Open water requires a unitless sea ice concentration strictly less than 1."
            ] ;
            sh:property [
                sh:path nltl:iceOfLandOriginPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "For open water, ice of land origin must be absent."
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceConditionClassification ?condition .
                        FILTER (?condition != nltl:openWaterIceCondition)
                    }
                """ ;
                sh:message "A non-open-water ice condition is not subject to the open-water concentration and land-origin-ice conditions."
            ]
        ]
    ) .
<END_SHACL>
