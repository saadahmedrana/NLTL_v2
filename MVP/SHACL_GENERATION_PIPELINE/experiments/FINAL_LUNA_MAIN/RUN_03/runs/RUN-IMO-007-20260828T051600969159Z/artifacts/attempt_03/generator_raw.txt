gen:IMO007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one ice condition classification."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceConditionClassification ;
                sh:hasValue nltl:openWaterIceCondition
            ] ;
            sh:property [
                sh:path nltl:seaIceConcentration ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:maxExclusive 1 ;
                        sh:message "Sea ice concentration must be a decimal strictly less than 1."
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:message "Sea ice concentration must use the unitless unit."
                    ]
                ] ;
                sh:message "Open water requires a unitless sea ice concentration below 1."
            ] ;
            sh:property [
                sh:path nltl:iceOfLandOriginPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "Open water requires ice of land origin to be absent."
            ]
        ]
        [
            sh:not [
                sh:property [
                    sh:path nltl:iceConditionClassification ;
                    sh:hasValue nltl:openWaterIceCondition
                ]
            ]
        ]
    ) ;
    sh:message "If the ice condition is open water, its concentration must be below 1 and no ice of land origin may be present."
