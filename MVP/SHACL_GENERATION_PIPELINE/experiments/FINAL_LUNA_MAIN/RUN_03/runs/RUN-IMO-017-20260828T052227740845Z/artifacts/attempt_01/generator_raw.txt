    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship must provide survey dates."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship must provide endorsement dates."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Ship Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "The ship must have a Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gs:PolarCertificateSupplementShape ;
        sh:message "The ship must have a Polar Certificate supplement with required equipment records."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gs:SolasCertificateScheduleShape ;
        sh:message "The ship must have an applicable SOLAS certificate schedule."
    ] ;
    sh:sparql [
        sh:message "Each SOLAS certificate schedule date must match the corresponding Polar Ship Certificate date by controlled date category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasSolasCertificateSchedule ?schedule .
                ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                ?record nltl:certificateScheduleDateType ?category ;
                        nltl:certificateScheduleDate ?scheduleDate .
                FILTER (
                    (
                        ?category = nltl:validityDateCategory &&
                        NOT EXISTS {
                            $this nltl:polarCertificateValidityDates ?shipDate .
                            FILTER (?shipDate = ?scheduleDate)
                        }
                    ) ||
                    (
                        ?category = nltl:surveyDateCategory &&
                        NOT EXISTS {
                            $this nltl:surveyDates ?shipDate .
                            FILTER (?shipDate = ?scheduleDate)
                        }
                    ) ||
                    (
                        ?category = nltl:endorsementDateCategory &&
                        NOT EXISTS {
                            $this nltl:endorsementDates ?shipDate .
                            FILTER (?shipDate = ?scheduleDate)
                        }
                    )
                )
            }
        """
    ] .
