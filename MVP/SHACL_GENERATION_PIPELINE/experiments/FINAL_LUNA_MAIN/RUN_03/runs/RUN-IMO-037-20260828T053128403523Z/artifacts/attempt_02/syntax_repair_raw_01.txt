<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO037ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCategoryValue ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:message "A valid polar ship category is required."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:or (
        gen:IMO037ApplicableShipBranch
        gen:IMO037CategoryCBranch
        gen:IMO037Pre2017Branch
    ) .

gen:IMO037ApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:or (
            [
                sh:hasValue nltl:polarShipCategoryA ;
            ]
            [
                sh:hasValue nltl:polarShipCategoryB ;
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionCase ;
        sh:node gen:IMO037LoadingConditionCaseShape
    ] .

gen:IMO037CategoryCBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC
    ] .

gen:IMO037Pre2017Branch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:maxInclusive "2016-12-31"^^xsd:date
    ] .

gen:IMO037LoadingConditionCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Each applicable loading condition case requires an identifier."
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "Each applicable loading condition case requires result evidence."
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Alternative-instrument applicability must be stated for each loading condition."
    ] ;
    sh:or (
        gen:IMO037StandardResidualBranch
        gen:IMO037AlternativeInstrumentBranch
    ) .

gen:IMO037StandardResidualBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:hasValue false
    ] ;
    sh:property gen:IMO037ResidualFactorPropertyShape .

gen:IMO037AlternativeInstrumentBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The alternative-instrument residual stability outcome must be compliant."
    ] .

gen:IMO037ResidualFactorPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:residualStabilityFactorSI ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:datatype xsd:decimal
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:hasValue qudt:UNITLESS
        ]
    ] ;
    sh:sparql [
        sh:message "The residual stability factor must be within 1e-6 of the expected dimensionless result 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:residualStabilityFactorSI ?quantityNode .
                ?quantityNode qudt:numericValue ?reportedValue .
                FILTER (
                    ABS(xsd:decimal(?reportedValue) - xsd:decimal("1.0")) >
                    xsd:decimal("0.000001")
                )
            }
        """
    ] .
<END_SHACL>
