<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:hasLoadingConditionCase ;
                sh:node gen:IMO_037_LoadingConditionCaseShape
            ]
        ]
        [
            sh:sparql [
                sh:message "The ship is outside the applicability condition for this requirement." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipCategory ?category ;
                              nltl:constructionDate ?date .
                        FILTER (
                            ?category NOT IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                            || ?date < "2017-01-01"^^xsd:date
                        )
                    }
                """
            ]
        ]
    ] .

gen:IMO_037_LoadingConditionCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:residualStabilityFactorSI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "The residual stability factor must equal the expected result 1.0 within the explicit expected-result-scaled tolerance." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:residualStabilityFactorSI ?quantity .
                        ?quantity qudt:numericValue ?reported ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            ?reported < (
                                "1.0"^^xsd:decimal
                                - (
                                    "0.000001"^^xsd:decimal
                                    * "1.0"^^xsd:decimal
                                    + "0.000000000001"^^xsd:decimal
                                )
                            )
                            ||
                            ?reported > (
                                "1.0"^^xsd:decimal
                                + (
                                    "0.000001"^^xsd:decimal
                                    * "1.0"^^xsd:decimal
                                    + "0.000000000001"^^xsd:decimal
                                )
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
