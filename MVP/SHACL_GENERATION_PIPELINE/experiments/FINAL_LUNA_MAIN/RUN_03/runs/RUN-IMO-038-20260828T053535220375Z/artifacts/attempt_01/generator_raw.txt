@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_038_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must state whether the damage centre is forward of maximum breadth."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Upper ice waterline length must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Upper ice waterline draught must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Longitudinal damage extent must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 760.0 ;
                sh:maxInclusive 760.0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Transverse penetration must equal 760 mm."
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Vertical damage extent must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Selected vertical damage position must be supplied as a single metre quantity."
    ] ;
    sh:sparql [
        sh:message "Damage extents and selected vertical damage position do not satisfy the required Polar Code formulas and range." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .

                ?lengthNode qudt:numericValue ?length ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?longitudinalNode qudt:numericValue ?longitudinal ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                ?verticalNode qudt:numericValue ?vertical ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                ?positionNode qudt:numericValue ?position ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    ABS(
                        ?longitudinal -
                        IF(
                            ?forward = true,
                            0.045 * ?length,
                            0.015 * ?length
                        )
                    ) >
                    (
                        ABS(
                            IF(
                                ?forward = true,
                                0.045 * ?length,
                                0.015 * ?length
                            )
                        ) * 0.000001 + 0.000000001
                    )
                    ||
                    ABS(
                        ?vertical -
                        IF(
                            0.20 * ?draught < ?longitudinal,
                            0.20 * ?draught,
                            ?longitudinal
                        )
                    ) >
                    (
                        ABS(
                            IF(
                                0.20 * ?draught < ?longitudinal,
                                0.20 * ?draught,
                                ?longitudinal
                            )
                        ) * 0.000001 + 0.000000001
                    )
                    ||
                    ?position > (1.20 * ?draught)
                )
            }
        """
    ] .
