<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO041Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hydraulicFluidFreezingOrViscosityControlPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the opening actuation type is hydraulic, hydraulic-fluid freezing or excessive-viscosity protection must be present and true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuationType .
                FILTER (STR(?actuationType) = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent ?protection .
                    FILTER (?protection = true)
                }
            }
        """
    ] .
<END_SHACL>
