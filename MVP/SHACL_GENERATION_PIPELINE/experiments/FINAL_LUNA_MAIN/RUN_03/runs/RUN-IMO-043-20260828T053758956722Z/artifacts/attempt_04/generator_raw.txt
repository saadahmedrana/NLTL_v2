<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:class nltl:machineryInstallationItem ;
        sh:node nltl:IMO043MachineryInstallationShape ;
        sh:message "Each linked machinery installation must be a machinery installation item and satisfy the hazard-coverage constraint."
    ] .

nltl:IMO043MachineryInstallationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceAccretionHazard
            nltl:snowAccumulationHazard
            nltl:seawaterIceIngestionHazard
            nltl:liquidFreezingOrViscosityIncreaseHazard
            nltl:seawaterIntakeTemperatureHazard
            nltl:snowIngestionHazard
        ) ;
        sh:minCount 1 ;
        sh:message "A machinery installation must identify at least one permitted applicable hazard."
    ] ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem ;
        sh:node nltl:IMO043AssociatedEquipmentShape ;
        sh:message "Each associated equipment item must be represented as an associated equipment item."
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:nodeKind sh:IRI ;
        sh:message "Protection evidence must be represented by an evidence artifact."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every applicable hazard and associated equipment item must have corresponding hazard-protection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasApplicableHazard ?hazard ;
                      nltl:hasAssociatedEquipment ?equipment .
                FILTER (
                    ?hazard IN (
                        nltl:iceAccretionHazard,
                        nltl:snowAccumulationHazard,
                        nltl:seawaterIceIngestionHazard,
                        nltl:liquidFreezingOrViscosityIncreaseHazard,
                        nltl:seawaterIntakeTemperatureHazard,
                        nltl:snowIngestionHazard
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:hasHazardProtectionEvidence ?evidence .
                }
            }
        """
    ] .

nltl:IMO043AssociatedEquipmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:nodeKind sh:IRI ;
        sh:message "Associated equipment must have protection evidence."
    ] .
<END_SHACL>
