<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_050_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fireSafetyComponentExposed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceAndSnowProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "An exposed fire-safety component must have ice and snow protection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:fireSafetyComponentExposed ?exposed .
                FILTER (?exposed = true^^xsd:boolean)
                FILTER NOT EXISTS {
                    $this nltl:iceAndSnowProtectionPresent ?protected .
                    FILTER (?protected = true^^xsd:boolean)
                }
            }
        """
    ] .
<END_SHACL>
