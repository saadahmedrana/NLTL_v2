<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO057ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each fire pump must be linked to a compartment with maintained temperature above 0 deg C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER NOT EXISTS {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:DEG_C .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value > 0
                    )
                }
            }
        """
    ] .
<END_SHACL>
