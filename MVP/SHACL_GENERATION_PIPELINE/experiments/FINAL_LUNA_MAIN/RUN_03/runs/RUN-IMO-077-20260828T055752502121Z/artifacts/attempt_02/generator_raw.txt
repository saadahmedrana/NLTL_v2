<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO_077_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The additional-equipment carriage applicability must be recorded as a single boolean value."
    ] ;
    sh:or (
        nltlgen:IMO_077_NotCarriedShape
        nltlgen:IMO_077_CarriedShape
    ) .

nltlgen:IMO_077_NotCarriedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IMO_077_CarriedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:personsCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single persons capacity requirement is required when additional equipment is carried in the survival craft."
    ] ;
    sh:property [
        sh:path nltl:additionalEquipmentCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single additional equipment capacity requirement is required when additional equipment is carried in the survival craft."
    ] ;
    sh:property [
        sh:path nltl:survivalCraftAvailableCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single survival craft available capacity is required when additional equipment is carried in the survival craft."
    ] ;
    sh:property [
        sh:path nltl:launchingApplianceCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single launching appliance capacity in kilograms is required when additional equipment is carried in the survival craft."
    ] ;
    sh:property [
        sh:path nltl:loadedSurvivalCraftRequirement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single loaded survival craft requirement in kilograms is required when additional equipment is carried in the survival craft."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Survival craft available capacity must be at least the persons capacity requirement plus the additional equipment capacity requirement, and launching appliance capacity must be at least the loaded survival craft requirement." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:additionalEquipmentCarriedInSurvivalCraft true .
    $this nltl:personsCapacityRequirement ?persons .
    $this nltl:additionalEquipmentCapacityRequirement ?additional .
    $this nltl:survivalCraftAvailableCapacity ?available .
    $this nltl:launchingApplianceCapacity ?launching .
    $this nltl:loadedSurvivalCraftRequirement ?loaded .
    ?launching qudt:numericValue ?launchingValue .
    ?loaded qudt:numericValue ?loadedValue .
    FILTER (
        ?available < (?persons + ?additional)
        || ?launchingValue < ?loadedValue
    )
}
""" ;
    ] .
<END_SHACL>
