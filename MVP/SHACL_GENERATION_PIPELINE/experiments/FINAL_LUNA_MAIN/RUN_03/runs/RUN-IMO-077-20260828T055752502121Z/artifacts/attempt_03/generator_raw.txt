<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO077Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The additional-equipment applicability value must be present exactly once and be boolean."
    ] ;
    sh:or (
        gen:IMO077NotApplicableShape
        gen:IMO077ApplicableShape
    ) .

gen:IMO077NotApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The non-applicable branch requires additionalEquipmentCarriedInSurvivalCraft to be false."
    ] .

gen:IMO077ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable branch requires additionalEquipmentCarriedInSurvivalCraft to be true."
    ] ;
    sh:property [
        sh:path nltl:personsCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The persons capacity requirement is required for the applicable calculation."
    ] ;
    sh:property [
        sh:path nltl:additionalEquipmentCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The additional equipment capacity requirement is required for the applicable calculation."
    ] ;
    sh:property [
        sh:path nltl:survivalCraftAvailableCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The available survival-craft capacity is required for the applicable calculation."
    ] ;
    sh:property [
        sh:path nltl:loadedSurvivalCraftRequirement ;
        sh:node gen:IMO077MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The loaded survival-craft requirement is required as a kilogram quantity."
    ] ;
    sh:property [
        sh:path nltl:launchingApplianceCapacity ;
        sh:node gen:IMO077MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The launching-appliance capacity is required as a kilogram quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The survival-craft capacity must be at least the persons capacity requirement plus the additional equipment capacity requirement, and the launching-appliance capacity must be at least the loaded survival-craft requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:personsCapacityRequirement ?persons .
                $this nltl:additionalEquipmentCapacityRequirement ?additional .
                $this nltl:survivalCraftAvailableCapacity ?available .
                $this nltl:loadedSurvivalCraftRequirement ?loaded .
                $this nltl:launchingApplianceCapacity ?launching .
                ?loaded qudt:numericValue ?loadedValue .
                ?launching qudt:numericValue ?launchingValue .
                FILTER (
                    ! ( ?available >= ( ?persons + ?additional ) )
                    || ! ( ?launchingValue >= ?loadedValue )
                )
            }
        """
    ] .

gen:IMO077MassQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloGM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use kilograms as its unit."
    ] .
<END_SHACL>
