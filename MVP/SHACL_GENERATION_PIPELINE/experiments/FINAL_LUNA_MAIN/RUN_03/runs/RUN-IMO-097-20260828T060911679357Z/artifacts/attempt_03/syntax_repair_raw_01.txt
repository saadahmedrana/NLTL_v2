<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-097-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating applicability must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat ;
        sh:node <urn:nltl:generated-shape:IMO-097-RequiredCraftShape>
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat ;
        sh:node <urn:nltl:generated-shape:IMO-097-RequiredCraftShape>
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft ;
        sh:node <urn:nltl:generated-shape:IMO-097-SurvivalCraftShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every released rescue boat or lifeboat must carry at least one ship-to-shore distress-alert device, one location-signal device, and one on-scene transmit/receive communication device when the ship operates in low air temperature." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature ?lowTemperature .
                FILTER (?lowTemperature = true)
                $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                ?craft nltl:releasedRescueBoatOrLifeboat ?released .
                FILTER (?released = true)
                FILTER (
                    NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?distressDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?shipToShoreDevice .
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-097-RequiredCraftShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The rescue boat or lifeboat release applicability must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasDistressAlertDevice ;
        sh:class nltl:equipmentItem
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasShipToShoreDistressAlertDevice ;
        sh:class nltl:shipToShoreDistressAlertDevice
    ] .

<urn:nltl:generated-shape:IMO-097-SurvivalCraftShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:survivalCraftHasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasShipToShoreDistressAlertDevice ;
        sh:class nltl:shipToShoreDistressAlertDevice
    ] .
<END_SHACL>
