<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_097_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide its low-air-temperature applicability state."
    ] ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide its rescue-boat or lifeboat release applicability state."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship operates in low air temperature and the rescue boats or lifeboats are released for evacuation, every linked rescue boat or lifeboat must have at least one ship-to-shore distress-alert device, one location-signal device, and one on-scene communication device." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipOperatesInLowAirTemperature ?lowTemperature ;
          nltl:releasedRescueBoatOrLifeboat ?released .
    FILTER (?lowTemperature = true && ?released = true)
    $this (nltl:hasLifeboat|nltl:hasRequiredRescueBoatOrLifeboat) ?craft .
    FILTER NOT EXISTS {
        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?distressDevice ;
               nltl:survivalCraftHasLocationSignalDevice ?locationDevice ;
               nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
    }
}
"""
    ] .

gen:IMO_097_LifeboatShape
    a sh:NodeShape ;
    sh:targetClass nltl:lifeboat ;
    sh:property [
        sh:path nltl:hasDistressAlertDevice ;
        sh:class nltl:equipmentItem
    ] .

gen:IMO_097_SurvivalCraftShape
    a sh:NodeShape ;
    sh:targetClass nltl:survivalCraft ;
    sh:property [
        sh:path nltl:survivalCraftHasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasShipToShoreDistressAlertDevice ;
        sh:class nltl:shipToShoreDistressAlertDevice
    ] .
<END_SHACL>
