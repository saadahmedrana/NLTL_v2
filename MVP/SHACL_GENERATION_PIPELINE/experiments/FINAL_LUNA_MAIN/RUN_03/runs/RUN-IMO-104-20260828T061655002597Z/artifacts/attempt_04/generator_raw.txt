<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization status."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:class nltl:crewMember ;
        sh:node gs:IMO_104_CrewMemberShape
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gs:IMO_104_CrewMemberShape
    ] ;
    sh:sparql [
        sh:message "Each crew member reached through one ship-to-crew path must be reached through both hasCrewMember and hasCrewMemberInventory." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                {
                    $this nltl:hasCrewMember ?crew .
                    FILTER NOT EXISTS {
                        $this nltl:hasCrewMemberInventory ?crew .
                    }
                }
                UNION
                {
                    $this nltl:hasCrewMemberInventory ?crew .
                    FILTER NOT EXISTS {
                        $this nltl:hasCrewMember ?crew .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every PWOM procedure or equipment item relevant to a crew member's assigned duties must be represented by that crew member's familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gs:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty ;
        sh:minCount 1 ;
        sh:node gs:IMO_104_AssignedDutyShape ;
        sh:message "Each crew member must have an assigned duty."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:minCount 1 ;
        sh:node gs:IMO_104_FamiliarizationRecordShape ;
        sh:message "Each crew member must have a PWOM familiarization record."
    ] .

gs:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node gs:IMO_104_ManualItemShape ;
        sh:message "Each assigned duty must identify relevant PWOM procedures or equipment."
    ] .

gs:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node gs:IMO_104_ManualItemShape ;
        sh:message "Each familiarization record must identify the PWOM items covered."
    ] .

gs:IMO_104_ManualItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .
<END_SHACL>
