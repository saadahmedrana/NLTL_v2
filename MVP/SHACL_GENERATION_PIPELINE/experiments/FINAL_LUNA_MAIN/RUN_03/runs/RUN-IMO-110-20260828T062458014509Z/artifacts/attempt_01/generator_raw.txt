<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:class qudt:QuantityValue ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue unit:TON_Metric ;
            sh:maxCount 1
        ]
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A qualifying Category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 tonnes must have compliant double-bottom protection status for the entire cargo tank length." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:deadweight ?deadweight ;
                      a nltl:oilTanker .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?shipType = "oil_tanker")
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                ?deadweight qudt:numericValue ?mass ;
                            qudt:unit unit:TON_Metric .
                FILTER (?mass < 5000)

                FILTER NOT EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?status .
                    FILTER (?status = true)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A qualifying Category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 tonnes must have compliant wing-tank protection status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:deadweight ?deadweight ;
                      a nltl:oilTanker .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?shipType = "oil_tanker")
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                ?deadweight qudt:numericValue ?mass ;
                            qudt:unit unit:TON_Metric .
                FILTER (?mass < 5000)

                FILTER NOT EXISTS {
                    $this nltl:wingTankProtectionStatus ?status .
                    FILTER (?status = true)
                }
            }
        """
    ] .
<END_SHACL>
