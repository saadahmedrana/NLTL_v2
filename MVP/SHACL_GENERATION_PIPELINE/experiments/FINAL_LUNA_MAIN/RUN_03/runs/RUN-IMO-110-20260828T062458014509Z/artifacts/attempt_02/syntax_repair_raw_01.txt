<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Applicable category A or B oil tankers under 5,000 tonnes deadweight constructed on or after 1 January 2017 shall have both required protection statuses exactly true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:oilTanker ;
                    nltl:shipCategory ?category ;
                    nltl:shipType ?shipType ;
                    nltl:deadweight ?deadweight ;
                    nltl:constructionDate ?constructionDate ;
                    nltl:entireCargoTankLengthDoubleBottomProtectionStatus true ;
                    nltl:wingTankProtectionStatus true .

                ?deadweight qudt:numericValue ?deadweightValue ;
                    qudt:unit unit:TON_Metric .

                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?shipType = "oil_tanker"^^xsd:string)
                FILTER (datatype(?deadweightValue) = xsd:decimal)
                FILTER (?deadweightValue < 5000)
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                FILTER (!EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?doubleBottomStatus .
                    FILTER NOT (
                        datatype(?doubleBottomStatus) = xsd:boolean &&
                        str(?doubleBottomStatus) = "true"
                    )
                })

                FILTER (!EXISTS {
                    $this nltl:wingTankProtectionStatus ?wingStatus .
                    FILTER NOT (
                        datatype(?wingStatus) = xsd:boolean &&
                        str(?wingStatus) = "true"
                    )
                })
            }
        """
    ] .
<END_SHACL>
