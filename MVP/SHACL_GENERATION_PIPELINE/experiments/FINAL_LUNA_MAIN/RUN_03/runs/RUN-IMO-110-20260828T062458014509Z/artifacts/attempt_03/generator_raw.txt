<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 metric tonnes, both required cargo-tank protection statuses must be exactly true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:shipCategory ?category .
  FILTER (
    ?category = nltl:polarShipCategoryA ||
    ?category = nltl:polarShipCategoryB
  )

  $this nltl:shipType ?shipType .
  FILTER (?shipType = "oil_tanker"^^xsd:string)

  $this nltl:constructionDate ?constructionDate .
  FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

  $this nltl:deadweight ?deadweightNode .
  ?deadweightNode qudt:numericValue ?deadweightValue .
  ?deadweightNode qudt:unit unit:TON_Metric .
  FILTER (
    datatype(?deadweightValue) = xsd:decimal &&
    ?deadweightValue < "5000"^^xsd:decimal
  )

  FILTER (
    NOT EXISTS {
      $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus
        "true"^^xsd:boolean
    } ||
    EXISTS {
      $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?doubleBottomStatus .
      FILTER (?doubleBottomStatus != "true"^^xsd:boolean)
    } ||
    NOT EXISTS {
      $this nltl:wingTankProtectionStatus
        "true"^^xsd:boolean
    } ||
    EXISTS {
      $this nltl:wingTankProtectionStatus ?wingTankStatus .
      FILTER (?wingTankStatus != "true"^^xsd:boolean)
    }
  )
}
"""
    ] .
<END_SHACL>
