<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship category must be supplied as Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A valid construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable oil residue tank or oily bilge water holding tank, individual tank capacity must be supplied in cubic metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipCategory ?category ;
        nltl:constructionDate ?date ;
        nltl:tankType ?tankType .
  FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
  FILTER (
    YEAR(?date) > 2017 ||
    (YEAR(?date) = 2017 && MONTH(?date) > 1) ||
    (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
  )
  FILTER (?tankType IN ("oil_residue_tank"^^xsd:string,
                        "oily_bilge_water_holding_tank"^^xsd:string))
  FILTER NOT EXISTS {
    $this nltl:individualTankCapacity ?capacity .
    ?capacity qudt:numericValue ?capacityValue ;
              qudt:unit unit:M3 .
    FILTER (datatype(?capacityValue) = xsd:decimal)
  }
}
""" ;
        sh:prefixes <urn:nltl:generated-shape:IMO-111>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable tank with an individual capacity greater than 30 m3, separation from the outer shell must be at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipCategory ?category ;
        nltl:constructionDate ?date ;
        nltl:tankType ?tankType ;
        nltl:individualTankCapacity ?capacity .
  ?capacity qudt:numericValue ?capacityValue ;
            qudt:unit unit:M3 .
  FILTER (datatype(?capacityValue) = xsd:decimal)
  FILTER (xsd:decimal(?capacityValue) > 30)
  FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
  FILTER (
    YEAR(?date) > 2017 ||
    (YEAR(?date) = 2017 && MONTH(?date) > 1) ||
    (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
  )
  FILTER (?tankType IN ("oil_residue_tank"^^xsd:string,
                        "oily_bilge_water_holding_tank"^^xsd:string))
  FILTER NOT EXISTS {
    $this nltl:tankOuterShellSeparation ?separation .
    ?separation qudt:numericValue ?separationValue ;
                qudt:unit unit:M .
    FILTER (datatype(?separationValue) = xsd:decimal)
    FILTER (xsd:decimal(?separationValue) >= 0.76)
  }
}
""" ;
        sh:prefixes <urn:nltl:generated-shape:IMO-111>
    ] .
</urn:nltl:generated-shape:IMO-111>
<END_SHACL>
