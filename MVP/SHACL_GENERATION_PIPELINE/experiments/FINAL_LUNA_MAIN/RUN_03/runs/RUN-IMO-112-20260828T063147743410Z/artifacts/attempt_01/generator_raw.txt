<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_112_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one operating area to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "The NLS or NLS-mixture discharge-to-sea record, when present, must be a single boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship operating in Arctic_waters must record that discharge of noxious liquid substances or mixtures containing them into the sea is false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (STR(?area) = "Arctic_waters")
                FILTER NOT EXISTS {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                    FILTER (
                        DATATYPE(?discharge) = xsd:boolean &&
                        ?discharge = false
                    )
                }
            }
        """
    ] .
<END_SHACL>
