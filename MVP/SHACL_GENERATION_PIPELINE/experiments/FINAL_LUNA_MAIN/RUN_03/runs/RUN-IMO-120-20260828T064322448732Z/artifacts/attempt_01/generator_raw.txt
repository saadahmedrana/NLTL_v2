<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Operating area is required to determine applicability." ;
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The food-waste discharge status is required to determine applicability." ;
    ] ;
    sh:sparql [
        sh:message "For Arctic waters with food-waste discharge to sea, the nearest land, ice-shelf, or fast-ice distance must be at least 12 nautical miles, and a distance from an area with ice concentration above one tenth must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .

                FILTER ( ?area = "Arctic_waters"^^xsd:string &&
                         ?discharge = true )

                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                    ?nearest rdf:type qudt:QuantityValue ;
                             qudt:numericValue ?nearestValue ;
                             qudt:unit unit:MI_N .
                    FILTER ( datatype(?nearestValue) = xsd:decimal &&
                             ?nearestValue >= "12"^^xsd:decimal )
                }

                FILTER NOT EXISTS {
                    $this nltl:nearbyIceConcentration ?ice .
                    ?ice rdf:type qudt:QuantityValue ;
                         qudt:numericValue ?iceValue ;
                         qudt:unit unit:UNITLESS .
                    FILTER ( datatype(?iceValue) = xsd:decimal &&
                             ?iceValue > "0.1"^^xsd:decimal )
                }

                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recordedDistance .
                    ?recordedDistance rdf:type qudt:QuantityValue ;
                                      qudt:numericValue ?recordedValue ;
                                      qudt:unit unit:MI_N .
                    FILTER ( datatype(?recordedValue) = xsd:decimal )
                }
            }
        """ ;
    ] .
<END_SHACL>
