<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_002_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ship type is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Length overall is required in metres to determine fishing-vessel applicability."
    ] ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Gross tonnage is required as a dimensionless quantity to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Trade status is required for pleasure-yacht applicability."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Chapter 9-1 applicability must be recorded as true for an applicable ship."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Chapter 9-1 applicability does not match the ship type, dimensions, gross tonnage, and trade status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipType ?shipType ;
                      nltl:lengthOverall ?lengthNode ;
                      nltl:grossTonnage ?tonnageNode ;
                      nltl:engagedInTrade ?engaged ;
                      nltl:polarCodeChapter9Dash1Applicable ?applicable .
                ?lengthNode qudt:numericValue ?length .
                ?tonnageNode qudt:numericValue ?tonnage .

                FILTER (
                    ?applicable != true ||
                    !(
                        (
                            ?shipType = "fishing_vessel" &&
                            ?length >= 24
                        ) ||
                        (
                            ?shipType = "pleasure_yacht" &&
                            ?tonnage >= 300 &&
                            ?engaged = false
                        ) ||
                        (
                            ?shipType = "cargo_ship" &&
                            ?tonnage >= 300 &&
                            ?tonnage < 500
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
