<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-015-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeChapter11Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When Polar Code chapter 11 is applicable, SOLAS chapter I certification must be recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter11Applicable ?applicable .
                FILTER (?applicable = true)
                FILTER NOT EXISTS {
                    $this nltl:solasChapterICertified ?certified .
                    FILTER (?certified = true)
                }
            }
        """
    ] .
<END_SHACL>
