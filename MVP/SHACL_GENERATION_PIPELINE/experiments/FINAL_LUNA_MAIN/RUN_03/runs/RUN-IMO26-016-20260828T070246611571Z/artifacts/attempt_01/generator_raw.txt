<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_016_Chapter11_1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship operating in polar waters, Chapter 11-1 applicability must be true and the ship particulars must establish one of the specified ship-type and threshold cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters true .

                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeChapter11Dash1Applicable true
                    }
                    ||
                    NOT EXISTS {
                        {
                            $this nltl:shipType "fishing_vessel"^^xsd:string .
                            $this nltl:lengthOverall ?lengthQuantity .
                            ?lengthQuantity qudt:numericValue ?lengthValue ;
                                             qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (
                                datatype(?lengthValue) = xsd:decimal &&
                                ?lengthValue >= 24
                            )
                        }
                        UNION
                        {
                            $this nltl:shipType "pleasure_yacht"^^xsd:string .
                            $this nltl:grossTonnage ?tonnageQuantity .
                            ?tonnageQuantity qudt:numericValue ?tonnageValue ;
                                             qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                            FILTER (
                                datatype(?tonnageValue) = xsd:decimal &&
                                ?tonnageValue >= 300
                            )
                            $this nltl:engagedInTrade false
                        }
                        UNION
                        {
                            $this nltl:shipType "cargo_ship"^^xsd:string .
                            $this nltl:grossTonnage ?cargoTonnageQuantity .
                            ?cargoTonnageQuantity qudt:numericValue ?cargoTonnageValue ;
                                                   qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                            FILTER (
                                datatype(?cargoTonnageValue) = xsd:decimal &&
                                ?cargoTonnageValue >= 300 &&
                                ?cargoTonnageValue < 500
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
