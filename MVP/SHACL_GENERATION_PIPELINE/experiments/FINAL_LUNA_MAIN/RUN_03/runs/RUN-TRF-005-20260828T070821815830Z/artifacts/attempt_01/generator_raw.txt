<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "Construction contract date must be an xsd:date when provided."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "Construction stage date must be an xsd:date when provided."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:message "Applicable ice class regulation edition must be an IRI of the permitted regulation-edition value class."
    ] ;
    sh:sparql [
        sh:message "Ships with a construction stage on or after 1 September 2003 and contracted before 1 January 2010 must identify the 2002 Ice Class Regulations as the applicable regulation edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate ;
                      nltl:constructionContractDate ?contractDate .

                FILTER (datatype(?stageDate) = xsd:date)
                FILTER (datatype(?contractDate) = xsd:date)

                FILTER (
                    YEAR(?stageDate) > 2003 ||
                    (YEAR(?stageDate) = 2003 && MONTH(?stageDate) > 9) ||
                    (YEAR(?stageDate) = 2003 && MONTH(?stageDate) = 9 && DAY(?stageDate) >= 1)
                )

                FILTER (
                    YEAR(?contractDate) < 2010 ||
                    (YEAR(?contractDate) = 2010 && MONTH(?contractDate) < 1) ||
                    (YEAR(?contractDate) = 2010 && MONTH(?contractDate) = 1 && DAY(?contractDate) < 1)
                )

                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2002 .
                }
            }
        """
    ] .
<END_SHACL>
