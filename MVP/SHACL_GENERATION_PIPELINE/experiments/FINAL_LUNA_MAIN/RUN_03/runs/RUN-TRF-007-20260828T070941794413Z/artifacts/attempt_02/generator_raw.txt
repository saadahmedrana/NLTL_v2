<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF007InputsShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice class is required to determine TRF-007 applicability."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single construction stage date is required to determine TRF-007 applicability."
    ] .

gs:TRF007NonApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with ice class IA or IA Super and a construction stage before 1 September 2003 must satisfy the applicable TRF-007 compliance branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                $this nltl:constructionStageDate ?constructionStageDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIa ,
                        nltl:iceClassIaSuper
                    )
                )
                FILTER (
                    ?constructionStageDate < "2003-09-01"^^xsd:date
                )
            }
        """
    ] .

gs:TRF007ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single delivery date is required for the TRF-007 twenty-year deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single assessment date is required to verify the TRF-007 deadline."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, the assessment date must not be later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:deliveryDate ?deliveryDate .
                $this nltl:assessmentDate ?assessmentDate .
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        &&
                        (
                            MONTH(?assessmentDate) > 1
                            ||
                            (
                                MONTH(?assessmentDate) = 1
                                &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    )
                )
            }
        """
    ] .

gs:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:node gs:TRF007InputsShape ;
    sh:or (
        [ sh:node gs:TRF007NonApplicableShape ]
        [ sh:node gs:TRF007ApplicableShape ]
    ) .
<END_SHACL>
