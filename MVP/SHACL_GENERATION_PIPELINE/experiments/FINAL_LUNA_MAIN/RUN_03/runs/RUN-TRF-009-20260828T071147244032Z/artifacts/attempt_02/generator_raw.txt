<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF009ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must identify its ice class."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA or IA Super ship constructed before 1 September 2003, assessmentDate, constructionStageDate, and deliveryDate are required to determine applicability and the compliance deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIa, nltl:iceClassIaSuper))
                $this nltl:constructionStageDate ?constructionDate .
                FILTER (
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        MONTH(?constructionDate) < 9
                    )
                )
                FILTER (
                    NOT EXISTS { $this nltl:assessmentDate ?assessmentDate } ||
                    NOT EXISTS { $this nltl:deliveryDate ?deliveryDate }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "From 1 January in the year twenty years after delivery, an applicable IA or IA Super ship must identify one of the permitted 2017 Ice Class Regulations compliance methods." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIa, nltl:iceClassIaSuper))
                $this nltl:constructionStageDate ?constructionDate .
                FILTER (
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        MONTH(?constructionDate) < 9
                    )
                )
                $this nltl:assessmentDate ?assessmentDate .
                $this nltl:deliveryDate ?deliveryDate .
                BIND(YEAR(?deliveryDate) + 20 AS ?deadlineYear)
                FILTER (
                    YEAR(?assessmentDate) > ?deadlineYear ||
                    (
                        YEAR(?assessmentDate) = ?deadlineYear &&
                        MONTH(?assessmentDate) >= 1
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?permittedMethod .
                        FILTER (
                            ?permittedMethod IN (
                                nltl:traficom2017Section3Point2Point2Method,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    } ||
                    EXISTS {
                        $this nltl:engineOutputComplianceMethod ?invalidMethod .
                        FILTER (
                            ?invalidMethod NOT IN (
                                nltl:traficom2017Section3Point2Point2Method,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
