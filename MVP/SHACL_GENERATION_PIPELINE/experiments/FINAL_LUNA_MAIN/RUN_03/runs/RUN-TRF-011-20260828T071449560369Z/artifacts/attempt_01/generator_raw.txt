<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node g:TRF011IntendedIceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node g:TRF011UpperIceWaterlineShape
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal the maximum vertical coordinate of intended ice-operating waterline points at every longitudinal position, with matching positions in both profiles." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            {
                SELECT ?this ?x ?uv
                WHERE {
                    ?this nltl:hasIntendedIceOperatingWaterline ?intended .
                    ?intended nltl:hasWaterlineProfilePoint ?ip .
                    ?ip nltl:longitudinalPosition ?ix ;
                        nltl:verticalCoordinate ?ivNode .
                    ?ix qudt:numericValue ?x .
                    ?ivNode qudt:numericValue ?iv .

                    OPTIONAL {
                        ?this nltl:hasUpperIceWaterline ?upper .
                        ?upper nltl:hasWaterlineProfilePoint ?up .
                        ?up nltl:longitudinalPosition ?ux ;
                            nltl:verticalCoordinate ?uvNode .
                        ?ux qudt:numericValue ?x .
                        ?uvNode qudt:numericValue ?uv .
                    }
                }
                GROUP BY ?this ?x ?uv
                HAVING (!BOUND(?uv) || MAX(?iv) != ?uv)
            }
            UNION
            {
                SELECT ?this ?x ?uv
                WHERE {
                    ?this nltl:hasUpperIceWaterline ?upper .
                    ?upper nltl:hasWaterlineProfilePoint ?up .
                    ?up nltl:longitudinalPosition ?ux ;
                        nltl:verticalCoordinate ?uvNode .
                    ?ux qudt:numericValue ?x .
                    ?uvNode qudt:numericValue ?uv .

                    OPTIONAL {
                        ?this nltl:hasIntendedIceOperatingWaterline ?intended .
                        ?intended nltl:hasWaterlineProfilePoint ?ip .
                        ?ip nltl:longitudinalPosition ?ix ;
                            nltl:verticalCoordinate ?ivNode .
                        ?ix qudt:numericValue ?x .
                        ?ivNode qudt:numericValue ?iv .
                    }
                }
                GROUP BY ?this ?x ?uv
                HAVING (!BOUND(?iv) || MAX(?iv) != ?uv)
            }
        """
    ] .

g:TRF011IntendedIceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node g:TRF011WaterlineProfilePointShape
    ] .

g:TRF011UpperIceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node g:TRF011WaterlineProfilePointShape
    ] .

g:TRF011WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape
    ] .

g:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
