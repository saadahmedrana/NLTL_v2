<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF012ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:minCount 1 ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF012IceWaterlineShape ;
        sh:message "The ship must represent at least one intended ice-operating waterline with profile points."
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:minCount 1 ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF012IceWaterlineShape ;
        sh:message "The externally determined lower ice waterline must be represented with profile points."
    ] .

gen:TRF012IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:minCount 1 ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:TRF012ProfilePointShape ;
        sh:message "Each represented ice waterline must contain profile points."
    ] .

gen:TRF012ProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:TRF012LengthQuantityShape ;
        sh:message "Each waterline profile point must have one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:TRF012LengthQuantityShape ;
        sh:message "Each waterline profile point must have one vertical coordinate in metres."
    ] .

gen:TRF012LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
