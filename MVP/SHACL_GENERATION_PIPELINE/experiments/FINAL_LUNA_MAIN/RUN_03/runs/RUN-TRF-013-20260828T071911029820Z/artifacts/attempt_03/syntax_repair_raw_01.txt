<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF013QuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF013IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline .

gen:TRF013Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipIceStrengthened ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:node gen:TRF013IceWaterlineShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:node gen:TRF013IceWaterlineShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:operatingForeDraught ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:operatingAftDraught ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumIceClassDraughtFore ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumIceClassDraughtFore ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumIceClassDraughtAft ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumIceClassDraughtAft ;
                sh:node gen:TRF013QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The operating fore and aft draughts must be within their inclusive ice-class draught bounds, and the upper and lower ice waterlines must be distinct." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:hasLowerIceWaterline ?lowerWaterline ;
                              nltl:hasUpperIceWaterline ?upperWaterline ;
                              nltl:operatingForeDraught ?operatingFore ;
                              nltl:operatingAftDraught ?operatingAft ;
                              nltl:minimumIceClassDraughtFore ?minimumFore ;
                              nltl:maximumIceClassDraughtFore ?maximumFore ;
                              nltl:minimumIceClassDraughtAft ?minimumAft ;
                              nltl:maximumIceClassDraughtAft ?maximumAft .

                        ?operatingFore qudt:numericValue ?operatingForeValue .
                        ?operatingAft qudt:numericValue ?operatingAftValue .
                        ?minimumFore qudt:numericValue ?minimumForeValue .
                        ?maximumFore qudt:numericValue ?maximumForeValue .
                        ?minimumAft qudt:numericValue ?minimumAftValue .
                        ?maximumAft qudt:numericValue ?maximumAftValue .

                        FILTER (
                            ?operatingForeValue < ?minimumForeValue ||
                            ?operatingForeValue > ?maximumForeValue ||
                            ?operatingAftValue < ?minimumAftValue ||
                            ?operatingAftValue > ?maximumAftValue ||
                            ?lowerWaterline = ?upperWaterline
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
