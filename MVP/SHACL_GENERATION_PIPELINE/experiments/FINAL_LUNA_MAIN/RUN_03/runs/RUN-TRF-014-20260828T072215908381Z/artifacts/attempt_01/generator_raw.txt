<urn:nltl:generated-shape:TRF014_NotApplicableBranch>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This branch applies when the summer load line fresh water draught is not above the upper ice waterline draught." ;

PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:summerLoadLineFreshWaterDraught ?summerNode ;
        nltl:upperIceWaterlineDraught ?uiwlNode .
  ?summerNode qudt:numericValue ?summer .
  ?uiwlNode qudt:numericValue ?uiwl .
  FILTER (?summer > ?uiwl)
}
"""
    ] .

<urn:nltl:generated-shape:TRF014_PostCutoffMarkingBranch>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "Ships built on or after 1 July 2007 must have the required warning triangle and ice-class draught mark." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  FILTER NOT EXISTS {
    $this nltl:constructionDate ?construction ;
          nltl:summerLoadLineFreshWaterDraught ?summerNode ;
          nltl:upperIceWaterlineDraught ?uiwlNode ;
          nltl:warningTrianglePresent true ;
          nltl:iceClassDraughtMarkPresent true ;
          nltl:iceClassDraughtMarkDraughtAmidships ?markNode .
    ?summerNode qudt:numericValue ?summer .
    ?uiwlNode qudt:numericValue ?uiwl .
    ?markNode qudt:numericValue ?markValue ;
              qudt:unit <http://qudt.org/vocab/unit/M> .
    FILTER (?construction >= "2007-07-01"^^xsd:date)
    FILTER (?summer > ?uiwl)
  }
}
"""
    ] .
