        sh:message "When navigating in ice, draught amidships must be at least the LIWL draught amidships." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true ;
                      nltl:draughtAmidships ?draughtQ ;
                      nltl:liwlDraughtAmidships ?liwlQ .
                ?draughtQ qudt:numericValue ?draught .
                ?liwlQ qudt:numericValue ?liwl .
                FILTER (?draught < ?liwl)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Forward draught must meet the level-ice-thickness formula using displacement at the greatest-displacement UIWL." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true ;
                      nltl:displacementAtUpperIceWaterline ?deltaQ ;
                      nltl:levelIceThickness ?hiQ ;
                      nltl:forwardDraught ?forwardQ ;
                      nltl:hasUpperIceWaterline ?selectedWL .
                ?deltaQ qudt:numericValue ?delta ;
                        qudt:unit <http://qudt.org/vocab/unit/TON_Metric> .
                ?hiQ qudt:numericValue ?hi ;
                     qudt:unit <http://qudt.org/vocab/unit/M> .
                ?forwardQ qudt:numericValue ?forward ;
                         qudt:unit <http://qudt.org/vocab/unit/M> .
                ?selectedWL nltl:waterlineDisplacement ?selectedDQ .
                ?selectedDQ qudt:numericValue ?selectedDelta ;
                            qudt:unit <http://qudt.org/vocab/unit/TON_Metric> .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?otherWL .
                    ?otherWL nltl:waterlineDisplacement ?otherDQ .
                    ?otherDQ qudt:numericValue ?otherDelta .
                    FILTER (?otherDelta > ?selectedDelta)
                }
                FILTER (?selectedDelta = ?delta)
                BIND (IF (((2 + (0.00025 * ?delta)) * ?hi) < (4 * ?hi),
                          ((2 + (0.00025 * ?delta)) * ?hi),
                          (4 * ?hi)) AS ?expected)
                BIND (((?expected * 0.000001) + 0.000000001) AS ?tolerance)
                FILTER ((?forward + ?tolerance) < ?expected)
            }
        """
    ] ;

        sh:message "Any ballast tank above the LIWL that is needed to reach the LIWL must have freezing prevention." ;

                      nltl:usedToReachLowerIceWaterline true .
                FILTER NOT EXISTS {
                    ?tank nltl:freezingPreventionPresent true
                }
            }
        """
    ] .
