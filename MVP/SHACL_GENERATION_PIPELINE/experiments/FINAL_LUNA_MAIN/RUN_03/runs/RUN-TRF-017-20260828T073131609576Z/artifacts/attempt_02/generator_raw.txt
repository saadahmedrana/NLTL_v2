<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF017Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propulsionOutputRestrictionApplies ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:restrictedPropulsionOutput ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:propulsionOutputRestrictionApplies ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionMachineryContinuousOutput ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The maximum continuous rating power must equal the restricted propulsion output when restriction applies, or the propulsion machinery continuous output otherwise." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:propulsionOutputRestrictionApplies ?flag .
    FILTER (
        (
            ?flag = true &&
            (
                NOT EXISTS {
                    $this nltl:restrictedPropulsionOutput ?restricted
                }
                ||
                NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?maximum
                }
                ||
                NOT EXISTS {
                    $this nltl:restrictedPropulsionOutput ?restricted .
                    ?restricted qudt:numericValue ?restrictedValue ;
                                qudt:unit unit:KiloW .
                    $this nltl:maximumContinuousRatingPower ?maximum .
                    ?maximum qudt:numericValue ?maximumValue ;
                             qudt:unit unit:KiloW .
                    FILTER (?restrictedValue = ?maximumValue)
                }
            )
        )
        ||
        (
            ?flag = false &&
            (
                NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?maximum
                }
                ||
                NOT EXISTS {
                    $this nltl:propulsionMachineryContinuousOutput ?machinery
                }
                ||
                NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?maximum .
                    ?maximum qudt:numericValue ?maximumValue ;
                             qudt:unit unit:KiloW .
                    $this nltl:propulsionMachineryContinuousOutput ?machinery .
                    ?machinery qudt:numericValue ?machineryValue ;
                              qudt:unit unit:KiloW .
                    FILTER (?maximumValue = ?machineryValue)
                }
            )
        )
    )
}
"""
    ] .
<END_SHACL>
