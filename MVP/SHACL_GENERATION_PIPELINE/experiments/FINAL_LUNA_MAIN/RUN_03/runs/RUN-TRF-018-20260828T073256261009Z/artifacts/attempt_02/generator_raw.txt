<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF018Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be at least the calculated required power and the applicable ice-class minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:calculatedRequiredPower ?calculatedQuantity ;
                      nltl:maximumContinuousRatingPower ?maximumQuantity .

                ?calculatedQuantity qudt:numericValue ?calculated ;
                                    qudt:unit unit:KiloW .
                ?maximumQuantity qudt:numericValue ?maximum ;
                                qudt:unit unit:KiloW .

                FILTER (
                    (
                        ?ice = nltl:iceClassIaSuper &&
                        (
                            ?maximum < ?calculated ||
                            ?maximum < 2800
                        )
                    ) ||
                    (
                        ?ice IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        (
                            ?maximum < ?calculated ||
                            ?maximum < 1000
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
