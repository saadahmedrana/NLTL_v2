<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityNoUnitShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NewtonQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityNoUnitShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:N ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KilowattQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityNoUnitShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MeterQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityNoUnitShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:node gen:QuantityNoUnitShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ApplicableIceWaterlineShape a sh:NodeShape ;
    sh:class nltl:iceWaterline .

gen:UpperIceWaterlineShape a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The upper ice waterline shall have an upper-waterline length."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineBreadth ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The upper ice waterline shall have an upper-waterline breadth."
    ] .

gen:ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2003-09-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceRch ;
        sh:node gen:NewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:node gen:QuantityNoUnitShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:node gen:QuantityNoUnitShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchControlType ;
        sh:class nltl:propellerPitchControlTypeValue ;
        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:class nltl:propulsionSystemTypeValue ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node gen:ApplicableIceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node gen:UpperIceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The applicable ship must have maximum continuous rating power at least as great as both externally calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mPower .
                ?mPower qudt:numericValue ?m .
                $this nltl:minimumRequiredPowerAtLowerIceWaterline ?lPower .
                ?lPower qudt:numericValue ?l .
                $this nltl:minimumRequiredPowerAtUpperIceWaterline ?uPower .
                ?uPower qudt:numericValue ?u .
                FILTER (?m < ?l || ?m < ?u)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The applicable branch requires the construction date and an eligible ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?d ;
                      nltl:iceClass ?c .
                FILTER (!(?d >= "2003-09-01"^^xsd:date &&
                          ?c IN (
                              nltl:iceClassIaSuper,
                              nltl:iceClassIa,
                              nltl:iceClassIb,
                              nltl:iceClassIc
                          )))
            }
        """
    ] .

gen:NonApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The non-applicable branch cannot be selected when the date and eligible ice class jointly establish applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?d ;
                      nltl:iceClass ?c .
                FILTER (?d >= "2003-09-01"^^xsd:date &&
                        ?c IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ))
            }
        """
    ] .

gen:TRF020Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [ sh:node gen:ApplicableShape ]
        [ sh:node gen:NonApplicableShape ]
    ) .
<END_SHACL>
