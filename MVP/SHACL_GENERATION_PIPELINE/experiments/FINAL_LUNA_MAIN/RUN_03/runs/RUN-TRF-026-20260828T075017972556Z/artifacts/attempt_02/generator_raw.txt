gen:TRF026ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property gen:TRF026IceClassPropertyShape ;
    sh:or (
        gen:TRF026NonIaSuperBranch
        gen:TRF026IaSuperWithoutBulbBranch
        gen:TRF026IaSuperWithBulbBranch
    ) .

gen:TRF026IceClassPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:iceClass ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:in (
        nltl:iceClassIaSuper
        nltl:iceClassIa
        nltl:iceClassIb
        nltl:iceClassIc
        nltl:iceClassIi
        nltl:iceClassIii
    ) ;
    sh:message "Exactly one permitted ice class is required." .
