<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF031QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:node gen:TRF031QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:node gen:TRF031QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:node gen:TRF031QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:nodeKind sh:BlankNodeOrIRI ;
                sh:node gen:TRF031QuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be lower than the yield point." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combinedBendingShearStress ?combinedQuantity ;
                       nltl:yieldPoint ?yieldQuantity .
                ?combinedQuantity qudt:numericValue ?combined .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?combined >= ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Yield shear stress must equal the yield point divided by the square root of 3 within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldShearStress ?reportedQuantity ;
                       nltl:yieldPoint ?yieldQuantity .
                ?reportedQuantity qudt:numericValue ?reported .
                ?yieldQuantity qudt:numericValue ?yield .
                BIND((?yield / 1.7320508075688772) AS ?expected)
                BIND(IF(ABS(?expected) * 0.000001 > 0.000000000001,
                        ABS(?expected) * 0.000001,
                        0.000000000001) AS ?tolerance)
                FILTER (ABS(?reported - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must be no greater than 0.9 times the yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                       nltl:allowableShearStress ?allowableQuantity ;
                       nltl:yieldPoint ?yieldQuantity .
                ?allowableQuantity qudt:numericValue ?allowable .
                ?yieldQuantity qudt:numericValue ?yield .
                BIND((0.9 * (?yield / 1.7320508075688772)) AS ?limit)
                FILTER (?allowable > ?limit)
            }
        """
    ] .
<END_SHACL>
