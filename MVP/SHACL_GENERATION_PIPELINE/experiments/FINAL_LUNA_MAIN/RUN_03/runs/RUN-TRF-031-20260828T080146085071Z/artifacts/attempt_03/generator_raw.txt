<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:StressQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:vonMisesYieldCriterion ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:node gen:StressQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be strictly lower than the yield point." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combinedBendingShearStress ?combinedQuantity ;
                      nltl:yieldPoint ?yieldQuantity .
                ?combinedQuantity qudt:numericValue ?combined .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?combined >= ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Reported yield shear stress must equal yield point divided by sqrt(3) within a relative tolerance of 1e-6 with a small absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldQuantity ;
                      nltl:yieldShearStress ?shearQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                ?shearQuantity qudt:numericValue ?reported .
                BIND (?yield / 1.7320508075688772 AS ?expected)
                FILTER (
                    ABS(?reported - ?expected) >
                    (0.000001 * ABS(?expected) + 0.000000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must not exceed 0.9 times yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:allowableShearStress ?allowableQuantity ;
                      nltl:yieldPoint ?yieldQuantity .
                ?allowableQuantity qudt:numericValue ?allowable .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?allowable > (0.9 * (?yield / 1.7320508075688772)))
            }
        """
    ] .
<END_SHACL>
