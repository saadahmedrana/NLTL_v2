<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-035-RuleLength>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship's rule length must be provided as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The classification society rule length must be provided as a single metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length must equal the classification society rule length." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ruleLength ?ruleQuantity .
                ?ruleQuantity qudt:numericValue ?ruleValue .
                $this nltl:classificationSocietyRuleLength ?societyQuantity .
                ?societyQuantity qudt:numericValue ?societyValue .
                FILTER (
                    datatype(?ruleValue) = xsd:decimal &&
                    datatype(?societyValue) = xsd:decimal &&
                    ?ruleValue != ?societyValue
                )
            }
        """
    ] .
<END_SHACL>
