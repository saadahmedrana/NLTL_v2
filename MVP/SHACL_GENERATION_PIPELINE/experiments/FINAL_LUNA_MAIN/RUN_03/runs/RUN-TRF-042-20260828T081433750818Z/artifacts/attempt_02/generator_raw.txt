<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMilliMQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NormalCorrosionBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Normal corrosion and abrasion addition must be exactly 2 mm."
    ] .

gen:LowerApprovedCoatingBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxExclusive 2
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A lower corrosion and abrasion addition must be supported by maintained special surface coating and approved material evidence."
    ] ;
    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved
    ] .

gen:PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:or (
        gen:NormalCorrosionBranch
        gen:LowerApprovedCoatingBranch
    ) ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable piecewise function of designIceLoadHeight divided by spacing, with a relative tolerance of 1e-6 and an absolute fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .

                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?f .

                FILTER (?s != 0)
                BIND (?h / ?s AS ?ratio)
                FILTER (?ratio > 0 && ?ratio <= 1.8)

                BIND (
                    IF(
                        ?ratio <= 1,
                        0.4 / ?ratio + 0.6,
                        1.4 - 0.4 * ?ratio
                    )
                    AS ?expected
                )

                BIND (
                    IF(
                        1e-6 * ABS(?expected) > 1e-12,
                        1e-6 * ABS(?expected),
                        1e-12
                    )
                    AS ?tolerance
                )

                FILTER (ABS(?f - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
