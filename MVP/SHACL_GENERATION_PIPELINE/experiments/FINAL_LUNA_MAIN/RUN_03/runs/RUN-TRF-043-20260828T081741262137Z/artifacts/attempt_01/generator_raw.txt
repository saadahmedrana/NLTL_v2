<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape ;
        sh:message "Each framing ice-strengthening termination case must conform to the termination requirements."
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MillimetreQuantityShape ;
        sh:message "The termination case must provide one extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:message "The termination adjacent boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:extensionBeyondAdjacentDeckOrTankBoundary qudt:numericValue ) ;
                sh:maxInclusive "250"^^xsd:decimal
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:extensionBeyondAdjacentDeckOrTankBoundary qudt:numericValue ) ;
                sh:minExclusive "250"^^xsd:decimal
            ]
        ]
    ) ;
    sh:message "When the extension is no more than 250 mm, termination at the adjacent boundary must be permitted." .

gen:TRF043MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qudt:MilliM
    ] .
<END_SHACL>
