<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF043_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:node [
            sh:property [
                sh:path nltl:terminationAdjacentBoundary ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:or (
                    [ sh:class nltl:deckStructure ]
                    [ sh:class nltl:tankBoundaryPlating ]
                    [ sh:class nltl:tankTop ]
                ] ;
                sh:message "A termination case must identify one adjacent deck, tank boundary plating, or tank top."
            ] ;
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:MilliM
                    ]
                ] ;
                sh:message "The termination extension must be a single quantity in millimetres."
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:datatype xsd:boolean ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "If the strengthening extends no more than 250 mm beyond the adjacent boundary, termination there must be permitted." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
                        ?quantity qudt:numericValue ?extension ;
                                  qudt:unit unit:MilliM .
                        FILTER (datatype(?extension) = xsd:decimal)
                        FILTER (?extension <= 250)
                        FILTER NOT EXISTS {
                            $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
                        }
                    }
                """
            ]
        }
    ] .
