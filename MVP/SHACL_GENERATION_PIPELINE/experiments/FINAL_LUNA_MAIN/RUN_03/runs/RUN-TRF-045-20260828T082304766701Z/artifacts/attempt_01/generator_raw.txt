<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-045_FrameOrdinaryScantlingsShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary frame scantlings may be used only when the frame span within the ice-strengthening zone is less than 15% of the frame span, with both quantities stated in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed ?used .
                FILTER (?used = true)
                FILTER NOT EXISTS {
                    $this nltl:frameSpanWithinIceStrengtheningZone ?withinQuantity .
                    $this nltl:span ?spanQuantity .
                    ?withinQuantity a qudt:QuantityValue ;
                        qudt:numericValue ?withinValue ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?spanQuantity a qudt:QuantityValue ;
                        qudt:numericValue ?spanValue ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?withinValue) = xsd:decimal)
                    FILTER (datatype(?spanValue) = xsd:decimal)
                    FILTER (?spanValue > 0)
                    FILTER ((?withinValue / ?spanValue) < 0.15)
                }
            }
        """
    ] .
<END_SHACL>
