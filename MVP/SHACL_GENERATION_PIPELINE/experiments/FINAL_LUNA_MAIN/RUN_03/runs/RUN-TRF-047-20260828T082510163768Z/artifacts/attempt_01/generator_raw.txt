<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047_FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each main frame or intermediate ice frame must have a strengthened part with a lower end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a ?frameType .
                FILTER (?frameType IN (nltl:mainFrame, nltl:intermediateIceFrame))
                FILTER NOT EXISTS {
                    $this nltl:hasStrengthenedPart ?part .
                    ?part nltl:hasLowerEnd ?lowerEnd .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The lower end must be attached to an approved supporting structure or use the permitted intermediate-frame horizontal connection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a ?frameType .
                FILTER (?frameType IN (nltl:mainFrame, nltl:intermediateIceFrame))
                $this nltl:hasStrengthenedPart ?part .
                ?part nltl:hasLowerEnd ?lowerEnd .

                FILTER NOT EXISTS {
                    ?lowerEnd nltl:hasAttachedSupportingStructure ?attached .
                    ?attached a ?attachedType .
                    FILTER (?attachedType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    ))
                }

                FILTER NOT EXISTS {
                    $this a nltl:intermediateIceFrame ;
                        nltl:supportingStructureAtOrBelowIceBeltLowerLimit true ;
                        nltl:sameScantlingsAsMainFrame true .
                    ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?terminatedBelow .
                    ?terminatedBelow a ?terminatedType .
                    FILTER (?terminatedType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    ))
                    ?lowerEnd nltl:hasHorizontalConnectionMember ?member .
                    ?member nltl:connectsToAdjacentMainFrame ?adjacentMain .
                    ?adjacentMain a nltl:mainFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each main frame must record that the main frame below the ice belt is strengthened." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:mainFrame .
                FILTER NOT EXISTS {
                    $this nltl:mainFrameBelowIceBeltStrengthened ?strengthened .
                    FILTER (
                        datatype(?strengthened) = xsd:boolean &&
                        ?strengthened = true
                    )
                }
            }
        """
    ] .
<END_SHACL>
