<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047> a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:class nltl:frameStrengthenedPart ;
        sh:message "A main or intermediate ice frame must identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:class nltl:frameEnd ;
        sh:message "A main or intermediate ice frame must identify its lower end."
    ] ;
    sh:sparql [
        sh:message "The lower end must be attached to an allowed supporting structure, or an applicable intermediate frame must satisfy the horizontal-member alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStrengthenedPart ?part .
                $this nltl:hasLowerEnd ?lower .
                FILTER NOT EXISTS {
                    {
                        ?lower nltl:hasAttachedSupportingStructure ?attached .
                        ?attached rdf:type ?attachedType .
                        FILTER (
                            ?attachedType IN (
                                nltl:deckStructure,
                                nltl:tankBoundaryPlating,
                                nltl:tankTop,
                                nltl:iceStringer
                            )
                        )
                    }
                    UNION
                    {
                        $this rdf:type nltl:intermediateIceFrame .
                        ?lower nltl:hasTerminationBelowSupportingStructure ?support .
                        ?support rdf:type ?supportType .
                        FILTER (
                            ?supportType IN (
                                nltl:deckStructure,
                                nltl:tankBoundaryPlating,
                                nltl:tankTop,
                                nltl:iceStringer
                            )
                        )
                        ?support nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
                        ?lower nltl:hasHorizontalConnectionMember ?member .
                        ?member nltl:connectsToAdjacentMainFrame ?adjacentMain .
                        $this nltl:sameScantlingsAsMainFrame true .
                        ?adjacentMain nltl:mainFrameBelowIceBeltStrengthened true .
                    }
                }
            }
        """
    ] .
</END_SHACL>
