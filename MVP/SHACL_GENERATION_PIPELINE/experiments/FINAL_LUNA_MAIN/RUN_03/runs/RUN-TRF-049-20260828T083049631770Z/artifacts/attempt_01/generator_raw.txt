<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF049_QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use millimetres."
    ] .

nltl:TRF049_BracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node nltl:TRF049_QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An installed bracket must have one thickness."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket must be stiffened against buckling."
    ] .

nltl:TRF049_FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [

        sh:minCount 1 ;

        sh:hasValue true ;
        sh:message "The frame attachment must be effectively confirmed."

nltl:TRF049_StructureAttachmentShape

    ] .

nltl:TRF049_LongitudinalSupportingStructureShape
    a sh:NodeShape ;

        sh:path nltl:hasSupportingStructureAttachment ;
        sh:minCount 1 ;
        sh:node nltl:TRF049_BracketedStructureAttachmentShape ;

        sh:path nltl:effectiveAttachmentConfirmed ;

        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:hasConnectionBracket ;

nltl:TRF049_InIceFrameShape

    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;

nltl:TRF049_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-strengthened-area applicability value is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false

nltl:TRF049_BracketThicknessComparisonShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each installed bracket must be at least as thick as the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                $this nltl:frameWebThickness ?frameThickness .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (xsd:decimal(?bracketValue) < xsd:decimal(?frameValue))
            }
        """
    ] .
<END_SHACL>
