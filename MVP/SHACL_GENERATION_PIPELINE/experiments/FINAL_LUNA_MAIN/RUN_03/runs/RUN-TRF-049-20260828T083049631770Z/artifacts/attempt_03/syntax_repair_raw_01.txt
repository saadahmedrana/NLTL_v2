@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF049_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "The frame must declare whether it is within the ice-strengthened area."
            ]
        ]
        [
            sh:node gs:TRF049_InIceFrameShape
        ]
    ) .

gs:TRF049_InIceFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable frame must be within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gs:TRF049_FrameAttachmentShape ;
        sh:message "An in-ice frame must have an effective attachment record."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting structure linked to an in-ice frame must be linked from each applicable frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?structure .
                FILTER NOT EXISTS {
                    ?attachment nltl:hasAttachedSupportingStructure ?structure .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each frame attachment to a supporting structure must be represented by the corresponding reverse attachment link." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasAttachedSupportingStructure ?structure .
                FILTER NOT EXISTS {
                    ?structure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] .

gs:TRF049_FrameAttachmentShape

    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;

        sh:message "Each applicable frame attachment must identify its supporting structure."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node gs:TRF049_ConnectionBracketShape ;
    ] .

gs:TRF049_ConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket must be stiffened against buckling."
    ] ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF049_MillimetreQuantityShape ;
        sh:message "An installed bracket must have a millimetre thickness."
    ] .

gs:TRF049_MillimetreQuantityShape

        sh:maxCount 1 ;
    ] ;
    sh:property [

        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
    ] .

gs:TRF049_LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:node gs:TRF049_InIceFrameShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A longitudinal frame must be attached by a bracket to every linked supporting web frame or bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?structure .
                FILTER NOT EXISTS {
                    ?attachment nltl:hasAttachedSupportingStructure ?structure ;
                                nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] .

gs:TRF049_TransverseFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:node gs:TRF049_InIceFrameShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                    ]
                ]
                [
                    sh:or (
                        [
                            sh:property [
                                sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket ) ;
                                sh:minCount 1 ;
                            ]
                        ]
                        [
                            sh:property [
                                sh:path ( nltl:hasFrameAttachment nltl:similarAttachmentConstructionPresent ) ;
                                sh:minCount 1 ;
                                sh:hasValue true ;
                            ]
                        ]
                    )
                ]
            )
        ]
    ) .

gs:TRF049_ThroughStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A frame running through a supporting structure must connect both sides of the web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount 2 .
                }
            }
        """
    ] .

gs:TRF049_ThicknessComparisonShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Installed bracket and frame web thicknesses must be present in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    $this nltl:withinIceStrengthenedArea true ;
                          nltl:hasFrameAttachment ?attachment ;
                          nltl:frameWebThickness ?frameThickness .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                    FILTER NOT EXISTS {
                        ?bracket nltl:bracketThickness ?bracketThicknessNode .
                        ?bracketThicknessNode qudt:numericValue ?bracketThickness ;
                                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    }
                }
                UNION
                {
                    $this nltl:withinIceStrengthenedArea true ;
                          nltl:hasFrameAttachment ?attachment ;
                          nltl:frameWebThickness ?frameThickness .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                    FILTER NOT EXISTS {
                        $this nltl:frameWebThickness ?frameThicknessNode .
                        ?frameThicknessNode qudt:numericValue ?frameThicknessValue ;
                                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An installed bracket must be at least as thick as the frame web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:hasFrameAttachment ?attachment ;
                      nltl:frameWebThickness ?frameThicknessNode .
                ?frameThicknessNode qudt:numericValue ?frameThickness ;
                                    qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThicknessNode .
                ?bracketThicknessNode qudt:numericValue ?bracketThickness ;
                                       qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (?bracketThickness < ?frameThickness)
            }
        """
    ] .
