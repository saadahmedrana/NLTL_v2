<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF055_NarrowDeckStripShape
    a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:or (
        gen:TRF055_NonApplicableShape
        gen:TRF055_ApplicableShape
    ) ;
    sh:message "The narrow deck strip must either be outside the stated applicability or satisfy the applicable deck-strip requirements." .

gen:TRF055_NonApplicableShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:abreastOfHatch ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:servesAsIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .

gen:TRF055_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:abreastOfHatch ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:servesAsIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForcePerLengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:actualSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:actualShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:evidenceStateApproved
            nltl:evidenceStateDraft
            nltl:evidenceStateExpired
            nltl:evidenceStateRejected
            nltl:evidenceStateRevoked
            nltl:evidenceStateSubmitted
            nltl:evidenceStateUnderReview
        )
    ] ;
    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF055_StandardHatchShape
        gen:TRF055_ApprovedReducedLoadShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual section modulus and actual shear area must meet their respective required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualZ ;
                      nltl:requiredSectionModulus ?requiredZ ;
                      nltl:actualShearArea ?actualA ;
                      nltl:requiredShearArea ?requiredA .
                ?actualZ qudt:numericValue ?actualZValue ;
                         qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?requiredZ qudt:numericValue ?requiredZValue ;
                           qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?actualA qudt:numericValue ?actualAValue ;
                         qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                ?requiredA qudt:numericValue ?requiredAValue ;
                           qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                FILTER (?actualZValue < ?requiredZValue || ?actualAValue < ?requiredAValue)
            }
        """
    ] .

gen:TRF055_StandardHatchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .
