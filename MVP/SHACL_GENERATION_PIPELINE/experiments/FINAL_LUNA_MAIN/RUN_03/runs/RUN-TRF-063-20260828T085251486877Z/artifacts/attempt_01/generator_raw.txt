<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A propeller count is required to determine whether the twin/triple-screw applicability branch applies."
    ] ;
    sh:or (
        gen:TRF063TwinShipBranch
        gen:TRF063TripleShipBranch
        gen:TRF063NonApplicableShipBranch
    ) .

gen:TRF063TwinShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape
    ] .

gen:TRF063TripleShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 3 ;
        sh:maxInclusive 3
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape
    ] .

gen:TRF063NonApplicableShipBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The propeller count must identify a non-twin/non-triple-screw ship for this requirement to be non-applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count .
                FILTER (?count != 2 && ?count != 3)
            }
        """
    ] .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:message "The side-propeller strengthening envelope must identify the tank top as its boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gen:TRF063ForwardExtentQuantityShape ;
        sh:message "The forward strengthening extent must be at least 1.5 m."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gen:TRF063AftExtentQuantityShape ;
        sh:message "The aft strengthening extent must be at least 1.5 m."
    ] .

gen:TRF063ForwardExtentQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF063AftExtentQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
