<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF064ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:detachedStrut ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether detached struts are used."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path ( nltl:hasSidePropellerShafting nltl:enclosedByPlatedBossing ) ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:platedBossingStructure
                ] ;
                sh:message "Side propeller shafting must be enclosed within plated bossing when detached struts are not used."
            ] ;
            sh:property [
                sh:path ( nltl:hasSternTube nltl:enclosedByPlatedBossing ) ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:platedBossingStructure
                ] ;
                sh:message "Stern tubes must be enclosed within plated bossing when detached struts are not used."
            ]
        ]
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:minCount 1 ;
                sh:node gen:TRF064DetachedStrutShape ;
                sh:message "A detached strut and evidence of its design, strength, and hull attachment are required when detached struts are used."
            ]
        ]
    ) .

gen:TRF064DetachedStrutShape
    a sh:NodeShape ;
    sh:class nltl:detachedStrutStructure ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:strutDesignEvidence
        ] ;
        sh:message "Detached strut design evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:strutStrengthEvidence
        ] ;
        sh:message "Detached strut strength evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:strutHullAttachmentEvidence
        ] ;
        sh:message "Detached strut hull-attachment evidence is required."
    ] .
<END_SHACL>
