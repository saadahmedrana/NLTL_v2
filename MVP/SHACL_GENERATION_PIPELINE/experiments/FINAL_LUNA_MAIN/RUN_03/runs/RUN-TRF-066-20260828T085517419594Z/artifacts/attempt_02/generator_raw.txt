<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF_066_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A ship must have exactly one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterline ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record exactly one lower ice waterline input."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in ( nltl:iceClassIa nltl:iceClassIaSuper )
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:property [
                        sh:path rdf:type ;
                        sh:hasValue nltl:iceKnife
                    ] ;
                    sh:property [
                        sh:path nltl:extendsBelowLowerIceWaterline ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ] ;
                sh:message "For ice class IA or IA Super, a practicable ice knife extending below the lower ice waterline must protect the rudder."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in ( nltl:iceClassIa nltl:iceClassIaSuper )
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:property [
                        sh:path rdf:type ;
                        sh:hasValue nltl:equivalentRudderProtectionMeans
                    ]
                ] ;
                sh:message "For ice class IA or IA Super, when an ice knife is not practicable, an equivalent rudder protection means must be provided."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
    ) .
<END_SHACL>
