<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-067> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must provide its ice class to determine applicability of the rudder protection arrangement requirement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IA and IA Super, the ship must have a rudder protection arrangement."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
                FILTER NOT EXISTS {
                    $this nltl:hasRudderProtectionArrangement ?arrangement .
                    ?arrangement rdf:type nltl:rudderProtectionArrangement .
                }
            }
        """
    ] .
<END_SHACL>
