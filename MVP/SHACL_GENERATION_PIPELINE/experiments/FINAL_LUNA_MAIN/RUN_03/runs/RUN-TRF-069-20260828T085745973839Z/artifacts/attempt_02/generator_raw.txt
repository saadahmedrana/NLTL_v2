<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one applicable Finnish-Swedish ice-class value."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The propeller type shall be open or ducted."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The pitch type shall be controllable or fixed."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The maximum ice torque shall be recorded as a decimal quantity in kN*m."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:designCondition ;
                    sh:property [
                        sh:path nltl:normalOperationalConditions ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:radialIceEntryOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "The propulsion-machinery design-condition rule applies only to the specified ice classes, propeller types, and pitch types." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:propellerType ?propellerType ;
                              nltl:pitchType ?pitchType .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            )
                            && ?propellerType IN ("open", "ducted")
                            && ?pitchType IN ("controllable", "fixed")
                        )
                    }
                """
            ]
        ]
    ) .
</BEGIN_SHACL>
