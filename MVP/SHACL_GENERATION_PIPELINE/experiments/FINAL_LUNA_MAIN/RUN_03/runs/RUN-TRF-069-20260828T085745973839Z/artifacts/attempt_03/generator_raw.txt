<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF069DesignConditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Normal operational conditions shall be recorded as true."
    ] ;
    sh:property [
        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Normal service-life propeller/ice load conditions shall be recorded as true."
    ] ;
    sh:property [
        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Fixed-pitch propeller reversal loads shall be recorded as included."
    ] ;
    sh:property [
        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Stopped-propeller dragging shall be recorded as outside the load model."
    ] ;
    sh:property [
        sh:path nltl:radialIceEntryOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Radial ice entry into a turned azimuthing thruster shall be recorded as outside the load model."
    ] .

gen:TRF069ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" )
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" )
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node gen:TRF069DesignConditionShape ;
        sh:message "An applicable propulsion-machinery design condition shall record all required load-model scope conditions."
    ] .

gen:TRF069NonApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The design-condition obligation applies only when the exact ice-class, propeller-type, and pitch-type conjunction is satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )
            }
        """
    ] .

gen:TRF069Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice class is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A pitch type is required to determine applicability."
    ] ;
    sh:or (
        gen:TRF069ApplicableShape
        gen:TRF069NonApplicableShape
    ) .
<END_SHACL>
