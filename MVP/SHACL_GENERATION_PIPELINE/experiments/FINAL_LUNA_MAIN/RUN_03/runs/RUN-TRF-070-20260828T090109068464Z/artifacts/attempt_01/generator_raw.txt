<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF070ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must be identified as having a main-propulsion thruster."
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:or (
            [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "azimuthing"
            ]
            [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "fixed"
            ]
        ) ;
        sh:message "The main-propulsion thruster type must be azimuthing or fixed."
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:KiloN
            ]
        ] ;
        sh:message "The ship must provide the propeller/ice interaction load in kN."
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:KiloN
            ]
        ] ;
        sh:message "The ship must provide the thruster-body/ice interaction load in kN."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MegaPA
            ]
        ] ;
        sh:message "The ship must provide the applicable local ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:designCondition ;
            sh:property [
                sh:path nltl:designConditionLocalStrengthCapacity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:KiloN
                    ]
                ] ;
                sh:message "Each design condition must provide the externally assessed local-strength capacity in kN."
            ]
        ] ;
        sh:message "The ship must link at least one design condition."
    ] ;
    sh:sparql [
        sh:message "The externally assessed local-strength capacity must meet the applicable local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:localIcePressure ?pressureQuantity .
                ?pressureQuantity qudt:numericValue ?pressureValue .
                ?pressureQuantity qudt:unit qudt:MegaPA .
                $this nltl:hasDesignCondition ?condition .
                ?condition nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacityValue .
                ?capacityQuantity qudt:unit qudt:KiloN .
                FILTER (?capacityValue >= ?pressureValue)
            }
        """
    ] .
<END_SHACL>
