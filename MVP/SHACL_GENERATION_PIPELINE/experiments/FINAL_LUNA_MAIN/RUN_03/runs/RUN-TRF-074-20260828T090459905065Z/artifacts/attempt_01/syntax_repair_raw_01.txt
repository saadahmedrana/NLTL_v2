<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074_QuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF074_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:minCount 1 ;
                sh:node gen:TRF074_QuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:minCount 1 ;
                sh:node gen:TRF074_QuantityValueShape
            ] ;
            sh:sparql [
                sh:message "For ice classes IB or IC, when the ballast-condition propeller top depth is below hi, the propulsion system shall be designed for ice class IA." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        ?depthNode qudt:numericValue ?depth ;
                            qudt:unit qudt:M .
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        ?hiNode qudt:numericValue ?hi ;
                            qudt:unit qudt:M .
                        FILTER (?depth < ?hi)
                        FILTER NOT EXISTS {
                            $this nltl:propulsionSystemDesignedForIceClass
                                nltl:iceClassIa
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
    ) .
<END_SHACL>
