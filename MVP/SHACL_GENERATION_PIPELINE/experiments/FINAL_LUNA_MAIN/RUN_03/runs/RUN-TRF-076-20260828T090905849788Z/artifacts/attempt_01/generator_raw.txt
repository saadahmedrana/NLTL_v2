<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF076ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must identify its propeller type as CP or FP for the applicable load-case rule."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:message "The required propeller blade load cases must be represented."
    ] ;
    sh:or (
        gen:TRF076CPShape
        gen:TRF076FPNonReversingShape
        gen:TRF076FPReversingShape
    ) .

gen:TRF076CPShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "CP"
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case1 .
                        ?case1 nltl:propellerBladeLoadCaseNumber 1 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case2 .
                        ?case2 nltl:propellerBladeLoadCaseNumber 2 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case3 .
                        ?case3 nltl:propellerBladeLoadCaseNumber 3 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case4 .
                        ?case4 nltl:propellerBladeLoadCaseNumber 4 .
                    }
                )
            }
        """ ;
        sh:message "CP propellers must cover propeller blade load cases 1 through 4."
    ] .
