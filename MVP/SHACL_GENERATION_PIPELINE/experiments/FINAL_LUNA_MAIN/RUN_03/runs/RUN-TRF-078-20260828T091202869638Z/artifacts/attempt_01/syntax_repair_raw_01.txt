gen:TRF078ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash6
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must have a Table 6-6 lookup case."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash7
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must have a Table 6-7 lookup case."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 multiplied by k2, k3, N_class, and n_n within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?classCase ;
                      nltl:hasTableLookupCase ?locationCase ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?actual .

                ?classCase nltl:tableReference nltl:traficomTable6Dash6 ;
                            nltl:selectedIceClassCycleCount ?nClassNode .
                ?locationCase nltl:tableReference nltl:traficomTable6Dash7 ;
                              nltl:selectedPropellerLocationFactor ?k1Node .

                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nnNode qudt:numericValue ?nn .
                ?nClassNode qudt:numericValue ?nClass .
                ?k1Node qudt:numericValue ?k1 .

                BIND((?k1 * ?k2 * ?k3 * ?nClass * ?nn) AS ?expected)
                FILTER(
                    ABS(xsd:decimal(?actual) - ?expected) >
                    (0.000001 * ABS(?expected) + 0.000001)
                )
            }
        """
    ] .

gen:TRF078TableLookupCaseShape

        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each table lookup case must have exactly one permitted table reference."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash6
