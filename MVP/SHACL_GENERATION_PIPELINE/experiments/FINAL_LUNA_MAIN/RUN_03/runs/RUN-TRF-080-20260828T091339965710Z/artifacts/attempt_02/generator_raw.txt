<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For a ship with a propeller ice interaction component, iceLoadCycleCount and propellerBladeCount must be present as integers, and effectiveIceLoadCycleCount must equal their product within a relative tolerance of 1e-6 plus an absolute tolerance of 1e-9, with unit UNITLESS." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component rdf:type nltl:propellerIceInteractionComponent .

                OPTIONAL {
                    $this nltl:iceLoadCycleCount ?nice .
                }
                OPTIONAL {
                    $this nltl:propellerBladeCount ?blades .
                }
                OPTIONAL {
                    $this nltl:effectiveIceLoadCycleCount ?effective .
                    ?effective qudt:numericValue ?actual .
                    ?effective qudt:unit ?unit .
                }

                FILTER (
                    !BOUND(?nice)
                    || datatype(?nice) != xsd:integer
                    || !BOUND(?blades)
                    || datatype(?blades) != xsd:integer
                    || !BOUND(?effective)
                    || !BOUND(?actual)
                    || datatype(?actual) != xsd:decimal
                    || !BOUND(?unit)
                    || ?unit != qudt:UNITLESS
                    || ABS(
                        ?actual
                        - (xsd:decimal(?nice) * xsd:decimal(?blades))
                    )
                    >
                    (
                        ABS(
                            xsd:decimal(?nice) * xsd:decimal(?blades)
                        ) * 0.000001
                        + 0.000000001
                    )
                )
            }
        """
    ] .
<END_SHACL>
