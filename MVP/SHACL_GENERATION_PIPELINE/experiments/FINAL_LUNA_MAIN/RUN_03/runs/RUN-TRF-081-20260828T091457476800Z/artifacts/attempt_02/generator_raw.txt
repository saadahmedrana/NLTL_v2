<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF081Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "Ice thrust and propeller shaft design thrust values must satisfy the TRF-081 formulas within a relative tolerance of 1e-6, with an absolute fallback of 1e-6 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce ?ffNode ;
                      nltl:backwardBladeForce ?fbNode ;
                      nltl:hydrodynamicBollardThrust ?bollardNode ;
                      nltl:forwardIceThrust ?tfNode ;
                      nltl:backwardIceThrust ?tbNode ;
                      nltl:forwardPropellerShaftDesignThrust ?forwardDesignNode ;
                      nltl:backwardPropellerShaftDesignThrust ?backwardDesignNode ;
                      nltl:propellerShaftDesignThrust ?designNode .

                ?ffNode qudt:numericValue ?ff .
                ?fbNode qudt:numericValue ?fb .
                ?bollardNode qudt:numericValue ?bollard .
                ?tfNode qudt:numericValue ?tf .
                ?tbNode qudt:numericValue ?tb .
                ?forwardDesignNode qudt:numericValue ?forwardDesign .
                ?backwardDesignNode qudt:numericValue ?backwardDesign .
                ?designNode qudt:numericValue ?design .

                BIND (1.1 * ?ff AS ?expectedTf)
                BIND (1.1 * ?fb AS ?expectedTb)
                BIND (?bollard + 2.2 * ?expectedTf AS ?forwardCandidate)
                BIND (1.5 * ?expectedTb AS ?backwardCandidate)
                BIND (IF(?forwardCandidate >= ?backwardCandidate, ?forwardCandidate, ?backwardCandidate) AS ?expectedDesign)

                BIND (IF(?expectedTf >= 0, ?expectedTf, -?expectedTf) AS ?absExpectedTf)
                BIND (IF(?expectedTb >= 0, ?expectedTb, -?expectedTb) AS ?absExpectedTb)
                BIND (IF(?forwardCandidate >= 0, ?forwardCandidate, -?forwardCandidate) AS ?absForwardCandidate)
                BIND (IF(?backwardCandidate >= 0, ?backwardCandidate, -?backwardCandidate) AS ?absBackwardCandidate)
                BIND (IF(?expectedDesign >= 0, ?expectedDesign, -?expectedDesign) AS ?absExpectedDesign)

                FILTER (
                    IF(?tf >= ?expectedTf, ?tf - ?expectedTf, ?expectedTf - ?tf)
                    >
                    IF(0.000001 * ?absExpectedTf > 0.000001, 0.000001 * ?absExpectedTf, 0.000001)
                    ||
                    IF(?tb >= ?expectedTb, ?tb - ?expectedTb, ?expectedTb - ?tb)
                    >
                    IF(0.000001 * ?absExpectedTb > 0.000001, 0.000001 * ?absExpectedTb, 0.000001)
                    ||
                    IF(?forwardDesign >= ?expectedDesign, ?forwardDesign - ?expectedDesign, ?expectedDesign - ?forwardDesign)
                    >
                    IF(0.000001 * ?absExpectedDesign > 0.000001, 0.000001 * ?absExpectedDesign, 0.000001)
                    ||
                    IF(?backwardDesign >= ?expectedDesign, ?backwardDesign - ?expectedDesign, ?expectedDesign - ?backwardDesign)
                    >
                    IF(0.000001 * ?absExpectedDesign > 0.000001, 0.000001 * ?absExpectedDesign, 0.000001)
                    ||
                    IF(?design >= ?expectedDesign, ?design - ?expectedDesign, ?expectedDesign - ?design)
                    >
                    IF(0.000001 * ?absExpectedDesign > 0.000001, 0.000001 * ?absExpectedDesign, 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
