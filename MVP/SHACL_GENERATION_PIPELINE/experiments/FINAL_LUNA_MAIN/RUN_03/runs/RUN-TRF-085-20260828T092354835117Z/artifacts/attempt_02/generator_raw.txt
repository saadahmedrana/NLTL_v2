<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF085KnownSpeedBranch
        gen:TRF085TableLookupBranch
    ) .

gen:TRF085KnownSpeedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must identify bollard propeller speed as known when the supplied speed is used."
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
            ]
        ] ;
        sh:message "A known bollard propeller speed at MCR must be supplied in revolutions per minute."
    ] .

gen:TRF085TableLookupBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The table-selection branch applies when bollard propeller speed is not known."
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "Propeller diameter must be supplied in metres for Table 6-9 selection."
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
            ]
        ] ;
        sh:message "The Table 6-9 result must be recorded as the bollard propeller speed in revolutions per minute."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:TRF085TableLookupCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When bollard propeller speed is unknown, each table lookup case must bind its input to the ship's propeller diameter and its result to the ship's bollard propeller speed, with the required units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:bollardPropellerSpeedKnown false .
                $this nltl:hasTableLookupCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:propellerDiameter ?diameter .
                    ?diameter qudt:unit <http://qudt.org/vocab/unit/M> .

                    $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed .
                    ?speed qudt:unit <http://qudt.org/vocab/unit/REV-PER-MIN> .

                    ?case nltl:lookupInputQuantity ?diameter ;
                          nltl:lookupResultQuantity ?speed .
                }
            }
        """
    ] .
