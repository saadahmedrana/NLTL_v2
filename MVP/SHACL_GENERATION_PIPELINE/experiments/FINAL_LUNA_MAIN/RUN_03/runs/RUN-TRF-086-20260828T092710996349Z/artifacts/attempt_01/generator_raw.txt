<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-086-LengthQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-086-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:pattern "^[^C].*$"
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:pattern "^C([^P].*|P.+)$"
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP"
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF-086-LengthQuantityShape>
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP"
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF-086-LengthQuantityShape>
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF-086-LengthQuantityShape>
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For a CP propeller with unknown pitch value, propellerPitchAtMcrBollard must equal 0.7 times propellerPitchAtMcrFreeRunning within the specified tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerType "CP" ;
                      nltl:pitchValueKnown false ;
                      nltl:propellerPitchAtMcrBollard ?bollard ;
                      nltl:propellerPitchAtMcrFreeRunning ?freeRunning .

                ?bollard qudt:numericValue ?bollardValue ;
                         qudt:unit unit:M .
                ?freeRunning qudt:numericValue ?freeRunningValue ;
                             qudt:unit unit:M .

                FILTER (
                    datatype(?bollardValue) = xsd:decimal &&
                    datatype(?freeRunningValue) = xsd:decimal
                )

                BIND((0.7 * ?freeRunningValue) AS ?expectedValue)
                BIND((0.000001 * ABS(?expectedValue) + 0.000000001) AS ?tolerance)

                FILTER(ABS(?bollardValue - ?expectedValue) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
