<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF091ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091LoadCaseShape ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:equals nltl:hasPropulsionLoadCase ;
        sh:message "The ice-milling load sequence case shall be the propulsion-line strength evaluation load case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091LoadCaseShape ;
        sh:class nltl:loadCase ;
        sh:message "A propulsion-line strength evaluation load case is required."
    ] .

gen:TRF091LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:minCount 1 ;
        sh:in ( nltl:propulsionLineStrengthEvaluationPurpose ) ;
        sh:message "The load case shall be designated for propulsion-line strength evaluation."
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:loadCasePurpose ;
            sh:hasValue nltl:stallingAnalysisPurposeValue
        ]
    ] .
<END_SHACL>
