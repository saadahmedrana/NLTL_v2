<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlg:TRF108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the SN-curve is known."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "When the SN-curve is known, its type must be recorded on the ship."
            ] ;
            sh:property [
                sh:path nltl:propellerBladeMaterialProperty ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "When the SN-curve is known, the blade material properties needed for correspondence must be recorded."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "When the SN-curve is known, the selected curve must be identified."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Evidence must demonstrate that the selected SN-curve corresponds to the blade material properties."
            ] ;
            sh:sparql [
                sh:message "The selected SN-curve, its type, blade material properties, and correspondence evidence must be asserted together on the ship." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:stressLifeCurveKnown true ;
                              nltl:stressLifeCurveType ?curveType ;
                              nltl:propellerBladeMaterialProperty ?material ;
                              nltl:stressLifeCurveSelection ?selectedCurve ;
                              nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:hasValue nltl:twoSlopeStressLifeCurve ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "When the SN-curve is unknown, the two-slope SN curve shall be used."
            ]
        ]
    ) .
<END_SHACL>
