@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-114-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:minCount 1 ;

                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A non-hemispherical impact case requires an equivalent impact sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?value ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed thruster resistance capacity must be at least the impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:thrusterIceImpactDemand ?demandNode ;
                      nltl:thrusterImpactCaseAppliesToBody ?body .
                ?demandNode qudt:numericValue ?demand ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?body nltl:thrusterResistanceCapacity ?capacityNode .
                ?capacityNode qudt:numericValue ?capacity ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                FILTER (
                    datatype(?demand) = xsd:decimal &&
                    datatype(?capacity) = xsd:decimal &&
                    ?capacity >= ?demand
                )
            }
        """
    ] .
