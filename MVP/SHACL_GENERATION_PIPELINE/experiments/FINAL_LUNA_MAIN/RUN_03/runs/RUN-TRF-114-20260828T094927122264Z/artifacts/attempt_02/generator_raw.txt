@prefix gs: <urn:nltl:generated-shape:> .

@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF114ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide a typical ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node gs:TRF114LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide an applicable thruster ice impact load case."
    ] .

gs:TRF114LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:in (
            nltl:traficomTable6Dash3Reference
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:in (
            nltl:traficomTable6Dash16Reference
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must reference TRAFICOM Table 6-16 for the impact load case."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The impact case must identify the thruster body and its resistance capacity."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:in (
            nltl:hemisphericalContactGeometry
            nltl:nonHemisphericalContactGeometry
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The contact geometry must be classified as hemispherical or non-hemispherical."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The load case must include evidence for the loaded impact area."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        sh:message "The externally assessed thruster resistance capacity must be at least the thruster ice impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:thrusterImpactCaseAppliesToBody ?body .
                        ?body nltl:thrusterResistanceCapacity ?capacity .
                        ?capacity qudt:numericValue ?capacityValue .
                        $this nltl:thrusterIceImpactDemand ?demand .
                        ?demand qudt:numericValue ?demandValue .
                        FILTER (
                            xsd:decimal(?capacityValue) >=
                            xsd:decimal(?demandValue)
                        )
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical contact geometry requires a valid ship-owned equivalent impact sphere radius and correspondence evidence on the load case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:contactGeometryClassification
                    nltl:nonHemisphericalContactGeometry .
                ?ship nltl:hasThrusterIceImpactLoadCase $this .
                FILTER (
                    NOT EXISTS {
                        ?ship nltl:equivalentImpactSphereRadius ?radius .
                        ?radius qudt:numericValue ?radiusValue .
                        ?radius qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (DATATYPE(?radiusValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:contactGeometryCorrespondenceEvidence ?evidence .
                        ?evidence a nltl:evidenceArtifact .
                    }
                )
            }
        """
    ] .
<END_SHACL>
