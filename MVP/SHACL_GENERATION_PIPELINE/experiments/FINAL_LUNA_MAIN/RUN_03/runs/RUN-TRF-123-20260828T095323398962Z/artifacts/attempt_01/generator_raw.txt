<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-123-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node <urn:nltl:generated-shape:TRF-123-OccasionalForceLoadCaseShape>
    ] .

<urn:nltl:generated-shape:TRF-123-OccasionalForceLoadCaseShape>
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The occasional force load case must confirm that the propeller blade is excluded from its scope."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-SafetyFactorQuantityShape> ;
        sh:message "The occasional force load case must provide the externally calculated component yield safety factor."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-AssessedComponentShape> ;
        sh:message "The applicable occasional force load case must identify the assessed components."
    ] ;
    sh:sparql [
        sh:message "The externally calculated occasional force component stress, yield strength, and safety factor do not satisfy the material yield limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:componentYieldSafetyFactor ?sf .
                ?sf qudt:numericValue ?sfValue .

                $this nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stress ;
                           nltl:hasMaterialProperties ?materialProperties .
                ?stress qudt:numericValue ?stressValue .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldStrength .
                ?yieldStrength qudt:numericValue ?yieldValue .

                FILTER (
                    ?stressValue * ?sfValue > ?yieldValue
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-123-AssessedComponentShape>
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-StressQuantityShape> ;
        sh:message "Each assessed component must provide its externally calculated occasional force stress."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-MaterialPropertiesShape> ;
        sh:message "Each assessed component must identify its material properties."
    ] .

<urn:nltl:generated-shape:TRF-123-MaterialPropertiesShape>
    a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-YieldStrengthQuantityShape> ;
        sh:message "The material properties must provide the component material yield strength."
    ] .

<urn:nltl:generated-shape:TRF-123-StressQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-123-YieldStrengthQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-123-SafetyFactorQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
