<urn:nltl:generated-shape:TRF-123-OccasionalForceLoadCaseShape>
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The occasional-force case must explicitly exclude propeller blades."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-ShipComponentShape> ;
        sh:message "The occasional-force case must identify each assessed component."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-123-DimensionlessQuantityValueShape> ;
        sh:message "The load case must provide the externally calculated component yield safety factor."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally calculated stress, material yield strength, and safety factor must be connected through the same occasional-force load case and assessed component." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:hasMaterialProperties ?materials .
                ?component nltl:occasionalForceComponentStress ?stress .
                ?materials nltl:componentMaterialYieldStrength ?yield .
                $this nltl:componentYieldSafetyFactor ?factor .
                ?stress qudt:numericValue ?stressValue .
                ?yield qudt:numericValue ?yieldValue .
                ?factor qudt:numericValue ?factorValue .
                FILTER (
                    !BOUND(?stressValue) ||
                    !BOUND(?yieldValue) ||
                    !BOUND(?factorValue)
                )
            }
        """
    ] .
