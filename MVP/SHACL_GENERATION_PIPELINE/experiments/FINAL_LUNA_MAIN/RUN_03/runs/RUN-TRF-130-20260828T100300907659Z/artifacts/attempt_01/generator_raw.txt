    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;

        sh:node [
            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

                sh:maxCount 1
            ] ;

                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;

            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node gen:InletChestInputShape
    ] ;
    sh:or (
        gen:StandardCoolingChestCaseShape
        gen:AlternativeCoolingChestCaseShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked inlet chest must report a recommended volume approximately equal to total relevant engine power divided by 750 kW, and its strainer open area must be at least four times the inlet pipe area. Volume comparison uses a relative tolerance of 1e-6 plus an absolute fallback of 1e-9 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?chest .
                ?chest a nltl:inletChest ;
                    nltl:recommendedChestVolume ?recommended ;
                    nltl:strainerOpenArea ?openArea ;
                    nltl:inletPipeArea ?pipeArea .
                $this nltl:totalRelevantEnginePower ?power .
                ?recommended qudt:numericValue ?recommendedValue ;
                    qudt:unit unit:M3 .
                ?power qudt:numericValue ?powerValue ;
                    qudt:unit unit:KiloW .
                ?openArea qudt:numericValue ?openValue ;
                    qudt:unit unit:M2 .
                ?pipeArea qudt:numericValue ?pipeValue ;
                    qudt:unit unit:M2 .
                FILTER (
                    ABS(?recommendedValue - (?powerValue / 750))
                        > (ABS(?powerValue / 750) * 0.000001 + 0.000000001)
                    || ?openValue < (4 * ?pipeValue)
                )
            }
        """
    ] .
