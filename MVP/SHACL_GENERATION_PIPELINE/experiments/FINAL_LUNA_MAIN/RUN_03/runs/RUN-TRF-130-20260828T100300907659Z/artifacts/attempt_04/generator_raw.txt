gs:CommonInletChestShape a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:RatioQuantityShape
    ] ;
    sh:sparql [
        sh:message "Strainer open area must be at least four times the inlet pipe area, and strainer open area ratio must equal strainer open area divided by inlet pipe area within a 1e-6 tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:strainerOpenArea ?areaNode .
                ?areaNode qudt:numericValue ?area .
                $this nltl:inletPipeArea ?pipeNode .
                ?pipeNode qudt:numericValue ?pipeArea .
                $this nltl:strainerOpenAreaRatio ?ratioNode .
                ?ratioNode qudt:numericValue ?ratio .
                BIND((?area / ?pipeArea) AS ?expectedRatio)
                BIND((?ratio - ?expectedRatio) AS ?difference)
                FILTER(
                    ?area < (4 * ?pipeArea) ||
                    !(
                        (
                            ?difference <= (?expectedRatio * 0.000001) &&
                            ?difference >= (-1 * ?expectedRatio * 0.000001)
                        ) ||
                        (
                            ?difference <= 0.000001 &&
                            ?difference >= -0.000001
                        )
                    )
                )
            }
        """
    ] .

        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Recommended chest volume must equal total relevant engine power divided by 750 within a 1e-6 relative tolerance, with a 1e-6 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this .
                ?ship nltl:totalRelevantEnginePower ?powerNode .
                ?powerNode qudt:numericValue ?power .
                $this nltl:recommendedChestVolume ?volumeNode .
                ?volumeNode qudt:numericValue ?volume .
                BIND((?power / 750) AS ?expectedVolume)
                BIND((?volume - ?expectedVolume) AS ?difference)
                FILTER(
                    !(
                        (
                            ?difference <= (?expectedVolume * 0.000001) &&
                            ?difference >= (-1 * ?expectedVolume * 0.000001)
                        ) ||
                        (
                            ?difference <= 0.000001 &&
                            ?difference >= -0.000001
                        )
                    )
                )
            }
        """
    ] .
