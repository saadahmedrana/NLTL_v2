gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice-class value."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:node gen:TRF133WarningTriangleShape ;
        sh:minCount 1 ;
        sh:message "The ship must have a conforming warning triangle marking."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:node gen:TRF133IceClassDraughtMarkingShape ;
        sh:minCount 1 ;
        sh:message "The ship must have a conforming ice-class draught marking."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The timber load line applicability selector is required to select the draught-mark reference."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:timberLoadLineVerticalReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:loadLineRingCentreReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "The timber load line selector and the corresponding draught-mark reference must form a permitted branch." .
