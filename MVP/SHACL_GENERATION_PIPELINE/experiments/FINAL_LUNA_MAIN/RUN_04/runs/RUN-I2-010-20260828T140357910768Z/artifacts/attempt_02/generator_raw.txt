<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-010> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue
    ] ;
    sh:sparql [
        sh:message "For a ship assigned PC6 or PC7 with a vertical-sided bow, every represented normal frame angle must be a decimal quantity in the inclusive range 0 to 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?pc .
                FILTER (
                    ?pc IN (
                        nltl:polarClassPc6,
                        nltl:polarClassPc7
                    )
                )
                $this nltl:verticalSidedBowForm ?vertical .
                FILTER (?vertical = true)
                $this nltl:normalFrameAngle ?angle .
                FILTER (
                    NOT EXISTS {
                        ?angle qudt:numericValue ?missingValue
                    }
                    ||
                    EXISTS {
                        ?angle qudt:numericValue ?nonDecimalValue .
                        FILTER (datatype(?nonDecimalValue) != xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?angle qudt:numericValue ?numericValue .
                        FILTER (
                            datatype(?numericValue) = xsd:decimal &&
                            (
                                ?numericValue < 0 ||
                                ?numericValue > 10
                            )
                        )
                    }
                    ||
                    NOT EXISTS {
                        ?angle qudt:unit qudt:DEG
                    }
                    ||
                    EXISTS {
                        ?angle qudt:unit ?wrongUnit .
                        FILTER (?wrongUnit != qudt:DEG)
                    }
                )
            }
        """
    ] .
<END_SHACL>
