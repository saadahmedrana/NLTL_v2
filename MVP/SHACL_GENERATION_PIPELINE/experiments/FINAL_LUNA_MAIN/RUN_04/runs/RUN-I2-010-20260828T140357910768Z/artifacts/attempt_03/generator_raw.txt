<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
                sh:message "The applicable ship must have Polar Class PC6 or PC7."
            ] ;
            sh:property [
                sh:path nltl:verticalSidedBowForm ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "The applicable ship must have a bow form with vertical sides."
            ] ;
            sh:property [
                sh:path nltl:bowVerticalSideStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "The applicable ship must record a vertical-sided bow status."
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethod ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "The applicable ship must identify the design ice force method."
            ] ;
            sh:property [
                sh:path nltl:normalFrameAngle ;
                sh:node gen:I2_010_NormalFrameAngleShape ;
                sh:message "Every represented considered-sub-region normal frame angle must be between 0 and 10 degrees inclusive."
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                ) ;
                sh:message "A ship outside the PC6 or PC7 applicability scope must have a Polar Class other than PC6 or PC7."
            ]
        ]
        [
            sh:property [
                sh:path nltl:verticalSidedBowForm ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:message "A ship outside the vertical-sided-bow applicability scope must record a non-vertical-sided bow form."
            ]
        ]
    ) .

gen:I2_010_NormalFrameAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 0 ;
        sh:maxInclusive 10 ;
        sh:message "The normal frame angle must be between 0 and 10 degrees inclusive."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The normal frame angle must use the degree unit."
    ] .
<END_SHACL>
