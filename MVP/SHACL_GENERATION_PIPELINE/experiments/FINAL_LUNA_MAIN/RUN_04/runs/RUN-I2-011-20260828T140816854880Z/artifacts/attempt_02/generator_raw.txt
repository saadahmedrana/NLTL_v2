gen:I2_011_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:in (
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_QuantityRatioShape ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive "0.6"^^xsd:decimal ;
                sh:maxInclusive "0.6"^^xsd:decimal
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_QuantityRatioShape ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive "1.3"^^xsd:decimal ;
                sh:maxInclusive "1.3"^^xsd:decimal
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_QuantityForceMNShape
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_QuantityForceMNShape
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_QuantityForceKNShape
    ] ;
    sh:sparql [
        sh:message "For an applicable PC6 or PC7 bulbous-bow ship, bowDesignForce must equal the part-IV force within a relative tolerance of 1e-6 (with an absolute tolerance of 1e-6 kN near zero) and must not be less than the part-III force." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?partIIINode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?partIVNode ;
                      nltl:bowDesignForce ?resultNode .

                ?partIIINode qudt:numericValue ?partIIIValue ;
                             qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                ?partIVNode qudt:numericValue ?partIVValue ;
                             qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                ?resultNode qudt:numericValue ?resultValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                BIND(?partIVValue * 1000 AS ?expectedResult)
                BIND(?partIIIValue * 1000 AS ?minimumResult)

                FILTER (
                    ?resultValue < ?minimumResult ||
                    ABS(?resultValue - ?expectedResult) >
                    IF(
                        ABS(?expectedResult) > 0.000001,
                        ABS(?expectedResult) * 0.000001,
                        0.000001
                    )
                )
            }
        """
    ] .
