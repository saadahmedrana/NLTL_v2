<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "bulbousBowPresent is required." ;
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:message "polarClass must be one of the permitted Polar Class values." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
            ] ;
            sh:property [
                sh:path nltl:shapeCoefficient ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "0.6"^^xsd:decimal ;
                        sh:maxInclusive "0.6"^^xsd:decimal ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:UNITLESS ;
                    ] ;
                ] ;
                sh:message "shapeCoefficient must be exactly 0.6 and dimensionless for the applicable case." ;
            ] ;
            sh:property [
                sh:path nltl:loadPatchAspectRatio ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "1.3"^^xsd:decimal ;
                        sh:maxInclusive "1.3"^^xsd:decimal ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:UNITLESS ;
                    ] ;
                ] ;
                sh:message "loadPatchAspectRatio must be exactly 1.3 and dimensionless for the applicable case." ;
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:MegaN ;
                    ] ;
                ] ;
                sh:message "The part-III bow design force is required in MN for the applicable case." ;
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:MegaN ;
                    ] ;
                ] ;
                sh:message "The part-IV bow design force is required in MN for the applicable case." ;
            ] ;
            sh:property [
                sh:path nltl:bowDesignForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:KiloN ;
                    ] ;
                ] ;
                sh:message "The bow design force is required in kN for the applicable case." ;
            ] ;
            sh:sparql [
                sh:message "For PC6 or PC7 ships with bulbous bows, bowDesignForce must equal the part-IV force within tolerance and must not be less than the part-III force." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                              nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode ;
                              nltl:bowDesignForce ?resultNode .
                        ?iiiNode qudt:numericValue ?iii .
                        ?ivNode qudt:numericValue ?iv .
                        ?resultNode qudt:numericValue ?reported .
                        BIND((?iv * 1000) AS ?expected)
                        BIND((?iii * 1000) AS ?lower)
                        BIND((?reported - ?expected) AS ?rawDelta)
                        BIND(IF(?rawDelta >= 0, ?rawDelta, 0 - ?rawDelta) AS ?delta)
                        BIND(IF(?expected >= 0, ?expected, 0 - ?expected) AS ?scale)
                        BIND(
                            IF(
                                (?scale * 0.000001) > 0.000001,
                                (?scale * 0.000001),
                                0.000001
                            )
                            AS ?tolerance
                        )
                        FILTER(
                            (?delta > ?tolerance) ||
                            (?reported < ?lower)
                        )
                    }
                """ ;
            ]
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:bulbousBowPresent ;
                        sh:hasValue false ;
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:polarClass ;
                        sh:in (
                            nltl:polarClassPc1
                            nltl:polarClassPc2
                            nltl:polarClassPc3
                            nltl:polarClassPc4
                            nltl:polarClassPc5
                        ) ;
                    ]
                ]
            )
        ]
    ) .
