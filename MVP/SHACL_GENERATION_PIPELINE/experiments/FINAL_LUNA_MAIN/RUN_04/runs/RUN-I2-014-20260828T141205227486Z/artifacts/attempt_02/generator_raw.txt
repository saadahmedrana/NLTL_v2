    sh:sparql [
        sh:message "Each bow subregion midpoint must equal LBOW/8, 3*LBOW/8, 5*LBOW/8, or 7*LBOW/8 within a 1e-6 relative tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lbq ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?lbq qudt:numericValue ?lb .
                ?case nltl:bowSubregionMidLengthPosition ?midq .
                ?midq qudt:numericValue ?mid .
                FILTER (
                    ABS(?mid - (?lb / 8)) >
                        IF(ABS(?lb / 8) > 0, 0.000001 * ABS(?lb / 8), 0.000001)
                    &&
                    ABS(?mid - (3 * ?lb / 8)) >
                        IF(ABS(3 * ?lb / 8) > 0, 0.000001 * ABS(3 * ?lb / 8), 0.000001)
                    &&
                    ABS(?mid - (5 * ?lb / 8)) >
                        IF(ABS(5 * ?lb / 8) > 0, 0.000001 * ABS(5 * ?lb / 8), 0.000001)
                    &&
                    ABS(?mid - (7 * ?lb / 8)) >
                        IF(ABS(7 * ?lb / 8) > 0, 0.000001 * ABS(7 * ?lb / 8), 0.000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The four bow subregion cases must cover all four equal-length midpoint positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lbq .
                ?lbq qudt:numericValue ?lb .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?c1 .
                        ?c1 nltl:bowSubregionMidLengthPosition ?m1q .
                        ?m1q qudt:numericValue ?m1 .
                        FILTER (
                            ABS(?m1 - (?lb / 8)) <=
                            IF(ABS(?lb / 8) > 0, 0.000001 * ABS(?lb / 8), 0.000001)
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?c2 .
                        ?c2 nltl:bowSubregionMidLengthPosition ?m2q .
                        ?m2q qudt:numericValue ?m2 .
                        FILTER (
                            ABS(?m2 - (3 * ?lb / 8)) <=
                            IF(ABS(3 * ?lb / 8) > 0, 0.000001 * ABS(3 * ?lb / 8), 0.000001)
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?c3 .
                        ?c3 nltl:bowSubregionMidLengthPosition ?m3q .
                        ?m3q qudt:numericValue ?m3 .
                        FILTER (
                            ABS(?m3 - (5 * ?lb / 8)) <=
                            IF(ABS(5 * ?lb / 8) > 0, 0.000001 * ABS(5 * ?lb / 8), 0.000001)
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?c4 .
                        ?c4 nltl:bowSubregionMidLengthPosition ?m4q .
                        ?m4q qudt:numericValue ?m4 .
                        FILTER (
                            ABS(?m4 - (7 * ?lb / 8)) <=
                            IF(ABS(7 * ?lb / 8) > 0, 0.000001 * ABS(7 * ?lb / 8), 0.000001)
                        )
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each selected maximum bow value must equal the maximum of the four reported case values within a 1e-6 relative tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?sfq ;
                      nltl:selectedMaximumBowLineLoad ?slq ;
                      nltl:selectedMaximumBowPressure ?spq .
                ?sfq qudt:numericValue ?sf .
                ?slq qudt:numericValue ?sl .
                ?spq qudt:numericValue ?sp .

                {
                    SELECT (MAX(?fv) AS ?maxF)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?fc .
                        ?fc nltl:bowSubregionCalculatedForce ?fq .
                        ?fq qudt:numericValue ?fv .
                    }
                }
                {
                    SELECT (MAX(?lv) AS ?maxL)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?lc .
                        ?lc nltl:bowSubregionCalculatedLineLoad ?lq .
                        ?lq qudt:numericValue ?lv .
                    }
                }
                {
                    SELECT (MAX(?pv) AS ?maxP)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?pc .
                        ?pc nltl:bowSubregionCalculatedPressure ?pq .
                        ?pq qudt:numericValue ?pv .
                    }
                }

                FILTER (
                    ABS(?sf - ?maxF) >
                    IF(ABS(?maxF) > 0, 0.000001 * ABS(?maxF), 0.000001)
                    ||
                    ABS(?sl - ?maxL) >
                    IF(ABS(?maxL) > 0, 0.000001 * ABS(?maxL), 0.000001)
                    ||
                    ABS(?sp - ?maxP) >
                    IF(ABS(?maxP) > 0, 0.000001 * ABS(?maxP), 0.000001)
                )
            }
        """
    ] .
