<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-015-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:node <urn:nltl:generated-shape:I2-015-BowCaseShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable bow subregion calculation case requires a valid upper ice waterline length L_UI on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasBowSubregionCalculationCase ?case .
                ?case nltl:bowFormApplicabilityClassification
                    nltl:i231vBowFormApplicability .
                OPTIONAL {
                    $this nltl:upperIceWaterlineLengthLUI ?lengthQuantity .
                    ?lengthQuantity qudt:numericValue ?lengthValue ;
                        qudt:unit qudt:M .
                    FILTER (datatype(?lengthValue) = xsd:decimal)
                }
                FILTER (!BOUND(?lengthQuantity) || !BOUND(?lengthValue))
            }
        """
    ] .

<urn:nltl:generated-shape:I2-015-BowCaseShape>
    a sh:NodeShape ;
    sh:class nltl:bowSubregionCalculationCase ;
    sh:property [
        sh:path nltl:bowFormApplicabilityClassification ;
        sh:class nltl:bowFormApplicabilityValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable bow subregion calculation case requires all declared formula inputs, the failure-factor lookup relationship, and the externally calculated bow shape coefficient with the canonical units and datatypes." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowFormApplicabilityClassification
                    nltl:i231vBowFormApplicability .

                OPTIONAL {
                    $this nltl:bowSubregionMidLengthPosition ?xQuantity .
                    ?xQuantity qudt:numericValue ?xValue ;
                        qudt:unit qudt:M .
                    FILTER (datatype(?xValue) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:bowSubregionWaterlineAngle ?alphaQuantity .
                    ?alphaQuantity qudt:numericValue ?alphaValue ;
                        qudt:unit qudt:DEG .
                    FILTER (datatype(?alphaValue) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:betaIPrime ?betaQuantity .
                    ?betaQuantity qudt:numericValue ?betaValue ;
                        qudt:unit qudt:UNITLESS .
                    FILTER (datatype(?betaValue) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:hasFailureClassFactorLookupCase ?lookup .
                    ?lookup a nltl:tableLookupCase ;
                        nltl:flexuralFailureClassFactor ?flexuralQuantity ;
                        nltl:crushingFailureClassFactor ?crushingQuantity .
                    ?flexuralQuantity qudt:numericValue ?flexuralValue ;
                        qudt:unit qudt:UNITLESS .
                    ?crushingQuantity qudt:numericValue ?crushingValue ;
                        qudt:unit qudt:UNITLESS .
                    FILTER (datatype(?flexuralValue) = xsd:decimal)
                    FILTER (datatype(?crushingValue) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:bowShapeCoefficient ?resultQuantity .
                    ?resultQuantity qudt:numericValue ?resultValue ;
                        qudt:unit qudt:UNITLESS .
                    FILTER (datatype(?resultValue) = xsd:decimal)
                }

                FILTER (
                    !BOUND(?xQuantity) ||
                    !BOUND(?xValue) ||
                    !BOUND(?alphaQuantity) ||
                    !BOUND(?alphaValue) ||
                    !BOUND(?betaQuantity) ||
                    !BOUND(?betaValue) ||
                    !BOUND(?lookup) ||
                    !BOUND(?flexuralQuantity) ||
                    !BOUND(?flexuralValue) ||
                    !BOUND(?crushingQuantity) ||
                    !BOUND(?crushingValue) ||
                    !BOUND(?resultQuantity) ||
                    !BOUND(?resultValue)
                )
            }
        """
    ] .
<END_SHACL>
