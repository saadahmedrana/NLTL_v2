    sh:property [
        sh:path nltl:hasIceLoadPatch ;

    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;

    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;

            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;

    sh:property [
        sh:path nltl:hasPeakPressureFactorLookupCase ;
        sh:class nltl:peakPressureFactorLookupCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-018-PeakLookupCaseShape>
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-018-TableLookupCaseShape>
    ] ;
    sh:sparql [
        sh:message "Average ice pressure must equal ice force divided by load patch height and length, with kN converted to MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?averageQ ;
                      nltl:iceForce ?forceQ ;
                      nltl:loadPatchHeight ?heightQ ;
                      nltl:loadPatchLength ?lengthQ .
                ?averageQ qudt:numericValue ?average .
                ?forceQ qudt:numericValue ?force .
                ?heightQ qudt:numericValue ?height .
                ?lengthQ qudt:numericValue ?length .
                BIND ((?force / (?height * ?length)) * 0.001 AS ?expected)
                FILTER (
                    ABS(?average - ?expected) >
                    IF(ABS(?expected) * 0.000001 > 0.000000001,
                       ABS(?expected) * 0.000001,
                       0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship peak pressure factor must match the selected factor from each applicable Table 3 lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor ?reportedQ ;
                      nltl:hasPeakPressureFactorLookupCase ?case .
                ?reportedQ qudt:numericValue ?reported .
                ?case nltl:selectedPeakPressureFactor ?selectedQ .
                ?selectedQ qudt:numericValue ?selected .
                FILTER (
                    ABS(?reported - ?selected) >
                    IF(ABS(?selected) * 0.000001 > 0.000000001,
                       ABS(?selected) * 0.000001,
                       0.000000001)
                )
            }
        """
    ] .
