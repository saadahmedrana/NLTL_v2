@prefix nltl: <https://w3id.org/nltl/vocab#> .

@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-018-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:minCount 1 ;
        sh:message "The ship must identify its design ice load patch."
    ] ;
    sh:property [
        sh:path nltl:iceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPeakPressureFactorLookupCase ;
        sh:class nltl:peakPressureFactorLookupCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-018-PeakLookupCaseShape>
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-018-TableLookupCaseShape>
    ] ;
    sh:sparql [
        sh:message "averageIcePressure must equal iceForce converted from kN to MN divided by loadPatchHeight times loadPatchLength, within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceForce ?forceQ ;
                      nltl:loadPatchHeight ?heightQ ;
                      nltl:loadPatchLength ?lengthQ ;
                      nltl:averageIcePressure ?averageQ .
                ?forceQ qudt:numericValue ?force .
                ?heightQ qudt:numericValue ?height .
                ?lengthQ qudt:numericValue ?length .
                ?averageQ qudt:numericValue ?reported .
                FILTER (?height > 0)
                FILTER (?length > 0)
                BIND ((?force * 0.001) / (?height * ?length) AS ?expected)
                FILTER (
                    (?reported > (?expected + (0.000001 * ?expected + 0.000001))) ||
                    (?reported < (?expected - (0.000001 * ?expected + 0.000001)))
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship peakPressureFactor must equal the selected peak pressure factor on its linked lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor ?resultQ ;
                      nltl:hasPeakPressureFactorLookupCase ?lookup .
                ?resultQ qudt:numericValue ?result .
                ?lookup nltl:selectedPeakPressureFactor ?selectedQ .
                ?selectedQ qudt:numericValue ?selected .
                FILTER (?result != ?selected)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each peak pressure factor lookup case must be the same linked case as a table lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPeakPressureFactorLookupCase ?peakCase .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?tableCase .
                    FILTER (?peakCase = ?tableCase)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-018-PeakLookupCaseShape>

        sh:path nltl:localizedStructuralMemberCategory ;

            nltl:stringerSideLongitudinalOrWebFrameCategory
            nltl:transverseFrameWithLoadDistributingStringersCategory
            nltl:transverseFrameWithoutLoadDistributingStringersCategory
            nltl:transverselyFramedPlatingCategory
        )
    ] ;
    sh:property [
        sh:path nltl:selectedPeakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

    ] .
