<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_019_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:node [
                            sh:class nltl:hullAreaValue ;
                            sh:property [
                                sh:path nltl:hullAreaFactor ;
                                sh:node [
                                    sh:class qudt:QuantityValue ;
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue unit:UNITLESS ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ] ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:sparql [
                        sh:message "selectedHullAreaFactor must equal the largest hull area factor among the hull areas spanned by the structural member" ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                            SELECT $this
                            WHERE {
                                $this nltl:hasSpannedHullArea ?area ;
                                      nltl:selectedHullAreaFactor ?selectedNode .
                                ?selectedNode qudt:numericValue ?selected .

                                {
                                    SELECT ?member (MAX(?factor) AS ?expected)
                                    WHERE {
                                        ?member nltl:hasSpannedHullArea ?memberArea .
                                        ?memberArea nltl:hullAreaFactor ?factorNode .
                                        ?factorNode qudt:numericValue ?factor .
                                    }
                                    GROUP BY ?member
                                }

                                FILTER (
                                    ABS(?selected - ?expected) >
                                    (1e-6 * ABS(?expected) + 1e-9)
                                )
                            }
                        """
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:maxCount 0
                    ]
                ]
            )
        ]
    ] .
<END_SHACL>
