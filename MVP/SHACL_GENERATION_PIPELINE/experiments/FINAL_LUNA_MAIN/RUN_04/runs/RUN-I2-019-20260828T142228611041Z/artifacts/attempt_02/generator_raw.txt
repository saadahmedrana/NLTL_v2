<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-019> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:hullAreaValue ;
                            sh:property [
                                sh:path nltl:hullAreaFactor ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:class qudt:QuantityValue ;
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue unit:UNITLESS ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:sparql [
                        sh:message "selectedHullAreaFactor must equal the maximum hullAreaFactor across the hull areas spanned by this structural member, within a relative tolerance of 1e-6 and an absolute fallback of 1e-12." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX sh: <http://www.w3.org/ns/shacl#>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                            SELECT $this
                            WHERE {
                                $this nltl:selectedHullAreaFactor ?selectedNode ;
                                      nltl:hasSpannedHullArea ?areaNode .
                                ?selectedNode qudt:numericValue ?selectedValue .
                                ?areaNode nltl:hullAreaFactor ?factorNode .
                                ?factorNode qudt:numericValue ?candidateValue .
                            }
                            GROUP BY $this
                            HAVING (
                                ABS(
                                    MAX(?candidateValue) -
                                    SAMPLE(?selectedValue)
                                ) >
                                (
                                    ABS(MAX(?candidateValue)) * 0.000001 +
                                    0.000000000001
                                )
                            )
                        """
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasSpannedHullArea ;
                        sh:maxCount 0
                    ] ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:maxCount 0
                    ]
                ]
            )
        ]
    ] .
<END_SHACL>
