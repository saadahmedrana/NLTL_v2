<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-021> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each plating thickness must equal the ice-load net plate thickness plus the corrosion-abrasion allowance, converted from millimetres to metres, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:iceLoadRequiredNetPlateThickness ?netQuantity ;
                         nltl:corrosionAbrasionAllowance ?allowanceQuantity ;
                         nltl:thickness ?thicknessQuantity .
                ?netQuantity qudt:numericValue ?net .
                ?allowanceQuantity qudt:numericValue ?allowance .
                ?thicknessQuantity qudt:numericValue ?thickness .
                BIND ((?net + ?allowance) / 1000 AS ?expected)
                BIND (
                    IF(
                        ((IF(?expected < 0, -?expected, ?expected)) * 0.000001) > 0.000000001,
                        (IF(?expected < 0, -?expected, ?expected)) * 0.000001,
                        0.000000001
                    ) AS ?tolerance
                )
                FILTER (
                    ?thickness < (?expected - ?tolerance) ||
                    ?thickness > (?expected + ?tolerance)
                )
            }
        """
    ] .
<END_SHACL>
