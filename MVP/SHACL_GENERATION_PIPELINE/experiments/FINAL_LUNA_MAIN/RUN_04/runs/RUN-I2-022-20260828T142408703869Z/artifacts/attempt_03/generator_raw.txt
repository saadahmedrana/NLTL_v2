gs:I2_022_NonApplicableAreaPlatingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:maxCount 1 ;
        sh:minCount 1 ;
        sh:node gs:I2_022_AngleAtLeast70Shape
    ] ;
    sh:property [
        sh:path nltl:platingHullAreaClassification ;
        sh:maxCount 1 ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:hullAreaValue
        ]
    ] ;
    sh:sparql [
        sh:message "A plating item with framing angle at least 70 degrees must have a non-applicable hull-area classification outside BI_b, M_b, and S_b when the formula is not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:platingHullAreaClassification ?area .
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
            }
        """
    ] .

gs:I2_022_ApplicableAreaShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:in (
        nltl:bowIntermediateBottomHullArea
        nltl:midbodyBottomHullArea
        nltl:sternBottomHullArea
    ) .
