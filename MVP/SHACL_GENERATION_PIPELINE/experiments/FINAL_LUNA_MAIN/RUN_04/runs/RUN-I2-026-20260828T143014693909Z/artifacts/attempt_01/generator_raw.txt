<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:minCount 1 ;
        sh:message "Each structural member shall identify its member support."
    ] ;
    sh:property [
        sh:path ( nltl:hasMemberSupport nltl:frameSupportCondition ) ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each member support shall have exactly one permitted frame support condition."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member shall state whether it terminates within the ice-strengthened area."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member terminating within an ice-strengthened area, or having demonstrated significant rotational restraint, shall use fixed frame support; continuous support or a connection bracket shall likewise be represented as fixed support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support ;
                      nltl:frameSupportCondition ?condition ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates .

                FILTER (
                    (?terminates = true &&
                     ?condition != nltl:fixedFrameSupport)
                    ||
                    EXISTS {
                        ?support nltl:demonstratedRotationalRestraint true .
                        FILTER (?condition != nltl:fixedFrameSupport)
                    }
                    ||
                    EXISTS {
                        $this nltl:continuousThroughSupport true .
                        FILTER (?condition != nltl:fixedFrameSupport)
                    }
                    ||
                    EXISTS {
                        ?support nltl:connectionBracketPresent true .
                        FILTER (?condition != nltl:fixedFrameSupport)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Fixed frame support shall be supported by continuous support, a connection bracket, or demonstrated significant rotational restraint with evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support ;
                      nltl:frameSupportCondition nltl:fixedFrameSupport .

                FILTER NOT EXISTS {
                    $this nltl:continuousThroughSupport true .
                }
                FILTER NOT EXISTS {
                    ?support nltl:connectionBracketPresent true .
                }
                FILTER NOT EXISTS {
                    ?support nltl:demonstratedRotationalRestraint true ;
                             nltl:significantRotationalRestraintEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:targetClass nltl:memberSupport ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Demonstrated significant rotational restraint shall have an evidence artifact." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember
    ] .
<END_SHACL>
