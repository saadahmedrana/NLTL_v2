<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_026_StructuralMemberShape
    ] .

gs:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node gs:I2_026_MemberSupportShape ;
        sh:message "Each structural member must identify its support conditions."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        ) ;
        sh:message "A structural member must specify either fixed or simple frame support."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Fixed support must be supported by continuity, a connection bracket, or demonstrated significant rotational restraint with evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                $this nltl:hasMemberSupport ?support .
                FILTER NOT EXISTS {
                    { $this nltl:continuousThroughSupport true . }
                    UNION
                    { ?support nltl:connectionBracketPresent true . }
                    UNION
                    {
                        ?support nltl:demonstratedRotationalRestraint true .
                        $this nltl:significantRotationalRestraintEvidence ?evidence .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A structural member terminating within an ice-strengthened area must have fixed support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:terminatesWithinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                }
            }
        """
    ] .

gs:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean
    ] .
<END_SHACL>
