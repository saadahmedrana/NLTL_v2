gen:I2_030_PlatingShape a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For oblique framing, framingAngleOmega must be strictly greater than 20 degrees and strictly less than 70 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:framingAngleOmega ?angle .
                ?angle qudt:numericValue ?angleValue .
                FILTER (?angleValue <= 20 || ?angleValue >= 70)
            }
        """
    ] .

    sh:class nltl:calculationCase ;

        sh:path nltl:plasticNeutralAxisHeight ;

        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:node gen:I2_030_InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:node gen:I2_030_InterpolationPointShape
    ] ;
    sh:sparql [
        sh:message "Interpolation lower and upper endpoint coordinates must be ordered strictly from lower to upper." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:interpolationLowerEndpoint ?lower .
                ?lower nltl:interpolationPointCoordinate ?lowerCoordinate .
                ?lowerCoordinate qudt:numericValue ?lowerValue .
                $this nltl:interpolationUpperEndpoint ?upper .
                ?upper nltl:interpolationPointCoordinate ?upperCoordinate .
                ?upperCoordinate qudt:numericValue ?upperValue .
                FILTER (?lowerValue >= ?upperValue)
            }
        """
    ] .
