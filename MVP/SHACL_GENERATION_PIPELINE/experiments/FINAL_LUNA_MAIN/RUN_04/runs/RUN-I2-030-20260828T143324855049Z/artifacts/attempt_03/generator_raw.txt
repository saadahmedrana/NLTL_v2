g:I2_030_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:I2_030_PlasticNeutralAxisHeightQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:class nltl:structuralMember
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The local-frame area must exceed the attached-plate-flange area for the declared calculation branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaQuantity .
                ?areaQuantity qudt:numericValue ?areaValue .
                ?member nltl:netAttachedShellPlateThickness ?thicknessQuantity .
                ?thicknessQuantity qudt:numericValue ?thicknessValue .
                ?member nltl:frameSpacing ?spacingQuantity .
                ?spacingQuantity qudt:numericValue ?spacingValue .
                FILTER (?areaValue * 100 <= 1000 * ?thicknessValue * ?spacingValue)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For oblique framing, the framing angle must be strictly between 20 and 70 degrees and paired interpolation endpoints must be supplied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleQuantity .
                ?angleQuantity qudt:numericValue ?angleValue .
                FILTER (?angleValue > 20 && ?angleValue < 70)
                FILTER NOT EXISTS {
                    $this nltl:interpolationLowerEndpoint ?lower .
                    $this nltl:interpolationUpperEndpoint ?upper .
                    ?lower nltl:interpolationPointCoordinate ?lowerCoordinate .
                    ?lower nltl:interpolationPointResult ?lowerResult .
                    ?upper nltl:interpolationPointCoordinate ?upperCoordinate .
                    ?upper nltl:interpolationPointResult ?upperResult .
                    ?lowerCoordinate qudt:numericValue ?lowerCoordinateValue .
                    ?upperCoordinate qudt:numericValue ?upperCoordinateValue .
                    FILTER (?lowerCoordinateValue < ?upperCoordinateValue)
                }
            }
        """
    ] .
