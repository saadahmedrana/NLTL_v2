<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-037-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:minCount 1 ;
        sh:message "The ship must identify the ice load patch used for the design."
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:class nltl:iceLoadPatchDesignCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-037-DesignCaseShape> ;
        sh:message "The ship must provide an ice load patch design case with minimization confirmation."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-037-StructuralMemberShape> ;
        sh:message "Each represented structural member must have the required ice-load design data."
    ] ;
    sh:sparql [
        sh:message "Each structural-member load case must represent a location at which the member capacity under the combined bending and shear effects is minimal." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship ;
                    nltl:hasIceLoadPatch ?patch ;
                    nltl:hasIceLoadPatchDesignCase ?designCase ;
                    nltl:hasStructuralMember ?member .

                ?patch a nltl:iceLoadPatch .
                ?designCase a nltl:iceLoadPatchDesignCase ;
                    nltl:memberCapacityMinimizationConfirmed true .

                ?member nltl:hasStructuralMemberLoadCase ?loadCase ;
                    nltl:memberCapacity ?memberQuantity .

                ?loadCase a nltl:loadCase ;
                    nltl:combinedShearBendingEffect ?combinedEffect ;
                    nltl:loadPatchApplicationLocation ?location .

                ?memberQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?capacity ;
                    qudt:unit unit:KiloN .

                $this nltl:hasStructuralMember ?otherMember .
                ?otherMember nltl:hasStructuralMemberLoadCase ?otherLoadCase ;
                    nltl:memberCapacity ?otherQuantity .

                ?otherLoadCase a nltl:loadCase ;
                    nltl:combinedShearBendingEffect ?otherCombinedEffect ;
                    nltl:loadPatchApplicationLocation ?location .

                ?otherQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?otherCapacity ;
                    qudt:unit unit:KiloN .

                FILTER (?otherCapacity < ?capacity)
            }
        """
    ] .

<urn:nltl:generated-shape:I2-037-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-037-LoadCaseShape> ;
        sh:message "Each structural member must have a load case for the ice-load design."
    ] ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-037-CapacityShape> ;
        sh:message "Each structural member must provide one capacity in kN."
    ] .

<urn:nltl:generated-shape:I2-037-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must identify the combined bending and shear effect."
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must identify the load patch application location."
    ] .

<urn:nltl:generated-shape:I2-037-DesignCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice load patch design case must confirm capacity minimization."
    ] .

<urn:nltl:generated-shape:I2-037-CapacityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
