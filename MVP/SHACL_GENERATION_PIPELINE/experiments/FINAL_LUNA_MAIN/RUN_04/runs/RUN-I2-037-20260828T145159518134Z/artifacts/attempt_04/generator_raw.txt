<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-037> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must identify the ice load patch to which the design cases apply."
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:class nltl:iceLoadPatchDesignCase ;
        sh:node [
            sh:property [
                sh:path nltl:memberCapacityMinimizationConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The ice-load-patch design case must confirm capacity minimization."
            ] ;
            sh:property [
                sh:path nltl:loadPatchApplicationLocation ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:combinedShearBendingEffect ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one ice-load-patch design case with the required minimization result and load-case operands."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node [
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:class nltl:loadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:loadPatchApplicationLocation ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:combinedShearBendingEffect ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:message "Each represented structural member must have a load case containing the patch location and combined bending-and-shear effect."
            ] ;
            sh:property [
                sh:path nltl:memberCapacity ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each represented structural member must provide its capacity in kN."
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The selected ice-load-patch location must be a location at which the same structural member's capacity under the combined bending-and-shear effect is minimal across its applicable load cases, and the selected case must be the identified ice-load-patch design case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasIceLoadPatch ?patch .
                $this nltl:hasIceLoadPatchDesignCase ?designCase .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberLoadCase ?selectedCase .
                FILTER (?selectedCase = ?designCase)

                ?selectedCase nltl:loadPatchApplicationLocation ?selectedLocation .
                ?selectedCase nltl:combinedShearBendingEffect ?selectedEffect .
                ?member nltl:memberCapacity ?selectedCapacity .
                ?selectedCapacity qudt:numericValue ?selectedValue .

                ?member nltl:hasStructuralMemberLoadCase ?alternativeCase .
                ?alternativeCase nltl:loadPatchApplicationLocation ?alternativeLocation .
                ?alternativeCase nltl:combinedShearBendingEffect ?alternativeEffect .
                ?member nltl:memberCapacity ?alternativeCapacity .
                ?alternativeCapacity qudt:numericValue ?alternativeValue .

                FILTER (?alternativeCase != ?selectedCase)
                FILTER (?alternativeValue < ?selectedValue)
            }
        """
    ] .
<END_SHACL>
