<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_047_Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For an ice-strengthened hull structure, gaugedThickness and netThickness must be present on the ship with decimal metre quantities, and steelRenewalRequired must be a boolean." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?structure .
                ?structure nltl:iceStrengthenedStructureStatus ?status .
                FILTER (?status = true && datatype(?status) = xsd:boolean)
                FILTER (
                    NOT EXISTS {
                        $this nltl:gaugedThickness ?g .
                        ?g qudt:numericValue ?gv ;
                           qudt:unit unit:M .
                        FILTER (datatype(?gv) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:netThickness ?n .
                        ?n qudt:numericValue ?nv ;
                           qudt:unit unit:M .
                        FILTER (datatype(?nv) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:steelRenewalRequired ?renewal .
                        FILTER (?renewal = true && datatype(?renewal) = xsd:boolean)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When gaugedThickness is less than netThickness plus 0.0005 m for an ice-strengthened hull structure, steelRenewalRequired must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?structure .
                ?structure nltl:iceStrengthenedStructureStatus ?status .
                FILTER (?status = true && datatype(?status) = xsd:boolean)

                $this nltl:gaugedThickness ?g .
                ?g qudt:numericValue ?gv ;
                   qudt:unit unit:M .
                FILTER (datatype(?gv) = xsd:decimal)

                $this nltl:netThickness ?n .
                ?n qudt:numericValue ?nv ;
                   qudt:unit unit:M .
                FILTER (datatype(?nv) = xsd:decimal)

                FILTER (?gv < (?nv + 0.0005))

                FILTER NOT EXISTS {
                    $this nltl:steelRenewalRequired ?renewal .
                    FILTER (?renewal = true && datatype(?renewal) = xsd:boolean)
                }
            }
        """
    ] .
<END_SHACL>
