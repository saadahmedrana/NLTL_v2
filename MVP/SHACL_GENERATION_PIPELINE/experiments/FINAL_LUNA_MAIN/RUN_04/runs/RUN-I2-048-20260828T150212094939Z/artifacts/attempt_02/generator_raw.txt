<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        )
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify exactly one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Each steel-grade requirement case must provide one as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify one permitted steel material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify one steel strength category."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify exactly one permitted actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1 ;
        sh:message "An applicable table selection must provide exactly one permitted required hull structural steel grade."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required grade must exactly match IACS UR I2 Table 8 for polar class, thickness, material class and strength category; the actual grade must have at least the required rank within the same grade family." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?pc .

                $this nltl:asBuiltPlateThickness ?thicknessNode ;
                      nltl:steelMaterialClass ?materialClass ;
                      nltl:steelStrengthCategory ?category ;
                      nltl:actualHullStructuralSteelGrade ?actual .

                ?thicknessNode qudt:numericValue ?th ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .

                OPTIONAL {
                    $this nltl:requiredHullStructuralSteelGrade ?required
                }

                FILTER (
                    ?th > 0 &&
                    ?th <= 50
                )

                BIND(
                    IF(?th <= 10, 1,
                    IF(?th <= 15, 2,
                    IF(?th <= 20, 3,
                    IF(?th <= 25, 4,
                    IF(?th <= 30, 5,
                    IF(?th <= 35, 6,
                    IF(?th <= 40, 7,
                    IF(?th <= 45, 8, 9)))))))) AS ?band
                )

                BIND(
                    (
                        ?category = nltl:normalStrengthSteelCategory &&
                        ?materialClass = nltl:steelMaterialClassThree &&
                        (
                            (?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3) && ?th > 35) ||
                            (?pc IN (nltl:polarClassPc4, nltl:polarClassPc5) && ?th > 40)
                        )
                    ) AS ?notApplicable
                )

                BIND(
                    IF(
                        ?category = nltl:normalStrengthSteelCategory,
                        IF(
                            ?materialClass = nltl:steelMaterialClassOne,
                            IF(?band <= 2, 1, IF(?band <= 7, 2, 3)),
                            IF(
                                ?materialClass = nltl:steelMaterialClassTwo,
                                IF(?band = 1, 1, IF(?band <= 7, 2, 3)),
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?band <= 2, 1, IF(?band <= 4, 2, 3)),
                                    3
                                )
                            )
                        ),
                        IF(
                            ?materialClass = nltl:steelMaterialClassOne,
                            IF(
                                ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                IF(?band <= 6, 1, 2),
                                IF(?band = 1, 1, IF(?band <= 7, 2, 3))
                            ),
                            IF(
                                ?materialClass = nltl:steelMaterialClassTwo,
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?band <= 4, 1, 2),
                                    IF(?band = 1, 1, IF(?band <= 4, 2, 3))
                                ),
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?band = 1, 1, IF(?band <= 4, 2, 3)),
                                    IF(
                                        (
                                            ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3) &&
                                            ?th > 35
                                        ) ||
                                        (
                                            ?pc IN (nltl:polarClassPc4, nltl:polarClassPc5) &&
                                            ?th > 40
                                        ),
                                        4,
                                        3
                                    )
                                )
                            )
                        )
                    ) AS ?requiredRank
                )

                BIND(
                    IF(
                        ?category = nltl:normalStrengthSteelCategory,
                        IF(?requiredRank = 1, nltl:steelGradeB,
                        IF(?requiredRank = 2, nltl:steelGradeD, nltl:steelGradeE)),
                        IF(?requiredRank = 1, nltl:steelGradeAh,
                        IF(?requiredRank = 2, nltl:steelGradeDh,
                        IF(?requiredRank = 3, nltl:steelGradeEh, nltl:steelGradeFh)))
                    ) AS ?expectedGrade
                )

                BIND(
                    IF(
                        ?actual = nltl:steelGradeB, 1,
                        IF(?actual = nltl:steelGradeD, 2,
                        IF(?actual = nltl:steelGradeE, 3,
                        IF(?actual = nltl:steelGradeAh, 1,
                        IF(?actual = nltl:steelGradeDh, 2,
                        IF(?actual = nltl:steelGradeEh, 3, 4)))))
                    ) AS ?actualRank
                )

                BIND(
                    (
                        (?category = nltl:normalStrengthSteelCategory &&
                         ?actual IN (
                            nltl:steelGradeB,
                            nltl:steelGradeD,
                            nltl:steelGradeE
                         ))
                        ||
                        (?category = nltl:highTensileSteelCategory &&
                         ?actual IN (
                            nltl:steelGradeAh,
                            nltl:steelGradeDh,
                            nltl:steelGradeEh,
                            nltl:steelGradeFh
                         ))
                    ) AS ?sameFamily
                )

                FILTER(
                    (?notApplicable && BOUND(?required)) ||
                    (
                        !?notApplicable &&
                        (
                            !BOUND(?required) ||
                            ?required != ?expectedGrade ||
                            !?sameFamily ||
                            ?actualRank < ?requiredRank
                        )
                    )
                )
            }
        """
    ] .
</urn:nltl:generated-shape:I2-048-CaseShape>
<END_SHACL>
