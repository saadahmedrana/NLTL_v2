<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single supported Polar Class is required."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade case must identify one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:message "The case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-ThicknessShape>
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The actual hull structural steel grade is required."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required grade must be the exact Table 8 grade for the selected Polar Class, thickness, material class, and strength category; the applicable result and family-scoped actual-grade rank are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
PREFIX qudt: <http://qudt.org/schema/qudt#>
SELECT $this
WHERE {
  $this nltl:asBuiltPlateThickness ?tv ;
        nltl:steelMaterialClass ?mc ;
        nltl:steelStrengthCategory ?sc ;
        nltl:actualHullStructuralSteelGrade ?actual .
  ?tv qudt:numericValue ?mm .
  ?ship nltl:hasSteelGradeRequirementCase $this ;
        nltl:polarClass ?pc .
  BIND(IF(?mm <= 10, 1,
       IF(?mm <= 15, 2,
       IF(?mm <= 20, 3,
       IF(?mm <= 25, 4,
       IF(?mm <= 30, 5,
       IF(?mm <= 35, 6,
       IF(?mm <= 40, 7,
       IF(?mm <= 45, 8,
       IF(?mm <= 50, 9, 0))))))))) AS ?band)
  BIND(?mc = nltl:steelMaterialClassOne AS ?m1)
  BIND(?mc = nltl:steelMaterialClassTwo AS ?m2)
  BIND(?mc = nltl:steelMaterialClassThree AS ?m3)
  BIND(?sc = nltl:normalStrengthSteelCategory AS ?normal)
  BIND(?sc = nltl:highTensileSteelCategory AS ?high)
  BIND(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3) AS ?pc13)
  BIND(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
               nltl:polarClassPc4, nltl:polarClassPc5) AS ?pc15)
  BIND(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5) AS ?pc45)
  BIND(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7) AS ?pc67)

  BIND(
    IF(?normal,
      IF(?m1,
        IF(?band <= 2, nltl:steelGradeB,
        IF(?band <= 4, nltl:steelGradeD,
        IF(?band <= 6, nltl:steelGradeE,
        IF(?band = 7, nltl:steelGradeD,
        IF(?pc15, nltl:steelGradeE,
        IF(?pc67, nltl:steelGradeD, nltl:steelGradeValue)))))),
      IF(?m2,
        IF(?band = 1, nltl:steelGradeB,
        IF(?band <= 4, nltl:steelGradeD,
        IF(?band <= 6, IF(?pc15, nltl:steelGradeE, nltl:steelGradeD),
        IF(?pc15, nltl:steelGradeE,
        IF(?pc67, nltl:steelGradeD, nltl:steelGradeValue))))),
        IF(?m3,
          IF(?band = 1,
            IF(?pc67, nltl:steelGradeB, nltl:steelGradeE),
          IF(?band <= 6,
            IF(?pc67, nltl:steelGradeD, nltl:steelGradeE),
          IF(?band <= 8,
            IF(?pc13, nltl:steelGradeValue,
            IF(?pc45, nltl:steelGradeE,
            IF(?pc67, nltl:steelGradeE, nltl:steelGradeValue))),
          IF(?pc67, nltl:steelGradeE, nltl:steelGradeValue)))),
        nltl:steelGradeValue),
    ?expectedNormal)

  BIND(
    IF(?high,
      IF(?m1,
        IF(?band <= 2, nltl:steelGradeAh,
        IF(?band <= 6,
          IF(?pc15, nltl:steelGradeDh, nltl:steelGradeAh),
        IF(?band = 7, nltl:steelGradeDh,
        IF(?pc15, nltl:steelGradeEh,
        IF(?pc67, nltl:steelGradeDh, nltl:steelGradeValue))))),
      IF(?m2,
        IF(?band = 1, nltl:steelGradeAh,
        IF(?band <= 4,
          IF(?pc15, nltl:steelGradeDh, nltl:steelGradeAh),
        IF(?band <= 7,
          IF(?pc15, nltl:steelGradeEh, nltl:steelGradeDh),
        IF(?pc15, nltl:steelGradeEh,
        IF(?pc67, nltl:steelGradeDh, nltl:steelGradeValue))))),
        IF(?m3,
          IF(?band = 1,
            IF(?pc67, nltl:steelGradeAh, nltl:steelGradeEh),
          IF(?band <= 6,
            IF(?pc67, nltl:steelGradeDh, nltl:steelGradeEh),
          IF(?band = 7,
            IF(?pc13, nltl:steelGradeFh,
            IF(?pc45, nltl:steelGradeEh,
            IF(?pc67, nltl:steelGradeEh, nltl:steelGradeValue))),
          IF(?pc67, nltl:steelGradeEh,
          IF(?pc13 || ?pc45, nltl:steelGradeFh, nltl:steelGradeValue))))),
        nltl:steelGradeValue),
    ?expectedHigh)

  BIND(IF(?normal, ?expectedNormal, ?expectedHigh) AS ?expected)
  BIND(
    ?normal && ?m3 && ?band >= 7 && ?band <= 9 && (?pc13 || (?pc45 && ?band = 9))
    AS ?notApplicable)

  FILTER(?band = 0 || (!?normal && !?high) || (!?m1 && !?m2 && !?m3) ||
         (!?pc13 && !?pc45 && !?pc67))
  UNION
  {
    FILTER(!?notApplicable)
    FILTER(!BOUND(?expected) || ?expected = nltl:steelGradeValue)
  }
  UNION
  {
    FILTER(!?notApplicable && BOUND(?expected) && ?expected != nltl:steelGradeValue)
    $this nltl:requiredHullStructuralSteelGrade ?required .
    FILTER(?required = ?expected)
    ?expected nltl:steelGradeOrderRank ?requiredRank .
    ?actual nltl:steelGradeOrderRank ?actualRank .
    FILTER(
      (?normal &&
       ?actual IN (nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE) &&
       ?required IN (nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE)) ||
      (?high &&
       ?actual IN (nltl:steelGradeAh, nltl:steelGradeDh, nltl:steelGradeEh, nltl:steelGradeFh) &&
       ?required IN (nltl:steelGradeAh, nltl:steelGradeDh, nltl:steelGradeEh, nltl:steelGradeFh))
    )
    FILTER(?actualRank >= ?requiredRank)
  }
}
""" ;
    ] .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
