gen:I2_054_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:node gen:I2_054_LengthQuantityShape ;
        sh:message "The ship must provide the upper ice waterline length." ;
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:node gen:I2_054_AngleQuantityShape ;
        sh:message "The ship must provide the stem angle." ;
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:node gen:I2_054_ForceQuantityShape ;
        sh:message "The ship must provide the design vertical ice force at the bow." ;
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_054_CalculationCaseShape ;
        sh:message "The ship must provide a calculation case for the externally determined bending-moment calculation." ;
    ] ;
    sh:property [
        sh:path nltl:hasHullGirderBendingMomentCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_054_HullGirderCaseShape ;
        sh:message "The ship must provide a hull-girder bending-moment calculation case." ;
    ] .

        sh:message "The calculation case must provide the declared interpolation-point structure." ;
    ] .

    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:minCount 1 ;
        sh:node gen:I2_054_DimensionlessQuantityShape ;
        sh:message "Each interpolation point must provide its coordinate." ;
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:node gen:I2_054_ResultQuantityShape ;
        sh:message "Each interpolation point must provide its externally determined result." ;
    ] .

    sh:property [
        sh:path nltl:caseBendingMomentDistributionFactor ;
        sh:minCount 1 ;
        sh:node gen:I2_054_DimensionlessQuantityShape ;

    sh:property [
        sh:path nltl:caseDesignVerticalIceBendingMoment ;
        sh:minCount 1 ;
        sh:node gen:I2_054_TorqueQuantityShape ;
        sh:message "The hull-girder case must provide the externally determined design vertical ice bending moment." ;
    ] .

    ] .
