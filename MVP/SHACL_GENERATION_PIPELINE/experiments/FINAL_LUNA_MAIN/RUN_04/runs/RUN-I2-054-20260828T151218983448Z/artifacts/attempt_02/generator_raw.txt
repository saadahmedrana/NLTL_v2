gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:sparql [
        sh:message "The calculation case must contain the five required C_m anchor points at aft, 0.5 L_UI, 0.7 L_UI, 0.95 L_UI, and forward, with C_m values 0, 1, 1, 0.3, and 0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this .
                ?ship nltl:upperIceWaterlineLengthLUI ?lengthValue .
                ?lengthValue qudt:numericValue ?length .

                $this nltl:hasInterpolationPoint ?aft .
                ?aft nltl:interpolationPointCoordinate ?aftCoordinate .
                ?aftCoordinate qudt:numericValue ?aftX .
                ?aft nltl:interpolationPointResult ?aftResult .
                ?aftResult qudt:numericValue ?aftCm .

                $this nltl:hasInterpolationPoint ?half .
                ?half nltl:interpolationPointCoordinate ?halfCoordinate .
                ?halfCoordinate qudt:numericValue ?halfX .
                ?half nltl:interpolationPointResult ?halfResult .
                ?halfResult qudt:numericValue ?halfCm .

                $this nltl:hasInterpolationPoint ?seven .
                ?seven nltl:interpolationPointCoordinate ?sevenCoordinate .
                ?sevenCoordinate qudt:numericValue ?sevenX .
                ?seven nltl:interpolationPointResult ?sevenResult .
                ?sevenResult qudt:numericValue ?sevenCm .

                $this nltl:hasInterpolationPoint ?ninetyFive .
                ?ninetyFive nltl:interpolationPointCoordinate ?ninetyFiveCoordinate .
                ?ninetyFiveCoordinate qudt:numericValue ?ninetyFiveX .
                ?ninetyFive nltl:interpolationPointResult ?ninetyFiveResult .
                ?ninetyFiveResult qudt:numericValue ?ninetyFiveCm .

                $this nltl:hasInterpolationPoint ?forward .
                ?forward nltl:interpolationPointCoordinate ?forwardCoordinate .
                ?forwardCoordinate qudt:numericValue ?forwardX .
                ?forward nltl:interpolationPointResult ?forwardResult .
                ?forwardResult qudt:numericValue ?forwardCm .

                FILTER (?aftX = 0.0 && ?aftCm = 0.0)
                FILTER (?halfX = 0.5 && ?halfCm = 1.0)
                FILTER (?sevenX = 0.7 && ?sevenCm = 1.0)
                FILTER (?ninetyFiveX = 0.95 && ?ninetyFiveCm = 0.3)
                FILTER (?forwardX = 1.0 && ?forwardCm = 0.0)
            }
        """
    ] .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 0.0 ;
                sh:maxInclusive 1.0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 0.0 ;
                sh:maxInclusive 1.0
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each interpolation point must follow the stated piecewise-linear C_m envelope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:interpolationPointCoordinate ?coordinate .
                ?coordinate qudt:numericValue ?x .
                $this nltl:interpolationPointResult ?result .
                ?result qudt:numericValue ?y .
                FILTER (
                    ?x < 0.0 ||
                    ?x > 1.0 ||
                    (?x <= 0.5 && ABS(?y - (2.0 * ?x)) > 0.000001) ||
                    (?x > 0.5 && ?x <= 0.7 && ABS(?y - 1.0) > 0.000001) ||
                    (?x > 0.7 && ?x <= 0.95 && ABS(?y - (1.0 - 0.7 * ((?x - 0.7) / 0.25))) > 0.000001) ||
                    (?x > 0.95 && ABS(?y) > 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
