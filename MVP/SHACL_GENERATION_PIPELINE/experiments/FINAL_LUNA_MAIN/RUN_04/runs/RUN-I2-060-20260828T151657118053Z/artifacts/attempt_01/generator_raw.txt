<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-060> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member in way of a cutout must not be recorded as unstable." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    ?member nltl:memberInWayOfCutout ?inWay .
    FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
    FILTER NOT EXISTS {
        $this nltl:memberInstability ?instability .
        FILTER(datatype(?instability) = xsd:boolean && ?instability = false)
    }
}
""" ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where stiffening is necessary for a member in way of a cutout, structural stiffening must be recorded." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    ?member nltl:memberInWayOfCutout ?inWay .
    ?member nltl:stiffeningNecessary ?necessary .
    FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
    FILTER(datatype(?necessary) = xsd:boolean && ?necessary = true)
    FILTER NOT EXISTS {
        $this nltl:structuralStiffening ?stiffening .
        FILTER(datatype(?stiffening) = xsd:string)
    }
}
""" ;
    ] .
<END_SHACL>
