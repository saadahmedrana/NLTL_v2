<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node nltlgen:CalculationCaseShape
    ] .

nltlgen:CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationMethodValue ;
        sh:in (
            nltl:directCalculationMethodValue
            nltl:prescribedAnalyticalProcedure
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:node [
            sh:property [
                sh:path nltl:thickness ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A direct calculation method must not replace the prescribed analytical procedure for shell-plating or local-frame hull structures." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?structure .
                ?structure rdf:type ?structureType .
                FILTER (
                    ?structureType IN (
                        nltl:plating,
                        nltl:structuralMember
                    )
                )
                $this nltl:calculationMethod nltl:directCalculationMethodValue .
            }
        """
    ] .
<END_SHACL>
