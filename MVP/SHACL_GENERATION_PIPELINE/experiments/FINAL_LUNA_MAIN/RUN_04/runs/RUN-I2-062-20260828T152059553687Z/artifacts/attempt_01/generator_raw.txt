<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive true ;
        sh:maxInclusive true ;
        sh:message "The ship must record that direct calculation is used."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked load carrying stringer or web frame forming part of a grillage system must identify that grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member rdf:type ?memberClass .
                FILTER (
                    ?memberClass = nltl:loadCarryingStringer ||
                    ?memberClass = nltl:webFrame
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage rdf:type nltl:grillageSystem .
                }
            }
        """
    ] .
<END_SHACL>
