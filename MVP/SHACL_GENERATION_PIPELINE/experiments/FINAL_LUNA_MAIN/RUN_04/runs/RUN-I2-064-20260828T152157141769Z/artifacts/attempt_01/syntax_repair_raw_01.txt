<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_064_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one yield-strength quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class nltl:calculationMethodValue ;
                    sh:hasValue nltl:linearCalculationMethodValue
                ]
            ]
        ] ;
        sh:message "The ship shall provide a calculation case using the linear calculation method."
    ] .

gen:I2_064_CalculationCaseShape a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:calculationMethodValue ;
            sh:hasValue nltl:linearCalculationMethodValue
        ] ;
        sh:message "Each calculation case shall identify the linear calculation method."
    ] .

gen:I2_064_StructuralMemberShape a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:bucklingCriteriaSatisfied ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable buckling criteria shall be satisfied for the structural member."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberWeb ;
        sh:node [
            sh:class nltl:structuralMemberWeb ;
            sh:property [
                sh:path nltl:nominalShearStress ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:MegaPA ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberFlange ;
        sh:node [
            sh:class nltl:structuralMemberFlange ;
            sh:property [
                sh:path nltl:nominalFlangeVonMisesStress ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:MegaPA ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal shear stress in a member web shall be less than the ship's yield strength divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this .
                ?ship nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                ?yieldQuantity qudt:unit qudt:MegaPA .
                $this nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stressValue .
                ?stressQuantity qudt:unit qudt:MegaPA .
                FILTER (?stressValue >= (?yieldValue / 1.7320508075688772))
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal flange von Mises stress shall be less than 1.15 times the ship's yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this .
                ?ship nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                ?yieldQuantity qudt:unit qudt:MegaPA .
                $this nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stressValue .
                ?stressQuantity qudt:unit qudt:MegaPA .
                FILTER (?stressValue >= (1.15 * ?yieldValue))
            }
        """
    ] .
<END_SHACL>
