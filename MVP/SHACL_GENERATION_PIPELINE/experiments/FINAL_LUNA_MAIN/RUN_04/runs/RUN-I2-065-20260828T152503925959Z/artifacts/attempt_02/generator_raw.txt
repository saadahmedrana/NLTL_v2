<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_065_NonlinearReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A calculation case selected for the nonlinear method must provide the required nonlinear-analysis readiness inputs, result, evidence, and units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod
                    nltl:nonlinearCalculationMethodValue .

                FILTER (
                    NOT EXISTS {
                        ?case a nltl:calculationCase .
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:fractureMargin ?fracture .
                        ?fracture qudt:numericValue ?fractureNumber ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?fractureNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:majorBucklingMargin ?buckling .
                        ?buckling qudt:numericValue ?bucklingNumber ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?bucklingNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:yieldingMargin ?yielding .
                        ?yielding qudt:numericValue ?yieldingNumber ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?yieldingNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentLateralDeformation ?lateral .
                        ?lateral qudt:numericValue ?lateralNumber ;
                            qudt:unit qudt:MilliM .
                        FILTER (datatype(?lateralNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                        ?outOfPlane qudt:numericValue ?outOfPlaneNumber ;
                            qudt:unit qudt:MilliM .
                        FILTER (datatype(?outOfPlaneNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:relevantStructuralDimension ?dimension .
                        ?dimension qudt:numericValue ?dimensionNumber ;
                            qudt:unit qudt:MilliM .
                        FILTER (datatype(?dimensionNumber) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:minorPermanentDeformationAssessmentSatisfied
                            true .
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                        ?evidence a nltl:evidenceArtifact .
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
