<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qudt-unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-065-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node <urn:nltl:generated-shape:I2-065-CalculationCaseShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a nonlinear calculation case is present, the required margins, deformation measurements, and relevant structural dimension must be recorded with the prescribed QUDT units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX qudt-unit: <http://qudt.org/vocab/unit/>

            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER (
                    !EXISTS {
                        ?this nltl:fractureMargin ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:UNITLESS .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:UNITLESS)
                        }
                    }
                    ||
                    !EXISTS {
                        ?this nltl:majorBucklingMargin ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:UNITLESS .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:UNITLESS)
                        }
                    }
                    ||
                    !EXISTS {
                        ?this nltl:yieldingMargin ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:UNITLESS .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:UNITLESS)
                        }
                    }
                    ||
                    !EXISTS {
                        ?this nltl:permanentLateralDeformation ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:MilliM .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:MilliM)
                        }
                    }
                    ||
                    !EXISTS {
                        ?this nltl:permanentOutOfPlaneDeformation ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:MilliM .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:MilliM)
                        }
                    }
                    ||
                    !EXISTS {
                        ?this nltl:relevantStructuralDimension ?q .
                        ?q qudt:numericValue ?number ;
                           qudt:unit qudt-unit:MilliM .
                        FILTER NOT EXISTS {
                            ?q qudt:unit ?otherUnit .
                            FILTER (?otherUnit != qudt-unit:MilliM)
                        }
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-065-CalculationCaseShape>
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            sh:property [
                sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ] ;
            sh:property [
                sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
<END_SHACL>
