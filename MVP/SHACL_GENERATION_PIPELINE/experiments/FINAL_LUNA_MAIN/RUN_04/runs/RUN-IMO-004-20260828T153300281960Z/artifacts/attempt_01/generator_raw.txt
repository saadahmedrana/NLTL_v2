<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship has sea ice, no more than one winter growth, and ice thickness from 0.3 m through 2.0 m, its ice type classification must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship ;
                      nltl:iceOriginType nltl:seaIceOrigin ;
                      nltl:winterGrowthCount ?growthCount ;
                      nltl:iceThickness ?thicknessNode .

                ?thicknessNode a qudt:QuantityValue ;
                                qudt:numericValue ?thickness ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    datatype(?growthCount) = xsd:integer &&
                    datatype(?thickness) = xsd:decimal &&
                    ?growthCount <= 1 &&
                    ?thickness >= 0.3 &&
                    ?thickness <= 2.0
                )

                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """ ;
    ] .
<END_SHACL>
