<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_004_Static a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer ;
        sh:maxInclusive 1 ;
        sh:message "Winter growth count must be an integer not greater than 1."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0.3"^^xsd:decimal ;
                sh:maxInclusive "2.0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice thickness must be a quantity from 0.3 m through 2.0 m using unit:M."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice origin is sea ice, winter growth count and ice thickness must establish applicability, and ice type classification must be exactly firstYearIce." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin .
                FILTER (
                    NOT EXISTS {
                        $this nltl:winterGrowthCount ?growth .
                        FILTER (
                            datatype(?growth) = xsd:integer &&
                            ?growth <= 1
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceThickness ?thicknessNode .
                        ?thicknessNode qudt:numericValue ?thickness ;
                            qudt:unit unit:M .
                        FILTER (
                            datatype(?thickness) = xsd:decimal &&
                            ?thickness >= "0.3"^^xsd:decimal &&
                            ?thickness <= "2.0"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceTypeClassification nltl:firstYearIce .
                    }
                    ||
                    EXISTS {
                        $this nltl:iceTypeClassification ?classification .
                        FILTER (?classification != nltl:firstYearIce)
                    }
                )
            }
        """
    ] .
<END_SHACL>
