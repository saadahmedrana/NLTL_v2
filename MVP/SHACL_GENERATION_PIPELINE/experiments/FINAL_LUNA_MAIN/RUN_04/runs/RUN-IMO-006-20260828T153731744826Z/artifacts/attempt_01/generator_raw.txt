<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_006_MediumFirstYearIceThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships classified as medium first-year ice must provide ice thickness from 0.7 m through 1.2 m, inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?thickness .
                    ?thickness rdf:type qudt:QuantityValue ;
                               qudt:numericValue ?value ;
                               qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.7"^^xsd:decimal &&
                        ?value <= "1.2"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
