<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-006-Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice type classification to determine the applicable ice-thickness requirement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as medium first-year ice must provide an ice thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?thickness .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every ice thickness for a ship classified as medium first-year ice must be a qudt:QuantityValue in metres and must be between 0.7 m and 1.2 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                $this nltl:iceThickness ?thickness .
                FILTER (
                    NOT EXISTS {
                        ?thickness a qudt:QuantityValue .
                    }
                    ||
                    NOT EXISTS {
                        ?thickness qudt:numericValue ?numericValue .
                    }
                    ||
                    NOT EXISTS {
                        ?thickness qudt:unit unit:M .
                    }
                    ||
                    EXISTS {
                        ?thickness qudt:unit ?otherUnit .
                        FILTER (?otherUnit != unit:M)
                    }
                    ||
                    EXISTS {
                        ?thickness qudt:numericValue ?outOfRangeValue .
                        FILTER (
                            ?outOfRangeValue < "0.7"^^xsd:decimal
                            ||
                            ?outOfRangeValue > "1.2"^^xsd:decimal
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
