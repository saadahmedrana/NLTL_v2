<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_009_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "A ship must provide an ice condition to determine whether the bergy-waters limits apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, every supplied land-origin ice concentration must be a dimensionless decimal quantity less than 1, and at least one valid quantity must be supplied; every supplied total ice concentration must be a dimensionless decimal quantity no greater than 1, and at least one valid quantity must be supplied." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER (?condition = "bergy waters"^^xsd:string)

    FILTER (
        NOT EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            FILTER NOT EXISTS {
                ?landQuantity a qudt:QuantityValue .
            }
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            FILTER NOT EXISTS {
                ?landQuantity qudt:numericValue ?landValue .
            }
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            ?landQuantity qudt:numericValue ?landValue .
            FILTER (datatype(?landValue) != xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            FILTER NOT EXISTS {
                ?landQuantity qudt:unit unit:UNITLESS .
            }
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            ?landQuantity qudt:unit ?landUnit .
            FILTER (?landUnit != unit:UNITLESS)
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            ?landQuantity qudt:numericValue ?landValue .
            FILTER (datatype(?landValue) = xsd:decimal && ?landValue >= 1)
        }
        ||
        NOT EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            ?landQuantity a qudt:QuantityValue ;
                qudt:numericValue ?landValue ;
                qudt:unit unit:UNITLESS .
            FILTER (datatype(?landValue) = xsd:decimal && ?landValue < 1)
        }
        ||
        NOT EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            FILTER NOT EXISTS {
                ?totalQuantity a qudt:QuantityValue .
            }
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            FILTER NOT EXISTS {
                ?totalQuantity qudt:numericValue ?totalValue .
            }
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            ?totalQuantity qudt:numericValue ?totalValue .
            FILTER (datatype(?totalValue) != xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            FILTER NOT EXISTS {
                ?totalQuantity qudt:unit unit:UNITLESS .
            }
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            ?totalQuantity qudt:unit ?totalUnit .
            FILTER (?totalUnit != unit:UNITLESS)
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            ?totalQuantity qudt:numericValue ?totalValue .
            FILTER (datatype(?totalValue) = xsd:decimal && ?totalValue > 1)
        }
        ||
        NOT EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            ?totalQuantity a qudt:QuantityValue ;
                qudt:numericValue ?totalValue ;
                qudt:unit unit:UNITLESS .
            FILTER (datatype(?totalValue) = xsd:decimal && ?totalValue <= 1)
        }
    )
}
""" ;
    ] .
<END_SHACL>
