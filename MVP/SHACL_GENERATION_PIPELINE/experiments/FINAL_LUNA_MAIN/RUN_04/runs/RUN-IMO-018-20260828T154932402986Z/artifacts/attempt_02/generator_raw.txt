<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-018> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice capability methodology is applicable, the certificate methodology reference and administration acceptance status must both be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCapabilityMethodologyApplicable ?applicable .
                FILTER (datatype(?applicable) = xsd:boolean && ?applicable = true)

                FILTER (
                    NOT EXISTS {
                        $this nltl:certificateMethodologyReferencePresent ?referencePresent .
                        FILTER (datatype(?referencePresent) = xsd:boolean && ?referencePresent = true)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:administrationAcceptanceStatus ?accepted .
                        FILTER (datatype(?accepted) = xsd:boolean && ?accepted = true)
                    }
                )
            }
        """
    ] .
<END_SHACL>
