<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_035_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarWaterOperationalManualRecord ;
        sh:class nltl:documentRecord ;
        sh:minCount 1 ;
        sh:message "The ship must identify a Polar Water Operational Manual record."
    ] ;
    sh:property [
        sh:path nltl:hasStabilityCalculationRecord ;
        sh:class nltl:documentRecord ;
        sh:minCount 1 ;
        sh:message "The ship must identify a stability calculation record."
    ] ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualIcingAllowanceInformationPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The PWOM must state that icing allowance information is present."
    ] ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualIcingAllowanceValues ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The PWOM icing allowance values must be provided."
    ] ;
    sh:property [
        sh:path nltl:stabilityCalculationIcingAllowanceValues ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The stability calculation icing allowance values must be provided."
    ] ;
    sh:sparql [
        sh:message "The icing allowance values in the PWOM must match the stability calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarWaterOperationalManualIcingAllowanceValues ?pwomValues .
                FILTER NOT EXISTS {
                    $this nltl:stabilityCalculationIcingAllowanceValues ?stabilityValues .
                    FILTER (?pwomValues = ?stabilityValues)
                }
            }
        """
    ] .
</END_SHACL>
