    sh:property [

        sh:maxCount 1 ;

                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "Upper ice waterline draught must be one metre-valued decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;

        sh:message "Damage extents and selected vertical position do not satisfy the required piecewise formulas and inclusive keel-to-1.20-draught range." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .

                ?lengthNode qudt:numericValue ?length ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?longNode qudt:numericValue ?longitudinal ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                ?verticalNode qudt:numericValue ?vertical ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?positionNode qudt:numericValue ?position ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .

                BIND(IF(?forward = true,
                        0.045 * ?length,
                        0.015 * ?length) AS ?expectedLongitudinal)
                BIND(0.20 * ?draught AS ?draughtLimit)
                BIND(IF(?draughtLimit < ?longitudinal,
                        ?draughtLimit,
                        ?longitudinal) AS ?expectedVertical)

                FILTER (
                    ABS(?longitudinal - ?expectedLongitudinal) >
                    (0.000001 * ABS(?expectedLongitudinal) + 0.000000001)
                    ||
                    ABS(?vertical - ?expectedVertical) >
                    (0.000001 * ABS(?expectedVertical) + 0.000000001)
                    ||
                    ?position < 0
                    ||
                    ?position > (1.20 * ?draught)
                )
            }
        """
    ] .
