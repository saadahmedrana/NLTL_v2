<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-043-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:class nltl:machineryInstallationItem ;
        sh:node <urn:nltl:generated-shape:IMO-043-MachineryInstallationShape>
    ] .

<urn:nltl:generated-shape:IMO-043-MachineryInstallationShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:iceAccretionHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable ice accretion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowAccumulationHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable snow accumulation hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIceIngestionHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable seawater ice ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:liquidFreezingOrViscosityIncreaseHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable liquid freezing or viscosity increase hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIntakeTemperatureHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable seawater intake temperature hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowIngestionHazard ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall cover the applicable snow ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem ;
        sh:node <urn:nltl:generated-shape:IMO-043-AssociatedEquipmentShape>
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Each machinery installation shall have evidence substantiating its hazard protection coverage."
    ] .

<urn:nltl:generated-shape:IMO-043-AssociatedEquipmentShape> a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each associated equipment item shall be linked to a machinery installation having hazard protection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?installation nltl:hasAssociatedEquipment $this .
                    ?installation nltl:hasHazardProtectionEvidence ?evidence .
                }
            }
        """
    ] .
<END_SHACL>
