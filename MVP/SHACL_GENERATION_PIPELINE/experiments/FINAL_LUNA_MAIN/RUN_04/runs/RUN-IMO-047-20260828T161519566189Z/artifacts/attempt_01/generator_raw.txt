<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:minCount 1 ;
        sh:message "The ship must identify its essential engine or engines."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Combustion-air temperature must be provided as a single decimal quantity in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion must be provided in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion must be provided in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion-air temperature must be within the engine manufacturer's minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureQuantity .
                $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .
                ?temperatureQuantity qudt:numericValue ?temperatureValue .
                ?minimumQuantity qudt:numericValue ?minimumValue .
                ?maximumQuantity qudt:numericValue ?maximumValue .
                FILTER (
                    ?temperatureValue < ?minimumValue ||
                    ?temperatureValue > ?maximumValue
                )
            }
        """
    ] .
