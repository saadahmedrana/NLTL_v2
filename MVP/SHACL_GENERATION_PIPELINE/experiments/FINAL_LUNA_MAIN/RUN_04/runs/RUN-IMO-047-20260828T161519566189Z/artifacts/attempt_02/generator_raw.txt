<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-047> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine
    ] ;
    sh:sparql [
        sh:message "For each ship with an essential engine, combustion-air temperature control must be present and the combustion-air temperature must be within the manufacturer minimum and maximum criteria, using Celsius quantity values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?engine .
                FILTER NOT EXISTS {
                    $this nltl:combustionAirTemperatureControlPresent true .
                    $this nltl:combustionAirTemperature ?temperatureNode .
                    ?temperatureNode qudt:numericValue ?temperature ;
                        qudt:unit unit:DEG_C .
                    $this nltl:manufacturerMinAirTemperature ?minimumNode .
                    ?minimumNode qudt:numericValue ?minimum ;
                        qudt:unit unit:DEG_C .
                    $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                    ?maximumNode qudt:numericValue ?maximum ;
                        qudt:unit unit:DEG_C .
                    FILTER(
                        datatype(?temperature) = xsd:decimal &&
                        datatype(?minimum) = xsd:decimal &&
                        datatype(?maximum) = xsd:decimal &&
                        ?temperature >= ?minimum &&
                        ?temperature <= ?maximum
                    )
                }
            }
        """
    ] .
<END_SHACL>
