<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO057StaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Every linked fire pump must have a containing compartment maintained above 0 deg C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER (
                    NOT EXISTS {
                        ?pump nltl:hasContainingCompartment ?missingCompartment
                    }
                    ||
                    EXISTS {
                        ?pump nltl:hasContainingCompartment ?compartment .
                        FILTER NOT EXISTS {
                            ?compartment nltl:maintainedTemperature ?temperature .
                            ?temperature a qudt:QuantityValue ;
                                qudt:unit unit:DEG_C ;
                                qudt:numericValue ?value .
                            FILTER (?value > 0)
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
