<urn:nltl:generated-shape:IMO-079> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one persons-on-board count."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one maximum expected rescue time in days."
    ] ;
    sh:property [
        sh:path nltl:availableEmergencyRationPerson ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [

        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            WHERE {
                $this nltl:personsOnBoard ?persons ;

                      nltl:availableEmergencyRationPerson ?rationQuantity .
                ?rescueQuantity qudt:numericValue ?rescueTime .
                ?rationQuantity qudt:numericValue ?rationCapacity .
                FILTER (?rationCapacity < (?persons * ?rescueTime))
            }
        """
    ] .
<END_SHACL>
