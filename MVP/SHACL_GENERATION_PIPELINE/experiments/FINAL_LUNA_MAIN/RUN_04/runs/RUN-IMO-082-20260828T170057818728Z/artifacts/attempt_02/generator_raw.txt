<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-082-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:bridgeConfigurationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single bridge configuration type is required to determine astern-view applicability."
    ] ;
    sh:property [
        sh:path nltl:clearViewAsternStatus ;
        sh:datatype xsd:boolean ;
        sh:message "Any clear astern-view status shall be an xsd:boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the bridge configuration requires an astern view, clearViewAsternStatus shall contain exactly one compliant xsd:boolean value."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bridgeConfigurationType
                    nltl:asternViewRequiredBridgeConfiguration .
                FILTER (
                    NOT EXISTS {
                        $this nltl:clearViewAsternStatus ?status .
                        FILTER (
                            datatype(?status) = xsd:boolean &&
                            ?status = true
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:clearViewAsternStatus ?status .
                        FILTER (
                            datatype(?status) != xsd:boolean ||
                            !(?status = true)
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:clearViewAsternStatus ?status1 .
                        $this nltl:clearViewAsternStatus ?status2 .
                        FILTER (?status1 != ?status2)
                    }
                )
            }
        """
    ] .
<END_SHACL>
