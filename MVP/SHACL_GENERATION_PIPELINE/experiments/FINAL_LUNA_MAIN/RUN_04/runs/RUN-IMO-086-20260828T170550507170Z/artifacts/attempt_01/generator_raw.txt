<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-086-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node [
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:independentFromHeadingMeans ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:nonMagneticHeadingMeans ;
                sh:minCount 1 ;
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means must be independent from another distinct non-magnetic heading means of the ship and connected to both main and emergency power." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasNonMagneticHeadingMeans ?means .
    FILTER (
        NOT EXISTS {
            ?means nltl:connectedToMainPower ?mainPower .
        }
        ||
        NOT EXISTS {
            ?means nltl:connectedToEmergencyPower ?emergencyPower .
        }
        ||
        NOT EXISTS {
            ?means nltl:independentFromHeadingMeans ?otherMeans .
            $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
            FILTER (?otherMeans != ?means)
        }
    )
}
""" ;
    ] .
<END_SHACL>
