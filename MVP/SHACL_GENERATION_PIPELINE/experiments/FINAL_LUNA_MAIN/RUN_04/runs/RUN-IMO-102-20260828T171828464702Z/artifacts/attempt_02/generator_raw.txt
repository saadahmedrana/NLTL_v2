<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO102ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "The ship must have exactly one controlled 12.3.1 ice condition."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:message "The ship must have exactly one controlled 12.3.1 ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO102CrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable crew member must have a training record whose required level and recorded level match the 12.3.1 matrix and whose STCW qualification is valid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role .
                FILTER (
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        (?shipType = nltl:tankerPolarTrainingShipType ||
                         ?shipType = nltl:passengerShipPolarTrainingShipType) &&
                        (?role = nltl:masterPolarTrainingRole ||
                         ?role = nltl:chiefMatePolarTrainingRole ||
                         ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole)
                    )
                    ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        (?role = nltl:masterPolarTrainingRole ||
                         ?role = nltl:chiefMatePolarTrainingRole)
                    )
                    ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                    )
                )
                FILTER NOT EXISTS {
                    ?crew nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?required ;
                            nltl:trainingRecordLevel ?recorded ;
                            nltl:stcwQualificationValid true .
                    FILTER (
                        (
                            ?ice = nltl:openWaterIceCondition &&
                            (?shipType = nltl:tankerPolarTrainingShipType ||
                             ?shipType = nltl:passengerShipPolarTrainingShipType) &&
                            ?required = nltl:basicPolarTrainingLevel &&
                            ?recorded = nltl:basicPolarTrainingLevel
                        )
                        ||
                        (
                            ?ice = nltl:otherWatersIceCondition &&
                            (?role = nltl:masterPolarTrainingRole ||
                             ?role = nltl:chiefMatePolarTrainingRole) &&
                            ?required = nltl:advancedPolarTrainingLevel &&
                            ?recorded = nltl:advancedPolarTrainingLevel
                        )
                        ||
                        (
                            ?ice = nltl:otherWatersIceCondition &&
                            ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                            ?required = nltl:basicPolarTrainingLevel &&
                            ?recorded = nltl:basicPolarTrainingLevel
                        )
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable crew member's training record must have the matrix-required level and a true STCW qualification status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role ;
                      nltl:hasCrewTrainingRecord ?record .
                ?record nltl:requiredTrainingLevel ?required ;
                        nltl:trainingRecordLevel ?recorded ;
                        nltl:stcwQualificationValid ?valid .
                FILTER (
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        (?shipType = nltl:tankerPolarTrainingShipType ||
                         ?shipType = nltl:passengerShipPolarTrainingShipType) &&
                        (?role = nltl:masterPolarTrainingRole ||
                         ?role = nltl:chiefMatePolarTrainingRole ||
                         ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole) &&
                        (?required != nltl:basicPolarTrainingLevel ||
                         ?recorded != nltl:basicPolarTrainingLevel ||
                         ?valid != true)
                    )
                    ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        (?role = nltl:masterPolarTrainingRole ||
                         ?role = nltl:chiefMatePolarTrainingRole) &&
                        (?required != nltl:advancedPolarTrainingLevel ||
                         ?recorded != nltl:advancedPolarTrainingLevel ||
                         ?valid != true)
                    )
                    ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                        (?required != nltl:basicPolarTrainingLevel ||
                         ?recorded != nltl:basicPolarTrainingLevel ||
                         ?valid != true)
                    )
                )
            }
        """
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:message "Each represented crew member must have exactly one controlled 12.3.1 training role."
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] .

gen:IMO102TrainingRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A training record must state one controlled required training level."
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A training record must state one controlled recorded training level."
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A training record must state whether the STCW qualification is valid."
    ] .
<END_SHACL>
