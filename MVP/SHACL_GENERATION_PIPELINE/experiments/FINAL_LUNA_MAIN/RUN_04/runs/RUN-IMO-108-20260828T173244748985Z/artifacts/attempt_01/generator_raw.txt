<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property gen:IMO108CategoryAnyProperty ;
    sh:property gen:IMO108ConstructionDateAnyProperty ;
    sh:property gen:IMO108AggregateCapacityAnyProperty ;
    sh:property gen:IMO108IndividualCapacityAnyProperty ;
    sh:property gen:IMO108SeparationAnyProperty ;
    sh:or (
        gen:IMO108ApplicableShape
        gen:IMO108CategoryCShape
        gen:IMO108DateBefore2017Shape
        gen:IMO108AggregateAtLeast600Shape
        gen:IMO108IndividualAtMost30Shape
    ) .

gen:IMO108CategoryAnyProperty
    a sh:PropertyShape ;
    sh:path nltl:shipCategory ;
    sh:nodeKind sh:IRI ;
    sh:in (
        nltl:polarShipCategoryA
        nltl:polarShipCategoryB
        nltl:polarShipCategoryC
    ) ;
    sh:maxCount 1 .

gen:IMO108ConstructionDateAnyProperty
    a sh:PropertyShape ;
    sh:path nltl:constructionDate ;
    sh:datatype xsd:date ;
    sh:maxCount 1 .

gen:IMO108AggregateCapacityAnyProperty
    a sh:PropertyShape ;
    sh:path nltl:aggregateOilFuelCapacity ;
    sh:maxCount 1 ;
    sh:node gen:IMO108VolumeQuantityShape .

gen:IMO108IndividualCapacityAnyProperty
    a sh:PropertyShape ;
    sh:path nltl:individualOilFuelTankCapacity ;
    sh:maxCount 1 ;
    sh:node gen:IMO108VolumeQuantityShape .

gen:IMO108SeparationAnyProperty
    a sh:PropertyShape ;
    sh:path nltl:oilFuelTankOuterShellSeparation ;
    sh:maxCount 1 ;
    sh:node gen:IMO108LengthQuantityShape .

gen:IMO108ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO108AggregateBelow600QuantityShape
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO108IndividualAbove30QuantityShape
    ] ;
    sh:property [
        sh:path nltl:oilFuelTankOuterShellSeparation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO108SeparationAtLeast076QuantityShape ;
        sh:message "Applicable oil fuel tanks must be separated from the outer shell by at least 0.76 m."
    ] .

gen:IMO108CategoryCShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO108DateBefore2017Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2016-12-31"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO108AggregateAtLeast600Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] .

gen:IMO108IndividualAtMost30Shape

    ] .
