<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_109_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, cargo tank outer shell separation must be a decimal metre QuantityValue of at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:shipType ?shipType ;
          nltl:constructionDate ?constructionDate ;
          nltl:cargoTankCarriesOil ?carriesOil .

    FILTER (
        ?category IN (
            nltl:polarShipCategoryA,
            nltl:polarShipCategoryB
        )
    )
    FILTER (datatype(?shipType) = xsd:string)
    FILTER (STR(?shipType) != "oil_tanker")
    FILTER (datatype(?constructionDate) = xsd:date)
    FILTER (
        YEAR(?constructionDate) > 2017
        || (
            YEAR(?constructionDate) = 2017
            && (
                MONTH(?constructionDate) > 1
                || (
                    MONTH(?constructionDate) = 1
                    && DAY(?constructionDate) >= 1
                )
            )
        )
    )
    FILTER (datatype(?carriesOil) = xsd:boolean)
    FILTER (?carriesOil = true)

    FILTER NOT EXISTS {
        $this nltl:cargoTankOuterShellSeparation ?separation .
        ?separation a qudt:QuantityValue ;
                    qudt:numericValue ?value ;
                    qudt:unit unit:M .
        FILTER (datatype(?value) = xsd:decimal)
        FILTER (?value >= "0.76"^^xsd:decimal)
    }
}
""" ;
    ] .
</END_SHACL>
