<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:hasValue "oil_tanker" ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship type must be oil_tanker."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1 ;
                sh:minCount 1 ;
                sh:maxExclusive "5000"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The deadweight must be a single metric-tonne quantity below 5,000 tonnes."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date must be on or after 1 January 2017."
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, entire cargo tank length double bottom protection status must be present exactly once as the boolean value true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?type ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?date .

                ?deadweight qudt:numericValue ?value ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?type = "oil_tanker"
                    && ?value < "5000"^^xsd:decimal
                    && ?date >= "2017-01-01"^^xsd:date
                )

                {
                    FILTER NOT EXISTS {
                        $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?status .
                    }
                }
                UNION
                {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?status1, ?status2 .
                    FILTER (?status1 != ?status2)
                }
                UNION
                {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?status .
                    FILTER (
                        datatype(?status) != xsd:boolean
                        || ?status != true
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, wing tank protection status must be present exactly once as the boolean value true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?type ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?date .

                ?deadweight qudt:numericValue ?value ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?type = "oil_tanker"
                    && ?value < "5000"^^xsd:decimal
                    && ?date >= "2017-01-01"^^xsd:date
                )

                {
                    FILTER NOT EXISTS {
                        $this nltl:wingTankProtectionStatus ?status .
                    }
                }
                UNION
                {
                    $this nltl:wingTankProtectionStatus ?status1, ?status2 .
                    FILTER (?status1 != ?status2)
                }
                UNION
                {
                    $this nltl:wingTankProtectionStatus ?status .
                    FILTER (
                        datatype(?status) != xsd:boolean
                        || ?status != true
                    )
                }
            }
        """
    ] .
<END_SHACL>
