<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be specified as a polar ship category value."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date must be specified as a date."
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The tank type must be specified."
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The individual tank capacity must be a volume in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, an oil residue tank or oily bilge water holding tank with individual capacity greater than 30 m3 must be separated from the outer shell by at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:tankType ?tankType ;
                      nltl:individualTankCapacity ?capacityNode .

                ?capacityNode qudt:numericValue ?capacity ;
                              qudt:unit unit:M3 .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (
                    ?tankType IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )
                FILTER (
                    YEAR(?date) > 2017 ||
                    (
                        YEAR(?date) = 2017 &&
                        (
                            MONTH(?date) > 1 ||
                            (MONTH(?date) = 1 && DAY(?date) >= 1)
                        )
                    )
                )
                FILTER (?capacity > 30)

                FILTER NOT EXISTS {
                    $this nltl:tankOuterShellSeparation ?separationNode .
                    ?separationNode qudt:numericValue ?separation ;
                                     qudt:unit unit:M .
                    FILTER (?separation >= 0.76)
                }
            }
        """
    ] .
<END_SHACL>
