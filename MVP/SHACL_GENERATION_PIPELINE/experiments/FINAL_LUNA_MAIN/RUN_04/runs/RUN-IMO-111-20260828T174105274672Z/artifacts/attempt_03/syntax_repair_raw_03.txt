<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one permitted polar ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 2017-01-01, every oil residue tank and oily bilge water holding tank with capacity greater than 30 m3 must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasComponent ?tank .
                ?tank rdf:type ?tankType .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (
                    ?tankType IN (
                        nltl:oilResidueTank,
                        nltl:oilyBilgeWaterHoldingTank
                    )
                )

                FILTER (
                    NOT EXISTS {
                        ?tank nltl:individualTankCapacity ?capacity .
                        ?capacity qudt:numericValue ?capacityValue ;
                                  qudt:unit unit:M3 .
                        FILTER (datatype(?capacityValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?tank nltl:individualTankCapacity ?largeCapacity .
                        ?largeCapacity qudt:numericValue ?largeCapacityValue ;
                                       qudt:unit unit:M3 .
                        FILTER (
                            datatype(?largeCapacityValue) = xsd:decimal &&
                            ?largeCapacityValue > 30
                        )
                        FILTER NOT EXISTS {
                            ?tank nltl:tankOuterShellSeparation ?separation .
                            ?separation qudt:numericValue ?separationValue ;
                                        qudt:unit unit:M .
                            FILTER (
                                datatype(?separationValue) = xsd:decimal &&
                                ?separationValue >= 0.76
                            )
                        }
                    }
                )
            }
        """ ;
    ] .
</END_SHACL>
