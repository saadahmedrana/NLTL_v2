<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_115_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:comminutedAndDisinfected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For comminuted and disinfected sewage, MARPOL compliance, the strict distance greater than 3 nautical miles, and the ice-area distance record with concentration exceeding 1/10 are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType ;
                      nltl:comminutedAndDisinfected ?comminuted .

                FILTER(
                    STR(?dischargeType) = "comminuted_and_disinfected"
                    && ?comminuted = true
                )

                FILTER(
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus ?status .
                        FILTER(
                            datatype(?status) = xsd:boolean
                            && ?status = true
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestIceShelfOrFastIce ?iceDistance .
                        ?iceDistance qudt:numericValue ?distanceValue ;
                                     qudt:unit unit:MI_N .
                        FILTER(
                            datatype(?distanceValue) = xsd:decimal
                            && ?distanceValue > "3"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?concentrationValue ;
                                       qudt:unit unit:UNITLESS .
                        FILTER(
                            datatype(?concentrationValue) = xsd:decimal
                            && ?concentrationValue > "0.1"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance qudt:numericValue ?areaDistanceValue ;
                                      qudt:unit unit:MI_N .
                        FILTER(datatype(?areaDistanceValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
</END_SHACL>
