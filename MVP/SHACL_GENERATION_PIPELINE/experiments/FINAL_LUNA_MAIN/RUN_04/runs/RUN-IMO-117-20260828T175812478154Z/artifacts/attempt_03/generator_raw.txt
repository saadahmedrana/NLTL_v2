<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO117Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage discharge uses a treatment plant, the required approval, MARPOL compliance status, and discharge-distance records must be present and conforming." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeUsesTreatmentPlant true .

                FILTER NOT EXISTS {
                    $this nltl:sewageTreatmentPlantApprovalStatus
                        nltl:evidenceStateApproved
                }

                FILTER NOT EXISTS {
                    $this nltl:sewageTreatmentPlantApprovalStatus ?approvalStatus .
                    FILTER(?approvalStatus != nltl:evidenceStateApproved)
                }

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus
                        nltl:evidenceStateApproved
                }

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certificationStatus .
                    FILTER(?certificationStatus != nltl:evidenceStateApproved)
                }

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certificationStatus1 .
                    $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certificationStatus2 .
                    FILTER(?certificationStatus1 != ?certificationStatus2)
                }

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation11Point1Point2DischargeComplianceStatus true
                }

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation11Point1Point2DischargeComplianceStatus ?dischargeStatus .
                    FILTER(?dischargeStatus != true)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasDischargeDistanceRecord ?record
                }

                FILTER NOT EXISTS {
                    $this nltl:hasDischargeDistanceRecord ?record .
                    FILTER NOT EXISTS {
                        ?record nltl:distanceToNearestLandIceShelfOrFastIce ?landIceDistance .
                        ?landIceDistance qudt:numericValue ?landIceValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER(datatype(?landIceValue) = xsd:decimal)
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasDischargeDistanceRecord ?record .
                    ?record nltl:distanceToNearestLandIceShelfOrFastIce ?landIceDistance .
                    FILTER NOT EXISTS {
                        ?landIceDistance qudt:numericValue ?landIceValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER(datatype(?landIceValue) = xsd:decimal)
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasDischargeDistanceRecord ?record .
                    FILTER NOT EXISTS {
                        ?record nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ?iceDistance .
                        ?iceDistance qudt:numericValue ?iceValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER(datatype(?iceValue) = xsd:decimal)
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasDischargeDistanceRecord ?record .
                    ?record nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ?iceDistance .
                    FILTER NOT EXISTS {
                        ?iceDistance qudt:numericValue ?iceValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER(datatype(?iceValue) = xsd:decimal)
                    }
                }
            }
        """
    ] .
<END_SHACL>
