<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO117ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage discharge uses a treatment plant, the required approved statuses and at least one discharge-distance record must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeUsesTreatmentPlant ?selector .
                FILTER (sameTerm(?selector, true))

                FILTER (
                    NOT EXISTS {
                        $this nltl:sewageTreatmentPlantApprovalStatus nltl:evidenceStateApproved .
                    }
                    ||
                    EXISTS {
                        $this nltl:sewageTreatmentPlantApprovalStatus ?approval .
                        FILTER (!sameTerm(?approval, nltl:evidenceStateApproved))
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus nltl:evidenceStateApproved .
                    }
                    ||
                    EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certification .
                        FILTER (!sameTerm(?certification, nltl:evidenceStateApproved))
                    }
                    ||
                    EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certification1 .
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ?certification2 .
                        FILTER (!sameTerm(?certification1, ?certification2))
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point2DischargeComplianceStatus ?dischargeStatus .
                        FILTER (sameTerm(?dischargeStatus, true))
                    }
                    ||
                    EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point2DischargeComplianceStatus ?dischargeStatus .
                        FILTER (!sameTerm(?dischargeStatus, true))
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDischargeDistanceRecord ?record .
                    }
                )
            }
        """
    ] .

gen:IMO117DischargeDistanceRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:dischargeDistanceRecords ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each discharge-distance record linked to an applicable ship must contain both required nautical-mile distance quantities." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasDischargeDistanceRecord $this .
                ?ship nltl:sewageDischargeUsesTreatmentPlant ?selector .
                FILTER (sameTerm(?selector, true))

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?landIceDistance .
                        ?landIceDistance qudt:numericValue ?landIceValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER (datatype(?landIceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ?iceDistance .
                        ?iceDistance qudt:numericValue ?iceValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER (datatype(?iceValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
