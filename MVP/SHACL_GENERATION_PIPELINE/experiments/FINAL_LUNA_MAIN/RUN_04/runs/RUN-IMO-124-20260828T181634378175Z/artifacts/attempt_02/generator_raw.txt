<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-124> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an Antarctic-area MARPOL Annex V regulation 6.1 discharge, the required ice distances, concentration data, and prohibition on food-waste discharge onto ice must conform." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .

                FILTER ( ?area = "Antarctic_area"^^xsd:string )
                FILTER ( ?discharge = true )

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestFastIce ?fastDistance .
                        ?fastDistance a qudt:QuantityValue ;
                                      qudt:numericValue ?fastValue ;
                                      qudt:unit unit:MI_N .
                        FILTER ( datatype(?fastValue) = xsd:decimal )
                        FILTER ( ?fastValue >= "12"^^xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance a qudt:QuantityValue ;
                                      qudt:numericValue ?areaValue ;
                                      qudt:unit unit:MI_N .
                        FILTER ( datatype(?areaValue) = xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceConcentration ?concentration .
                        ?concentration a qudt:QuantityValue ;
                                       qudt:numericValue ?concentrationValue ;
                                       qudt:unit unit:UNITLESS .
                        FILTER ( datatype(?concentrationValue) = xsd:decimal )
                        FILTER ( ?concentrationValue > "0.1"^^xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearbyConcentration .
                        ?nearbyConcentration a qudt:QuantityValue ;
                                              qudt:numericValue ?nearbyValue ;
                                              qudt:unit unit:UNITLESS .
                        FILTER ( datatype(?nearbyValue) = xsd:decimal )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:foodWasteDischargeOntoIce ?foodDischarge .
                        FILTER ( datatype(?foodDischarge) = xsd:boolean )
                        FILTER ( ?foodDischarge = false )
                    }
                )
            }
        """
    ] .
<END_SHACL>
