<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_124_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a ship operating in Antarctic_area when a MARPOL Annex V regulation 6.1 discharge occurs, the nearest-fast-ice distance must be at least 12 nautical miles, the concentration identifying the relevant ice area must exceed 0.1, the corresponding distance must be recorded, and food waste must not be discharged onto ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge ;
                      nltl:distanceToNearestFastIce ?fastIceDistance ;
                      nltl:nearbyIceConcentration ?nearbyConcentration ;
                      nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?iceAreaDistance ;
                      nltl:foodWasteDischargeOntoIce ?foodWasteDischarge .

                ?fastIceDistance qudt:numericValue ?fastIceValue ;
                                 qudt:unit unit:MI_N .
                ?nearbyConcentration qudt:numericValue ?nearbyValue ;
                                      qudt:unit unit:UNITLESS .
                ?iceAreaDistance qudt:numericValue ?iceAreaDistanceValue ;
                                  qudt:unit unit:MI_N .

                FILTER(
                    STR(?area) = "Antarctic_area" &&
                    ?discharge = true &&
                    ?fastIceValue >= 12 &&
                    ?nearbyValue > 0.1 &&
                    ?foodWasteDischarge = false
                )
            }
        """
    ] .
<END_SHACL>
