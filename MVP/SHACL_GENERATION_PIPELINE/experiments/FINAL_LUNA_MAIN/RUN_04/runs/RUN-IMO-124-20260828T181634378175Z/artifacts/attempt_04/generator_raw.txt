<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO124AntarcticDischargeShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an Antarctic MARPOL Annex V regulation 6.1 discharge, the minimum distance, relevant ice-area distance record, and prohibition on food-waste discharge onto ice must all conform." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .

                FILTER (
                    ?area = "Antarctic_area"^^xsd:string &&
                    ?discharge = true
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestFastIce ?fastDistance .
                        ?fastDistance qudt:numericValue ?fastValue ;
                                      qudt:unit unit:MI_N .
                        FILTER (?fastValue >= 12)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?concentrationValue ;
                                       qudt:unit unit:UNITLESS .
                        FILTER (?concentrationValue > 0.1)

                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance qudt:numericValue ?areaDistanceValue ;
                                      qudt:unit unit:MI_N .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:foodWasteDischargeOntoIce ?foodWaste .
                        FILTER (?foodWaste = false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
