<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO26_012_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:node gen:IMO26_012_HeadingMeansShape
    ] ;
    sh:sparql [
        sh:message "Each represented non-magnetic heading means must be independent from another distinct non-magnetic heading means of the same ship." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?other .
                    FILTER (?other != ?means)
                    ?means nltl:independentFromHeadingMeans ?other .
                }
            }
        """
    ] .

gen:IMO26_012_HeadingMeansShape
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:nodeKind sh:IRI ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each represented non-magnetic heading means must be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each represented non-magnetic heading means must be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each represented non-magnetic heading means must declare an independent heading means."
    ] .
<END_SHACL>
