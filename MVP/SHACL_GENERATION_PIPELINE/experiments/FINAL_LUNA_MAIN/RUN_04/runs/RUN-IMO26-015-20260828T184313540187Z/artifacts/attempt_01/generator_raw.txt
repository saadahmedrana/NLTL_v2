<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_015_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship's SOLAS chapter I certification status is required to determine chapter 11 applicability."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter11Applicable ;
        sh:datatype xsd:boolean ;
    ] ;
    sh:sparql [
        sh:message "Polar Code chapter 11 may apply only when the ship is certified in accordance with SOLAS chapter I." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter11Applicable ?applicable .
                FILTER (?applicable = true)
                FILTER NOT EXISTS {
                    $this nltl:solasChapterICertified true .
                }
            }
        """
    ] .
<END_SHACL>
