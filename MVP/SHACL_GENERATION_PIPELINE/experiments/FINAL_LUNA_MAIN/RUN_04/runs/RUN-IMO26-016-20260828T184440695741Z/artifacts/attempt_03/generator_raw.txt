            PREFIX qudt: <http://qudt.org/schema/qudt/>

                        ?operates = true &&

                                            datatype(?lengthValue) = xsd:decimal &&
                                            ?lengthValue >= 24

                                    }
                                )

                                (
                                    EXISTS {
                                        $this rdf:type nltl:pleasureYacht .
                                        $this nltl:grossTonnage ?yachtTonnageQuantity .
                                        ?yachtTonnageQuantity qudt:numericValue ?yachtTonnage ;
                                                            qudt:unit qudt:UNITLESS .
                                        $this nltl:engagedInTrade false .

                                            datatype(?yachtTonnage) = xsd:decimal &&

                                    }

                                            datatype(?cargoTonnage) = xsd:decimal &&

                                    }
                                )

                        (
                            (
                                EXISTS { $this rdf:type nltl:fishingVessel . }
                                &&
                                NOT EXISTS {
                                    $this nltl:lengthOverall ?fishingLengthQuantity .
                                    ?fishingLengthQuantity qudt:numericValue ?fishingLength ;
                                                            qudt:unit qudt:M .
                                    FILTER (
                                        datatype(?fishingLength) = xsd:decimal &&
                                        ?fishingLength < 24
                                    )
                                }
                            )
                            ||
                            (
                                EXISTS { $this rdf:type nltl:pleasureYacht . }
                                &&
                                NOT EXISTS {
                                    $this nltl:grossTonnage ?pleasureTonnageQuantity .
                                    ?pleasureTonnageQuantity qudt:numericValue ?pleasureTonnage ;
                                                              qudt:unit qudt:UNITLESS .
                                    $this nltl:engagedInTrade false .
                                    FILTER (
                                        datatype(?pleasureTonnage) = xsd:decimal &&
                                        ?pleasureTonnage >= 300
                                    )
                                }

                                NOT EXISTS {
                                    $this nltl:grossTonnage ?pleasureLowTonnageQuantity .
                                    ?pleasureLowTonnageQuantity qudt:numericValue ?pleasureLowTonnage ;
                                                               qudt:unit qudt:UNITLESS .
                                    FILTER (
                                        datatype(?pleasureLowTonnage) = xsd:decimal &&
                                        ?pleasureLowTonnage < 300
                                    )
                                }
                            )
                            ||
                            (
                                EXISTS { $this rdf:type nltl:cargoShip . }
                                &&
                                NOT EXISTS {
                                    $this nltl:grossTonnage ?cargoQualifyingQuantity .
                                    ?cargoQualifyingQuantity qudt:numericValue ?cargoQualifyingTonnage ;

                                        ?cargoQualifyingTonnage < 500

                                                                 qudt:unit qudt:UNITLESS .

                                }

                    )
