<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

[
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:sparql [
        sh:message "A ship with a construction stage date on or after 2003-09-01 and a contract date before 2010-01-01 must specify exactly the 2002 ice-class regulation edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate ;
                      nltl:constructionContractDate ?contractDate .

                FILTER (
                    (
                        YEAR(?stageDate) > 2003 ||
                        (
                            YEAR(?stageDate) = 2003 &&
                            (
                                MONTH(?stageDate) > 9 ||
                                (
                                    MONTH(?stageDate) = 9 &&
                                    DAY(?stageDate) >= 1
                                )
                            )
                        )
                    ) &&
                    YEAR(?contractDate) < 2010
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                              nltl:iceClassRegulationEdition2002
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?edition .
                        FILTER (
                            ?edition != nltl:iceClassRegulationEdition2002
                        )
                    }
                )
            }
        """
    ]
] .
<END_SHACL>
