<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify its intended ice-operating waterline and its profile points."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify an upper ice waterline and its profile points."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal, pointwise, the envelope of the highest intended ice-operating waterline points at each longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?this
WHERE {
    ?this nltl:hasIntendedIceOperatingWaterline ?intended ;
          nltl:hasUpperIceWaterline ?upper .

    {
        SELECT ?intended ?x (MAX(?yv) AS ?maxY)
        WHERE {
            ?intended nltl:hasWaterlineProfilePoint ?intendedPoint .
            ?intendedPoint nltl:longitudinalPosition ?xQuantity ;
                           nltl:verticalCoordinate ?yQuantity .
            ?xQuantity qudt:numericValue ?x .
            ?yQuantity qudt:numericValue ?yv .
        }
        GROUP BY ?intended ?x
    }

    {
        ?upper nltl:hasWaterlineProfilePoint ?upperPoint .
        ?upperPoint nltl:longitudinalPosition ?upperXQuantity ;
                    nltl:verticalCoordinate ?upperYQuantity .
        ?upperXQuantity qudt:numericValue ?upperX .
        ?upperYQuantity qudt:numericValue ?upperY .
        FILTER (?upperX = ?x && ?upperY != ?maxY)
    }
    UNION
    {
        FILTER NOT EXISTS {
            ?upper nltl:hasWaterlineProfilePoint ?matchingPoint .
            ?matchingPoint nltl:longitudinalPosition ?matchingXQuantity ;
                           nltl:verticalCoordinate ?matchingYQuantity .
            ?matchingXQuantity qudt:numericValue ?matchingX .
            ?matchingYQuantity qudt:numericValue ?matchingY .
            FILTER (?matchingX = ?x && ?matchingY = ?maxY)
        }
    }
}
""" ;
    ] .

gen:TRF_011_IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_ProfilePointShape ;
        sh:message "Each represented ice waterline must contain at least one waterline profile point."
    ] .

gen:TRF_011_ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF_011_LengthQuantityShape ;
        sh:message "Each waterline profile point must have one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF_011_LengthQuantityShape ;
        sh:message "Each waterline profile point must have one vertical coordinate in metres."
    ] .

gen:TRF_011_LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
