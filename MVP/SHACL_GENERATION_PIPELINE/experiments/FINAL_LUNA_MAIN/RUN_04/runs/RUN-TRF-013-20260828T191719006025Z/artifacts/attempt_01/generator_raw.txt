<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF013LengthQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF013ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that it is ice-strengthened when this requirement applies."
    ] ;

    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "An ice class is required for the applicable ice-strengthened ship."
    ] ;

    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:message "The applicable ship must identify its upper ice waterline."
    ] ;

    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:message "The applicable ship must identify its lower ice waterline."
    ] ;

    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The operating draught at the fore perpendicular is required in metres."
    ] ;

    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The operating draught at the aft perpendicular is required in metres."
    ] ;

    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The maximum ice-class draught at the fore perpendicular is required in metres."
    ] ;

    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The maximum ice-class draught at the aft perpendicular is required in metres."
    ] ;

    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The minimum ice-class draught at the fore perpendicular is required in metres."
    ] ;

    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node gen:TRF013LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The minimum ice-class draught at the aft perpendicular is required in metres."
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "At the fore perpendicular, the operating draught must be between the minimum and maximum ice-class draughts, and at the aft perpendicular it must likewise be between its corresponding bounds." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingForeDraught/qudt:numericValue ?operatingFore ;
                       nltl:operatingAftDraught/qudt:numericValue ?operatingAft ;
                       nltl:maximumIceClassDraughtFore/qudt:numericValue ?maximumFore ;
                       nltl:maximumIceClassDraughtAft/qudt:numericValue ?maximumAft ;
                       nltl:minimumIceClassDraughtFore/qudt:numericValue ?minimumFore ;
                       nltl:minimumIceClassDraughtAft/qudt:numericValue ?minimumAft .
                FILTER (
                    !(
                        ?minimumFore <= ?operatingFore &&
                        ?operatingFore <= ?maximumFore &&
                        ?minimumAft <= ?operatingAft &&
                        ?operatingAft <= ?maximumAft
                    )
                )
            }
        """
    ] .
<END_SHACL>
