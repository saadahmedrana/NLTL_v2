<urn:nltl:generated-shape:TRF-014-PostCutoffNoMarkingShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:minInclusive "2007-07-01"^^xsd:date
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a post-cutoff ship, marking is not required only when the summer load line fresh-water draught is not higher than the UIWL." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:summerLoadLineFreshWaterDraught ?summer .
                $this nltl:upperIceWaterlineDraught ?uiwl .
                ?summer qudt:numericValue ?summerValue .
                ?uiwl qudt:numericValue ?uiwlValue .
                FILTER ( ?summerValue > ?uiwlValue )
            }
        """
    ] .
