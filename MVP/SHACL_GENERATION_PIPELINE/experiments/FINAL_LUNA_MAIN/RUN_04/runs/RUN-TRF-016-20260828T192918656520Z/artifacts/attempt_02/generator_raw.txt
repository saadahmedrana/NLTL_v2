<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix qudt-unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF_016_UpperIceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasUpperIceWaterline ]
            nltl:displacementAtUpperIceWaterline
        ) ;
        sh:node gen:TRF_016_MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked upper ice waterline must have exactly one ship-owned displacement at upper ice waterline with metric-ton units."
    ] .

gen:TRF_016_LowerIceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline .

gen:TRF_016_MassQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue qudt-unit:TON_Metric ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF_016_LengthQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue qudt-unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF_016_BallastTankNotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:TRF_016_BallastTankNotNeededShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:TRF_016_BallastTankProtectedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:TRF_016_BallastTankShape
    a sh:NodeShape ;
    sh:class nltl:ballastTank ;
    sh:or (
        gen:TRF_016_BallastTankNotApplicableShape
        gen:TRF_016_BallastTankNotNeededShape
        gen:TRF_016_BallastTankProtectedShape
    ) .

gen:TRF_016_NavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node gen:TRF_016_LowerIceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node gen:TRF_016_UpperIceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node gen:TRF_016_MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node gen:TRF_016_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node gen:TRF_016_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node gen:TRF_016_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node gen:TRF_016_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node gen:TRF_016_BallastTankShape
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, draught amidships must reach the LIWL draught and forward draught must meet the level-ice-thickness displacement limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:draughtAmidships ?draughtQ ;
                       nltl:liwlDraughtAmidships ?liwlQ ;
                       nltl:forwardDraught ?forwardQ ;
                       nltl:displacementAtUpperIceWaterline ?displacementQ ;
                       nltl:levelIceThickness ?thicknessQ .

                ?draughtQ q:numericValue ?draught .
                ?liwlQ q:numericValue ?liwl .
                ?forwardQ q:numericValue ?forward .
                ?displacementQ q:numericValue ?displacement .
                ?thicknessQ q:numericValue ?thickness .

                BIND(
                    IF(
                        ((2 + (0.00025 * ?displacement)) * ?thickness)
                        <
                        (4 * ?thickness),
                        ((2 + (0.00025 * ?displacement)) * ?thickness),
                        (4 * ?thickness)
                    )
                    AS ?minimumForward
                )

                FILTER(
                    (?draught < ?liwl)
                    ||
                    (?forward < ?minimumForward)
                )
            }
        """
    ] .

gen:TRF_016_NotNavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:TRF_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF_016_NotNavigatingInIceShape
        gen:TRF_016_NavigatingInIceShape
    ) .
<END_SHACL>
