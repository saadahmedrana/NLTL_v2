<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-016-UIWLShape>
    a sh:NodeShape ;
    sh:class nltl:iceWaterline .

<urn:nltl:generated-shape:TRF-016-BallastTankShape>
    a sh:NodeShape ;
    sh:class nltl:ballastTank ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "A ballast tank above the LIWL that is needed to reach the LIWL must have freezing prevention." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:situatedAboveLowerIceWaterline ?above ;
                      nltl:usedToReachLowerIceWaterline ?needed .
                FILTER (?above = true && ?needed = true)
                FILTER NOT EXISTS {
                    $this nltl:freezingPreventionPresent true
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-016-ActiveShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-UIWLShape>
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node <urn:nltl:generated-shape:TRF-016-BallastTankShape>
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI
    ] ;
    sh:property [
        sh:path nltl:waterlineDisplacement ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path ( nltl:draughtAmidships qudt:numericValue ) ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path ( nltl:draughtAmidships qudt:unit ) ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] ;
    sh:property [
        sh:path ( nltl:liwlDraughtAmidships qudt:numericValue ) ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path ( nltl:liwlDraughtAmidships qudt:unit ) ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] ;
    sh:property [
        sh:path ( nltl:forwardDraught qudt:numericValue ) ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path ( nltl:forwardDraught qudt:unit ) ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] ;
    sh:property [
        sh:path ( nltl:levelIceThickness qudt:numericValue ) ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path ( nltl:levelIceThickness qudt:unit ) ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] ;
    sh:property [
        sh:path ( nltl:waterlineDisplacement qudt:numericValue ) ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path ( nltl:waterlineDisplacement qudt:unit ) ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric>
    ] ;
    sh:sparql [
        sh:message "Forward draught must be at least min((2 + 0.00025 * UIWL displacement) * level ice thickness, 4 * level ice thickness), and draught amidships must reach the LIWL draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:draughtAmidships ?draughtNode ;
                      nltl:liwlDraughtAmidships ?liwlNode ;
                      nltl:forwardDraught ?forwardNode ;
                      nltl:levelIceThickness ?iceNode ;
                      nltl:waterlineDisplacement ?displacementNode .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?liwlNode qudt:numericValue ?liwl ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?forwardNode qudt:numericValue ?forward ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .
                ?iceNode qudt:numericValue ?ice ;
                         qudt:unit <http://qudt.org/vocab/unit/M> .
                ?displacementNode qudt:numericValue ?displacement ;
                                   qudt:unit <http://qudt.org/vocab/unit/TON_Metric> .
                BIND (MIN(
                    (2 + (0.00025 * ?displacement)) * ?ice,
                    4 * ?ice
                ) AS ?requiredForward)
                FILTER (?draught < ?liwl || ?forward < ?requiredForward)
            }
        """
    ] .
