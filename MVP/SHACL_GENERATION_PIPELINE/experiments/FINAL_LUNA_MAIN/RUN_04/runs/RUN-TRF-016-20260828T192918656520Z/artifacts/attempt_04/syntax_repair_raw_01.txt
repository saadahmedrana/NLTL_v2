@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-016-MassQuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-016-LIWLShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-016-UIWLShape>
    a sh:NodeShape ;
    sh:class nltl:iceWaterline .

<urn:nltl:generated-shape:TRF-016-BallastTankShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "A ballast tank situated above the LIWL and needed to reach the LIWL shall have freezing prevention present." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#situatedAboveLowerIceWaterline> ?above ;
                      <https://w3id.org/nltl/vocab#usedToReachLowerIceWaterline> ?needed ;
                      <https://w3id.org/nltl/vocab#freezingPreventionPresent> ?prevention .
                FILTER (?above = true && ?needed = true && ?prevention != true)
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-016-UpperIceWaterlineShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-016-UIWLShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-016-LIWLShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node <urn:nltl:generated-shape:TRF-016-BallastTankShape>
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-016-MassQuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:waterlineDisplacement ;
        sh:node <urn:nltl:generated-shape:TRF-016-MassQuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, draught amidships shall be at least the LIWL draught amidships." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#navigatingInIce> true ;
                      <https://w3id.org/nltl/vocab#draughtAmidships> ?draughtNode ;
                      <https://w3id.org/nltl/vocab#hasLowerIceWaterline> ?liwl .
                ?draughtNode <http://qudt.org/schema/qudt/numericValue> ?draught ;
                             <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?liwl <https://w3id.org/nltl/vocab#liwlDraughtAmidships> ?liwlNode .
                ?liwlNode <http://qudt.org/schema/qudt/numericValue> ?liwlDraught ;
                          <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                FILTER (?draught < ?liwlDraught)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Forward draught shall be at least the applicable level-ice-thickness formula result." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#forwardDraught> ?forwardNode ;
                      <https://w3id.org/nltl/vocab#waterlineDisplacement> ?displacementNode ;
                      <https://w3id.org/nltl/vocab#levelIceThickness> ?thicknessNode ;
                      <https://w3id.org/nltl/vocab#hasUpperIceWaterline> ?uiwl .
                ?forwardNode <http://qudt.org/schema/qudt/numericValue> ?forward ;
                             <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?displacementNode <http://qudt.org/schema/qudt/numericValue> ?displacement ;
                                  <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/TON_Metric> .
                ?thicknessNode <http://qudt.org/schema/qudt/numericValue> ?thickness ;
                                <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?uiwl a <https://w3id.org/nltl/vocab#iceWaterline> .
                BIND (2 + (0.00025 * ?displacement) AS ?uncapped)
                BIND (4 * ?thickness AS ?cap)
                BIND (IF(?uncapped < ?cap, ?uncapped, ?cap) AS ?expected)
                FILTER (?forward < ?expected)
            }
        """
    ] .

</END_SHACL>
