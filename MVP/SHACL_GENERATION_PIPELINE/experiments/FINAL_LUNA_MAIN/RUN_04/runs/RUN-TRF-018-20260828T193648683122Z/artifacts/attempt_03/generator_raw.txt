<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:CalculatedRequiredPowerPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:calculatedRequiredPower ;
    sh:nodeKind sh:IRI ;
    sh:class qudt:QuantityValue ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloW ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MaximumContinuousRatingPowerPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:maximumContinuousRatingPower ;
    sh:nodeKind sh:IRI ;
    sh:class qudt:QuantityValue ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloW ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceClassIaSuperCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iceClassIaSuper ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property gen:CalculatedRequiredPowerPropertyShape ;
    sh:property gen:MaximumContinuousRatingPowerPropertyShape .

gen:IceClassIaIbIcCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property gen:CalculatedRequiredPowerPropertyShape ;
    sh:property gen:MaximumContinuousRatingPowerPropertyShape .

gen:IceClassIiIiiCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF018ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [ sh:node gen:IceClassIaSuperCaseShape ]
        [ sh:node gen:IceClassIaIbIcCaseShape ]
        [ sh:node gen:IceClassIiIiiCaseShape ]
    ) ;
    sh:sparql [
        sh:message "For applicable ice classes, maximumContinuousRatingPower must be at least calculatedRequiredPower and the applicable regulatory minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?ratingNode .

                ?calculatedNode qudt:numericValue ?calculatedPower ;
                                qudt:unit qudt:KiloW .

                ?ratingNode qudt:numericValue ?ratingPower ;
                            qudt:unit qudt:KiloW .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        (
                            ?ratingPower < ?calculatedPower ||
                            ?ratingPower < "2800"^^xsd:decimal
                        )
                    ) ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        (
                            ?ratingPower < ?calculatedPower ||
                            ?ratingPower < "1000"^^xsd:decimal
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
