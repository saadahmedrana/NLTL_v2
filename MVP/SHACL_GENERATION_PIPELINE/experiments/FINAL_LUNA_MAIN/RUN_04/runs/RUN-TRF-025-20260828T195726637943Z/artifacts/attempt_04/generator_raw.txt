gen:TRF025Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one controlled ice class is required before applicability is selected."
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one bulbous-bow boolean is required before applicability is selected."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false
            ] ;
            sh:node gen:TRF025ApplicableShape
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:in (
                            nltl:iceClassIa
                            nltl:iceClassIb
                            nltl:iceClassIc
                            nltl:iceClassIi
                            nltl:iceClassIii
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bulbousBowPresent ;
                        sh:hasValue true
                    ]
                ]
            )
        ]
    ) .
