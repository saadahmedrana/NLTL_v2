        sh:path nltl:coefficientG3 ;

        sh:node nltlgen:CoefficientG3ValueShape

    sh:property [
        sh:path nltl:clampedHullGeometryTerm ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:ClampedHullGeometryValueShape
    ] ;
    sh:sparql [
        sh:message "clampedHullGeometryTerm must equal the clamped value of ((shipLength * iceClassDraught) / shipBreadth squared) cubed, within a relative tolerance of 1e-6 plus an absolute fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthValue ;
                      nltl:iceClassDraught ?draughtValue ;
                      nltl:shipBreadth ?breadthValue ;
                      nltl:clampedHullGeometryTerm ?reportedValue .

                ?lengthValue qudt:numericValue ?length .
                ?draughtValue qudt:numericValue ?draught .
                ?breadthValue qudt:numericValue ?breadth .
                ?reportedValue qudt:numericValue ?reported .

                BIND(
                    (
                        (?length * ?draught) /
                        (?breadth * ?breadth)
                    ) *
                    (
                        (?length * ?draught) /
                        (?breadth * ?breadth)
                    ) *
                    (
                        (?length * ?draught) /
                        (?breadth * ?breadth)
                    )
                    AS ?raw
                )
                BIND(
                    IF(?raw < 5, 5,
                        IF(?raw > 20, 20, ?raw)
                    )
                    AS ?expected
                )
                BIND(
                    IF(?reported >= ?expected,
                        ?reported - ?expected,
                        ?expected - ?reported
                    )
                    AS ?difference
                )
                BIND(
                    (?expected * 0.000001) + 0.000000000001
                    AS ?tolerance
                )
                FILTER(?difference > ?tolerance)
            }
        """
    ] .
