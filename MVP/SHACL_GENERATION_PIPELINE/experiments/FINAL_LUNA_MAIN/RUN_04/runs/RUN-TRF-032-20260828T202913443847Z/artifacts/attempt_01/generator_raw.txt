<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_032_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship that is not ice strengthened, when the regulation-derived scantling is less than the classification-society-required scantling, the selected design scantling must equal the classification-society-required scantling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:regulationRequiredScantling ?regulationScantling ;
                      nltl:classificationSocietyRequiredScantling ?classificationScantling .

                ?regulationScantling qudt:numericValue ?regulationValue .
                ?classificationScantling qudt:numericValue ?classificationValue .

                FILTER ( ?regulationValue < ?classificationValue )

                FILTER NOT EXISTS {
                    $this nltl:selectedDesignScantling ?selectedScantling .
                    ?selectedScantling qudt:numericValue ?selectedValue .
                    FILTER ( ?selectedValue = ?classificationValue )
                }
            }
        """
    ] .
<END_SHACL>
