<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF032Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF032NotIceStrengthenedBranch
        gen:TRF032IceStrengthenedBranch
    ) ;
    sh:sparql [
        sh:message "When the ship is not ice strengthened and the regulation-required scantling is less than the classification-society-required scantling, the selected design scantling must equal the classification-society-required scantling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus ?iceStatus .
                $this nltl:regulationRequiredScantling ?regulationQuantity .
                ?regulationQuantity qudt:numericValue ?regulationValue .
                $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                ?classificationQuantity qudt:numericValue ?classificationValue .
                $this nltl:selectedDesignScantling ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                FILTER (
                    ?iceStatus = false &&
                    ?regulationValue < ?classificationValue &&
                    ?selectedValue != ?classificationValue
                )
            }
        """
    ] .

gen:TRF032NotIceStrengthenedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gen:TRF032IceStrengthenedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
