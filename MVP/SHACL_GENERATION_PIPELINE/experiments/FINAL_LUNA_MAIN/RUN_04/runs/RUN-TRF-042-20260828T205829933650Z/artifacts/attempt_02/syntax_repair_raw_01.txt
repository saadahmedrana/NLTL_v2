@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:HeightQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SpacingQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;

    ] ;
    sh:property [

        sh:hasValue <http://qudt.org/vocab/unit/M> ;

    ] .

gen:CorrosionNormalBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node [
            sh:node gen:LengthQuantityShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive 2.0 ;
                sh:maxInclusive 2.0
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:CorrosionLowerApprovedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node [
            sh:node gen:LengthQuantityShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:maxExclusive 2.0
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF042PlatingShape a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:HeightQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "designIceLoadHeight is required in metres."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:SpacingQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "spacing is required in metres."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "longitudinalPlatingFactorF2 is required as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1 ;
        sh:message "materialApprovalStatus values must be canonical evidenceState individuals."
    ] ;
    sh:or (
        gen:CorrosionNormalBranch
        gen:CorrosionLowerApprovedBranch
    ) ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable f2 formula within max(1e-6*ABS(expected),1e-12)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?actual .
                FILTER (
                    ?s = 0 ||
                    ?h / ?s > 1.8 ||
                    ABS(
                        ?actual -
                        IF(
                            (?h / ?s) <= 1,
                            0.4 / (?h / ?s) + 0.6,
                            1.4 - 0.4 * (?h / ?s)
                        )
                    ) >
                    IF(
                        1e-6 * ABS(
                            IF(
                                (?h / ?s) <= 1,
                                0.4 / (?h / ?s) + 0.6,
                                1.4 - 0.4 * (?h / ?s)
                            )
                        ) > 1e-12,
                        1e-6 * ABS(
                            IF(
                                (?h / ?s) <= 1,
                                0.4 / (?h / ?s) + 0.6,
                                1.4 - 0.4 * (?h / ?s)
                            )
                        ),
                        1e-12
                    )
                )
            }
        """
    ] .
<END_SHACL>
