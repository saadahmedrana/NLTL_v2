<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF042_PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "designIceLoadHeight must be a single metre quantity value."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "spacing must be a single metre quantity value."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "longitudinalPlatingFactorF2 must be a single dimensionless quantity value."
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "corrosionAbrasionAddition must be a single millimetre quantity value."
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1 ;
        sh:message "materialApprovalStatus values must be canonical evidenceState individuals."
    ] ;
    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable f2 formula for 0 < designIceLoadHeight/spacing <= 1.8, within the stated tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .

                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?actual .

                FILTER (?s > 0.0)
                FILTER (?h > 0.0)

                BIND((?h / ?s) AS ?ratio)
                FILTER (?ratio <= 1.8)

                BIND(
                    IF(?ratio <= 1.0,
                       (0.4 / ?ratio) + 0.6,
                       1.4 - (0.4 * ?ratio))
                    AS ?expected
                )
                BIND(
                    IF(?expected >= 0.0, ?expected, 0.0 - ?expected)
                    AS ?expectedAbs
                )
                BIND(
                    IF((0.000001 * ?expectedAbs) >= 0.000000000001,
                       (0.000001 * ?expectedAbs),
                       0.000000000001)
                    AS ?tolerance
                )

                FILTER (
                    (?actual < (?expected - ?tolerance)) ||
                    (?actual > (?expected + ?tolerance))
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A corrosionAbrasionAddition below 2 mm requires a maintained special surface coating and approved material evidence; additions above 2 mm are not permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc .

                FILTER (
                    (?tc > 2.0) ||
                    (
                        (?tc < 2.0) &&
                        NOT EXISTS {
                            $this nltl:specialSurfaceCoatingMaintained true .
                        }
                    ) ||
                    (
                        (?tc < 2.0) &&
                        NOT EXISTS {
                            $this nltl:materialApprovalStatus nltl:evidenceStateApproved .
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
