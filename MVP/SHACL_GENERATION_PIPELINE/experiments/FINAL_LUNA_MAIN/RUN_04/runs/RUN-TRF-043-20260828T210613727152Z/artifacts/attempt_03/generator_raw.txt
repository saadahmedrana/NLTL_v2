<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MeterQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TerminationBoundaryShape
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:tankBoundaryPlating ]
        [ sh:class nltl:tankTop ]
    ) .

gen:FramingIceStrengtheningTerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:node gen:TerminationBoundaryShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extensionQuantity .
                ?extensionQuantity qudt:numericValue ?extension ;
                    qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (
                    ?extension <= 250 &&
                    NOT EXISTS {
                        $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true
                    }
                )
            }
        """
    ] .

gen:ShipFramingIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:FramingIceStrengtheningTerminationCaseShape
    ] .

gen:FrameVerticalExtentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The framing vertical extents do not satisfy the applicable Table 4-6 alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:hullRegion ?hullRegion ;
                    nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQuantity ;
                    nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowQuantity ;
                    nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .
                ?aboveQuantity qudt:numericValue ?above ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                ?belowQuantity qudt:numericValue ?below ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    !(
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?hullRegion = nltl:bowRegion &&
                            ?reachesTankTop = true &&
                            ?above >= 1.2 &&
                            ?below >= 2.0
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?hullRegion = nltl:midbodyRegion &&
                            ?above >= 1.2 &&
                            ?below >= 2.0
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?hullRegion = nltl:sternRegion &&
                            ?above >= 1.2 &&
                            ?below >= 1.6
                        ) ||
                        (
                            (
                                ?iceClass = nltl:iceClassIa ||
                                ?iceClass = nltl:iceClassIb ||
                                ?iceClass = nltl:iceClassIc
                            ) &&
                            ?hullRegion = nltl:bowRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.6
                        ) ||
                        (
                            (
                                ?iceClass = nltl:iceClassIa ||
                                ?iceClass = nltl:iceClassIb ||
                                ?iceClass = nltl:iceClassIc
                            ) &&
                            ?hullRegion = nltl:midbodyRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.3
                        ) ||
                        (
                            (
                                ?iceClass = nltl:iceClassIa ||
                                ?iceClass = nltl:iceClassIb ||
                                ?iceClass = nltl:iceClassIc
                            ) &&
                            ?hullRegion = nltl:sternRegion &&
                            ?above >= 1.0 &&
                            ?below >= 1.0
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
