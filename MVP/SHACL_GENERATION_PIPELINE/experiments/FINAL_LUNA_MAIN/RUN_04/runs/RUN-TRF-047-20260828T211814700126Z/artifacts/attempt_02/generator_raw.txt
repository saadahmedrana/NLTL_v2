<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF_047_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable main frame or intermediate ice frame must have a strengthened part and a lower end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                {
                    $this a nltl:mainFrame .
                }
                UNION
                {
                    $this a nltl:intermediateIceFrame .
                }
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasStrengthenedPart ?strengthenedPart .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasLowerEnd ?lowerEnd .
                    }
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A main frame lower end must be attached to a deck, tank boundary plating, tank top, or ice stringer." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:mainFrame ;
                      nltl:hasLowerEnd ?lowerEnd .
                FILTER NOT EXISTS {
                    ?lowerEnd nltl:hasAttachedSupportingStructure ?supportingStructure .
                    ?supportingStructure a ?supportingStructureType .
                    FILTER (
                        ?supportingStructureType IN (
                            nltl:deckStructure,
                            nltl:tankBoundaryPlating,
                            nltl:tankTop,
                            nltl:iceStringer
                        )
                    )
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A qualifying intermediate ice frame lower end must have an allowed attached structure or a fully qualified horizontal connection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:intermediateIceFrame ;
                      nltl:hasLowerEnd ?lowerEnd .
                FILTER (
                    NOT EXISTS {
                        ?lowerEnd nltl:hasAttachedSupportingStructure ?supportingStructure .
                        ?supportingStructure a ?supportingStructureType .
                        FILTER (
                            ?supportingStructureType IN (
                                nltl:deckStructure,
                                nltl:tankBoundaryPlating,
                                nltl:tankTop,
                                nltl:iceStringer
                            )
                        )
                    }
                    &&
                    NOT EXISTS {
                        $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true ;
                              nltl:sameScantlingsAsMainFrame true .
                        ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?supportingStructure ;
                                  nltl:hasHorizontalConnectionMember ?horizontalMember .
                        ?horizontalMember nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                        ?adjacentMainFrame a nltl:mainFrame .
                    }
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A main frame below the lower edge of the ice belt must be ice strengthened." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:mainFrame .
                FILTER NOT EXISTS {
                    $this nltl:mainFrameBelowIceBeltStrengthened true .
                }
            }
        """ ;
    ] .
<END_SHACL>
