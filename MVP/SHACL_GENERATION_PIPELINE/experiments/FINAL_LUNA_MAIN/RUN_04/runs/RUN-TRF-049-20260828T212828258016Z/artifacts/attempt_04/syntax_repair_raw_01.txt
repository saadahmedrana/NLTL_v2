                sh:path nltl:hasFrameAttachment ;
                sh:class nltl:frameAttachmentRecord ;
                sh:minCount 1 ;
                sh:node s:FrameAttachmentShape
            ] ;
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:or (
                    [
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:property [
                            sh:path nltl:webPlateConnectionSideCount ;
                            sh:datatype xsd:integer ;
                            sh:minCount 1 ;
                            sh:maxCount 1 ;
                            sh:hasValue 2
                        ]
                    ]
                    [
                        sh:datatype xsd:boolean ;
                        sh:hasValue false
                    ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked supporting structure must be represented on the corresponding frame attachment, with the reverse attachment relationship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?structure .
                FILTER NOT EXISTS {
                    ?attachment nltl:hasAttachedSupportingStructure ?structure .
                    ?structure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every installed connection bracket must be at least as thick as the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .
