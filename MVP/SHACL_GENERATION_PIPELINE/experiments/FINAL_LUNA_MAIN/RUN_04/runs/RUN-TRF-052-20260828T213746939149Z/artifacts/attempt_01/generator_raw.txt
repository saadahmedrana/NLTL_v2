                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceClass ;

            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

        [
            sh:property [
                sh:path nltl:frameAsymmetrical ;

            sh:property [

    sh:sparql [
        sh:message "The antitripping support region does not satisfy the required frame-span and ice-class scope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpan ?spanNode .
                ?spanNode qudt:numericValue ?span .
                $this nltl:iceClass ?ice .
                FILTER (
                    (
                        ?span > 4 &&
                        (
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIaSuper &&
                        (
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIa &&
                        (
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        (
                            ?ice = nltl:iceClassIb ||
                            ?ice = nltl:iceClassIc
                        ) &&
                        NOT EXISTS {
                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                        }
                    )
                )
            }
        """
    ] .
