<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF052Shape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasAntitrippingSupportRegion ;
        sh:class nltl:hullRegionValue
    ] ;
    sh:sparql [
        sh:message "The frame does not satisfy the TRF-052 applicability, spacing, or antitripping-support-region requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:equivalentSupportStatusByDirectCalculation ?alternative .
                    FILTER(?alternative = true)
                }
                FILTER(
                    !EXISTS {
                        $this nltl:frameAsymmetrical ?asymmetrical .
                        FILTER(?asymmetrical = true || ?asymmetrical = false)
                    }
                    ||
                    !EXISTS {
                        $this nltl:frameWebAngleToShell ?angleNode .
                        ?angleNode qudt:numericValue ?angleValue .
                        ?angleNode qudt:unit unit:DEG .
                    }
                    ||
                    (
                        (
                            EXISTS {
                                $this nltl:frameAsymmetrical ?asymmetricalTrue .
                                FILTER(?asymmetricalTrue = true)
                            }
                            ||
                            EXISTS {
                                $this nltl:frameWebAngleToShell ?angleNodeApplicable .
                                ?angleNodeApplicable qudt:numericValue ?angleApplicable .
                                ?angleNodeApplicable qudt:unit unit:DEG .
                                FILTER(?angleApplicable < 90)
                            }
                        )
                        &&
                        !EXISTS {
                            $this nltl:antitrippingSupportSpacing ?spacingNode .
                            ?spacingNode qudt:numericValue ?spacingValue .
                            ?spacingNode qudt:unit unit:MilliM .
                            FILTER(?spacingValue <= 1300)
                        }
                    )
                    ||
                    !EXISTS {
                        $this nltl:frameSpan ?spanNode .
                        ?spanNode qudt:numericValue ?spanValue .
                        ?spanNode qudt:unit unit:M .
                    }
                    ||
                    (
                        EXISTS {
                            $this nltl:frameSpan ?largeSpanNode .
                            ?largeSpanNode qudt:numericValue ?largeSpanValue .
                            ?largeSpanNode qudt:unit unit:M .
                            FILTER(?largeSpanValue > 4)
                        }
                        &&
                        (
                            !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion }
                            ||
                            !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                            ||
                            !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion }
                            ||
                            !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    )
                    ||
                    (
                        EXISTS {
                            $this nltl:frameSpan ?shortSpanNode .
                            ?shortSpanNode qudt:numericValue ?shortSpanValue .
                            ?shortSpanNode qudt:unit unit:M .
                            FILTER(?shortSpanValue <= 4)
                        }
                        &&
                        (
                            !EXISTS {
                                $this nltl:iceClass ?iceClassValue .
                            }
                            ||
                            (
                                EXISTS { $this nltl:iceClass nltl:iceClassIaSuper }
                                &&
                                (
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion }
                                    ||
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                                    ||
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion }
                                    ||
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                                )
                            )
                            ||
                            (
                                EXISTS { $this nltl:iceClass nltl:iceClassIa }
                                &&
                                (
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion }
                                    ||
                                    !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                                )
                            )
                            ||
                            (
                                (
                                    EXISTS { $this nltl:iceClass nltl:iceClassIb }
                                    ||
                                    EXISTS { $this nltl:iceClass nltl:iceClassIc }
                                )
                                &&
                                !EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion }
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
