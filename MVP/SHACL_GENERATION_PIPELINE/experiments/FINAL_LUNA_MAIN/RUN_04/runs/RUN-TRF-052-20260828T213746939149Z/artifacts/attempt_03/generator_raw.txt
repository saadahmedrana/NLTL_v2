<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF052Shape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        gs:TRF052DirectCalculationAlternativeShape
        gs:TRF052PrescriptiveShape
    ) .

gs:TRF052DirectCalculationAlternativeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gs:TRF052PrescriptiveShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:property [
        sh:path nltl:hasAntitrippingSupportRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
            nltl:otherIceStrengthenedAreaValue
        )
    ] ;
    sh:sparql [
        sh:message "The frame does not satisfy the applicable antitripping spacing or support-region requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:frameAsymmetrical ?asym ;
                      nltl:frameWebAngleToShell ?angleNode ;
                      nltl:frameSpan ?spanNode ;
                      nltl:iceClass ?ice .

                ?angleNode qudt:numericValue ?angle .
                ?spanNode qudt:numericValue ?span .

                FILTER (
                    (
                        (?asym = true || ?angle < 90) &&
                        NOT EXISTS {
                            ?this nltl:antitrippingSupportSpacing ?spacingNode .
                            ?spacingNode qudt:numericValue ?spacing ;
                                         qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                            FILTER ( ?spacing <= 1300 )
                        }
                    )
                    ||
                    (
                        ?span > 4 &&
                        (
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:sternRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIaSuper &&
                        (
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:sternRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIa &&
                        (
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:bowRegion } ||
                            NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                        )
                    )
                    ||
                    (
                        ?span <= 4 &&
                        (
                            ?ice = nltl:iceClassIb ||
                            ?ice = nltl:iceClassIc
                        ) &&
                        NOT EXISTS { ?this nltl:hasAntitrippingSupportRegion nltl:bowRegion }
                    )
                )
            }
        """
    ] .

    a sh:PropertyShape ;
    sh:path nltl:antitrippingSupportSpacing ;
    sh:maxCount 1 ;
    sh:node [
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:hasValue <http://qudt.org/vocab/unit/MilliM>
        ]
    ] .
<END_SHACL>
