<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF052Shape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property gs:TRF052SpacingShape ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The direct-calculation alternative status must be recorded as a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:maxCount 1 ;
        sh:node gs:TRF052AngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:maxCount 1 ;
        sh:node gs:TRF052SpanQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:property [
        sh:path nltl:hasAntitrippingSupportRegion ;
        sh:class nltl:hullRegionValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame does not satisfy the applicable antitripping support spacing and region-scope requirement, or the direct-calculation alternative is not valid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:equivalentSupportStatusByDirectCalculation ?alternativeStatus .
                FILTER (
                    ?alternativeStatus = true
                    ||
                    EXISTS {
                        $this nltl:frameAsymmetrical ?asymmetrical .
                        $this nltl:frameWebAngleToShell ?angleQuantity .
                        ?angleQuantity qudt:numericValue ?angle .
                        ?angleQuantity qudt:unit <http://qudt.org/vocab/unit/DEG> .
                        $this nltl:frameSpan ?spanQuantity .
                        ?spanQuantity qudt:numericValue ?span .
                        ?spanQuantity qudt:unit <http://qudt.org/vocab/unit/M> .
                        $this nltl:iceClass ?iceClass .

                        FILTER (
                            ?asymmetrical = true
                            || ?angle < 90
                        )

                        FILTER EXISTS {
                            $this nltl:antitrippingSupportSpacing ?spacingQuantity .
                            ?spacingQuantity qudt:numericValue ?spacing .
                            ?spacingQuantity qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                            FILTER ( ?spacing <= 1300 )
                        }

                        FILTER (
                            (
                                ?span > 4
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                }
                                ||
                                ?span > 4
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                                }
                                ||
                                ?span > 4
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
                                }
                                ||
                                ?span > 4
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
                                }
                            )
                            ||
                            (
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIaSuper
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                }
                                ||
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIaSuper
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                                }
                                ||
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIaSuper
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
                                }
                                ||
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIaSuper
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
                                }
                            )
                            ||
                            (
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIa
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                }
                                ||
                                ?span <= 4
                                &&
                                ?iceClass = nltl:iceClassIa
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                                }
                            )
                            ||
                            (
                                ?span <= 4
                                &&
                                (
                                    ?iceClass = nltl:iceClassIb
                                    ||
                                    ?iceClass = nltl:iceClassIc
                                )
                                &&
                                NOT EXISTS {
                                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                                }
                            )
                        )
                    }
                )
            }
        """
    ] .

gs:TRF052SpacingShape
    a sh:PropertyShape ;
    sh:path nltl:antitrippingSupportSpacing ;
    sh:maxCount 1 ;
    sh:node gs:TRF052SpacingQuantityShape .

gs:TRF052SpacingQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF052SpanQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF052AngleQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
