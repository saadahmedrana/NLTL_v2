gen:TRF055_NarrowDeckStripShape

        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:actualShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:requiredShearArea ;

        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:evidenceStateApproved
            nltl:evidenceStateDraft
            nltl:evidenceStateExpired
            nltl:evidenceStateRejected
            nltl:evidenceStateRevoked
            nltl:evidenceStateSubmitted
            nltl:evidenceStateUnderReview
        )
    ] ;
    sh:or (
        gen:TRF055_NotVeryLongHatchShape
        gen:TRF055_ApprovedVeryLongHatchShape
    ) ;
    sh:sparql [
        sh:message "The actual section modulus and actual shear area must meet their respective required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualZ .
                ?actualZ qudt:numericValue ?actualZValue .
                ?actualZ qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                $this nltl:requiredSectionModulus ?requiredZ .
                ?requiredZ qudt:numericValue ?requiredZValue .
                ?requiredZ qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                $this nltl:actualShearArea ?actualA .
                ?actualA qudt:numericValue ?actualAValue .
                ?actualA qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                $this nltl:requiredShearArea ?requiredA .
                ?requiredA qudt:numericValue ?requiredAValue .
                ?requiredA qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                FILTER (
                    ?actualZValue < ?requiredZValue ||
                    ?actualAValue < ?requiredAValue
                )
            }
        """
    ] .

gen:TRF055_ApprovedVeryLongHatchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:permittedReducedLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_ForcePerLengthQuantityShape ;
        sh:message "A very long hatch requires an approved permitted reduced line load."
    ] ;
    sh:property [
        sh:path nltl:reducedLineLoadApprovedByClassificationSociety ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The reduced line load for a very long hatch must be approved by the classification society."
    ] ;
    sh:sparql [
        sh:message "For a very long hatch, the permitted reduced line load must be at least 0.10 kN/m and less than 0.15 kN/m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:permittedReducedLineLoad ?load .
                ?load qudt:numericValue ?loadValue .
                ?load qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                FILTER (?loadValue < 0.10 || ?loadValue >= 0.15)
            }
        """
    ] .
