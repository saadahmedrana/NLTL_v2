<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF057WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:iceLoadSourceType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:hasValue nltl:iceStringerLoadSource ]
            [ sh:hasValue nltl:longitudinalFramingLoadSource ]
        ) ;
        sh:message "The ice load source must be an ice stringer or longitudinal framing."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadLengthParameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "loadLengthParameter must equal twice webFrameSpacing within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:webFrameSpacing ?spacingNode ;
                      nltl:loadLengthParameter ?lengthNode .
                ?spacingNode qudt:numericValue ?spacing .
                ?lengthNode qudt:numericValue ?length .
                BIND (2 * ?spacing AS ?expected)
                FILTER (
                    ABS(?length - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "webFrameIceLoad must equal 1000 times 1.8 times the capped pressure-height product times webFrameSpacing within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:iceLoadHeight ?heightNode ;
                      nltl:webFrameSpacing ?spacingNode ;
                      nltl:webFrameIceLoad ?loadNode .
                ?pressureNode qudt:numericValue ?pressure .
                ?heightNode qudt:numericValue ?height .
                ?spacingNode qudt:numericValue ?spacing .
                ?loadNode qudt:numericValue ?load .
                BIND (?pressure * ?height AS ?pressureHeight)
                BIND (
                    1000 * 1.8 *
                    IF(?pressureHeight < 0.15, 0.15, ?pressureHeight) *
                    ?spacing
                    AS ?expected
                )
                FILTER (
                    ABS(?load - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
