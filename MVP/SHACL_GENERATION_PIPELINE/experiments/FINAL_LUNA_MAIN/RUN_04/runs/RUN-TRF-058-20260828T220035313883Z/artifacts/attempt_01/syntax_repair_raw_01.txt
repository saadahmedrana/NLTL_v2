<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF058WebFrameShape a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:message "supportedStringerOutsideIceBelt, when present, must be a boolean."
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "iceLoadForce must have one decimal numeric value in kN."
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "stringerOutsideIceBeltHeight must have one decimal numeric value in m."
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "stringerSpan must have one decimal numeric value in m."
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "adjustedIceLoadForce must have one decimal numeric value in kN."
    ] ;
    sh:sparql [
        sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt ?outside ;
                      nltl:iceLoadForce ?iceNode ;
                      nltl:stringerOutsideIceBeltHeight ?heightNode ;
                      nltl:stringerSpan ?spanNode ;
                      nltl:adjustedIceLoadForce ?adjustedNode .

                ?outside a xsd:boolean .
                ?iceNode qudt:numericValue ?ice .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?adjustedNode qudt:numericValue ?adjusted .

                FILTER (
                    ?outside = true &&
                    (
                        ?span = 0 ||
                        ABS(
                            ?adjusted - (?ice * (1 - (?height / ?span)))
                        ) >
                        (
                            ABS(?ice * (1 - (?height / ?span))) * 0.000001 + 0.000000001
                        )
                    )
                )
            }
        """
    ] .
