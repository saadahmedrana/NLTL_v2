gen:TRF058WebFrameShape

    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "supportedStringerOutsideIceBelt must be explicitly recorded as true or false." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:iceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                ] ;
                sh:message "When the supported stringer is outside the ice belt, iceLoadForce must be supplied in kN." ;
            ] ;
            sh:property [
                sh:path nltl:stringerOutsideIceBeltHeight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                ] ;
                sh:message "When the supported stringer is outside the ice belt, stringerOutsideIceBeltHeight must be supplied in m." ;
            ] ;
            sh:property [
                sh:path nltl:stringerSpan ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                ] ;
                sh:message "When the supported stringer is outside the ice belt, stringerSpan must be supplied in m." ;
            ] ;
            sh:property [
                sh:path nltl:adjustedIceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                ] ;
                sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must be supplied in kN." ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:hasValue false ;
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For a supported stringer outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within the specified numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true ;
                      nltl:iceLoadForce ?forceNode ;
                      nltl:stringerOutsideIceBeltHeight ?heightNode ;
                      nltl:stringerSpan ?spanNode ;
                      nltl:adjustedIceLoadForce ?adjustedNode .

                ?forceNode qudt:numericValue ?force ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?heightNode qudt:numericValue ?height ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spanNode qudt:numericValue ?span ;
                           qudt:unit <http://qudt.org/vocab/unit/M> .
                ?adjustedNode qudt:numericValue ?adjusted ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                FILTER (
                    (?span = 0) ||
                    (
                        (?span != 0) &&
                        (
                            (
                                (?force * (1 - (?height / ?span))) >= 0 &&
                                (
                                    (?adjusted - (?force * (1 - (?height / ?span))) >
                                        ((?force * (1 - (?height / ?span))) * 0.000001 + 0.000001))
                                    ||
                                    ((?force * (1 - (?height / ?span))) - ?adjusted >
                                        ((?force * (1 - (?height / ?span))) * 0.000001 + 0.000001))
                                )
                            )
                            ||
                            (
                                (?force * (1 - (?height / ?span))) < 0 &&
                                (
                                    (?adjusted - (?force * (1 - (?height / ?span))) >
                                        ((0 - (?force * (1 - (?height / ?span)))) * 0.000001 + 0.000001))
                                    ||
                                    ((?force * (1 - (?height / ?span))) - ?adjusted >
                                        ((0 - (?force * (1 - (?height / ?span)))) * 0.000001 + 0.000001))
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .
