<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:or (
        gen:TRF058SupportedOutsideShape
        gen:TRF058SupportedInsideShape
    ) .

gen:TRF058SupportedOutsideShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:node gen:TRF058ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:node gen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:node gen:TRF058ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan) within the permitted tolerance, with a nonzero stringerSpan." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true ;
                      nltl:iceLoadForce ?forceNode ;
                      nltl:stringerOutsideIceBeltHeight ?heightNode ;
                      nltl:stringerSpan ?spanNode ;
                      nltl:adjustedIceLoadForce ?adjustedNode .

                ?forceNode qudt:numericValue ?force .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?adjustedNode qudt:numericValue ?adjusted .

                FILTER (?span != 0)
                BIND ((?force * (1 - (?height / ?span))) AS ?expected)

                FILTER (
                    (
                        ?expected > 0 &&
                        (
                            (
                                ?adjusted > ?expected &&
                                (?adjusted - ?expected) > (?expected * 0.000001)
                            ) ||
                            (
                                ?expected >= ?adjusted &&
                                (?expected - ?adjusted) > (?expected * 0.000001)
                            )
                        )
                    ) ||
                    (
                        ?expected <= 0 &&
                        (
                            (
                                ?adjusted > ?expected &&
                                (?adjusted - ?expected) > 0.000001
                            ) ||
                            (
                                ?expected >= ?adjusted &&
                                (?expected - ?adjusted) > 0.000001
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:TRF058SupportedInsideShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

    ] ;

    ] .
