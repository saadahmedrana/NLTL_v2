@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:path nltl:propellerCount ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2
            ]
            [
                sh:path nltl:propellerCount ;
                sh:minInclusive 3 ;
                sh:maxInclusive 3
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape ;
        sh:minCount 1
    ] ;
    sh:message "A twin- or triple-screw ship must provide a side-propeller strengthening envelope with the required tank-top boundary and extents." .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:message "The strengthening envelope boundary must be a tank top."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:node gen:TRF063ForwardExtentQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The strengthening forward extent must be at least 1.5 m."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:node gen:TRF063AftExtentQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The strengthening aft extent must be at least 1.5 m."
    ] .

gen:TRF063ForwardExtentQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF063AftExtentQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
