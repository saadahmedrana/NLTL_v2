<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF064ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasSidePropellerShafting ;
        sh:node gen:TRF064SidePropellerShaftingShape ;
        sh:message "Each side-propeller shafting assembly must be enclosed by a plated bossing."
    ] ;
    sh:property [
        sh:path nltl:hasSternTube ;
        sh:node gen:TRF064SternTubeShape ;
        sh:message "Each side-propeller stern tube must be enclosed by a plated bossing."
    ] ;
    sh:property [
        sh:path nltl:hasDetachedStrut ;
        sh:node gen:TRF064DetachedStrutShape ;
        sh:message "Each detached strut must have design, strength, and hull-attachment evidence."
    ] .

gen:TRF064SidePropellerShaftingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:class nltl:platedBossingStructure ;
        sh:message "The side-propeller shafting must be enclosed by a plated bossing."
    ] .

gen:TRF064SternTubeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:class nltl:platedBossingStructure ;
        sh:message "The stern tube must be enclosed by a plated bossing."
    ] .

gen:TRF064DetachedStrutShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutDesignEvidence ;
        sh:message "Detached-strut design evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutStrengthEvidence ;
        sh:message "Detached-strut strength evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutHullAttachmentEvidence ;
        sh:message "Detached-strut hull-attachment evidence is required."
    ] .
<END_SHACL>
