<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF069Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice class is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single propeller type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single pitch type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The maximum ice torque input must be provided in kN*m."
    ] ;
    sh:or (
        gen:TRF069ApplicableBranch
        gen:TRF069NonApplicableBranch
    ) .

gen:TRF069ApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:in ( "open" "ducted" )
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:in ( "controllable" "fixed" )
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node gen:TRF069DesignConditionShape
    ] .

gen:TRF069DesignConditionShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design-load scope must cover normal operational conditions."
    ] ;
    sh:property [
        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design-load scope must cover normal service-life propeller and ice loads."
    ] ;
    sh:property [
        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Fixed-pitch propeller reversal loads must be included."
    ] ;
    sh:property [
        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stopped-propeller dragging through ice must be recorded as outside the prescribed load model."
    ] ;
    sh:property [
        sh:path nltl:radialIceEntryOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Radial ice entry into a turned azimuthing thruster must be recorded as outside the prescribed load model."
    ] .

gen:TRF069NonApplicableBranch a sh:NodeShape ;
    sh:sparql [
        sh:message "The non-applicable branch must represent at least one propeller, pitch, or ice-class value outside the stated applicability set." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )
            }
        """
    ] .
<END_SHACL>
