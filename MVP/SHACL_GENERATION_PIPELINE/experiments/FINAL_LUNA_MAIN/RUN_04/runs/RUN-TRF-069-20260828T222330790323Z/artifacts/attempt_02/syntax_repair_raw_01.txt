<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one ice class in the applicable set."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify exactly one propeller type."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify exactly one propeller pitch type."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one maximum ice torque quantity in kN*m."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable propulsion-machinery configuration, the linked design condition must record normal operational conditions, normal service-life propeller/ice loads, stopped-propeller dragging outside the load model, and radial ice entry outside the load model." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT ?this
            WHERE {
                ?this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType ;
                      nltl:hasDesignCondition ?designCondition .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )

                FILTER (
                    NOT EXISTS {
                        ?designCondition nltl:normalOperationalConditions true
                    }
                    ||
                    NOT EXISTS {
                        ?designCondition nltl:normalServiceLifePropellerIceLoadCondition true
                    }
                    ||
                    NOT EXISTS {
                        ?designCondition nltl:stoppedPropellerDraggingOutsideLoadModel true
                    }
                    ||
                    NOT EXISTS {
                        ?designCondition nltl:radialIceEntryOutsideLoadModel true
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable fixed-pitch configuration, the linked design condition must record that fixed-pitch propeller reversal loads are included." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT ?this
            WHERE {
                ?this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType "fixed" ;
                      nltl:hasDesignCondition ?designCondition .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                )

                FILTER NOT EXISTS {
                    ?designCondition nltl:fixedPitchPropellerReversalLoadIncluded true
                }
            }
        """
    ] .
<END_SHACL>
