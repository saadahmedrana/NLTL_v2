<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of the propeller submersion rule."
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For ice classes IB and IC, valid ballast propeller-top depth and hi values are required; if the propeller top is above hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        ?depthNode qudt:numericValue ?depthValue .
                        ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?depthValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        ?hiNode qudt:numericValue ?hiValue .
                        ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?hiValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode2 .
                        ?depthNode2 qudt:numericValue ?depthValue2 .
                        ?depthNode2 qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?depthValue2) = xsd:decimal)

                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode2 .
                        ?hiNode2 qudt:numericValue ?hiValue2 .
                        ?hiNode2 qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?hiValue2) = xsd:decimal)

                        FILTER (?depthValue2 < ?hiValue2)

                        FILTER NOT EXISTS {
                            $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
