<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF078QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity value must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use the unitless unit."
    ] .

nltl:TRF078TableLookupCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each table lookup case must have exactly one table reference."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Table 6-6 lookup case must have one permitted ice-class selection key."
    ] ;
    sh:property [
        sh:path nltl:tableLookupPropellerLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:propellerLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Table 6-7 lookup case must have one propeller-location selection key."
    ] ;
    sh:property [
        sh:path nltl:selectedIceClassCycleCount ;
        sh:nodeKind sh:IRI ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Table 6-6 lookup case must provide one selected ice-class cycle count."
    ] ;
    sh:property [
        sh:path nltl:selectedPropellerLocationFactor ;
        sh:nodeKind sh:IRI ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Table 6-7 lookup case must provide one selected propeller-location factor."
    ] .

nltl:TRF078ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:node nltl:TRF078TableLookupCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the applicable table lookup cases."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:nodeKind sh:IRI ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one unitless k2 coefficient."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:nodeKind sh:IRI ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one unitless k3 coefficient."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:nodeKind sh:IRI ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one unitless propeller blade load-spectrum count factor."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one ice load cycle count."
    ] ;
    sh:sparql [
        sh:message "The ship must have a Table 6-6 lookup case referencing traficomTable6Dash6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                    ?case nltl:tableReference nltl:traficomTable6Dash6 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship must have a Table 6-7 lookup case referencing traficomTable6Dash7." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                    ?case nltl:tableReference nltl:traficomTable6Dash7 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 times k2 times k3 times N_class times n_n, within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship ;
                    nltl:hasTableLookupCase ?case6 ;
                    nltl:hasTableLookupCase ?case7 ;
                    nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                    nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                    nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                    nltl:iceLoadCycleCount ?nice .

                ?case6 nltl:tableReference nltl:traficomTable6Dash6 ;
                    nltl:iceClass ?iceClass ;
                    nltl:selectedIceClassCycleCount ?nclassNode .

                ?case7 nltl:tableReference nltl:traficomTable6Dash7 ;
                    nltl:tableLookupPropellerLocation ?location ;
                    nltl:selectedPropellerLocationFactor ?k1Node .

                ?k1Node qudt:numericValue ?k1 .
                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nclassNode qudt:numericValue ?nclass .
                ?nnNode qudt:numericValue ?nn .

                BIND((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (
                    ABS(?nice - ?expected) >
                    IF(
                        ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001
                    )
                )
            }
        """
    ] .
<END_SHACL>
