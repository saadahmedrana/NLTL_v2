<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qudtunit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF083ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:rotationalSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:TRF083RotationalSpeedQuantityShape ;
        sh:message "The ship must provide the propeller rotational speed at maximum continuous rating in bollard condition."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:rotationalSpeedKnown ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:rotationalSpeedKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node nltlgen:TRF083UnknownSpeedLookupCaseShape
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "When the bollard propeller speed is unknown, the Table 6-9 lookup case must attribute its selected result to the ship's required bollard propeller rotational-speed quantity." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX qudtunit: <http://qudt.org/vocab/unit/>

                    SELECT $this
                    WHERE {
                        $this nltl:rotationalSpeedKnown false ;
                              nltl:hasTableLookupCase ?lookupCase ;
                              nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speedQuantity .
                        ?lookupCase nltl:tableLookupApplied true ;
                                    nltl:tableReference nltl:traficomTable6Dash9 .
                        ?speedQuantity qudt:numericValue ?speedValue ;
                                        qudt:unit qudtunit:REV-PER-MIN .
                    }
                """
            ]
        ]
    ) .

nltlgen:TRF083RotationalSpeedQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qudtunit:REV-PER-MIN
    ] .

nltlgen:TRF083UnknownSpeedLookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash9
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF083EvidenceArtifactShape
    ] .

nltlgen:TRF083EvidenceArtifactShape
    a sh:NodeShape ;
    sh:class nltl:evidenceArtifact .
<END_SHACL>
