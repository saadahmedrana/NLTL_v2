<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF104ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:pyramidStrengthPrinciple ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship shall record the pyramid strength principle for the propulsion line design."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:message "The ship shall identify its propeller shaft line component or components."
    ] ;
    sh:sparql [
        sh:message "When propeller blade loss is represented, no linked propeller shaft line component may have significant damage from blade loss." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerBladeLoss true .
                $this nltl:hasPropellerShaftLineComponent ?component .
                FILTER (
                    NOT EXISTS {
                        ?component nltl:significantDamageFromBladeLoss false
                    }
                    ||
                    EXISTS {
                        ?component nltl:significantDamageFromBladeLoss ?damage .
                        FILTER (?damage != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
