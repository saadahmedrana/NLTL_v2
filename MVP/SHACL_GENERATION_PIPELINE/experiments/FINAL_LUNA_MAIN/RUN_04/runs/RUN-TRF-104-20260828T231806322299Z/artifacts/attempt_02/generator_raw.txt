<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-104-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:pyramidStrengthPrinciple ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship shall record application of the pyramid strength principle."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:message "Propeller blade loss shall be represented as an xsd:boolean value."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:node <urn:nltl:generated-shape:TRF-104-PropellerShaftLineComponentShape>
    ] ;
    sh:sparql [
        sh:message "When propeller blade loss is true, every linked propeller shaft line component shall have significantDamageFromBladeLoss explicitly set to false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerBladeLoss true ;
                      nltl:hasPropellerShaftLineComponent ?component .
                FILTER (
                    NOT EXISTS {
                        ?component nltl:significantDamageFromBladeLoss false
                    }
                    ||
                    EXISTS {
                        ?component nltl:significantDamageFromBladeLoss true
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-104-PropellerShaftLineComponentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:significantDamageFromBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:message "Significant damage from propeller blade loss shall be represented as an xsd:boolean value."
    ] .
<END_SHACL>
