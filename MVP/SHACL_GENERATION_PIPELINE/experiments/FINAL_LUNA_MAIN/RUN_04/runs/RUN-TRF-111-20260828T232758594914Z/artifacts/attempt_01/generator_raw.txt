<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qv: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

qv:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

qv:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:node qv:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

qv:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:node qv:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

qv:ShipComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:fatigueSafetyFactor ;
        sh:node qv:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked ship component must provide its externally calculated fatigue safety factor."
    ] .

qv:LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:in (
            nltl:extremeOperationalLoadCase
            nltl:fatigueLoadCase
            nltl:bladeFailureLoadCase
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each propulsion load case must identify an allowed shaft-line load-case type."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:node qv:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each propulsion load case must provide its externally calculated component yield safety factor."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The component yield safety factor must be at least 1.3 for extreme operational loads and at least 1.0 for blade failure loads." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:shaftLineLoadCaseType ?caseType .
                FILTER (
                    (
                        ?caseType = nltl:extremeOperationalLoadCase &&
                        NOT EXISTS {
                            ?this nltl:componentYieldSafetyFactor ?sf .
                            ?sf qudt:numericValue ?value .
                            FILTER (datatype(?value) = xsd:decimal && ?value >= 1.3)
                        }
                    )
                    ||
                    (
                        ?caseType = nltl:bladeFailureLoadCase &&
                        NOT EXISTS {
                            ?this nltl:componentYieldSafetyFactor ?sf .
                            ?sf qudt:numericValue ?value .
                            FILTER (datatype(?value) = xsd:decimal && ?value >= 1.0)
                        }
                    )
                )
            }
        """
    ] .

qv:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:node qv:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the extreme ice force input in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:message "The ship must identify the applicable propeller shaft-line components."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node qv:ShipComponentShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify the ship components for which fatigue safety factors are assessed."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:node qv:LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the applicable propulsion load cases and their externally calculated safety factors."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a fatigue load case is represented, every linked ship component must have a fatigue safety factor of at least 1.5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                FILTER NOT EXISTS {
                    ?this nltl:hasComponent ?component .
                    FILTER NOT EXISTS {
                        ?component nltl:fatigueSafetyFactor ?sf .
                        ?sf qudt:numericValue ?value .
                        FILTER (datatype(?value) = xsd:decimal && ?value >= 1.5)
                    }
                }
            }
        """
    ] .
</BEGIN_SHACL>
