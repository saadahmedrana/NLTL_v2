<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class q:QuantityValue ;
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase
        ] ;
        sh:message "The ship shall provide at least one propulsion load case."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ] ;
        sh:message "The ship shall identify the regulated propeller shaft-line components."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:shipComponent
        ] ;
        sh:message "The ship shall identify the applicable ship-component owners."
    ] ;
    sh:sparql [
        sh:message "When a fatigue load case applies, each linked ship component shall provide a fatigue safety factor of at least 1.5 with a dimensionless unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?fatigueCase .
                ?fatigueCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasComponent ?component .
                ?component a nltl:shipComponent .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor q:numericValue ?value ;
                            q:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= 1.5
                    )
                }
            }
        """
    ] .

gs:TRF111LoadCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:loadCase ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:shaftLineLoadCaseTypeValue ;
                    sh:in (
                        nltl:extremeOperationalLoadCase
                    )
                ]
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:node [
                    sh:class q:QuantityValue ;
                    sh:property [
                        sh:path q:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:minInclusive 1.3
                    ] ;
                    sh:property [
                        sh:path q:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:shaftLineLoadCaseTypeValue ;
                    sh:in (
                        nltl:fatigueLoadCase
                    )
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:shaftLineLoadCaseTypeValue ;
                    sh:in (
                        nltl:bladeFailureLoadCase
                    )
                ]
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:node [
                    sh:class q:QuantityValue ;
                    sh:property [
                        sh:path q:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:minInclusive 1.0
                    ] ;
                    sh:property [
                        sh:path q:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "Each propulsion load case shall identify an allowed load-case type and meet the applicable direct safety-factor check." .
<END_SHACL>
