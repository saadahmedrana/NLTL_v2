<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-112-ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:bladeFailureUltimateLoad ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:combinedAxialBendingTorsionLoad ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:not [
                        sh:property [
                            sh:path nltl:shaftLineLoadCaseType ;
                            sh:hasValue nltl:bladeFailureLoadCase
                        ]
                    ]
                ]
            )
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable total blade failure load case, bending and torsional yield safety factors of the linked shaft-line components must be at least 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                FILTER (
                    NOT EXISTS {
                        ?component nltl:bendingYieldSafetyFactor ?bending .
                        ?bending qudt:numericValue ?bendingValue .
                        ?bending qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?bendingValue) = xsd:decimal &&
                            ?bendingValue >= 1
                        )
                    }
                    ||
                    NOT EXISTS {
                        ?component nltl:torsionalYieldSafetyFactor ?torsional .
                        ?torsional qudt:numericValue ?torsionalValue .
                        ?torsional qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?torsionalValue) = xsd:decimal &&
                            ?torsionalValue >= 1
                        )
                    }
                )
            }
        """
    ]
    .
<END_SHACL>
