<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship must provide damping for the thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloGM
            ]
        ] ;
        sh:message "The ship must provide water added mass in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:HZ
            ]
        ] ;
        sh:message "The ship must provide the thruster global natural frequency in hertz."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked thruster body must provide ship attachment stiffness and calculation cases for both longitudinal and transverse vibration directions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?thruster .
                ?thruster rdf:type nltl:thrusterBody .
                FILTER (
                    NOT EXISTS {
                        ?thruster nltl:shipAttachmentStiffness ?stiffness .
                        ?stiffness rdf:type qudt:QuantityValue .
                        ?stiffness qudt:numericValue ?stiffnessValue .
                        ?stiffness qudt:unit unit:KiloN-PER-M .
                        FILTER (datatype(?stiffnessValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?thruster nltl:hasNaturalFrequencyCalculationCase ?longitudinalCase .
                        ?longitudinalCase rdf:type nltl:calculationCase .
                        ?longitudinalCase nltl:vibrationDirection nltl:longitudinalDirection .
                    }
                    ||
                    NOT EXISTS {
                        ?thruster nltl:hasNaturalFrequencyCalculationCase ?transverseCase .
                        ?transverseCase rdf:type nltl:calculationCase .
                        ?transverseCase nltl:vibrationDirection nltl:transverseDirection .
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
