<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF123ComplexReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The occasional-force load case must identify that propeller blades are excluded from its scope."
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:class nltl:shipComponent ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:occasionalForceComponentStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:hasMaterialProperties ;
                        sh:class nltl:materialProperties ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path nltl:componentMaterialYieldStrength ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:class qudt:QuantityValue ;
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The externally calculated occasional-force component stress, including the declared safety factor, must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasOccasionalForceLoadCase ?case .
                ?case nltl:propellerBladeExcludedFromOccasionalForceScope true .
                ?case nltl:componentYieldSafetyFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factor ;
                                 qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?case nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (?stress * ?factor > ?yield)
            }
        """
    ] .
<END_SHACL>
