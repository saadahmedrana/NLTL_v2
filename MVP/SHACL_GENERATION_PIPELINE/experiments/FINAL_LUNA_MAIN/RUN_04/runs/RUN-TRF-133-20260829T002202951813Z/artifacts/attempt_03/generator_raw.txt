<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice class."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:node nltl:TRF133IceClassDraughtMarkingShape ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one ice-class draught marking."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:node nltl:TRF133WarningTriangleMarkingShape ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one warning triangle marking."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Timber-load-line applicability must be explicitly recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:timberLoadLineVerticalReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:loadLineRingCentreReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ) .

nltl:TRF133IceClassDraughtMarkingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:draughtMarkAftOffset ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "540"^^xsd:decimal ;
                sh:maxInclusive "540"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-class draught mark must be 540 mm abaft of the applicable reference."
    ] ;
    sh:property [
        sh:path nltl:draughtMarkAftReferencePoint ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:loadLineRingCentreReference
            nltl:timberLoadLineVerticalReference
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:letterDimensionsEqualLoadLineMark ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-class draught-mark letter dimensions must equal those of the load-line mark."
    ] .

nltl:TRF133WarningTriangleMarkingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:warningTriangleUpperEdgeVerticallyAboveIceMark ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The warning triangle upper edge must be vertically above the ICE mark."
    ] ;
    sh:property [
        sh:path nltl:verticalOffsetAboveSummerFreshWaterLoadLine ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "1000"^^xsd:decimal ;
                sh:maxInclusive "1000"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The warning triangle upper edge must be 1000 mm above the summer fresh-water load line."
    ] ;
    sh:property [
        sh:path nltl:markingAtOrBelowDeckLine ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The warning triangle must not be higher than the deck line."
    ] ;
    sh:property [
        sh:path nltl:warningTriangleSideLength ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "300"^^xsd:decimal ;
                sh:maxInclusive "300"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The warning triangle sides must be 300 mm long."
    ] .
<END_SHACL>
