<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_001_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a construction contract date is supplied, UR I2 Rev.4 applicability must be true on or after 2021-01-01 and false before that date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    DATATYPE(?date) = xsd:date
                    &&
                    (
                        (
                            (
                                YEAR(?date) > 2021
                                ||
                                (YEAR(?date) = 2021 && MONTH(?date) > 1)
                                ||
                                (YEAR(?date) = 2021 && MONTH(?date) = 1 && DAY(?date) >= 1)
                            )
                            &&
                            (
                                NOT EXISTS {
                                    $this nltl:urI2RevisionApplicability ?applicable .
                                    FILTER (?applicable = true)
                                }
                            )
                        )
                        ||
                        (
                            (
                                YEAR(?date) < 2021
                                ||
                                (YEAR(?date) = 2021 && MONTH(?date) < 1)
                                ||
                                (YEAR(?date) = 2021 && MONTH(?date) = 1 && DAY(?date) < 1)
                            )
                            &&
                            (
                                NOT EXISTS {
                                    $this nltl:urI2RevisionApplicability ?applicable .
                                    FILTER (?applicable = false)
                                }
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
