gs:I2_009_ApplicableBowFormShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:node gs:I2_009_AngleShape
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:node gs:I2_009_AngleShape
    ] ;
    sh:sparql [
        sh:message "For the applicable bow-form branch, buttock angle gamma shall be greater than 0 degrees and less than 80 degrees, and normal frame angle beta shall be greater than 10 degrees." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  FILTER NOT EXISTS {
    $this nltl:buttockAngle ?gammaQuantity .
    ?gammaQuantity qudt:numericValue ?gamma .
    $this nltl:normalFrameAngle ?betaQuantity .
    ?betaQuantity qudt:numericValue ?beta .
    FILTER (?gamma > 0 && ?gamma < 80 && ?beta > 10)
  }
}
"""
    ] .

gs:I2_009_NonApplicableBowFormShape
    a sh:NodeShape ;
    sh:property [

        sh:node gs:I2_009_AngleShape
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:node gs:I2_009_AngleShape
    ] ;
    sh:sparql [
        sh:message "For the non-applicable bow-form branch, the supplied angle evidence shall show gamma at or below 0 degrees, gamma at or above 80 degrees, or beta at or below 10 degrees." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  FILTER NOT EXISTS {
    $this nltl:buttockAngle ?gammaQuantity .
    ?gammaQuantity qudt:numericValue ?gamma .
    $this nltl:normalFrameAngle ?betaQuantity .
    ?betaQuantity qudt:numericValue ?beta .
    FILTER (?gamma <= 0 || ?gamma >= 80 || ?beta <= 10)
  }
}
"""
    ] .
