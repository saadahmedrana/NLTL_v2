<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:I2_010_NonApplicableShape
        gen:I2_010_ApplicableShape
    ) .

gen:I2_010_NonApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
        ) ;
        sh:message "A non-applicable ship must have a Polar Class other than PC6 or PC7."
    ] .

gen:I2_010_ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:in (
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:message "The applicable ship must be assigned Polar Class PC6 or PC7."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An applicable ship must record the design ice force method."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:verticalSidedBowForm ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:bowVerticalSideStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:normalFrameAngle ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:minInclusive 0 ;
                        sh:maxInclusive 10 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:DEG ;
                    ]
                ] ;
                sh:message "Every supplied considered-subregion normal frame angle must be between 0 and 10 degrees inclusive and use unit DEG."
            ]
        ]
    ) .

</END_SHACL>
