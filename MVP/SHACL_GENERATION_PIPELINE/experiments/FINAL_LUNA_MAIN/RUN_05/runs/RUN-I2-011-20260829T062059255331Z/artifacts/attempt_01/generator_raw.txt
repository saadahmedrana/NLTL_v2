<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_BowDesignForceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
                sh:minCount 1 ;
                sh:message "A Polar Class of PC6 or PC7 is required to establish applicability."
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:message "Bulbous-bow status is required and must be true for the applicable case."
            ] ;
            sh:property [
                sh:path nltl:shapeCoefficient ;
                sh:minCount 1 ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:datatype xsd:decimal ;
                    sh:minInclusive "0.6"^^xsd:decimal ;
                    sh:maxInclusive "0.6"^^xsd:decimal
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:hasValue unit:UNITLESS ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:message "The applicable shape coefficient must be 0.6 and dimensionless."
            ] ;
            sh:property [
                sh:path nltl:loadPatchAspectRatio ;
                sh:minCount 1 ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:datatype xsd:decimal ;
                    sh:minInclusive "1.3"^^xsd:decimal ;
                    sh:maxInclusive "1.3"^^xsd:decimal
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:hasValue unit:UNITLESS ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:message "The applicable load-patch aspect ratio must be 1.3 and dimensionless."
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
                sh:minCount 1 ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:datatype xsd:decimal
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:hasValue unit:MegaN ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:message "The force from I2.3.2.1(iii) must be supplied in MN."
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
                sh:minCount 1 ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:datatype xsd:decimal
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:hasValue unit:MegaN ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:message "The force from I2.3.2.1(iv) must be supplied in MN."
            ] ;
            sh:property [
                sh:path nltl:bowDesignForce ;
                sh:minCount 1 ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:datatype xsd:decimal
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:hasValue unit:KiloN ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:message "The bow design force must be supplied in kN."
            ] ;
            sh:sparql [
                sh:message "For an applicable PC6 or PC7 bulbous-bow ship, the bow design force must equal the I2.3.2.1(iv) force within a relative tolerance of 1e-6 with a 1e-6 kN absolute fallback and must not be less than the I2.3.2.1(iii) force." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iii ;
                              nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?iv ;
                              nltl:bowDesignForce ?result .
                        ?iii qudt:numericValue ?iiiValue ;
                             qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                        ?iv qudt:numericValue ?ivValue ;
                            qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                        ?result qudt:numericValue ?resultValue ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (
                            ?resultValue < (1000 * ?iiiValue)
                            ||
                            ABS(?resultValue - (1000 * ?ivValue))
                                > ((0.000001 * ABS(1000 * ?ivValue)) + 0.000001)
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                ) ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ]
        ]
    ] .
<END_SHACL>
