<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_BowDesignForceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:I2_011_ApplicableShape
        gen:I2_011_NonApplicablePolarShape
        gen:I2_011_NonApplicableBulbousBowShape
    ) ;
    sh:sparql [
        sh:message "For an applicable PC6 or PC7 ship with a bulbous bow, bowDesignForce must equal the I2.3.2.1(iv) force within a bounded tolerance and must not be less than the I2.3.2.1(iii) force; MN operands are converted to kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:bulbousBowPresent ?bulbous ;
                      nltl:bowDesignForce ?reportedNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?minimumNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?expectedNode .

                ?reportedNode qudt:numericValue ?reported .
                ?minimumNode qudt:numericValue ?minimumMn .
                ?expectedNode qudt:numericValue ?expectedMn .

                FILTER (
                    ?bulbous = true &&
                    (?polarClass = nltl:polarClassPc6 ||
                     ?polarClass = nltl:polarClassPc7)
                )

                BIND(?minimumMn * 1000 AS ?minimumKn)
                BIND(?expectedMn * 1000 AS ?expectedKn)
                BIND(
                    IF(?expectedKn >= 0.001,
                       ?expectedKn * 0.000001,
                       0.001)
                    AS ?toleranceKn
                )

                FILTER (
                    ?reported < ?minimumKn ||
                    ?reported < (?expectedKn - ?toleranceKn) ||
                    ?reported > (?expectedKn + ?toleranceKn)
                )
            }
        """
    ] .

gen:I2_011_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarClassPc6
            nltl:polarClassPc7
        )
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_MegaNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_MegaNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_011_KiloNewtonQuantityShape
    ] .
