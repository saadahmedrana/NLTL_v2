gen:I2_014_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide its bow length in metres."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "4"^^xsd:integer ;
        sh:maxInclusive "4"^^xsd:integer ;
        sh:message "The bow region must be divided into four subregions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:node gen:I2_014_BowSubregionCaseShape ;
        sh:message "The ship must provide exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;

                sh:maxCount 1

                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1

        ] ;
        sh:message "The selected maximum bow force must be provided in MN."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN-M-PER-M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The selected maximum bow line load must be provided in MN/m."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The selected maximum bow pressure must be provided in MPa."
    ] ;
    sh:sparql [
        sh:message "The selected maximum bow force must equal the maximum force reported for the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedNode .
                ?selectedNode qudt:numericValue ?selectedValue .
                FILTER (
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?greaterCase .
                        ?greaterCase nltl:bowSubregionCalculatedForce ?greaterNode .
                        ?greaterNode qudt:numericValue ?greaterValue .
                        FILTER (?greaterValue > ?selectedValue)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?equalCase .
                        ?equalCase nltl:bowSubregionCalculatedForce ?equalNode .
                        ?equalNode qudt:numericValue ?equalValue .
                        FILTER (?equalValue = ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected maximum bow line load must equal the maximum line load reported for the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

                        ?equalNode qudt:numericValue ?equalValue .
                        FILTER (?equalValue = ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected maximum bow pressure must equal the maximum pressure reported for the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
