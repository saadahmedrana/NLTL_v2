<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_019_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_019_StructuralMemberShape
    ] .

gs:I2_019_StructuralMemberShape a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 1 ;
                sh:node gs:I2_019_HullAreaValueShape
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:I2_019_SelectedFactorShape
            ] ;
            sh:sparql [
                sh:message "selectedHullAreaFactor must equal the maximum hullAreaFactor over all spanned hull areas within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:selectedHullAreaFactor ?selectedNode .
                        ?selectedNode qudt:numericValue ?selectedValue .
                        {
                            SELECT ?member (MAX(?factorValue) AS ?expectedValue)
                            WHERE {
                                ?member nltl:hasSpannedHullArea ?area .
                                ?area nltl:hullAreaFactor ?factorNode .
                                ?factorNode qudt:numericValue ?factorValue .
                            }
                            GROUP BY ?member
                        }
                        FILTER (
                            ABS(?selectedValue - ?expectedValue) >
                            (0.000001 * ABS(?expectedValue) + 0.000001)
                        )
                        FILTER (?member = $this)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:maxCount 0
            ]
        ]
    ] .

gs:I2_019_HullAreaValueShape a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:I2_019_HullAreaFactorShape
    ] .

gs:I2_019_HullAreaFactorShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:I2_019_SelectedFactorShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
