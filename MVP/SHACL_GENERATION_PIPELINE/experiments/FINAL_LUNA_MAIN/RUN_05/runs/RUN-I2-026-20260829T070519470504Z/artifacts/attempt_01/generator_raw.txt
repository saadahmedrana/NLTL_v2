<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:I2_026_MemberSupportShape
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:minCount 1 ;
        sh:message "A framing member must specify fixed or simple support." 
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Fixed support requires continuity through the support, a connection bracket, or demonstrated significant rotational restraint." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                FILTER NOT EXISTS {
                    $this nltl:continuousThroughSupport true
                }
                FILTER NOT EXISTS {
                    $this nltl:hasMemberSupport ?supportBracket .
                    ?supportBracket nltl:connectionBracketPresent true
                }
                FILTER NOT EXISTS {
                    $this nltl:hasMemberSupport ?supportRestraint .
                    ?supportRestraint nltl:demonstratedRotationalRestraint true
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing terminating within an ice-strengthened area must have fixed support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:terminatesWithinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:frameSupportCondition nltl:fixedFrameSupport
                }
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Demonstrated significant rotational restraint requires supporting evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence
                }
            }
        """
    ] .
<END_SHACL>
