<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node nltl:I2_026_StructuralMemberShape
    ] .

nltl:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node nltl:I2_026_MemberSupportBaseShape ;
        sh:message "Each structural member must identify a support and its support condition."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:I2_026_FixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:I2_026_BracketFixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:I2_026_RestraintFixedSupportShape
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact ;
                sh:message "Demonstrated significant rotational restraint requires structural-member evidence."
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:I2_026_FixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:I2_026_SimpleSupportShape
            ]
        ]
    ) ;
    sh:message "Support fixity must be fixed for continuity, a connection bracket, demonstrated rotational restraint, or termination within an ice-strengthened area; otherwise it must be simple support." .

nltl:I2_026_MemberSupportBaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:frameSupportConditionValue
    ] .

nltl:I2_026_FixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] .

nltl:I2_026_BracketFixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] .

nltl:I2_026_RestraintFixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] .

nltl:I2_026_SimpleSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:simpleFrameSupport
    ] .
<END_SHACL>
