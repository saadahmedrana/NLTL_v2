<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:fixedFrameSupport
            ]
            [
                sh:hasValue nltl:simpleFrameSupport
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:I2_026_MemberSupportShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The support condition must be fixed when continuity, a connection bracket, demonstrated restraint, or termination within the ice-strengthened area requires fixity; otherwise it must be simple support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support ;
                      nltl:continuousThroughSupport ?continuous ;
                      nltl:frameSupportCondition ?condition ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates .
                ?support nltl:connectionBracketPresent ?bracket ;
                         nltl:demonstratedRotationalRestraint ?restraint .
                FILTER (
                    (
                        (?continuous = true ||
                         ?bracket = true ||
                         ?restraint = true ||
                         ?terminates = true)
                        && ?condition != nltl:fixedFrameSupport
                    )
                    ||
                    (
                        ?continuous = false &&
                        ?bracket = false &&
                        ?restraint = false &&
                        ?terminates = false &&
                        ?condition != nltl:simpleFrameSupport
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Demonstrated significant rotational restraint requires evidence on the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:class nltl:memberSupport ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_026_EvidenceShape
    a sh:NodeShape ;
    sh:class nltl:evidenceArtifact .
<END_SHACL>
