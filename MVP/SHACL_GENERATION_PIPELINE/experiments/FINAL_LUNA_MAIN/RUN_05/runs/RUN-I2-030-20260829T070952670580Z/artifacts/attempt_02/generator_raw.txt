<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-030-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:interpolationLowerEndpoint ;
                sh:node [
                    sh:class nltl:interpolationPoint
                ]
            ] ;
            sh:property [
                sh:path nltl:interpolationUpperEndpoint ;
                sh:node [
                    sh:class nltl:interpolationPoint
                ]
            ] ;
            sh:property [
                sh:path nltl:plasticNeutralAxisHeight ;
                sh:node <urn:nltl:generated-shape:I2-030-LengthQuantity>
            ] ;
            sh:property [
                sh:path nltl:sectionCalculationCaseStructuralMember ;
                sh:class nltl:structuralMember ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The plastic neutral axis height is required when 100 times the local-frame flange area exceeds 1000 times the net attached shell plate thickness times frame spacing." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:sectionCalculationCaseStructuralMember ?member .
                        ?member nltl:netLocalFrameFlangeArea ?area ;
                                nltl:netAttachedShellPlateThickness ?thickness ;
                                nltl:frameSpacing ?spacing .
                        ?area qudt:numericValue ?areaValue .
                        ?thickness qudt:numericValue ?thicknessValue .
                        ?spacing qudt:numericValue ?spacingValue .
                        FILTER (100 * ?areaValue > 1000 * ?thicknessValue * ?spacingValue)
                        FILTER NOT EXISTS {
                            $this nltl:plasticNeutralAxisHeight ?neutralAxis .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "For framing angles strictly between 20 and 70 degrees, both interpolation endpoints are required." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:sectionCalculationCaseStructuralMember ?member .
                        ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                        ?localCase nltl:sectionCalculationCasePlating ?plating .
                        ?plating nltl:framingAngleOmega ?angle .
                        ?angle qudt:numericValue ?angleValue .
                        FILTER (?angleValue > 20 && ?angleValue < 70)
                        FILTER NOT EXISTS {
                            $this nltl:interpolationLowerEndpoint ?lower ;
                                  nltl:interpolationUpperEndpoint ?upper .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Interpolation endpoints are applicable only for framing angles strictly between 20 and 70 degrees." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:sectionCalculationCaseStructuralMember ?member .
                        ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                        ?localCase nltl:sectionCalculationCasePlating ?plating .
                        ?plating nltl:framingAngleOmega ?angle .
                        ?angle qudt:numericValue ?angleValue .
                        FILTER (?angleValue <= 20 || ?angleValue >= 70)
                        {
                            $this nltl:interpolationLowerEndpoint ?endpoint .
                        }
                        UNION
                        {
                            $this nltl:interpolationUpperEndpoint ?endpoint .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The calculation case structural member must be the structural member that links to the local-frame section calculation case." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT $this
                    WHERE {
                        $this nltl:sectionCalculationCaseStructuralMember ?member .
                        FILTER NOT EXISTS {
                            ?member nltl:hasLocalFrameSectionCalculationCase $this .
                        }
                    }
                """
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-030-StructuralMemberShape>
    ] .
