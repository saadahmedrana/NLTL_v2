<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_031_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or [
            rdf:first [ sh:hasValue nltl:flatBarSection ] ;
            rdf:rest [
                rdf:first [ sh:hasValue nltl:profileSection ] ;
                rdf:rest rdf:nil
            ]
        ] ;
        sh:message "The ship must identify an applicable frame profile type."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ] ;
        sh:message "The required patch load must be represented as a kilonewton quantity."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The plastic collapse mechanism must be recorded."
    ] .

gs:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:or [
            rdf:first [ sh:hasValue nltl:bottomStructureLocation ] ;
            rdf:rest [
                rdf:first [ sh:hasValue nltl:sideStructureLocation ] ;
                rdf:rest rdf:nil
            ]
        ] ;
        sh:message "Each applicable structural member must be identified as being in a bottom or side structure."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ] ;
        sh:minCount 1 ;
        sh:message "Each applicable structural member must provide its plastic strength in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ] ;
        sh:minCount 1 ;
        sh:message "The midspan load defining plastic collapse must be provided in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node gs:I2_031_LoadCaseShape ;
        sh:message "Each applicable structural member must have a structural-member load case."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:plasticStrength ?strength .
                ?strength qudt:numericValue ?strengthValue .
                FILTER (
                    xsd:decimal(?demandValue) >
                    xsd:decimal(?strengthValue)
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a bottom structural member, the patch-load dimension b must be parallel with the frame direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:structuralLocation nltl:bottomStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:hasStructuralMemberLoadCase ?loadCase .
                    ?loadCase nltl:loadPatchOrientation
                              nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] .

gs:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:message "Each structural-member load case must provide the combined shear and bending demand."
    ] .
<END_SHACL>
