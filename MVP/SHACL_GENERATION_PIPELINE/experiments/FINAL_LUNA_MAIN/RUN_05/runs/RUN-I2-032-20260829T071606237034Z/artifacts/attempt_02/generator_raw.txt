        sh:message "The actual shear area must satisfy the calculated required shear area, and the reported required transverse frame shear area must match the formula within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:averageIcePressure/qudt:numericValue ?pavg ;
                      nltl:frameSpan/qudt:numericValue ?frameSpan ;
                      nltl:loadPatchHeight/qudt:numericValue ?patchHeight ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:shearArea/qudt:numericValue ?actualM2 ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength .

                ?member nltl:frameSpacing/qudt:numericValue ?spacing ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredTransverseFrameShearArea/qudt:numericValue ?reportedAt .

                BIND(
                    10000 * 0.5 *
                    IF(?frameSpan < ?patchHeight, ?frameSpan, ?patchHeight) *
                    ?spacing * (?af * ?ppf * ?pavg) /
                    (0.577 * ?yieldStrength)
                    AS ?expectedAt
                )
                BIND(?expectedAt * 0.000001 + 0.000001 AS ?tolerance)
                BIND(?actualM2 * 10000 AS ?actualCm2)

                FILTER(
                    ?actualCm2 < ?expectedAt ||
                    ?reportedAt < (?expectedAt - ?tolerance) ||
                    ?reportedAt > (?expectedAt + ?tolerance)
                )
            }
        """
    ] .
