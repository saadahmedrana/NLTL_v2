            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this ;
                      nltl:averageIcePressure ?pressureQ ;
                      nltl:frameSpan ?frameSpanQ ;
                      nltl:loadPatchHeight ?heightQ ;
                      nltl:peakPressureFactor ?peakFactorQ ;
                      nltl:shearArea ?shearQ ;
                      nltl:span ?spanQ ;
                      nltl:yieldStrength ?yieldQ .

                $this nltl:frameSpacing ?spacingQ ;
                      nltl:selectedHullAreaFactor ?areaFactorQ ;
                      nltl:requiredTransverseFrameShearArea ?requiredQ .

                ?pressureQ qudt:numericValue ?pressure .
                ?frameSpanQ qudt:numericValue ?frameSpan .
                ?heightQ qudt:numericValue ?height .
                ?peakFactorQ qudt:numericValue ?peakFactor .
                ?shearQ qudt:numericValue ?shear .
                ?spanQ qudt:numericValue ?span .
                ?yieldQ qudt:numericValue ?yield .
                ?spacingQ qudt:numericValue ?spacing .
                ?areaFactorQ qudt:numericValue ?areaFactor .
                ?requiredQ qudt:numericValue ?required .

                BIND(IF(?frameSpan < ?height, ?frameSpan, ?height) AS ?ll)
                BIND((10000 * 0.5 * ?ll * ?spacing *
                      (?areaFactor * ?peakFactor * ?pressure) /
                      (0.577 * ?yield)) AS ?expected)

                FILTER(
                    (?shear * 10000 < ?expected) ||
                    (ABS(?required - ?expected) >
                     IF(ABS(?expected) < 0.000000001,
                        0.000000001,
                        ABS(?expected) * 0.000001)) ||
                    (ABS(?span - ?frameSpan) >
                     IF(ABS(?frameSpan) < 0.000000001,
                        0.000000001,
                        ABS(?frameSpan) * 0.000001))
                )
            }
        """
