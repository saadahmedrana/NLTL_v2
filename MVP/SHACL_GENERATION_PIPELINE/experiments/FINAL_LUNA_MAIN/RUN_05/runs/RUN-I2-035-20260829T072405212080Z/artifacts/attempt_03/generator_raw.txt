    sh:property [

        sh:maxCount 1 ;

            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;

                sh:maxCount 1

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "The actual shear area must be at least the formula-derived required area, and each reported required area must match the formula within a relative tolerance of 1e-6 with a 1e-9 cm2 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pNode ;
                      nltl:frameSpan ?spanNode ;
                      nltl:peakPressureFactor ?ppfNode ;
                      nltl:shearArea ?awNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:hasStructuralMember ?member .

                ?pNode qudt:numericValue ?p .
                ?spanNode qudt:numericValue ?span .
                ?ppfNode qudt:numericValue ?ppf .
                ?awNode qudt:numericValue ?aw .
                ?yieldNode qudt:numericValue ?yield .

                ?member nltl:frameSpacing ?spacingNode ;
                        nltl:selectedHullAreaFactor ?afNode ;
                        nltl:requiredLongitudinalFrameShearArea ?reportedNode .

                ?spacingNode qudt:numericValue ?spacing .
                ?afNode qudt:numericValue ?af .
                ?reportedNode qudt:numericValue ?reportedAl .

                BIND (?span / ?spacing AS ?bPrime)
                BIND (1 - (0.3 / ?bPrime) AS ?ko)
                BIND (
                    IF (?bPrime < 2,
                        ?span * (1 - (0.25 * ?bPrime)),
                        ?spacing
                    ) AS ?b2
                )
                BIND (?ko * ?b2 AS ?b1)
                BIND (100 * 100 * (?af * ?ppf * ?p) * 0.5 * ?b1 * ?span / (0.577 * ?yield) AS ?expectedAlCm2)
                BIND (?expectedAlCm2 / 10000 AS ?expectedAlM2)
                BIND ((?expectedAlCm2 * 0.000001) + 0.000000001 AS ?tolerance)

                FILTER (
                    (?reportedAl - ?expectedAlCm2 > ?tolerance) ||
                    (?expectedAlCm2 - ?reportedAl > ?tolerance) ||
                    (?aw < ?expectedAlM2)
                )
            }
        """
    ] .
