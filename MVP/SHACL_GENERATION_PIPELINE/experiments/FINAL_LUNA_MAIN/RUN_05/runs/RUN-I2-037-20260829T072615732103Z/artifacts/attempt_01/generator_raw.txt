<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_037_StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify structural members covered by the ice-load design requirement."
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:node gen:I2_037_IceLoadPatchDesignCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice-load-patch design case."
    ] .

gen:I2_037_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_037_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "Each structural member must have a load case for the ice-load assessment."
    ] ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural member must provide its capacity as a single kN quantity."
    ] .

gen:I2_037_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must identify the combined bending-and-shear effect."
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must identify the load-patch application location."
    ] .

gen:I2_037_IceLoadPatchDesignCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design case must confirm that the load patch is applied where combined member capacity is minimised."
    ] .
<END_SHACL>
