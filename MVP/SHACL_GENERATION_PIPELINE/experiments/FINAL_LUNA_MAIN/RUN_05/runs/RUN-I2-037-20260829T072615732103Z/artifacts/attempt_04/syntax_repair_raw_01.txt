<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_037_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceLoadPatch
        ]
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceLoadPatchDesignCase ;
            sh:property [
                sh:path nltl:memberCapacityMinimizationConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked web frame or load-carrying stringer must have a load case, combined bending-and-shear effect, application location, kN member capacity, and linked ice-patch design confirmation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                {
                    ?member rdf:type nltl:webFrame .
                }
                UNION
                {
                    ?member rdf:type nltl:loadCarryingStringer .
                }
                FILTER NOT EXISTS {
                    ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                    ?loadCase rdf:type nltl:loadCase ;
                              nltl:combinedShearBendingEffect ?effect ;
                              nltl:loadPatchApplicationLocation ?location .
                    FILTER NOT EXISTS {
                        ?loadCase nltl:combinedShearBendingEffect ?badEffect .
                        FILTER (datatype(?badEffect) != xsd:string)
                    }
                    FILTER NOT EXISTS {
                        ?loadCase nltl:loadPatchApplicationLocation ?badLocation .
                        FILTER (datatype(?badLocation) != xsd:string)
                    }
                    ?member nltl:memberCapacity ?capacity .
                    ?capacity rdf:type qudt:QuantityValue ;
                              qudt:numericValue ?capacityValue ;
                              qudt:unit unit:KiloN .
                    FILTER (datatype(?capacityValue) = xsd:decimal)
                    FILTER NOT EXISTS {
                        ?capacity qudt:unit ?badUnit .
                        FILTER (?badUnit != unit:KiloN)
                    }
                    FILTER NOT EXISTS {
                        ?capacity qudt:numericValue ?badValue .
                        FILTER (datatype(?badValue) != xsd:decimal)
                    }
                    $this nltl:hasIceLoadPatch ?patch .
                    ?patch rdf:type nltl:iceLoadPatch .
                    $this nltl:hasIceLoadPatchDesignCase ?designCase .
                    ?designCase rdf:type nltl:iceLoadPatchDesignCase ;
                                nltl:memberCapacityMinimizationConfirmed true .
                }
            }
        """
    ] .
</END_SHACL>
