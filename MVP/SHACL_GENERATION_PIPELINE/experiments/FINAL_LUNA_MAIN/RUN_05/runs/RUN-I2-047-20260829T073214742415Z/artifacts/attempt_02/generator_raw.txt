<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:I2-047#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SteelRenewalForIceStrengthenedStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasComponent nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:gaugedThickness ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:netThickness ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:steelRenewalRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:hasComponent nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For an ice strengthened structure, steelRenewalRequired must reflect whether gauged thickness is less than net thickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?structure ;
                      nltl:gaugedThickness ?gaugedNode ;
                      nltl:netThickness ?netNode ;
                      nltl:steelRenewalRequired ?renewal .

                ?structure nltl:iceStrengthenedStructureStatus true .
                ?gaugedNode <http://qudt.org/schema/qudt/numericValue> ?gauged .
                ?netNode <http://qudt.org/schema/qudt/numericValue> ?net .

                FILTER (
                    (?renewal = true && ! (?gauged < (?net + 0.0005))) ||
                    (?renewal = false && (?gauged < (?net + 0.0005)))
                )
            }
        """
    ] .
<END_SHACL>
