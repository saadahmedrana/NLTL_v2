<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_047_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:or (
        gen:I2_047_NotIceStrengthenedShape
        gen:I2_047_IceStrengthenedShape
    ) .

gen:I2_047_NotIceStrengthenedShape
    a sh:NodeShape ;
    sh:property [
        sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
        sh:hasValue false ;
    ] .

gen:I2_047_IceStrengthenedShape
    a sh:NodeShape ;
    sh:property [
        sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_047_GaugedThicknessQuantityShape ;
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_047_NetThicknessQuantityShape ;
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        sh:message "For an ice strengthened structure, steel renewal shall be true exactly when gauged thickness is less than net thickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating/nltl:iceStrengthenedStructureStatus true ;
                       nltl:gaugedThickness ?gaugedQuantity ;
                       nltl:netThickness ?netQuantity ;
                       nltl:steelRenewalRequired ?renewal .

                ?gaugedQuantity qudt:numericValue ?gauged ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?netQuantity qudt:numericValue ?net ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    (?gauged < (?net + "0.0005"^^xsd:decimal) && ?renewal != true)
                    ||
                    (?gauged >= (?net + "0.0005"^^xsd:decimal) && ?renewal != false)
                )
            }
        """
    ] .

gen:I2_047_GaugedThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gen:I2_047_NetThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
