<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one permitted Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade case must identify exactly one assessed plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must provide exactly one permitted actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade case must provide one as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must provide exactly one permitted material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must provide exactly one permitted steel strength category."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual grade must satisfy the Table 8 required grade for the case selectors and must be ranked within the same strength category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required ;
                      nltl:asBuiltPlateThickness ?thickness ;
                      nltl:steelMaterialClass ?material ;
                      nltl:steelStrengthCategory ?strength .
                ?thickness qudt:numericValue ?t .
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?polar .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .

                BIND(
                    IF(?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3), "PC1-3",
                    IF(?polar IN (nltl:polarClassPc4, nltl:polarClassPc5), "PC4-5",
                    IF(?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), "PC1-5", "PC6-7")))
                    AS ?pcGroup
                ) .
                BIND(
                    IF(?t > 0 && ?t <= 10, 1,
                    IF(?t > 10 && ?t <= 25, 2,
                    IF(?t > 25 && ?t <= 35, 3,
                    IF(?t > 35 && ?t <= 45, 4,
                    IF(?t > 45 && ?t <= 50, 5, 0)))))
                    AS ?band
                ) .
                BIND(
                    IF(?strength = nltl:normalStrengthSteelCategory,
                        IF(?band = 1,
                            IF(?material = nltl:steelMaterialClassThree && ?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeE, nltl:steelGradeB),
                        IF(?band = 2,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeE, nltl:steelGradeD),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeD, nltl:steelGradeB)),
                        IF(?band = 3,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeE, nltl:steelGradeD),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeD, nltl:steelGradeB)),
                        IF(?band = 4,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeE, nltl:steelGradeD),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeD, nltl:steelGradeB)),
                        IF(?band = 5,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup = "PC6-7", nltl:steelGradeE, nltl:steelGradeB),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeE, nltl:steelGradeD)),
                            nltl:steelGradeB))))),
                        IF(?band = 1,
                            IF(?material = nltl:steelMaterialClassThree && ?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeEh, nltl:steelGradeAh),
                        IF(?band = 2,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeEh, nltl:steelGradeDh),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeDh, nltl:steelGradeAh)),
                        IF(?band = 3,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeEh, nltl:steelGradeDh),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeDh, nltl:steelGradeAh)),
                        IF(?band = 4,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeEh, nltl:steelGradeDh),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeDh, nltl:steelGradeAh)),
                        IF(?band = 5,
                            IF(?material = nltl:steelMaterialClassThree,
                                IF(?pcGroup IN ("PC1-3", "PC4-5"), nltl:steelGradeEh, nltl:steelGradeAh),
                                IF(?pcGroup = "PC1-5", nltl:steelGradeEh, nltl:steelGradeDh)),
                            nltl:steelGradeAh)))))
                    AS ?expected
                ) .
                FILTER(?required = ?expected && ?actualRank < ?requiredRank)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable Table 8 selection must have the corresponding required hull structural steel grade; a NOT_APPLICABLE selection must not have one." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:asBuiltPlateThickness ?thickness ;
                      nltl:steelMaterialClass ?material ;
                      nltl:steelStrengthCategory ?strength .
                ?thickness qudt:numericValue ?t .
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?polar .
                BIND(
                    IF(?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3), "PC1-3",
                    IF(?polar IN (nltl:polarClassPc4, nltl:polarClassPc5), "PC4-5",
                    IF(?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3, nltl:polarClassPc4, nltl:polarClassPc5), "PC1-5", "PC6-7")))
                    AS ?pcGroup
                ) .
                BIND(
                    IF(?t > 0 && ?t <= 10, 1,
                    IF(?t > 10 && ?t <= 25, 2,
                    IF(?t > 25 && ?t <= 35, 3,
                    IF(?t > 35 && ?t <= 45, 4,
                    IF(?t > 45 && ?t <= 50, 5, 0)))))
                    AS ?band
                ) .
                FILTER(
                    ?strength = nltl:normalStrengthSteelCategory &&
                    ?material = nltl:steelMaterialClassThree &&
                    ?pcGroup IN ("PC1-3", "PC4-5") &&
                    ?band = 5
                )
                FILTER EXISTS {
                    $this nltl:requiredHullStructuralSteelGrade ?required
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The as-built thickness must fall within a represented Table 8 thickness band." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:asBuiltPlateThickness ?thickness .
                ?thickness qudt:numericValue ?t .
                FILTER(?t <= 0 || ?t > 50)
            }
        """
    ] .
<END_SHACL>
