ex:I2_048_RankShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasSteelGradeRequirementCase ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual hull structural steel grade must have rank greater than or equal to the required grade within the same steel strength category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:steelStrengthCategory ?fam ;
                      nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                FILTER (?actualRank < ?requiredRank)
                FILTER (
                    (?fam = nltl:normalStrengthSteelCategory &&
                     ?actual IN (nltl:steelGradeB,nltl:steelGradeD,nltl:steelGradeE) &&
                     ?required IN (nltl:steelGradeB,nltl:steelGradeD,nltl:steelGradeE))
                    ||
                    (?fam = nltl:highTensileSteelCategory &&
                     ?actual IN (nltl:steelGradeAh,nltl:steelGradeDh,nltl:steelGradeEh,nltl:steelGradeFh) &&
                     ?required IN (nltl:steelGradeAh,nltl:steelGradeDh,nltl:steelGradeEh,nltl:steelGradeFh))
                )
            }
        """
    ] .
