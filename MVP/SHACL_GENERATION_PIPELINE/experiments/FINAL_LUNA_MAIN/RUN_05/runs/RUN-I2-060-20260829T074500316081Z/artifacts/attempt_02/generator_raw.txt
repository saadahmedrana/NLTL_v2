<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_060_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:structuralMember ;
        sh:node gs:I2_060_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member in way of a cutout requires the ship-owned memberInstability value to be explicitly false and not true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout .
                ?cutout a nltl:cutout .
                ?member nltl:memberInWayOfCutout true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:memberInstability false
                    }
                    ||
                    EXISTS {
                        $this nltl:memberInstability true
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where stiffening is necessary for a member in way of a cutout, the ship must provide a nonempty structuralStiffening value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout .
                ?cutout a nltl:cutout .
                ?member nltl:memberInWayOfCutout true .
                ?member nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (
                        DATATYPE(?stiffening) = xsd:string
                        &&
                        STRLEN(STR(?stiffening)) > 0
                    )
                }
            }
        """
    ] .

gs:I2_060_StructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCutout ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:cutout
    ] ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean
    ] .
<END_SHACL>
