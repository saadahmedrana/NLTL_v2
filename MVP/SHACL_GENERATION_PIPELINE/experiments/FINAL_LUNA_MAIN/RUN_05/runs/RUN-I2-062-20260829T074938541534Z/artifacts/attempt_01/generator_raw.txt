<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record that direct calculation is used."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:qualifiedValueShape [
            sh:class nltl:loadCarryingStringer ;
            sh:property [
                sh:path nltl:partOfGrillageSystem ;
                sh:class nltl:grillageSystem ;
                sh:minCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship shall identify a load carrying stringer forming part of a grillage system."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:qualifiedValueShape [
            sh:class nltl:webFrame ;
            sh:property [
                sh:path nltl:partOfGrillageSystem ;
                sh:class nltl:grillageSystem ;
                sh:minCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship shall identify a web frame forming part of a grillage system."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage a nltl:grillageSystem .
                }
            }
        """ ;
        sh:message "Each represented load carrying stringer or web frame shall form part of a grillage system."
    ] .
<END_SHACL>
