<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_064_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one yield-strength value in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:linearCalculationMethodValue ;
                sh:message "The calculation case shall identify the linear calculation method."
            ]
        ] ;
        sh:message "The ship shall provide a calculation case for the applicable linear structural calculation."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasStructuralMemberWeb ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:structuralMemberWeb ;
                    sh:property [
                        sh:path nltl:nominalShearStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:message "The member web shall provide one nominal shear stress in MPa."
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                        sh:message "The applicable buckling criteria shall be recorded as satisfied for the member web."
                    ] ;
                    sh:sparql [
                        a sh:SPARQLConstraint ;
                        sh:message "Nominal web shear stress shall be less than yield strength divided by sqrt(3)." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>

                            SELECT $this
                            WHERE {
                                $this nltl:nominalShearStress ?stressQuantity .
                                ?stressQuantity qudt:numericValue ?stress .
                                ?member nltl:hasStructuralMemberWeb $this .
                                ?ship nltl:hasStructuralMember ?member ;
                                      nltl:yieldStrength ?yieldQuantity .
                                ?yieldQuantity qudt:numericValue ?yield .
                                FILTER (?stress >= (?yield / 1.7320508075688772))
                            }
                        """
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberFlange ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:structuralMemberFlange ;
                    sh:property [
                        sh:path nltl:nominalFlangeVonMisesStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:message "The member flange shall provide one nominal von Mises stress in MPa."
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                        sh:message "The applicable buckling criteria shall be recorded as satisfied for the member flange."
                    ] ;
                    sh:sparql [
                        a sh:SPARQLConstraint ;
                        sh:message "Nominal flange von Mises stress shall be less than 1.15 times yield strength." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>

                            SELECT $this
                            WHERE {
                                $this nltl:nominalFlangeVonMisesStress ?stressQuantity .
                                ?stressQuantity qudt:numericValue ?stress .
                                ?member nltl:hasStructuralMemberFlange $this .
                                ?ship nltl:hasStructuralMember ?member ;
                                      nltl:yieldStrength ?yieldQuantity .
                                ?yieldQuantity qudt:numericValue ?yield .
                                FILTER (?stress >= (1.15 * ?yield))
                            }
                        """
                    ]
                ]
            ]
        ]
    ] .
<END_SHACL>
