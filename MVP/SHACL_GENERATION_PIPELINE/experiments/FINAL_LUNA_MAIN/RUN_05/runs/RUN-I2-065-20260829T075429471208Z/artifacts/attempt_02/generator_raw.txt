<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_065_CalculationCaseShape ;
        sh:class nltl:calculationCase ;
        sh:message "Each linked calculation case must follow the declared nonlinear-calculation case model."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:hasValue nltl:nonlinearCalculationMethodValue
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:fractureMargin ;
                sh:minCount 1 ;
                sh:node gen:I2_065_DimensionlessQuantityShape
            ] ;
            sh:property [
                sh:path nltl:majorBucklingMargin ;
                sh:minCount 1 ;
                sh:node gen:I2_065_DimensionlessQuantityShape
            ] ;
            sh:property [
                sh:path nltl:yieldingMargin ;
                sh:minCount 1 ;
                sh:node gen:I2_065_DimensionlessQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:in (
                            nltl:directCalculationMethodValue
                            nltl:linearCalculationMethodValue
                            nltl:prescribedAnalyticalProcedure
                        )
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "When a linked calculation case uses the nonlinear method, the ship must provide the required acceptance margins." .

gen:I2_065_CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:message "A calculation case must declare its calculation method."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            sh:property [
                sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact ;
                sh:message "A nonlinear calculation case must contain acceptance evidence."
            ] ;
            sh:property [
                sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "The minor permanent deformation assessment must be recorded as satisfied."
            ] ;
            sh:property [
                sh:path nltl:permanentLateralDeformation ;
                sh:minCount 1 ;
                sh:node gen:I2_065_LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:permanentOutOfPlaneDeformation ;
                sh:minCount 1 ;
                sh:node gen:I2_065_LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:relevantStructuralDimension ;
                sh:minCount 1 ;
                sh:node gen:I2_065_LengthQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:in (
                    nltl:directCalculationMethodValue
                    nltl:linearCalculationMethodValue
                    nltl:prescribedAnalyticalProcedure
                )
            ]
        ]
    ) ;
    sh:message "A nonlinear calculation case must provide the declared method, evidence, assessment result, and deformation-related inputs." .

gen:I2_065_DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .

gen:I2_065_LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
