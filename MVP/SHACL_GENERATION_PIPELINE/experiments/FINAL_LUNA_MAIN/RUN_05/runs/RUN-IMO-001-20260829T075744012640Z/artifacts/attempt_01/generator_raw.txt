<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-001-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare a design ice condition to determine applicability of the category rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for at least medium first-year ice with possible old-ice inclusions must have Polar Ship Category A." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions .
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryA .
                }
            }
        """
    ] .
<END_SHACL>
