<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-001_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition."
    ] ;
    sh:sparql [
        sh:message "A ship designed for at least medium first-year ice with possible old-ice inclusions must have exactly Category A as its ship category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions .
                {
                    FILTER NOT EXISTS {
                        $this nltl:shipCategory nltl:polarShipCategoryA .
                    }
                }
                UNION
                {
                    $this nltl:shipCategory ?category .
                    FILTER (?category != nltl:polarShipCategoryA)
                }
            }
        """
    ] .
<END_SHACL>
