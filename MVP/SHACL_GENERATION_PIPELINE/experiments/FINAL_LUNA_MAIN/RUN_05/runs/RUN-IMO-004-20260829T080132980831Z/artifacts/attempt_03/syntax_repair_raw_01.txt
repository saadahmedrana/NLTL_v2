<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer ;
        sh:message "Any supplied winter growth count must be an integer."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:message "Any supplied ice thickness must have a decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:message "Any supplied ice thickness must use metres."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m, the ice type classification must be exactly first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
           nltl:winterGrowthCount ?growth ;
           nltl:iceThickness ?thicknessNode .
    ?thicknessNode qudt:numericValue ?thickness ;
                   qudt:unit unit:M .
    FILTER (
        ?growth <= 1 &&
        ?thickness >= "0.3"^^xsd:decimal &&
        ?thickness <= "2.0"^^xsd:decimal
    )
    FILTER (
        NOT EXISTS {
            $this nltl:iceTypeClassification nltl:firstYearIce .
        }
        ||
        EXISTS {
            $this nltl:iceTypeClassification ?classification .
            FILTER (?classification != nltl:firstYearIce)
        }
    )
}
"""
    ] .
</END_SHACL>
