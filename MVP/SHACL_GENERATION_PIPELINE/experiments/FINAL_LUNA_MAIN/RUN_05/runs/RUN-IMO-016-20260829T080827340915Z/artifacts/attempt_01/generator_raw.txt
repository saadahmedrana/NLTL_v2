<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must state the language used for the Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:node gen:IMO_016_CertificateShape ;
        sh:message "The ship must have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:in ( "English" "French" "Spanish" )
            ]
        ]
        [
            sh:property [
                sh:path nltl:approvedTranslationPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:message "If the certificate language is not English, French, or Spanish, an approved translation into one of those languages must be present." .

gen:IMO_016_CertificateShape
    a sh:NodeShape ;
    sh:class nltl:polarShipCertificateForm ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:class nltl:certificateFormModelValue ;
        sh:message "The Polar Ship Certificate must use the Appendix I certificate form model."
    ] .
<END_SHACL>
