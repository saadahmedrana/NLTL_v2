<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate validity date is required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate survey date is required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate endorsement date is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Ship Certificate supplement shall be present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "The ship shall have a Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasScheduleShape ;
        sh:message "The ship shall have a SOLAS certificate schedule."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017SupplementShape ;
        sh:message "The Polar Ship Certificate supplement record is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate dates and endorsements shall match the corresponding SOLAS certificate schedule dates." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                {
                    $this nltl:polarCertificateValidityDates ?d .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule/
                              nltl:hasCertificateScheduleDateRecord ?r .
                        ?r nltl:certificateScheduleDate ?d ;
                           nltl:certificateScheduleDateType nltl:validityDateCategory .
                    }
                }
                UNION
                {
                    $this nltl:surveyDates ?d .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule/
                              nltl:hasCertificateScheduleDateRecord ?r .
                        ?r nltl:certificateScheduleDate ?d ;
                           nltl:certificateScheduleDateType nltl:surveyDateCategory .
                    }
                }
                UNION
                {
                    $this nltl:endorsementDates ?d .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule/
                              nltl:hasCertificateScheduleDateRecord ?r .
                        ?r nltl:certificateScheduleDate ?d ;
                           nltl:certificateScheduleDateType nltl:endorsementDateCategory .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule/
                          nltl:hasCertificateScheduleDateRecord ?r .
                    ?r nltl:certificateScheduleDate ?d ;
                       nltl:certificateScheduleDateType nltl:validityDateCategory .
                    FILTER NOT EXISTS {
                        $this nltl:polarCertificateValidityDates ?d .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule/
                          nltl:hasCertificateScheduleDateRecord ?r .
                    ?r nltl:certificateScheduleDate ?d ;
                       nltl:certificateScheduleDateType nltl:surveyDateCategory .
                    FILTER NOT EXISTS {
                        $this nltl:surveyDates ?d .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule/
                          nltl:hasCertificateScheduleDateRecord ?r .
                    ?r nltl:certificateScheduleDate ?d ;
                       nltl:certificateScheduleDateType nltl:endorsementDateCategory .
                    FILTER NOT EXISTS {
                        $this nltl:endorsementDates ?d .
                    }
                }
            }
        """
    ] .

gen:IMO017SolasScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017DateRecordShape ;
        sh:message "The SOLAS certificate schedule shall contain certificate schedule date records."
    ] .

gen:IMO017DateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record shall contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall identify a validity, survey, or endorsement date category."
    ] .

gen:IMO017SupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Ship Certificate supplement shall record the equipment required by the Code."
    ] .
<END_SHACL>
