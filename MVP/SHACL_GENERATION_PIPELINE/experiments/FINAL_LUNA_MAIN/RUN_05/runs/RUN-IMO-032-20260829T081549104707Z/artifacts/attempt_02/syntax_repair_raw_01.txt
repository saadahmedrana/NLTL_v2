<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO032ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryA ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node gen:IMO032ApprovalABShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryB ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node gen:IMO032ApprovalABShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:shipIceStrengthened ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node gen:IMO032ApprovalCShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:shipIceStrengthened ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "The ship shall satisfy the applicable category-conditioned scantling approval requirements." .

gen:IMO032ApprovalABShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1
    ] ;
    sh:message "The approval record shall be approved by an authorized authority and use an Organization-acceptable or equivalent-safety standard." .

gen:IMO032ApprovalCShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:hasValue "acceptable standard adequate for operating ice type and concentration" ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:node gen:IMO032IceConcentrationShape
    ] ;
    sh:message "The approval record for an ice-strengthened category C ship shall include an authorized approval, an adequate acceptable standard, operating ice type, and operating ice concentration." .

gen:IMO032IceConcentrationShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
