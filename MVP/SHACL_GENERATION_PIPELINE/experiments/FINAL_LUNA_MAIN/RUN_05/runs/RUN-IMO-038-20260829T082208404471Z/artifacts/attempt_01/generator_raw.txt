    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;

            sh:property [

    sh:property [

    sh:sparql [
        sh:message "The calculated damage extents or selected vertical damage position do not satisfy the Polar Code formula and inclusive range requirements." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?selectedNode .

                ?lengthNode <http://qudt.org/schema/qudt/numericValue> ?length .
                ?draughtNode <http://qudt.org/schema/qudt/numericValue> ?draught .
                ?longitudinalNode <http://qudt.org/schema/qudt/numericValue> ?longitudinal .
                ?verticalNode <http://qudt.org/schema/qudt/numericValue> ?vertical .
                ?selectedNode <http://qudt.org/schema/qudt/numericValue> ?selected .

                BIND(
                    IF(
                        ?forward = true,
                        0.045 * ?length,
                        0.015 * ?length
                    )
                    AS ?expectedLongitudinal
                )

                BIND(
                    IF(
                        0.20 * ?draught < ?expectedLongitudinal,
                        0.20 * ?draught,
                        ?expectedLongitudinal
                    )
                    AS ?expectedVertical
                )

                FILTER(
                    ABS(?longitudinal - ?expectedLongitudinal)
                        > (0.000001 * ABS(?expectedLongitudinal) + 0.000000001)
                    ||
                    ABS(?vertical - ?expectedVertical)
                        > (0.000001 * ABS(?expectedVertical) + 0.000000001)
                    ||
                    ?selected < 0
                    ||
                    ?selected > (1.20 * ?draught)
                )
            }
        """
    ] .
