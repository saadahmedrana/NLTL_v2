<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO040RelevantHatchOrDoorShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasIceOrSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:message "Each relevant hatch or door shall have an associated ice or snow removal or prevention means."
    ] .

gen:IMO040ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:message "The ship shall provide a boolean ice-accretion applicability value."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceAccretionLikely ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceAccretionLikely ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasRelevantHatchOrDoor ;
                sh:node gen:IMO040RelevantHatchOrDoorShape
            ]
        ]
    ) .
