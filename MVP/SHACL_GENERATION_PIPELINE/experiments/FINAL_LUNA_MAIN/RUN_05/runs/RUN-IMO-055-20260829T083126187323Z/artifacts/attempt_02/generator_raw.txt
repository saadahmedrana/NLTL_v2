<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-055-ship-valve-protection> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Each ship must provide whether the valve is exposed." ;
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each ship must provide the valve type." ;
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean ;
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean ;
    ] ;
    sh:sparql [
        sh:message "An exposed isolating, pressure, or vacuum valve must have ice-accretion protection and remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:valveExposed ?exposed ;
                      nltl:valveType ?type .
                OPTIONAL {
                    $this nltl:iceAccretionProtectionPresent ?protected .
                }
                OPTIONAL {
                    $this nltl:continuouslyAccessible ?accessible .
                }
                FILTER (
                    ?exposed = true &&
                    ?type IN ("isolating", "pressure", "vacuum") &&
                    (
                        !BOUND(?protected) ||
                        ?protected != true ||
                        !BOUND(?accessible) ||
                        ?accessible != true
                    )
                )
            }
        """ ;
    ] .
<END_SHACL>
