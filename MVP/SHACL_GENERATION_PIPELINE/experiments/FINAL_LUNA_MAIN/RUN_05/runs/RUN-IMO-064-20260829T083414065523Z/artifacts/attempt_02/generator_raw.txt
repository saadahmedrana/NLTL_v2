<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO-064-StaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "The ship must provide the persons-on-board count needed to establish thermal-protection coverage."
    ] ;
    sh:property [
        sh:path nltl:assignedThermalProtection ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Assigned thermal protection must be recorded as provided."
    ] ;
    sh:property [
        sh:path nltl:intendedVoyage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The intended voyage must be recorded for the thermal-protection adequacy assessment."
    ] ;
    sh:property [
        sh:path nltl:anticipatedCold ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated cold conditions must be recorded for the thermal-protection adequacy assessment."
    ] ;
    sh:property [
        sh:path nltl:anticipatedWind ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated wind conditions must be recorded for the thermal-protection adequacy assessment."
    ] ;
    sh:property [
        sh:path nltl:immersionPotential ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Potential for immersion in polar water must be recorded for the thermal-protection adequacy assessment."
    ] ;
    sh:property [
        sh:path nltl:thermalProtectionApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Thermal-protection adequacy approval must have approved status."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Approved thermal-protection adequacy evidence must be tied to the ship's intended voyage, anticipated cold and wind, immersion potential, person-on-board count, and assigned thermal protection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                FILTER NOT EXISTS {
                    $this nltl:personOnBoard ?persons ;
                          nltl:assignedThermalProtection true ;
                          nltl:intendedVoyage ?voyage ;
                          nltl:anticipatedCold ?cold ;
                          nltl:anticipatedWind ?wind ;
                          nltl:immersionPotential ?immersion ;
                          nltl:thermalProtectionApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
