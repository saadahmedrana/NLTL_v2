<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_066_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship exposure-to-ice-accretion applicability assessment is required and must be a single boolean."
    ] ;
    sh:property [
        sh:path nltl:hasProtectedItem ;
        sh:class nltl:protectedItem
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "Every supplied protected item in an applicable category must have associated ice or snow removal or prevention means." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:shipExposedToIceAccretion true .
                        $this nltl:hasProtectedItem ?item .
                        ?item nltl:protectedItemCategory ?category .
                        FILTER (
                            ?category IN (
                                nltl:escapeRouteProtectedItem,
                                nltl:musterStationProtectedItem,
                                nltl:embarkationAreaProtectedItem,
                                nltl:survivalCraftProtectedItem,
                                nltl:launchingApplianceProtectedItem,
                                nltl:survivalCraftAccessProtectedItem
                            )
                        )
                        FILTER NOT EXISTS {
                            ?item nltl:hasIceSnowRemovalOrPreventionMeans ?means .
                            ?means a nltl:iceSnowRemovalOrPreventionMeans .
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
