<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO072ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltlgen:IMO072ExtendedDarknessTrueBranch
        nltlgen:IMO072ExtendedDarknessFalseBranch
    ) .

nltlgen:IMO072ExtendedDarknessTrueBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must indicate extended darkness operation when the conditional requirement applies."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:node nltlgen:IMO072LifeboatShape ;
        sh:message "Each lifeboat must have an assigned searchlight suitable for continuous use during extended darkness operation."
    ] .

nltlgen:IMO072ExtendedDarknessFalseBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "The ship must provide the extended darkness operation applicability value."
    ] .

nltlgen:IMO072LifeboatShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAssignedSearchlight ;
        sh:minCount 1 ;
        sh:node nltlgen:IMO072SearchlightShape ;
        sh:message "Each lifeboat must have at least one assigned searchlight."
    ] .

nltlgen:IMO072SearchlightShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:searchlightContinuousUseSuitabilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The assigned searchlight must be suitable for continuous use."
    ] .
<END_SHACL>
