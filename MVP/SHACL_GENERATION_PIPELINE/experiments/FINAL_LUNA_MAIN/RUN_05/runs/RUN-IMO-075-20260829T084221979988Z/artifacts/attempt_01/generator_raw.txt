<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-075-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "personalOrGroupSurvivalEquipmentRequired must be provided as exactly one boolean value."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:IMO-075-NotRequiredBranch>
        <urn:nltl:generated-shape:IMO-075-RequiredBranch>
    ) .

<urn:nltl:generated-shape:IMO-075-NotRequiredBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue false
    ] .

<urn:nltl:generated-shape:IMO-075-RequiredBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "personsOnBoard is required when personal or group survival equipment is required."
    ] ;
    sh:property [
        sh:path nltl:availableSurvivalEquipmentCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "availableSurvivalEquipmentCapacity is required when personal or group survival equipment is required."
    ] ;
    sh:property [
        sh:path nltl:storageAccessibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival equipment must be stored in an easily accessible location."
    ] ;
    sh:sparql [
        sh:message "availableSurvivalEquipmentCapacity must be at least the ceiling of 110 percent of personsOnBoard." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:personalOrGroupSurvivalEquipmentRequired true ;
                      nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (?capacity < CEIL(?persons * 1.10))
            }
        """
    ] .
<END_SHACL>
