gen:IMO_075_SurvivalEquipmentCapacityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "personalOrGroupSurvivalEquipmentRequired must be false when the survival-equipment rule is not applicable."
            ]
        ]
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "personalOrGroupSurvivalEquipmentRequired must be recorded as true when the survival-equipment rule is applicable."
            ] ;
            sh:property [
                sh:path nltl:personsOnBoard ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The number of persons on board is required to determine survival-equipment capacity."
            ] ;
            sh:property [
                sh:path nltl:availableSurvivalEquipmentCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Available survival equipment capacity is required when personal or group survival equipment is required."
            ] ;
            sh:property [
                sh:path nltl:storageAccessibilityStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Survival equipment must be stored in an easily accessible location."
            ] ;
            sh:sparql [
                sh:message "Available survival equipment capacity must be at least 110% of the persons on board, rounded up to the next integer." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:personsOnBoard ?persons ;
                              nltl:availableSurvivalEquipmentCapacity ?capacity .
                        FILTER (?capacity * 10 < ?persons * 11)
                    }
                """
            ]
        )
    .
