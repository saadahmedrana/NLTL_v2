<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO086ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gen:IMO086HeadingMeansShape ;
        sh:message "The ship shall have at least two non-magnetic heading means, each meeting the required power and independence constraints."
    ] .

gen:IMO086HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be independent from another non-magnetic heading means."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                ?ship nltl:hasNonMagneticHeadingMeans $this .
                ?ship nltl:hasNonMagneticHeadingMeans ?other .
                $this nltl:independentFromHeadingMeans ?other .
                FILTER (?other != $this)
            }
        """ ;
        sh:message "The independent heading means relation shall identify another distinct non-magnetic heading means belonging to the same ship."
    ] .
<END_SHACL>
