<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_086_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:message "The ship shall have at least two non-magnetic heading means."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship shall have two distinct non-magnetic heading means, each connected to main and emergency power, with reciprocal independence relations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?m1 .
                    $this nltl:hasNonMagneticHeadingMeans ?m2 .
                    FILTER (?m1 != ?m2)

                    ?m1 a nltl:nonMagneticHeadingMeans ;
                        nltl:connectedToMainPower ?main1 ;
                        nltl:connectedToEmergencyPower ?emergency1 ;
                        nltl:independentFromHeadingMeans ?m2 .

                    ?m2 a nltl:nonMagneticHeadingMeans ;
                        nltl:connectedToMainPower ?main2 ;
                        nltl:connectedToEmergencyPower ?emergency2 ;
                        nltl:independentFromHeadingMeans ?m1 .

                    ?main1 a nltl:benchmarkEntity .
                    ?emergency1 a nltl:benchmarkEntity .
                    ?main2 a nltl:benchmarkEntity .
                    ?emergency2 a nltl:benchmarkEntity .
                }
            }
        """
    ] .
<END_SHACL>
