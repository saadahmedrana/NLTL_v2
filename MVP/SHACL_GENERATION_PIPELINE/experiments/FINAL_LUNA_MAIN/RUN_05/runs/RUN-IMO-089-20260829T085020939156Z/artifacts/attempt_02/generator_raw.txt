<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_089_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship is involved in an icebreaker escort operation, it must have a manually initiated flashing red stopped light visible from astern, with a visibility range of at least two nautical miles and compliant stern-light arcs." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    FILTER EXISTS {
        $this nltl:icebreakerEscortOperation true .
    }
    FILTER NOT EXISTS {
        $this nltl:manuallyInitiatedFlashingRedStoppedLightPresent true ;
              nltl:lightVisibleFromAstern true ;
              nltl:visibilityRange ?quantity ;
              nltl:colregSternLightArcComplianceStatus true .

        ?quantity a qudt:QuantityValue ;
                  qudt:numericValue ?value ;
                  qudt:unit unit:MI_N .

        FILTER (
            datatype(?value) = xsd:decimal &&
            ?value >= 2
        )
    }
}
"""
    ] .
<END_SHACL>
