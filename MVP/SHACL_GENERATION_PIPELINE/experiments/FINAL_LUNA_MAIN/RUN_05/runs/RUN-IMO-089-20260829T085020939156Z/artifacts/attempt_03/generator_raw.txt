<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_089_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "When icebreaker escort operation is true, the required stopped-light conditions must be satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icebreakerEscortOperation true .

                FILTER (
                    NOT EXISTS {
                        $this nltl:manuallyInitiatedFlashingRedStoppedLightPresent true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:lightVisibleFromAstern true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:visibilityRange ?range .
                        ?range qudt:unit unit:MI_N .
                        ?range qudt:numericValue ?value .
                        FILTER (isNumeric(?value) && ?value >= "2"^^xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:colregSternLightArcComplianceStatus true .
                    }
                )
            }
        """
    ] .
<END_SHACL>
