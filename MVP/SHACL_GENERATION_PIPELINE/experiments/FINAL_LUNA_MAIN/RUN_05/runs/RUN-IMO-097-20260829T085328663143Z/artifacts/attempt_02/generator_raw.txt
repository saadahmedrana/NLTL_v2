<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_097_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether it operates in low air temperature."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat ;
        sh:minCount 1 ;
        sh:message "The ship must provide its lifeboat relationship."
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft ;
        sh:minCount 1 ;
        sh:message "The ship must provide its other-survival-craft relationship."
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat ;
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship operates in low air temperature, every linked released rescue boat or lifeboat must also be linked as a survival craft and must carry at least one distress-alert device, one location-signal device, and one on-scene communication device, with the declared distress-alert relationship present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true .
                $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasSurvivalCraft ?craft .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?distressDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?shipToShoreDevice .
                    }
                )
            }
        """
    ] .
</END_SHACL>
