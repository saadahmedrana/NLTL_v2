<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-097> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:minCount 1 ;
        sh:message "The ship must record its lifeboat relationship."
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:minCount 1 ;
        sh:message "The ship must record its other-survival-craft relationship."
    ] ;
    sh:sparql [
        sh:message "When the ship operates in low air temperature, every released rescue boat or lifeboat must carry a distress-alert device, and the corresponding survival-craft node must carry a location-signal device, an on-scene communication device, and a ship-to-shore distress-alert device." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:shipOperatesInLowAirTemperature true .
                $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                ?craft nltl:releasedRescueBoatOrLifeboat true .
                FILTER (
                    NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?distressDevice .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasSurvivalCraft ?craft .
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?shipToShoreDevice .
                    }
                )
            }
        """
    ] .
</END_SHACL>
