@prefix gen: <urn:nltl:generated-shape:> .

@prefix qv: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_099_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record that a battery-life management procedure is present."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record that the procedure is implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qv:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qv:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qv:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DAY
            ]
        ] ;
        sh:message "Communication equipment availability shall be provided as a single duration in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qv:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qv:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qv:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DAY
            ]
        ] ;
        sh:message "The maximum expected rescue time shall be provided as a single duration in days."
    ] ;
    sh:sparql [
        sh:message "Communication equipment shall remain available for at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qv: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availability .
                ?availability qv:numericValue ?availabilityValue .
                ?availability qv:unit unit:DAY .
                $this nltl:maximumExpectedRescueTime ?rescueTime .
                ?rescueTime qv:numericValue ?rescueValue .
                ?rescueTime qv:unit unit:DAY .
                FILTER (xsd:decimal(?availabilityValue) < xsd:decimal(?rescueValue))
            }
        """
    ] .
