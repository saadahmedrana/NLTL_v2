<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node nltl:IMO_104_CrewMemberShape
    ] ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization."
    ] ;
    sh:sparql [
        sh:message "Each assigned duty's relevant PWOM procedure or equipment shall have a corresponding crew-member familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

nltl:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty ;
        sh:node nltl:IMO_104_AssignedDutyShape ;
        sh:minCount 1 ;
        sh:message "Each crew member shall have an assigned duty."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:node nltl:IMO_104_FamiliarizationRecordShape ;
        sh:minCount 1 ;
        sh:message "Each crew member shall have a PWOM familiarization record."
    ] .

nltl:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:node nltl:IMO_104_ManualItemShape
    ] .

nltl:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:node nltl:IMO_104_ManualItemShape ;
        sh:minCount 1 ;
        sh:message "A familiarization record shall identify the relevant PWOM procedure or equipment."
    ] .

nltl:IMO_104_ManualItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .

<END_SHACL>
