<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization status."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:node g:IMO_104_CrewMemberShape
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node g:IMO_104_CrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each relevant PWOM procedure or equipment item assigned to a crew member must have a corresponding familiarization record item." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this (nltl:hasCrewMember|nltl:hasCrewMemberInventory) ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

g:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:node g:IMO_104_AssignedDutyShape
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:node g:IMO_104_FamiliarizationRecordShape
    ] .

g:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node g:IMO_104_ManualItemShape ;
        sh:message "An assigned duty must identify the relevant PWOM procedure or equipment."
    ] .

g:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node g:IMO_104_ManualItemShape ;
        sh:message "A familiarization record must identify the relevant PWOM procedure or equipment."
    ] .

g:IMO_104_ManualItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .
<END_SHACL>
