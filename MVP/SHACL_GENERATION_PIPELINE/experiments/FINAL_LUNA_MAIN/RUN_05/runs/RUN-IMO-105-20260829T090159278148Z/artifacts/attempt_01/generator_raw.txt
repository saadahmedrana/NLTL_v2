<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO_105_ArcticOilDischargeShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, oil or oily mixture discharge to sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (
                    datatype(?area) = xsd:string &&
                    STR(?area) = "Arctic_waters"
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                        FILTER (
                            datatype(?discharge) = xsd:boolean &&
                            ?discharge = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?nonconformingDischarge .
                        FILTER (
                            !(
                                datatype(?nonconformingDischarge) = xsd:boolean &&
                                ?nonconformingDischarge = false
                            )
                        )
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
