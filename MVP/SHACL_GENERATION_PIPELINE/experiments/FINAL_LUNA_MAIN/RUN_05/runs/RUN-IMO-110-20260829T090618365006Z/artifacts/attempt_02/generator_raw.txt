<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        gen:IMO_110_ApplicableCase
        gen:IMO_110_NonApplicableCase
    ) .

gen:IMO_110_ApplicableCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path rdf:type ;
        sh:hasValue nltl:oilTanker ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:maxInclusive "9999-12-31"^^xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive "5000"^^xsd:decimal
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable oil tankers must have compliant double-bottom protection over the entire cargo tank length."
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable oil tankers must have compliant wing-tank protection."
    ] .

gen:IMO_110_NonApplicableCase
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The applicability conditions are not satisfied, or the ship is not an oil tanker." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                $this nltl:constructionDate ?constructionDate .
                $this nltl:deadweight ?deadweightNode .
                ?deadweightNode qudt:numericValue ?deadweightValue .
                ?deadweightNode qudt:unit unit:TON_Metric .
                FILTER (
                    ?category NOT IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    || ?constructionDate < "2017-01-01"^^xsd:date
                    || ?deadweightValue >= "5000"^^xsd:decimal
                    || NOT EXISTS {
                        $this a nltl:oilTanker .
                    }
                )
            }
        """
    ] .
<END_SHACL>
