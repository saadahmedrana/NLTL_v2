gen:IMO118Shape a sh:NodeShape ;

        gen:IMO118Pre2017Shape
        gen:IMO118CategoryApplicableShape
        gen:IMO118PassengerApplicableShape
    ) ;
    sh:message "For ships constructed on or after 1 January 2017, category or passenger-ship applicability and the sewage discharge prohibition or compliance exception must be established." .

    sh:property [
        sh:path nltl:constructionDate ;

        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1
    ] ;
    sh:or (
        gen:IMO118DischargeProhibitedShape
        gen:IMO118ComplianceExceptionShape
    ) .

gen:IMO118PassengerApplicableShape a sh:NodeShape ;
    sh:class nltl:passengerShip ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1
    ] ;
    sh:or (
        gen:IMO118DischargeProhibitedShape
        gen:IMO118ComplianceExceptionShape
    ) .

    sh:property [
        sh:path nltl:sewageDischargeToSea ;

        sh:minCount 1
    ] .

    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] .
