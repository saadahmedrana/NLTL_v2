@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-119> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extendedPeriodOperation ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeOccurs ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState
    ] ;
    sh:property [
        sh:path nltl:marpolCertificationStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState
    ] ;
    sh:sparql [
        sh:message "For a Category A or B ship with ice concentration above 1/10 operating for an extended period while sewage discharge occurs, an approved sewage treatment plant and Administration discharge approval are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:iceConcentration ?iceQuantity ;
                      nltl:extendedPeriodOperation true ;
                      nltl:sewageDischargeOccurs true .

                ?iceQuantity qudt:numericValue ?iceValue ;
                             qudt:unit unit:UNITLESS .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?iceValue > 1
                )

                FILTER NOT EXISTS {
                    $this nltl:approvedSewageTreatmentPlantUsed true
                }

                FILTER NOT EXISTS {
                    $this nltl:administrationDischargeApprovalStatus
                           nltl:evidenceStateApproved
                }
            }
        """
    ] .
