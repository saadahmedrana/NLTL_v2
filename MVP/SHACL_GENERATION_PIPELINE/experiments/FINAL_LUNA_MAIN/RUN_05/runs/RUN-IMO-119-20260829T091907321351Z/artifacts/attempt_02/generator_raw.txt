<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_119_SewageDischargeApprovalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the Category A/B, ice concentration, extended-operation, and sewage-discharge conditions apply, the required treatment approval and MARPOL certification statuses must be approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:iceConcentration ?ice ;
                      nltl:extendedPeriodOperation true ;
                      nltl:sewageDischargeOccurs true .

                ?ice qudt:numericValue ?iceValue ;
                     qudt:unit unit:UNITLESS .

                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (datatype(?iceValue) = xsd:decimal)
                FILTER (?iceValue > 1)

                FILTER (
                    NOT EXISTS {
                        $this nltl:approvedSewageTreatmentPlantUsed true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus
                            nltl:evidenceStateApproved
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolCertificationStatus
                            nltl:evidenceStateApproved
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:administrationDischargeApprovalStatus
                            nltl:evidenceStateApproved
                    }
                )
            }
        """
    ] .
<END_SHACL>
