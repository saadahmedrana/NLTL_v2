<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_005_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:headingOrPositionSystem ;
        sh:datatype xsd:string ;
        sh:message "Each represented heading-reference or position-fixing system must be a string."
    ] ;
    sh:property [
        sh:path nltl:intendedArea ;
        sh:datatype xsd:string ;
        sh:message "The represented intended area must be a string."
    ] ;
    sh:property [
        sh:path nltl:intendedAreaSuitabilityApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState ;
        sh:in ( nltl:evidenceStateApproved ) ;
        sh:message "Any intended-area suitability approval status must be Approved."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When heading-reference or position-fixing systems are represented, the ship must provide an intended area and an Approved suitability status for those systems." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:headingOrPositionSystem ?system .
                FILTER (
                    NOT EXISTS {
                        $this nltl:intendedArea ?area .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:intendedAreaSuitabilityApprovalStatus nltl:evidenceStateApproved .
                    }
                )
            }
        """
    ] .
<END_SHACL>
