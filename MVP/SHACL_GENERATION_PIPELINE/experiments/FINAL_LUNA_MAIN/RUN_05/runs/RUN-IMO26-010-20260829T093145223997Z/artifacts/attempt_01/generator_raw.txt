<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO26_010_ApplicableBranch
        gen:IMO26_010_NotApplicableBranch
    ) ;
    sh:message "The ship must identify whether below-hull sensor projection applies; where it applies, the sensors must be protected against ice." .

gen:IMO26_010_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredSensorProjectsBelowHull ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:sensorIceProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Sensors projecting below the hull must be protected against ice."
    ] .

gen:IMO26_010_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredSensorProjectsBelowHull ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] .
