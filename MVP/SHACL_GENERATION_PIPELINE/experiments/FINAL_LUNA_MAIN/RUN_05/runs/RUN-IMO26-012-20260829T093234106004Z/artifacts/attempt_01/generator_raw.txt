<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When at least two non-magnetic heading means are represented, each must be connected to main and emergency power and be independent from another heading means of the same ship." ;
        sh:select """
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?first .
                    $this nltl:hasNonMagneticHeadingMeans ?second .
                    FILTER (?first != ?second)
                }
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER (
                    NOT EXISTS {
                        ?means nltl:connectedToMainPower ?mainPower .
                    }
                    ||
                    NOT EXISTS {
                        ?means nltl:connectedToEmergencyPower ?emergencyPower .
                    }
                    ||
                    NOT EXISTS {
                        ?means nltl:independentFromHeadingMeans ?otherMeans .
                        $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                        FILTER (?otherMeans != ?means)
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
