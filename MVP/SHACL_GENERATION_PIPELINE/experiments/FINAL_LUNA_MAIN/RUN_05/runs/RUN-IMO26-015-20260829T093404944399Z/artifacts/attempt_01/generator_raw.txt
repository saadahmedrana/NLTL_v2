        sh:datatype xsd:boolean ;
        sh:message "SOLAS chapter I certification status, when recorded, must be a boolean."
    ] ;

        sh:message "Polar Code chapter 11 may be applicable only to a ship certified in accordance with SOLAS chapter I." ;

                $this nltl:polarCodeChapter11Applicable ?applicable .

                }
            }
        """
    ] .
