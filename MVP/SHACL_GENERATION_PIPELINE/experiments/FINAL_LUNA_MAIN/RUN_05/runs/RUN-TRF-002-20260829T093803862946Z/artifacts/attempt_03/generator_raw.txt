<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-002> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one construction contract date to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ships contracted on or after 2019-01-01 and before 2021-07-05, the applicable ice-class regulation edition must be exactly the 2017 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    ?date >= "2019-01-01"^^xsd:date &&
                    ?date < "2021-07-05"^^xsd:date
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                            nltl:iceClassRegulationEdition2017
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?otherEdition .
                        FILTER (
                            ?otherEdition !=
                            nltl:iceClassRegulationEdition2017
                        )
                    }
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The 2017 ice-class regulation edition must not be asserted for ships contracted outside the 2019-01-01 inclusive to 2021-07-05 exclusive interval." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    ?date < "2019-01-01"^^xsd:date ||
                    ?date >= "2021-07-05"^^xsd:date
                )
                $this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRegulationEdition2017 .
            }
        """ ;
    ] .
<END_SHACL>
