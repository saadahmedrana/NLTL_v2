<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF003ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A ship's construction contract date is required to determine whether the 2010 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships contracted for construction on or after 1 January 2012 but before 1 January 2019 shall identify the 2010 Ice Class Regulations as the applicable regulation edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    ?date >= "2012-01-01"^^xsd:date &&
                    ?date < "2019-01-01"^^xsd:date
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2010 .
                }
            }
        """
    ] .
<END_SHACL>
