<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship whose construction stage was on or after 1 September 2003 and whose construction contract was before 1 January 2010, the applicable ice class regulation edition must be the 2002 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stage ;
                       nltl:constructionContractDate ?contract .

                FILTER (
                    YEAR(?stage) > 2003 ||
                    (
                        YEAR(?stage) = 2003 &&
                        (
                            MONTH(?stage) > 9 ||
                            (MONTH(?stage) = 9 && DAY(?stage) >= 1)
                        )
                    )
                )

                FILTER (
                    YEAR(?contract) < 2010 ||
                    (
                        YEAR(?contract) = 2010 &&
                        (
                            MONTH(?contract) < 1 ||
                            (MONTH(?contract) = 1 && DAY(?contract) < 1)
                        )
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2002
                }

                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition ?edition .
                    FILTER (?edition != nltl:iceClassRegulationEdition2002)
                }
            }
        """
    ] .
<END_SHACL>
