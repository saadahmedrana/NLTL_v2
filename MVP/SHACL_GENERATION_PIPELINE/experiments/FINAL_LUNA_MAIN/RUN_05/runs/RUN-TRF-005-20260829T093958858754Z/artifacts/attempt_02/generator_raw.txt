<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "Construction contract date must be an xsd:date when supplied."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "Construction stage date must be an xsd:date when supplied."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship whose construction stage began on or after 1 September 2003 and whose construction contract predates 1 January 2010, the applicable ice class regulation edition must be exactly the 2002 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate ;
                      nltl:constructionContractDate ?contractDate .
                FILTER (
                    ?stageDate >= "2003-09-01"^^xsd:date &&
                    ?contractDate < "2010-01-01"^^xsd:date
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                            nltl:iceClassRegulationEdition2002
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?edition .
                        FILTER (?edition != nltl:iceClassRegulationEdition2002)
                    }
                )
            }
        """
    ] .
<END_SHACL>
