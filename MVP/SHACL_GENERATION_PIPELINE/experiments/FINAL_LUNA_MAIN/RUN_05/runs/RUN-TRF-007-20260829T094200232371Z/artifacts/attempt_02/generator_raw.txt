<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must provide exactly one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one construction-stage date."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2003-09-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must provide exactly one delivery date."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must provide exactly one assessment date."
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "An applicable ship must identify section 3.2.2 or section 3.2.4 as its compliance method."
            ] ;
            sh:sparql [
                sh:message "For an applicable ship, compliance must be established no later than 1 January in the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        BIND (YEAR(?deliveryDate) + 20 AS ?deadlineYear)
                        FILTER (
                            YEAR(?assessmentDate) > ?deadlineYear
                            || (
                                YEAR(?assessmentDate) = ?deadlineYear
                                && MONTH(?assessmentDate) > 1
                            )
                            || (
                                YEAR(?assessmentDate) = ?deadlineYear
                                && MONTH(?assessmentDate) = 1
                                && DAY(?assessmentDate) > 1
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:in (
                            nltl:iceClassIb
                            nltl:iceClassIc
                            nltl:iceClassIi
                            nltl:iceClassIii
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:constructionStageDate ;
                        sh:datatype xsd:date ;
                        sh:minInclusive "2003-09-01"^^xsd:date
                    ]
                ]
            )
        ]
    ) ;
    sh:message "The conditional requirements apply to IA Super or IA ships constructed before 1 September 2003; such ships must satisfy the stated delivery-date deadline and use one permitted compliance method."
.
<END_SHACL>
