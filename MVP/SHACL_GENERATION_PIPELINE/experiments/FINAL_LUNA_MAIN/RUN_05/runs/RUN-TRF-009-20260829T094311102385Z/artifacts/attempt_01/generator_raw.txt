<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable IA or IA Super ship, construction stage date, delivery date, assessment date, and a permitted 2017 compliance method are required; the deadline must not have passed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIa, nltl:iceClassIaSuper))

                FILTER (
                    NOT EXISTS {
                        $this nltl:constructionStageDate ?constructionDate .
                        FILTER (
                            datatype(?constructionDate) = xsd:date &&
                            (
                                YEAR(?constructionDate) < 2003 ||
                                (
                                    YEAR(?constructionDate) = 2003 &&
                                    (
                                        MONTH(?constructionDate) < 9 ||
                                        (
                                            MONTH(?constructionDate) = 9 &&
                                            DAY(?constructionDate) < 1
                                        )
                                    )
                                )
                            )
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:deliveryDate ?deliveryDate .
                        FILTER (datatype(?deliveryDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?assessmentDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (
                            ?method = nltl:traficom2017Section3Point2Point2Method ||
                            ?method = nltl:traficom2017Section3Point2Point4Method
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:constructionStageDate ?constructionDate2 ;
                              nltl:deliveryDate ?deliveryDate2 ;
                              nltl:assessmentDate ?assessmentDate2 .
                        FILTER (
                            datatype(?constructionDate2) = xsd:date &&
                            datatype(?deliveryDate2) = xsd:date &&
                            datatype(?assessmentDate2) = xsd:date &&
                            (
                                YEAR(?constructionDate2) < 2003 ||
                                (
                                    YEAR(?constructionDate2) = 2003 &&
                                    (
                                        MONTH(?constructionDate2) < 9 ||
                                        (
                                            MONTH(?constructionDate2) = 9 &&
                                            DAY(?constructionDate2) < 1
                                        )
                                    )
                                )
                            ) &&
                            (
                                YEAR(?assessmentDate2) > YEAR(?deliveryDate2) + 20 ||
                                (
                                    YEAR(?assessmentDate2) = YEAR(?deliveryDate2) + 20 &&
                                    (
                                        MONTH(?assessmentDate2) > 1 ||
                                        (
                                            MONTH(?assessmentDate2) = 1 &&
                                            DAY(?assessmentDate2) > 1
                                        )
                                    )
                                )
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
