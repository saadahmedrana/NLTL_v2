<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IA or IA Super must have a valid construction-stage date; a date on or after 2003-09-01 is non-applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate .
                    FILTER (datatype(?constructionDate) = xsd:date)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA or IA Super ship must have delivery and assessment dates, use a permitted 2017 compliance method, and comply by 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:constructionStageDate ?constructionDate .
                FILTER (?ice IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (?constructionDate < "2003-09-01"^^xsd:date)

                FILTER (
                    NOT EXISTS {
                        $this nltl:deliveryDate ?deliveryDate .
                        FILTER (datatype(?deliveryDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?assessmentDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (?method IN (
                            nltl:traficom2017Section3Point2Point2Method,
                            nltl:traficom2017Section3Point2Point4Method
                        ))
                    }
                    ||
                    EXISTS {
                        $this nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?deliveryDate) = xsd:date)
                        FILTER (datatype(?assessmentDate) = xsd:date)
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                &&
                                (
                                    MONTH(?assessmentDate) > 1
                                    ||
                                    (
                                        MONTH(?assessmentDate) = 1
                                        &&
                                        DAY(?assessmentDate) > 1
                                    )
                                )
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
