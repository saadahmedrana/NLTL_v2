<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:class nltl:iceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape> ;
        sh:minCount 1 ;
        sh:message "The ship must identify the intended ice-operating waterline data used for the UIWL envelope."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape> ;
        sh:minCount 1 ;
        sh:message "The ship must identify an upper ice waterline with profile points."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must contain exactly the maximum vertical coordinate of the intended ice-operating waterline points at every longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                {
                    SELECT ?x
                           (MAX(?intY) AS ?expectedY)
                           (MAX(?upperY) AS ?upperMaxY)
                           (MIN(?upperY) AS ?upperMinY)
                           (COUNT(?intY) AS ?intCount)
                           (COUNT(?upperY) AS ?upperCount)
                    WHERE {
                        {
                            ?this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
                            ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                            ?intendedPoint nltl:longitudinalPosition ?intendedXValue ;
                                           nltl:verticalCoordinate ?intendedYValue .
                            ?intendedXValue qudt:numericValue ?x ;
                                            qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?intendedYValue qudt:numericValue ?intY ;
                                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                        UNION
                        {
                            ?this nltl:hasUpperIceWaterline ?upperWaterline .
                            ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                            ?upperPoint nltl:longitudinalPosition ?upperXValue ;
                                        nltl:verticalCoordinate ?upperYValue .
                            ?upperXValue qudt:numericValue ?x ;
                                         qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?upperYValue qudt:numericValue ?upperY ;
                                         qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                    }
                    GROUP BY ?x
                }
                FILTER (
                    ?intCount = 0 ||
                    ?upperCount = 0 ||
                    ?expectedY != ?upperMaxY ||
                    ?upperMinY != ?upperMaxY
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-011-IceWaterlineShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node <urn:nltl:generated-shape:TRF-011-ProfilePointShape> ;
        sh:minCount 1 ;
        sh:message "Each ice waterline used by the requirement must contain waterline profile points."
    ] .

<urn:nltl:generated-shape:TRF-011-ProfilePointShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must provide one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must provide one vertical coordinate in metres."
    ] .

<urn:nltl:generated-shape:TRF-011-LengthQuantityShape> a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length quantity must contain one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length quantity must use metres as its unit."
    ] .
<END_SHACL>
