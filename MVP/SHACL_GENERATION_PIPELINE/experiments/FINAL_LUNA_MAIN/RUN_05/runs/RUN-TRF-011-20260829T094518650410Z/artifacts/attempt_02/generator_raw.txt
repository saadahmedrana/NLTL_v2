<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF011WaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF011WaterlineShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must contain, at every longitudinal position, the maximum vertical coordinate of the intended ice-operating waterlines, and both waterlines must cover the same positions." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?this
WHERE {
  {
    SELECT ?this ?x
           (MAX(?ivn) AS ?imax)
           (MAX(?uvn) AS ?umax)
           (COUNT(DISTINCT ?ip) AS ?icount)
           (COUNT(DISTINCT ?up) AS ?ucount)
    WHERE {
      {
        ?this nltl:hasIntendedIceOperatingWaterline ?iw .
        ?iw nltl:hasWaterlineProfilePoint ?ip .
        ?ip nltl:longitudinalPosition ?ixq ;
            nltl:verticalCoordinate ?iyq .
        ?ixq qudt:numericValue ?x ;
             qudt:unit <http://qudt.org/vocab/unit/M> .
        ?iyq qudt:numericValue ?ivn ;
             qudt:unit <http://qudt.org/vocab/unit/M> .
      }
      UNION
      {
        ?this nltl:hasUpperIceWaterline ?uw .
        ?uw nltl:hasWaterlineProfilePoint ?up .
        ?up nltl:longitudinalPosition ?uxq ;
            nltl:verticalCoordinate ?uyq .
        ?uxq qudt:numericValue ?x ;
             qudt:unit <http://qudt.org/vocab/unit/M> .
        ?uyq qudt:numericValue ?uvn ;
             qudt:unit <http://qudt.org/vocab/unit/M> .
      }
    }
    GROUP BY ?this ?x
  }
  FILTER(
    ?icount = 0 ||
    ?ucount = 0 ||
    !BOUND(?imax) ||
    !BOUND(?umax) ||
    ABS(?imax - ?umax) >
      (1e-6 *
       IF(ABS(?imax) > ABS(?umax), ABS(?imax), ABS(?umax))
       + 1e-9)
  )
}
""" ;
    ] .

gs:TRF011WaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gs:TRF011ProfilePointShape
    ] .

gs:TRF011ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF011LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF011LengthQuantityShape
    ] .

gs:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
