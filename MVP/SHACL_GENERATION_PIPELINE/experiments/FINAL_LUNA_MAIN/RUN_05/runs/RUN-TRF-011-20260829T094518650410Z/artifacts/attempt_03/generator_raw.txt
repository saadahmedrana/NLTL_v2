<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node g:TRF011IceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011IceWaterlineShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must be the pointwise envelope of the highest intended ice-operating waterline points at every longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                {
                    SELECT ?this ?x
                           (MAX(?ivn) AS ?imax)
                           (MIN(?uvn) AS ?umin)
                           (MAX(?uvn) AS ?umax)
                           (COUNT(?ivn) AS ?icount)
                           (COUNT(?uvn) AS ?ucount)
                    WHERE {
                        {
                            ?this nltl:hasIntendedIceOperatingWaterline ?iw .
                            ?iw nltl:hasWaterlineProfilePoint ?ip .
                            ?ip nltl:longitudinalPosition ?ipxq .
                            ?ipxq qudt:numericValue ?x .
                            ?ip nltl:verticalCoordinate ?ivq .
                            ?ivq qudt:numericValue ?ivn .
                            OPTIONAL {
                                ?this nltl:hasUpperIceWaterline ?uw .
                                ?uw nltl:hasWaterlineProfilePoint ?up .
                                ?up nltl:longitudinalPosition ?upxq .
                                ?upxq qudt:numericValue ?upx .
                                FILTER (?upx = ?x)
                                ?up nltl:verticalCoordinate ?uvq .
                                ?uvq qudt:numericValue ?uvn .
                            }
                        }
                        UNION
                        {
                            ?this nltl:hasUpperIceWaterline ?uw .
                            ?uw nltl:hasWaterlineProfilePoint ?up .
                            ?up nltl:longitudinalPosition ?upxq .
                            ?upxq qudt:numericValue ?x .
                            ?up nltl:verticalCoordinate ?uvq .
                            ?uvq qudt:numericValue ?uvn .
                            OPTIONAL {
                                ?this nltl:hasIntendedIceOperatingWaterline ?iw .
                                ?iw nltl:hasWaterlineProfilePoint ?ip .
                                ?ip nltl:longitudinalPosition ?ipxq .
                                ?ipxq qudt:numericValue ?ipx .
                                FILTER (?ipx = ?x)
                                ?ip nltl:verticalCoordinate ?ivq .
                                ?ivq qudt:numericValue ?ivn .
                            }
                        }
                    }
                    GROUP BY ?this ?x
                    HAVING (
                        COUNT(?ivn) = 0 ||
                        COUNT(?uvn) = 0 ||
                        ABS(?umin - ?imax) > (0.000001 * ABS(?imax) + 0.000000001) ||
                        ABS(?umax - ?imax) > (0.000001 * ABS(?imax) + 0.000000001)
                    )
                }
            }
        """
    ] .

g:TRF011IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node g:TRF011ProfilePointShape
    ] .

g:TRF011ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape
    ] .

g:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
