<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NonApplicableShipShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1
    ] .

gen:ApplicableShipShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class nltl:iceClassValue
        ] ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class nltl:iceWaterline
        ] ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class nltl:iceWaterline
        ] ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "When the ship is ice-strengthened, each operating draught must be between its corresponding minimum and maximum ice-class draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingForeDraught ?of .
                ?of qudt:numericValue ?ofv .
                $this nltl:maximumIceClassDraughtFore ?maxf .
                ?maxf qudt:numericValue ?maxfv .
                $this nltl:minimumIceClassDraughtFore ?minf .
                ?minf qudt:numericValue ?minfv .
                $this nltl:operatingAftDraught ?oa .
                ?oa qudt:numericValue ?oav .
                $this nltl:maximumIceClassDraughtAft ?maxa .
                ?maxa qudt:numericValue ?maxav .
                $this nltl:minimumIceClassDraughtAft ?mina .
                ?mina qudt:numericValue ?minav .
                FILTER (
                    ?ofv < ?minfv ||
                    ?ofv > ?maxfv ||
                    ?oav < ?minav ||
                    ?oav > ?maxav
                )
            }
        """
    ] .

gen:TRF013ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:NonApplicableShipShape
        gen:ApplicableShipShape
    ) .
<END_SHACL>
