<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:NotIceStrengthenedBranch
        gen:IceStrengthenedBranch
    ) .

gen:NotIceStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:IceStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice class is required when the ship is ice-strengthened."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The upper ice waterline must be identified for an ice-strengthened ship."
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The lower ice waterline must be identified for an ice-strengthened ship."
    ] ;
    sh:property [
        sh:path nltl:draughtAtForePerpendicular ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship draught at the fore perpendicular is required to determine the ice-class draught bounds."
    ] ;
    sh:property [
        sh:path nltl:draughtAtAftPerpendicular ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship draught at the aft perpendicular is required to determine the ice-class draught bounds."
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:nodeKind sh:IRI ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an ice-strengthened ship, each operating draught must be inclusively between its corresponding minimum and maximum ice-class draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipIceStrengthened true .
                $this nltl:minimumIceClassDraughtFore ?minForeNode .
                $this nltl:operatingForeDraught ?opForeNode .
                $this nltl:maximumIceClassDraughtFore ?maxForeNode .
                $this nltl:minimumIceClassDraughtAft ?minAftNode .
                $this nltl:operatingAftDraught ?opAftNode .
                $this nltl:maximumIceClassDraughtAft ?maxAftNode .
                ?minForeNode qudt:numericValue ?minFore .
                ?opForeNode qudt:numericValue ?opFore .
                ?maxForeNode qudt:numericValue ?maxFore .
                ?minAftNode qudt:numericValue ?minAft .
                ?opAftNode qudt:numericValue ?opAft .
                ?maxAftNode qudt:numericValue ?maxAft .
                FILTER (
                    ?minFore > ?opFore ||
                    ?opFore > ?maxFore ||
                    ?minAft > ?opAft ||
                    ?opAft > ?maxAft
                )
            }
        """
    ] .

<END_SHACL>
