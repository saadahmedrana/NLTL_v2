<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SalinityQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:gramPerKilogramSalinityUnit ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF015Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:intendedRouteSeaWaterSalinity ;
        sh:node gen:SalinityQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadingCalculationSeaWaterSalinity ;
        sh:node gen:SalinityQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The loading-calculation sea-water salinity must equal the intended-route sea-water salinity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:intendedRouteSeaWaterSalinity ?intended .
                ?intended qudt:numericValue ?intendedValue .
                $this nltl:loadingCalculationSeaWaterSalinity ?loading .
                ?loading qudt:numericValue ?loadingValue .
                FILTER (?intendedValue != ?loadingValue)
            }
        """
    ] ;
    sh:or [
        rdf:first [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:operatingDraught ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:operatingTrim ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineDraught ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumPermittedIceTrim ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "When navigating in ice, operating draught and trim must not exceed their applicable limits." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:operatingDraught ?draught .
                        ?draught qudt:numericValue ?draughtValue .
                        $this nltl:upperIceWaterlineDraught ?waterline .
                        ?waterline qudt:numericValue ?waterlineValue .
                        $this nltl:operatingTrim ?trim .
                        ?trim qudt:numericValue ?trimValue .
                        $this nltl:maximumPermittedIceTrim ?maximumTrim .
                        ?maximumTrim qudt:numericValue ?maximumTrimValue .
                        FILTER (
                            ?draughtValue > ?waterlineValue ||
                            ?trimValue > ?maximumTrimValue
                        )
                    }
                """
            ]
        ] ;
        rdf:rest [
            rdf:first [
                sh:property [
                    sh:path nltl:navigatingInIce ;
                    sh:datatype xsd:boolean ;
                    sh:hasValue false ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ]
            ] ;
            rdf:rest rdf:nil
        ]
    ] .
<END_SHACL>
