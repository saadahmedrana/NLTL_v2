                        BIND (4 * ?hi AS ?cap)

                        BIND (IF(?expected * 0.000001 > 0.000000001,
                                  ?expected * 0.000001,
                                  0.000000001) AS ?tolerance)
                        FILTER (?forward + ?tolerance < ?expected)
                    }
                """
