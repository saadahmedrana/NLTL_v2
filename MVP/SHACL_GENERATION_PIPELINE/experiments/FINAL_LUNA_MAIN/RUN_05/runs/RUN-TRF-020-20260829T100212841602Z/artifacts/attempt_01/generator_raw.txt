                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:constructionStageDate ?date .
                            $this nltl:iceClass ?class .
                            FILTER (
                                YEAR(?date) > 2003 ||
                                (YEAR(?date) = 2003 && MONTH(?date) > 9) ||
                                (YEAR(?date) = 2003 && MONTH(?date) = 9 && DAY(?date) >= 1)

                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:maximumContinuousRatingPower ?mcr .
                        ?mcr qudt:numericValue ?mcrValue .
                        $this nltl:minimumRequiredPowerAtLowerIceWaterline ?lower .
                        ?lower qudt:numericValue ?lowerValue .
                        $this nltl:minimumRequiredPowerAtUpperIceWaterline ?upper .
                        ?upper qudt:numericValue ?upperValue .
                        FILTER (
                            xsd:decimal(?mcrValue) < xsd:decimal(?lowerValue) ||
                            xsd:decimal(?mcrValue) < xsd:decimal(?upperValue)
