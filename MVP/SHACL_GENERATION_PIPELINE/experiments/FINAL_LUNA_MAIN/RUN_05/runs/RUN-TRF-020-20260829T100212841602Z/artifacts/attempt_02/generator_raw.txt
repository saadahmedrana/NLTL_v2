gen:ApplicableTRF020 a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:node gen:QuantityDimensionless ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceRch ;
        sh:node gen:QuantityN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:node gen:QuantityN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:node gen:QuantityN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:QuantityM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchControlType ;
        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node gen:LowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node gen:UpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:node gen:QuantityKiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:node gen:QuantityKiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:QuantityKiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, constructionStageDate must be on or after 2003-09-01 and iceClass must be IA Super, IA, IB, or IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date ;
                      nltl:iceClass ?class .
                FILTER (
                    !(
                        ?date >= "2003-09-01"^^xsd:date &&
                        ?class IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        )
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be at least both externally calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrNode ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerNode ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline ?upperNode .
                ?mcrNode qudt:numericValue ?mcr .
                ?lowerNode qudt:numericValue ?lower .
                ?upperNode qudt:numericValue ?upper .
                FILTER (?mcr < ?lower || ?mcr < ?upper)
            }
        """
    ] .

    ] ;
    sh:property [

            nltl:iceClassIaSuper

            nltl:iceClassIii

        sh:message "The non-applicable branch requires a constructionStageDate before 2003-09-01." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            WHERE {
                $this nltl:constructionStageDate ?date .
                FILTER (?date >= "2003-09-01"^^xsd:date)
            }
        """
    ] .
