<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRegulationEdition2002 .
                ?this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:nodeKind sh:Literal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "An applicable IB or IC ship must provide exactly one maximum continuous rating power quantity with one numeric value and one kW unit."
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:nodeKind sh:Literal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "An applicable IB or IC ship must provide exactly one required power quantity with one numeric value and one kW unit."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?maximumPower .
                ?maximumPower qudt:numericValue ?maximumValue ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .

                $this nltl:requiredPowerUnder1985Rules ?requiredPower .
                ?requiredPower qudt:numericValue ?requiredValue ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .

                FILTER (
                    !isNumeric(?maximumValue) ||
                    !isNumeric(?requiredValue) ||
                    ?maximumValue < ?requiredValue
                )
            }
        """ ;
        sh:message "Maximum continuous rating power must be numeric and at least the required power under the 1985 rules."
    ] .
<END_SHACL>
