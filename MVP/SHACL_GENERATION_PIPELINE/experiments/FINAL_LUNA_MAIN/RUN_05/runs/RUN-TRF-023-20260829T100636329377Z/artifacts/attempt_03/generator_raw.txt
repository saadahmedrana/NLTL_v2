@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF023Shape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRuleEdition1985 .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
            }
        """ ;
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW>
            ]
        ] ;
        sh:message "An applicable ship must provide exactly one maximum continuous rating power quantity with one numeric value and exactly one kW unit."
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW>
            ]
        ] ;
        sh:message "An applicable ship must provide exactly one required power quantity with one numeric value and exactly one kW unit."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, maximum continuous rating power must be at least the required power under the 1985 rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:maximumContinuousRatingPower ?maximumQuantity ;
                      nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                ?maximumQuantity qudt:numericValue ?maximumPower .
                ?requiredQuantity qudt:numericValue ?requiredPower .
                FILTER ( ?maximumPower >= ?requiredPower )
            }
        """
    ] .
