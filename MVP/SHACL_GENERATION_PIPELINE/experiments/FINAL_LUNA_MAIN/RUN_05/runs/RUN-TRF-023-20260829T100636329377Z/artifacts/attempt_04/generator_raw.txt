<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:PowerQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:nodeKind sh:Literal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single numeric value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use kW as its unit."
    ] .

gen:ApplicableShipShape a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRuleEdition1985 .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ships require exactly one maximum continuous rating power in kW."
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ships require exactly one required power under the 1985 rules in kW."
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be at least the required power under the 1985 rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?maximumQuantity ;
                      nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                ?maximumQuantity qudt:numericValue ?maximumPower .
                ?requiredQuantity qudt:numericValue ?requiredPower .
                FILTER (?maximumPower < ?requiredPower)
            }
        """
    ] .
<END_SHACL>
