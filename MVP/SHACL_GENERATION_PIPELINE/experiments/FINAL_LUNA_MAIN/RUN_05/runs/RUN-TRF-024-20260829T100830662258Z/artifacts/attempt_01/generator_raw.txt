<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-024> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice class to determine applicability of the twenty-year requirement."
    ] ;
    sh:sparql [
        sh:message "For ice class IA Super or IA, construction stage date is required to determine whether the conditional requirement applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For ice class IA Super or IA, delivery date is required to determine the twenty-year deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For ice class IA Super or IA, an assessment date is required to verify compliance by the applicable deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:assessmentDate ?assessmentDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ship retaining ice class IA Super or IA with construction before 1 September 2003 must comply by 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice .
                FILTER (?ice IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate ;
                          nltl:deliveryDate ?deliveryDate ;
                          nltl:assessmentDate ?assessmentDate .
                    FILTER (?constructionDate < "2003-09-01"^^xsd:date)
                    FILTER (
                        YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                        ||
                        (
                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                            &&
                            (
                                MONTH(?assessmentDate) > 1
                                ||
                                (MONTH(?assessmentDate) = 1 && DAY(?assessmentDate) > 1)
                            )
                        )
                    )
                }
            }
        """
    ] .
<END_SHACL>
