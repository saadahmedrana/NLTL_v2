@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-024-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:message "An ice class value must be an IRI classified as an ice-class value."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "The construction stage date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:message "The delivery date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:message "The assessment date must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA Super or IA applicable to this rule, constructionStageDate is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionStageDate .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, deliveryDate is required to determine the twenty-year deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))
                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, assessmentDate is required to evaluate the deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))
                FILTER NOT EXISTS {
                    $this nltl:assessmentDate ?assessmentDate .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship constructed before 1 September 2003, the assessment date must not be later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .

                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))

                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        MONTH(?constructionStageDate) < 9
                    )
                )

                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        MONTH(?assessmentDate) > 1
                    ) ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        MONTH(?assessmentDate) = 1 &&
                        DAY(?assessmentDate) > 1
                    )
                )
            }
        """
    ] .
