gen:TRF025ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "Exactly one permitted ice class is required."
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one bulbous-bow presence value is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:node gen:TRF025CoefficientF1Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:node gen:TRF025CoefficientF2Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:node gen:TRF025CoefficientF3Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:node gen:TRF025CoefficientF4Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:node gen:TRF025CoefficientG1Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:node gen:TRF025CoefficientG2Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:node gen:TRF025CoefficientG3Shape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:node gen:TRF025PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:node gen:TRF025PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:node gen:TRF025PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:node gen:TRF025NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "C1 must be supplied as a Newton quantity value."
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:node gen:TRF025NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "C2 must be supplied as a Newton quantity value."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
    ) .
