<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF025ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceClassValue ;
        sh:message "Exactly one ice class is required before applicability can be determined."
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Exactly one bulbous-bow presence value is required before applicability can be determined."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025F1QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025F2QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025F3QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025F4QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025G1QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025G2QuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025G3QuantityShape
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025BreadthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025DraughtQuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025C1QuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF025C2QuantityShape
            ] ;
            sh:sparql [
                sh:message "For an IA Super ship without a bulb, C1 and C2 must match their declared formula results within relative tolerance 1e-6 with an absolute fallback of 1e-6." ;
                sh:select """PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass nltl:iceClassIaSuper ;
        nltl:bulbousBowPresent false ;
        nltl:coefficientF1 ?f1n ;
        nltl:coefficientF2 ?f2n ;
        nltl:coefficientF3 ?f3n ;
        nltl:coefficientF4 ?f4n ;
        nltl:coefficientG1 ?g1n ;
        nltl:coefficientG2 ?g2n ;
        nltl:coefficientG3 ?g3n ;
        nltl:shipBreadth ?bn ;
        nltl:shipLength ?ln ;
        nltl:iceClassDraught ?tn ;
        nltl:brashIceResistanceCoefficientC1 ?c1n ;
        nltl:brashIceResistanceCoefficientC2 ?c2n .
    ?f1n qudt:numericValue ?f1 .
    ?f2n qudt:numericValue ?f2 .
    ?f3n qudt:numericValue ?f3 .
    ?f4n qudt:numericValue ?f4 .
    ?g1n qudt:numericValue ?g1 .
    ?g2n qudt:numericValue ?g2 .
    ?g3n qudt:numericValue ?g3 .
    ?bn qudt:numericValue ?b .
    ?ln qudt:numericValue ?l .
    ?tn qudt:numericValue ?t .
    ?c1n qudt:numericValue ?c1 .
    ?c2n qudt:numericValue ?c2 .
    FILTER(
        ?b > 0 &&
        ?l > 0 &&
        ?t > 0 &&
        (
            ABS(
                ?c1 -
                (
                    ?f1 * (?b * ?l) / (2 * ?t / ?b + 1) +
                    1.84 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                )
            ) >
            0.000001 *
            IF(
                ABS(
                    ?f1 * (?b * ?l) / (2 * ?t / ?b + 1) +
                    1.84 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                ) > 1,
                ABS(
                    ?f1 * (?b * ?l) / (2 * ?t / ?b + 1) +
                    1.84 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                ),
                1
            )
        ) ||
        ABS(
            ?c2 -
            (
                3.52 * (?g1 + ?g2 * ?b) +
                ?g3 * (1 + 1.2 * ?t / ?b) * ?b * ?b /
                POWER(?l, 0.5)
            )
        ) >
        0.000001 *
        IF(
            ABS(
                3.52 * (?g1 + ?g2 * ?b) +
                ?g3 * (1 + 1.2 * ?t / ?b) * ?b * ?b /
                POWER(?l, 0.5)
            ) > 1,
            ABS(
                3.52 * (?g1 + ?g2 * ?b) +
                ?g3 * (1 + 1.2 * ?t / ?b) * ?b * ?b /
                POWER(?l, 0.5)
            ),
            1
        )
    )
}"""
            ]
        ]
    ) .
