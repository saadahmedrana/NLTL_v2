        sh:datatype xsd:boolean ;

        sh:maxCount 1

    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:minCount 1 ;
        sh:node gen:TRF030DirectAnalysisCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:node gen:TRF030LoadLengthCaseShape
            ] ;
            sh:sparql [
                sh:message "When load length is not determined from the arrangement, several distinct direct-analysis cases with paired load length and area-factor values are required." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:hasDirectAnalysisCase ?case1 .
                        ?this nltl:hasDirectAnalysisCase ?case2 .
                        FILTER (?case1 != ?case2)
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The direct analyses must include distinct cases at the UIWL, 0.5h0 below the LIWL, and intermediate vertical positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this nltl:hasDirectAnalysisCase ?upperCase .
                ?upperCase nltl:verticalLoadPositionType nltl:upperIceWaterlineCenteredPosition .
                ?this nltl:hasDirectAnalysisCase ?halfCase .
                ?halfCase nltl:verticalLoadPositionType nltl:halfLoadHeightBelowLowerIceWaterlinePosition .
                ?this nltl:hasDirectAnalysisCase ?intermediateCase .
                ?intermediateCase nltl:verticalLoadPositionType nltl:intermediateVerticalPosition .
                FILTER (?upperCase != ?halfCase)
                FILTER (?upperCase != ?intermediateCase)
                FILTER (?halfCase != ?intermediateCase)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The direct analyses must include several distinct horizontal locations, including a mid-span or spacing-centred location." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this nltl:hasDirectAnalysisCase ?centredCase .
                ?centredCase nltl:horizontalLoadPositionType nltl:midSpanOrSpacingCenteredPosition .
                ?this nltl:hasDirectAnalysisCase ?otherCase .
                ?otherCase nltl:horizontalLoadPositionType nltl:otherHorizontalPosition .
                FILTER (?centredCase != ?otherCase)
            }
        """
    ] .
