<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF031Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide its yield point in MPa."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide combined bending and shear stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide yield shear stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must state whether the direct calculation is based on beam theory."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:datatype qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Beam-theory calculations must provide allowable shear stress in MPa."
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be lower than the yield point, and beam-theory allowable shear stress must not exceed 0.9 times yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldPoint ;
                      nltl:combinedBendingShearStress ?combinedStress ;
                      nltl:yieldShearStress ?yieldShearStress ;
                      nltl:beamTheoryUsed ?beamTheoryUsed .

                ?yieldPoint qudt:numericValue ?yieldPointValue .
                ?combinedStress qudt:numericValue ?combinedStressValue .
                ?yieldShearStress qudt:numericValue ?yieldShearStressValue .

                FILTER (
                    datatype(?yieldPointValue) = xsd:decimal &&
                    datatype(?combinedStressValue) = xsd:decimal &&
                    datatype(?yieldShearStressValue) = xsd:decimal
                )

                FILTER (
                    ?combinedStressValue >= ?yieldPointValue ||
                    ABS(?yieldShearStressValue - (?yieldPointValue / 1.7320508075688772)) >
                    (0.000001 * ABS(?yieldPointValue / 1.7320508075688772) + 0.000000000001)
                )

                {
                    FILTER (?beamTheoryUsed = true)
                    $this nltl:allowableShearStress ?allowableStress .
                    ?allowableStress qudt:numericValue ?allowableStressValue .
                    FILTER (
                        ?allowableStressValue >
                        (0.9 * ?yieldShearStressValue)
                    )
                }
                UNION
                {
                    FILTER (?beamTheoryUsed = false)
                }
            }
        """
    ] .
<END_SHACL>
