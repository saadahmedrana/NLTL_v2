<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be lower than the yield point." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldPoint ?yieldNode .
                ?combinedNode qudt:numericValue ?combined .
                ?yieldNode qudt:numericValue ?yield .
                FILTER (?combined >= ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Yield shear stress must equal yield point divided by square root of three within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldNode ;
                      nltl:yieldShearStress ?shearNode .
                ?yieldNode qudt:numericValue ?yield .
                ?shearNode qudt:numericValue ?shear .
                BIND (?yield / 1.7320508075688772 AS ?expected)
                FILTER (
                    ABS(?shear - ?expected) >
                    (ABS(?expected) * 0.000001 + 0.000000001)
                )
            }
        """
    ] ;
    sh:or (
        gen:TRF031NonBeamCase
        gen:TRF031BeamCase
    ) .

gen:TRF031NonBeamCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:TRF031BeamCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:allowableShearStress ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must be no larger than 0.9 times the yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:allowableShearStress ?allowableNode ;
                      nltl:yieldPoint ?yieldNode .
                ?allowableNode qudt:numericValue ?allowable .
                ?yieldNode qudt:numericValue ?yield .
                FILTER (?allowable > (0.9 * (?yield / 1.7320508075688772)))
            }
        """
    ] .
<END_SHACL>
