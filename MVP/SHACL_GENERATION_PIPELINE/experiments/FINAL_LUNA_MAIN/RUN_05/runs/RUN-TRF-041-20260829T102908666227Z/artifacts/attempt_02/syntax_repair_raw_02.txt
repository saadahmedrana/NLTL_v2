    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "Corrosion and abrasion addition is required in millimetres."
    ] ;
    sh:property [

        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;

    sh:property [

    sh:or (

                sh:node gs:PressureQuantityShape ;
                sh:minCount 1
            ] ;
