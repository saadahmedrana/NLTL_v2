<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-043-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:framingIceStrengtheningTerminationCase
    ] .

<urn:nltl:generated-shape:TRF-043-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing ice-strengthening extents do not satisfy the applicable Table 4-6 ice-class and hull-region thresholds." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:iceClass ?iceClass ;
                          nltl:hullRegion ?region ;
                          nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                          nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .

                    ?above qudt:numericValue ?aboveValue ;
                           qudt:unit unit:M .
                    ?below qudt:numericValue ?belowValue ;
                           qudt:unit unit:M .

                    FILTER (
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            xsd:decimal(?aboveValue) >= 1.2 &&
                            (
                                (
                                    ?region = nltl:bowRegion &&
                                    xsd:decimal(?belowValue) >= 2.0 &&
                                    EXISTS {
                                        $this nltl:reachesTankTopOrBelowFloorTop true
                                    }
                                ) ||
                                (
                                    ?region = nltl:midbodyRegion &&
                                    xsd:decimal(?belowValue) >= 2.0
                                ) ||
                                (
                                    ?region = nltl:sternRegion &&
                                    xsd:decimal(?belowValue) >= 1.6
                                )
                            )
                        ) ||
                        (
                            (
                                ?iceClass = nltl:iceClassIa ||
                                ?iceClass = nltl:iceClassIb ||
                                ?iceClass = nltl:iceClassIc
                            ) &&
                            xsd:decimal(?aboveValue) >= 1.0 &&
                            (
                                (
                                    ?region = nltl:bowRegion &&
                                    xsd:decimal(?belowValue) >= 1.6 &&
                                    EXISTS {
                                        $this nltl:reachesTankTopOrBelowFloorTop true
                                    }
                                ) ||
                                (
                                    ?region = nltl:midbodyRegion &&
                                    xsd:decimal(?belowValue) >= 1.3
                                ) ||
                                (
                                    ?region = nltl:sternRegion &&
                                    xsd:decimal(?belowValue) >= 1.0
                                )
                            )
                        )
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:minCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable framing-strengthening termination case must permit termination at its adjacent boundary when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .
                ?extension qudt:numericValue ?value ;
                           qudt:unit unit:MilliM .
                FILTER (xsd:decimal(?value) <= 250)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true
                }
            }
        """
    ] .
<END_SHACL>
