<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

<urn:nltl:generated-shape:TRF043ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase
    ] .

<urn:nltl:generated-shape:TRF043FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The framing strengthening above UIWL is below the Table 4-6 value for the applicable ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ic ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?q .
                ?q qudt:numericValue ?v .
                FILTER (
                    (?ic = nltl:iceClassIaSuper && ?v < 1.2) ||
                    ((?ic = nltl:iceClassIa ||
                      ?ic = nltl:iceClassIb ||
                      ?ic = nltl:iceClassIc) && ?v < 1.0)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The framing strengthening below LIWL is below the Table 4-6 value for the applicable ice class and hull region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ic ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?q .
                ?q qudt:numericValue ?v .
                FILTER (
                    (?ic = nltl:iceClassIaSuper &&
                     ?region = nltl:bowRegion && ?v < 2.0) ||
                    (?ic = nltl:iceClassIaSuper &&
                     ?region = nltl:midbodyRegion && ?v < 2.0) ||
                    (?ic = nltl:iceClassIaSuper &&
                     ?region = nltl:sternRegion && ?v < 1.6) ||
                    ((?ic = nltl:iceClassIa ||
                      ?ic = nltl:iceClassIb ||
                      ?ic = nltl:iceClassIc) &&
                     ?region = nltl:bowRegion && ?v < 1.6) ||
                    ((?ic = nltl:iceClassIa ||
                      ?ic = nltl:iceClassIb ||
                      ?ic = nltl:iceClassIc) &&
                     ?region = nltl:midbodyRegion && ?v < 1.3) ||
                    ((?ic = nltl:iceClassIa ||
                      ?ic = nltl:iceClassIb ||
                      ?ic = nltl:iceClassIc) &&
                     ?region = nltl:sternRegion && ?v < 1.0)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "An IA Super bow frame reaching the tank top or below the floor top must record true evidence for that extent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:hullRegion nltl:bowRegion .
                FILTER NOT EXISTS {
                    $this nltl:reachesTankTopOrBelowFloorTop true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "When an upper bow ice belt is required, framing must reach its top." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperBowIceBeltRequired true .
                FILTER NOT EXISTS {
                    $this nltl:reachesUpperBowIceBeltTop true .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF043TerminationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:or (
            [
                sh:class nltl:deckStructure
            ]
            [
                sh:class nltl:tankBoundaryPlating
            ]
            [
                sh:class nltl:tankTop
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For an extension of 250 mm or less, termination at the adjacent boundary must be permitted and the boundary must be a deck, tank boundary plating, or tank top." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?q .
                ?q qudt:numericValue ?extension .
                FILTER (?extension <= 250)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
                }
                FILTER NOT EXISTS {
                    $this nltl:terminationAdjacentBoundary ?boundary .
                    ?boundary rdf:type ?boundaryType .
                    FILTER (
                        ?boundaryType = nltl:deckStructure ||
                        ?boundaryType = nltl:tankBoundaryPlating ||
                        ?boundaryType = nltl:tankTop
                    )
                }
            }
        """
    ] .
<END_SHACL>
