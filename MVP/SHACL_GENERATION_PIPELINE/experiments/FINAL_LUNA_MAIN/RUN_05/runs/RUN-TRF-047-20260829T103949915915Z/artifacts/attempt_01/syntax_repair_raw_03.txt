<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:message "A frame shall identify its strengthened part."
    ] .

<urn:nltl:generated-shape:TRF-047-TransverseFrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:node [
            a sh:NodeShape ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:hasAttachedSupportingStructure ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:or (
                                [ sh:class nltl:deckStructure ]
                                [ sh:class nltl:tankBoundaryPlating ]
                                [ sh:class nltl:tankTop ]
                                [ sh:class nltl:iceStringer ]
                            )
                        ] ;
                        sh:message "The lower end shall be attached to a deck, tank boundary plating, tank top, or ice stringer."
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasTerminationBelowSupportingStructure ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:supportingStructure
                        ] ;
                        sh:message "The conditional connection shall identify the supporting structure below which the frame terminates."
                    ] ;
                    sh:property [
                        sh:path nltl:hasHorizontalConnectionMember ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:horizontalConnectionMember ;
                            sh:property [
                                sh:path nltl:connectsToAdjacentMainFrame ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:class nltl:mainFrame
                                ] ;
                                sh:message "The horizontal member shall connect to an adjacent main frame."
                            ]
                        ] ;
                        sh:message "The conditional connection shall identify a horizontal connection member."
                    ]
                ]
            )
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a lower end terminates below supporting structure, the supporting structure shall be at or below the lower limit of the ice belt, the main frame below the ice belt shall be strengthened, and the horizontal member shall have the same scantlings as the main frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasLowerEnd ?lowerEnd .
                ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?supportingStructure .
                FILTER (
                    NOT EXISTS {
                        $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:mainFrameBelowIceBeltStrengthened true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:sameScantlingsAsMainFrame true .
                    }
                )
            }
        """
    ] .

</END_SHACL>
