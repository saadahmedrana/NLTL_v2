<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF049FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node gen:TRF049FrameAttachmentShape
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path rdf:type ;
                        sh:hasValue nltl:longitudinalFrame
                    ] ;
                    sh:property [
                        sh:path nltl:hasSupportingWebFrame ;
                        sh:minCount 1 ;
                        sh:class nltl:supportingStructure ;
                        sh:node gen:TRF049SupportingStructureShape
                    ] ;
                    sh:property [
                        sh:path nltl:hasSupportingBulkhead ;
                        sh:minCount 1 ;
                        sh:class nltl:supportingStructure ;
                        sh:node gen:TRF049SupportingStructureShape
                    ]
                ]
                [
                    sh:property [
                        sh:path rdf:type ;
                        sh:hasValue nltl:transverseFrame
                    ] ;
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1
                    ] ;
                    sh:or (
                        [
                            sh:property [
                                sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket ) ;
                                sh:minCount 1 ;
                                sh:class nltl:connectionBracket ;
                                sh:node gen:TRF049ConnectionBracketShape
                            ]
                        ]
                        [
                            sh:property [
                                sh:path ( nltl:hasFrameAttachment nltl:similarAttachmentConstructionPresent ) ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:hasValue true
                            ]
                        ]
                    )
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each installed bracket must be at least as thick as the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

gen:TRF049FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure ;
        sh:node gen:TRF049SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:TRF049ConnectionBracketShape
    ] .

gen:TRF049SupportingStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachment
    ] .

gen:TRF049ConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:node gen:TRF049MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] .

gen:TRF049MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
