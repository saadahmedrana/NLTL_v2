<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:FrameAntitrippingSupportShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:equivalentSupportStatusByDirectCalculation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:and (
                [
                    sh:or (
                        [
                            sh:property [
                                sh:path nltl:frameAsymmetrical ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue true
                            ] ;
                            sh:property [
                                sh:path nltl:antitrippingSupportSpacing ;
                                sh:node gen:AntitrippingSpacingQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:sparql [
                                sh:message "Antitripping support spacing must not exceed 1300 mm when the frame is asymmetrical." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:antitrippingSupportSpacing ?q .
                                        ?q qudt:numericValue ?v .
                                        FILTER ( ?v > "1300"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameAsymmetrical ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue false
                            ] ;
                            sh:property [
                                sh:path nltl:frameWebAngleToShell ;
                                sh:node gen:AngleQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:antitrippingSupportSpacing ;
                                sh:node gen:AntitrippingSpacingQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:sparql [
                                sh:message "Antitripping support spacing must not exceed 1300 mm when the frame web angle to the shell is below 90 degrees." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameWebAngleToShell ?a .
                                        ?a qudt:numericValue ?av .
                                        $this nltl:antitrippingSupportSpacing ?q .
                                        ?q qudt:numericValue ?v .
                                        FILTER ( ?av < "90"^^xsd:decimal && ?v > "1300"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameAsymmetrical ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue false
                            ] ;
                            sh:property [
                                sh:path nltl:frameWebAngleToShell ;
                                sh:node gen:AngleQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:sparql [
                                sh:message "A frame that is not asymmetrical and has a web angle of at least 90 degrees is outside the conditional spacing rule." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameWebAngleToShell ?a .
                                        ?a qudt:numericValue ?av .
                                        FILTER ( ?av < "90"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                    )
                ]
                [
                    sh:or (
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:sparql [
                                sh:message "For frame spans greater than 4 m, the applicable all-regions and all-ice-classes scope must be represented by the approved vocabulary." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameSpan ?s .
                                        ?s qudt:numericValue ?v .
                                        FILTER ( ?v <= "4"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIaSuper
                            ] ;
                            sh:sparql [
                                sh:message "The frame span must be no greater than 4 m for the IA Super branch." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameSpan ?s .
                                        ?s qudt:numericValue ?v .
                                        FILTER ( ?v > "4"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIa
                            ] ;
                            sh:property [
                                sh:path nltl:hasAntitrippingSupportRegion ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:bowRegion
                            ] ;
                            sh:property [
                                sh:path nltl:hasAntitrippingSupportRegion ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:midbodyRegion
                            ] ;
                            sh:sparql [
                                sh:message "The frame span must be no greater than 4 m for the IA regional-scope branch." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameSpan ?s .
                                        ?s qudt:numericValue ?v .
                                        FILTER ( ?v > "4"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIb
                            ] ;
                            sh:property [
                                sh:path nltl:hasAntitrippingSupportRegion ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:bowRegion
                            ] ;
                            sh:sparql [
                                sh:message "The frame span must be no greater than 4 m for the IB regional-scope branch." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameSpan ?s .
                                        ?s qudt:numericValue ?v .
                                        FILTER ( ?v > "4"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIc
                            ] ;
                            sh:property [
                                sh:path nltl:hasAntitrippingSupportRegion ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:bowRegion
                            ] ;
                            sh:sparql [
                                sh:message "The frame span must be no greater than 4 m for the IC regional-scope branch." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:frameSpan ?s .
                                        ?s qudt:numericValue ?v .
                                        FILTER ( ?v > "4"^^xsd:decimal )
                                    }
                                """
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIi
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:frameSpan ;
                                sh:node gen:SpanQuantityShape ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path nltl:iceClass ;
                                sh:minCount 1 ;
                                sh:hasValue nltl:iceClassIii
                            ]
                        ]
                    )
                )
            )
        ]
    ) .

gen:AntitrippingSpacingQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:SpanQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:AngleQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG>
    ] .
<END_SHACL>
