<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DecimalQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMetreQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureMegaPascalQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LineLoadQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AreaQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM2> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SectionModulusQuantityShape
    a sh:NodeShape ;
    sh:node gen:DecimalQuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM3> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:or (
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:icePressure ;
                sh:node gen:PressureMegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designIceLoadHeight ;
                sh:node gen:LengthMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:span ;
                sh:node gen:LengthMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:PressureMegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToIceBelt ;
                sh:node gen:LengthMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToAdjacentIceStringer ;
                sh:node gen:LengthMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designLineLoad ;
                sh:node gen:LineLoadQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:outsideIceBeltFactor ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:sectionModulus ;
                sh:node gen:SectionModulusQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredShearArea ;
                sh:node gen:AreaQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The derived design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy TRF-054." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:icePressure ?pNode ;
                              nltl:designIceLoadHeight ?hNode ;
                              nltl:span ?lNode ;
                              nltl:frameMomentFactorM ?mNode ;
                              nltl:yieldStrength ?yNode ;
                              nltl:distanceToIceBelt ?hsNode ;
                              nltl:distanceToAdjacentIceStringer ?lsNode ;
                              nltl:designLineLoad ?qNode ;
                              nltl:outsideIceBeltFactor ?fNode ;
                              nltl:sectionModulus ?zNode ;
                              nltl:requiredShearArea ?aNode .
                        ?pNode qudt:numericValue ?p .
                        ?hNode qudt:numericValue ?h .
                        ?lNode qudt:numericValue ?l .
                        ?mNode qudt:numericValue ?m .
                        ?yNode qudt:numericValue ?y .
                        ?hsNode qudt:numericValue ?hs .
                        ?lsNode qudt:numericValue ?ls .
                        ?qNode qudt:numericValue ?q .
                        ?fNode qudt:numericValue ?f .
                        ?zNode qudt:numericValue ?z .
                        ?aNode qudt:numericValue ?a .
                        BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?qMn)
                        BIND((?qMn * 1000) AS ?expectedQ)
                        BIND((1 - (?hs / ?ls)) AS ?expectedF)
                        BIND((0.80 * 1.8 * ?qMn * ?l * ?l / (?m * ?y) * ?expectedF * 1000000) AS ?expectedZ)
                        BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?qMn * ?l / (2 * ?y) * ?expectedF * 10000) AS ?expectedA)
                        FILTER (
                            ABS(?q - ?expectedQ) > (1e-6 * IF(ABS(?expectedQ) > 1e-9, ABS(?expectedQ), 1e-9))
                            ||
                            ABS(?f - ?expectedF) > (1e-6 * IF(ABS(?expectedF) > 1e-9, ABS(?expectedF), 1e-9))
                            ||
                            ABS(?z - ?expectedZ) > (1e-6 * IF(ABS(?expectedZ) > 1e-9, ABS(?expectedZ), 1e-9))
                            ||
                            ABS(?a - ?expectedA) > (1e-6 * IF(ABS(?expectedA) > 1e-9, ABS(?expectedA), 1e-9))
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ] .
<END_SHACL>
