<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF059WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:maximumCalculatedShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameShearFactorAlpha ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:freeFlangeToWebAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:CentiM2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "requiredShearArea must equal the calculated effective shear area, with alpha selected consistently from Table 4-8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumCalculatedShearForce ?qNode ;
                      nltl:webFrameShearFactorAlpha ?alphaNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:freeFlangeToWebAreaRatio ?ratioNode ;
                      nltl:requiredShearArea ?areaNode .

                ?qNode qudt:numericValue ?q .
                ?alphaNode qudt:numericValue ?alpha .
                ?sigmaNode qudt:numericValue ?sigma .
                ?ratioNode qudt:numericValue ?ratio .
                ?areaNode qudt:numericValue ?area .

                FILTER (?sigma > 0)
                FILTER (
                    NOT (
                        (?ratio = 0.0 && ?alpha = 1.5) ||
                        (?ratio = 0.2 && ?alpha = 1.23) ||
                        (?ratio = 0.4 && ?alpha = 1.16) ||
                        (?ratio = 0.6 && ?alpha = 1.11) ||
                        (?ratio = 0.8 && ?alpha = 1.09) ||
                        (?ratio = 1.0 && ?alpha = 1.07) ||
                        (?ratio = 1.2 && ?alpha = 1.06) ||
                        (?ratio = 1.4 && ?alpha = 1.05) ||
                        (?ratio = 1.6 && ?alpha = 1.05) ||
                        (?ratio = 1.8 && ?alpha = 1.04) ||
                        (?ratio = 2.0 && ?alpha = 1.04)
                    )
                )

                BIND (1.7320508075688772 * ?alpha * 1.1 * (?q / 1000.0) / ?sigma * 10000.0 AS ?expected)
                BIND (1e-6 * ABS(?expected) AS ?relativeTolerance)
                BIND (IF(?relativeTolerance > 1e-9, ?relativeTolerance, 1e-9) AS ?tolerance)

                FILTER (ABS(?area - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
