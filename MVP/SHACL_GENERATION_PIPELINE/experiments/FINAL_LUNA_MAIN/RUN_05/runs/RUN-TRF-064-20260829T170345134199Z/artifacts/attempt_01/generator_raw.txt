<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:StrutDesignEvidenceShape
    a sh:NodeShape ;
    sh:class nltl:strutDesignEvidence .

gen:StrutHullAttachmentEvidenceShape
    a sh:NodeShape ;
    sh:class nltl:strutHullAttachmentEvidence .

gen:StrutStrengthEvidenceShape
    a sh:NodeShape ;
    sh:class nltl:strutStrengthEvidence .

gen:DetachedStrutShape
    a sh:NodeShape ;
    sh:class nltl:detachedStrutStructure ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:minCount 1 ;
        sh:node gen:StrutDesignEvidenceShape ;
        sh:message "A detached strut shall have design evidence."
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:node gen:StrutHullAttachmentEvidenceShape ;
        sh:message "A detached strut shall have hull-attachment evidence."
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:minCount 1 ;
        sh:node gen:StrutStrengthEvidenceShape ;
        sh:message "A detached strut shall have strength evidence."
    ] .

gen:PlatedBossingShape
    a sh:NodeShape ;
    sh:class nltl:platedBossingStructure .

gen:SidePropellerShaftingEnclosedShape
    a sh:NodeShape ;
    sh:class nltl:sidePropellerShaftingAssembly ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:node gen:PlatedBossingShape ;
        sh:message "Side propeller shafting shall be enclosed within plated bossing."
    ] .

gen:SternTubeEnclosedShape
    a sh:NodeShape ;
    sh:class nltl:sternTube ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:node gen:PlatedBossingShape ;
        sh:message "The stern tube shall be enclosed within plated bossing."
    ] .

gen:SidePropellerShaftingShape
    a sh:NodeShape ;
    sh:class nltl:sidePropellerShaftingAssembly .

gen:SternTubeShape
    a sh:NodeShape ;
    sh:class nltl:sternTube .

gen:TRF064ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerShafting ;
                sh:minCount 1 ;
                sh:node gen:SidePropellerShaftingEnclosedShape
            ] ;
            sh:property [
                sh:path nltl:hasSternTube ;
                sh:minCount 1 ;
                sh:node gen:SternTubeEnclosedShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerShafting ;
                sh:minCount 1 ;
                sh:node gen:SidePropellerShaftingShape
            ] ;
            sh:property [
                sh:path nltl:hasSternTube ;
                sh:minCount 1 ;
                sh:node gen:SternTubeShape
            ] ;
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:minCount 1 ;
                sh:node gen:DetachedStrutShape ;
                sh:message "A detached strut shall have design, strength, and hull-attachment evidence."
            ]
        ]
    ) .

</END_SHACL>
