@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF064Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasSidePropellerShafting ;
        sh:class nltl:sidePropellerShaftingAssembly ;
        sh:node gen:TRF064EnclosedComponentShape
    ] ;
    sh:property [
        sh:path nltl:hasSternTube ;
        sh:class nltl:sternTube ;
        sh:node gen:TRF064EnclosedComponentShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:detachedStrut ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:minCount 1 ;
                sh:class nltl:detachedStrutStructure ;
                sh:node gen:TRF064DetachedStrutShape
            ]
        ]
    ) ;
    sh:message "The ship must consistently indicate whether detached struts are used and satisfy the applicable enclosure or detached-strut evidence requirements." .

gen:TRF064EnclosedComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:class nltl:platedBossingStructure ;
        sh:message "The side-propeller shafting or stern tube must be enclosed within plated bossing."
    ] .

gen:TRF064DetachedStrutShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutDesignEvidence ;
        sh:node gen:TRF064DocumentEvidenceShape ;
        sh:message "Detached struts require design evidence."
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutStrengthEvidence ;
        sh:node gen:TRF064DocumentEvidenceShape ;
        sh:message "Detached struts require strength evidence."
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutHullAttachmentEvidence ;
        sh:node gen:TRF064DocumentEvidenceShape ;
        sh:message "Detached struts require hull-attachment evidence."
    ] .

gen:TRF064DocumentEvidenceShape
    a sh:NodeShape ;
    sh:class nltl:documentEvidence .
