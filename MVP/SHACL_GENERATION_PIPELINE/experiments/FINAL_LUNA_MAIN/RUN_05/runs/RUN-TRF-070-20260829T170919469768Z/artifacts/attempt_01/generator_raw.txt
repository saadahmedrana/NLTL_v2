gs:TRF070ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF070QuantityForceKiloNShape
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF070QuantityForceKiloNShape
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node gs:TRF070QuantityPressureMegaPAShape
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:designCondition ;
            sh:property [
                sh:path nltl:designConditionLocalStrengthCapacity ;
                sh:minCount 1 ;
                sh:node gs:TRF070QuantityForceKiloNShape
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The externally assessed local-strength capacity shall be at least the applicable local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:localIcePressure ?pressure .
                ?pressure qudt:numericValue ?pressureValue .
                $this nltl:hasDesignCondition ?condition .
                ?condition nltl:designConditionLocalStrengthCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .
