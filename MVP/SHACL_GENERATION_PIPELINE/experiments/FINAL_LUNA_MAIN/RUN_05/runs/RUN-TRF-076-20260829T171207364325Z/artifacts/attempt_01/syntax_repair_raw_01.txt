<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:PropellerBladeLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Each linked propeller blade load case must have an integer load case number."
    ] .

gen:LoadCase1Shape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 1 ;
        sh:maxInclusive 1
    ] .

gen:LoadCase2Shape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2
    ] .

gen:LoadCase3Shape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 3 ;
        sh:maxInclusive 3
    ] .

gen:LoadCase4Shape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4
    ] .

gen:LoadCase5Shape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 5 ;
        sh:maxInclusive 5
    ] .

gen:Cases1To4Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:PropellerBladeLoadCaseShape ;
        sh:qualifiedValueShape gen:LoadCase1Shape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Load case 1 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:PropellerBladeLoadCaseShape ;
        sh:qualifiedValueShape gen:LoadCase2Shape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Load case 2 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:PropellerBladeLoadCaseShape ;
        sh:qualifiedValueShape gen:LoadCase3Shape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Load case 3 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:PropellerBladeLoadCaseShape ;
        sh:qualifiedValueShape gen:LoadCase4Shape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Load case 4 must be covered."
    ] .

gen:CPBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:hasValue "CP"
    ] ;
    sh:node gen:Cases1To4Shape .

gen:FPReversingBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:hasValue "FP"
    ] ;
    sh:property [
        sh:path nltl:propellerReversingCapability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:node gen:Cases1To4Shape ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:PropellerBladeLoadCaseShape ;
        sh:qualifiedValueShape gen:LoadCase5Shape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Load case 5 must be covered for a reversing FP propeller."
    ] .

gen:FPNonReversingBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:hasValue "FP"
    ] ;
    sh:property [
        sh:path nltl:propellerReversingCapability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:node gen:Cases1To4Shape .

gen:TRF076PropellerBladeLoadCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in ( "CP" "FP" ) ;
        sh:message "The propeller type must be recorded as CP or FP."
    ] ;
    sh:or (
        gen:CPBranchShape
        gen:FPReversingBranchShape
        gen:FPNonReversingBranchShape
    ) .
<END_SHACL>
