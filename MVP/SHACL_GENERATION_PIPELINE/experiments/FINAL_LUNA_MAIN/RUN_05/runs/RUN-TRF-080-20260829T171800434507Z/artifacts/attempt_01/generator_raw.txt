<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-080> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceLoadCycleCount is required as an integer." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "propellerBladeCount is required as an integer." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "effectiveIceLoadCycleCount must provide one decimal quantity value in unitless form." ;
    ] ;
    sh:sparql [
        sh:message "effectiveIceLoadCycleCount must equal iceLoadCycleCount multiplied by propellerBladeCount within a relative tolerance of 1e-6, with an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?quantity .
                ?quantity qudt:numericValue ?reported .
                BIND((xsd:decimal(?nice) * xsd:decimal(?blades)) AS ?expected)
                BIND(
                    IF(
                        (ABS(?expected) * 0.000001) > 0.000000001,
                        (ABS(?expected) * 0.000001),
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?reported - ?expected) > ?tolerance)
            }
        """ ;
    ] .
<END_SHACL>
