<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-080> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] ;
    sh:sparql [
        sh:message "For a ship with a propeller ice interaction component, every supplied effective ice load cycle count must equal iceLoadCycleCount multiplied by propellerBladeCount within an expected-result-scaled tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component a nltl:propellerIceInteractionComponent .

                $this nltl:iceLoadCycleCount ?niceQuantity .
                ?niceQuantity qudt:numericValue ?nice .

                $this nltl:propellerBladeCount ?blades .

                $this nltl:effectiveIceLoadCycleCount ?resultQuantity .
                ?resultQuantity qudt:numericValue ?reported .

                BIND((?nice * ?blades) AS ?expected)
                BIND((?expected * 0.000001 + 0.000001) AS ?tolerance)

                FILTER (
                    ?reported < (?expected - ?tolerance)
                    ||
                    ?reported > (?expected + ?tolerance)
                )
            }
        """
    ] .
<END_SHACL>
