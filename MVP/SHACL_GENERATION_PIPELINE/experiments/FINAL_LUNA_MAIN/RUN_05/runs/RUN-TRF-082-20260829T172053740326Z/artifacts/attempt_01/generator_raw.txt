<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF082ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardThrustKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether hydrodynamic bollard thrust is known."
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:node gen:TRF082ThrustQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Hydrodynamic bollard thrust must be provided as a single kN quantity."
    ] ;
    sh:or (
        gen:TRF082KnownThrustBranchShape
        gen:TRF082UnknownThrustBranchShape
    ) .

gen:TRF082ThrustQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF082KnownThrustBranchShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardThrustKnown ;
        sh:hasValue true
    ] .

gen:TRF082UnknownThrustBranchShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardThrustKnown ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TRF082TableLookupCaseShape ;
        sh:minCount 1 ;
        sh:message "When hydrodynamic bollard thrust is not known, a Table 6-8 lookup case with selection evidence is required."
    ] .

gen:TRF082TableLookupCaseShape a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The Table 6-8 lookup case must record that the lookup was applied."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash8 ;
        sh:minCount 1 ;
        sh:message "The lookup case must reference TRAFICOM Table 6-8."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The Table 6-8 selection must have supporting evidence."
    ] .
<END_SHACL>
