@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single propeller type is required to determine whether the CP rule applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:pattern "^(?!CP$).*"
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller, pitchValueKnown must be true when the bollard-condition pitch is directly specified."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "For a CP propeller with known pitch, the MCR bollard-condition pitch must be provided in metres."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller using the calculated value, pitchValueKnown must be false."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The free-running MCR propeller pitch must be provided in metres for the CP fallback calculation."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The calculated MCR bollard-condition propeller pitch must be provided in metres."
            ] ;
            sh:sparql [
                sh:message "For a CP propeller with unknown pitch, propellerPitchAtMcrBollard must equal 0.7 times propellerPitchAtMcrFreeRunning within a relative tolerance of 1e-6, with an absolute tolerance of 1e-9 near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:propellerType "CP" ;
                              nltl:pitchValueKnown false ;
                              nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
                              nltl:propellerPitchAtMcrBollard ?bollardQuantity .

                        ?freeQuantity qudt:numericValue ?freeValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/M> .

                        ?bollardQuantity qudt:numericValue ?bollardValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/M> .

                        BIND("0.7"^^xsd:decimal * ?freeValue AS ?expectedValue)
                        BIND(ABS(?bollardValue - ?expectedValue) AS ?difference)
                        BIND(
                            IF(
                                "0.000001"^^xsd:decimal * ABS(?expectedValue)
                                > "0.000000001"^^xsd:decimal,
                                "0.000001"^^xsd:decimal * ABS(?expectedValue),
                                "0.000000001"^^xsd:decimal
                            )
                            AS ?tolerance
                        )

                        FILTER(?difference > ?tolerance)
                    }
                """
            ]
        ]
    ) .
