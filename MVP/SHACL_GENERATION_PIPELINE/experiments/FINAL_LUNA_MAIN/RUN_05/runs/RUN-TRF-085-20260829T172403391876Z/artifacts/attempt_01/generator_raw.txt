<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PropellerSpeedMcrBollardShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerDiameterQuantityShape ;
        sh:message "A ship must provide its propeller diameter in metres."
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerSpeedQuantityShape ;
        sh:message "A ship must provide the propeller rotational speed at MCR in bollard condition."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:BollardSpeedTableLookupCaseShape ;
                sh:message "When the bollard propeller speed is not known, a Table 6-9 lookup case is required."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The Table 6-9 lookup input must equal the ship's propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:lookupInputQuantity ?lookupInput .
                ?lookupInput qudt:numericValue ?lookupValue .
                $this nltl:propellerDiameter ?diameter .
                ?diameter qudt:numericValue ?diameterValue .
                FILTER (xsd:decimal(?lookupValue) != xsd:decimal(?diameterValue))
            }
        """
    ] .

        sh:maxCount 1

gen:PropellerSpeedQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
    ] .

    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LookupInputQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
