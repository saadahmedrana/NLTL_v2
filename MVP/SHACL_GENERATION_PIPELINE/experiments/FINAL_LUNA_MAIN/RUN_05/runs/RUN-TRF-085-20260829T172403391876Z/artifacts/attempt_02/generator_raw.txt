<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:tableLookupApplied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:lookupInputQuantity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:lookupResultQuantity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:tableReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue nltl:traficomTable6Dash9
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "When bollard propeller speed is unknown, the propeller speed must equal the Table 6-9 lookup result for the propeller diameter." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:bollardPropellerSpeedKnown false ;
                              nltl:propellerDiameter ?diameter ;
                              nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed ;
                              nltl:hasTableLookupCase ?case .
                        ?diameter qudt:numericValue ?diameterValue .
                        ?speed qudt:numericValue ?speedValue .
                        ?case nltl:tableLookupApplied true ;
                              nltl:lookupInputQuantity ?input ;
                              nltl:lookupResultQuantity ?result ;
                              nltl:tableReference nltl:traficomTable6Dash9 .
                        ?input qudt:numericValue ?inputValue .
                        ?result qudt:numericValue ?resultValue .
                        FILTER (
                            ?diameterValue = ?inputValue &&
                            ?speedValue = ?resultValue
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
