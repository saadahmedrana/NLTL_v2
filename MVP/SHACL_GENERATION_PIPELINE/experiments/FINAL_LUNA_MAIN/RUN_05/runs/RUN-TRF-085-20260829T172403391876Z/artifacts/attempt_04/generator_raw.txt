<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerDiameterQuantityShape
    ] ;
    sh:or (
        gen:TRF085KnownSpeedShape
        gen:TRF085LookupSpeedShape
    ) .

gen:PropellerDiameterQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:PropellerSpeedQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
    ] .

gen:TRF085KnownSpeedShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerSpeedQuantityShape
    ] .

gen:TRF085LookupSpeedShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerSpeedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:TRF085TableLookupCaseShape
    ] ;
    sh:sparql [
        sh:message "In the Table 6-9 alternative, the ship diameter and propeller speed must equal the input and selected result of the same lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameterQuantity .
                ?diameterQuantity qudt:numericValue ?diameterValue .

                $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speedQuantity .
                ?speedQuantity qudt:numericValue ?speedValue .

                $this nltl:hasTableLookupCase ?lookupCase .
                ?lookupCase nltl:lookupInputQuantity ?inputQuantity ;
                            nltl:lookupResultQuantity ?resultQuantity .
                ?inputQuantity qudt:numericValue ?inputValue .
                ?resultQuantity qudt:numericValue ?resultValue .

                FILTER (
                    ?diameterValue != ?inputValue ||
                    ?speedValue != ?resultValue
                )
            }
        """
    ] .

gen:TRF085TableLookupCaseShape a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash9
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] .

gen:LookupQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
