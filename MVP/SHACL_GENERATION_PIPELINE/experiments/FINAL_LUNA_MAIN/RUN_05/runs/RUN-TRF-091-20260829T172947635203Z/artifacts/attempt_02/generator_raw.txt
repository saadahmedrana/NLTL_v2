<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF091Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceMillingSequenceCase ;
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
                sh:minCount 1 ;
                sh:message "The ice-milling load case shall have propulsion-line strength evaluation as its purpose."
            ] ;
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:not [
                    sh:hasValue nltl:stallingAnalysisPurposeValue
                ] ;
                sh:message "The ice-milling load case shall not have stalling analysis as its purpose."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:class nltl:loadCase
    ] ;
    sh:sparql [
        sh:message "Each ice-milling sequence case shall be the same load-case node used for the propulsion-line strength evaluation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasIceMillingSequenceCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?case .
                }
            }
        """
    ] .
<END_SHACL>
