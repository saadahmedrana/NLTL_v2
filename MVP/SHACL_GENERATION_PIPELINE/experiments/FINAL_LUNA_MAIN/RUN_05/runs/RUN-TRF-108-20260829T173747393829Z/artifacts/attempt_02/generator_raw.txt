<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-108> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue "unknown" ;
                sh:message "An unknown SN-curve type must be explicitly recorded."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:twoSlopeStressLifeCurve ;
                sh:message "When the SN-curve type is unknown, the two-slope SN curve shall be selected."
            ]
        ]
        [
            sh:property [
                sh:path nltl:stressLifeCurveType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A known SN-curve type shall be recorded."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A known SN-curve type shall have a selected curve."
            ] ;
            sh:property [
                sh:path nltl:propellerBladeMaterialProperty ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "A known SN-curve selection shall correspond to the blade material properties."
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Evidence shall establish correspondence between the selected SN curve and the blade material properties."
            ] ;
            sh:sparql [
                sh:message "A known SN-curve type must not be represented as unknown and must have a material-corresponding selection with evidence." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:stressLifeCurveType ?curveType .
                        FILTER (STR(?curveType) != "unknown")
                        $this nltl:stressLifeCurveSelection ?selection .
                        $this nltl:propellerBladeMaterialProperty ?material .
                        $this nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
