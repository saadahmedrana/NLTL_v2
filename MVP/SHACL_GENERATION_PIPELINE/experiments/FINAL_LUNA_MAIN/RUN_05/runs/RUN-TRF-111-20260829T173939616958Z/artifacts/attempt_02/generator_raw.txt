<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:message "The ship must identify its propeller shaft line component."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:node [
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:class nltl:shaftLineLoadCaseTypeValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each propulsion load case must identify a shaft line load case type."
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.3 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:message "The extreme operational load case requires a yield safety factor of at least 1.3."
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase ;
                        sh:minCount 1
                    ] ;
                    sh:sparql [
                        sh:message "The fatigue load case requires a ship component fatigue safety factor of at least 1.5." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                FILTER NOT EXISTS {
                                    ?ship nltl:hasPropulsionLoadCase $this .
                                    ?ship nltl:hasComponent ?component .
                                    ?component nltl:fatigueSafetyFactor ?factor .
                                    ?factor qudt:numericValue ?value .
                                    ?factor qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                    FILTER (
                                        datatype(?value) = xsd:decimal &&
                                        ?value >= 1.5
                                    )
                                }
                            }
                        """
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.0 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:message "The blade failure load case requires a yield safety factor of at least 1.0."
                    ]
                ]
            )
        ] ;
        sh:minCount 1 ;
        sh:message "The ship must provide applicable propulsion load cases."
    ] .
<END_SHACL>
