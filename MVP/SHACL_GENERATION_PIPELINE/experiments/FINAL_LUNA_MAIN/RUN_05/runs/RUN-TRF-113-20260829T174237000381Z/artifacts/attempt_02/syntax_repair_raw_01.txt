<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-113> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:steeringMechanism
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship shall identify a steering mechanism designed for the required blade-loss plastic-bending condition."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:thrusterFitting
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship shall identify a thruster fitting designed for the required blade-loss plastic-bending condition."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:thrusterBody
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship shall identify a thruster body designed for the required blade-loss plastic-bending condition."
    ] ;
    sh:property [
        sh:path nltl:bladePlasticBending ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minLength 1 ;
        sh:message "A non-empty design statement for resistance to plastic blade bending without damage is required."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeOrientation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minLength 1 ;
        sh:message "The propeller blade orientation producing the maximum component load shall be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:maximumComponentLoadBladeOrientationConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The blade-loss load case shall confirm that the recorded orientation produces the maximum component load."
            ]
        ] ;
        sh:message "A propulsion blade-loss load case with maximum-component-load orientation confirmation is required."
    ] ;
    sh:property [
        sh:path nltl:thrusterBladeLossDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "Evidence supporting the thruster blade-loss design assessment is required."
    ] ;
    sh:sparql [
        sh:message "Each represented steering mechanism, thruster fitting, and thruster body shall be covered by the recorded plastic-bending design statement and the confirmed maximum-load blade-loss case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component a ?componentType .
                FILTER (?componentType IN (
                    nltl:steeringMechanism,
                    nltl:thrusterFitting,
                    nltl:thrusterBody
                ))
                FILTER NOT EXISTS {
                    $this nltl:bladePlasticBending ?design .
                    FILTER (
                        datatype(?design) = xsd:string &&
                        STRLEN(STR(?design)) > 0
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:propellerBladeOrientation ?orientation .
                    FILTER (
                        datatype(?orientation) = xsd:string &&
                        STRLEN(STR(?orientation)) > 0
                    )
                    $this nltl:hasPropulsionLoadCase ?case .
                    ?case nltl:maximumComponentLoadBladeOrientationConfirmed true .
                }
            }
        """
    ] .

</END_SHACL>
