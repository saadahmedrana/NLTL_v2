<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node <urn:nltl:generated-shape:TRF-120-ThrusterBodyShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each thruster natural-frequency calculation must account for damping, water added mass, and ship attachment stiffness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                FILTER (
                    NOT EXISTS {
                        $this nltl:damping ?damping
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:waterAddedMass ?waterAddedMass
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterNaturalFrequency ?naturalFrequency
                    }
                    ||
                    NOT EXISTS {
                        ?body nltl:shipAttachmentStiffness ?attachmentStiffness
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Thruster global natural-frequency calculation cases must cover both longitudinal and transverse directions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                FILTER (
                    NOT EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?longitudinalCase .
                        ?longitudinalCase nltl:vibrationDirection nltl:longitudinalDirection
                    }
                    ||
                    NOT EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?transverseCase .
                        ?transverseCase nltl:vibrationDirection nltl:transverseDirection
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-120-ThrusterBodyShape>
    a sh:NodeShape ;
    sh:class nltl:thrusterBody ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-PER-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:node <urn:nltl:generated-shape:TRF-120-CalculationCaseShape>
    ] .

<urn:nltl:generated-shape:TRF-120-CalculationCaseShape>
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:vibrationDirectionValue ;
            sh:in (
                nltl:longitudinalDirection
                nltl:transverseDirection
            )
        ]
    ] .
<END_SHACL>
