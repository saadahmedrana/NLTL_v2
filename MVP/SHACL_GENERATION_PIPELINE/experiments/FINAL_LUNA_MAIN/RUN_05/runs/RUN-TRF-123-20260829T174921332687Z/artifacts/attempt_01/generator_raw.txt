@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-123-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node [
            sh:class nltl:occasionalForceLoadCase ;
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:message "The occasional-force load case must confirm that propeller blades are excluded from this scope."
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:shipComponent ;
                    sh:property [
                        sh:path nltl:occasionalForceComponentStress ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:hasMaterialProperties ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:materialProperties ;
                            sh:property [
                                sh:path nltl:componentMaterialYieldStrength ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ] ;
                                sh:minCount 1
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally calculated occasional-force stress exceeds the component yield strength after applying the declared safety factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this ?loadCase ?component ?material
            WHERE {
                $this nltl:hasOccasionalForceLoadCase ?loadCase .
                ?loadCase nltl:occasionalForceCaseAssessedComponent ?component .
                ?loadCase nltl:componentYieldSafetyFactor ?factorValue .
                ?factorValue qudt:numericValue ?factor .
                ?component nltl:occasionalForceComponentStress ?stressValue .
                ?stressValue qudt:numericValue ?stress .
                ?component nltl:hasMaterialProperties ?material .
                ?material nltl:componentMaterialYieldStrength ?yieldValue .
                ?yieldValue qudt:numericValue ?yield .
                FILTER (?stress * ?factor > ?yield)
            }
        """
    ] .
