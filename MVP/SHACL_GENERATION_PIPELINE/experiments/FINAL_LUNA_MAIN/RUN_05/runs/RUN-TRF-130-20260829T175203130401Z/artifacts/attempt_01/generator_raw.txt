gen:TRF130NormalBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:inletChest ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "At least one inlet chest must satisfy the location, volume, ice-space, discharge, and strainer requirements." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?c .
                ?c a nltl:inletChest ;
                   nltl:seaInletLocation "near centreline and well aft if possible"^^xsd:string ;
                   nltl:iceAccumulationSpaceAboveInlet true ;
                   nltl:fullCapacityDischargePipeConnected true ;
                   nltl:recommendedChestVolume ?rv ;
                   nltl:strainerOpenArea ?sa ;
                   nltl:inletPipeArea ?ia .
                $this nltl:totalRelevantEnginePower ?tp .
                ?rv qudt:numericValue ?rvn ;
                    qudt:unit <http://qudt.org/vocab/unit/M3> .
                ?sa qudt:numericValue ?san ;
                    qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?ia qudt:numericValue ?ian ;
                    qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?tp qudt:numericValue ?tpn ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                FILTER (
                    datatype(?rvn) = xsd:decimal &&
                    datatype(?san) = xsd:decimal &&
                    datatype(?ian) = xsd:decimal &&
                    datatype(?tpn) = xsd:decimal &&
                    ABS(?rvn - (?tpn / 750)) <=
                      (0.000001 * ABS(?tpn / 750) + 0.000000001) &&
                    ?san >= (4 * ?ian)
                )
            }
        """
    ] .

gen:TRF130AlternativeBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:inletChest ;
        sh:minCount 2 ;
        sh:qualifiedValueShape [
            sh:class nltl:inletChest ;
            sh:property [
                sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 2
    ] ;
    sh:sparql [
        sh:message "At least two alternating inlet chests must retain the required location, discharge, and strainer arrangements." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?c .
                ?c a nltl:inletChest ;
                   nltl:alternatingCoolingWaterIntakeAndDischarge true ;
                   nltl:seaInletLocation "near centreline and well aft if possible"^^xsd:string ;
                   nltl:fullCapacityDischargePipeConnected true ;
                   nltl:strainerOpenArea ?sa ;
                   nltl:inletPipeArea ?ia .
                ?sa qudt:numericValue ?san ;
                    qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?ia qudt:numericValue ?ian ;
                    qudt:unit <http://qudt.org/vocab/unit/M2> .
                FILTER (
                    datatype(?san) = xsd:decimal &&
                    datatype(?ian) = xsd:decimal &&
                    ?san >= (4 * ?ian)
                )
            }
            GROUP BY $this
            HAVING (COUNT(DISTINCT ?c) >= 2)
        """
    ] .
