<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF130Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerQuantityShape
    ] ;
    sh:or (
        gen:TRF130NormalBranchShape
        gen:TRF130AlternativeBranchShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable inlet chest must satisfy the required volume and strainer-area calculations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:totalRelevantEnginePower ?totalPower .
                ?totalPower qudt:numericValue ?totalPowerValue .
                $this nltl:hasComponent ?chest .
                ?chest a nltl:inletChest .
                ?chest nltl:recommendedChestVolume ?recommendedVolume .
                ?recommendedVolume qudt:numericValue ?recommendedVolumeValue .
                ?chest nltl:strainerOpenArea ?openArea .
                ?openArea qudt:numericValue ?openAreaValue .
                ?chest nltl:inletPipeArea ?pipeArea .
                ?pipeArea qudt:numericValue ?pipeAreaValue .
                ?chest nltl:strainerOpenAreaRatio ?areaRatio .
                ?areaRatio qudt:numericValue ?areaRatioValue .
                FILTER (
                    ABS(?recommendedVolumeValue - (?totalPowerValue / 750)) <=
                    0.000001 * ABS(?totalPowerValue / 750)
                    ||
                    (
                        ABS(?totalPowerValue / 750) < 0.000001 &&
                        ABS(?recommendedVolumeValue - (?totalPowerValue / 750)) <= 0.000001
                    )
                )
                FILTER (?openAreaValue >= (4 * ?pipeAreaValue))
                FILTER (
                    ABS(?areaRatioValue - (?openAreaValue / ?pipeAreaValue)) <=
                    0.000001 * ABS(?openAreaValue / ?pipeAreaValue)
                    ||
                    (
                        ABS(?openAreaValue / ?pipeAreaValue) < 0.000001 &&
                        ABS(?areaRatioValue - (?openAreaValue / ?pipeAreaValue)) <= 0.000001
                    )
                )
            }
        """
    ] .

gen:TRF130NormalBranchShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node gen:InletChestShape
    ] .
