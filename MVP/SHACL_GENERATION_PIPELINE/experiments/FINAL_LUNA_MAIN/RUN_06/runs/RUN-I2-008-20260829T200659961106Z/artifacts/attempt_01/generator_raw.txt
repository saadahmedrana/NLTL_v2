gen:I2_008_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue ;
        sh:minCount 1 ;
        sh:message "A Polar Class value is required to establish applicability."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_008_BowSubregionCaseShape ;
        sh:message "At least one bow subregion calculation case is required."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:node gen:I2_008_PressureQuantityShape ;
        sh:message "The externally calculated average ice pressure is required."
    ] ;
    sh:property [
        sh:path nltl:lineLoad ;
        sh:minCount 1 ;
        sh:node gen:I2_008_LineLoadQuantityShape ;
        sh:message "The externally calculated line load is required."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:node gen:I2_008_LengthQuantityShape ;
        sh:message "The load patch height input is required."
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:node gen:I2_008_LengthQuantityShape ;
        sh:message "The load patch length input is required."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node gen:I2_008_PressureQuantityShape ;
        sh:message "The local ice pressure input is required."
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:node gen:I2_008_DimensionlessQuantityShape ;
        sh:message "The externally calculated shape coefficient is required."
    ] ;
    sh:property [
        sh:path nltl:totalGlancingImpactForce ;
        sh:minCount 1 ;
        sh:node gen:I2_008_ForceQuantityShape ;
        sh:message "The externally calculated total glancing impact force is required."
    ] .

        sh:path nltl:bowShapeCoefficient ;

    ] ;
    sh:property [
        sh:path nltl:glancingImpactLineLoad ;
        sh:minCount 1 ;
        sh:node gen:I2_008_GlancingLineLoadQuantityShape ;
        sh:message "Each bow subregion case requires an externally calculated glancing impact line load."
    ] ;
    sh:property [
        sh:path nltl:glancingImpactPressure ;
        sh:minCount 1 ;

        sh:message "Each bow subregion case requires an externally calculated glancing impact pressure."
    ] .
