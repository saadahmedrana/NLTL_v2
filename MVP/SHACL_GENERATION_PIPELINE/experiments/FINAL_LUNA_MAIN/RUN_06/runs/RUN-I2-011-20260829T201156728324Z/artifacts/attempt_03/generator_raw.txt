<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:I2_011_ApplicableShape
        gen:I2_011_NonApplicableShape
    ) .

gen:I2_011_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( nltl:polarClassPc6 nltl:polarClassPc7 )
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaN>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaN>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0.6"^^xsd:decimal ;
                sh:maxInclusive "0.6"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "1.3"^^xsd:decimal ;
                sh:maxInclusive "1.3"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For an applicable PC6 or PC7 ship with a bulbous bow, bowDesignForce must be at least the method (iv) force and the method (iii) lower-bound force after conversion from kN to MN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:polarClass ?pc ;
                      nltl:bulbousBowPresent true ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode ;
                      nltl:shapeCoefficient ?shapeNode ;
                      nltl:loadPatchAspectRatio ?arNode ;
                      nltl:bowDesignForce ?reportedNode .

                ?pc a nltl:polarClassValue .
                ?iiiNode qudt:numericValue ?iiiMn ;
                          qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                ?ivNode qudt:numericValue ?ivMn ;
                         qudt:unit <http://qudt.org/vocab/unit/MegaN> .
                ?shapeNode qudt:numericValue ?shape ;
                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?arNode qudt:numericValue ?ar ;
                         qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?reportedNode qudt:numericValue ?reportedKn ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                FILTER (?pc IN (nltl:polarClassPc6, nltl:polarClassPc7))

                BIND ((?iiiMn * (?shape / 0.6) * (?ar / 1.3)) AS ?lowerBoundMn)
                BIND ((?reportedKn / 1000) AS ?reportedMn)

                FILTER (
                    ?reportedMn <
                    ?ivMn -
                    (0.000001 * IF(ABS(?ivMn) > 0.000000001, ABS(?ivMn), 0.000000001))
                    ||
                    ?reportedMn <
                    ?lowerBoundMn -
                    (0.000001 * IF(ABS(?lowerBoundMn) > 0.000000001, ABS(?lowerBoundMn), 0.000000001))
                )
            }
        """
    ] .

gen:I2_011_NonApplicableShape
    a sh:NodeShape ;
    sh:or (
        gen:I2_011_NonApplicableBulbousBowShape
        gen:I2_011_NonApplicablePolarClassShape
    ) .

gen:I2_011_NonApplicableBulbousBowShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        )
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

    ] .
<END_SHACL>
