    sh:sparql [
        sh:message "Each bow subregion midpoint must correspond, within an explicit relative tolerance of 1e-6 plus 1e-9, to one of LBOW/8, 3*LBOW/8, 5*LBOW/8, or 7*LBOW/8, with all four midpoint positions represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthNode .
                ?lengthNode qudt:numericValue ?length .
                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case1 .
                    ?case1 nltl:bowSubregionMidLengthPosition ?midNode1 .
                    ?midNode1 qudt:numericValue ?mid1 .
                    FILTER (
                        ?mid1 >= (?length / 8.0 - ((?length / 8.0) * 0.000001 + 0.000000001)) &&
                        ?mid1 <= (?length / 8.0 + ((?length / 8.0) * 0.000001 + 0.000000001))
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case2 .
                    ?case2 nltl:bowSubregionMidLengthPosition ?midNode2 .
                    ?midNode2 qudt:numericValue ?mid2 .
                    FILTER (
                        ?mid2 >= (3.0 * ?length / 8.0 - ((3.0 * ?length / 8.0) * 0.000001 + 0.000000001)) &&
                        ?mid2 <= (3.0 * ?length / 8.0 + ((3.0 * ?length / 8.0) * 0.000001 + 0.000000001))
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case3 .
                    ?case3 nltl:bowSubregionMidLengthPosition ?midNode3 .
                    ?midNode3 qudt:numericValue ?mid3 .
                    FILTER (
                        ?mid3 >= (5.0 * ?length / 8.0 - ((5.0 * ?length / 8.0) * 0.000001 + 0.000000001)) &&
                        ?mid3 <= (5.0 * ?length / 8.0 + ((5.0 * ?length / 8.0) * 0.000001 + 0.000000001))
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case4 .
                    ?case4 nltl:bowSubregionMidLengthPosition ?midNode4 .
                    ?midNode4 qudt:numericValue ?mid4 .
                    FILTER (
                        ?mid4 >= (7.0 * ?length / 8.0 - ((7.0 * ?length / 8.0) * 0.000001 + 0.000000001)) &&
                        ?mid4 <= (7.0 * ?length / 8.0 + ((7.0 * ?length / 8.0) * 0.000001 + 0.000000001))
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected maximum bow force, line load, and pressure must equal the corresponding case maximums within a tolerance scaled from each expected maximum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForceNode ;
                      nltl:selectedMaximumBowLineLoad ?selectedLineNode ;
                      nltl:selectedMaximumBowPressure ?selectedPressureNode .
                ?selectedForceNode qudt:numericValue ?selectedForce .
                ?selectedLineNode qudt:numericValue ?selectedLine .
                ?selectedPressureNode qudt:numericValue ?selectedPressure .
                {
                    SELECT ?ship (MAX(?force) AS ?maxForce)
                           (MAX(?lineLoad) AS ?maxLineLoad)
                           (MAX(?pressure) AS ?maxPressure)
                    WHERE {
                        ?ship nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?forceNode ;
                              nltl:bowSubregionCalculatedLineLoad ?lineNode ;
                              nltl:bowSubregionCalculatedPressure ?pressureNode .
                        ?forceNode qudt:numericValue ?force .
                        ?lineNode qudt:numericValue ?lineLoad .
                        ?pressureNode qudt:numericValue ?pressure .
                    }
                    GROUP BY ?ship
                }
                FILTER (?ship = $this)
                FILTER (
                    ?selectedForce < (?maxForce - (?maxForce * 0.000001 + 0.000000001)) ||
                    ?selectedForce > (?maxForce + (?maxForce * 0.000001 + 0.000000001)) ||
                    ?selectedLine < (?maxLineLoad - (?maxLineLoad * 0.000001 + 0.000000001)) ||
                    ?selectedLine > (?maxLineLoad + (?maxLineLoad * 0.000001 + 0.000000001)) ||
                    ?selectedPressure < (?maxPressure - (?maxPressure * 0.000001 + 0.000000001)) ||
                    ?selectedPressure > (?maxPressure + (?maxPressure * 0.000001 + 0.000000001))
                )
            }
        """
    ] .
