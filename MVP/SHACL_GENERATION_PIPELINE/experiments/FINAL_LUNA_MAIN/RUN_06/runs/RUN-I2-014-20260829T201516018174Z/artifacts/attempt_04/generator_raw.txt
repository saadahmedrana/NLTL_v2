        sh:message "The four bow subregion midpoints must cover LBOW/8, 3*LBOW/8, 5*LBOW/8, and 7*LBOW/8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    SELECT $this (COUNT(DISTINCT ?slot) AS ?slotCount)
                    WHERE {
                        $this nltl:bowLength ?lengthNode ;
                              nltl:hasBowSubregionCalculationCase ?case .
                        ?lengthNode qudt:numericValue ?length .
                        ?case nltl:bowSubregionMidLengthPosition ?positionNode .
                        ?positionNode qudt:numericValue ?position .
                        BIND(
                            IF(
                                ?position >= ((?length / 8) - (((?length / 8) * 0.000001) + 0.000000001)) &&
                                ?position <= ((?length / 8) + (((?length / 8) * 0.000001) + 0.000000001)),
                                1,
                                IF(
                                    ?position >= ((3 * ?length / 8) - (((3 * ?length / 8) * 0.000001) + 0.000000001)) &&
                                    ?position <= ((3 * ?length / 8) + (((3 * ?length / 8) * 0.000001) + 0.000000001)),
                                    2,
                                    IF(
                                        ?position >= ((5 * ?length / 8) - (((5 * ?length / 8) * 0.000001) + 0.000000001)) &&
                                        ?position <= ((5 * ?length / 8) + (((5 * ?length / 8) * 0.000001) + 0.000000001)),
                                        3,
                                        IF(

                                            ?position <= ((7 * ?length / 8) + (((7 * ?length / 8) * 0.000001) + 0.000000001)),
                                            4,
                                            0
                                        )
                                    )
                                )
                            ) AS ?slot
                        )
                        FILTER (?slot > 0)
                    }
                    GROUP BY $this
                }
                FILTER (?slotCount != 4)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected bow force, line load, and pressure must equal the corresponding case maxima within a tolerance scaled from each expected maximum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                              nltl:bowSubregionCalculatedPressure ?pressureNode .

                        ?lineLoadNode qudt:numericValue ?lineLoad .
                        ?pressureNode qudt:numericValue ?pressure .
                    }
                    GROUP BY $this
                }

                      nltl:selectedMaximumBowLineLoad ?selectedLineLoadNode ;
                      nltl:selectedMaximumBowPressure ?selectedPressureNode .
                ?selectedForceNode qudt:numericValue ?selectedForce .
                ?selectedLineLoadNode qudt:numericValue ?selectedLineLoad .
                ?selectedPressureNode qudt:numericValue ?selectedPressure .
                FILTER (
                    ?selectedForce < (?maxForce - ((?maxForce * 0.000001) + 0.000000001)) ||

                    ?selectedPressure > (?maxPressure + ((?maxPressure * 0.000001) + 0.000000001))
                )
            }
        """
    ] .
