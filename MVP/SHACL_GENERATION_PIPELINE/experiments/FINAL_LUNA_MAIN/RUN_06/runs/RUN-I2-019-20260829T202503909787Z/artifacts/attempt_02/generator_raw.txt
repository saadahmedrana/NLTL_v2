<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_019_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2_019_StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2_019_StructuralMemberShape>
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:hullAreaValue ;
                    sh:property [
                        sh:path nltl:hullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "selectedHullAreaFactor must equal the largest hullAreaFactor among the hull areas spanned by the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:selectedHullAreaFactor ?selectedNode .
                ?selectedNode qudt:numericValue ?selectedValue .
                FILTER (
                    EXISTS {
                        $this nltl:hasSpannedHullArea ?greaterArea .
                        ?greaterArea nltl:hullAreaFactor ?greaterFactorNode .
                        ?greaterFactorNode qudt:numericValue ?greaterValue .
                        FILTER (?greaterValue > ?selectedValue)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasSpannedHullArea ?matchingArea .
                        ?matchingArea nltl:hullAreaFactor ?matchingFactorNode .
                        ?matchingFactorNode qudt:numericValue ?matchingValue .
                        FILTER (?matchingValue = ?selectedValue)
                    }
                )
            }
        """
    ] .
<END_SHACL>
