<urn:nltl:generated-shape:I2-022-PlatingShape>
    a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-022-FramingAngleOmegaShape>
    ] ;
    sh:sparql [
        sh:message "Framing angle omega must be at least 70 degrees for the applicable transversely-framed plating rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:framingAngleOmega ?quantity .
                ?quantity qudt:numericValue ?value .
                FILTER (?value < 70)
            }
        """
    ] ;
    sh:property [
        sh:path nltl:platingHullAreaClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowIntermediateBottomHullArea
            nltl:midbodyBottomHullArea
            nltl:sternBottomHullArea
        ) ;
        sh:message "Plating hull area classification must be BI_b, M_b, or S_b."
    ] ;
    sh:property [
        sh:path nltl:platingSupportedByStructuralMember ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-022-StructuralMemberShape> ;
        sh:message "Applicable plating must identify a supporting structural member with the required calculation inputs."
    ] ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-022-IceLoadRequiredNetPlateThicknessShape> ;
        sh:message "Applicable plating must provide the externally computed ice-load required net plate thickness."
    ] .
