<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A structural member must state whether it is continuous through its support."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A structural member must state whether it terminates within an ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:class nltl:frameSupportConditionValue
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:fixedFrameSupport
            ]
            [
                sh:hasValue nltl:simpleFrameSupport
            ]
        ) ;
        sh:message "The frame support condition must be stated as fixed or simple support."
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:I2_026_MemberSupportShape
    ] ;
    sh:sparql [
        sh:message "Support fixity must be fixed for continuous members, members with a connection bracket, members with demonstrated significant rotational restraint, or members terminating within an ice-strengthened area; otherwise it must be simple. Demonstrated rotational restraint must have evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?support
            WHERE {
                $this nltl:hasMemberSupport ?support ;
                      nltl:continuousThroughSupport ?continuous ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates .

                ?support nltl:connectionBracketPresent ?bracket ;
                         nltl:demonstratedRotationalRestraint ?restraint .

                FILTER (
                    (
                        (
                            ?continuous = true ||
                            ?bracket = true ||
                            ?restraint = true ||
                            ?terminates = true
                        )
                        &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:fixedFrameSupport
                        }
                    )
                    ||
                    (
                        ?continuous = false &&
                        ?bracket = false &&
                        ?restraint = false &&
                        ?terminates = false &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:simpleFrameSupport
                        }
                    )
                    ||
                    (
                        ?restraint = true &&
                        NOT EXISTS {
                            ?support nltl:significantRotationalRestraintEvidence ?evidence
                        }
                    )
                )
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:class nltl:memberSupport ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A member support must state whether a connection bracket is present."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A member support must state whether significant rotational restraint is demonstrated."
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] .
<END_SHACL>
