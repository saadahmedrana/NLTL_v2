<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026_StructuralMemberShape ;
        sh:message "Each linked structural member must satisfy the framing-member support condition."
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026_MemberSupportShape ;
        sh:message "A structural member must identify its supporting member support."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        ) ;
        sh:message "The frame support condition must be identified as fixed or simple."
    ] ;
    sh:or (
        gen:I2_026_FixedMemberShape
        gen:I2_026_SimpleMemberShape
    ) ;
    sh:sparql [
        sh:message "A framing member terminating within an ice-strengthened area must have fixed support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:terminatesWithinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Evidence must be recorded when significant rotational restraint is demonstrated." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean
    ] .

gen:I2_026_FixedMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node gen:I2_026_BracketSupportShape
            ]
        ]
    ) .

gen:I2_026_BracketSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] .

gen:I2_026_SimpleMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:simpleFrameSupport
    ] ;
    sh:sparql [
        sh:message "Simple support may be assumed only where continuous support, a connection bracket, and demonstrated rotational restraint are not present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                {
                    $this nltl:continuousThroughSupport true .
                }
                UNION
                {
                    $this nltl:hasMemberSupport ?bracketSupport .
                    ?bracketSupport nltl:connectionBracketPresent true .
                }
                UNION
                {
                    $this nltl:hasMemberSupport ?restraintSupport .
                    ?restraintSupport nltl:demonstratedRotationalRestraint true .
                }
            }
        """
    ] .
<END_SHACL>
