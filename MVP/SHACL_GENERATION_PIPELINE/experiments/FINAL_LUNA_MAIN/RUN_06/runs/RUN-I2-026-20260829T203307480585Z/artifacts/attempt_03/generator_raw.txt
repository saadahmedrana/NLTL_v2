<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026_MemberSupportShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        sh:message "A structural member with demonstrated rotational restraint requires significant rotational restraint evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each member support must have fixed support when continuity, a connection bracket, demonstrated rotational restraint, or termination within an ice-strengthened area applies; otherwise it must have simple support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:continuousThroughSupport ?continuous ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates ;
                      nltl:hasMemberSupport ?support .
                ?support nltl:connectionBracketPresent ?bracket ;
                         nltl:demonstratedRotationalRestraint ?restraint ;
                         nltl:frameSupportCondition ?condition .
                FILTER (
                    (
                        (
                            ?continuous = true ||
                            ?bracket = true ||
                            ?restraint = true ||
                            ?terminates = true
                        ) &&
                        ?condition != nltl:fixedFrameSupport
                    ) ||
                    (
                        ?continuous = false &&
                        ?bracket = false &&
                        ?restraint = false &&
                        ?terminates = false &&
                        ?condition != nltl:simpleFrameSupport
                    )
                )
            }
        """
    ] .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
