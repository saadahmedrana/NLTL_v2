gen:ShipSectionModulusReadinessShape

    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;

        sh:message "The ship must provide a calculation case for the local-frame section calculation."
    ] .

    sh:property [

    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointReadinessShape ;

    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;

    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
