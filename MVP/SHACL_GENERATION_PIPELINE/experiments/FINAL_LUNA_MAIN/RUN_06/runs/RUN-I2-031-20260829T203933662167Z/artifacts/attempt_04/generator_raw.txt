<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-031-StructuralMemberShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    a sh:NodeShape ;
    sh:or (
        <urn:nltl:generated-shape:I2-031-BottomMemberShape>
        <urn:nltl:generated-shape:I2-031-SideMemberShape>
    ) ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthValue .
                ?strengthValue qudt:numericValue ?strength .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandValue .
                ?demandValue qudt:numericValue ?demand .
                FILTER (?demand > ?strength)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The plastic strength must agree with the midspan plastic collapse load within the specified tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthValue .
                ?strengthValue qudt:numericValue ?strength .
                $this nltl:midspanPlasticCollapseLoad ?midspanValue .
                ?midspanValue qudt:numericValue ?midspan .
                FILTER (ABS(?strength - ?midspan) >
                        (ABS(?midspan) * 0.000001 + 0.000000001))
            }
        """
    ] .

<urn:nltl:generated-shape:I2-031-BottomMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:bottomStructureLocation ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-031-BottomLoadCaseShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:I2-031-SideMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:transverseFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-031-GenericLoadCaseShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:I2-031-BottomLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-GenericLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
