<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-034-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-034-MemberShape>
    ] .

<urn:nltl:generated-shape:I2-034-MemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        <urn:nltl:generated-shape:I2-034-ApplicableMemberShape>
        <urn:nltl:generated-shape:I2-034-NonLongitudinalMemberShape>
        <urn:nltl:generated-shape:I2-034-NonSideMemberShape>
    ) .

<urn:nltl:generated-shape:I2-034-ApplicableMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-034-LoadCaseShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable longitudinal side frame, every represented load-case demand must not exceed the member midspan plastic-collapse load, and plastic strength must agree with that load within the stated numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:midspanPlasticCollapseLoad ?collapseQuantity ;
                      nltl:plasticStrength ?strengthQuantity ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .

                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .

                ?collapseQuantity qudt:numericValue ?collapseValue ;
                                  qudt:unit qudt:KiloN .
                ?strengthQuantity qudt:numericValue ?strengthValue ;
                                  qudt:unit qudt:KiloN .
                ?demandQuantity qudt:numericValue ?demandValue .

                FILTER (
                    ?demandValue > ?collapseValue
                    ||
                    ABS(?strengthValue - ?collapseValue)
                    > (1e-6 * ABS(?collapseValue) + 1e-9)
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-034-NonLongitudinalMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue false
    ] .

<urn:nltl:generated-shape:I2-034-NonSideMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:not [
            sh:hasValue nltl:sideStructureLocation
        ]
    ] .

<urn:nltl:generated-shape:I2-034-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-DemandQuantityShape>
    ] .

<urn:nltl:generated-shape:I2-034-DemandQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
