<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node g:I2_034_StructuralMemberShape
    ] .

g:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The structural member must be identified as longitudinal for this requirement."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:minCount 1 ;
        sh:message "The longitudinal structural member must be located in the side structure."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node g:I2_034_CollapseLoadQuantityShape ;
        sh:minCount 1 ;
        sh:message "The structural member must provide its midspan plastic collapse load."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node g:I2_034_PlasticStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The structural member must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node g:I2_034_LoadCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented load-case combined shear-and-bending demand must not exceed the member's midspan plastic collapse load." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:hasStructuralMemberLoadCase ?case .
    ?case nltl:combinedShearAndBendingDemand ?demandQuantity .
    ?demandQuantity qudt:numericValue ?demand .
    $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
    ?collapseQuantity qudt:numericValue ?collapse .
    FILTER (?demand > ?collapse)
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The member plastic strength must correspond to its midspan plastic collapse load within the permitted numerical tolerance." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:plasticStrength ?strengthQuantity .
    ?strengthQuantity qudt:numericValue ?strength .
    $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
    ?collapseQuantity qudt:numericValue ?collapse .
    FILTER (
        (
            ?collapse >= 0 &&
            (
                (?strength < ?collapse - 0.000000000001 &&
                 ?strength < ?collapse - (0.000001 * ?collapse))
                ||
                (?strength > ?collapse + 0.000000000001 &&
                 ?strength > ?collapse + (0.000001 * ?collapse))
            )
        )
        ||
        (
            ?collapse < 0 &&
            (
                (?strength < ?collapse - 0.000000000001 &&
                 ?strength < ?collapse + (0.000001 * ?collapse))
                ||
                (?strength > ?collapse + 0.000000000001 &&
                 ?strength > ?collapse - (0.000001 * ?collapse))
            )
        )
    )
}
"""
    ] .

g:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node g:I2_034_DemandQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each represented load case must provide its combined shear-and-bending demand."
    ] .

g:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

g:I2_034_CollapseLoadQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

g:I2_034_PlasticStrengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
