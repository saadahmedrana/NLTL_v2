<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-034-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must provide longitudinal-frame orientation evidence."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must provide structural-location evidence."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-034-LoadCaseShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable longitudinal side structural member, every represented load-case demand must not exceed the midspan plastic collapse load, and plastic strength must correspond to that load within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:hasStructuralMemberLoadCase ?loadCase ;
                      nltl:midspanPlasticCollapseLoad ?collapseQuantity ;
                      nltl:plasticStrength ?strengthQuantity .

                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand .

                ?collapseQuantity qudt:numericValue ?collapse .
                ?strengthQuantity qudt:numericValue ?strength .

                FILTER(
                    (?demand > ?collapse)
                    ||
                    (
                        ?collapse >= 1
                        &&
                        (
                            ?strength < (?collapse - (0.000001 * ?collapse))
                            ||
                            ?strength > (?collapse + (0.000001 * ?collapse))
                        )
                    )
                    ||
                    (
                        ?collapse < 1
                        &&
                        (
                            ?strength < (?collapse - 0.000001)
                            ||
                            ?strength > (?collapse + 0.000001)
                        )
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-034-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node <urn:nltl:generated-shape:I2-034-DecimalQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case for an applicable member must provide a combined shear-and-bending demand."
    ] .

<urn:nltl:generated-shape:I2-034-DecimalQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
