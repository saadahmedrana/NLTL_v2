        sh:message "The actual shear area must be at least the calculated required longitudinal frame shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressureQuantity ;
                      nltl:frameSpan ?spanQuantity ;
                      nltl:peakPressureFactor ?peakFactorQuantity ;
                      nltl:shearArea ?actualAreaQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .

                ?pressureQuantity qudt:numericValue ?pressure .
                ?spanQuantity qudt:numericValue ?b .
                ?peakFactorQuantity qudt:numericValue ?peakFactor .
                ?actualAreaQuantity qudt:numericValue ?actualArea .
                ?yieldQuantity qudt:numericValue ?yieldStrength .

                ?member nltl:frameSpacing ?spacingQuantity ;
                        nltl:selectedHullAreaFactor ?hullFactorQuantity .

                ?spacingQuantity qudt:numericValue ?spacing .
                ?hullFactorQuantity qudt:numericValue ?hullFactor .

                BIND((?b / ?spacing) AS ?bPrime)
                BIND(
                    IF(
                        ?bPrime < 2,
                        ?b * (1 - 0.25 * ?bPrime),
                        ?spacing
                    )
                    AS ?b2
                )
                BIND((1 - 0.3 / ?bPrime) * ?b2 AS ?b1)
                BIND(
                    10000 * (?hullFactor * ?peakFactor * ?pressure)
                    * 0.5 * ?b1 * ?spacing
                    / (0.577 * ?yieldStrength)
                    AS ?requiredAreaCm2
                )

                FILTER ((?actualArea * 10000) < ?requiredAreaCm2)
            }
        """
    ] .
