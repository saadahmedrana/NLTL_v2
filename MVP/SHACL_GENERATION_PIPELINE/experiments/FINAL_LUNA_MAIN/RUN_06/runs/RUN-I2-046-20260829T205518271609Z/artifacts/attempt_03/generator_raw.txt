gen:I2_046_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClassRequirementsApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "polarClassRequirementsApplicable must be recorded exactly once as a boolean."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClassRequirementsApplicable ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
                sh:message "An applicable ship must have one valid controlled Polar Class value."
            ] ;
            sh:property [
                sh:path nltl:hasInternalIceStrengthenedStructure ;
                sh:node gen:I2_046_InternalStructureShape ;
                sh:message "Every linked internal ice-strengthened structure must satisfy the corrosion/abrasion addition constraint."
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClassRequirementsApplicable ;
                sh:hasValue false
            ]
        ]
    ] .

gen:I2_046_InternalStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:internalStructureCorrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_046_QuantityValueShape ;
        sh:message "Each applicable internal ice-strengthened structure must have exactly one corrosion/abrasion addition of at least 1.0 mm."
    ] .

gen:I2_046_QuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.0"^^xsd:decimal ;
        sh:message "The corrosion/abrasion addition numeric value must be a decimal of at least 1.0."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM ;
        sh:message "The corrosion/abrasion addition must use exactly one unit, MilliM."
    ] .
