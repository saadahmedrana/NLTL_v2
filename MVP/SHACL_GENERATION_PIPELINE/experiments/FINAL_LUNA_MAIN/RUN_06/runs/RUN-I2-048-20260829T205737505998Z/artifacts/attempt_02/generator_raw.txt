<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qtu: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one permitted Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:class nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade case must identify exactly one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The steel-grade case must refer exactly once to IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qtu:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade case must have one as-built plate thickness quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must have exactly one permitted material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade case must have exactly one permitted steel-strength category."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeShape> ;
        sh:message "Each represented steel-grade case must state exactly one actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeShape>
    ] ;
    sh:sparql [
        sh:message "The Table 8 selection is missing, is supplied for a non-applicable combination, or the actual grade is below the required grade or belongs to a different strength family." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?pc .

                $this nltl:asBuiltPlateThickness ?thicknessNode ;
                      nltl:steelMaterialClass ?material ;
                      nltl:steelStrengthCategory ?category ;
                      nltl:actualHullStructuralSteelGrade ?actual .

                ?thicknessNode qudt:numericValue ?thickness .

                BIND(
                    (?material = nltl:steelMaterialClassThree &&
                     ?category = nltl:normalStrengthSteelCategory &&
                     (
                       (?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3) && ?thickness > 35) ||
                       (?pc IN (nltl:polarClassPc4, nltl:polarClassPc5) && ?thickness > 45)
                     ))
                    AS ?notApplicable
                )

                BIND(
                    IF(
                        ?category = nltl:normalStrengthSteelCategory,
                        IF(
                            ?material = nltl:steelMaterialClassThree,
                            IF(
                                ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                IF(?thickness <= 10, 1, IF(?thickness <= 35, 2, 3)),
                                3
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassOne,
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?thickness <= 35, 1, 2),
                                    IF(?thickness <= 35, 2, 3)
                                ),
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?thickness <= 30, 1, 2),
                                    IF(?thickness <= 30, 2, 3)
                                )
                            )
                        ),
                        IF(
                            ?material = nltl:steelMaterialClassThree,
                            IF(
                                ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                IF(?thickness <= 10, 1, IF(?thickness <= 35, 2, 3)),
                                IF(
                                    ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3),
                                    IF(?thickness <= 35, 3, 4),
                                    IF(?thickness <= 45, 3, 4)
                                )
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassOne,
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?thickness <= 35, 1, 2),
                                    IF(?thickness <= 35, 2, 3)
                                ),
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?thickness <= 30, 1, 2),
                                    IF(?thickness <= 30, 2, 3)
                                )
                            )
                        )
                    )
                    AS ?expectedRank
                )

                FILTER(
                    (
                        ?notApplicable &&
                        EXISTS {
                            $this nltl:requiredHullStructuralSteelGrade ?suppliedRequired
                        }
                    )
                    ||
                    (
                        !?notApplicable &&
                        NOT EXISTS {
                            $this nltl:requiredHullStructuralSteelGrade ?required ;
                            FILTER(
                                (
                                    ?category = nltl:normalStrengthSteelCategory &&
                                    (
                                        (?expectedRank = 1 && ?required = nltl:steelGradeB) ||
                                        (?expectedRank = 2 && ?required = nltl:steelGradeD) ||
                                        (?expectedRank = 3 && ?required = nltl:steelGradeE)
                                    )
                                )
                                ||
                                (
                                    ?category = nltl:highTensileSteelCategory &&
                                    (
                                        (?expectedRank = 1 && ?required = nltl:steelGradeAh) ||
                                        (?expectedRank = 2 && ?required = nltl:steelGradeDh) ||
                                        (?expectedRank = 3 && ?required = nltl:steelGradeEh) ||
                                        (?expectedRank = 4 && ?required = nltl:steelGradeFh)
                                    )
                                )
                            )
                        }
                    )
                    ||
                    (
                        !?notApplicable &&
                        EXISTS {
                            $this nltl:requiredHullStructuralSteelGrade ?required ;
                                  nltl:actualHullStructuralSteelGrade ?actual .
                            ?required nltl:steelGradeOrderRank ?requiredRank .
                            ?actual nltl:steelGradeOrderRank ?actualRank .
                            FILTER(
                                ?actualRank < ?requiredRank ||
                                (
                                    ?category = nltl:normalStrengthSteelCategory &&
                                    ?actual NOT IN (
                                        nltl:steelGradeB,
                                        nltl:steelGradeD,
                                        nltl:steelGradeE
                                    )
                                ) ||
                                (
                                    ?category = nltl:highTensileSteelCategory &&
                                    ?actual NOT IN (
                                        nltl:steelGradeAh,
                                        nltl:steelGradeDh,
                                        nltl:steelGradeEh,
                                        nltl:steelGradeFh
                                    )
                                )
                            )
                        }
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-048-GradeShape>
    a sh:NodeShape ;
    sh:class nltl:steelGradeValue ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade value used by the requirement must have exactly one integer order rank."
    ] .
<END_SHACL>
