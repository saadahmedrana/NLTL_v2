<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-ThicknessShape>
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        )
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        )
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeValueShape>
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048-GradeValueShape>
    ] ;
    sh:sparql [
        sh:message "An applicable Table 8 combination requires one required hull structural steel grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:polarClass ?pc .
                $this nltl:asBuiltPlateThickness ?q ;
                      nltl:steelMaterialClass ?mat ;
                      nltl:steelStrengthCategory ?cat .
                ?q qudt:numericValue ?t ;
                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (
                    !(
                        ?cat = nltl:normalStrengthSteelCategory &&
                        ?mat = nltl:steelMaterialClassThree &&
                        (
                            ((?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)) && ?t > 35) ||
                            ((?pc IN (nltl:polarClassPc4, nltl:polarClassPc5)) && ?t > 45)
                        )
                    )
                    &&
                    NOT EXISTS {
                        $this nltl:requiredHullStructuralSteelGrade ?required .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A NOT_APPLICABLE Table 8 combination must not contain a required hull structural steel grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:polarClass ?pc .
                $this nltl:asBuiltPlateThickness ?q ;
                      nltl:steelMaterialClass nltl:steelMaterialClassThree ;
                      nltl:steelStrengthCategory nltl:normalStrengthSteelCategory ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?q qudt:numericValue ?t ;
                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (
                    ((?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)) && ?t > 35) ||
                    ((?pc IN (nltl:polarClassPc4, nltl:polarClassPc5)) && ?t > 45)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual grade must be at least the Table 8 required grade within the same steel-strength family, and the required grade must match the selected Table 8 rank." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:polarClass ?pc .
                $this nltl:asBuiltPlateThickness ?q ;
                      nltl:steelMaterialClass ?mat ;
                      nltl:steelStrengthCategory ?cat ;
                      nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?q qudt:numericValue ?t ;
                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                BIND(
                    IF(
                        ?cat = nltl:normalStrengthSteelCategory,
                        IF(
                            ?mat = nltl:steelMaterialClassOne,
                            IF(?t <= 10, 1, IF(?t <= 35, 2, 3)),
                            IF(
                                ?mat = nltl:steelMaterialClassTwo,
                                IF(?t <= 10, 1, IF(?t <= 25, 2, 3)),
                                IF(
                                    ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3),
                                    IF(?t <= 35, 3, 0),
                                    IF(
                                        ?pc IN (nltl:polarClassPc4, nltl:polarClassPc5),
                                        IF(?t <= 45, 3, 0),
                                        IF(?t <= 10, 1, IF(?t <= 25, 2, 3))
                                    )
                                )
                            )
                        ),
                        IF(
                            ?mat = nltl:steelMaterialClassOne,
                            IF(
                                ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                IF(?t <= 35, 1, IF(?t <= 50, 2, 3)),
                                IF(?t <= 10, 1, IF(?t <= 35, 2, 3))
                            ),
                            IF(
                                ?mat = nltl:steelMaterialClassTwo,
                                IF(
                                    ?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
                                    IF(?t <= 25, 1, IF(?t <= 50, 2, 3)),
                                    IF(?t <= 10, 1, IF(?t <= 25, 2, 3))
                                ),
                                IF(
                                    ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3),
                                    IF(?t <= 35, 3, 4),
                                    IF(
                                        ?pc IN (nltl:polarClassPc4, nltl:polarClassPc5),
                                        IF(?t <= 45, 3, 4),
                                        IF(?t <= 10, 1, IF(?t <= 25, 2, 3))
                                    )
                                )
                            )
                        )
                    ) AS ?expectedRank
                )
                FILTER (
                    ?expectedRank = 0 ||
                    ?requiredRank != ?expectedRank ||
                    ?actualRank < ?requiredRank ||
                    (
                        ?cat = nltl:normalStrengthSteelCategory &&
                        (
                            ?actual NOT IN (
                                nltl:steelGradeB,
                                nltl:steelGradeD,
                                nltl:steelGradeE
                            ) ||
                            ?required NOT IN (
                                nltl:steelGradeB,
                                nltl:steelGradeD,
                                nltl:steelGradeE
                            )
                        )
                    ) ||
                    (
                        ?cat = nltl:highTensileSteelCategory &&
                        (
                            ?actual NOT IN (
                                nltl:steelGradeAh,
                                nltl:steelGradeDh,
                                nltl:steelGradeEh,
                                nltl:steelGradeFh
                            ) ||
                            ?required NOT IN (
                                nltl:steelGradeAh,
                                nltl:steelGradeDh,
                                nltl:steelGradeEh,
                                nltl:steelGradeFh
                            )
                        )
                    )
                )
            }
        """
    ] .
