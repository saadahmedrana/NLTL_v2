<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:hasCutout ;
                sh:class nltl:cutout ;
                sh:minCount 1 ;
                sh:message "Each represented structural member shall be linked to a cutout."
            ] ;
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The structural member shall be identified as being in way of a cutout."
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Whether stiffening is necessary shall be recorded for the structural member."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Loads carried by a member in way of a cutout shall not cause instability."
    ] ;
    sh:sparql [
        sh:message "Structural stiffening shall be recorded when stiffening is necessary." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (datatype(?stiffening) = xsd:string)
                }
            }
        """
    ] .
</urn:nltl:generated-shape:I2-060-ShipShape>
<END_SHACL>
