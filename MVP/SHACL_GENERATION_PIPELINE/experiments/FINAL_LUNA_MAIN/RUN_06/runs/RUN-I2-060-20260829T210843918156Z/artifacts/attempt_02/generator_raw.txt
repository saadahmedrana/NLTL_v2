<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "A member in way of a cut-out must be recorded as not causing instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                       nltl:hasCutout ?cutout .
                FILTER (
                    NOT EXISTS {
                        $this nltl:memberInstability ?validValue .
                        FILTER (
                            datatype(?validValue) = xsd:boolean &&
                            ?validValue = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:memberInstability ?invalidValue .
                        FILTER (
                            datatype(?invalidValue) != xsd:boolean ||
                            ?invalidValue != false
                        )
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Where stiffening is necessary for a member in way of a cut-out, structural stiffening must be recorded as a string." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                       nltl:hasCutout ?cutout ;
                       nltl:stiffeningNecessary true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:structuralStiffening ?validValue .
                        FILTER (
                            isLiteral(?validValue) &&
                            datatype(?validValue) = xsd:string
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:structuralStiffening ?invalidValue .
                        FILTER (
                            !isLiteral(?invalidValue) ||
                            datatype(?invalidValue) != xsd:string
                        )
                    }
                )
            }
        """
    ] .

gen:StructuralMemberShape a sh:NodeShape ;
    sh:or (
        [
            sh:and (
                [
                    sh:path nltl:memberInWayOfCutout ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue false
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:path nltl:memberInWayOfCutout ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
                [
                    sh:path nltl:hasCutout ;
                    sh:minCount 1
                ]
                [
                    sh:or (
                        [
                            sh:path nltl:stiffeningNecessary ;
                            sh:datatype xsd:boolean ;
                            sh:minCount 1 ;
                            sh:maxCount 1 ;
                            sh:hasValue false
                        ]
                        [
                            sh:path nltl:stiffeningNecessary ;
                            sh:datatype xsd:boolean ;
                            sh:minCount 1 ;
                            sh:maxCount 1 ;
                            sh:hasValue true
                        ]
                    )
                ]
            )
        ]
    ) ;
    sh:message "Each linked structural member must state whether it is in way of a cut-out; applicable members must identify a cut-out and state whether stiffening is necessary." .
<END_SHACL>
