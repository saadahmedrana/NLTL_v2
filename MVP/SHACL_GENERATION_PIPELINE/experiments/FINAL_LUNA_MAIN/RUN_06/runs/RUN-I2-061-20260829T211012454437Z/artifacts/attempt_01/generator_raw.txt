<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-061-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-CalculationCaseShape> ;
        sh:message "Each ship shall identify a calculation case for the prescribed hull-structure procedure."
    ] .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:class nltl:hullStructure ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-HullStructureShape> ;
        sh:message "Each calculation case shall identify the assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( nltl:prescribedAnalyticalProcedure ) ;
        sh:message "The calculation case shall use the prescribed analytical procedure and shall not use direct calculation as its alternative."
    ] .

<urn:nltl:generated-shape:I2-061-HullStructureShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-ThicknessQuantityShape> ;
        sh:message "The assessed hull structure shall provide its thickness."
    ] .

<urn:nltl:generated-shape:I2-061-ThicknessQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness shall have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness shall be expressed in metres."
    ] .
<END_SHACL>
