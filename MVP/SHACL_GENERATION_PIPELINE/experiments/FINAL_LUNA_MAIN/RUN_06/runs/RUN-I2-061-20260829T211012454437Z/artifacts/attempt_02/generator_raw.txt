<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-061-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:node <urn:nltl:generated-shape:I2-061-CalculationCaseShape> ;
        sh:message "Each represented calculation case must follow the prescribed calculation-case model."
    ] .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "A calculation case must identify its assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A calculation case must identify its calculation method."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For shell-plating or local-frame cases governed by I2.4, I2.6, or I2.7, the prescribed analytical procedure is required and direct calculation may not be used as an alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?structure .
                ?structure a ?structureClass .
                FILTER (
                    ?structureClass IN (
                        nltl:plating,
                        nltl:structuralMember
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:calculationMethod
                            nltl:prescribedAnalyticalProcedure
                    }
                    ||
                    EXISTS {
                        $this nltl:calculationMethod
                            nltl:directCalculationMethodValue
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-061-PlatingShape>
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Applicable shell plating must provide thickness in metres."
    ] .

<urn:nltl:generated-shape:I2-061-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Applicable local-frame structural members must provide thickness in metres."
    ] .
<END_SHACL>
