<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The ship must record that the required direct calculation is used."
    ] ;
    sh:sparql [
        sh:message "Every load carrying stringer or web frame linked from the ship must be identified as part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType = nltl:loadCarryingStringer ||
                    ?memberType = nltl:webFrame
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage a nltl:grillageSystem .
                }
            }
        """
    ] .
<END_SHACL>
