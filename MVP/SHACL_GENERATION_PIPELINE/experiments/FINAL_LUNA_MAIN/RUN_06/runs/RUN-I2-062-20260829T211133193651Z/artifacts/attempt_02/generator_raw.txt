<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_062_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked load carrying stringer or web frame forming part of a grillage system must identify a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member rdf:type ?memberType .
                FILTER (?memberType IN (
                    nltl:loadCarryingStringer,
                    nltl:webFrame
                ))
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage rdf:type nltl:grillageSystem .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with a linked load carrying stringer or web frame forming part of a grillage system must record that direct calculation was used." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member rdf:type ?memberType .
                FILTER (?memberType IN (
                    nltl:loadCarryingStringer,
                    nltl:webFrame
                ))
                ?member nltl:partOfGrillageSystem ?grillage .
                ?grillage rdf:type nltl:grillageSystem .
                FILTER NOT EXISTS {
                    $this nltl:directCalculationUsed ?used .
                    FILTER (?used = true)
                }
            }
        """
    ] .
<END_SHACL>
