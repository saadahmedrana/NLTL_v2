<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO_002_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition to determine the applicable category rule."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its Polar Code category to determine compliance with the conditional category rule."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:thinFirstYearIceWithPossibleOldIceInclusions
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryB
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryA
            ]
        ]
        [
            sh:sparql [
                sh:message "The ship is outside the thin first-year ice applicability branch because its design ice condition is not thin first-year ice with possible old-ice inclusions." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:designIceCondition ?iceCondition .
                            FILTER (?iceCondition != nltl:thinFirstYearIceWithPossibleOldIceInclusions)
                        }
                    }
                """
            ]
        ]
    ) ;
    sh:message "If a ship is not category A and is designed to operate in at least thin first-year ice with possible old-ice inclusions, its category must be category B." .
<END_SHACL>
