<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceOriginValue ;
        sh:message "Ice origin is required to determine whether the first-year ice rule applies."
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Winter growth count is required and must be an integer."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ] ;
        sh:message "Ice thickness must be a single decimal quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m, the ice type must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                       nltl:winterGrowthCount ?growthCount ;
                       nltl:iceThickness ?thicknessNode .
                ?thicknessNode qudt:numericValue ?thickness ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    ?growthCount <= 1 &&
                    ?thickness >= 0.3 &&
                    ?thickness <= 2.0
                )
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .

</END_SHACL>
