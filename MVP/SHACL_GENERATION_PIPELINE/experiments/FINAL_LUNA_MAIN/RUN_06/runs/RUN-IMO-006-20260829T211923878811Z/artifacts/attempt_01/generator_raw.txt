<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO006Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceTypeValue
        ] ;
        sh:message "A ship must provide an ice type classification to determine applicability of the medium first-year ice thickness range."
    ] ;
    sh:sparql [
        sh:message "For medium first-year ice, ice thickness must be a decimal quantity in metres from 0.7 m through 1.2 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?validQuantity .
                        ?validQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?validValue ;
                            qudt:unit unit:M .
                        FILTER (
                            datatype(?validValue) = xsd:decimal &&
                            ?validValue >= 0.7 &&
                            ?validValue <= 1.2
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?invalidQuantity .
                        FILTER NOT EXISTS {
                            ?invalidQuantity a qudt:QuantityValue ;
                                qudt:numericValue ?invalidValue ;
                                qudt:unit unit:M .
                            FILTER (
                                datatype(?invalidValue) = xsd:decimal &&
                                ?invalidValue >= 0.7 &&
                                ?invalidValue <= 1.2
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
