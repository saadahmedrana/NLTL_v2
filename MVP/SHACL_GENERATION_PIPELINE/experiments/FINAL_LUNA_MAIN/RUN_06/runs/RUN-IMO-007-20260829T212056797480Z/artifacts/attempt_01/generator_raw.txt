<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_007_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice condition classification is required to determine whether the open-water condition applies."
    ] ;
    sh:or (
        gen:IMO_007_OpenWaterBranch
        gen:IMO_007_NonOpenWaterBranch
    ) .

gen:IMO_007_OpenWaterBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:hasValue nltl:openWaterIceCondition ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:seaIceConcentration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxExclusive 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For open water, sea ice concentration must be a unitless decimal less than 1."
    ] ;
    sh:property [
        sh:path nltl:iceOfLandOriginPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "For open water, ice of land origin must be recorded as absent."
    ] .

gen:IMO_007_NonOpenWaterBranch
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This branch applies when the recorded ice condition is not open water." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?condition .
                FILTER (?condition != nltl:openWaterIceCondition)
            }
        """
    ] .
<END_SHACL>
