<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:iceTypeValue
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships classified as thin first-year ice shall have an ice thickness from 0.3 m through 0.7 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceTypeClassification nltl:thinFirstYearIce .
    FILTER NOT EXISTS {
        $this nltl:iceThickness ?quantity .
        ?quantity a qudt:QuantityValue ;
            qudt:numericValue ?value ;
            qudt:unit unit:M .
        FILTER (
            datatype(?value) = xsd:decimal &&
            ?value >= "0.3"^^xsd:decimal &&
            ?value <= "0.7"^^xsd:decimal
        )
    }
}
""" ;
    ] .
<END_SHACL>
