<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008_ThinFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as thin first-year ice must have only ice thickness quantities with one decimal numeric value in metres, inclusive from 0.3 m to 0.7 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:thinFirstYearIce .
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?thickness
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?thickness .
                        FILTER NOT EXISTS {
                            ?thickness qudt:numericValue ?numericValue
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?thickness .
                        FILTER NOT EXISTS {
                            ?thickness qudt:unit unit:M
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?thickness .
                        ?thickness qudt:numericValue ?numericValue .
                        FILTER (datatype(?numericValue) != xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?thickness .
                        ?thickness qudt:numericValue ?numericValue .
                        FILTER (
                            datatype(?numericValue) = xsd:decimal &&
                            (
                                ?numericValue < "0.3"^^xsd:decimal ||
                                ?numericValue > "0.7"^^xsd:decimal
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?thickness .
                        ?thickness qudt:unit ?suppliedUnit .
                        FILTER (?suppliedUnit != unit:M)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
