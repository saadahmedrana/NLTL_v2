<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:IMO016ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:node g:IMO016CertificateShape ;
        sh:message "The ship shall have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The certificate language shall be recorded."
    ] ;
    sh:property [
        sh:path nltl:approvedTranslationPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the certificate language is neither English, French, nor Spanish, an approved translation shall be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (DATATYPE(?language) = xsd:string)
                FILTER (STR(?language) NOT IN ("English", "French", "Spanish"))
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent ?translation .
                    FILTER (?translation = true)
                }
            }
        """
    ] .

g:IMO016CertificateShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate shall use the Appendix I model."
    ] .
<END_SHACL>
