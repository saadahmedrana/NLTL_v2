<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-017-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "The ship shall have a Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-017-SolasScheduleShape> ;
        sh:message "The ship shall have an applicable SOLAS certificate schedule."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-017-SupplementShape> ;
        sh:message "The Polar Ship Certificate shall include a supplement with required equipment records."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement shall be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "Polar Ship Certificate validity dates are required and shall be dates."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "Polar Ship Certificate survey dates are required and shall be dates."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "Polar Ship Certificate endorsement dates are required and shall be dates."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each Polar Ship Certificate validity, survey, or endorsement date shall match a correspondingly categorized date in the applicable SOLAS certificate schedule." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this ?shipDate ?value .
    FILTER (
        ?shipDate = nltl:polarCertificateValidityDates ||
        ?shipDate = nltl:surveyDates ||
        ?shipDate = nltl:endorsementDates
    )
    FILTER NOT EXISTS {
        $this nltl:hasSolasCertificateSchedule ?schedule .
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDate ?recordDate ;
                nltl:certificateScheduleDateType ?category .
        FILTER (
            (?shipDate = nltl:polarCertificateValidityDates &&
             ?category = nltl:validityDateCategory) ||
            (?shipDate = nltl:surveyDates &&
             ?category = nltl:surveyDateCategory) ||
            (?shipDate = nltl:endorsementDates &&
             ?category = nltl:endorsementDateCategory)
        )
        FILTER (?value = ?recordDate)
    }
}
""" ;
    ] .

<urn:nltl:generated-shape:IMO-017-SolasScheduleShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-017-DateRecordShape> ;
        sh:message "The SOLAS certificate schedule shall contain categorized schedule-date records."
    ] .

<urn:nltl:generated-shape:IMO-017-DateRecordShape>
    a sh:NodeShape ;
    sh:class nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "Each certificate schedule date record shall contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall have a permitted date category."
    ] .

<urn:nltl:generated-shape:IMO-017-SupplementShape>
    a sh:NodeShape ;
    sh:class nltl:polarCertificateSupplement ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Certificate supplement shall record required equipment."
    ] .
<END_SHACL>
