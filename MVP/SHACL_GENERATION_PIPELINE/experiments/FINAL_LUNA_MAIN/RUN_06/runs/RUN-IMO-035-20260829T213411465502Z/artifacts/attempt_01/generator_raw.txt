@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-035> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarWaterOperationalManualRecord ;
        sh:class nltl:documentRecord ;
        sh:minCount 1 ;
        sh:message "A Polar Water Operational Manual record must be linked."
    ] ;
    sh:property [
        sh:path nltl:hasStabilityCalculationRecord ;
        sh:class nltl:documentRecord ;
        sh:minCount 1 ;
        sh:message "A stability calculation record must be linked."
    ] ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualIcingAllowanceInformationPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The PWOM must state that icing allowance information is present."
    ] ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualIcingAllowanceValues ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "PWOM icing allowance values must be provided."
    ] ;
    sh:property [
        sh:path nltl:stabilityCalculationIcingAllowanceValues ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stability calculation icing allowance values must be provided."
    ] ;
    sh:sparql [
        sh:message "The PWOM icing allowance values must match the stability calculation icing allowance values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:polarWaterOperationalManualIcingAllowanceValues ?pwomValues .
                $this nltl:stabilityCalculationIcingAllowanceValues ?stabilityValues .
                FILTER (?pwomValues != ?stabilityValues)
            }
        """
    ] .
