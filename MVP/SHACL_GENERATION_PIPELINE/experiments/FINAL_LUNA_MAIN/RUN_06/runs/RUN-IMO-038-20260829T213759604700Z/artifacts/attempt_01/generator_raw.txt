<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 760 ;
        sh:maxInclusive 760
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SelectedVerticalPositionQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO038Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Damage-center position relative to maximum breadth is required." 
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:node gen:MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Transverse penetration must be exactly 760 mm."
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:node gen:SelectedVerticalPositionQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Longitudinal and vertical damage extents must satisfy the specified piecewise formulas within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longNode ;
                      nltl:verticalDamageExtent ?verticalNode .

                ?lengthNode qudt:numericValue ?lengthValue .
                ?draughtNode qudt:numericValue ?draughtValue .
                ?longNode qudt:numericValue ?longValue .
                ?verticalNode qudt:numericValue ?verticalValue .

                BIND(IF(?forward,
                        0.045 * ?lengthValue,
                        0.015 * ?lengthValue) AS ?expectedLong)
                BIND(IF((0.20 * ?draughtValue) < ?expectedLong,
                        0.20 * ?draughtValue,
                        ?expectedLong) AS ?expectedVertical)

                FILTER(
                    ABS(?longValue - ?expectedLong) >
                    (0.000001 * ABS(?expectedLong) + 0.000001)
                    ||
                    ABS(?verticalValue - ?expectedVertical) >
                    (0.000001 * ABS(?expectedVertical) + 0.000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected vertical damage position must be between the keel and 1.20 times the upper ice waterline draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:selectedVerticalDamagePosition ?positionNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode .

                ?positionNode qudt:numericValue ?positionValue .
                ?draughtNode qudt:numericValue ?draughtValue .

                FILTER(?positionValue > (1.20 * ?draughtValue))
            }
        """
    ] .
<END_SHACL>
