<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_038_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The damage center position indicator must be supplied as one boolean value."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Upper ice waterline length must be one metre-based quantity."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Upper ice waterline draught must be one metre-based quantity."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Longitudinal damage extent must be one metre-based quantity."
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 760 ;
                sh:maxInclusive 760
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Transverse penetration must equal 760 mm."
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Vertical damage extent must be one metre-based quantity."
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Selected vertical damage position must be at or above the keel."
    ] ;
    sh:sparql [
        sh:message "Longitudinal damage extent must equal 4.5% or 1.5% of upper ice waterline length, according to the damage-center position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:longitudinalDamageExtent ?extentNode .
                ?lengthNode qudt:numericValue ?length .
                ?extentNode qudt:numericValue ?extent .
                BIND(IF(?forward = true, 0.045 * ?length, 0.015 * ?length) AS ?expected)
                BIND(IF(?expected >= 0, ?expected, -?expected) AS ?absoluteExpected)
                BIND(IF((?absoluteExpected * 0.000001) >= 0.000000001,
                        ?absoluteExpected * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER (?extent < (?expected - ?tolerance) ||
                        ?extent > (?expected + ?tolerance))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Vertical damage extent must equal the lesser of 20% of upper ice waterline draught and the longitudinal damage extent, and the selected vertical position must not exceed 120% of draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .
                ?draughtNode qudt:numericValue ?draught .
                ?longitudinalNode qudt:numericValue ?longitudinal .
                ?verticalNode qudt:numericValue ?vertical .
                ?positionNode qudt:numericValue ?position .
                BIND(0.20 * ?draught AS ?draughtLimit)
                BIND(IF(?draughtLimit <= ?longitudinal,
                        ?draughtLimit,
                        ?longitudinal) AS ?expectedVertical)
                BIND(IF(?expectedVertical >= 0,
                        ?expectedVertical,
                        -?expectedVertical) AS ?absoluteExpectedVertical)
                BIND(IF((?absoluteExpectedVertical * 0.000001) >= 0.000000001,
                        ?absoluteExpectedVertical * 0.000001,
                        0.000000001) AS ?verticalTolerance)
                FILTER (?vertical < (?expectedVertical - ?verticalTolerance) ||
                        ?vertical > (?expectedVertical + ?verticalTolerance) ||
                        ?position > (1.20 * ?draught))
            }
        """
    ] .
