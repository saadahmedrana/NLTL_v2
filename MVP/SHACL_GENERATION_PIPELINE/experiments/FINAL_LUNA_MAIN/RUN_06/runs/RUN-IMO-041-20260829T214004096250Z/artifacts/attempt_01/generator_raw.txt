<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-041> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Opening actuation type is required to determine applicability of the hydraulic-fluid protection condition."
    ] ;
    sh:property [
        sh:path nltl:hydraulicFluidFreezingOrViscosityControlPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships with hydraulically operated hatches or doors shall provide means to prevent freezing or excessive viscosity of hydraulic fluids." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuation .
                FILTER (STR(?actuation) = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent ?protection .
                    FILTER (?protection = true)
                }
            }
        """
    ] .
<END_SHACL>
