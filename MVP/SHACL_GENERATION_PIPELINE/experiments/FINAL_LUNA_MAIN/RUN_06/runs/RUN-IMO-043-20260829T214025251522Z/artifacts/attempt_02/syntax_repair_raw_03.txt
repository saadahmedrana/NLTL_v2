<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-043_ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:node <urn:nltl:generated-shape:IMO-043_MachineryInstallationShape>
    ] .

<urn:nltl:generated-shape:IMO-043_MachineryInstallationShape> a sh:NodeShape ;
    sh:class nltl:machineryInstallationItem ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceAccretionHazard
            nltl:snowAccumulationHazard
            nltl:seawaterIceIngestionHazard
            nltl:liquidFreezingOrViscosityIncreaseHazard
            nltl:seawaterIntakeTemperatureHazard
            nltl:snowIngestionHazard
        )
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "Each machinery installation, including its associated equipment, must have hazard-protection evidence."
    ] ;
    sh:sparql [
        sh:message "Each associated equipment item must be covered by the hazard-protection evidence recorded for its machinery installation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasAssociatedEquipment ?equipment .
                FILTER NOT EXISTS {
                    $this nltl:hasHazardProtectionEvidence ?evidence .
                }
            }
        """
    ] .
<END_SHACL>
