<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO047ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:IMO047ApplicableShipShape
        gs:IMO047NonApplicableShipShape
    ) .

gs:IMO047ApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable ship shall record combustion-air temperature control as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:node gs:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship shall provide its combustion-air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:node gs:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship shall provide the engine manufacturer's minimum air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:node gs:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship shall provide the engine manufacturer's maximum air temperature."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combustion-air temperature shall be within the engine manufacturer's minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?engine .
                FILTER NOT EXISTS {
                    $this nltl:combustionAirTemperature ?temperatureQuantity .
                    ?temperatureQuantity qudt:numericValue ?temperature .
                    $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                    ?minimumQuantity qudt:numericValue ?minimum .
                    $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .
                    ?maximumQuantity qudt:numericValue ?maximum .
                    FILTER (
                        ?temperature >= ?minimum &&
                        ?temperature <= ?maximum
                    )
                }
            }
        """
    ] .

gs:IMO047NonApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:maxCount 0
    ] .

gs:IMO047TemperatureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
    ] .
<END_SHACL>
