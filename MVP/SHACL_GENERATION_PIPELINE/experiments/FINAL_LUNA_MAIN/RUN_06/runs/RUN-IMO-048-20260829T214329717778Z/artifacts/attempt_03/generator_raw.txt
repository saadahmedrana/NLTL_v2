<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-048_ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarServiceTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide its polar service temperature in degrees Celsius."
    ] ;
    sh:sparql [
        sh:message "Each exposed machinery or foundation material must be paired with approval evidence from an authorized authority, using an accepted or equivalent-safety standard and the ship's polar service temperature basis." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this rdf:type nltl:ship .
                $this nltl:exposedMachineryOrFoundationMaterial ?material .
                FILTER NOT EXISTS {
                    $this nltl:hasEvidence ?evidence .
                    ?evidence rdf:type nltl:approvalEvidence ;
                        nltl:exposedMachineryOrFoundationMaterial ?material ;
                        nltl:approvingAuthority ?authority ;
                        nltl:approvalStandard ?standard ;
                        nltl:materialApprovalStatus nltl:evidenceStateApproved ;
                        nltl:polarServiceTemperature ?evidenceTemperature .
                    FILTER (
                        ?authority IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                    FILTER (
                        ?standard IN (
                            "standard acceptable to the Organization",
                            "another standard offering an equivalent level of safety"
                        )
                    )
                    $this nltl:polarServiceTemperature ?shipTemperature .
                    ?shipTemperature qudt:numericValue ?shipValue ;
                        qudt:unit unit:DEG_C .
                    ?evidenceTemperature qudt:numericValue ?evidenceValue ;
                        qudt:unit unit:DEG_C .
                    FILTER (?evidenceValue = ?shipValue)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-048_ApprovalEvidenceShape> a sh:NodeShape ;
    sh:targetClass nltl:approvalEvidence ;
    sh:property [
        sh:path nltl:exposedMachineryOrFoundationMaterial ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Approval evidence must identify one exposed machinery or foundation material."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:message "Approval evidence must identify an authorized approving authority."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:message "Approval evidence must identify an accepted or equivalent-safety standard."
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Approval evidence must have approved material status."
    ] ;
    sh:property [
        sh:path nltl:polarServiceTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Approval evidence must record the polar service temperature basis in degrees Celsius."
    ] .
<END_SHACL>
