nltl:IMO049ABMachineryComponentShape a sh:NodeShape ;
    sh:class nltl:machineryComponent ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node nltl:IMO049ABApprovalRecordShape ;
        sh:message "Each applicable machinery component must have approval evidence."
    ] .

    sh:class nltl:machineryComponent ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;

        sh:message "Each applicable ice-strengthened Category C machinery component must have approval evidence."
    ] .

nltl:IMO049ABApprovalRecordShape a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The scantling approval evidence must have approved status."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The approval must identify an authorized approving authority."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The approval must use an Organization-acceptable or equivalent-safety standard."
    ] .
