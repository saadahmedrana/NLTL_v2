@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-066-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must state whether it is exposed to ice accretion."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:node [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:in (
                            nltl:escapeRouteProtectedItem
                            nltl:musterStationProtectedItem
                            nltl:embarkationAreaProtectedItem
                            nltl:survivalCraftProtectedItem
                            nltl:launchingApplianceProtectedItem
                            nltl:survivalCraftAccessProtectedItem
                        ) ;
                        sh:message "Each protected item must have one permitted protected-item category."
                    ] ;
                    sh:property [
                        sh:path nltl:hasIceSnowRemovalOrPreventionMeans ;
                        sh:minCount 1 ;
                        sh:message "Each protected item must have associated ice or snow removal or prevention means."
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "For a ship exposed to ice accretion, all six specified protected-item categories must have associated ice or snow removal or prevention means." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        FILTER (
                            EXISTS {
                                $this nltl:hasProtectedItem ?escapeRouteItem .
                                ?escapeRouteItem
                                    nltl:protectedItemCategory nltl:escapeRouteProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?escapeRouteMeans .
                            }
                            &&
                            EXISTS {
                                $this nltl:hasProtectedItem ?musterStationItem .
                                ?musterStationItem
                                    nltl:protectedItemCategory nltl:musterStationProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?musterStationMeans .
                            }
                            &&
                            EXISTS {
                                $this nltl:hasProtectedItem ?embarkationAreaItem .
                                ?embarkationAreaItem
                                    nltl:protectedItemCategory nltl:embarkationAreaProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?embarkationAreaMeans .
                            }
                            &&
                            EXISTS {
                                $this nltl:hasProtectedItem ?survivalCraftItem .
                                ?survivalCraftItem
                                    nltl:protectedItemCategory nltl:survivalCraftProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?survivalCraftMeans .
                            }
                            &&
                            EXISTS {
                                $this nltl:hasProtectedItem ?launchingApplianceItem .
                                ?launchingApplianceItem
                                    nltl:protectedItemCategory nltl:launchingApplianceProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?launchingApplianceMeans .
                            }
                            &&
                            EXISTS {
                                $this nltl:hasProtectedItem ?survivalCraftAccessItem .
                                ?survivalCraftAccessItem
                                    nltl:protectedItemCategory nltl:survivalCraftAccessProtectedItem ;
                                    nltl:hasIceSnowRemovalOrPreventionMeans ?survivalCraftAccessMeans .
                            }
                        )
                    }
                """
            ]
        ]
    ) .
