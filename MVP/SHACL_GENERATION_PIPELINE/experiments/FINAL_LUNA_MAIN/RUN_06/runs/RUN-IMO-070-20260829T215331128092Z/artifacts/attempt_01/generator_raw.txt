<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-070> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A passenger ship shall record compatible survival equipment for every person on board." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this a nltl:ship ;
          a nltl:passengerShip .
    {
        FILTER NOT EXISTS {
            $this nltl:sizeCompatibilityStatus true .
        }
    }
    UNION
    {
        $this nltl:hasPersonOnBoardMember ?person .
        FILTER NOT EXISTS {
            ?person a nltl:personOnBoardMember ;
                    nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
            ?assignment a nltl:personalSurvivalEquipmentAssignment ;
                        nltl:assignedPersonalSurvivalEquipment ?equipment .
            ?equipment a nltl:personalSurvivalEquipment .
            {
                ?equipment a nltl:immersionSuitEquipment .
            }
            UNION
            {
                ?equipment a nltl:thermalProtectiveAidEquipment .
            }
            $this nltl:sizeCompatibilityStatus true .
        }
    }
}
""" ;
    ] .
<END_SHACL>
