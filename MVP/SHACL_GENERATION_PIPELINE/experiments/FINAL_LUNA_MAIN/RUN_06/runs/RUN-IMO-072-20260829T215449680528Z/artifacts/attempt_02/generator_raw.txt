@prefix nltl: <https://w3id.org/nltl/vocab#> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_072_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide one extended-darkness operation applicability value."
    ] ;
    sh:or (
        gen:IMO_072_NotApplicableShape
        gen:IMO_072_ApplicableShape
    ) .

gen:IMO_072_NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:IMO_072_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:node gen:IMO_072_LifeboatShape ;
        sh:message "Every lifeboat on a ship operating in extended darkness must have an assigned suitable searchlight."
    ] .

gen:IMO_072_LifeboatShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAssignedSearchlight ;
        sh:minCount 1 ;
        sh:node gen:IMO_072_SearchlightShape ;
        sh:message "Each lifeboat must have an assigned searchlight."
    ] .

gen:IMO_072_SearchlightShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:searchlightContinuousUseSuitabilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each assigned searchlight must be suitable for continuous use."
    ] .
