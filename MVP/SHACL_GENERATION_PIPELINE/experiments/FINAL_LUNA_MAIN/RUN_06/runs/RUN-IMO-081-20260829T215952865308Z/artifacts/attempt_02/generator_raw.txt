<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO081ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice-strengthening status is required to determine applicability."
    ] ;
    sh:or (
        gen:IMO081NonApplicableGuard
        gen:IMO081ApplicableShape
    ) .

gen:IMO081NonApplicableGuard a sh:NodeShape ;
    sh:sparql [
        sh:message "This applicability branch applies only when the ship is not constructed on or after 1 January 2017 while ice strengthened." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date ;
                      nltl:shipIceStrengthened ?ice .
                FILTER (
                    ?date >= "2017-01-01"^^xsd:date &&
                    ?ice = true
                )
            }
        """
    ] .

gen:IMO081ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:IMO081TwoIndependentEchoDevices
        gen:IMO081OneEchoDeviceTwoTransducers
    ) .

gen:IMO081TwoIndependentEchoDevices a sh:NodeShape ;
    sh:property [
        sh:path nltl:independentEchoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:minCount 1 ;
        sh:message "An applicable ship shall have at least two independent echo-sounding devices."
    ] .

gen:IMO081OneEchoDeviceTwoTransducers a sh:NodeShape ;
    sh:property [
        sh:path nltl:echoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 1 ;
        sh:minCount 1 ;
        sh:message "An applicable ship shall have at least one echo-sounding device."
    ] ;
    sh:property [
        sh:path nltl:independentTransducerCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:minCount 1 ;
        sh:message "An applicable ship shall have at least two independent transducers."
    ] .
<END_SHACL>
