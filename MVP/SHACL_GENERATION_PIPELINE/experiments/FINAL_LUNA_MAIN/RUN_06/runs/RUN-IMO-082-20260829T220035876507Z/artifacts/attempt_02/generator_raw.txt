@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-082-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bridge configuration required to determine astern-view applicability must be provided."
    ] ;
    sh:sparql [
        sh:message "A clear view astern is required when the bridge configuration is astern-view-required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:bridgeConfigurationType ?configuration .
                FILTER (?configuration = nltl:asternViewRequiredBridgeConfiguration)
                FILTER NOT EXISTS {
                    $this nltl:clearViewAsternStatus ?clearView .
                    FILTER (?clearView = true)
                }
            }
        """
    ] .
