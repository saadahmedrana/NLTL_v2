<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO086ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gs:IMO086HeadingMeansShape ;
        sh:message "The ship shall have at least two non-magnetic heading means."
    ] .

gs:IMO086HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the main power source."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the emergency power source."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall identify an independent heading means."
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means shall have an independence relation to a distinct non-magnetic heading means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?ship nltl:hasNonMagneticHeadingMeans $this .
                    $this nltl:independentFromHeadingMeans ?other .
                    FILTER (?other != $this)
                    ?ship nltl:hasNonMagneticHeadingMeans ?other .
                }
            }
        """
    ] .

</END_SHACL>
