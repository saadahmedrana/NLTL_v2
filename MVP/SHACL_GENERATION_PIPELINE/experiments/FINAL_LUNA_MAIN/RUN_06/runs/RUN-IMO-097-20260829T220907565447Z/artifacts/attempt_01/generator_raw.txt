<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO097ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating status must be recorded as a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The released rescue boat or lifeboat status must be recorded as a single boolean value."
    ] ;
    sh:sparql [
        sh:message "When the ship operates in low air temperature and a rescue boat or lifeboat is released, every represented survival craft must carry a ship-to-shore distress alert device, a location signal device, and an on-scene communication device." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true ;
                      nltl:releasedRescueBoatOrLifeboat true ;
                      nltl:hasSurvivalCraft ?craft .
                FILTER (
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?distressDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                    }
                )
            }
        """
    ] .
<END_SHACL>
