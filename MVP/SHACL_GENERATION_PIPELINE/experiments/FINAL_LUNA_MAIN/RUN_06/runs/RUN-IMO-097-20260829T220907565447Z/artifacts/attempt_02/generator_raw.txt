<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-097-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating applicability must be recorded."
    ] ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The rescue-boat or lifeboat release applicability must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship operates in low air temperature and the rescue boat or lifeboat is released, each required rescue boat or lifeboat must have at least one ship-to-shore distress-alert device, one location-signal device, and one on-scene transmit/receive communication device." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true ;
                      nltl:releasedRescueBoatOrLifeboat true .
                {
                    $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?device .
                    }
                }
                UNION
                {
                    $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?device .
                    }
                }
                UNION
                {
                    $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?device .
                    }
                }
                UNION
                {
                    $this nltl:hasLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?device .
                    }
                }
                UNION
                {
                    $this nltl:hasLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?device .
                    }
                }
                UNION
                {
                    $this nltl:hasLifeboat ?craft .
                    FILTER NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?device .
                    }
                }
            }
        """
    ] .

</END_SHACL>
