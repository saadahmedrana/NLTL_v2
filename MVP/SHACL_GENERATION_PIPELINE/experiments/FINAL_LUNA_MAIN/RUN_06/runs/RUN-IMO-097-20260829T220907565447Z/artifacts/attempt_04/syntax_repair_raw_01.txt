<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO097ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether it operates in low air temperature."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft
    ] ;
    sh:or (
        gen:IMO097ApplicableCase
        gen:IMO097NonApplicableCase
    ) .

gen:IMO097ApplicableCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each required released rescue boat or lifeboat must have at least one ship-to-shore distress-alert device, one location-signal device, and one on-scene transmit/receive communication device." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true .
                $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                FILTER (
                    NOT EXISTS {
                        ?craft nltl:hasDistressAlertDevice ?distressDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                    }
                    ||
                    NOT EXISTS {
                        ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?shipToShoreDevice .
                    }
                )
            }
        """
    ] .

gen:IMO097NonApplicableCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:IMO097LifeboatShape
    a sh:NodeShape ;
    sh:targetClass nltl:lifeboat ;
    sh:property [
        sh:path nltl:hasDistressAlertDevice
    ] .

gen:IMO097SurvivalCraftShape
    a sh:NodeShape ;
    sh:targetClass nltl:survivalCraft ;
    sh:property [
        sh:path nltl:survivalCraftHasLocationSignalDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasOnSceneCommunicationDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasShipToShoreDistressAlertDevice
    ] .
<END_SHACL>
