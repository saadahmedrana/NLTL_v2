<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-101-PolarRoutePlanCoverageShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarRoutePlan ;
        sh:class nltl:polarRoutePlan ;
        sh:minCount 1 ;
        sh:message "The ship must have a polar route plan."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked polar route plan must have voyage planning topic coverage exactly true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:hasPolarRoutePlan ?plan .
    FILTER NOT EXISTS {
        ?plan nltl:voyagePlanningTopicCoverage ?coverage .
        FILTER (
            datatype(?coverage) = xsd:boolean
            && ?coverage = true
        )
    }
}
"""
    ] .
<END_SHACL>
