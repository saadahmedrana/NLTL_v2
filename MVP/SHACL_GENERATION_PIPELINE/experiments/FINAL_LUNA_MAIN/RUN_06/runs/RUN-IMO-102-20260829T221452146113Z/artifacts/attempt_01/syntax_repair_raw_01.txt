<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-102-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "The ship must have exactly one controlled ice condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:message "The ship must have exactly one controlled polar training ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node [
            sh:property [
                sh:path nltl:polarTrainingCrewRoleClassification ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:masterPolarTrainingRole
                    nltl:chiefMatePolarTrainingRole
                    nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                ) ;
                sh:message "Each represented crew member must have exactly one controlled polar training crew role."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable crew member must have a training record whose required level matches its recorded level and whose STCW qualification is valid." ;
        sh:select '''
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT DISTINCT ?this
            WHERE {
                ?this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew ;
                      nltl:iceConditionClassification ?selectedIce ;
                      nltl:polarTrainingShipTypeClassification ?selectedShipType .

                ?crew nltl:polarTrainingCrewRoleClassification ?role .

                {
                    FILTER (
                        ?selectedIce = nltl:iceFreeIceCondition &&
                        ?selectedShipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType,
                            nltl:otherShipPolarTrainingType
                        ) &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    )
                    BIND(nltl:polarTrainingNotApplicable AS ?expectedLevel)
                }
                UNION
                {
                    FILTER (
                        ?selectedIce = nltl:openWaterIceCondition &&
                        ?selectedShipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType
                        ) &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    )
                    BIND(nltl:basicPolarTrainingLevel AS ?expectedLevel)
                }
                UNION
                {
                    FILTER (
                        ?selectedIce = nltl:openWaterIceCondition &&
                        ?selectedShipType = nltl:otherShipPolarTrainingType &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    )
                    BIND(nltl:polarTrainingNotApplicable AS ?expectedLevel)
                }
                UNION
                {
                    FILTER (
                        ?selectedIce = nltl:otherWatersIceCondition &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole
                        ) &&
                        ?selectedShipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType,
                            nltl:otherShipPolarTrainingType
                        )
                    )
                    BIND(nltl:advancedPolarTrainingLevel AS ?expectedLevel)
                }
                UNION
                {
                    FILTER (
                        ?selectedIce = nltl:otherWatersIceCondition &&
                        ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                        ?selectedShipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType,
                            nltl:otherShipPolarTrainingType
                        )
                    )
                    BIND(nltl:basicPolarTrainingLevel AS ?expectedLevel)
                }

                FILTER (?expectedLevel != nltl:polarTrainingNotApplicable)

                FILTER NOT EXISTS {
                    ?crew nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?expectedLevel ;
                            nltl:trainingRecordLevel ?expectedLevel ;
                            nltl:stcwQualificationValid true .
                }
            }
        '''
    ] .
</urn:nltl:generated-shape:IMO-102-ShipShape>
<END_SHACL>
