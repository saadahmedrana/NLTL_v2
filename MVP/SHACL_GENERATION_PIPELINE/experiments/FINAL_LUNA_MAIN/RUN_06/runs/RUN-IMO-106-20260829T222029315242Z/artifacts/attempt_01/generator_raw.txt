<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO_106_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:class nltl:dischargedMaterialTypeValue ;
        sh:message "A ship must provide its discharged material classification."
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition must be recorded as not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable false^^xsd:boolean .
                }
            }
        """
    ] .
<END_SHACL>
