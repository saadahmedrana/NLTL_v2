<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_108_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 2017-01-01, with aggregate oil fuel capacity below 600 m3 and individual oil fuel tank capacity above 30 m3, the oil fuel tank outer shell separation must be at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:aggregateOilFuelCapacity ?aggregateQuantity ;
                      nltl:individualOilFuelTankCapacity ?individualQuantity .

                ?aggregateQuantity qudt:numericValue ?aggregateValue ;
                                    qudt:unit unit:M3 .
                ?individualQuantity qudt:numericValue ?individualValue ;
                                     qudt:unit unit:M3 .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (
                    YEAR(?constructionDate) > 2017 ||
                    (
                        YEAR(?constructionDate) = 2017 &&
                        (
                            MONTH(?constructionDate) > 1 ||
                            (
                                MONTH(?constructionDate) = 1 &&
                                DAY(?constructionDate) >= 1
                            )
                        )
                    )
                )
                FILTER (?aggregateValue < "600"^^xsd:decimal)
                FILTER (?individualValue > "30"^^xsd:decimal)

                FILTER NOT EXISTS {
                    $this nltl:oilFuelTankOuterShellSeparation ?separationQuantity .
                    ?separationQuantity qudt:numericValue ?separationValue ;
                                         qudt:unit unit:M .
                    FILTER (
                        datatype(?separationValue) = xsd:decimal &&
                        ?separationValue >= "0.76"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
