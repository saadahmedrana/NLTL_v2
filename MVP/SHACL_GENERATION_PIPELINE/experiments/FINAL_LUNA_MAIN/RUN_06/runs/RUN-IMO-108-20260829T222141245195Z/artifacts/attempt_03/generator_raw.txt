<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO108Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one permitted polar ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date."
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The aggregate oil fuel capacity must be a decimal quantity in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each represented individual oil fuel tank capacity must be a decimal quantity in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:oilFuelTankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Any represented oil fuel tank outer-shell separation must be a decimal quantity in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, every represented non-small oil fuel tank must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )

                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)

                $this nltl:aggregateOilFuelCapacity ?aggregate .
                ?aggregate a qudt:QuantityValue ;
                    qudt:numericValue ?aggregateValue ;
                    qudt:unit unit:M3 .
                FILTER (
                    datatype(?aggregateValue) = xsd:decimal &&
                    ?aggregateValue < 600
                )

                $this nltl:individualOilFuelCapacity ?individual .
                ?individual a qudt:QuantityValue ;
                    qudt:numericValue ?individualValue ;
                    qudt:unit unit:M3 .
                FILTER (
                    datatype(?individualValue) = xsd:decimal &&
                    ?individualValue > 30
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?separation .
                        ?separation a qudt:QuantityValue ;
                            qudt:numericValue ?separationValue ;
                            qudt:unit unit:M .
                        FILTER (
                            datatype(?separationValue) = xsd:decimal &&
                            ?separationValue >= 0.76
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?malformedSeparation .
                        FILTER NOT EXISTS {
                            ?malformedSeparation a qudt:QuantityValue ;
                                qudt:numericValue ?validSeparationValue ;
                                qudt:unit unit:M .
                            FILTER (datatype(?validSeparationValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?belowSeparation .
                        ?belowSeparation a qudt:QuantityValue ;
                            qudt:numericValue ?belowValue ;
                            qudt:unit unit:M .
                        FILTER (
                            datatype(?belowValue) = xsd:decimal &&
                            ?belowValue < 0.76
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
