<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_109_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship category is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:cargoTankCarriesOil ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Cargo-tank oil service information is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B non-oil-tanker ship constructed on or after 1 January 2017 whose cargo tanks carry oil, the cargo tank outer shell separation must be at least 0.76 m and use unit M." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true .

                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?shipType != "oil_tanker")
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                FILTER NOT EXISTS {
                    $this nltl:cargoTankOuterShellSeparation ?separation .
                    ?separation a qudt:QuantityValue ;
                                qudt:numericValue ?value ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (?value >= 0.76)
                }
            }
        """
    ] .
<END_SHACL>
