<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO-112Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must specify its operating area so the Arctic waters discharge prohibition can be evaluated."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of noxious liquid substances or mixtures containing them must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (
                    datatype(?area) = xsd:string &&
                    STR(?area) = "Arctic_waters"
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                        FILTER (
                            datatype(?discharge) = xsd:boolean &&
                            ?discharge = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                        FILTER (!(
                            datatype(?discharge) = xsd:boolean &&
                            ?discharge = false
                        ))
                    }
                )
            }
        """
    ] .
<END_SHACL>
