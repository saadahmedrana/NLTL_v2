@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_116_Sewage_Ice_Distance_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:notComminutedOrDisinfected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage is not comminuted or disinfected, MARPOL Annex IV regulation 11.1.1 compliance, a distance greater than 12 nautical miles from ice shelf or fast ice, and the applicable ice-area distance record are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship ;
                      nltl:sewageDischargeType ?dischargeType .

                FILTER (?dischargeType = "not_comminuted_or_disinfected"^^xsd:string)

                FILTER (
                    NOT EXISTS {
                        $this nltl:notComminutedOrDisinfected "true"^^xsd:boolean
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus
                              "true"^^xsd:boolean
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestIceShelfOrFastIce ?iceDistance .
                        ?iceDistance qudt:numericValue ?distanceValue ;
                                     qudt:unit unit:MI_N .
                        FILTER (?distanceValue > "12"^^xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?concentrationValue ;
                                       qudt:unit unit:UNITLESS .
                        FILTER (?concentrationValue > "1"^^xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance qudt:numericValue ?areaDistanceValue ;
                                      qudt:unit unit:MI_N .
                    }
                )
            }
        """
    ] .
