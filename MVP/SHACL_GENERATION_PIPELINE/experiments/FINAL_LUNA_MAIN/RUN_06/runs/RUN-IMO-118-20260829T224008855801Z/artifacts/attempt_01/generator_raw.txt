<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-118> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine whether the sewage-discharge prohibition applies."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:message "A valid polar ship category is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "For a ship constructed on or after 1 January 2017, passenger-ship applicability evidence is required when the category is not A or B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryA .
                }
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryB .
                }
                FILTER NOT EXISTS {
                    $this rdf:type nltl:passengerShip .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Sewage discharge to sea must be false for an applicable ship unless compliance with paragraph 4.2.1.3 is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)
                {
                    $this nltl:shipCategory ?category .
                    FILTER (?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    ))
                }
                UNION
                {
                    $this rdf:type nltl:passengerShip .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus true .
                }
                FILTER NOT EXISTS {
                    $this nltl:sewageDischargeToSea false .
                }
            }
        """
    ] .
<END_SHACL>
