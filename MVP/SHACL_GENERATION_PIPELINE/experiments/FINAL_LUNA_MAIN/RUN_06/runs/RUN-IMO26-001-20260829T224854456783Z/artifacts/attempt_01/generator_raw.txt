<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-001> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must record whether it is certified in accordance with SOLAS chapter I."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Applicable ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Polar Code chapter 9 applicability must be true for a SOLAS chapter I certified ship, and must not be true for a ship without that certification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    $this nltl:solasChapterICertified true .
                    FILTER NOT EXISTS {
                        $this nltl:polarCodeChapter9Applicable true .
                    }
                }
                UNION
                {
                    $this nltl:polarCodeChapter9Applicable true .
                    FILTER NOT EXISTS {
                        $this nltl:solasChapterICertified true .
                    }
                }
            }
        """
    ] .
<END_SHACL>
