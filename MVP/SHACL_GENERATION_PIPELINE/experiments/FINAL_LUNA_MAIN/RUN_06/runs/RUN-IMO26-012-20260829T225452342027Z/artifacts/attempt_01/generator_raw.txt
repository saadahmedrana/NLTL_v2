<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012_ShipHeadingMeansShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node [
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to the main power source."
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to the emergency power source."
            ] ;
            sh:property [
                sh:path nltl:independentFromHeadingMeans ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:nonMagneticHeadingMeans ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall have an independence relationship to another heading means."
            ]
        ] ;
        sh:message "The ship shall have at least two non-magnetic heading means."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means shall be independently related to another distinct heading means belonging to the same ship and shall have both required power connections." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
SELECT $this
WHERE {
  ?this nltl:hasNonMagneticHeadingMeans ?means .
  FILTER (
    NOT EXISTS {
      ?means nltl:connectedToMainPower ?mainPower .
    }
    ||
    NOT EXISTS {
      ?means nltl:connectedToEmergencyPower ?emergencyPower .
    }
    ||
    NOT EXISTS {
      ?means nltl:independentFromHeadingMeans ?otherMeans .
      FILTER (?otherMeans != ?means)
      ?this nltl:hasNonMagneticHeadingMeans ?otherMeans .
    }
  )
}
"""
    ] .
<END_SHACL>
