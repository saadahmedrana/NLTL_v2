@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-012-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For a ship of at least 500 gross tonnage, at least two distinct non-magnetic heading means are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:grossTonnage ?quantity .
    ?quantity qudt:numericValue ?grossTonnage ;
              qudt:unit qudt:UNITLESS .
    FILTER(?grossTonnage >= 500)
    FILTER NOT EXISTS {
        $this nltl:hasNonMagneticHeadingMeans ?means1 .
        ?means1 a nltl:nonMagneticHeadingMeans .
        $this nltl:hasNonMagneticHeadingMeans ?means2 .
        ?means2 a nltl:nonMagneticHeadingMeans .
        FILTER(?means1 != ?means2)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Each required non-magnetic heading means must be independent from another distinct heading means and connected to both main and emergency power." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:grossTonnage ?quantity .
    ?quantity qudt:numericValue ?grossTonnage ;
              qudt:unit qudt:UNITLESS .
    FILTER(?grossTonnage >= 500)
    $this nltl:hasNonMagneticHeadingMeans ?means .
    ?means a nltl:nonMagneticHeadingMeans .
    FILTER NOT EXISTS {
        ?means nltl:connectedToMainPower ?mainPower ;
               nltl:connectedToEmergencyPower ?emergencyPower ;
               nltl:independentFromHeadingMeans ?otherMeans .
        ?mainPower a nltl:benchmarkEntity .
        ?emergencyPower a nltl:benchmarkEntity .
        ?this nltl:hasNonMagneticHeadingMeans ?otherMeans .
        ?otherMeans a nltl:nonMagneticHeadingMeans .
        FILTER(?otherMeans != ?means)
    }
}
"""
    ] .
