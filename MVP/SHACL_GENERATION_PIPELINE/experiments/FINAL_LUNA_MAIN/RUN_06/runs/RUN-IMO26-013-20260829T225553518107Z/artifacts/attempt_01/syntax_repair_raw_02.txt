<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-013> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Ships proceeding to latitudes over 80 degrees require at least one GNSS compass or equivalent connected to both main and emergency power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plannedOrActualLatitude ?latitudeNode .
                ?latitudeNode qudt:numericValue ?latitudeValue .
                FILTER (
                    datatype(?latitudeValue) = xsd:decimal
                    && ?latitudeValue > 80
                )

                FILTER NOT EXISTS {
                    $this nltl:gnssCompassOrEquivalentCount ?count .
                    FILTER (?count < 1)
                }
                FILTER NOT EXISTS {
                    $this nltl:gnssCompassOrEquivalentCount ?count .
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToMainPower ?mainPower .
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToEmergencyPower ?emergencyPower .
                }
            }
        """
    ] .
</END_SHACL>
