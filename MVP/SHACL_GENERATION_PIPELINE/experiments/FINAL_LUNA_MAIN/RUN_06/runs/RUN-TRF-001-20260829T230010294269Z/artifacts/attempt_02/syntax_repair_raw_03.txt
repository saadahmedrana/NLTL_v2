<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF001Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide one construction contract date to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021 ;
        sh:minCount 1 ;
        sh:message "The 2021 Ice Class Regulations section 1.8 provision must apply."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021 ;
        sh:minCount 1 ;
        sh:message "The 2021 Ice Class Regulations chapter 2 provision must apply."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRegulationEdition2021 ;
                sh:minCount 1 ;
                sh:message "Ships contracted on or after 5 July 2021 must identify the 2021 Ice Class Regulations edition."
            ] ;
            sh:sparql [
                sh:message "The 2021 Ice Class Regulations edition applies to ships contracted on or after 5 July 2021." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?date .
                        FILTER (
                            YEAR(?date) > 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) > 7) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) >= 5)
                        )
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "Ships contracted before 5 July 2021 must not assert the 2021 Ice Class Regulations edition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?date .
                        FILTER (
                            YEAR(?date) < 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) < 7) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) < 5)
                        )
                        FILTER NOT EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRegulationEdition2021 .
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
