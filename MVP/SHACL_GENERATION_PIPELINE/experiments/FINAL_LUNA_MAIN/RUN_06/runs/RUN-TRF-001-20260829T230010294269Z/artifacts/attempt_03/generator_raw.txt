<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021 ;
        sh:minCount 1 ;
        sh:message "The section 1.8 provision of the 2021 Ice Class Regulations must apply."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021 ;
        sh:minCount 1 ;
        sh:message "The chapter 2 Ice Class Draught provision of the 2021 Ice Class Regulations must apply."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxExclusive "2021-07-05"^^xsd:date
            ] ;
            sh:sparql [
                sh:message "A ship contracted before 5 July 2021 must not assert the 2021 Ice Class Regulations edition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?date .
                        FILTER (
                            ?date >= "2021-07-05"^^xsd:date
                            || EXISTS {
                                $this nltl:applicableIceClassRegulationEdition
                                    nltl:iceClassRegulationEdition2021 .
                            }
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:minInclusive "2021-07-05"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRegulationEdition2021 ;
                sh:minCount 1 ;
                sh:message "Ships contracted on or after 5 July 2021 must identify the 2021 Ice Class Regulations edition."
            ]
        ]
    ) .
<END_SHACL>
