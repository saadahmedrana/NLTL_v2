<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF002ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2017 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        sh:message "For a ship contracted on or after 1 January 2019 and before 5 July 2021, the applicable Ice Class regulation edition must be the 2017 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (datatype(?date) = xsd:date)
                FILTER (
                    YEAR(?date) > 2019 ||
                    (YEAR(?date) = 2019 && MONTH(?date) > 1) ||
                    (YEAR(?date) = 2019 && MONTH(?date) = 1 && DAY(?date) >= 1)
                )
                FILTER (
                    YEAR(?date) < 2021 ||
                    (YEAR(?date) = 2021 && MONTH(?date) < 7) ||
                    (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) < 5)
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                            nltl:iceClassRegulationEdition2017 .
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?edition .
                        FILTER (?edition != nltl:iceClassRegulationEdition2017)
                    }
                )
            }
        """
    ] .
<END_SHACL>
