<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF005ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:or (
        [
            sh:node gen:TRF005ApplicableBranch
        ]
        [
            sh:node gen:TRF005NonApplicableBranch
        ]
    ) .

gen:TRF005ApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iceClassRegulationEdition2002 ;
        sh:message "Ships within the stated construction-date scope shall identify the 2002 Ice Class Regulations as the applicable edition."
    ] ;
    sh:sparql [
        sh:message "The applicability dates do not establish the applicable branch for the 2002 Ice Class Regulations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?stage ;
                          nltl:constructionContractDate ?contract .
                    FILTER (
                        ?stage >= "2003-09-01"^^xsd:date &&
                        ?contract < "2010-01-01"^^xsd:date
                    )
                }
            }
        """
    ] .

gen:TRF005NonApplicableBranch a sh:NodeShape ;
    sh:sparql [
        sh:message "The ship falls within the applicability dates for the 2002 Ice Class Regulations and must use the applicable branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stage ;
                      nltl:constructionContractDate ?contract .
                FILTER (
                    ?stage >= "2003-09-01"^^xsd:date &&
                    ?contract < "2010-01-01"^^xsd:date
                )
            }
        """
    ] .
<END_SHACL>
