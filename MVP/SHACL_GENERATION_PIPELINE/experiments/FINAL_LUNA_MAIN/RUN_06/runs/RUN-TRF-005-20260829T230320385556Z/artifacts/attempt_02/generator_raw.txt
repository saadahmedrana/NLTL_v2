<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005_Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRegulationEdition2002 ;
                sh:minCount 1 ;
                sh:message "Ships within the stated date scope must identify the 2002 Ice Class Regulations as the applicable edition."
            ] ;
            sh:sparql [
                sh:message "The 2002 Ice Class Regulations apply when the construction stage date is on or after 1 September 2003 and the construction contract date is before 1 January 2010." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:constructionStageDate ?stageDate ;
                                  nltl:constructionContractDate ?contractDate .
                            FILTER (
                                ?stageDate >= "2003-09-01"^^xsd:date &&
                                ?contractDate < "2010-01-01"^^xsd:date
                            )
                        }
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "The non-applicable alternative is permitted only outside the date scope for the 2002 Ice Class Regulations and must not identify the 2002 edition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER (
                            EXISTS {
                                $this nltl:constructionStageDate ?stageDate ;
                                      nltl:constructionContractDate ?contractDate .
                                FILTER (
                                    ?stageDate >= "2003-09-01"^^xsd:date &&
                                    ?contractDate < "2010-01-01"^^xsd:date
                                )
                            } ||
                            EXISTS {
                                $this nltl:applicableIceClassRegulationEdition
                                      nltl:iceClassRegulationEdition2002
                            }
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
