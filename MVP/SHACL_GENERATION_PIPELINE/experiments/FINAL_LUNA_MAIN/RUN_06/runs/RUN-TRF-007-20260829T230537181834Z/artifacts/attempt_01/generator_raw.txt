<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF007Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice class to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship must provide its construction-stage date to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship must provide its delivery date to determine the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship must provide an assessment date for deadline evaluation."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA Super or IA ship constructed before 1 September 2003 must have a permitted 2017 compliance method and must comply by 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )

                FILTER (
                    YEAR(?constructionDate) < 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) < 9
                    )
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) = 9
                        && DAY(?constructionDate) < 1
                    )
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (
                            ?method IN (
                                nltl:traficom2017Section3Point2Point2Method,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    }
                    ||
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        && MONTH(?assessmentDate) > 1
                    )
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        && MONTH(?assessmentDate) = 1
                        && DAY(?assessmentDate) > 1
                    )
                )
            }
        """
    ] .
<END_SHACL>
