<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_011_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF_011_IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify its intended ice-operating waterline."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF_011_IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify its upper ice waterline."
    ] ;
    sh:sparql [
        sh:message "At every longitudinal position represented by either waterline, the UIWL must exist and all of its vertical coordinates must equal the maximum vertical coordinate of the intended ice-operating waterlines within the stated numerical tolerance." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?this
WHERE {
    {
        SELECT ?this ?position
               (COUNT(DISTINCT ?intendedPoint) AS ?intendedCount)
               (COUNT(DISTINCT ?upperPoint) AS ?upperCount)
               (MAX(?intendedValue) AS ?intendedMaximum)
               (MIN(?upperValue) AS ?upperMinimum)
               (MAX(?upperValue) AS ?upperMaximum)
        WHERE {
            {
                ?this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
                ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                ?intendedPoint nltl:longitudinalPosition ?intendedPositionValue ;
                                nltl:verticalCoordinate ?intendedVerticalValue .
                ?intendedPositionValue qudt:numericValue ?position .
                ?intendedVerticalValue qudt:numericValue ?intendedValue .
            }
            UNION
            {
                ?this nltl:hasUpperIceWaterline ?upperWaterline .
                ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                ?upperPoint nltl:longitudinalPosition ?upperPositionValue ;
                            nltl:verticalCoordinate ?upperVerticalValue .
                ?upperPositionValue qudt:numericValue ?position .
                ?upperVerticalValue qudt:numericValue ?upperValue .
            }
        }
        GROUP BY ?this ?position
    }
    BIND((ABS(?intendedMaximum) * 0.000001) + 0.000000001 AS ?tolerance)
    FILTER(
        IF(
            ?intendedCount = 0,
            true,
            IF(
                ?upperCount = 0,
                true,
                (ABS(?upperMinimum - ?intendedMaximum) > ?tolerance) ||
                (ABS(?upperMaximum - ?intendedMaximum) > ?tolerance)
            )
        )
    )
}
"""
    ] .

gen:TRF_011_IceWaterlineShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:TRF_011_ProfilePointShape ;
        sh:minCount 1 ;
        sh:message "Each represented ice waterline must contain at least one profile point."
    ] .

gen:TRF_011_ProfilePointShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:node gen:TRF_011_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each waterline profile point must have a longitudinal position."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:node gen:TRF_011_LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each waterline profile point must have a vertical coordinate."
    ] .

gen:TRF_011_LengthQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
