<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:NotIceStrengthenedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IceStrengthenedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path ( nltl:hasUpperIceWaterline nltl:draughtAtForePerpendicular ) ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path ( nltl:hasUpperIceWaterline nltl:draughtAtAftPerpendicular ) ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path ( nltl:hasLowerIceWaterline nltl:draughtAtForePerpendicular ) ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path ( nltl:hasLowerIceWaterline nltl:draughtAtAftPerpendicular ) ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Operating fore and aft draughts must be between the corresponding minimum and maximum ice-class draughts, and those bounds must correspond to the lower and upper ice waterline draughts." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?upper ;
                          nltl:hasLowerIceWaterline ?lower ;
                          nltl:operatingForeDraught ?opFore ;
                          nltl:operatingAftDraught ?opAft ;
                          nltl:maximumIceClassDraughtFore ?maxFore ;
                          nltl:maximumIceClassDraughtAft ?maxAft ;
                          nltl:minimumIceClassDraughtFore ?minFore ;
                          nltl:minimumIceClassDraughtAft ?minAft .

                    ?upper nltl:draughtAtForePerpendicular ?upperFore ;
                           nltl:draughtAtAftPerpendicular ?upperAft .
                    ?lower nltl:draughtAtForePerpendicular ?lowerFore ;
                           nltl:draughtAtAftPerpendicular ?lowerAft .

                    ?opFore qudt:numericValue ?opForeValue .
                    ?opAft qudt:numericValue ?opAftValue .
                    ?maxFore qudt:numericValue ?maxForeValue .
                    ?maxAft qudt:numericValue ?maxAftValue .
                    ?minFore qudt:numericValue ?minForeValue .
                    ?minAft qudt:numericValue ?minAftValue .
                    ?upperFore qudt:numericValue ?upperForeValue .
                    ?upperAft qudt:numericValue ?upperAftValue .
                    ?lowerFore qudt:numericValue ?lowerForeValue .
                    ?lowerAft qudt:numericValue ?lowerAftValue .

                    FILTER (
                        ?minForeValue <= ?opForeValue &&
                        ?opForeValue <= ?maxForeValue &&
                        ?minAftValue <= ?opAftValue &&
                        ?opAftValue <= ?maxAftValue &&
                        ?maxForeValue = ?upperForeValue &&
                        ?maxAftValue = ?upperAftValue &&
                        ?minForeValue = ?lowerForeValue &&
                        ?minAftValue = ?lowerAftValue
                    )
                }
            }
        """
    ] .

nltlgen:TRF013ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        nltlgen:NotIceStrengthenedShape
        nltlgen:IceStrengthenedShape
    ) .
<END_SHACL>
