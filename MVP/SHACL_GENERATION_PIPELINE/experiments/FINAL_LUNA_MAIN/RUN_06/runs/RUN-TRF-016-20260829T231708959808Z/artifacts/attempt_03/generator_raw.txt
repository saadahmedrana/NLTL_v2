<urn:nltl:generated-shape:TRF-016-NavigatingInIceShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node <urn:nltl:generated-shape:TRF-016-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-016-MassQuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The highest point of the propeller must be submerged."
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node <urn:nltl:generated-shape:TRF-016-BallastTankShape>
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-016-UpperIceWaterlineShape>
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, draught amidships must be at least the LIWL draught amidships, and forward draught must meet the applicable level-ice-thickness formula."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:draughtAmidships/qudt:numericValue ?draught ;
                      nltl:liwlDraughtAmidships/qudt:numericValue ?liwl ;
                      nltl:forwardDraught/qudt:numericValue ?forward ;
                      nltl:levelIceThickness/qudt:numericValue ?hi ;
                      nltl:displacementAtUpperIceWaterline/qudt:numericValue ?delta .
                FILTER (?draught < ?liwl)
                UNION
                {
                    BIND (IF ((2.0 + (0.00025 * ?delta)) * ?hi < 4.0 * ?hi,
                              (2.0 + (0.00025 * ?delta)) * ?hi,
                              4.0 * ?hi) AS ?requiredForward)
                    FILTER (?forward < ?requiredForward)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship displacement at the upper ice waterline must correspond to the greatest-displacement represented upper ice waterline."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:displacementAtUpperIceWaterline/qudt:numericValue ?selected .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?waterline .
                    ?waterline nltl:waterlineDisplacement/qudt:numericValue ?candidate .
                    FILTER (?candidate = ?selected)
                }
                UNION
                {
                    $this nltl:hasUpperIceWaterline ?waterline .
                    ?waterline nltl:waterlineDisplacement/qudt:numericValue ?candidate .
                    FILTER (?candidate > ?selected)
                }
            }
        """
    ] .
