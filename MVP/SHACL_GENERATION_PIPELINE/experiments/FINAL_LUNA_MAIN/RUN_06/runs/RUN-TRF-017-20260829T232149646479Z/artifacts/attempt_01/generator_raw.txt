<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF017Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Maximum continuous rating power P must be provided in kW."
    ] ;

    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Continuous propulsion machinery output must be provided in kW."
    ] ;

    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:or (
            [
                sh:hasValue false
            ]
            [
                sh:hasValue true ;
                sh:property [
                    sh:path nltl:restrictedPropulsionOutput ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node [
                        sh:class qudt:QuantityValue ;
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue unit:KiloW ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ]
            ]
        ] ;
        sh:message "The restriction-applicability value must select a valid output case; a restricted output is required when the restriction applies."
    ] ;

    sh:sparql [
        sh:message "Maximum continuous rating power P must equal the continuous propulsion machinery output." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?pNode .
                ?pNode qudt:numericValue ?p .

                $this nltl:propulsionMachineryContinuousOutput ?cNode .
                ?cNode qudt:numericValue ?c .

                FILTER (?p != ?c)
            }
        """
    ] ;

    sh:sparql [
        sh:message "When propulsion output is restricted, maximum continuous rating power P must equal the restricted propulsion output." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propulsionOutputRestrictionApplies ?applies .
                FILTER (?applies = true)

                $this nltl:maximumContinuousRatingPower ?pNode .
                ?pNode qudt:numericValue ?p .

                $this nltl:restrictedPropulsionOutput ?rNode .
                ?rNode qudt:numericValue ?r .

                FILTER (?p != ?r)
            }
        """
    ] .
<END_SHACL>
