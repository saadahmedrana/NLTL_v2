<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA or IA Super, the applicable regulation edition must be 2002; for ice class IB or IC, the applicable edition must be 1985 and maximum continuous rating power must meet the required power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIa
                        &&
                        NOT EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRegulationEdition2002
                        }
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper
                        &&
                        NOT EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRegulationEdition2002
                        }
                    )
                    ||
                    (
                        (
                            ?iceClass = nltl:iceClassIb
                            ||
                            ?iceClass = nltl:iceClassIc
                        )
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:applicableIceClassRegulationEdition
                                    nltl:iceClassRuleEdition1985
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:maximumContinuousRatingPower ?mcr .
                                ?mcr qudt:numericValue ?mcrValue ;
                                    qudt:unit qudt:KiloW .
                                FILTER ( datatype(?mcrValue) = xsd:decimal )
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:requiredPowerUnder1985Rules ?required .
                                ?required qudt:numericValue ?requiredValue ;
                                    qudt:unit qudt:KiloW .
                                FILTER ( datatype(?requiredValue) = xsd:decimal )
                            }
                            ||
                            EXISTS {
                                $this nltl:maximumContinuousRatingPower ?mcr2 .
                                ?mcr2 qudt:numericValue ?mcrValue2 ;
                                    qudt:unit qudt:KiloW .
                                $this nltl:requiredPowerUnder1985Rules ?required2 .
                                ?required2 qudt:numericValue ?requiredValue2 ;
                                    qudt:unit qudt:KiloW .
                                FILTER ( datatype(?mcrValue2) = xsd:decimal )
                                FILTER ( datatype(?requiredValue2) = xsd:decimal )
                                FILTER ( ?mcrValue2 < ?requiredValue2 )
                            }
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
