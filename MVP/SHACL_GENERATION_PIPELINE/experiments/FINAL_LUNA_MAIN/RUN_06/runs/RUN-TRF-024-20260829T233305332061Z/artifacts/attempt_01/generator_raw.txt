<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF024Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in ( nltl:iceClassIaSuper nltl:iceClassIa ) ;
                sh:minCount 1 ;
                sh:message "An IA Super or IA ice class is required for the applicable branch."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:maxInclusive "2003-08-31"^^xsd:date ;
                sh:minCount 1 ;
                sh:message "The construction stage date must be before 1 September 2003."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A delivery date is required to determine the twenty-year deadline."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "An assessment date is required to evaluate the deadline."
            ] ;
            sh:sparql [
                sh:message "The ship must comply no later than 1 January in the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:deliveryDate ?delivery ;
                              nltl:assessmentDate ?assessment .
                        FILTER (
                            YEAR(?assessment) < YEAR(?delivery) + 20
                            ||
                            (
                                YEAR(?assessment) = YEAR(?delivery) + 20
                                && MONTH(?assessment) = 1
                                && DAY(?assessment) = 1
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1 ;
                sh:message "The ship is outside the IA Super or IA applicability branch."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in ( nltl:iceClassIaSuper nltl:iceClassIa ) ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2003-09-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:message "The ship is outside the applicability branch because construction was not before 1 September 2003."
            ]
        ]
    ) ;
    sh:message "The conditional ice-class retention deadline requirement is not satisfied or cannot be classified for this ship." .
<END_SHACL>
