<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "beamTheoryUsed must be provided as a single boolean applicability value."
    ] ;

    sh:property [
        sh:path nltl:yieldPoint ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "yieldPoint must be a single MPa quantity with a decimal numeric value."
    ] ;

    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "combinedBendingShearStress must be a single MPa quantity with a decimal numeric value."
    ] ;

    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "yieldShearStress must be a single MPa quantity with a decimal numeric value."
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "allowableShearStress must be a single MPa quantity when beam theory is used."
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;

    sh:sparql [
        sh:message "combinedBendingShearStress must be lower than yieldPoint, and yieldShearStress must equal yieldPoint divided by sqrt(3) within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldPointNode ;
                      nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldShearStress ?yieldShearNode .

                ?yieldPointNode qudt:numericValue ?yieldPointValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?combinedNode qudt:numericValue ?combinedValue ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?yieldShearNode qudt:numericValue ?yieldShearValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .

                BIND((?yieldPointValue / 1.7320508075688772) AS ?expectedYieldShear)

                FILTER (
                    (?combinedValue >= ?yieldPointValue)
                    ||
                    (ABS(?yieldShearValue - ?expectedYieldShear) >
                     ((ABS(?expectedYieldShear) * 0.000001) + 0.000001))
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "When beam theory is used, allowableShearStress must be no larger than 0.9 times yieldPoint divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:yieldPoint ?yieldPointNode ;
                      nltl:allowableShearStress ?allowableNode .

                ?yieldPointNode qudt:numericValue ?yieldPointValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?allowableNode qudt:numericValue ?allowableValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .

                BIND((0.9 * (?yieldPointValue / 1.7320508075688772)) AS ?maximumAllowable)

                FILTER (?allowableValue > ?maximumAllowable)
            }
        """
    ] .
<END_SHACL>
