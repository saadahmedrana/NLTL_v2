<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:vonMisesYieldCriterion ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:node nltlgen:PressureQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be strictly lower than the yield point, and yield shear stress must equal yield point divided by sqrt(3) within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:vonMisesYieldCriterion ?criterion ;
                      nltl:yieldPoint ?yieldNode ;
                      nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldShearStress ?yieldShearNode .
                ?yieldNode qudt:numericValue ?yieldPoint ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?combinedNode qudt:numericValue ?combinedStress ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?yieldShearNode qudt:numericValue ?yieldShear ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                BIND((?yieldPoint / 1.7320508075688772) AS ?expectedYieldShear)
                BIND((1e-6 * MAX(?expectedYieldShear, -?expectedYieldShear)) AS ?relativeTolerance)
                BIND(MAX(?relativeTolerance, 1e-9) AS ?tolerance)
                BIND((?expectedYieldShear - ?tolerance) AS ?lowerBound)
                BIND((?expectedYieldShear + ?tolerance) AS ?upperBound)
                FILTER (
                    ?combinedStress >= ?yieldPoint ||
                    ?yieldShear < ?lowerBound ||
                    ?yieldShear > ?upperBound
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must not exceed 0.9 times yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:vonMisesYieldCriterion ?criterion ;
                      nltl:yieldPoint ?yieldNode ;
                      nltl:allowableShearStress ?allowableNode .
                ?yieldNode qudt:numericValue ?yieldPoint ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?allowableNode qudt:numericValue ?allowableStress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                BIND((0.9 * (?yieldPoint / 1.7320508075688772)) AS ?maximumAllowableStress)
                FILTER (?allowableStress > ?maximumAllowableStress)
            }
        """
    ] .
<END_SHACL>
