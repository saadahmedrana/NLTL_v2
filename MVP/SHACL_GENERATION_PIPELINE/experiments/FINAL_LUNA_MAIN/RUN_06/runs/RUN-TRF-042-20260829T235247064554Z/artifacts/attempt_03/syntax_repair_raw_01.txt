<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlg:LengthMillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:LengthMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:YieldStrengthBaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:YieldStrengthNormalShape
    a sh:NodeShape ;
    sh:node nltlg:YieldStrengthBaseShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 235 ;
        sh:maxInclusive 235
    ] .

nltlg:YieldStrengthHighShape
    a sh:NodeShape ;
    sh:node nltlg:YieldStrengthBaseShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 315
    ] .

nltlg:PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;

    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:datatype qudt:QuantityValue ;
        sh:node nltlg:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "designIceLoadHeight is required in metres."
    ] ;

    sh:property [
        sh:path nltl:spacing ;
        sh:datatype qudt:QuantityValue ;
        sh:node nltlg:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "spacing is required in metres."
    ] ;

    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:datatype qudt:QuantityValue ;
        sh:node nltlg:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "longitudinalPlatingFactorF2 is required as a dimensionless quantity."
    ] ;

    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:datatype qudt:QuantityValue ;
        sh:node nltlg:LengthMillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "corrosionAbrasionAddition is required in millimetres."
    ] ;

    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:normalStrengthHullStructuralSteel ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node nltlg:YieldStrengthNormalShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:highStrengthHullStructuralSteel ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node nltlg:YieldStrengthHighShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;

    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the declared piecewise function of designIceLoadHeight divided by spacing, within the required tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?f .
                FILTER (
                    ?s = 0 ||
                    (?s != 0 &&
                     (
                        ?h / ?s < 0 ||
                        ?h / ?s > 1.8 ||
                        ABS(
                            ?f -
                            IF(
                                ?h / ?s <= 1,
                                0.4 / (?h / ?s) + 0.6,
                                1.4 - 0.4 * (?h / ?s)
                            )
                        ) >
                        IF(
                            0.000001 *
                            ABS(
                                IF(
                                    ?h / ?s <= 1,
                                    0.4 / (?h / ?s) + 0.6,
                                    1.4 - 0.4 * (?h / ?s)
                                )
                            ) >
                            0.000000000001,
                            0.000001 *
                            ABS(
                                IF(
                                    ?h / ?s <= 1,
                                    0.4 / (?h / ?s) + 0.6,
                                    1.4 - 0.4 * (?h / ?s)
                                )
                            ),
                            0.000000000001
                        )
                     )
                    )
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "corrosionAbrasionAddition must not exceed 2 mm; a lower value requires maintained special surface coating and approved material evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc .
                FILTER (
                    ?tc > 2 ||
                    (
                        ?tc < 2 &&
                        (
                            NOT EXISTS {
                                $this nltl:specialSurfaceCoatingMaintained true
                            } ||
                            NOT EXISTS {
                                $this nltl:materialApprovalStatus nltl:evidenceStateApproved
                            }
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
