<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gs:TRF043TerminationCaseShape
    ] .

gs:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF043ExtensionQuantityShape ;
        sh:message "Each framing ice-strengthening termination case must state the extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:message "The termination adjacent boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:sparql [
        sh:message "When the extension beyond the adjacent boundary is no more than 250 mm, termination at that boundary must be permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
                ?quantity qudt:numericValue ?extensionValue .
                ?quantity qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (xsd:decimal(?extensionValue) <= 250.0)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ?permitted .
                    FILTER (datatype(?permitted) = xsd:boolean && ?permitted = true)
                }
            }
        """
    ] .

gs:TRF043ExtensionQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

</END_SHACL>
