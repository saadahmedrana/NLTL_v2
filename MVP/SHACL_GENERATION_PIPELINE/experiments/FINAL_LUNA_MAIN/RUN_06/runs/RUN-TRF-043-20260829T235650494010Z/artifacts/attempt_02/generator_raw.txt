<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node g:TRF043TerminationCaseShape ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:message "Each framing ice-strengthening termination case linked to the ship must conform to the termination-case constraints."
    ] .

g:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node g:TRF043MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A termination case must specify one extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "The termination permission, when recorded, must be a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:class nltl:deckStructure
            ]
            [
                sh:class nltl:tankBoundaryPlating
            ]
            [
                sh:class nltl:tankTop
            ]
        ) ;
        sh:message "The adjacent termination boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:sparql [
        sh:message "When the extension beyond the adjacent boundary is no more than 250 mm, termination at that boundary must be permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
                ?quantity qudt:numericValue ?extension .
                ?quantity qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER ( ?extension <= "250"^^xsd:decimal )
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted
                        "true"^^xsd:boolean .
                }
            }
        """
    ] .

g:TRF043MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The extension quantity must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The extension quantity must use millimetres."
    ] .

<END_SHACL>
