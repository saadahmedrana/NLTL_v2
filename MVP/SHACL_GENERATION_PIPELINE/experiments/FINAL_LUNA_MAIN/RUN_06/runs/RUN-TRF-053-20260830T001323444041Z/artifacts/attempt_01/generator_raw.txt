<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    ] ;
    sh:property [

nltlgen:TRF053IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltlgen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltlgen:ForcePerLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltlgen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltlgen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "designLineLoad, sectionModulus, and requiredShearArea must satisfy the TRF-053 formulas; calculated decimal results are checked with a relative tolerance of 1e-6 plus an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:designLineLoad ?dNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?dNode qudt:numericValue ?d .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF(?p * ?h < 0.15, 0.15, ?p * ?h) AS ?designLoad)
                BIND(0.9 * 1.8 * ?designLoad * ?l * ?l / (?m * ?sigma) * 1000000 AS ?expectedZ)
                BIND(1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?designLoad * ?l / (2 * ?sigma) * 10000 AS ?expectedA)

                FILTER(
                    ABS(?d - ?designLoad) > (0.000001 * ABS(?designLoad) + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) > (0.000001 * ABS(?expectedZ) + 0.000000001)
                    ||
                    ABS(?a - ?expectedA) > (0.000001 * ABS(?expectedA) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
