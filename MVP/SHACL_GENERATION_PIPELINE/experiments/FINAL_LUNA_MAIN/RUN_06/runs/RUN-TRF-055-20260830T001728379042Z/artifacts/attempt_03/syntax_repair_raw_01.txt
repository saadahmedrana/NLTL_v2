    a sh:NodeShape ;
    sh:property [
        sh:path nltl:actualSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_ActualSectionModulusValueShape
    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_RequiredSectionModulusValueShape
    ] ;
    sh:property [
        sh:path nltl:actualShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_ActualShearAreaValueShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_RequiredShearAreaValueShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual section modulus and actual shear area must each be at least their corresponding required value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualZ .
                ?actualZ qudt:numericValue ?actualZNumeric .
                $this nltl:requiredSectionModulus ?requiredZ .
                ?requiredZ qudt:numericValue ?requiredZNumeric .
                $this nltl:actualShearArea ?actualA .
                ?actualA qudt:numericValue ?actualANumeric .
                $this nltl:requiredShearArea ?requiredA .
                ?requiredA qudt:numericValue ?requiredANumeric .
                FILTER (
                    ?actualZNumeric < ?requiredZNumeric ||
                    ?actualANumeric < ?requiredANumeric
                )
            }
        """
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:veryLongHatchOpening ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:veryLongHatchOpening ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:permittedReducedLineLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF055_PermittedReducedLineLoadValueShape
            ] ;
            sh:property [
                sh:path nltl:reducedLineLoadApprovedByClassificationSociety ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:message "For a very long hatch opening, the permitted reduced line load must be at least 0.10 kN/m and less than 0.15 kN/m, with classification-society approval."
        ]
    ) .
