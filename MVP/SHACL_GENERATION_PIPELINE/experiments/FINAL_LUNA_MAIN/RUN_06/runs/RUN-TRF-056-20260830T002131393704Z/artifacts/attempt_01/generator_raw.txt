<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-056-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:node <urn:nltl:generated-shape:TRF-056-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node <urn:nltl:generated-shape:TRF-056-WeatherdeckHatchShape>
    ] ;
    sh:sparql [
        sh:message "A ship breadth is required when the ship has a weatherdeck hatch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                FILTER NOT EXISTS {
                    $this nltl:shipBreadth ?breadth
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-056-WeatherdeckHatchShape>
    a sh:NodeShape ;
    sh:class nltl:weatherdeckHatch ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-056-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:shipSideDeflection ;
        sh:node <urn:nltl:generated-shape:TRF-056-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hasHatchCoverDesignEvidence ;
        sh:node [
            sh:class nltl:hatchCoverDesignEvidence
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchFittingDesignEvidence ;
        sh:node [
            sh:class nltl:hatchFittingDesignEvidence
        ]
    ] ;
    sh:sparql [
        sh:message "Ship-side deflection must be represented for a weatherdeck hatch longer than half the ship breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hatchOpeningLength ?opening .
                ?opening qudt:numericValue ?openingLength ;
                         qudt:unit unit:M .
                ?ship nltl:hasWeatherdeckHatch $this ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth ;
                                 qudt:unit unit:M .
                FILTER(
                    datatype(?openingLength) = xsd:decimal &&
                    datatype(?breadth) = xsd:decimal &&
                    ?openingLength > (?breadth / 2)
                )
                FILTER NOT EXISTS {
                    $this nltl:shipSideDeflection ?deflection
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch-cover design evidence must exist for an applicable weatherdeck hatch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hatchOpeningLength ?opening .
                ?opening qudt:numericValue ?openingLength ;
                         qudt:unit unit:M .
                ?ship nltl:hasWeatherdeckHatch $this ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth ;
                                 qudt:unit unit:M .
                FILTER(
                    datatype(?openingLength) = xsd:decimal &&
                    datatype(?breadth) = xsd:decimal &&
                    ?openingLength > (?breadth / 2)
                )
                FILTER NOT EXISTS {
                    $this nltl:hasHatchCoverDesignEvidence ?evidence
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch-fitting design evidence must exist for an applicable weatherdeck hatch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hatchOpeningLength ?opening .
                ?opening qudt:numericValue ?openingLength ;
                         qudt:unit unit:M .
                ?ship nltl:hasWeatherdeckHatch $this ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth ;
                                 qudt:unit unit:M .
                FILTER(
                    datatype(?openingLength) = xsd:decimal &&
                    datatype(?breadth) = xsd:decimal &&
                    ?openingLength > (?breadth / 2)
                )
                FILTER NOT EXISTS {
                    $this nltl:hasHatchFittingDesignEvidence ?evidence
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-056-LengthQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .

<END_SHACL>
