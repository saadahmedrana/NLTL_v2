<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_057_WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "webFrameIceLoad must equal 1.8 times the greater of icePressure multiplied by iceLoadHeight and 0.15, multiplied by webFrameSpacing, with the result converted from MN to kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:iceLoadHeight ?heightNode ;
                      nltl:webFrameSpacing ?spacingNode ;
                      nltl:webFrameIceLoad ?loadNode .

                ?pressureNode qudt:numericValue ?pressure .
                ?heightNode qudt:numericValue ?height .
                ?spacingNode qudt:numericValue ?spacing .
                ?loadNode qudt:numericValue ?reportedLoad .

                BIND (
                    1.8 *
                    IF(
                        (?pressure * ?height) < 0.15,
                        0.15,
                        (?pressure * ?height)
                    ) *
                    ?spacing *
                    1000
                    AS ?expectedLoad
                )

                FILTER (
                    ABS(?reportedLoad - ?expectedLoad) >
                    IF(
                        (0.000001 * ABS(?expectedLoad)) > 0.000001,
                        (0.000001 * ABS(?expectedLoad)),
                        0.000001
                    )
                )
            }
        """
    ] .
<END_SHACL>
