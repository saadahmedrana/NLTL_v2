<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        nltlgen:TRF058ApplicableShape
        nltlgen:TRF058NotApplicableShape
    ) .

nltlgen:TRF058ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:node nltlgen:TRF058ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:node nltlgen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node nltlgen:TRF058LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:node nltlgen:TRF058ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceLoadForce ?forceNode ;
                      nltl:stringerOutsideIceBeltHeight ?heightNode ;
                      nltl:stringerSpan ?spanNode ;
                      nltl:adjustedIceLoadForce ?adjustedNode .

                ?forceNode qudt:numericValue ?force .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?adjustedNode qudt:numericValue ?adjusted .

                BIND (?force * (1 - (?height / ?span)) AS ?expected)
                BIND (IF(ABS(?expected) * 0.000001 > 0.000000001,
                         ABS(?expected) * 0.000001,
                         0.000000001) AS ?tolerance)

                FILTER (
                    ?adjusted < (?expected - ?tolerance) ||
                    ?adjusted > (?expected + ?tolerance)
                )
            }
        """
    ] .

nltlgen:TRF058NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:hasValue false
    ] .

nltlgen:TRF058ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF058LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
