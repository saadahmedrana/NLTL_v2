<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

<urn:nltl:generated-shape:TRF-063-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:in ( 2 3 )
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
                sh:minCount 1 ;
                sh:class nltl:sidePropellerStrengtheningEnvelope ;
                sh:node <urn:nltl:generated-shape:TRF-063-EnvelopeShape>
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "A non-twin or non-triple-screw ship must not have a propeller count of 2 or 3 in the non-applicable branch." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this nltl:propellerCount ?count .
                        FILTER (?count = 2 || ?count = 3)
                    }
                """
            ]
        ]
    ] .

<urn:nltl:generated-shape:TRF-063-EnvelopeShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:minCount 1 ;
        sh:class nltl:tankTop ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-063-ForwardExtentQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-063-AftExtentQuantityShape>
    ] .

<urn:nltl:generated-shape:TRF-063-ForwardExtentQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.5
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

<urn:nltl:generated-shape:TRF-063-AftExtentQuantityShape>
