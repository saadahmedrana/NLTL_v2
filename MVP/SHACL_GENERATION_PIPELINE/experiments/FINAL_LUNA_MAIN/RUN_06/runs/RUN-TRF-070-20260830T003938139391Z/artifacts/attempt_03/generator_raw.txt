<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:KiloNQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPAQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PropellerIceInteractionLoadPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:propellerIceInteractionLoad ;
    sh:node gen:KiloNQuantityShape ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:message "The applicable propeller/ice interaction load is required in kN." .

gen:ThrusterBodyIceInteractionLoadPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:thrusterBodyIceInteractionLoad ;
    sh:node gen:KiloNQuantityShape ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:message "The applicable thruster-body ice-interaction load is required in kN." .

gen:LocalIcePressurePropertyShape
    a sh:PropertyShape ;
    sh:path nltl:localIcePressure ;
    sh:node gen:MegaPAQuantityShape ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:message "The applicable local ice pressure is required in MPa." .

gen:DesignConditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The azimuthing-thruster design condition must represent normal operational conditions."
    ] ;
    sh:property [
        sh:path nltl:designConditionExpectedLoad ;
        sh:node gen:KiloNQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable design-condition expected load is required in kN."
    ] ;
    sh:property [
        sh:path nltl:designConditionLocalStrengthCapacity ;
        sh:node gen:KiloNQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The externally assessed local-strength capacity is required in kN."
    ] ;
    sh:sparql [
        sh:message "The design-condition local-strength capacity must be at least the local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasDesignCondition $this .
                ?ship nltl:localIcePressure ?pressure .
                ?pressure q:numericValue ?pressureValue .
                $this nltl:designConditionLocalStrengthCapacity ?capacity .
                ?capacity q:numericValue ?capacityValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .

gen:TRF070Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:in ("azimuthing" "fixed") ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:hasValue "fixed" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property gen:PropellerIceInteractionLoadPropertyShape ;
            sh:property gen:ThrusterBodyIceInteractionLoadPropertyShape ;
            sh:property gen:LocalIcePressurePropertyShape
        ]
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:hasValue "azimuthing" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property gen:PropellerIceInteractionLoadPropertyShape ;
            sh:property gen:ThrusterBodyIceInteractionLoadPropertyShape ;
            sh:property gen:LocalIcePressurePropertyShape ;
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:node gen:DesignConditionShape ;
                sh:minCount 1 ;
                sh:message "An applicable azimuthing-thruster design condition is required."
            ] ;
            sh:sparql [
                sh:message "For an azimuthing main-propulsion thruster, the normal-condition expected load must equal the thruster-body ice-interaction load." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX q: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasDesignCondition ?condition .
                        ?condition nltl:normalOperationalConditions true .
                        FILTER NOT EXISTS {
                            ?condition nltl:designConditionExpectedLoad ?expected .
                            ?expected q:numericValue ?expectedValue .
                            $this nltl:thrusterBodyIceInteractionLoad ?bodyLoad .
                            ?bodyLoad q:numericValue ?bodyValue .
                            FILTER (?expectedValue = ?bodyValue)
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
