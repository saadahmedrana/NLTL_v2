<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SafePropellerDepthCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The ballast-condition propeller top depth must be at least hi." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                ?depthNode qudt:numericValue ?depth .
                $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                ?hiNode qudt:numericValue ?hi .
                FILTER (?depth < ?hi)
            }
        """
    ] .

gen:InsufficientPropellerDepthCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemDesignedForIceClass ;
        sh:hasValue nltl:iceClassIa ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "When propeller submersion is below hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                ?depthNode qudt:numericValue ?depth .
                $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                ?hiNode qudt:numericValue ?hi .
                FILTER (?depth >= ?hi)
            }
        """
    ] .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii ;
                sh:minCount 1
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:hasValue nltl:iceClassIb ;
                        sh:minCount 1
                    ]
                ]
                [
                    sh:or (
                        [ sh:node gen:SafePropellerDepthCase ]
                        [ sh:node gen:InsufficientPropellerDepthCase ]
                    )
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:hasValue nltl:iceClassIc ;
                        sh:minCount 1
                    ]
                ]
                [
                    sh:or (
                        [ sh:node gen:SafePropellerDepthCase ]
                        [ sh:node gen:InsufficientPropellerDepthCase ]
                    )
                ]
            )
        ]
    ) .
<END_SHACL>
