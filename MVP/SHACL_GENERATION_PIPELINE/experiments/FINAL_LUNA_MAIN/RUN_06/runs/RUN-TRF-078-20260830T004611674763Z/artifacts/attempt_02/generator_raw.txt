<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TableLookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each table lookup case must have exactly one table reference."
    ] .

gen:Table6Dash6CaseShape
    a sh:NodeShape ;
    sh:node gen:TableLookupCaseShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash6 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedIceClassCycleCount ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:Table6Dash7CaseShape
    a sh:NodeShape ;
    sh:node gen:TableLookupCaseShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash7 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:tableLookupPropellerLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:propellerLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedPropellerLocationFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TableLookupCaseShape ;
        sh:qualifiedValueShape gen:Table6Dash6CaseShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must have a Table 6-6 lookup case with an ice-class selection and selected cycle count."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TableLookupCaseShape ;
        sh:qualifiedValueShape gen:Table6Dash7CaseShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must have a Table 6-7 lookup case with a propeller-location selection and selected factor."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 times k2 times k3 times N_class times n_n within a relative tolerance of 1e-6."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case6 ;
                      nltl:hasTableLookupCase ?case7 ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?result .

                ?case6 nltl:tableReference nltl:traficomTable6Dash6 ;
                        nltl:selectedIceClassCycleCount ?nclassNode .

                ?case7 nltl:tableReference nltl:traficomTable6Dash7 ;
                        nltl:selectedPropellerLocationFactor ?k1Node .

                ?k1Node qudt:numericValue ?k1 .
                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nclassNode qudt:numericValue ?nclass .
                ?nnNode qudt:numericValue ?nn .

                BIND((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER(
                    ABS(xsd:decimal(?result) - ?expected) >
                    (1e-6 * ABS(?expected) + 1e-9)
                )
            }
        """
    ] .
<END_SHACL>
