<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Propeller type is required to determine whether the CP-propeller rule applies."
    ] ;
    sh:or (
        gen:TRF084NonCPShape
        gen:TRF084CPKnownShape
        gen:TRF084CPUnknownShape
    ) .

gen:TRF084NonCPShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The CP-propeller applicability branch was not established." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER NOT EXISTS {
                    $this nltl:propellerType "CP" .
                }
            }
        """
    ] .

gen:TRF084CPKnownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:hasValue "CP" ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with known pitch, the MCR bollard-condition pitch is required."
    ] .

gen:TRF084CPUnknownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:hasValue "CP" ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrFreeRunning ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with unknown pitch, the MCR free-running pitch is required."
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with unknown pitch, the MCR bollard-condition pitch is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a CP propeller with unknown pitch, the MCR bollard-condition pitch must equal 0.7 times the MCR free-running pitch within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 m." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:propellerType "CP" ;
                      nltl:pitchValueKnown false ;
                      nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
                      nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                ?freeQuantity qudt:numericValue ?freeValue ;
                               qudt:unit unit:M .
                ?bollardQuantity qudt:numericValue ?bollardValue ;
                                  qudt:unit unit:M .
                FILTER (
                    ABS(?bollardValue - (0.7 * ?freeValue)) >
                    (
                        0.000001 * ABS(0.7 * ?freeValue) +
                        0.000000001
                    )
                )
            }
        """
    ] .
<END_SHACL>
