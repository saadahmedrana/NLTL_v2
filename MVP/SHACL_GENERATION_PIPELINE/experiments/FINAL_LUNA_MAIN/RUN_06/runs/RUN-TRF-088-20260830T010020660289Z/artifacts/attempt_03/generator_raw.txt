<urn:nltl:generated-shape:TRF088ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:maximumEngineTorqueKnown ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:maximumEngineTorqueKnown ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF088TableLookupCaseShape>
            ] ;
            sh:property [
                sh:path nltl:maximumEngineTorque ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF088TorqueQuantityShape>
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When maximum engine torque is unknown, the reported torque must equal the value selected from TRAFICOM Table 6-10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:maximumEngineTorqueKnown false .
                $this nltl:maximumEngineTorque ?torque .
                ?torque qudt:numericValue ?reportedValue ;
                        qudt:unit qudt:KiloN-M .
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:tableReference nltl:traficomTable6Dash10 .
                nltl:traficomTable6Dash10 qudt:numericValue ?selectedValue ;
                                           qudt:unit qudt:KiloN-M .
                FILTER (?reportedValue != ?selectedValue)
            }
        """
    ] .
