<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF091ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091IceMillingSequenceCaseShape ;
        sh:message "The ship shall identify an ice milling sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091PropulsionLoadCaseShape ;
        sh:message "The ship shall identify a propulsion load case for the strength evaluation."
    ] .

gen:TRF091IceMillingSequenceCaseShape
    a sh:NodeShape .

gen:TRF091PropulsionLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
        sh:message "The propulsion load case shall have the strength-evaluation purpose."
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:loadCasePurpose ;
            sh:hasValue nltl:stallingAnalysisPurposeValue ;
            sh:message "The propulsion load case shall not be designated for stalling analysis."
        ]
    ] .
<END_SHACL>
