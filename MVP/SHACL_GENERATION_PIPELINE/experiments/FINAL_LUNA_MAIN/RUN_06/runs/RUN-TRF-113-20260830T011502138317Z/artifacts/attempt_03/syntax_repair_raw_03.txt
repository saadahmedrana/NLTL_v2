<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_113_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:steeringMechanism
        ] ;
        sh:message "The ship must identify a steering mechanism for the blade-loss plastic-bending assessment."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:thrusterFitting
        ] ;
        sh:message "The ship must identify the thruster fitting for the blade-loss plastic-bending assessment."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:thrusterBody
        ] ;
        sh:message "The ship must identify the thruster body for the blade-loss plastic-bending assessment."
    ] ;
    sh:property [
        sh:path nltl:thrusterBladeLossDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "The ship must link evidence for the thruster blade-loss design assessment."
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:maximumComponentLoadBladeOrientationConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "The load case must truthfully confirm the propeller blade orientation producing the maximum component load."
            ]
        ] ;
        sh:message "The ship must provide a load case identifying the maximum-load blade orientation."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The steering mechanism, thruster fitting, and thruster body must be represented as distinct components in the blade-loss assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?steering .
                    ?steering a nltl:steeringMechanism .
                    $this nltl:hasComponent ?fitting .
                    ?fitting a nltl:thrusterFitting .
                    $this nltl:hasComponent ?body .
                    ?body a nltl:thrusterBody .
                    FILTER (
                        ?steering != ?fitting &&
                        ?steering != ?body &&
                        ?fitting != ?body
                    )
                }
            }
        """
    ] .
</END_SHACL>
