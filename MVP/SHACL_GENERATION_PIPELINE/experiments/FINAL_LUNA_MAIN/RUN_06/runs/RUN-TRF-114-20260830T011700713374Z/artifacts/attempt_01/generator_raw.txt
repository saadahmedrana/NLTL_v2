<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-114-ship> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide an ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node [
            sh:class nltl:thrusterIceImpactLoadCase ;
            sh:property [
                sh:path nltl:designIceBlockTableReference ;
                sh:hasValue nltl:traficomTable6Dash3Reference ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must reference TRAFICOM Table 6-3 for the design ice block."
            ] ;
            sh:property [
                sh:path nltl:impactLoadCaseTableReference ;
                sh:hasValue nltl:traficomTable6Dash16Reference ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must reference TRAFICOM Table 6-16 for its load case."
            ] ;
            sh:property [
                sh:path nltl:thrusterImpactCaseAppliesToBody ;
                sh:class nltl:thrusterBody ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:thrusterResistanceCapacity ;
                        sh:class qudt:QuantityValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoad ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactDemand ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "The impact case must provide evidence for the loaded impact area."
            ] ;
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:or (
                    [
                        sh:hasValue nltl:hemisphericalContactGeometry
                    ]
                    [
                        sh:hasValue nltl:nonHemisphericalContactGeometry ;
                        sh:property [
                            sh:path nltl:contactGeometryCorrespondenceEvidence ;
                            sh:class nltl:evidenceArtifact ;
                            sh:minCount 1 ;
                            sh:message "Non-hemispherical contact geometry requires correspondence evidence."
                        ]
                    ]
                ) ;
                sh:message "The impact case must classify its contact geometry as hemispherical or non-hemispherical."
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "A non-hemispherical contact geometry requires an equivalent impact sphere radius on the ship." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    SELECT $this
                    WHERE {
                        $this nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                        ?ship nltl:hasThrusterIceImpactLoadCase $this .
                        FILTER NOT EXISTS {
                            ?ship nltl:equivalentImpactSphereRadius ?radius .
                            ?radius qudt:numericValue ?value .
                            ?radius qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                    }
                """
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The externally assessed thruster resistance capacity must be at least the thruster ice impact demand." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    SELECT $this
                    WHERE {
                        $this nltl:thrusterIceImpactDemand ?demand .
                        ?demand qudt:numericValue ?demandValue .
                        $this nltl:thrusterImpactCaseAppliesToBody ?body .
                        ?body nltl:thrusterResistanceCapacity ?capacity .
                        ?capacity qudt:numericValue ?capacityValue .
                        FILTER (?capacityValue < ?demandValue)
                    }
                """
            ]
        ]
    ;
    sh:message "Each linked thruster ice impact case must provide the declared applicability, load, resistance, body, geometry, and evidence information."
    .

            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
