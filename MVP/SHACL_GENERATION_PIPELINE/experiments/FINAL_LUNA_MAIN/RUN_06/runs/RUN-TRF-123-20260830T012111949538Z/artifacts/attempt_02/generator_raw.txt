<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_123_ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:occasionalForceLoadCase ;
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each occasional-force load case must explicitly confirm exclusion of propeller blades."
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:shipComponent ;
                    sh:property [
                        sh:path nltl:occasionalForceComponentStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:hasMaterialProperties ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:materialProperties ;
                            sh:property [
                                sh:path nltl:componentMaterialYieldStrength ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Occasional-force component stress must not exceed the material yield strength after applying the yield safety factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasOccasionalForceLoadCase ?case .
                ?case nltl:propellerBladeExcludedFromOccasionalForceScope true .
                ?case nltl:componentYieldSafetyFactor ?factorValue .
                ?factorValue qudt:numericValue ?factor .
                ?case nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressValue .
                ?stressValue qudt:numericValue ?stress .
                ?component nltl:hasMaterialProperties ?material .
                ?material nltl:componentMaterialYieldStrength ?yieldValue .
                ?yieldValue qudt:numericValue ?yield .
                FILTER (?stress > (?yield / ?factor))
            }
        """
    ] .
<END_SHACL>
