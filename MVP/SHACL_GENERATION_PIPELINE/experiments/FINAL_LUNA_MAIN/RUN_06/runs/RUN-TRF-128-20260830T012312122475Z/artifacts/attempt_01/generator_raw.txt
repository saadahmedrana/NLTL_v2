    sh:property [

        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "0.5"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
