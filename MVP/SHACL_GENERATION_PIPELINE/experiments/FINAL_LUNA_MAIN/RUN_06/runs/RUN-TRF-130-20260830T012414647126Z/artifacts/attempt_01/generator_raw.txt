            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this .
                ?ship nltl:totalRelevantEnginePower ?powerNode .
                ?powerNode qudt:numericValue ?power .
                $this nltl:recommendedChestVolume ?volumeNode .
                ?volumeNode qudt:numericValue ?volume .
                $this nltl:strainerOpenArea ?openNode .
                ?openNode qudt:numericValue ?open .
                $this nltl:inletPipeArea ?pipeNode .
                ?pipeNode qudt:numericValue ?pipe .
                $this nltl:strainerOpenAreaRatio ?ratioNode .
                ?ratioNode qudt:numericValue ?ratio .
                BIND(?power / 750 AS ?expectedVolume)
                BIND(?open / ?pipe AS ?expectedRatio)
                FILTER(
                    ABS(?volume - ?expectedVolume) >
                    (ABS(?expectedVolume) * 0.000001 + 0.000000001)
                    ||
                    ?open < (4 * ?pipe)
                    ||
                    ?ratio < 4
                    ||
                    ABS(?ratio - ?expectedRatio) >
                    (ABS(?expectedRatio) * 0.000001 + 0.000000001)
                )
            }
        """
