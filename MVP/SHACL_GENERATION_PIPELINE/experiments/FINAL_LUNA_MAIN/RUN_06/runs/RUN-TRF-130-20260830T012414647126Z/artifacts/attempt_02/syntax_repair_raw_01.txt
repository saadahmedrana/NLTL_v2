gen:NormalQualifyingInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "near_centreline_and_well_aft_if_possible"
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:RatioQuantityShape
    ] ;
    sh:sparql [
        sh:message "The qualifying inlet chest must have the required approximate volume and strainer open area." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  ?ship nltl:hasComponent $this ;
        nltl:totalRelevantEnginePower ?powerNode .
  ?powerNode qudt:numericValue ?powerValue .
  $this nltl:recommendedChestVolume ?volumeNode ;
        nltl:strainerOpenArea ?openAreaNode ;
        nltl:inletPipeArea ?pipeAreaNode .
  ?volumeNode qudt:numericValue ?volumeValue .
  ?openAreaNode qudt:numericValue ?openAreaValue .
  ?pipeAreaNode qudt:numericValue ?pipeAreaValue .
  BIND((?powerValue / 750) AS ?expectedVolume)
  FILTER(
    ABS(?volumeValue - ?expectedVolume) <=
    (ABS(?expectedVolume) * 0.000001 + 0.000001)
  )
  FILTER(?openAreaValue >= (4 * ?pipeAreaValue))
}
"""
    ] .

PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:strainerOpenArea ?openAreaNode ;
        nltl:inletPipeArea ?pipeAreaNode .
  ?openAreaNode qudt:numericValue ?openAreaValue .
  ?pipeAreaNode qudt:numericValue ?pipeAreaValue .
  FILTER(?openAreaValue < (4 * ?pipeAreaValue))
