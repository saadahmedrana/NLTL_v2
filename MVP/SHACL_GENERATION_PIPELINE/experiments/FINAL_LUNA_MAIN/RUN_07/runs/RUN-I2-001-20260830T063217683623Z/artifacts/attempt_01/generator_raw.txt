<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-001> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A ship must provide its construction contract date to determine UR I2 Rev.4 applicability."
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "A ship must provide the derived UR I2 Rev.4 applicability state."
    ] ;
    sh:sparql [
        sh:message "UR I2 Rev.4 applicability must be true for ships contracted for construction on or after 1 January 2021 and false for earlier ships." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate ;
                      nltl:urI2RevisionApplicability ?applicable .
                FILTER (
                    (?contractDate >= "2021-01-01"^^xsd:date && ?applicable != true) ||
                    (?contractDate < "2021-01-01"^^xsd:date && ?applicable != false)
                )
            }
        """
    ] .
<END_SHACL>
