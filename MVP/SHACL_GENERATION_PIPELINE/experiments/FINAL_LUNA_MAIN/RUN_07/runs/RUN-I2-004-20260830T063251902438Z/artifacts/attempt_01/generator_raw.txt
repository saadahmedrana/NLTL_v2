<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-004-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterline ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the upper ice waterline." ;
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:class nltl:upperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ]
                ] ;
            ]
        ] ;
        sh:message "Each upper ice waterline case must provide its displacement in metric kilotonnes." ;
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The ship displacement at the upper ice waterline must be provided in metric kilotonnes." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline displacement must equal the displacement of one of the linked waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?reportedQuantity .
                ?reportedQuantity qudt:numericValue ?reportedValue .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?case .
                    ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                    ?caseQuantity qudt:numericValue ?caseValue .
                    FILTER (?caseValue = ?reportedValue)
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline displacement must be the greatest displacement among the linked waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?reportedQuantity .
                ?reportedQuantity qudt:numericValue ?reportedValue .
                $this nltl:hasUpperIceWaterlineCase ?case .
                ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseValue .
                FILTER (?caseValue > ?reportedValue)
            }
        """ ;
    ] .
<END_SHACL>
