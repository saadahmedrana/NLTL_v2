<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_009_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:class nltl:hullAreaAssignment ;
        sh:minCount 1 ;
        sh:node gs:I2_009_HullAreaAssignmentShape ;
        sh:message "A ship must provide a hull-area assignment for this requirement."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:I2_009_ButtockAngleShape ;
        sh:message "The buttock angle must be greater than 0 degrees and less than 80 degrees."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:I2_009_NormalFrameAngleShape ;
        sh:message "The normal frame angle must be greater than 10 degrees."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For every other ice-strengthened area, average ice pressure, load-patch aspect ratio, load-patch height, and load-patch length must be supplied with the required units; the aspect ratio must equal 3.6." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:hasHullAreaAssignment ?assignment .
  ?assignment nltl:assignedHullArea nltl:otherIceStrengthenedArea .
  FILTER (
    NOT EXISTS {
      $this nltl:averageIcePressure ?pressure .
      ?pressure qudt:numericValue ?pressureValue ;
                qudt:unit qudt:MegaPA .
      FILTER(datatype(?pressureValue) = xsd:decimal)
    }
    ||
    NOT EXISTS {
      $this nltl:loadPatchAspectRatio ?aspect .
      ?aspect qudt:numericValue ?aspectValue ;
             qudt:unit qudt:UNITLESS .
      FILTER(
        datatype(?aspectValue) = xsd:decimal &&
        ?aspectValue >= 3.6 &&
        ?aspectValue <= 3.6
      )
    }
    ||
    NOT EXISTS {
      $this nltl:nonBowLoadPatchHeight ?height .
      ?height qudt:numericValue ?heightValue ;
             qudt:unit qudt:M .
      FILTER(datatype(?heightValue) = xsd:decimal)
    }
    ||
    NOT EXISTS {
      $this nltl:nonBowLoadPatchLength ?length .
      ?length qudt:numericValue ?lengthValue ;
             qudt:unit qudt:M .
      FILTER(datatype(?lengthValue) = xsd:decimal)
    }
  )
}
"""
    ] .

gs:I2_009_HullAreaAssignmentShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaAssignment ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowIntermediateBottomHullArea
            nltl:midbodyBottomHullArea
            nltl:sternBottomHullArea
            nltl:otherIceStrengthenedArea
        ) ;
        sh:message "The hull-area assignment must identify one permitted hull area."
    ] .

gs:I2_009_ButtockAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0 ;
        sh:maxExclusive 80
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:I2_009_NormalFrameAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 10
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
