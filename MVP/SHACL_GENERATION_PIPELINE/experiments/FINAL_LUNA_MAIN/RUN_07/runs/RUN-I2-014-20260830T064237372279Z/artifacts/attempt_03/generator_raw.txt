            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthValue .
                ?lengthValue qudt:numericValue ?lbow .

                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case1 .
                    ?case1 nltl:bowSubregionMidLengthPosition ?position1 .
                    ?position1 qudt:numericValue ?p1 .
                    FILTER (
                        ABS(?p1 - (?lbow / 8)) <=
                        (ABS(?lbow / 8) * 0.000001 + 0.000000001)
                    )
                }

                    $this nltl:hasBowSubregionCalculationCase ?case2 .
