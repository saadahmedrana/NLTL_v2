<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_029_CalculationCaseShape ;
        sh:message "Each linked calculation case must satisfy the section modulus and shear area conditions."
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be included in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be excluded from the shear area."
    ] ;
    sh:or (
        gen:I2_029_FlangeExcludedCaseShape
        gen:I2_029_FlangeIncludedCaseShape
    ) .

gen:I2_029_FlangeExcludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Flange material inclusion in the shear area must be explicitly represented."
    ] .

gen:I2_029_FlangeIncludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Flange material inclusion in the shear area must be explicitly represented."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:minCount 1 ;
        sh:node gen:I2_029_FittedFlangeMemberShape ;
        sh:message "A calculation case including flange material in shear area must link to a fitted-flange structural member."
    ] .

gen:I2_029_FittedFlangeMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The linked structural member must have a fitted flange."
    ] .
<END_SHACL>
