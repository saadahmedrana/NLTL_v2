gen:NonObliqueCalculationCaseShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "A non-oblique calculation case cannot have a framing angle strictly between 20 and 70 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angle .
                ?angle qudt:numericValue ?value .
                FILTER ( ?value > "20"^^xsd:decimal && ?value < "70"^^xsd:decimal )
            }
        """
    ] .

gen:ObliqueCalculationCaseShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "An oblique framing calculation case requires a framing angle strictly between 20 and 70 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angle .
                ?angle qudt:numericValue ?value .
                FILTER ( ?value <= "20"^^xsd:decimal || ?value >= "70"^^xsd:decimal )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node gen:MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:sparql [
            sh:message "The local-frame area must exceed the attached-plate-flange area before the plastic neutral axis branch applies." ;
            sh:select """
                PREFIX nltl: <https://w3id.org/nltl/vocab#>
                PREFIX qudt: <http://qudt.org/schema/qudt/>
                PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                SELECT $this
                WHERE {
                    $this nltl:sectionCalculationCaseStructuralMember ?member .
                    ?member nltl:netLocalFrameFlangeArea ?area .
                    ?area qudt:numericValue ?areaValue .
                    ?member nltl:netAttachedShellPlateThickness ?thickness .
                    ?thickness qudt:numericValue ?thicknessValue .
                    ?member nltl:frameSpacing ?spacing .
                    ?spacing qudt:numericValue ?spacingValue .
                    FILTER ( 100 * ?areaValue <= 1000 * ?thicknessValue * ?spacingValue )
                }
            """
        ]
    ] ;
    sh:or (
        [ sh:node gen:NonObliqueCalculationCaseShape ]
        [ sh:node gen:ObliqueCalculationCaseShape ]
    ) .
