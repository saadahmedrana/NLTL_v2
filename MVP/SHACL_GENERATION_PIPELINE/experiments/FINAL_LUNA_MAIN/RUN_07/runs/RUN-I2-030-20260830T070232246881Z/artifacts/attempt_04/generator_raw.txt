gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:netEffectivePlasticSectionModulus ;
        sh:node gen:CubicCentimetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "When the local-frame area exceeds 1000 times the attached shell thickness and frame spacing, the calculation case shall provide the plastic neutral axis height and net effective plastic section modulus." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?localCase nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaNode ;
                        nltl:netAttachedShellPlateThickness ?thicknessNode ;
                        nltl:frameSpacing ?spacingNode .
                ?areaNode qudt:numericValue ?area .
                ?thicknessNode qudt:numericValue ?thickness .
                ?spacingNode qudt:numericValue ?spacing .
                FILTER (
                    100 * ?area > 1000 * ?thickness * ?spacing
                )
                FILTER NOT EXISTS {
                    $this nltl:plasticNeutralAxisHeight ?zNode .
                    ?zNode qudt:numericValue ?zValue .
                }
            }
        """ ;
    ] ;
    sh:sparql [
        sh:message "When the local-frame area exceeds 1000 times the attached shell thickness and frame spacing, the calculation case shall provide the net effective plastic section modulus." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?localCase nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaNode ;
                        nltl:netAttachedShellPlateThickness ?thicknessNode ;
                        nltl:frameSpacing ?spacingNode .
                ?areaNode qudt:numericValue ?area .
                ?thicknessNode qudt:numericValue ?thickness .
                ?spacingNode qudt:numericValue ?spacing .
                FILTER (
                    100 * ?area > 1000 * ?thickness * ?spacing
                )
                FILTER NOT EXISTS {
                    $this nltl:netEffectivePlasticSectionModulus ?zNode .
                    ?zNode qudt:numericValue ?zValue .
                }
            }
        """ ;
    ] ;
    sh:sparql [
        sh:message "For oblique framing with 20 degrees less than framing angle omega and framing angle omega less than 70 degrees, both interpolation endpoints shall be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleNode .
                ?angleNode qudt:numericValue ?angle .
                FILTER (
                    ?angle > 20 &&
                    ?angle < 70
                )
                FILTER NOT EXISTS {
                    $this nltl:interpolationLowerEndpoint ?lower .
                }
            }
        """ ;
    ] ;
    sh:sparql [
        sh:message "For oblique framing with 20 degrees less than framing angle omega and framing angle omega less than 70 degrees, both interpolation endpoints shall be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleNode .
                ?angleNode qudt:numericValue ?angle .
                FILTER (
                    ?angle > 20 &&
                    ?angle < 70
                )
                FILTER NOT EXISTS {
                    $this nltl:interpolationUpperEndpoint ?upper .
                }
            }
        """ ;
    ] .
