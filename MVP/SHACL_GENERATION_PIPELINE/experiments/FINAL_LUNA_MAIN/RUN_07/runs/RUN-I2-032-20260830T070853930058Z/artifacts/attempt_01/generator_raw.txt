    sh:sparql [
        sh:message "The required transverse frame shear area must equal the formula result within a relative tolerance of 1e-6, and the actual shear area must be at least the calculated required area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this ;
                      nltl:averageIcePressure ?pressureNode ;
                      nltl:frameSpan ?frameSpanNode ;
                      nltl:loadPatchHeight ?heightNode ;
                      nltl:peakPressureFactor ?peakFactorNode ;
                      nltl:shearArea ?actualAreaNode ;
                      nltl:yieldStrength ?yieldNode .

                $this nltl:frameSpacing ?spacingNode ;
                      nltl:selectedHullAreaFactor ?areaFactorNode ;
                      nltl:requiredTransverseFrameShearArea ?requiredAreaNode .

                ?pressureNode qudt:numericValue ?pressure .
                ?frameSpanNode qudt:numericValue ?frameSpan .
                ?heightNode qudt:numericValue ?height .
                ?peakFactorNode qudt:numericValue ?peakFactor .
                ?actualAreaNode qudt:numericValue ?actualArea .
                ?yieldNode qudt:numericValue ?yieldStrength .
                ?spacingNode qudt:numericValue ?spacing .
                ?areaFactorNode qudt:numericValue ?areaFactor .
                ?requiredAreaNode qudt:numericValue ?requiredArea .

                BIND(IF(?frameSpan <= ?height, ?frameSpan, ?height) AS ?ll)
                BIND((5000 * ?ll * ?spacing * ?areaFactor * ?peakFactor * ?pressure / (0.577 * ?yieldStrength)) AS ?expectedArea)

                FILTER (
                    ABS(?requiredArea - ?expectedArea) >
                    (ABS(?expectedArea) * 0.000001 + 0.000001)
                    ||
                    (?actualArea * 10000 < ?expectedArea)
                )
            }
        """
    ] .
