<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_047_SteelRenewalShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked plating structure must declare whether it is ice strengthened."
    ] ;
    sh:sparql [
        sh:message "For an ice-strengthened plating structure, gaugedThickness and netThickness must be metre-valued QUDT quantities and steelRenewalRequired must be a boolean." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:iceStrengthenedStructureStatus true .

                FILTER (
                    NOT EXISTS {
                        $this nltl:gaugedThickness ?gauged .
                        ?gauged a qudt:QuantityValue ;
                                qudt:numericValue ?gaugedValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?gaugedValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:netThickness ?net .
                        ?net a qudt:QuantityValue ;
                             qudt:numericValue ?netValue ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?netValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:steelRenewalRequired ?renewal .
                        FILTER (datatype(?renewal) = xsd:boolean)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For an ice-strengthened plating structure, steelRenewalRequired must be true exactly when gaugedThickness is less than netThickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:gaugedThickness ?gauged ;
                      nltl:netThickness ?net ;
                      nltl:steelRenewalRequired ?renewal .
                ?plating nltl:iceStrengthenedStructureStatus true .
                ?gauged a qudt:QuantityValue ;
                        qudt:numericValue ?gaugedValue ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                ?net a qudt:QuantityValue ;
                     qudt:numericValue ?netValue ;
                     qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (datatype(?gaugedValue) = xsd:decimal)
                FILTER (datatype(?netValue) = xsd:decimal)
                FILTER (
                    (?gaugedValue < (?netValue + 0.0005) && ?renewal != true)
                    ||
                    (?gaugedValue >= (?netValue + 0.0005) && ?renewal != false)
                )
            }
        """
    ] .
<END_SHACL>
