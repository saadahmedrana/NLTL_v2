nltlg:I2_051_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:hasValue nltl:positiveShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For positive shear force between the aft end and 0.6 LUI from aft, the longitudinal distribution factor must be Cf = 0.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?lui .
                $this nltl:hullGirderLongitudinalPositionFromAft ?position ;
                      nltl:longitudinalDistributionFactor ?factor ;
                      nltl:shearForceDirection nltl:positiveShearForce .
                ?position qudt:numericValue ?p .
                ?lui qudt:numericValue ?l .
                ?factor qudt:numericValue ?c .
                FILTER (
                    ?p >= "0.0"^^xsd:decimal &&
                    ?p <= ("0.6"^^xsd:decimal * ?l) &&
                    ?c != "0.0"^^xsd:decimal
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force between 0.9 LUI from aft and the forward end of LUI, the longitudinal distribution factor must be Cf = 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?lui .
                $this nltl:hullGirderLongitudinalPositionFromAft ?position ;
                      nltl:longitudinalDistributionFactor ?factor ;
                      nltl:shearForceDirection nltl:positiveShearForce .
                ?position qudt:numericValue ?p .
                ?lui qudt:numericValue ?l .
                ?factor qudt:numericValue ?c .
                FILTER (
                    ?p >= ("0.9"^^xsd:decimal * ?l) &&
                    ?p <= ?l &&
                    ?c != "1.0"^^xsd:decimal
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal Cf multiplied by the design vertical ice force at bow, with MegaN converted to KiloN; comparison uses a relative tolerance of 1e-6 with a 1e-9 KiloN absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:designVerticalIceForceAtBow ?fib ;
                      nltl:designVerticalIceShearForce ?fi .
                $this nltl:longitudinalDistributionFactor ?cf .
                ?fib qudt:numericValue ?fibValue .
                ?fi qudt:numericValue ?fiValue .
                ?cf qudt:numericValue ?cfValue .
                BIND (?cfValue * ?fibValue * "1000.0"^^xsd:decimal AS ?expected)
                FILTER (
                    ABS(?fiValue - ?expected) >
                    ("0.000001"^^xsd:decimal * ABS(?expected) +
                     "0.000000001"^^xsd:decimal)
                )
            }
        """
    ] .
