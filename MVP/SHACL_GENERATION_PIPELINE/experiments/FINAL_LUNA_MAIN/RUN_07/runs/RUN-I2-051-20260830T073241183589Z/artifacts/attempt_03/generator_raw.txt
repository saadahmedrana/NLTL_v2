    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;

                sh:path qudt:numericValue ;

            sh:property [

                sh:hasValue unit:KiloN ;

            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

    ] ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

                sh:maxCount 1

            sh:property [
                sh:path qudt:unit ;

                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:hullGirderLongitudinalPositionFromAft ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:shearForceDirection ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:positiveShearForce
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal the longitudinal distribution factor times the design vertical ice force at bow, with MegaN converted to KiloN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQ ;
                      nltl:designVerticalIceShearForce ?fiQ ;
                      nltl:longitudinalDistributionFactor ?cfQ ;
                      nltl:hasCalculationCase ?case .
                ?fibQ <http://qudt.org/schema/qudt/numericValue> ?fib .
                ?fiQ <http://qudt.org/schema/qudt/numericValue> ?fi .
                ?cfQ <http://qudt.org/schema/qudt/numericValue> ?cf .
                ?case nltl:shearForceDirection nltl:positiveShearForce .
                BIND (?cf * ?fib * 1000.0 AS ?expected)
                FILTER (ABS(?fi - ?expected) > ((ABS(?expected) * 0.000001) + 0.000000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force between the aft end and 0.6 times the upper ice waterline length from aft, the longitudinal distribution factor must be exactly 0.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineLengthLUI ?luiQ ;
                      nltl:longitudinalDistributionFactor ?cfQ ;
                      nltl:hasCalculationCase ?case .
                ?luiQ <http://qudt.org/schema/qudt/numericValue> ?lui .
                ?cfQ <http://qudt.org/schema/qudt/numericValue> ?cf .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                       nltl:hullGirderLongitudinalPositionFromAft ?positionQ .
                ?positionQ <http://qudt.org/schema/qudt/numericValue> ?position .
                FILTER (?position >= 0.0 && ?position <= (0.6 * ?lui))
                FILTER (?cf < 0.0 || ?cf > 0.0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force between 0.9 times the upper ice waterline length from aft and its forward end, the longitudinal distribution factor must be exactly 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineLengthLUI ?luiQ ;
                      nltl:longitudinalDistributionFactor ?cfQ ;
                      nltl:hasCalculationCase ?case .
                ?luiQ <http://qudt.org/schema/qudt/numericValue> ?lui .
                ?cfQ <http://qudt.org/schema/qudt/numericValue> ?cf .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                       nltl:hullGirderLongitudinalPositionFromAft ?positionQ .
                ?positionQ <http://qudt.org/schema/qudt/numericValue> ?position .
                FILTER (?position >= (0.9 * ?lui) && ?position <= ?lui)
                FILTER (?cf < 1.0 || ?cf > 1.0)
            }
        """
    ] .
