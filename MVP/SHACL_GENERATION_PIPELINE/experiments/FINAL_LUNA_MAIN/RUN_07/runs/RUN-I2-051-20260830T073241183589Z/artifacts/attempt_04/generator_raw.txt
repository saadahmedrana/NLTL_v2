<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_051_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hullGirderPosition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node nltl:I2_051_CalculationCaseShape
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal the longitudinal distribution factor multiplied by the design vertical ice force at bow, converted from MN to kN; comparison uses a relative tolerance of 1e-6 with a 1e-9 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibNode ;
                      nltl:designVerticalIceShearForce ?fiNode ;
                      nltl:longitudinalDistributionFactor ?cfNode .
                ?fibNode qudt:numericValue ?fib .
                ?fiNode qudt:numericValue ?fi .
                ?cfNode qudt:numericValue ?cf .
                BIND ((?cf * ?fib * 1000.0) AS ?expected)
                FILTER (ABS(?fi - ?expected) >
                    ((0.000001 * ABS(?expected)) + 0.000000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force between the aft end and 0.6 LUI from aft, the longitudinal distribution factor must be exactly 0.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:upperIceWaterlineLengthLUI ?luiNode .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionNode .
                ?luiNode qudt:numericValue ?lui .
                ?positionNode qudt:numericValue ?position .
                FILTER (?position >= (0.0 * ?lui) &&
                        ?position <= (0.6 * ?lui))
                FILTER NOT EXISTS {
                    $this nltl:longitudinalDistributionFactor ?cfNode .
                    ?cfNode qudt:numericValue ?cf .
                    FILTER (?cf >= 0.0 && ?cf <= 0.0)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force between 0.9 LUI from aft and the forward end of LUI, the longitudinal distribution factor must be exactly 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:upperIceWaterlineLengthLUI ?luiNode .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionNode .
                ?luiNode qudt:numericValue ?lui .
                ?positionNode qudt:numericValue ?position .
                FILTER (?position >= (0.9 * ?lui) &&
                        ?position <= ?lui)
                FILTER NOT EXISTS {
                    $this nltl:longitudinalDistributionFactor ?cfNode .
                    ?cfNode qudt:numericValue ?cf .
                    FILTER (?cf >= 1.0 && ?cf <= 1.0)
                }
            }
        """
    ] .

nltl:I2_051_CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:positiveShearForce
    ] .

<END_SHACL>
