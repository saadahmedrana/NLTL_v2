<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node g:CalculationCaseShape
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship shall provide an interpolated value for the linear-interpolation result."
    ] .

g:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:node g:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case shall provide one hull-girder longitudinal position from aft."
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:node g:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node g:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case shall identify one lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node g:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case shall identify one upper interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:node g:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case shall provide one interpolated longitudinal distribution factor."
    ] ;
    sh:sparql [
        sh:message "The interpolation case shall link distinct lower and upper points, contain the input coordinate within their ordered interval, and report the linear-interpolation result within the endpoint envelope using a relative tolerance of 1e-6 with a 1e-6 absolute fallback." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hullGirderLongitudinalPositionFromAft ?xq .
  ?xq qudt:numericValue ?x .

  $this nltl:interpolationLowerEndpoint ?lo .
  $this nltl:interpolationUpperEndpoint ?hi .
  $this nltl:hasInterpolationPoint ?lo .
  $this nltl:hasInterpolationPoint ?hi .
  FILTER (?lo != ?hi)

  ?lo nltl:interpolationCoordinate ?lcq .
  ?lo nltl:interpolationPointValue ?lvq .
  ?hi nltl:interpolationCoordinate ?ucq .
  ?hi nltl:interpolationPointValue ?uvq .
  ?lcq qudt:numericValue ?lc .
  ?lvq qudt:numericValue ?lv .
  ?ucq qudt:numericValue ?uc .
  ?uvq qudt:numericValue ?uv .

  $this nltl:interpolatedLongitudinalDistributionFactor ?rq .
  ?rq qudt:numericValue ?r .

  BIND(
    ?lv + ((?x - ?lc) / (?uc - ?lc)) * (?uv - ?lv)
    AS ?expected
  )

  FILTER (
    ?lc >= ?uc
    || ?x < ?lc
    || ?x > ?uc
    || (
      (?lv <= ?uv && (?r < ?lv || ?r > ?uv))
      ||
      (?lv > ?uv && (?r < ?uv || ?r > ?lv))
    )
    || ABS(?r - ?expected) >
       (0.000001 * ABS(?expected) + 0.000001)
  )
}
"""
    ] .

g:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:node g:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each interpolation point shall provide one interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:node g:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each interpolation point shall provide one interpolation point value."
    ] .

g:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

g:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

g:QuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
