<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_055_MPaQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_055_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:node gen:I2_055_MPaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide one design stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:node gen:I2_055_MPaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide one permissible stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Table 9 strength criteria must be satisfied."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The design stress must not exceed the permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designStress ?designQuantity .
                ?designQuantity qudt:numericValue ?designValue .
                $this nltl:permissibleStress ?permissibleQuantity .
                ?permissibleQuantity qudt:numericValue ?permissibleValue .
                FILTER (?designValue > ?permissibleValue)
            }
        """
    ] .
<END_SHACL>
