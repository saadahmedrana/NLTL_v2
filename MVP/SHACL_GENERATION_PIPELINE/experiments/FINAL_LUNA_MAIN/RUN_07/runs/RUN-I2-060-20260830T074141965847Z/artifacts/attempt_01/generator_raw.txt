<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    ] ;
    sh:sparql [
        sh:message "A structural stiffening description is required when a linked member in way of a cutout requires stiffening." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                        nltl:stiffeningNecessary true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:structuralStiffening ?value .
                        FILTER (datatype(?value) = xsd:string)
                    }
                    ||
                    EXISTS {
                        $this nltl:structuralStiffening ?invalidValue .
                        FILTER (datatype(?invalidValue) != xsd:string)
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasCutout ;
                sh:minCount 1 ;
                sh:class nltl:cutout
            ] ;
            sh:property [
                sh:path nltl:memberInstability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:stiffeningNecessary ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:stiffeningNecessary ;
                        sh:hasValue true
                    ]
                ]
            )
        ]
    ) .
<END_SHACL>
