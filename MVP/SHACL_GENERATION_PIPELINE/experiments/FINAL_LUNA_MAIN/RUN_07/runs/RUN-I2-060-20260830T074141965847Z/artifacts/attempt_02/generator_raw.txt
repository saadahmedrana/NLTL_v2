<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_060_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_060_StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "A member in way of a cutout must not be unstable; ship-owned memberInstability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                        nltl:hasCutout ?cutout .
                FILTER (
                    NOT EXISTS {
                        $this nltl:memberInstability false
                    }
                    ||
                    EXISTS {
                        $this nltl:memberInstability true
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ship-owned structural stiffening is required when stiffening is necessary for a member in way of a cutout." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                        nltl:hasCutout ?cutout ;
                        nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (datatype(?stiffening) = xsd:string)
                }
            }
        """
    ] .

gen:I2_060_StructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasCutout ;
                sh:class nltl:cutout ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
<END_SHACL>
