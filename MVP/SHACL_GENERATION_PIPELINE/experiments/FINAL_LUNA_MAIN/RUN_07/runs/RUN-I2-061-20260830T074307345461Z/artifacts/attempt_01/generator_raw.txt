<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_061_CalculationCaseShape ;
        sh:message "The ship must identify a calculation case governed by the prescribed analytical procedure."
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_061_HullStructureShape ;
        sh:message "The calculation case must identify its assessed shell-plating or local-frame hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "The calculation case must use the prescribed analytical procedure and not a direct calculation."
    ] .

gen:I2_061_HullStructureShape
    a sh:NodeShape ;
    sh:class nltl:hullStructure ;
    sh:or (
        [ sh:class nltl:plating ]
        [ sh:class nltl:structuralMember ]
    ) ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_061_ThicknessQuantityShape ;
        sh:message "The assessed hull structure must provide its thickness in metres."
    ] .

gen:I2_061_ThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness must use metres as its unit."
    ] .
<END_SHACL>
