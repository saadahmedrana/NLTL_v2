<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-065-ship> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:nonlinearCalculationMethod ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A nonlinear calculation method must be identified to establish applicability of the nonlinear-analysis readiness requirement."
    ] ;
    sh:property [
        sh:path nltl:bucklingResponse ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The nonlinear analysis record must state that buckling is reliably captured."
    ] ;
    sh:property [
        sh:path nltl:plasticDeformation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The nonlinear analysis record must state that plastic deformation is reliably captured."
    ] ;
    sh:property [
        sh:path nltl:fractureMargin ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:majorBucklingMargin ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldingMargin ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase
    ] ;
    sh:sparql [
        sh:message "An applicable nonlinear calculation case must use the nonlinear calculation method and provide acceptance evidence, satisfied minor-deformation assessment, permanent lateral and out-of-plane deformation, and the relevant structural dimension in the prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:nonlinearCalculationMethod ?method
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasCalculationCase ?case .
                        ?case a nltl:calculationCase ;
                              nltl:calculationMethod nltl:nonlinearCalculationMethodValue
                    }
                    ||
                    EXISTS {
                        $this nltl:hasCalculationCase ?badCase .
                        ?badCase a nltl:calculationCase ;
                                 nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
                        FILTER (
                            NOT EXISTS {
                                ?badCase nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                                ?evidence a nltl:evidenceArtifact
                            }
                            ||
                            NOT EXISTS {
                                ?badCase nltl:minorPermanentDeformationAssessmentSatisfied true
                            }
                            ||
                            NOT EXISTS {
                                ?badCase nltl:permanentLateralDeformation ?lateral .
                                ?lateral qudt:numericValue ?lateralValue ;
                                         qudt:unit qudt:MilliM .
                                FILTER (datatype(?lateralValue) = xsd:decimal)
                            }
                            ||
                            NOT EXISTS {
                                ?badCase nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                                ?outOfPlane qudt:numericValue ?outOfPlaneValue ;
                                            qudt:unit qudt:MilliM .
                                FILTER (datatype(?outOfPlaneValue) = xsd:decimal)
                            }
                            ||
                            NOT EXISTS {
                                ?badCase nltl:relevantStructuralDimension ?dimension .
                                ?dimension qudt:numericValue ?dimensionValue ;
                                           qudt:unit qudt:MilliM .
                                FILTER (datatype(?dimensionValue) = xsd:decimal)
                            }
                        )
                    }
                )
            }
        """
    ] .

</END_SHACL>
