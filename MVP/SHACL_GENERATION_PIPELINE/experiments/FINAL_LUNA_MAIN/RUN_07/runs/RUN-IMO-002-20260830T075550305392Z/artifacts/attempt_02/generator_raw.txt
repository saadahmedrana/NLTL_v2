<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_002_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:thinFirstYearIceWithPossibleOldIceInclusions
            nltl:mediumFirstYearIceWithPossibleOldIceInclusions
            nltl:iceConditionLessSevereThanCategoryAAndB
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "A ship must have exactly one permitted design ice condition." ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:message "A ship must have exactly one permitted Polar Code category." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryA ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:iceConditionLessSevereThanCategoryAAndB
                    nltl:iceFreeIceCondition
                    nltl:openWaterIceCondition
                    nltl:otherWatersIceCondition
                ) ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions
                ) ;
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryB ;
            ]
        ]
    ) ;
    sh:message "A non-category-A ship designed for at least thin first-year ice, including possible old-ice inclusions, must be category B." .
<END_SHACL>
