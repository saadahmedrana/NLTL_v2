<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-003-ShipCategoryC>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:openWaterIceCondition
                    nltl:iceConditionLessSevereThanCategoryAAndB
                )
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:polarShipCategoryValue ;
                sh:minCount 1 ;
                sh:hasValue nltl:polarShipCategoryC
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For open water or ice conditions less severe than categories A and B, shipCategory must be exactly Category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipCategory nltl:polarShipCategoryC .
                    }
                    ||
                    EXISTS {
                        $this nltl:shipCategory ?declaredCategory .
                        FILTER (?declaredCategory != nltl:polarShipCategoryC)
                    }
                )
            }
        """
    ] .
<END_SHACL>
