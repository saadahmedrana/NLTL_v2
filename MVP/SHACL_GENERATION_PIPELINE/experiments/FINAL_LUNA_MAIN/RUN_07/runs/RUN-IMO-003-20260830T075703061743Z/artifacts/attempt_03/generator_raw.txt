<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-003-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition to determine applicability of the Category C rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship designed for open water or ice conditions less severe than categories A and B, shipCategory must be exactly Category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipCategory nltl:polarShipCategoryC
                    }
                    ||
                    EXISTS {
                        $this nltl:shipCategory ?declaredCategory .
                        FILTER (?declaredCategory != nltl:polarShipCategoryC)
                    }
                )
            }
        """
    ] .
<END_SHACL>
