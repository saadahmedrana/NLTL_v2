<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_StaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship has sea ice of no more than one winter growth and thickness from 0.3 m through 2.0 m, its ice type classification must be first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
          nltl:winterGrowthCount ?growth ;
          nltl:iceThickness ?thicknessNode .
    ?thicknessNode qudt:numericValue ?thicknessValue ;
                   qudt:unit unit:M .
    FILTER (
        datatype(?growth) = xsd:integer &&
        ?growth <= 1 &&
        datatype(?thicknessValue) = xsd:decimal &&
        ?thicknessValue >= 0.3 &&
        ?thicknessValue <= 2.0
    )
    FILTER NOT EXISTS {
        $this nltl:iceTypeClassification nltl:firstYearIce .
    }
}
""" ;
    ] .
<END_SHACL>
