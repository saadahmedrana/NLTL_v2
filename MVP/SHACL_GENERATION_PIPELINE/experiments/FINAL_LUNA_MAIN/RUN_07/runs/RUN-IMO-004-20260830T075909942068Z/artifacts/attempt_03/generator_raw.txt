<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:iceOriginValue
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Applicable ships must have exactly the first-year ice classification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                      nltl:winterGrowthCount ?growth ;
                      nltl:iceThickness ?thickness .

                ?thickness qudt:numericValue ?thicknessValue ;
                           qudt:unit unit:M .

                FILTER (
                    ?growth <= 1 &&
                    ?thicknessValue >= 0.3 &&
                    ?thicknessValue <= 2.0
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:iceTypeClassification nltl:firstYearIce
                    }
                    ||
                    EXISTS {
                        $this nltl:iceTypeClassification ?classification .
                        FILTER (?classification != nltl:firstYearIce)
                    }
                )
            }
        """
    ] .
<END_SHACL>
