<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_006_MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceTypeValue
        ] ;
        sh:message "A ship must provide an ice type classification."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For medium first-year ice, ice thickness must be a QUDT length quantity in metres between 0.7 m and 1.2 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?q0 .
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?q1 .
                        FILTER NOT EXISTS {
                            ?q1 a qudt:QuantityValue ;
                                qudt:numericValue ?v ;
                                qudt:unit unit:M .
                            FILTER (
                                datatype(?v) = xsd:decimal
                                && ?v >= "0.7"^^xsd:decimal
                                && ?v <= "1.2"^^xsd:decimal
                            )
                            FILTER NOT EXISTS {
                                ?q1 qudt:unit ?otherUnit .
                                FILTER (?otherUnit != unit:M)
                            }
                            FILTER NOT EXISTS {
                                ?q1 qudt:numericValue ?badValue .
                                FILTER (
                                    datatype(?badValue) != xsd:decimal
                                    || ?badValue < "0.7"^^xsd:decimal
                                    || ?badValue > "1.2"^^xsd:decimal
                                )
                            }
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
