<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:IMO_006_MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as medium first-year ice must provide at least one valid ice thickness quantity; every supplied quantity must be a decimal metre value between 0.7 m and 1.2 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?q0 .
                        ?q0 a qudt:QuantityValue .
                        ?q0 qudt:numericValue ?v0 .
                        ?q0 qudt:unit unit:M .
                        FILTER (
                            datatype(?v0) = xsd:decimal &&
                            ?v0 >= "0.7"^^xsd:decimal &&
                            ?v0 <= "1.2"^^xsd:decimal
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?q1 .
                        FILTER (
                            NOT EXISTS {
                                ?q1 a qudt:QuantityValue
                            }
                            ||
                            NOT EXISTS {
                                ?q1 qudt:numericValue ?v1
                            }
                            ||
                            EXISTS {
                                ?q1 qudt:numericValue ?v2 .
                                FILTER (
                                    !(datatype(?v2) = xsd:decimal &&
                                      ?v2 >= "0.7"^^xsd:decimal &&
                                      ?v2 <= "1.2"^^xsd:decimal)
                                )
                            }
                            ||
                            NOT EXISTS {
                                ?q1 qudt:unit unit:M
                            }
                            ||
                            EXISTS {
                                ?q1 qudt:unit ?u1 .
                                FILTER (?u1 != unit:M)
                            }
                        )
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
