<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO007OpenWaterShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship's ice condition is open water, sea ice concentration must be a decimal below 1 with unitless units, and ice of land origin must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?condition .
                FILTER (?condition = nltl:openWaterIceCondition)
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?quantity .
                        ?quantity qudt:numericValue ?concentration ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?concentration) = xsd:decimal)
                        FILTER (?concentration < 1)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent ?present .
                        FILTER (datatype(?present) = xsd:boolean)
                        FILTER (?present = false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
