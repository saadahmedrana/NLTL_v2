<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO007OpenWaterShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open water, sea ice concentration must have a qualifying unitless value below 1 and no supplied concentration may be 1 or greater; ice of land origin must be explicitly false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER (
                    EXISTS {
                        $this nltl:seaIceConcentration ?badConcentration .
                        ?badConcentration qudt:numericValue ?badValue .
                        FILTER (?badValue >= 1)
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:unit ?unit .
                        FILTER (?unit != qudt:UNITLESS)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?qualifyingConcentration .
                        ?qualifyingConcentration qudt:unit qudt:UNITLESS ;
                            qudt:numericValue ?qualifyingValue .
                        FILTER (?qualifyingValue < 1)
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?state .
                        FILTER (?state = true)
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?state .
                        FILTER (datatype(?state) != xsd:boolean)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent false
                    }
                )
            }
        """
    ] .
<END_SHACL>
