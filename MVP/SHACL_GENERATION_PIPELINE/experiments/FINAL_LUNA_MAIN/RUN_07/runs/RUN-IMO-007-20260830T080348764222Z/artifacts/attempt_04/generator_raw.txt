<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO_007_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship classified as open water, sea ice concentration must have a unitless quantity value less than 1, with no conflicting or non-unit concentration value, and ice of land origin must be explicitly false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER (
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?value .
                        FILTER (?value >= 1)
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:unit ?unit .
                        FILTER (?unit != <http://qudt.org/vocab/unit/UNITLESS>)
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        FILTER NOT EXISTS {
                            ?concentration qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:unit <http://qudt.org/vocab/unit/UNITLESS> ;
                                      qudt:numericValue ?value .
                        FILTER (?value < 1)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent false .
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?present .
                        FILTER (?present != false)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
