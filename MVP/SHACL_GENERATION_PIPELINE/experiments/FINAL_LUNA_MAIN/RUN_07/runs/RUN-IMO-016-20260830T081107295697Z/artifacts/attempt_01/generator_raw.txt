<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO016PolarShipCertificateShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path ( nltl:hasPolarShipCertificate nltl:certificateFormModel ) ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate shall use the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate language is required to determine whether an approved translation is needed."
    ] ;
    sh:sparql [
        sh:message "If the certificate language is not English, French, or Spanish, an approved translation shall be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (?language NOT IN ("English", "French", "Spanish"))
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent true
                }
            }
        """
    ] .
<END_SHACL>
