<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A Polar Ship Certificate form is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO017SolasScheduleShape ;
        sh:message "A SOLAS certificate schedule with date records is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO017SupplementShape ;
        sh:message "A Polar Certificate supplement with required equipment records is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every SOLAS certificate schedule date must match the corresponding Polar Ship Certificate date, and every Polar Ship Certificate date must have a corresponding SOLAS schedule record of the correct category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:polarCertificateValidityDates ?date
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:surveyDates ?date
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:endorsementDates ?date
                    }
                }
                UNION
                {
                    $this nltl:polarCertificateValidityDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
                UNION
                {
                    $this nltl:surveyDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
                UNION
                {
                    $this nltl:endorsementDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
            }
        """
    ] .

gen:IMO017SolasScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO017DateRecordShape ;
        sh:message "The SOLAS certificate schedule must contain certificate schedule date records."
    ] .

gen:IMO017DateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each certificate schedule date record must have one permitted date category."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each certificate schedule date record must contain one xsd:date value."
    ] .

gen:IMO017SupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Certificate supplement must contain required equipment records."
    ] .
<END_SHACL>
