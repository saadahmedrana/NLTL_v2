gs:IMO_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:IMO_037_ApplicableShipBranch
        gs:IMO_037_NonApplicableShipBranch
    ) .

gs:IMO_037_ApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryA
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionCase ;
        sh:node gs:IMO_037_LoadingConditionCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Category A ships constructed on or after 2017-01-01 require the loading-condition residual stability outcome or the permitted alternative-instrument outcome." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date .
                FILTER (
                    ?category = nltl:polarShipCategoryA &&
                    (
                        YEAR(?date) > 2017 ||
                        (YEAR(?date) = 2017 && MONTH(?date) > 1) ||
                        (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
                    )
                )
            }
        """
    ] .

gs:IMO_037_NonApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A non-applicable ship must establish that its category is not A or B, or that its construction date is before 2017-01-01." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    ) &&
                    YEAR(?date) > 2017 ||
                    (
                        ?category IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        ) &&
                        YEAR(?date) = 2017 &&
                        MONTH(?date) = 1 &&
                        DAY(?date) >= 1
                    )
                )
            }
        """
    ] .

gs:IMO_037_StandardResidualStabilityOutcome
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:residualStabilityFactorSI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:IMO_037_ResidualStabilityQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally calculated residual stability factor must be within the inclusive 1e-6 tolerance scaled from the expected value 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:residualStabilityFactorSI ?quantity .
                ?quantity qudt:numericValue ?value .
                FILTER (?value < 0.999999 || ?value > 1.000001)
            }
        """
    ] .
