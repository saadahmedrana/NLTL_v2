<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TransversePenetrationQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "760.0"^^xsd:decimal ;
        sh:maxInclusive "760.0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO038Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:node gen:LengthMQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:node gen:TransversePenetrationQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "longitudinalDamageExtent must equal 4.5% of upperIceWaterlineLength when the damage center is forward of maximum breadth, otherwise 1.5%, within the permitted numerical tolerance" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:longitudinalDamageExtent ?extentNode .
                ?lengthNode qudt:numericValue ?length .
                ?extentNode qudt:numericValue ?extent .
                FILTER (datatype(?length) = xsd:decimal)
                FILTER (datatype(?extent) = xsd:decimal)
                BIND(IF(?forward = true, "0.045"^^xsd:decimal * ?length,
                                      "0.015"^^xsd:decimal * ?length) AS ?expected)
                FILTER (ABS(?extent - ?expected) >
                        (ABS(?expected) * "0.000001"^^xsd:decimal +
                         "0.000000001"^^xsd:decimal))
            }
        """
    ] ;
    sh:sparql [
        sh:message "verticalDamageExtent must equal the lesser of 20% of upperIceWaterlineDraught and longitudinalDamageExtent within the permitted numerical tolerance" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode .
                ?draughtNode qudt:numericValue ?draught .
                ?longitudinalNode qudt:numericValue ?longitudinal .
                ?verticalNode qudt:numericValue ?vertical .
                FILTER (datatype(?draught) = xsd:decimal)
                FILTER (datatype(?longitudinal) = xsd:decimal)
                FILTER (datatype(?vertical) = xsd:decimal)
                BIND("0.20"^^xsd:decimal * ?draught AS ?draughtLimit)
                BIND(IF(?draughtLimit < ?longitudinal, ?draughtLimit, ?longitudinal) AS ?expected)
                FILTER (ABS(?vertical - ?expected) >
                        (ABS(?expected) * "0.000001"^^xsd:decimal +
                         "0.000000001"^^xsd:decimal))
            }
        """
    ] ;
    sh:sparql [
        sh:message "selectedVerticalDamagePosition must be between the keel and 120% of upperIceWaterlineDraught" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .
                ?draughtNode qudt:numericValue ?draught .
                ?positionNode qudt:numericValue ?position .
                FILTER (datatype(?draught) = xsd:decimal)
                FILTER (datatype(?position) = xsd:decimal)
                FILTER (?position < "0.0"^^xsd:decimal ||
                        ?position > "1.20"^^xsd:decimal * ?draught)
            }
        """
    ] .
<END_SHACL>
