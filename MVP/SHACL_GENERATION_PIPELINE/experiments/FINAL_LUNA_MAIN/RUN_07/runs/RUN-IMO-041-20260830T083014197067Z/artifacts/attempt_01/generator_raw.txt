<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-041> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must specify its opening actuation type so the hydraulic-fluid protection requirement can be evaluated."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Hydraulically operated hatches or doors require means to prevent freezing or excessive viscosity of liquids." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuationType .
                FILTER (STR(?actuationType) = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent ?protectionPresent .
                    FILTER (?protectionPresent = true)
                }
            }
        """
    ] .
<END_SHACL>
