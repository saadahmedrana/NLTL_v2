<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltlg:IMO_043_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:node nltlg:IMO_043_MachineryInstallationShape ;
        sh:message "Each linked machinery installation must represent protection coverage for all specified applicable hazards and have protection evidence."
    ] .

nltlg:IMO_043_MachineryInstallationShape
    a sh:NodeShape ;
    sh:class nltl:machineryInstallationItem ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:in (
            nltl:iceAccretionHazard
            nltl:snowAccumulationHazard
            nltl:seawaterIceIngestionHazard
            nltl:liquidFreezingOrViscosityIncreaseHazard
            nltl:seawaterIntakeTemperatureHazard
            nltl:snowIngestionHazard
        ) ;
        sh:minCount 6 ;
        sh:maxCount 6 ;
        sh:message "The machinery installation must identify all six applicable protection hazards."
    ] ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem ;
        sh:node nltlg:IMO_043_AssociatedEquipmentShape ;
        sh:message "Each linked associated equipment item must be represented under the covered machinery installation."
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Each machinery installation and its linked associated equipment must have protection evidence."
    ] .

nltlg:IMO_043_AssociatedEquipmentShape
    a sh:NodeShape ;
    sh:class nltl:equipmentItem .
<END_SHACL>
