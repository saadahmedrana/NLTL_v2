<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-047-QuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:IMO-047-EngineValueShape>
    a sh:NodeShape ;
    sh:class nltl:essentialEngine .

<urn:nltl:generated-shape:IMO-047-ApplicableShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-047-EngineValueShape>
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-047-QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-047-QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-047-QuantityValueShape>
    ] ;
    sh:sparql [
        sh:message "Combustion air temperature must be within the manufacturer minimum and maximum air-temperature criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureNode .
                ?temperatureNode qudt:numericValue ?temperatureValue .
                $this nltl:manufacturerMinAirTemperature ?minimumNode .
                ?minimumNode qudt:numericValue ?minimumValue .
                $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                ?maximumNode qudt:numericValue ?maximumValue .
                FILTER (
                    xsd:decimal(?temperatureValue) < xsd:decimal(?minimumValue)
                    || xsd:decimal(?temperatureValue) > xsd:decimal(?maximumValue)
                )
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-047-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:maxCount 0
            ]
        ]
        [
            sh:node <urn:nltl:generated-shape:IMO-047-ApplicableShape>
        ]
    ) .
<END_SHACL>
