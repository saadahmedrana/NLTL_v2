<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-050> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fireSafetyComponentExposed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceAndSnowProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When a fire safety component is exposed, ice and snow protection must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:fireSafetyComponentExposed true .
                FILTER NOT EXISTS {
                    $this nltl:iceAndSnowProtectionPresent true .
                }
            }
        """
    ] .
<END_SHACL>
