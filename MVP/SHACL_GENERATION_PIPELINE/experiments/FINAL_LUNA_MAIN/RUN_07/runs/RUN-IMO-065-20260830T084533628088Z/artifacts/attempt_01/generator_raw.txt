<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-065-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:abandonmentEnvironment ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must identify the assessed abandonment environment(s)."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:node <urn:nltl:generated-shape:IMO-065-TimeQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The maximum expected rescue time must be provided in days."
    ] ;
    sh:property [
        sh:path nltl:survivalSupportDuration ;
        sh:node <urn:nltl:generated-shape:IMO-065-TimeQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The survival support duration must be provided in days."
    ] ;
    sh:property [
        sh:path nltl:habitableEnvironmentPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must provide a habitable environment."
    ] ;
    sh:property [
        sh:path nltl:weatherProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must protect persons from cold, wind, and sun."
    ] ;
    sh:property [
        sh:path nltl:accommodationSpacePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must provide accommodation space."
    ] ;
    sh:property [
        sh:path nltl:sustenanceMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must provide means of sustenance."
    ] ;
    sh:property [
        sh:path nltl:safeAccessExitPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must provide safe access and exit points."
    ] ;
    sh:property [
        sh:path nltl:rescueCommunicationPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources must provide communication with rescue assets."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The survival support duration must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:survivalSupportDuration ?supportNode .
                ?supportNode qudt:numericValue ?supportValue .
                $this nltl:maximumExpectedRescueTime ?rescueNode .
                ?rescueNode qudt:numericValue ?rescueValue .
                FILTER(datatype(?supportValue) = xsd:decimal)
                FILTER(datatype(?rescueValue) = xsd:decimal)
                FILTER(?supportValue < ?rescueValue)
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-065-TimeQuantityShape> a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qudt:DAY
    ] .
<END_SHACL>
