<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_073_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboat ;
        sh:node gen:IMO_073_LifeboatShape ;
        sh:message "Each supplied lifeboat must use an allowed lifeboat type when a type is represented."
    ] .

gen:IMO_073_LifeboatShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:lifeboatTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboatTypeValue ;
        sh:in (
            nltl:partiallyEnclosedLifeboat
            nltl:totallyEnclosedLifeboat
        ) ;
        sh:message "A lifeboat type must be partially enclosed or totally enclosed."
    ] .
<END_SHACL>
