<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_075_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gs:IMO_075_ApplicableShape
        gs:IMO_075_NotApplicableShape
    ) .

gs:IMO_075_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:availableSurvivalEquipmentCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:storageAccessibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:sparql [
        sh:message "Available survival equipment capacity must be at least 110 percent of the persons on board, rounded up." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (?capacity < CEIL(?persons * 1.10))
            }
        """
    ] .

gs:IMO_075_NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue false
    ] .
<END_SHACL>
