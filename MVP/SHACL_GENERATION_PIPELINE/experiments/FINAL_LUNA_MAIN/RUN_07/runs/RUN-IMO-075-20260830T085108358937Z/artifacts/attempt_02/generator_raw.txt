<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO075ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether personal or group survival equipment is required."
    ] ;
    sh:or (
        gs:IMO075ApplicableBranch
        gs:IMO075NotApplicableBranch
    ) .

gs:IMO075ApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "When survival equipment is required, the persons-on-board count must be provided."
    ] ;
    sh:property [
        sh:path nltl:availableSurvivalEquipmentCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "When survival equipment is required, available survival equipment capacity must be provided."
    ] ;
    sh:property [
        sh:path nltl:storageAccessibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Required survival equipment must be stored in an easily accessible location."
    ] ;
    sh:property [
        sh:path nltl:storageDistanceToMusterOrEmbarkation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:IMO075StorageDistanceQuantityShape ;
        sh:message "The storage location must include its distance to a muster or embarkation station."
    ] ;
    sh:sparql [
        sh:message "Available survival equipment capacity must be at least 110 percent of the persons on board." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:personalOrGroupSurvivalEquipmentRequired true ;
                      nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (?capacity * 10 < ?persons * 11)
            }
        """
    ] .

gs:IMO075NotApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:hasValue false
    ] .

gs:IMO075StorageDistanceQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
