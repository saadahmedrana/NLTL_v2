<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_079_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Persons on board must be provided as a single integer value."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Maximum expected rescue time must be provided as a single day quantity."
    ] ;
    sh:property [
        sh:path nltl:availableEmergencyRationPerson ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Available emergency ration capacity must be provided as a single day quantity."
    ] ;
    sh:sparql [
        sh:message "Available emergency ration person-days must be at least persons on board multiplied by maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:personsOnBoard ?persons ;
                      nltl:maximumExpectedRescueTime ?rescueQuantity ;
                      nltl:availableEmergencyRationPerson ?rationQuantity .
                ?rescueQuantity qudt:numericValue ?rescueTime .
                ?rationQuantity qudt:numericValue ?ration .
                FILTER (?ration < (?persons * ?rescueTime))
            }
        """
    ] .
<END_SHACL>
