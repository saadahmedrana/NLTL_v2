<urn:nltl:generated-shape:IMO_082_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bridge configuration required to determine astern-view applicability shall be provided."
    ] ;
    sh:sparql [
        sh:message "A clear view astern shall be confirmed when the bridge configuration requires it." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bridgeConfigurationType
                      nltl:asternViewRequiredBridgeConfiguration .
                FILTER NOT EXISTS {
                    $this nltl:clearViewAsternStatus ?status .
                    FILTER (
                        datatype(?status) = xsd:boolean &&
                        ?status = true
                    )
                }
            }
        """
    ] .
