<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO098ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO098ApplicableBranch
        gen:IMO098NotApplicableBranch
    ) .

gen:IMO098ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that it operates in low air temperature when this requirement applies."
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft ;
        sh:node gen:IMO098OtherSurvivalCraftShape
    ] .

gen:IMO098NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] .

gen:IMO098OtherSurvivalCraftShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one location-signal device."
    ] ;
    sh:property [
        sh:path nltl:hasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one on-scene transmit/receive communication device."
    ] .
<END_SHACL>
