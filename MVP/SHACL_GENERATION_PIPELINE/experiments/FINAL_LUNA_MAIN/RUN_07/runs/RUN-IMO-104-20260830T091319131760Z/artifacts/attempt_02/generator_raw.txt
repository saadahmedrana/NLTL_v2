<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization." 
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gen:CrewMemberShape
    ] .

gen:CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty ;
        sh:node gen:AssignedDutyShape
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:node gen:FamiliarizationRecordShape ;
        sh:minCount 1 ;
        sh:message "Each crew member shall have a familiarization record."
    ] ;
    sh:sparql [
        sh:message "Each PWOM procedure or equipment item relevant to an assigned duty shall be covered by the crew member's familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    $this nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gen:AssignedDutyShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:node gen:ManualItemShape
    ] .

gen:FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:node gen:ManualItemShape ;
        sh:minCount 1 ;
        sh:message "The familiarization record shall cover at least one PWOM procedure or equipment item."
    ] .

gen:ManualItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .
<END_SHACL>
