<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_105_Arctic_Oil_Discharge_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship must identify its operating area to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of oil or oily mixtures into the sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (STR(?area) = "Arctic_waters")
                FILTER EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                    FILTER (
                        datatype(?discharge) = xsd:boolean
                        && ?discharge = false
                    )
                }
                FILTER ( ! EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea ?other .
                    FILTER ( ! (
                        datatype(?other) = xsd:boolean
                        && ?other = false
                    ) )
                } )
            }
        """ ;
    ] .
<END_SHACL>
