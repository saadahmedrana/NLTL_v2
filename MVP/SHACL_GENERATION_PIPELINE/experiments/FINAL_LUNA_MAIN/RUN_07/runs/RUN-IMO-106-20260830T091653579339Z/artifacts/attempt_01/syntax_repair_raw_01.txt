<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_106_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition applicability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?requiredValue .
                        FILTER (?requiredValue = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?conflictingValue .
                        FILTER (?conflictingValue != false)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
