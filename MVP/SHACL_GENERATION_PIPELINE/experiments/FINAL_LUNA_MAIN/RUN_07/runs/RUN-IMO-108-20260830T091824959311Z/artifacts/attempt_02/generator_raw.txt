<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_108_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:oilFuelTankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Category A or B ships constructed on or after 2017-01-01, with aggregate oil fuel capacity below 600 m3 and an individual oil fuel tank capacity above 30 m3, the oil fuel tank outer shell separation must be at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipCategory ?category .
    FILTER (?category IN (
        nltl:polarShipCategoryA,
        nltl:polarShipCategoryB
    ))

    $this nltl:constructionDate ?constructionDate .
    FILTER (
        datatype(?constructionDate) = xsd:date &&
        ?constructionDate >= "2017-01-01"^^xsd:date
    )

    $this nltl:aggregateOilFuelCapacity ?aggregateQuantity .
    ?aggregateQuantity qudt:numericValue ?aggregateValue ;
                       qudt:unit unit:M3 .
    FILTER (
        datatype(?aggregateValue) = xsd:decimal &&
        ?aggregateValue < "600"^^xsd:decimal
    )

    $this nltl:individualOilFuelTankCapacity ?individualQuantity .
    ?individualQuantity qudt:numericValue ?individualValue ;
                        qudt:unit unit:M3 .
    FILTER (
        datatype(?individualValue) = xsd:decimal &&
        ?individualValue > "30"^^xsd:decimal
    )

    FILTER NOT EXISTS {
        $this nltl:oilFuelTankOuterShellSeparation ?separationQuantity .
        ?separationQuantity qudt:numericValue ?separationValue ;
                            qudt:unit unit:M .
        FILTER (
            datatype(?separationValue) = xsd:decimal &&
            ?separationValue >= "0.76"^^xsd:decimal
        )
    }
}
"""
    ] .
