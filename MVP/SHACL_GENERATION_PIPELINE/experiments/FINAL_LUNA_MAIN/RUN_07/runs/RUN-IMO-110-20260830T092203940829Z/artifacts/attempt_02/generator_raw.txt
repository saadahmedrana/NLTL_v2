<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B oil tanker of less than 5,000 tonnes deadweight constructed on or after 1 January 2017, both required cargo-tank protection statuses must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this rdf:type nltl:oilTanker ;
                      nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:deadweight ?deadweightNode .

                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (STR(?shipType) = "oil_tanker")
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                ?deadweightNode qudt:numericValue ?deadweightValue ;
                                qudt:unit unit:TON_Metric .
                FILTER (datatype(?deadweightValue) = xsd:decimal)
                FILTER (?deadweightValue < 5000)

                FILTER NOT EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true
                }
                FILTER NOT EXISTS {
                    $this nltl:wingTankProtectionStatus true
                }
            }
        """
    ] .
<END_SHACL>
