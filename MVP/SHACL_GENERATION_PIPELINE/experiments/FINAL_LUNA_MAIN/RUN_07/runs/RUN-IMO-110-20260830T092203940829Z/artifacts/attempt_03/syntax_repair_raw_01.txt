<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipType ;
                sh:datatype xsd:string ;
                sh:hasValue "oil_tanker" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:deadweight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:maxExclusive "5000"^^xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:TON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Applicable ships must have compliant entire cargo-tank-length double-bottom protection."
            ] ;
            sh:property [
                sh:path nltl:wingTankProtectionStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Applicable ships must have compliant wing-tank protection."
            ]
        ]
        [
            sh:sparql [
                sh:message "The conditional protection requirement is applicable unless the ship is demonstrably outside at least one applicability condition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        FILTER (
                            EXISTS {
                                $this nltl:shipCategory ?category .
                                FILTER (
                                    ?category NOT IN (
                                        nltl:polarShipCategoryA,
                                        nltl:polarShipCategoryB
                                    )
                                )
                            }
                            ||
                            EXISTS {
                                $this nltl:shipType ?type .
                                FILTER (?type != "oil_tanker")
                            }
                            ||
                            EXISTS {
                                $this nltl:deadweight ?deadweight .
                                ?deadweight qudt:numericValue ?value .
                                ?deadweight qudt:unit unit:TON_Metric .
                                FILTER (?value >= "5000"^^xsd:decimal)
                            }
                            ||
                            EXISTS {
                                $this nltl:constructionDate ?date .
                                FILTER (?date < "2017-01-01"^^xsd:date)
                            }
                        )
                    }
                """
            ]
        ]
    ) .
</END_SHACL>
