<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1 ;
        sh:message "Construction date must be a single xsd:date when supplied."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:maxCount 1 ;
        sh:message "Ship category must be one of the supplied polar ship category values when supplied."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, every oil residue or oily bilge water holding tank must have a valid individual tank capacity quantity in cubic metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipCategory ?category ;
        nltl:constructionDate ?date ;
        nltl:hasComponent ?tank .
  FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
  FILTER (?date >= "2017-01-01"^^xsd:date)
  ?tank a ?tankType .
  FILTER (?tankType IN (nltl:oilResidueTank, nltl:oilyBilgeWaterHoldingTank))
  FILTER NOT EXISTS {
    ?tank nltl:individualTankCapacity ?capacity .
    ?capacity a qudt:QuantityValue ;
              qudt:numericValue ?capacityValue ;
              qudt:unit unit:M3 .
    FILTER (datatype(?capacityValue) = xsd:decimal)
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, each oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have outer-shell separation of at least 0.76 m, with a valid metre quantity." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipCategory ?category ;
        nltl:constructionDate ?date ;
        nltl:hasComponent ?tank .
  FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
  FILTER (?date >= "2017-01-01"^^xsd:date)
  ?tank a ?tankType .
  FILTER (?tankType IN (nltl:oilResidueTank, nltl:oilyBilgeWaterHoldingTank))
  ?tank nltl:individualTankCapacity ?capacity .
  ?capacity a qudt:QuantityValue ;
            qudt:numericValue ?capacityValue ;
            qudt:unit unit:M3 .
  FILTER (datatype(?capacityValue) = xsd:decimal)
  FILTER (?capacityValue > 30)
  FILTER NOT EXISTS {
    ?tank nltl:tankOuterShellSeparation ?separation .
    ?separation a qudt:QuantityValue ;
                qudt:numericValue ?separationValue ;
                qudt:unit unit:M .
    FILTER (datatype(?separationValue) = xsd:decimal)
    FILTER (?separationValue >= 0.76)
  }
}
"""
    ] .
<END_SHACL>
