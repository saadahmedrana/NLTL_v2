<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one Category A or Category B ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date on or after 2017-01-01."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have a paired outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasComponent ?tank .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                ?tank a ?tankType ;
                      nltl:individualTankCapacity ?capacityNode .
                FILTER (
                    ?tankType IN (
                        nltl:oilResidueTank,
                        nltl:oilyBilgeWaterHoldingTank
                    )
                )

                ?capacityNode qudt:numericValue ?capacityValue ;
                              qudt:unit unit:M3 .
                FILTER (?capacityValue > 30)

                FILTER NOT EXISTS {
                    ?tank nltl:tankOuterShellSeparation ?separationNode .
                    ?separationNode qudt:numericValue ?separationValue ;
                                    qudt:unit unit:M .
                    FILTER (?separationValue >= 0.76)
                }
            }
        """
    ] .

                sh:maxCount 1

        ]
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
