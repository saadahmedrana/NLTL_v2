<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall record its operating area."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall record whether food waste is discharged to sea."
    ] ;
    sh:sparql [
        sh:message "For Arctic_waters when food waste is discharged to sea, the nearest land ice shelf or fast ice shall be at least 12 nautical miles away, the qualifying ice-concentration information shall be recorded, and the distance from areas with ice concentration above one tenth shall be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .

                FILTER (
                    datatype(?area) = xsd:string &&
                    str(?area) = "Arctic_waters" &&
                    datatype(?discharge) = xsd:boolean &&
                    ?discharge = true
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:arcticWaters ?aw .
                        FILTER (
                            datatype(?aw) = xsd:string &&
                            str(?aw) = "Arctic_waters"
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                        ?nearest a qudt:QuantityValue ;
                                 qudt:numericValue ?nearestValue ;
                                 qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal &&
                            ?nearestValue >= 12
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceConcentration ?ice .
                        ?ice a qudt:QuantityValue ;
                             qudt:numericValue ?iceValue ;
                             qudt:unit unit:UNITLESS .
                        FILTER (datatype(?iceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearby .
                        ?nearby a qudt:QuantityValue ;
                                qudt:numericValue ?nearbyValue ;
                                qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?nearbyValue) = xsd:decimal &&
                            ?nearbyValue > 1
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recordedDistance .
                        ?recordedDistance a qudt:QuantityValue ;
                                          qudt:numericValue ?recordedValue ;
                                          qudt:unit unit:MI_N .
                        FILTER (datatype(?recordedValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
</END_SHACL>
