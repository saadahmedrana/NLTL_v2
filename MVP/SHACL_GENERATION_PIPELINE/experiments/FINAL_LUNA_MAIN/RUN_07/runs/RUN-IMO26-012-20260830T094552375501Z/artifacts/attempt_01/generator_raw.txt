<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node <urn:nltl:generated-shape:IMO26-012-HeadingMeansShape> ;
        sh:message "The ship shall have at least two non-magnetic heading means, with the required power and independence relationships."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means shall be independent from a distinct non-magnetic heading means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    FILTER (?otherMeans != ?means)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall identify an independent heading means."
    ] .
<END_SHACL>
