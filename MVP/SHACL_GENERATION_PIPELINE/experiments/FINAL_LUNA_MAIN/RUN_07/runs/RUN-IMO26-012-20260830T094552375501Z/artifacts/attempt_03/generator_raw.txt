<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO26_012_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO26_012_ApplicableShipShape
        gen:IMO26_012_NonApplicableShipShape
    ) .

gen:IMO26_012_ApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:minInclusive 500
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue qudt:UNITLESS ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gen:IMO26_012_HeadingMeansShape
    ] .

gen:IMO26_012_NonApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:maxExclusive 500
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue qudt:UNITLESS ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] .

gen:IMO26_012_HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means must be independent from another distinct heading means linked to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?ship nltl:hasNonMagneticHeadingMeans $this .
                    ?ship nltl:hasNonMagneticHeadingMeans ?other .
                    $this nltl:independentFromHeadingMeans ?other .
                    FILTER (?other != $this)
                }
            }
        """
    ] .
<END_SHACL>
