<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_012_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:node gen:IMO26_012_ApplicableShape
        ]
        [
            sh:node gen:IMO26_012_NonApplicableShape
        ]
    ) .

gen:IMO26_012_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO26_012_GrossTonnageAtLeast500 ;
        sh:message "Gross tonnage must be at least 500 for this applicability branch."
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gen:IMO26_012_HeadingMeansShape ;
        sh:message "Ships of at least 500 gross tonnage shall have at least two non-magnetic heading means."
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means shall be independent from another heading means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?other .
                    $this nltl:hasNonMagneticHeadingMeans ?other .
                    FILTER (?other != ?means)
                }
            }
        """
    ] .

gen:IMO26_012_NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO26_012_GrossTonnageBelow500 ;
        sh:message "Gross tonnage must be below 500 for the non-applicable branch."
    ] .

gen:IMO26_012_GrossTonnageAtLeast500
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "500"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <https://qudt.org/vocab/unit/UNITLESS>
    ] .

gen:IMO26_012_GrossTonnageBelow500
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:maxExclusive "500"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <https://qudt.org/vocab/unit/UNITLESS>
    ] .

gen:IMO26_012_HeadingMeansShape
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Each non-magnetic heading means shall be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Each non-magnetic heading means shall be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Each non-magnetic heading means shall have an independence relation."
    ] .
<END_SHACL>
