<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_016_Chapter11_1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must record whether it operates in polar waters."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship operating in polar waters, Chapter 11-1 applicability must be true when the ship is a qualifying fishing vessel, pleasure yacht, or cargo ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters true ;
                      nltl:shipType ?shipType .

                {
                    FILTER (STR(?shipType) = "fishing_vessel")
                    $this nltl:lengthOverall ?lengthQuantity .
                    ?lengthQuantity qudt:numericValue ?lengthValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (DATATYPE(?lengthValue) = xsd:decimal)
                    FILTER (?lengthValue >= 24)
                }
                UNION
                {
                    FILTER (STR(?shipType) = "pleasure_yacht")
                    $this nltl:grossTonnage ?tonnageQuantity ;
                          nltl:engagedInTrade false .
                    ?tonnageQuantity qudt:numericValue ?tonnageValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (DATATYPE(?tonnageValue) = xsd:decimal)
                    FILTER (?tonnageValue >= 300)
                }
                UNION
                {
                    FILTER (STR(?shipType) = "cargo_ship")
                    $this nltl:grossTonnage ?tonnageQuantity .
                    ?tonnageQuantity qudt:numericValue ?tonnageValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (DATATYPE(?tonnageValue) = xsd:decimal)
                    FILTER (?tonnageValue >= 300 && ?tonnageValue < 500)
                }

                FILTER NOT EXISTS {
                    $this nltl:polarCodeChapter11Dash1Applicable true
                }
            }
        """
    ] .
<END_SHACL>
