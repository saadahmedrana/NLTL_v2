<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-002-ship>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "A construction contract date is required to determine whether the 2017 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        sh:message "Ships contracted for construction on or after 1 January 2019 and before 5 July 2021 shall identify the 2017 Ice Class Regulations as the applicable edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    YEAR(?date) > 2019
                    || (
                        YEAR(?date) = 2019
                        && (
                            MONTH(?date) > 1
                            || (MONTH(?date) = 1 && DAY(?date) >= 1)
                        )
                    )
                )
                FILTER (
                    YEAR(?date) < 2021
                    || (
                        YEAR(?date) = 2021
                        && (
                            MONTH(?date) < 7
                            || (MONTH(?date) = 7 && DAY(?date) < 5)
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2017 .
                }
            }
        """
    ] .
</urn:nltl:generated-shape:TRF-002-ship>
<END_SHACL>
