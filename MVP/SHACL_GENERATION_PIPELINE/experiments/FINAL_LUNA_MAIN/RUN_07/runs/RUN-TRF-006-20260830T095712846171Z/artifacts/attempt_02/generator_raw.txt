<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF_006_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction-stage date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:ownerRequested2008EngineOutputRequirements ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The owner's request concerning the 2008 engine-output requirements must be recorded."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF_006_ApplicableConstructionCohort
        gen:TRF_006_NonApplicableConstructionCohort
    ) ;
    sh:or (
        gen:TRF_006_OwnerRequested2008
        gen:TRF_006_OwnerDidNotRequest2008
    ) .

gen:TRF_006_ApplicableConstructionCohort
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:minInclusive "1986-11-01"^^xsd:date ;
        sh:maxExclusive "2003-09-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:hasValue nltl:iceClassRuleEdition1985 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ships in the applicable construction cohort must use the 1985 ice-class rule edition."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ships in the applicable construction cohort must have an ice class."
    ] .

gen:TRF_006_NonApplicableConstructionCohort
    a sh:NodeShape ;
    sh:or (
        [
            sh:path nltl:constructionStageDate ;
            sh:maxExclusive "1986-11-01"^^xsd:date
        ]
        [
            sh:path nltl:constructionStageDate ;
            sh:minInclusive "2003-09-01"^^xsd:date
        ]
    ) .

gen:TRF_006_OwnerRequested2008
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ownerRequested2008EngineOutputRequirements ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputRegulationEdition ;
        sh:hasValue nltl:iceClassRegulationEdition2008 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "When the owner requests the 2008 requirements, the engine-output edition must be the 2008 Ice Class Regulations."
    ] .

gen:TRF_006_OwnerDidNotRequest2008
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ownerRequested2008EngineOutputRequirements ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
