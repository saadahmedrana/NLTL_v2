<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice class for applicability assessment."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship of ice class IA Super or IA, a valid construction stage date is required to determine whether the conditional requirement applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate .
                    FILTER (datatype(?constructionDate) = xsd:date)
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA Super or IA ship must have a construction stage date before 1 September 2003, delivery date, assessment date not later than the applicable deadline, and exactly one permitted 2017 compliance method." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa
                ))

                $this nltl:constructionStageDate ?constructionDate .
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (?constructionDate < "2003-09-01"^^xsd:date)

                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                    FILTER (datatype(?deliveryDate) = xsd:date)
                }

                FILTER NOT EXISTS {
                    $this nltl:assessmentDate ?assessmentDate .
                    FILTER (datatype(?assessmentDate) = xsd:date)
                }

                FILTER NOT EXISTS {
                    $this nltl:engineOutputComplianceMethod ?permittedMethod .
                    FILTER (?permittedMethod IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    ))
                }

                FILTER EXISTS {
                    $this nltl:engineOutputComplianceMethod ?disallowedMethod .
                    FILTER (?disallowedMethod NOT IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    ))
                }

                FILTER EXISTS {
                    $this nltl:deliveryDate ?deliveryDate2 .
                    $this nltl:assessmentDate ?assessmentDate2 .
                    FILTER (datatype(?deliveryDate2) = xsd:date)
                    FILTER (datatype(?assessmentDate2) = xsd:date)
                    FILTER (
                        YEAR(?assessmentDate2) > YEAR(?deliveryDate2) + 20
                        ||
                        (
                            YEAR(?assessmentDate2) = YEAR(?deliveryDate2) + 20
                            &&
                            (
                                MONTH(?assessmentDate2) > 1
                                ||
                                (
                                    MONTH(?assessmentDate2) = 1
                                    && DAY(?assessmentDate2) > 1
                                )
                            )
                        )
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
