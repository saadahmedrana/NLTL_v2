<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                ) ;
                sh:message "An applicable ship must have exactly one ice class: IA or IA Super."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must have exactly one construction stage date."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must have exactly one delivery date."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must have exactly one assessment date."
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "An applicable ship must have exactly one permitted 2017 Ice Class Regulations compliance method."
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The construction stage must precede 1 September 2003, and assessment must not be later than 1 January of the year twenty years after delivery." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate ;
          nltl:assessmentDate ?assessmentDate .
    FILTER (
        YEAR(?constructionDate) > 2003
        || (
            YEAR(?constructionDate) = 2003
            && (
                MONTH(?constructionDate) > 9
                || (
                    MONTH(?constructionDate) = 9
                    && DAY(?constructionDate) >= 1
                )
            )
        )
        || YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
        || (
            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
            && (
                MONTH(?assessmentDate) > 1
                || (
                    MONTH(?assessmentDate) = 1
                    && DAY(?assessmentDate) > 1
                )
            )
        )
    )
}
"""
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:message "A non-applicable ship must have exactly one supplied ice class outside IA and IA Super."
            ]
        ]
    ) .
<END_SHACL>
