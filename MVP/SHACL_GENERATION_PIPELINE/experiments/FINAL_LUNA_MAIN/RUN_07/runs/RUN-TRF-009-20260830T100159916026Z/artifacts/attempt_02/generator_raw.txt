<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A ship must have one permitted Finnish-Swedish ice class."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have one construction stage date for applicability assessment."
    ] ;
    sh:sparql [
        sh:message "For an IA Super or IA ship constructed before 1 September 2003, delivery date, assessment date, and a permitted 2017 compliance method are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?ice ;
          nltl:constructionStageDate ?constructionDate .
    FILTER (
        ?ice = nltl:iceClassIaSuper ||
        ?ice = nltl:iceClassIa
    )
    FILTER (
        YEAR(?constructionDate) < 2003 ||
        (
            YEAR(?constructionDate) = 2003 &&
            (
                MONTH(?constructionDate) < 9 ||
                (MONTH(?constructionDate) = 9 && DAY(?constructionDate) < 1)
            )
        )
    )
    FILTER NOT EXISTS {
        $this nltl:deliveryDate ?deliveryDate .
    }
    FILTER NOT EXISTS {
        $this nltl:assessmentDate ?assessmentDate .
    }
    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod ?method .
        FILTER (
            ?method = nltl:traficom2017Section3Point2Point2Method ||
            ?method = nltl:traficom2017Section3Point2Point4Method
        )
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "The assessment date must be no later than 1 January of the year twenty years after delivery." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?ice ;
          nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate ;
          nltl:assessmentDate ?assessmentDate .
    FILTER (
        ?ice = nltl:iceClassIaSuper ||
        ?ice = nltl:iceClassIa
    )
    FILTER (
        YEAR(?constructionDate) < 2003 ||
        (
            YEAR(?constructionDate) = 2003 &&
            (
                MONTH(?constructionDate) < 9 ||
                (MONTH(?constructionDate) = 9 && DAY(?constructionDate) < 1)
            )
        )
    )
    FILTER (
        YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
        (
            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
            (
                MONTH(?assessmentDate) > 1 ||
                (
                    MONTH(?assessmentDate) = 1 &&
                    DAY(?assessmentDate) > 1
                )
            )
        )
    )
}
"""
    ] .
<END_SHACL>
