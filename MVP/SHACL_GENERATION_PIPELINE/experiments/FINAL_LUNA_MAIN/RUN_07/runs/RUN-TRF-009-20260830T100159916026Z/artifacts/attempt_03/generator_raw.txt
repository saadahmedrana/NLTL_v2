<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice class so applicability can be determined."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A ship must provide a valid construction stage date so applicability can be determined."
    ] ;
    sh:sparql [
        sh:message "For an IA or IA Super ship constructed before 1 September 2003, delivery date, assessment date, and a permitted 2017 compliance method are required; the assessment date must be no later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIa ,
                        nltl:iceClassIaSuper
                    )
                )
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (
                    YEAR(?constructionDate) < 2003
                    || (
                        YEAR(?constructionDate) = 2003
                        && (
                            MONTH(?constructionDate) < 9
                            || (
                                MONTH(?constructionDate) = 9
                                && DAY(?constructionDate) < 1
                            )
                        )
                    )
                )

                FILTER (
                    !EXISTS {
                        $this nltl:deliveryDate ?deliveryDate .
                        FILTER (datatype(?deliveryDate) = xsd:date)
                    }
                    ||
                    !EXISTS {
                        $this nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?assessmentDate) = xsd:date)
                    }
                    ||
                    !EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (
                            ?method IN (
                                nltl:traficom2017Section3Point2Point2Method ,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            datatype(?deliveryDate) = xsd:date
                            && datatype(?assessmentDate) = xsd:date
                        )
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                && MONTH(?assessmentDate) > 1
                            )
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                && MONTH(?assessmentDate) = 1
                                && DAY(?assessmentDate) > 1
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
