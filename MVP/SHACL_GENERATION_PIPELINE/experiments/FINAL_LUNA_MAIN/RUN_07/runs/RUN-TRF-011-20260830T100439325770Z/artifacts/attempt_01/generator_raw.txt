<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF011IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify its intended ice-operating waterline."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF011IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify its upper ice waterline."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must equal, in both directions at every longitudinal position, the maximum vertical coordinate of the intended ice-operating waterline points, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    {
        SELECT $this ?x
        WHERE {
            {
                $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
                ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                ?intendedPoint nltl:longitudinalPosition ?intendedPosition .
                ?intendedPosition <http://qudt.org/schema/qudt/numericValue> ?x .
                ?intendedPoint nltl:verticalCoordinate ?intendedVertical .
                ?intendedVertical <http://qudt.org/schema/qudt/numericValue> ?iv .
            }
            UNION
            {
                $this nltl:hasUpperIceWaterline ?upperWaterline .
                ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                ?upperPoint nltl:longitudinalPosition ?upperPosition .
                ?upperPosition <http://qudt.org/schema/qudt/numericValue> ?x .
                ?upperPoint nltl:verticalCoordinate ?upperVertical .
                ?upperVertical <http://qudt.org/schema/qudt/numericValue> ?uv .
            }
        }
        GROUP BY $this ?x
        HAVING (
            COUNT(DISTINCT ?intendedPoint) = 0 ||
            COUNT(DISTINCT ?upperPoint) = 0 ||
            (
                COUNT(DISTINCT ?intendedPoint) > 0 &&
                COUNT(DISTINCT ?upperPoint) > 0 &&
                ABS(MAX(?iv) - MAX(?uv)) >
                (
                    (ABS(MAX(?iv)) + ABS(MAX(?uv))) *
                    "0.000001"^^xsd:decimal +
                    "0.000000001"^^xsd:decimal
                )
            )
        )
    }
}
""" ;
    ] .

gen:TRF011IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:TRF011ProfilePointShape ;
        sh:minCount 1 ;
        sh:message "Each represented ice waterline must contain profile points."
    ] .

gen:TRF011ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:node gen:TRF011LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must have one longitudinal position."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:node gen:TRF011LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must have one vertical coordinate."
    ] .

gen:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
