@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-016-MassQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric>
    ] .

<urn:nltl:generated-shape:TRF-016-LengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

<urn:nltl:generated-shape:TRF-016-BallastTankShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasBallastTank ;
    sh:class nltl:ballastTank ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "A ballast tank situated above the LIWL and needed to reach it must have freezing prevention." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#situatedAboveLowerIceWaterline> true ;
                      <https://w3id.org/nltl/vocab#usedToReachLowerIceWaterline> true .
                FILTER NOT EXISTS {
                    $this <https://w3id.org/nltl/vocab#freezingPreventionPresent> true .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-016-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-MassQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:waterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-MassQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-016-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:propellerHighestPointSubmerged ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "When navigating in ice, draught amidships must be at least the LIWL draught amidships." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#draughtAmidships> ?draughtNode ;
                              <https://w3id.org/nltl/vocab#liwlDraughtAmidships> ?liwlNode .
                        ?draughtNode <http://qudt.org/schema/qudt/numericValue> ?draught .
                        ?liwlNode <http://qudt.org/schema/qudt/numericValue> ?liwl .
                        FILTER (?draught < ?liwl)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Displacement at the UIWL must equal the ship waterline displacement." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#displacementAtUpperIceWaterline> ?uiwlNode ;
                              <https://w3id.org/nltl/vocab#waterlineDisplacement> ?waterlineNode .
                        ?uiwlNode <http://qudt.org/schema/qudt/numericValue> ?uiwl .
                        ?waterlineNode <http://qudt.org/schema/qudt/numericValue> ?waterline .
                        FILTER (?uiwl != ?waterline)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Forward draught must meet the level-ice-thickness formula using UIWL displacement." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl/vocab#forwardDraught> ?forwardNode ;
                              <https://w3id.org/nltl/vocab#displacementAtUpperIceWaterline> ?displacementNode ;
                              <https://w3id.org/nltl/vocab#levelIceThickness> ?thicknessNode .
                        ?forwardNode <http://qudt.org/schema/qudt/numericValue> ?forward .
                        ?displacementNode <http://qudt.org/schema/qudt/numericValue> ?displacement .
                        ?thicknessNode <http://qudt.org/schema/qudt/numericValue> ?thickness .
                        BIND ((2 + (0.00025 * ?displacement)) * ?thickness AS ?uncapped)
                        BIND (4 * ?thickness AS ?cap)
                        BIND (IF(?uncapped < ?cap, ?uncapped, ?cap) AS ?minimum)
                        FILTER (?forward < ?minimum)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
