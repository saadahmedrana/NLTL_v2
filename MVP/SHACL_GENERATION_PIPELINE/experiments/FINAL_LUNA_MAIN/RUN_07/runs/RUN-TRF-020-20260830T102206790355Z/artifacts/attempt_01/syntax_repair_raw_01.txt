<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-020-ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "An ice-class value is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, all declared calculation inputs, waterline relationships, and externally calculated minimum-power results must be present with the required quantity representation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date ;
                      nltl:iceClass ?class .
                FILTER (?date >= "2003-09-01"^^xsd:date)
                FILTER (?class IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER NOT EXISTS {
                    $this nltl:engineOutputCoefficientKe ?keNode ;
                          nltl:brashIceChannelResistanceAtLowerIceWaterline ?rchLowerNode ;
                          nltl:brashIceChannelResistanceAtUpperIceWaterline ?rchUpperNode ;
                          nltl:propellerDiameter ?diameterNode ;
                          nltl:propellerCount ?count ;
                          nltl:propellerPitchControlType ?pitch ;
                          nltl:propulsionSystemType ?propulsion ;
                          nltl:maximumContinuousRatingPower ?mcrNode ;
                          nltl:minimumRequiredPowerAtLowerIceWaterline ?minLowerNode ;
                          nltl:minimumRequiredPowerAtUpperIceWaterline ?minUpperNode ;
                          nltl:hasLowerIceWaterline ?lowerWaterline ;
                          nltl:hasUpperIceWaterline ?upperWaterline .
                    $lowerWaterline a nltl:iceWaterline .
                    $upperWaterline a nltl:iceWaterline ;
                                    nltl:upperIceWaterlineBreadth ?breadthNode ;
                                    nltl:upperIceWaterlineLength ?lengthNode .
                    ?keNode qudt:numericValue ?keValue ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?rchLowerNode qudt:numericValue ?rchLowerValue .
                    ?rchUpperNode qudt:numericValue ?rchUpperValue .
                    ?diameterNode qudt:numericValue ?diameterValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?mcrNode qudt:numericValue ?mcrValue ;
                             qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    ?minLowerNode qudt:numericValue ?minLowerValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    ?minUpperNode qudt:numericValue ?minUpperValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    ?breadthNode qudt:numericValue ?breadthValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?lengthNode qudt:numericValue ?lengthValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?keValue) = xsd:decimal)
                    FILTER (datatype(?rchLowerValue) = xsd:decimal)
                    FILTER (datatype(?rchUpperValue) = xsd:decimal)
                    FILTER (datatype(?diameterValue) = xsd:decimal)
                    FILTER (datatype(?mcrValue) = xsd:decimal)
                    FILTER (datatype(?minLowerValue) = xsd:decimal)
                    FILTER (datatype(?minUpperValue) = xsd:decimal)
                    FILTER (datatype(?breadthValue) = xsd:decimal)
                    FILTER (datatype(?lengthValue) = xsd:decimal)
                    FILTER (datatype(?count) = xsd:integer)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, maximum continuous rating power must be at least both externally calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date ;
                      nltl:iceClass ?class ;
                      nltl:maximumContinuousRatingPower ?mcrNode ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerNode ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline ?upperNode .
                FILTER (?date >= "2003-09-01"^^xsd:date)
                FILTER (?class IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                ?mcrNode qudt:numericValue ?mcr ;
                         qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?lowerNode qudt:numericValue ?lower ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?upperNode qudt:numericValue ?upper ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                FILTER (datatype(?mcr) = xsd:decimal)
                FILTER (datatype(?lower) = xsd:decimal)
                FILTER (datatype(?upper) = xsd:decimal)
                FILTER (?mcr < ?lower || ?mcr < ?upper)
            }
        """
    ] .
</END_SHACL>
