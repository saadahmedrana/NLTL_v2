<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF023ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship with ice class IB or IC under the 1985 ice class rules, maximum continuous rating power and required power under the 1985 rules must each be present exactly once as decimal kW quantities, and maximum continuous rating power must be at least the required power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )
                $this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRuleEdition1985 .

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?mcrNode .
                        ?mcrNode a qudt:QuantityValue ;
                            qudt:numericValue ?mcrValue ;
                            qudt:unit qudt:KiloW .
                        FILTER (DATATYPE(?mcrValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPowerUnder1985Rules ?requiredNode .
                        ?requiredNode a qudt:QuantityValue ;
                            qudt:numericValue ?requiredValue ;
                            qudt:unit qudt:KiloW .
                        FILTER (DATATYPE(?requiredValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?invalidMcrNode .
                        FILTER NOT EXISTS {
                            ?invalidMcrNode a qudt:QuantityValue ;
                                qudt:numericValue ?invalidMcrValue ;
                                qudt:unit qudt:KiloW .
                            FILTER (DATATYPE(?invalidMcrValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:requiredPowerUnder1985Rules ?invalidRequiredNode .
                        FILTER NOT EXISTS {
                            ?invalidRequiredNode a qudt:QuantityValue ;
                                qudt:numericValue ?invalidRequiredValue ;
                                qudt:unit qudt:KiloW .
                            FILTER (DATATYPE(?invalidRequiredValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?mcrNode1, ?mcrNode2 .
                        FILTER (?mcrNode1 != ?mcrNode2)
                    }
                    ||
                    EXISTS {
                        $this nltl:requiredPowerUnder1985Rules
                            ?requiredNode1, ?requiredNode2 .
                        FILTER (?requiredNode1 != ?requiredNode2)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?mcrNode .
                        ?mcrNode a qudt:QuantityValue ;
                            qudt:numericValue ?mcrValue ;
                            qudt:unit qudt:KiloW .
                        FILTER (DATATYPE(?mcrValue) = xsd:decimal)

                        $this nltl:requiredPowerUnder1985Rules ?requiredNode .
                        ?requiredNode a qudt:QuantityValue ;
                            qudt:numericValue ?requiredValue ;
                            qudt:unit qudt:KiloW .
                        FILTER (DATATYPE(?requiredValue) = xsd:decimal)

                        FILTER (?mcrValue < ?requiredValue)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
