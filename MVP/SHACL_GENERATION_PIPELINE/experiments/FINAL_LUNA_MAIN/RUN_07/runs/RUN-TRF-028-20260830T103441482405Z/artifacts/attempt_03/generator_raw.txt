<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-028> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEvidence ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] ;
    sh:sparql [
        sh:message "When alternative calculation or model-test evidence is used, the ship must have approved status, an approved revocation status, and at least one valid Ke or RCH result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEvidence ?evidence .
                ?evidence a ?evidenceType .
                FILTER (
                    ?evidenceType = nltl:alternativeCalculationEvidence ||
                    ?evidenceType = nltl:modelTestEvidence
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:approvalStatus ?approval .
                        FILTER (
                            datatype(?approval) = xsd:boolean &&
                            ?approval = "true"^^xsd:boolean
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:approvalRevocationStatus nltl:evidenceStateApproved .
                    }
                    ||
                    NOT EXISTS {
                        {
                            $this nltl:engineOutputCoefficientKe ?result .
                        }
                        UNION
                        {
                            $this nltl:brashIceChannelResistanceRch ?result .
                        }
                        ?result qudt:numericValue ?numericValue ;
                                qudt:unit ?unit .
                        FILTER (
                            datatype(?numericValue) = xsd:decimal &&
                            (
                                ?unit = <http://qudt.org/vocab/unit/UNITLESS> ||
                                ?unit = <http://qudt.org/vocab/unit/N>
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
