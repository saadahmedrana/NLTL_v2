<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF034_EffectiveMemberCrossSectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:effectiveMemberCrossSection ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/CentiM3> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The effective member cross section must have one section modulus expressed in cubic centimetres."
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M2> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The effective member cross section must have one shear area expressed in square metres."
    ] ;
    sh:property [
        sh:path nltl:hasClassificationSocietySectionPropertyCalculationEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        sh:message "The effective member cross section must be linked from the applicable hull structure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?owner nltl:hasEffectiveMemberCrossSection $this .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A member not normal to the plating must have classification society section property calculation evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                ?owner nltl:hasEffectiveMemberCrossSection $this .
                ?owner nltl:memberNormalToPlating false .
                FILTER NOT EXISTS {
                    $this nltl:hasClassificationSocietySectionPropertyCalculationEvidence ?evidence .
                }
            }
        """
    ] .

nltl:TRF034_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:memberNormalToPlating ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member normal-to-plating status must be provided for the applicable ice stringer."
    ] ;
    sh:property [
        sh:path nltl:hasEffectiveMemberCrossSection ;
        sh:class nltl:effectiveMemberCrossSection ;
        sh:minCount 1 ;
        sh:message "The applicable ice stringer must identify its effective member cross section."
    ] .
<END_SHACL>
