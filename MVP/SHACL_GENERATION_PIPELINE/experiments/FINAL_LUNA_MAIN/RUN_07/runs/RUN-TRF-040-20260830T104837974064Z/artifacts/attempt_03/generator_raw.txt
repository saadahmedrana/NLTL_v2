<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF040Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted ice-class value."
    ] ;

    sh:property [
        sh:path nltl:designOperationSpeedInIce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:thickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Sidescuttles shall not be situated in the ice belt."
    ] ;

    sh:property [
        sh:path nltl:bowRegionEquivalent ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:forefootShellStrengtheningStatus ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:midbodyEquivalent ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:upperBowIceBeltStrengtheningStatus ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:bulwarkStrengthStatus ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:freeingPortStrengthStatus ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
        sh:datatype xsd:boolean
    ] ;

    sh:sparql [
        sh:message "For ice class IA Super, forefoot strengthening shall cover the specified five-main-frame-spacing extent and be equivalent to the bow-region strengthening." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER (
                    NOT EXISTS {
                        $this nltl:spacing ?spacingNode .
                        ?spacingNode qudt:numericValue ?spacingValue ;
                                      qudt:unit unit:M .
                        FILTER (datatype(?spacingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:forefootShellStrengtheningStatus true .
                    }
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "For ice class IA Super or IA, open-water service speed shall establish whether the upper-bow strengthening condition applies; at speeds of at least 18 knots both upper-bow and midbody-equivalent strengthening shall be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:openWaterServiceSpeed ?speedNode .
                        ?speedNode qudt:numericValue ?speedValue ;
                                   qudt:unit unit:KN .
                        FILTER (datatype(?speedValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:openWaterServiceSpeed ?applicableSpeedNode .
                        ?applicableSpeedNode qudt:numericValue ?applicableSpeedValue ;
                                              qudt:unit unit:KN .
                        FILTER (
                            datatype(?applicableSpeedValue) = xsd:decimal &&
                            ?applicableSpeedValue >= 18
                        )
                        FILTER NOT EXISTS {
                            $this nltl:upperBowIceBeltStrengtheningStatus true .
                        }
                        FILTER NOT EXISTS {
                            $this nltl:midbodyEquivalent true .
                        }
                    }
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "When the weather deck is below the upper limit of the ice belt, both bulwark and freeing-port strength shall meet the ice-belt shell requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                FILTER NOT EXISTS {
                    $this nltl:bulwarkStrengthStatus true .
                }
                FILTER NOT EXISTS {
                    $this nltl:freeingPortStrengthStatus true .
                }
            }
        """
    ] .
<END_SHACL>
