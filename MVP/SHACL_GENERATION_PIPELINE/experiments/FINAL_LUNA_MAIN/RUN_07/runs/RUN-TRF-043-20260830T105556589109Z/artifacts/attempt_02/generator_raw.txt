<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF043MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:minCount 1
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:TRF043MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The termination case must state the extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The termination case must identify a deck, tank boundary plating, or tank top."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary is permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
                ?quantity qudt:numericValue ?extension .
                FILTER(?extension <= 250)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
                }
            }
        """
    ] .

gen:TRF043FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node gen:TRF043QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node gen:TRF043QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing vertical extents must satisfy the applicable Table 4-6 ice-class and hull-region values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                       nltl:hullRegion ?region ;
                       nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQuantity ;
                       nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowQuantity ;
                       nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .
                ?aboveQuantity qudt:numericValue ?above .
                ?belowQuantity qudt:numericValue ?below .
                FILTER(
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?above != 1.2
                    ) ||
                    (
                        (?iceClass = nltl:iceClassIa ||
                         ?iceClass = nltl:iceClassIb ||
                         ?iceClass = nltl:iceClassIc) &&
                        ?above != 1.0
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:bowRegion &&
                        ?reachesTankTop != true
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:bowRegion &&
                        ?reachesTankTop = true &&
                        ?below != 2.0
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:midbodyRegion &&
                        ?below != 2.0
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:sternRegion &&
                        ?below != 1.6
                    ) ||
                    (
                        (?iceClass = nltl:iceClassIa ||
                         ?iceClass = nltl:iceClassIb ||
                         ?iceClass = nltl:iceClassIc) &&
                        ?region = nltl:bowRegion &&
                        ?below != 1.6
                    ) ||
                    (
                        (?iceClass = nltl:iceClassIa ||
                         ?iceClass = nltl:iceClassIb ||
                         ?iceClass = nltl:iceClassIc) &&
                        ?region = nltl:midbodyRegion &&
                        ?below != 1.3
                    ) ||
                    (
                        (?iceClass = nltl:iceClassIa ||
                         ?iceClass = nltl:iceClassIb ||
                         ?iceClass = nltl:iceClassIc) &&
                        ?region = nltl:sternRegion &&
                        ?below != 1.0
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an upper bow ice belt is required, the framing must reach its top." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperBowIceBeltRequired true .
                FILTER NOT EXISTS {
                    $this nltl:reachesUpperBowIceBeltTop true .
                }
            }
        """
    ] .
<END_SHACL>
