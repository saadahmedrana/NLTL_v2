<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-043-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:minCount 1 ;
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the extension is no more than 250 mm, termination at an adjacent deck, tank boundary plating, or tank top must be permitted and the applicable boundary must be identified." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .
                ?extension q:numericValue ?extensionValue ;
                           q:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER ( ?extensionValue <= 250.0 )

                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
                }

                FILTER NOT EXISTS {
                    $this nltl:terminationAdjacentBoundary ?boundary .
                    ?boundary a ?boundaryClass .
                    FILTER (
                        ?boundaryClass = nltl:deckStructure ||
                        ?boundaryClass = nltl:tankBoundaryPlating ||
                        ?boundaryClass = nltl:tankTop
                    )
                }
            }
        """ ;
    ] .

<urn:nltl:generated-shape:TRF-043-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing vertical extents must meet the applicable Table 4-6 lower bounds, including the IA Super bow condition selected by reachesTankTopOrBelowFloorTop." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:reachesTankTopOrBelowFloorTop ?reaches ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .

                ?above q:numericValue ?aboveValue ;
                       q:unit <http://qudt.org/vocab/unit/M> .
                ?below q:numericValue ?belowValue ;
                       q:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?aboveValue >= 1.2 &&
                        (
                            (?region = nltl:bowRegion &&
                             (
                                (?reaches = true && ?belowValue >= 2.0) ||
                                (?reaches = false && ?belowValue >= 1.6)
                             )) ||
                            (?region = nltl:midbodyRegion && ?belowValue >= 2.0) ||
                            (?region = nltl:sternRegion && ?belowValue >= 1.6)
                        )
                    ) ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        ?aboveValue >= 1.0 &&
                        (
                            (?region = nltl:bowRegion && ?belowValue >= 1.6) ||
                            (?region = nltl:midbodyRegion && ?belowValue >= 1.3) ||
                            (?region = nltl:sternRegion && ?belowValue >= 1.0)
                        )
                    )
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where an upper bow ice belt is required, framing ice strengthening must reach at least the top of that ice belt." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperBowIceBeltRequired true ;
                      nltl:reachesUpperBowIceBeltTop ?top ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above .

                ?top q:numericValue ?topValue ;
                     q:unit <http://qudt.org/vocab/unit/M> .
                ?above q:numericValue ?aboveValue ;
                       q:unit <http://qudt.org/vocab/unit/M> .

                FILTER ( ?aboveValue < ?topValue )
            }
        """ ;
    ] .
<END_SHACL>
