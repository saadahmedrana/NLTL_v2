<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:class nltl:frameStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A frame must identify its strengthened part."
    ] ;
    sh:property [
        sh:path (
            nltl:hasStrengthenedPart
            nltl:hasLowerEnd
        ) ;
        sh:minCount 1 ;
        sh:message "The strengthened part must identify its lower end."
    ] ;
    sh:sparql [
        sh:message "The lower end of every strengthened part must be attached to an allowed deck, tank plating, tank top, or ice stringer, or an applicable intermediate ice frame must satisfy the permitted horizontal-member alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStrengthenedPart ?strengthenedPart .
                {
                    FILTER NOT EXISTS {
                        ?strengthenedPart nltl:hasLowerEnd ?lowerEnd .
                    }
                }
                UNION
                {
                    ?strengthenedPart nltl:hasLowerEnd ?lowerEnd .
                    FILTER NOT EXISTS {
                        {
                            ?lowerEnd nltl:hasAttachedSupportingStructure ?attachedStructure .
                            ?attachedStructure rdf:type ?attachedStructureClass .
                            FILTER (
                                ?attachedStructureClass IN (
                                    nltl:deckStructure,
                                    nltl:tankBoundaryPlating,
                                    nltl:tankTop,
                                    nltl:iceStringer
                                )
                            )
                        }
                        UNION
                        {
                            $this rdf:type nltl:intermediateIceFrame .
                            ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?terminationStructure .
                            ?terminationStructure rdf:type ?terminationClass .
                            FILTER (
                                ?terminationClass IN (
                                    nltl:deckStructure,
                                    nltl:tankBoundaryPlating,
                                    nltl:tankTop,
                                    nltl:iceStringer
                                )
                            )
                            $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
                            ?lowerEnd nltl:hasHorizontalConnectionMember ?horizontalMember .
                            ?horizontalMember nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                            $this nltl:sameScantlingsAsMainFrame true .
                            $this nltl:mainFrameBelowIceBeltStrengthened true .
                        }
                    }
                }
            }
        """
    ] .
</END_SHACL>
