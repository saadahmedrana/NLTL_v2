<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF049FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:webPlateConnectionSideCount ;
                        sh:datatype xsd:integer ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ]
                ]
            ) ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:node [
                    sh:property [
                        sh:path nltl:effectiveAttachmentConfirmed ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:hasAttachedSupportingStructure ;
                        sh:minCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Each frame attachment must identify a supporting structure and have the reciprocal supporting-structure attachment relationship." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasFrameAttachment ?attachment .
                        FILTER NOT EXISTS {
                            ?attachment nltl:hasAttachedSupportingStructure ?structure .
                            ?structure nltl:hasSupportingStructureAttachment ?attachment .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Every supporting web frame declared for a frame must be paired with a reciprocal frame attachment." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasSupportingWebFrame ?structure .
                        FILTER NOT EXISTS {
                            $this nltl:hasFrameAttachment ?attachment .
                            ?attachment nltl:hasAttachedSupportingStructure ?structure .
                            ?structure nltl:hasSupportingStructureAttachment ?attachment .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Every supporting bulkhead declared for a frame must be paired with a reciprocal frame attachment." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasSupportingBulkhead ?structure .
                        FILTER NOT EXISTS {
                            $this nltl:hasFrameAttachment ?attachment .
                            ?attachment nltl:hasAttachedSupportingStructure ?structure .
                            ?structure nltl:hasSupportingStructureAttachment ?attachment .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "A longitudinal frame must be attached by a bracket to every declared supporting web frame." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this a nltl:longitudinalFrame ;
                              nltl:hasSupportingWebFrame ?structure .
                        FILTER NOT EXISTS {
                            $this nltl:hasFrameAttachment ?attachment .
                            ?attachment nltl:hasAttachedSupportingStructure ?structure ;
                                       nltl:hasConnectionBracket ?bracket .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "A longitudinal frame must be attached by a bracket to every declared supporting bulkhead." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this a nltl:longitudinalFrame ;
                              nltl:hasSupportingBulkhead ?structure .
                        FILTER NOT EXISTS {
                            $this nltl:hasFrameAttachment ?attachment .
                            ?attachment nltl:hasAttachedSupportingStructure ?structure ;
                                       nltl:hasConnectionBracket ?bracket .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar construction on each attachment." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this a nltl:transverseFrame ;
                              nltl:terminatesAtDeckOrIceStringer true ;
                              nltl:hasFrameAttachment ?attachment .
                        FILTER NOT EXISTS {
                            ?attachment nltl:hasConnectionBracket ?bracket .
                        }
                        FILTER NOT EXISTS {
                            ?attachment nltl:similarAttachmentConstructionPresent true .
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "A frame running through a supporting structure must have exactly two web-plate connection sides for each paired supporting structure attachment." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:passesThroughSupportingStructure true ;
                              nltl:webPlateConnectionSideCount ?count ;
                              nltl:hasFrameAttachment ?attachment .
                        ?attachment nltl:hasAttachedSupportingStructure ?structure .
                        FILTER NOT EXISTS {
                            ?structure nltl:hasSupportingStructureAttachment ?attachment .
                        }
                        FILTER (?count != 2)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Bracket thickness must be at least the frame web thickness, and the bracket edge must be stiffened against buckling." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    SELECT $this
                    WHERE {
                        $this nltl:hasFrameAttachment ?attachment ;
                              nltl:frameWebThickness ?webThickness .
                        ?attachment nltl:hasConnectionBracket ?bracket .
                        FILTER NOT EXISTS {
                            ?bracket nltl:bracketEdgeStiffened true ;
                                     nltl:bucklingStiffening true ;
                                     nltl:bracketThickness ?bracketThickness .
                            ?webThickness qudt:numericValue ?webValue ;
                                          qudt:unit unit:MilliM .
                            ?bracketThickness qudt:numericValue ?bracketValue ;
                                             qudt:unit unit:MilliM .
                            FILTER (?bracketValue >= ?webValue)
                        }
                    }
                """
            ]
        ]
    ) .

gen:TRF049TransverseSelectorShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF049BracketShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasConnectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:TRF049FrameWebThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

<END_SHACL>
