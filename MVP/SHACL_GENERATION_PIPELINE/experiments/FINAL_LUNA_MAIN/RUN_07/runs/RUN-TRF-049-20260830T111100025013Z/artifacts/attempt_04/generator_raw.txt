<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF049_FrameShape a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node nltl:TRF049_FrameAttachmentShape
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:webPlateConnectionSideCount ;
                        sh:datatype xsd:integer ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ]
                ]
            )
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each in-area frame attachment must be paired with its supporting structure, reciprocal supporting-structure attachment, and effective-attachment confirmation." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:withinIceStrengthenedArea true .
  $this nltl:hasFrameAttachment ?attachment .
  ?attachment nltl:hasAttachedSupportingStructure ?structure .
  FILTER NOT EXISTS {
    ?structure nltl:hasSupportingStructureAttachment ?attachment .
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every represented supporting-structure attachment for an in-area frame must be linked from that frame and back to the same supporting structure." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
  $this nltl:withinIceStrengthenedArea true .
  ?structure nltl:hasSupportingStructureAttachment ?attachment .
  ?attachment nltl:hasAttachedSupportingStructure ?structure .
  FILTER NOT EXISTS {
    $this nltl:hasFrameAttachment ?attachment .
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bracket must be at least as thick as the frame web plate and its edge must be stiffened against buckling." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:withinIceStrengthenedArea true .
  $this nltl:hasFrameAttachment ?attachment .
  ?attachment nltl:hasConnectionBracket ?bracket .
  ?bracket nltl:bracketThickness ?bracketThickness .
  ?bracketThickness qudt:numericValue ?bracketValue .
  $this nltl:frameWebThickness ?webThickness .
  ?webThickness qudt:numericValue ?webValue .
  FILTER (?bracketValue < ?webValue)
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bracket requires a frame web thickness value." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
  $this nltl:withinIceStrengthenedArea true .
  $this nltl:hasFrameAttachment ?attachment .
  ?attachment nltl:hasConnectionBracket ?bracket .
  FILTER NOT EXISTS {
    $this nltl:frameWebThickness ?webThickness .
  }
}
"""
    ] .

nltl:TRF049_FrameAttachmentShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasSupportingStructureAttachment
        ] ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node nltl:TRF049_ConnectionBracketShape
    ] .

nltl:TRF049_SupportingStructureShape a sh:NodeShape ;
    sh:targetClass nltl:supportingStructure ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachmentRecord
    ] .

nltl:TRF049_ConnectionBracketShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF049_BracketThicknessShape
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

nltl:TRF049_BracketThicknessShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF049_LongitudinalFrameShape a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasSupportingWebFrame ;
                sh:minCount 1 ;
                sh:class nltl:supportingStructure
            ] ;
            sh:property [
                sh:path nltl:hasSupportingBulkhead ;
                sh:minCount 1 ;
                sh:class nltl:supportingStructure
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting web frame and bulkhead of an in-area longitudinal frame must be paired with a bracketed frame attachment." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
  $this nltl:withinIceStrengthenedArea true .
  {
    $this nltl:hasSupportingWebFrame ?structure .
  }
  UNION
  {
    $this nltl:hasSupportingBulkhead ?structure .
  }
  FILTER NOT EXISTS {
    $this nltl:hasFrameAttachment ?attachment .
    ?attachment nltl:hasAttachedSupportingStructure ?structure .
    ?attachment nltl:hasConnectionBracket ?bracket .
  }
}
"""
    ] .

nltl:TRF049_TransverseFrameShape a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path [
                            sh:inversePath nltl:hasFrameAttachment
                        ] ;
                        sh:minCount 1 ;
                        sh:or (
                            [
                                sh:property [
                                    sh:path nltl:hasConnectionBracket ;
                                    sh:minCount 1
                                ]
                            ]
                            [
                                sh:property [
                                    sh:path nltl:similarAttachmentConstructionPresent ;
                                    sh:datatype xsd:boolean ;
                                    sh:minCount 1 ;
                                    sh:maxCount 1 ;
                                    sh:hasValue true
                                ]
                            ]
                        )
                    ]
                ]
            )
        ]
    ) .
<END_SHACL>
