<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF050_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachmentRecord ;
        sh:node gen:TRF050_FrameAttachmentRecordShape ;
        sh:message "Each frame must have a frame attachment record." ;
    ] .

gen:TRF050_FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:class nltl:weldTypeValue ;
        sh:message "The frame must be attached to the shell by a double continuous weld." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#boolean> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#boolean> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#boolean> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "Scalloping is not allowed unless crossing a shell plate butt." ;
            ]
        ]
    ) .

</END_SHACL>
