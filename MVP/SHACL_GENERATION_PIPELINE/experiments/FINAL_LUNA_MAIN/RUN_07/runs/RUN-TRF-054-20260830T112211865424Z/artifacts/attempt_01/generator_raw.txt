gen:TRF054IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;

    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToAdjacentIceStringer ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:ForcePerLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:sparql [
        sh:message "The design line load, outside-ice-belt factor, section modulus, or required shear area does not satisfy the declared calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:stringerSpan/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hs ;
                      nltl:distanceToAdjacentIceStringer/qudt:numericValue ?ls ;
                      nltl:designLineLoad/qudt:numericValue ?reportedLoad ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?reportedFactor ;
                      nltl:sectionModulus/qudt:numericValue ?reportedZ ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedA .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?loadMNPerM)
                BIND((?loadMNPerM * 1000) AS ?expectedLoad)
                BIND((1 - (?hs / ?ls)) AS ?expectedFactor)
                BIND((0.80 * 1.8 * ?loadMNPerM * ?l * ?l /
                      (?m * ?sigma) * ?expectedFactor * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?loadMNPerM * ?l / (2 * ?sigma) *
                      ?expectedFactor * 10000) AS ?expectedA)

                FILTER(
                    ABS(?reportedLoad - ?expectedLoad) >
                        (ABS(?expectedLoad) * 0.000001 + 0.000001)
                    ||
                    ABS(?reportedFactor - ?expectedFactor) >
                        (ABS(?expectedFactor) * 0.000001 + 0.000001)
                    ||
                    ABS(?reportedZ - ?expectedZ) >
                        (ABS(?expectedZ) * 0.000001 + 0.000001)
                    ||
                    ABS(?reportedA - ?expectedA) >
                        (ABS(?expectedA) * 0.000001 + 0.000001)
                )
            }
        """
    ] .
