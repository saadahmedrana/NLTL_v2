<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF054_QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF054_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:locatedOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:supportsIceStrengthenedFrames ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltl:TRF054_IcePressureShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltl:TRF054_DesignHeightShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltl:TRF054_SpanShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltl:TRF054_MomentFactorShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltl:TRF054_YieldStrengthShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node nltl:TRF054_IceBeltDistanceShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node nltl:TRF054_StringerSpanShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltl:TRF054_DesignLineLoadShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node nltl:TRF054_OutsideFactorShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltl:TRF054_SectionModulusShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltl:TRF054_ShearAreaShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable ice stringer, the reported line load, outside-ice-belt factor, section modulus, and shear area must satisfy the declared calculations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true ;
                      nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?syNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?qNode ;
                      nltl:outsideIceBeltFactor ?fNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?syNode qudt:numericValue ?sy .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?qNode qudt:numericValue ?q .
                ?fNode qudt:numericValue ?f .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND (?p * ?h AS ?product)
                BIND (IF(?product < 0.15, 0.15, ?product) AS ?load)
                BIND (1 - (?hs / ?ls) AS ?outsideFactor)
                BIND (1000 * ?load AS ?expectedQ)
                BIND (0.80 * 1.8 * ?load * ?l * ?l / (?m * ?sy) * ?outsideFactor * 1000000 AS ?expectedZ)
                BIND (1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?load * ?l / (2 * ?sy) * ?outsideFactor * 10000 AS ?expectedA)

                FILTER (
                    ?q > (?expectedQ + (0.000001 * ?expectedQ) + 0.000000001) ||
                    ?q < (?expectedQ - (0.000001 * ?expectedQ) - 0.000000001) ||
                    ?f > (?outsideFactor + (0.000001 * ?outsideFactor) + 0.000000001) ||
                    ?f < (?outsideFactor - (0.000001 * ?outsideFactor) - 0.000000001) ||
                    ?z > (?expectedZ + (0.000001 * ?expectedZ) + 0.000000001) ||
                    ?z < (?expectedZ - (0.000001 * ?expectedZ) - 0.000000001) ||
                    ?a > (?expectedA + (0.000001 * ?expectedA) + 0.000000001) ||
                    ?a < (?expectedA - (0.000001 * ?expectedA) - 0.000000001)
                )
            }
        """
    ] .

nltl:TRF054_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:or (
        nltl:TRF054_ApplicableShape
        nltl:TRF054_NotApplicableOutsideShape
        nltl:TRF054_NotApplicableSupportShape
    ) .
