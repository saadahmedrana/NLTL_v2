<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF056ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node gen:TRF056WeatherdeckHatchCase
    ] .

gen:TRF056WeatherdeckHatchCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:or (
        gen:TRF056NonApplicableHatch
        gen:TRF056ApplicableHatch
    ) .

gen:TRF056NonApplicableHatch a sh:NodeShape ;
    sh:sparql [
        sh:message "A hatch opening longer than half the ship breadth requires the applicable hatch design constraints." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasWeatherdeckHatch $this ;
                      nltl:shipBreadth ?breadth .
                $this nltl:hatchOpeningLength ?opening .
                ?breadth qudt:numericValue ?breadthValue ;
                         qudt:unit unit:M .
                ?opening qudt:numericValue ?openingValue ;
                         qudt:unit unit:M .
                FILTER (?openingValue > (?breadthValue / 2))
            }
        """
    ] .

gen:TRF056ApplicableHatch a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipSideDeflection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchCoverDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:hatchCoverDesignEvidence
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchFittingDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:hatchFittingDesignEvidence
        ]
    ] ;
    sh:sparql [
        sh:message "For a hatch opening longer than half the ship breadth, side deflection and both hatch design evidence records are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasWeatherdeckHatch $this ;
                      nltl:shipBreadth ?breadth .
                $this nltl:hatchOpeningLength ?opening .
                ?breadth qudt:numericValue ?breadthValue ;
                         qudt:unit unit:M .
                ?opening qudt:numericValue ?openingValue ;
                         qudt:unit unit:M .
                FILTER (?openingValue <= (?breadthValue / 2))
            }
        """
    ] .
<END_SHACL>
