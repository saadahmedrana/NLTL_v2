<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF056ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node gen:TRF056WeatherdeckHatchCase
    ] ;
    sh:sparql [
        sh:message "For a weatherdeck hatch longer than half the ship breadth, ship-side deflection and both hatch design evidence relationships must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth .
                $this nltl:hasWeatherdeckHatch ?hatch .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingLength .
                FILTER (?openingLength > (?breadth / 2))
                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflection .
                        ?deflection a qudt:QuantityValue ;
                            qudt:numericValue ?deflectionValue ;
                            qudt:unit unit:M .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                        ?coverEvidence a nltl:hatchCoverDesignEvidence .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                        ?fittingEvidence a nltl:hatchFittingDesignEvidence .
                    }
                )
            }
        """
    ] .

gen:TRF056WeatherdeckHatchCase a sh:NodeShape ;
    sh:class nltl:weatherdeckHatch ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shipSideDeflection ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchCoverDesignEvidence ;
        sh:node [
            sh:class nltl:hatchCoverDesignEvidence
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchFittingDesignEvidence ;
        sh:node [
            sh:class nltl:hatchFittingDesignEvidence
        ]
    ] .
<END_SHACL>
