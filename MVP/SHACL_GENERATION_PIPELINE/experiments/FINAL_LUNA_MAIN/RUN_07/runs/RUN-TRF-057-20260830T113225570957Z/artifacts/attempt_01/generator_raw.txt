@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF057WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:MegaPA
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloN
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "webFrameIceLoad must equal 1.8 times max(icePressure multiplied by iceLoadHeight, 0.15) multiplied by iceLoadHeight and webFrameSpacing, converted from MN to kN, within a relative tolerance of 1e-6 with a small absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:iceLoadHeight ?heightNode ;
                      nltl:webFrameSpacing ?spacingNode ;
                      nltl:webFrameIceLoad ?loadNode .

                ?pressureNode qudt:numericValue ?pressure .
                ?heightNode qudt:numericValue ?height .
                ?spacingNode qudt:numericValue ?spacing .
                ?loadNode qudt:numericValue ?reported .

                BIND(?pressure * ?height AS ?pressureHeight)
                BIND(IF(?pressureHeight < 0.15, 0.15, ?pressureHeight) AS ?minimumLineLoad)
                BIND(1000 * 1.8 * ?minimumLineLoad * ?height * ?spacing AS ?expected)
                BIND(ABS(?expected) * 0.000001 + 0.000001 AS ?tolerance)

                FILTER(ABS(?reported - ?expected) > ?tolerance)
            }
        """
    ] .
