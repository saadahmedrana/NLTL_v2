<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-063-Ship>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single integer propeller count is required to determine whether the twin- or triple-screw applicability condition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a twin- or triple-screw ship, every side-propeller strengthening envelope must have a tank-top boundary and forward and aft extents of at least 1.5 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerCount ?propellerCount .
                FILTER (?propellerCount = 2 || ?propellerCount = 3)
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasSidePropellerStrengtheningEnvelope ?anyEnvelope .
                    }
                    ||
                    EXISTS {
                        $this nltl:hasSidePropellerStrengtheningEnvelope ?envelope .
                        FILTER NOT EXISTS {
                            ?envelope nltl:strengtheningEnvelopeBoundary ?boundary ;
                                      nltl:strengtheningForwardExtent ?forwardQuantity ;
                                      nltl:strengtheningAftExtent ?aftQuantity .
                            ?boundary a nltl:tankTop .
                            ?forwardQuantity qudt:numericValue ?forwardValue ;
                                             qudt:unit qudt:M .
                            ?aftQuantity qudt:numericValue ?aftValue ;
                                         qudt:unit qudt:M .
                            FILTER (
                                datatype(?forwardValue) = xsd:decimal &&
                                datatype(?aftValue) = xsd:decimal &&
                                ?forwardValue >= 1.5 &&
                                ?aftValue >= 1.5
                            )
                        }
                    }
                )
            }
        """ ;
    ] .
</END_SHACL>
