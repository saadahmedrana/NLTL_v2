<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_066_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide its ice class to determine rudder protection applicability."
    ] ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IA and IA Super, iceKnifePracticable must be provided as a boolean." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIa ||
                    ?iceClass = nltl:iceClassIaSuper
                )
                FILTER NOT EXISTS {
                    $this nltl:iceKnifePracticable ?practicable .
                    FILTER (
                        datatype(?practicable) = xsd:boolean &&
                        (?practicable = true || ?practicable = false)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an ice knife is practicable for ice class IA or IA Super, the rudder protection arrangement must include an ice knife extending below the lower ice waterline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:iceKnifePracticable ?practicable .
                FILTER (
                    (?iceClass = nltl:iceClassIa ||
                     ?iceClass = nltl:iceClassIaSuper) &&
                    datatype(?practicable) = xsd:boolean &&
                    ?practicable = true
                )
                FILTER NOT EXISTS {
                    $this nltl:hasRudderProtectionArrangement ?arrangement .
                    {
                        ?arrangement a nltl:iceKnife ;
                                     nltl:extendsBelowLowerIceWaterline true .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an ice knife is not practicable for ice class IA or IA Super, the rudder protection arrangement must provide equivalent rudder protection means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:iceKnifePracticable ?practicable .
                FILTER (
                    (?iceClass = nltl:iceClassIa ||
                     ?iceClass = nltl:iceClassIaSuper) &&
                    datatype(?practicable) = xsd:boolean &&
                    ?practicable = false
                )
                FILTER NOT EXISTS {
                    $this nltl:hasRudderProtectionArrangement ?arrangement .
                    ?arrangement a nltl:equivalentRudderProtectionMeans .
                }
            }
        """
    ] .
<END_SHACL>
