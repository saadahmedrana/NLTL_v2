<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-066> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA or IA Super, LIWL and a valid boolean practicability selector are required; practicable ice-knife or equivalent rudder protection must then be represented." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:iceClass ?iceClass .
  FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))

  FILTER (
    NOT EXISTS {
      $this nltl:lowerIceWaterline ?liwl .
    }
    ||
    EXISTS {
      $this nltl:lowerIceWaterline ?liwl .
      FILTER (datatype(?liwl) != xsd:string)
    }
    ||
    NOT EXISTS {
      $this nltl:iceKnifePracticable ?practicable .
    }
    ||
    EXISTS {
      $this nltl:iceKnifePracticable ?practicable .
      FILTER (datatype(?practicable) != xsd:boolean)
    }
    ||
    (
      EXISTS {
        $this nltl:iceKnifePracticable ?practicable .
        FILTER (datatype(?practicable) = xsd:boolean && ?practicable = true)
      }
      &&
      NOT EXISTS {
        $this nltl:hasRudderProtectionArrangement ?arrangement .
        ?arrangement rdf:type nltl:iceKnife .
        ?arrangement nltl:extendsBelowLowerIceWaterline true .
      }
    )
    ||
    (
      EXISTS {
        $this nltl:iceKnifePracticable ?practicable .
        FILTER (datatype(?practicable) = xsd:boolean && ?practicable = false)
      }
      &&
      NOT EXISTS {
        $this nltl:hasRudderProtectionArrangement ?arrangement .
        ?arrangement rdf:type nltl:equivalentRudderProtectionMeans .
      }
    )
    ||
    (
      NOT EXISTS {
        $this nltl:iceKnifePracticable ?practicable .
        FILTER (datatype(?practicable) = xsd:boolean && ?practicable = true)
      }
      &&
      NOT EXISTS {
        $this nltl:iceKnifePracticable ?practicable .
        FILTER (datatype(?practicable) = xsd:boolean && ?practicable = false)
      }
    )
  }
}
""" ;
    ] .
<END_SHACL>
