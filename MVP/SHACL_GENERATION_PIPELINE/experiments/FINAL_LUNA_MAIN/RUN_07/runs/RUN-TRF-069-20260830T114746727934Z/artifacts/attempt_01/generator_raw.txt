<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall declare one permitted ice class for propulsion-machinery rule applicability."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall declare whether the propeller is open or ducted."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall declare whether the propeller has controllable or fixed pitch."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one maximum ice torque quantity in kN*m."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable propulsion-machinery configuration, every linked design condition shall record normal service-life propeller/ice loads, inclusion of fixed-pitch reversal loads, and exclusion of stopped-propeller dragging and radial ice-entry loads from the prescribed model." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    &&
                    ?propellerType IN ("open", "ducted")
                    &&
                    ?pitchType IN ("controllable", "fixed")
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:hasDesignCondition ?anyCondition .
                    }
                    ||
                    EXISTS {
                        $this nltl:hasDesignCondition ?condition .
                        FILTER NOT EXISTS {
                            ?condition nltl:normalOperationalConditions true ;
                                       nltl:normalServiceLifePropellerIceLoadCondition true ;
                                       nltl:fixedPitchPropellerReversalLoadIncluded true ;
                                       nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                                       nltl:radialIceEntryOutsideLoadModel true .
                        }
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
