<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF076ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller type is required to determine whether the load-case rule applies."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For CP and FP propellers, load cases 1-4 must be covered; for an FP reversing propeller, load case 5 must also be covered, and the reversing capability must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (STR(?type) = "CP" || STR(?type) = "FP")

                FILTER (
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case1 .
                        ?case1 nltl:propellerBladeLoadCaseNumber ?number1 .
                        FILTER (?number1 = 1)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case2 .
                        ?case2 nltl:propellerBladeLoadCaseNumber ?number2 .
                        FILTER (?number2 = 2)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case3 .
                        ?case3 nltl:propellerBladeLoadCaseNumber ?number3 .
                        FILTER (?number3 = 3)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case4 .
                        ?case4 nltl:propellerBladeLoadCaseNumber ?number4 .
                        FILTER (?number4 = 4)
                    }
                    ||
                    (
                        STR(?type) = "FP"
                        &&
                        NOT EXISTS {
                            $this nltl:propellerReversingCapability ?reversing .
                            FILTER (datatype(?reversing) = xsd:boolean)
                        }
                    )
                    ||
                    (
                        STR(?type) = "FP"
                        &&
                        EXISTS {
                            $this nltl:propellerReversingCapability ?reversingTrue .
                            FILTER (
                                datatype(?reversingTrue) = xsd:boolean
                                && ?reversingTrue = true
                            )
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:hasPropellerBladeLoadCase ?case5 .
                            ?case5 nltl:propellerBladeLoadCaseNumber ?number5 .
                            FILTER (?number5 = 5)
                        }
                    )
                )
            }
        """
    ] .

gen:TRF076PropellerBladeLoadCaseShape a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasPropellerBladeLoadCase ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Each represented propeller blade load case must have an integer load-case number."
    ] .
<END_SHACL>
