@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF078ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 2 ;
        sh:maxCount 2 ;
        sh:node gen:TRF078TableLookupCaseShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF078DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF078DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF078DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Exactly one Table 6-6 lookup case is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                    ?case nltl:tableReference nltl:traficomTable6Dash6 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Exactly one Table 6-7 lookup case is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                    ?case nltl:tableReference nltl:traficomTable6Dash7 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Only one Table 6-6 lookup case is permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case1 .
                $this nltl:hasTableLookupCase ?case2 .
                ?case1 nltl:tableReference nltl:traficomTable6Dash6 .
                ?case2 nltl:tableReference nltl:traficomTable6Dash6 .
                FILTER (?case1 != ?case2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Only one Table 6-7 lookup case is permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case1 .
                $this nltl:hasTableLookupCase ?case2 .
                ?case1 nltl:tableReference nltl:traficomTable6Dash7 .
                ?case2 nltl:tableReference nltl:traficomTable6Dash7 .
                FILTER (?case1 != ?case2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The Table 6-6 case must contain an ice-class selection and selected ice-class cycle count." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:tableReference nltl:traficomTable6Dash6 .
                FILTER (
                    NOT EXISTS { ?case nltl:iceClass ?iceClass }
                    ||
                    NOT EXISTS { ?case nltl:selectedIceClassCycleCount ?count }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The Table 6-7 case must contain a propeller-location selection and selected location factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:tableReference nltl:traficomTable6Dash7 .
                FILTER (
                    NOT EXISTS { ?case nltl:tableLookupPropellerLocation ?location }
                    ||
                    NOT EXISTS { ?case nltl:selectedPropellerLocationFactor ?factor }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 multiplied by k2, k3, N_class, and n_n within the specified tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?classCase ;
                      nltl:hasTableLookupCase ?locationCase ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?reported .

                ?classCase nltl:tableReference nltl:traficomTable6Dash6 ;
                            nltl:selectedIceClassCycleCount ?nclassNode .

                ?locationCase nltl:tableReference nltl:traficomTable6Dash7 ;
                               nltl:selectedPropellerLocationFactor ?k1Node .

                ?k1Node qudt:numericValue ?k1 .
                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nclassNode qudt:numericValue ?nclass .
                ?nnNode qudt:numericValue ?nn .

                BIND ((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (
                    ABS(xsd:decimal(?reported) - ?expected)
                    >
                    (ABS(?expected) * 0.000001 + 0.000001)
                )
            }
        """
    ] .

gen:TRF078TableLookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:tableLookupPropellerLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:propellerLocationValue ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedIceClassCycleCount ;
        sh:nodeKind sh:IRI ;
        sh:maxCount 1 ;
        sh:node gen:TRF078CountQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedPropellerLocationFactor ;
        sh:nodeKind sh:IRI ;
        sh:maxCount 1 ;
        sh:node gen:TRF078DimensionlessQuantityShape
    ] .
