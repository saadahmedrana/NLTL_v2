    sh:sparql [
        sh:message "Propeller diameter must be non-zero to determine the immersion function." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?dNode .
                ?dNode qudt:numericValue ?d .
                FILTER (?d = 0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The immersion function and submersion factor k2 must satisfy the specified formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?hNode ;
                      nltl:iceLoadCycleCoefficientK2 ?kNode ;
                      nltl:propellerDiameter ?dNode ;
                      nltl:propellerImmersionFunction ?fNode ;
                      nltl:propellerShaftCentrelineDepth ?h0Node .
                ?hNode qudt:numericValue ?h .
                ?kNode qudt:numericValue ?k .
                ?dNode qudt:numericValue ?d .
                ?fNode qudt:numericValue ?fReported .
                ?h0Node qudt:numericValue ?h0 .
                FILTER (?d != 0)
                BIND (((?h0 - ?h) / (?d / 2)) - 1 AS ?fExpected)
                BIND (
                    IF(
                        ?fExpected < 0,
                        0.8 - ?fExpected,
                        IF(
                            ?fExpected <= 1,
                            0.8 - (0.4 * ?fExpected),
                            IF(
                                ?fExpected <= 2.5,
                                0.6 - (0.2 * ?fExpected),
                                0.1
                            )
                        )
                    ) AS ?kExpected
                )
                FILTER (
                    ABS(?fReported - ?fExpected) >
                    ((0.000001 * ABS(?fExpected)) + 0.000000001)
                    ||
                    ABS(?k - ?kExpected) >
                    ((0.000001 * ABS(?kExpected)) + 0.000000001)
                )
            }
        """
    ] .
