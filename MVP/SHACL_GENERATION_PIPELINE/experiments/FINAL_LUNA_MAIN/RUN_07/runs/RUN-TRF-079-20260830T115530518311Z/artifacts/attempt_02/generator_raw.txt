            sh:class qudt:QuantityValue ;

                sh:maxCount 1

            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;

            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

            ] ;
            sh:property [

            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;

            sh:property [

                sh:maxCount 1

            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The propeller immersion function and ice load cycle coefficient must satisfy the declared formulae, using a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?hIceNode ;
                      nltl:propellerDiameter ?diameterNode ;
                      nltl:propellerShaftCentrelineDepth ?depthNode ;
                      nltl:propellerImmersionFunction ?fNode ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node .
                ?hIceNode qudt:numericValue ?hIce .
                ?diameterNode qudt:numericValue ?diameter .
                ?depthNode qudt:numericValue ?depth .
                ?fNode qudt:numericValue ?fObserved .
                ?k2Node qudt:numericValue ?k2Observed .
                BIND((?depth - ?hIce) / (?diameter / 2) - 1 AS ?fExpected)
                BIND(
                    IF(
                        ?fExpected < 0,
                        0.8 - ?fExpected,
                        IF(
                            ?fExpected <= 1,
                            0.8 - (0.4 * ?fExpected),
                            IF(
                                ?fExpected <= 2.5,
                                0.6 - (0.2 * ?fExpected),
                                0.1
                            )
                        )
                    ) AS ?k2Expected
                )
                BIND(
                    IF(
                        ?fExpected >= 0,
                        (0.000001 * ?fExpected) + 0.000000001,
                        (0.000001 * (-1 * ?fExpected)) + 0.000000001
                    ) AS ?fTolerance
                )
                BIND(
                    IF(
                        ?k2Expected >= 0,
                        (0.000001 * ?k2Expected) + 0.000000001,
                        (0.000001 * (-1 * ?k2Expected)) + 0.000000001
                    ) AS ?k2Tolerance
                )
                FILTER(
                    ?fObserved < (?fExpected - ?fTolerance)
                    || ?fObserved > (?fExpected + ?fTolerance)
                    || ?k2Observed < (?k2Expected - ?k2Tolerance)
                    || ?k2Observed > (?k2Expected + ?k2Tolerance)
                )
            }
        """
    ] .
