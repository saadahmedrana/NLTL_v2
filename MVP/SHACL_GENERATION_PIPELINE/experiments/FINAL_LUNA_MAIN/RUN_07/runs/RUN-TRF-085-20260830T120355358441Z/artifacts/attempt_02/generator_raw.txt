            sh:sparql [
                sh:message "The Table 6-9 lookup input must represent the ship's propeller diameter." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this WHERE {
                        ?this nltl:propellerDiameter ?diameter .
                        ?diameter qudt:numericValue ?diameterValue .
                        ?this nltl:hasTableLookupCase ?case .
                        ?case nltl:lookupInputQuantity ?input .
                        ?input qudt:numericValue ?inputValue .
                        FILTER(
                            ABS(?diameterValue - ?inputValue) >

                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The Table 6-9 lookup result must equal the selected bollard propeller speed." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this WHERE {
                        ?this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed .
                        ?speed qudt:numericValue ?speedValue .
                        ?this nltl:hasTableLookupCase ?case .
                        ?case nltl:lookupResultQuantity ?result .
                        ?result qudt:numericValue ?resultValue .
                        FILTER(
                            ABS(?speedValue - ?resultValue) >
                            0.000001 * IF(ABS(?speedValue) > 1, ABS(?speedValue), 1)
                        )
                    }
                """

        ]
