<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF091ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091IceMillingCaseShape ;
        sh:message "The ship shall identify the ice-milling load-sequence case used for propulsion-line strength evaluation."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091PropulsionLoadCaseShape ;
        sh:message "The ship shall identify the propulsion-line strength-evaluation load case."
    ] .

gen:TRF091IceMillingCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
        sh:minCount 1 ;
        sh:message "The ice-milling load-sequence case shall have propulsion-line strength evaluation as its purpose."
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:loadCasePurpose ;
            sh:hasValue nltl:stallingAnalysisPurposeValue
        ]
    ] .

gen:TRF091PropulsionLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
        sh:minCount 1 ;
        sh:message "The propulsion load case shall have propulsion-line strength evaluation as its purpose."
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:loadCasePurpose ;
            sh:hasValue nltl:stallingAnalysisPurposeValue
        ]
    ] .
<END_SHACL>
