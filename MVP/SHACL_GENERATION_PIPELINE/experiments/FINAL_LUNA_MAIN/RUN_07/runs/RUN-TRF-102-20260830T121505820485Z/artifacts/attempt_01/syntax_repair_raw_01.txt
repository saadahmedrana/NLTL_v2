gen:TRF102Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedSpindleTorqueStressAnalysis ;
                sh:class nltl:approvalRecord ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumSpindleTorque ;
                sh:node gen:MaximumSpindleTorqueQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An approved spindle torque stress analysis must provide the maximum spindle torque."
            ]
        ]
        [
            sh:property [
                sh:path nltl:bladeExpandedAreaRatio ;
                sh:node gen:BladeExpandedAreaRatioQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerBladeCount ;
                sh:node gen:PropellerBladeCountValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:leadingEdgeChordPortionAtZeroPointEightRadius ;
                sh:node gen:ChordLengthQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:trailingEdgeChordPortionAtZeroPointEightRadius ;
                sh:node gen:ChordLengthQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:extremeIceForce ;
                sh:node gen:ExtremeIceForceQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeFailureSpindleTorqueFactor ;
                sh:node gen:BladeFailureSpindleTorqueFactorQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeFailureSpindleTorque ;
                sh:node gen:BladeFailureSpindleTorqueQuantity ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The blade failure spindle torque factor and torque must satisfy the declared formulas within a relative tolerance of 1e-6, with an absolute fallback of 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:bladeExpandedAreaRatio ?earNode ;
                              nltl:propellerBladeCount ?z ;
                              nltl:leadingEdgeChordPortionAtZeroPointEightRadius ?leNode ;
                              nltl:trailingEdgeChordPortionAtZeroPointEightRadius ?teNode ;
                              nltl:extremeIceForce ?forceNode ;
                              nltl:bladeFailureSpindleTorqueFactor ?factorNode ;
                              nltl:bladeFailureSpindleTorque ?torqueNode .
                        ?earNode qudt:numericValue ?ear .
                        ?leNode qudt:numericValue ?le .
                        ?teNode qudt:numericValue ?te .
                        ?forceNode qudt:numericValue ?force .
                        ?factorNode qudt:numericValue ?factor .
                        ?torqueNode qudt:numericValue ?torque .
                        BIND ((4.0 * ?ear) / xsd:decimal(?z) AS ?ratio)
                        BIND (0.7 * (1.0 - (?ratio * ?ratio * ?ratio)) AS ?rawFactor)
                        BIND (IF(?rawFactor < 0.3, 0.3, ?rawFactor) AS ?expectedFactor)
                        BIND (IF(?le > (0.8 * ?te), ?le, (0.8 * ?te)) AS ?chordFactor)
                        BIND (?chordFactor * ?expectedFactor * ?force AS ?expectedTorque)
                        FILTER (
                            ABS(?factor - ?expectedFactor) >
                            (0.000001 * ABS(?expectedFactor) + 0.000000001)
                            ||
                            ABS(?torque - ?expectedTorque) >
                            (0.000001 * ABS(?expectedTorque) + 0.000000001)
                        )
                    }
                """
            ]
        ]
    ) .
