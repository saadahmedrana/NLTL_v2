<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF108Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall state whether the SN-curve is known."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the SN-curve type applicability input."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The blade material properties shall be provided for SN-curve correspondence."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide one selected SN-curve type."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Evidence shall demonstrate correspondence between the selected SN-curve type and the blade material properties."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:hasValue nltl:twoSlopeStressLifeCurve
            ]
        ]
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue true
            ]
        ]
    ) .
<END_SHACL>
