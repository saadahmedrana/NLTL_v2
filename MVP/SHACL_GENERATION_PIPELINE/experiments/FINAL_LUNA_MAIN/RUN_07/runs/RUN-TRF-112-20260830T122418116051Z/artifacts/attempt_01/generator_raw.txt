<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-112-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:message "The ship shall identify the applicable propeller shaft line component." ;
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:class nltl:loadCase ;
        sh:message "The ship shall identify the propulsion load case used to establish applicability and compliance." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a blade-failure load case, the externally calculated ultimate and combined loads and the bending and torsional yield safety factors shall be provided with the required units; both safety factors shall be at least 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .

                FILTER NOT EXISTS {
                    ?loadCase nltl:bladeFailureUltimateLoad ?ultimateLoad .
                    ?ultimateLoad qudt:numericValue ?ultimateValue ;
                                  qudt:unit qudt:KiloN .
                    FILTER (datatype(?ultimateValue) = xsd:decimal)

                    ?loadCase nltl:combinedAxialBendingTorsionLoad ?combinedLoad .
                    ?combinedLoad qudt:numericValue ?combinedValue ;
                                  qudt:unit qudt:KiloN .
                    FILTER (datatype(?combinedValue) = xsd:decimal)

                    ?component nltl:bendingYieldSafetyFactor ?bendingFactor .
                    ?bendingFactor qudt:numericValue ?bendingValue ;
                                   qudt:unit qudt:UNITLESS .
                    FILTER (
                        datatype(?bendingValue) = xsd:decimal &&
                        ?bendingValue >= 1.0
                    )

                    ?component nltl:torsionalYieldSafetyFactor ?torsionalFactor .
                    ?torsionalFactor qudt:numericValue ?torsionalValue ;
                                     qudt:unit qudt:UNITLESS .
                    FILTER (
                        datatype(?torsionalValue) = xsd:decimal &&
                        ?torsionalValue >= 1.0
                    )
                }
            }
        """ ;
    ] .

<urn:nltl:generated-shape:TRF-112-PropulsionLoadCaseShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasPropulsionLoadCase ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:shaftLineLoadCaseTypeValue ;
        sh:message "Each propulsion load case shall identify one shaft-line load-case type so that applicability can be determined." ;
    ] .
<END_SHACL>
