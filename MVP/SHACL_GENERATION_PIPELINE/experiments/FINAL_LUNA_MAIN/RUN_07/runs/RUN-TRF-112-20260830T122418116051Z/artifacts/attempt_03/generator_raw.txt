<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-112-ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class nltl:shaftLineLoadCaseTypeValue
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable blade-failure load case must provide valid ultimate and combined axial, bending, and torsion load quantities in kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                {
                    FILTER NOT EXISTS {
                        ?loadCase nltl:bladeFailureUltimateLoad ?missingUltimateLoad
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?loadCase nltl:combinedAxialBendingTorsionLoad ?missingCombinedLoad
                    }
                }
                UNION
                {
                    ?loadCase ?operand ?quantity .
                    FILTER (
                        ?operand = nltl:bladeFailureUltimateLoad ||
                        ?operand = nltl:combinedAxialBendingTorsionLoad
                    )
                    FILTER NOT EXISTS {
                        ?quantity qudt:numericValue ?numericValue
                    }
                }
                UNION
                {
                    ?loadCase ?operand ?quantity .
                    FILTER (
                        ?operand = nltl:bladeFailureUltimateLoad ||
                        ?operand = nltl:combinedAxialBendingTorsionLoad
                    )
                    FILTER NOT EXISTS {
                        ?quantity qudt:unit qudt:KiloN
                    }
                }
                UNION
                {
                    ?loadCase ?operand ?quantity .
                    FILTER (
                        ?operand = nltl:bladeFailureUltimateLoad ||
                        ?operand = nltl:combinedAxialBendingTorsionLoad
                    )
                    ?quantity qudt:unit ?unit .
                    FILTER (?unit != qudt:KiloN)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each reported bending and torsional yield safety factor on every applicable propeller shaft line component must have a numeric value, unit 1, and a value of at least 1.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                {
                    FILTER NOT EXISTS {
                        ?component nltl:bendingYieldSafetyFactor ?missingBendingFactor
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?component nltl:torsionalYieldSafetyFactor ?missingTorsionalFactor
                    }
                }
                UNION
                {
                    ?component ?factor ?quantity .
                    FILTER (
                        ?factor = nltl:bendingYieldSafetyFactor ||
                        ?factor = nltl:torsionalYieldSafetyFactor
                    )
                    FILTER NOT EXISTS {
                        ?quantity qudt:numericValue ?numericValue
                    }
                }
                UNION
                {
                    ?component ?factor ?quantity .
                    FILTER (
                        ?factor = nltl:bendingYieldSafetyFactor ||
                        ?factor = nltl:torsionalYieldSafetyFactor
                    )
                    FILTER NOT EXISTS {
                        ?quantity qudt:unit qudt:UNITLESS
                    }
                }
                UNION
                {
                    ?component ?factor ?quantity .
                    FILTER (
                        ?factor = nltl:bendingYieldSafetyFactor ||
                        ?factor = nltl:torsionalYieldSafetyFactor
                    )
                    ?quantity qudt:unit ?unit .
                    FILTER (?unit != qudt:UNITLESS)
                }
                UNION
                {
                    ?component ?factor ?quantity .
                    FILTER (
                        ?factor = nltl:bendingYieldSafetyFactor ||
                        ?factor = nltl:torsionalYieldSafetyFactor
                    )
                    ?quantity qudt:numericValue ?numericValue .
                    FILTER (?numericValue < 1.0)
                }
            }
        """
    ] .
<END_SHACL>
