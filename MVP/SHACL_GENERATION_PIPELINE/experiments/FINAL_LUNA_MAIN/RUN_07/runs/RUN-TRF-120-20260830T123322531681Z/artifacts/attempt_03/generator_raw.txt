<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF120Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql <urn:nltl:generated-shape:TRF120Constraint> .

<urn:nltl:generated-shape:TRF120Constraint>
    a sh:SPARQLConstraint ;
    sh:message "For each thruster-body natural-frequency calculation case, longitudinal or transverse vibration direction must be declared; the ship must provide damping, water added mass, and thruster natural frequency; and the thruster body must provide ship attachment stiffness with the canonical quantity units." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX qudt: <http://qudt.org/schema/qudt/>
        PREFIX sh: <http://www.w3.org/ns/shacl#>
        PREFIX unit: <http://qudt.org/vocab/unit/>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

        SELECT DISTINCT ?this
        WHERE {
            ?this nltl:hasComponent ?body .
            ?body a nltl:thrusterBody ;
                  nltl:hasNaturalFrequencyCalculationCase ?case .

            FILTER (
                NOT EXISTS {
                    ?case nltl:vibrationDirection ?validDirection .
                    FILTER (
                        ?validDirection = nltl:longitudinalDirection ||
                        ?validDirection = nltl:transverseDirection
                    )
                }
                ||
                EXISTS {
                    ?case nltl:vibrationDirection ?invalidDirection .
                    FILTER (
                        ?invalidDirection != nltl:longitudinalDirection &&
                        ?invalidDirection != nltl:transverseDirection
                    )
                }
                ||
                NOT EXISTS {
                    ?this nltl:damping ?damping .
                    FILTER (datatype(?damping) = xsd:string)
                }
                ||
                EXISTS {
                    ?this nltl:damping ?invalidDamping .
                    FILTER (datatype(?invalidDamping) != xsd:string)
                }
                ||
                NOT EXISTS {
                    ?this nltl:waterAddedMass ?waterMass .
                    ?waterMass a qudt:QuantityValue ;
                               qudt:numericValue ?waterMassValue ;
                               qudt:unit unit:KiloGM .
                    FILTER (datatype(?waterMassValue) = xsd:decimal)
                }
                ||
                EXISTS {
                    ?this nltl:waterAddedMass ?invalidWaterMass .
                    FILTER NOT EXISTS {
                        ?invalidWaterMass a qudt:QuantityValue ;
                                           qudt:numericValue ?validWaterMassValue ;
                                           qudt:unit unit:KiloGM .
                        FILTER (datatype(?validWaterMassValue) = xsd:decimal)
                    }
                }
                ||
                NOT EXISTS {
                    ?this nltl:thrusterNaturalFrequency ?frequency .
                    ?frequency a qudt:QuantityValue ;
                               qudt:numericValue ?frequencyValue ;
                               qudt:unit unit:HZ .
                    FILTER (datatype(?frequencyValue) = xsd:decimal)
                }
                ||
                EXISTS {
                    ?this nltl:thrusterNaturalFrequency ?invalidFrequency .
                    FILTER NOT EXISTS {
                        ?invalidFrequency a qudt:QuantityValue ;
                                           qudt:numericValue ?validFrequencyValue ;
                                           qudt:unit unit:HZ .
                        FILTER (datatype(?validFrequencyValue) = xsd:decimal)
                    }
                }
                ||
                NOT EXISTS {
                    ?body nltl:shipAttachmentStiffness ?stiffness .
                    ?stiffness a qudt:QuantityValue ;
                               qudt:numericValue ?stiffnessValue ;
                               qudt:unit unit:KiloN-PER-M .
                    FILTER (datatype(?stiffnessValue) = xsd:decimal)
                }
                ||
                EXISTS {
                    ?body nltl:shipAttachmentStiffness ?invalidStiffness .
                    FILTER NOT EXISTS {
                        ?invalidStiffness a qudt:QuantityValue ;
                                           qudt:numericValue ?validStiffnessValue ;
                                           qudt:unit unit:KiloN-PER-M .
                        FILTER (datatype(?validStiffnessValue) = xsd:decimal)
                    }
                }
            )
        }
    """ .
<END_SHACL>
