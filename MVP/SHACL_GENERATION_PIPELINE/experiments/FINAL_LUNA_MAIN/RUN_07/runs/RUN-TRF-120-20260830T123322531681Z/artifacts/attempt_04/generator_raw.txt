<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each linked thruster natural-frequency calculation case, both longitudinal and transverse directions must be represented and the ship damping, water added mass, thruster natural frequency, and linked thruster-body attachment stiffness inputs must be present with their canonical datatypes and units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this rdf:type nltl:ship .
                $this nltl:hasComponent ?body .
                ?body rdf:type nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .

                FILTER (
                    NOT EXISTS {
                        $this nltl:damping ?damping .
                        FILTER (datatype(?damping) = xsd:string)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:waterAddedMass ?waterMass .
                        ?waterMass rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?waterMassValue ;
                            qudt:unit unit:KiloGM .
                        FILTER (datatype(?waterMassValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterNaturalFrequency ?naturalFrequency .
                        ?naturalFrequency rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?naturalFrequencyValue ;
                            qudt:unit unit:HZ .
                        FILTER (datatype(?naturalFrequencyValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?body nltl:shipAttachmentStiffness ?attachmentStiffness .
                        ?attachmentStiffness rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?attachmentStiffnessValue ;
                            qudt:unit unit:KiloN-PER-M .
                        FILTER (datatype(?attachmentStiffnessValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:vibrationDirection nltl:longitudinalDirection
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:vibrationDirection nltl:transverseDirection
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
