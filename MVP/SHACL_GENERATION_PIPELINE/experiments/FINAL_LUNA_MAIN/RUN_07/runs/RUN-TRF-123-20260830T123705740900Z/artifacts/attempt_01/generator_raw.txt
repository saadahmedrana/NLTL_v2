<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF123ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node gs:TRF123OccasionalForceLoadCaseShape ;
        sh:message "The ship must provide an occasional-force load case for the required analysis."
    ] .

gs:TRF123OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The occasional-force load case must confirm that propeller blades are excluded from its scope."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The occasional-force load case must provide its externally calculated component yield safety factor as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:node gs:TRF123AssessedComponentShape ;
        sh:message "The occasional-force load case must identify each assessed component."
    ] ;
    sh:sparql [
        sh:message "The externally calculated occasional-force component stress, including the safety margin, must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:occasionalForceCaseAssessedComponent ?component .
                $this nltl:componentYieldSafetyFactor/qudt:numericValue ?safetyFactor .
                ?component nltl:occasionalForceComponentStress/qudt:numericValue ?stress .
                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength/qudt:numericValue ?yieldStrength .
                FILTER (
                    ?stress * ?safetyFactor > ?yieldStrength
                )
            }
        """
    ] .

gs:TRF123AssessedComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each assessed component must provide its externally calculated occasional-force stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:class nltl:materialProperties ;
        sh:minCount 1 ;
        sh:node gs:TRF123MaterialPropertiesShape ;
        sh:message "Each assessed component must identify its material properties."
    ] .

gs:TRF123MaterialPropertiesShape
    a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The component material properties must provide the material yield strength in MPa."
    ] .
<END_SHACL>
