<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF128Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:maxInclusive "0.5"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "Compressor charge time must be at most 1 hour unless the ship is ice class IA Super and requires propulsion-engine reversal for astern." ;
                sh:select '''
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?ic ;
                              nltl:propulsionEngineReversalRequiredForAstern ?rev ;
                              nltl:compressorChargeTime ?charge .
                        ?charge qudt:numericValue ?v ;
                                qudt:unit unit:HR .
                        FILTER (
                            (?ic = nltl:iceClassIaSuper && ?rev = true)
                            || ?v > "1"^^xsd:decimal
                        )
                    }
                '''
            ]
        ]
    ) .
</urn:nltl:generated-shape:TRF128Shape>
<END_SHACL>
