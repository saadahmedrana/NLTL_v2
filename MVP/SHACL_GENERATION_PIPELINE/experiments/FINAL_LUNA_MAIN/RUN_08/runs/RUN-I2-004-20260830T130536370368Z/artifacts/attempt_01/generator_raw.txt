<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_004_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:KiloTON_Metric
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloTON_Metric
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship displacement must equal the displacement of the upper ice waterline case having the greatest displacement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?dui .
                ?dui qudt:numericValue ?duiValue ;
                     qudt:unit unit:KiloTON_Metric .

                $this nltl:hasUpperIceWaterlineCase ?matchingCase .
                ?matchingCase nltl:waterlineCaseDisplacement ?matchingDisplacement .
                ?matchingDisplacement qudt:numericValue ?matchingValue ;
                                      qudt:unit unit:KiloTON_Metric .
                FILTER (?matchingValue = ?duiValue)

                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?greaterCase .
                    ?greaterCase nltl:waterlineCaseDisplacement ?greaterDisplacement .
                    ?greaterDisplacement qudt:numericValue ?greaterValue ;
                                          qudt:unit unit:KiloTON_Metric .
                    FILTER (?greaterValue > ?duiValue)
                }
            }
        """
    ] .
<END_SHACL>
