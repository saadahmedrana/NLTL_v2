            nltl:polarClassPc6
            nltl:polarClassPc7

        sh:message "A valid Polar Class value is required to establish applicability."

    sh:property [
        sh:path nltl:bowShape ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:averageIcePressure ;

    sh:property [

    sh:property [
        sh:path nltl:loadPatchLength ;

    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "Local ice pressure is required as an engineering input."
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape ;
