<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the hull area."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_PressureQuantityShape ;
        sh:message "Average ice pressure must be supplied in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_AspectRatioQuantityShape ;
        sh:message "The load patch aspect ratio must be exactly 3.6 and dimensionless."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_LengthQuantityShape ;
        sh:message "The non-bow load patch height must be supplied in metres."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_LengthQuantityShape ;
        sh:message "The non-bow load patch length must be supplied in metres."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_ButtockAngleShape ;
        sh:message "The buttock angle must be greater than 0 and less than 80 degrees."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_009_NormalFrameAngleShape ;
        sh:message "The normal frame angle must be greater than 10 degrees."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability must be explicitly recorded."
    ] ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:minCount 1 ;
        sh:node gen:I2_009_HullAreaAssignmentShape ;
        sh:message "The ship must provide a hull-area assignment."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow-form applicability must agree with the buttock-angle and normal-frame-angle conditions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability ?app .
                FILTER (
                    (?app = true &&
                     NOT EXISTS {
                         $this nltl:buttockAngle ?gammaNode .
                         ?gammaNode qudt:numericValue ?gamma .
                         ?gammaNode qudt:unit <http://qudt.org/vocab/unit/DEG> .
                         $this nltl:normalFrameAngle ?betaNode .
                         ?betaNode qudt:numericValue ?beta .
                         ?betaNode qudt:unit <http://qudt.org/vocab/unit/DEG> .
                         FILTER ( ?gamma > 0 && ?gamma < 80 && ?beta > 10 )
                     })
                    ||
                    (?app = false &&
                     EXISTS {
                         $this nltl:buttockAngle ?gammaNode2 .
                         ?gammaNode2 qudt:numericValue ?gamma2 .
                         ?gammaNode2 qudt:unit <http://qudt.org/vocab/unit/DEG> .
                         $this nltl:normalFrameAngle ?betaNode2 .
                         ?betaNode2 qudt:numericValue ?beta2 .
                         ?betaNode2 qudt:unit <http://qudt.org/vocab/unit/DEG> .
                         FILTER ( ?gamma2 > 0 && ?gamma2 < 80 && ?beta2 > 10 )
                     })
                )
            }
        """
    ] .

gen:I2_009_HullAreaAssignmentShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaAssignment ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:hullAreaValue ;
        sh:in (
            nltl:otherIceStrengthenedArea
            nltl:bowIntermediateBottomHullArea
            nltl:midbodyBottomHullArea
            nltl:sternBottomHullArea
        ) ;
        sh:message "The assigned hull area must be an allowed ice-strengthened hull-area value."
    ] .

gen:I2_009_PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_009_AspectRatioQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 3.6 ;
        sh:maxInclusive 3.6
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_009_LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_009_ButtockAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0 ;
        sh:maxExclusive 80
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_009_NormalFrameAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 10
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
