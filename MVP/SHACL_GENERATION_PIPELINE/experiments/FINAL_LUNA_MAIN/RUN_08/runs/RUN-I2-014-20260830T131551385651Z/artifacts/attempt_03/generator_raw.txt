    sh:targetClass nltl:ship ;

    sh:property [

    sh:property [

    sh:property [

    ] ;

    sh:property [
        sh:path nltl:selectedMaximumBowForce ;

                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

    sh:property [

    sh:property [

    sh:sparql [
        sh:message "The four bow subregion mid-length positions must represent four distinct equal subdivisions of bowLength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthQuantity ;
                      nltl:hasBowSubregionCalculationCase ?c1, ?c2, ?c3, ?c4 .
                ?lengthQuantity qudt:numericValue ?length .

                ?c1 nltl:bowSubregionMidLengthPosition ?p1 .
                ?c2 nltl:bowSubregionMidLengthPosition ?p2 .
                ?c3 nltl:bowSubregionMidLengthPosition ?p3 .
                ?c4 nltl:bowSubregionMidLengthPosition ?p4 .

                ?p1 qudt:numericValue ?m1 .
                ?p2 qudt:numericValue ?m2 .
                ?p3 qudt:numericValue ?m3 .
                ?p4 qudt:numericValue ?m4 .

                FILTER (?c1 != ?c2 && ?c1 != ?c3 && ?c1 != ?c4 &&
                        ?c2 != ?c3 && ?c2 != ?c4 && ?c3 != ?c4)

                FILTER (
                    (
                        ((?m1 - (?length / 8)) <= ((?length / 8) * 0.000001 + 0.000000001) &&
                         (?m1 - (?length / 8)) >= -((?length / 8) * 0.000001 + 0.000000001)) &&
                        ((?m2 - (?length / 8)) <= ((?length / 8) * 0.000001 + 0.000000001) &&
                         (?m2 - (?length / 8)) >= -((?length / 8) * 0.000001 + 0.000000001)))
                    ||
                    (
                        ((?m1 - (3 * ?length / 8)) <= ((3 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m1 - (3 * ?length / 8)) >= -((3 * ?length / 8) * 0.000001 + 0.000000001)) &&
                        ((?m2 - (3 * ?length / 8)) <= ((3 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m2 - (3 * ?length / 8)) >= -((3 * ?length / 8) * 0.000001 + 0.000000001)))
                    ||
                    (
                        ((?m1 - (5 * ?length / 8)) <= ((5 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m1 - (5 * ?length / 8)) >= -((5 * ?length / 8) * 0.000001 + 0.000000001)) &&
                        ((?m2 - (5 * ?length / 8)) <= ((5 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m2 - (5 * ?length / 8)) >= -((5 * ?length / 8) * 0.000001 + 0.000000001)))
                    ||
                    (
                        ((?m1 - (7 * ?length / 8)) <= ((7 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m1 - (7 * ?length / 8)) >= -((7 * ?length / 8) * 0.000001 + 0.000000001)) &&
                        ((?m2 - (7 * ?length / 8)) <= ((7 * ?length / 8) * 0.000001 + 0.000000001) &&
                         (?m2 - (7 * ?length / 8)) >= -((7 * ?length / 8) * 0.000001 + 0.000000001)))
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "The four bow subregion mid-length positions must collectively match the four equal-subdivision midpoint positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?lengthQuantity qudt:numericValue ?length .
                ?case nltl:bowSubregionMidLengthPosition ?position .
                ?position qudt:numericValue ?midpoint .

                FILTER (
                    (?midpoint - (?length / 8) > ((?length / 8) * 0.000001 + 0.000000001) ||
                     ?midpoint - (?length / 8) < -((?length / 8) * 0.000001 + 0.000000001)) &&
                    (?midpoint - (3 * ?length / 8) > ((3 * ?length / 8) * 0.000001 + 0.000000001) ||
                     ?midpoint - (3 * ?length / 8) < -((3 * ?length / 8) * 0.000001 + 0.000000001)) &&
                    (?midpoint - (5 * ?length / 8) > ((5 * ?length / 8) * 0.000001 + 0.000000001) ||
                     ?midpoint - (5 * ?length / 8) < -((5 * ?length / 8) * 0.000001 + 0.000000001)) &&
                    (?midpoint - (7 * ?length / 8) > ((7 * ?length / 8) * 0.000001 + 0.000000001) ||
                     ?midpoint - (7 * ?length / 8) < -((7 * ?length / 8) * 0.000001 + 0.000000001))
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each selected bow maximum must equal one reported subregion value and be no less than every reported value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasBowSubregionCalculationCase ?c1, ?c2, ?c3, ?c4 ;
                      nltl:selectedMaximumBowForce ?sfq ;
                      nltl:selectedMaximumBowLineLoad ?sqlq ;
                      nltl:selectedMaximumBowPressure ?spq .

                ?c1 nltl:bowSubregionCalculatedForce ?f1 ;
                    nltl:bowSubregionCalculatedLineLoad ?q1 ;
                    nltl:bowSubregionCalculatedPressure ?p1 .
                ?c2 nltl:bowSubregionCalculatedForce ?f2 ;
                    nltl:bowSubregionCalculatedLineLoad ?q2 ;
                    nltl:bowSubregionCalculatedPressure ?p2 .
                ?c3 nltl:bowSubregionCalculatedForce ?f3 ;
                    nltl:bowSubregionCalculatedLineLoad ?q3 ;
                    nltl:bowSubregionCalculatedPressure ?p3 .
                ?c4 nltl:bowSubregionCalculatedForce ?f4 ;
                    nltl:bowSubregionCalculatedLineLoad ?q4 ;
                    nltl:bowSubregionCalculatedPressure ?p4 .

                ?f1 qudt:numericValue ?fv1 .
                ?f2 qudt:numericValue ?fv2 .
                ?f3 qudt:numericValue ?fv3 .
                ?f4 qudt:numericValue ?fv4 .
                ?q1 qudt:numericValue ?qv1 .
                ?q2 qudt:numericValue ?qv2 .
                ?q3 qudt:numericValue ?qv3 .
                ?q4 qudt:numericValue ?qv4 .
                ?p1 qudt:numericValue ?pv1 .
                ?p2 qudt:numericValue ?pv2 .
                ?p3 qudt:numericValue ?pv3 .
                ?p4 qudt:numericValue ?pv4 .
                ?sfq qudt:numericValue ?sf .
                ?sqlq qudt:numericValue ?sql .
                ?spq qudt:numericValue ?sp .

                FILTER (
                    (?sf < ?fv1 || ?sf < ?fv2 || ?sf < ?fv3 || ?sf < ?fv4 ||
                     (?sf != ?fv1 && ?sf != ?fv2 && ?sf != ?fv3 && ?sf != ?fv4)) ||
                    (?sql < ?qv1 || ?sql < ?qv2 || ?sql < ?qv3 || ?sql < ?qv4 ||
                     (?sql != ?qv1 && ?sql != ?qv2 && ?sql != ?qv3 && ?sql != ?qv4)) ||
                    (?sp < ?pv1 || ?sp < ?pv2 || ?sp < ?pv3 || ?sp < ?pv4 ||
                     (?sp != ?pv1 && ?sp != ?pv2 && ?sp != ?pv3 && ?sp != ?pv4))
                )
            }
        """
    ] .
