<urn:nltl:generated-shape:I2-015-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:node <urn:nltl:generated-shape:I2-015-BowSubregionCalculationCaseShape>
    ] ;
    sh:sparql [
        sh:message "For every applicable I2.3.1(v) bow calculation case, the ship must provide beta i prime and upper ice waterline length with their prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasBowSubregionCalculationCase ?case .
                ?case nltl:bowFormApplicabilityClassification
                    nltl:i231vBowFormApplicability .
                FILTER (
                    NOT EXISTS {
                        $this nltl:betaIPrime ?beta .
                        ?beta a qudt:QuantityValue ;
                              qudt:numericValue ?betaValue ;
                              qudt:unit qudt:UNITLESS .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:upperIceWaterlineLengthLUI ?length .
                        ?length a qudt:QuantityValue ;
                                qudt:numericValue ?lengthValue ;
                                qudt:unit qudt:M .
                    }
                )
            }
        """
    ] .
