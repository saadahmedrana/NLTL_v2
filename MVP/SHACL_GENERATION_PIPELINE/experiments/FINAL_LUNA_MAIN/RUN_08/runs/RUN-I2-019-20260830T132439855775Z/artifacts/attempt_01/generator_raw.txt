<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:HullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gs:StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:node gs:HullAreaValueShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A structural member spanning hull areas must have a selected hull area factor equal to the largest spanned hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSpannedHullArea ?area .
                ?area nltl:hullAreaFactor ?factor .
                ?factor qudt:numericValue ?factorValue .
                FILTER NOT EXISTS {
                    $this nltl:hasSpannedHullArea ?otherArea .
                    ?otherArea nltl:hullAreaFactor ?otherFactor .
                    ?otherFactor qudt:numericValue ?otherFactorValue .
                    FILTER (?otherFactorValue > ?factorValue)
                }
                FILTER NOT EXISTS {
                    $this nltl:selectedHullAreaFactor ?selected .
                    ?selected qudt:numericValue ?selectedValue .
                    FILTER (?selectedValue = ?factorValue)
                }
            }
        """
    ] .

gs:I2_019_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberShape
    ] .
<END_SHACL>
