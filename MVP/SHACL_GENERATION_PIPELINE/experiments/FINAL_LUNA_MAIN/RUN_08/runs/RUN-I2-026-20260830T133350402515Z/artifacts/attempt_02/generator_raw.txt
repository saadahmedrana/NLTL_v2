<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:message "Each structural member must satisfy the support-fixity rule."
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A structural member must have one frame support condition."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:SupportInputShape ;
    ] ;
    sh:or (
        gen:ContinuousFixedBranch
        gen:BracketFixedBranch
        gen:DemonstratedRestraintFixedBranch
        gen:TerminatingFixedBranch
        gen:NonTerminatingSimpleBranch
    ) .

gen:ContinuousFixedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:SupportInputShape ;
    ] .

gen:BracketFixedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:BracketSupportShape ;
    ] .

gen:DemonstratedRestraintFixedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:DemonstratedRestraintSupportShape ;
    ] .

gen:TerminatingFixedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:NeitherSupportShape ;
    ] .

gen:NonTerminatingSimpleBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:simpleFrameSupport ;
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:NeitherSupportShape ;
    ] .

gen:SupportInputShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:or (
        gen:DemonstratedEvidenceShape
        gen:NoDemonstratedRestraintShape
    ) .

gen:DemonstratedEvidenceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Demonstrated significant rotational restraint requires evidence."
    ] .

gen:NoDemonstratedRestraintShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue false ;
    ] .

gen:BracketSupportShape
    a sh:NodeShape ;
    sh:node gen:SupportInputShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:hasValue true ;
    ] .

gen:DemonstratedRestraintSupportShape
    a sh:NodeShape ;
    sh:node gen:SupportInputShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
    ] .

gen:NeitherSupportShape
    a sh:NodeShape ;
    sh:node gen:SupportInputShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue false ;
    ] .
<END_SHACL>
