<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:I2-026:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:message "Each structural member must declare one permitted frame support condition."
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:MemberSupportShape
    ] ;
    sh:or (
        gen:FixedByContinuityShape
        gen:FixedByConnectionBracketShape
        gen:FixedByDemonstratedRestraintShape
        gen:FixedByTerminationShape
        gen:SimpleSupportShape
    ) .

gen:MemberSupportShape
    a sh:NodeShape ;
    sh:or (
        gen:ConnectionBracketSupportShape
        gen:DemonstratedRestraintSupportShape
        gen:NonFixingSupportShape
    ) .

gen:ConnectionBracketSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] .

gen:DemonstratedRestraintSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] .

gen:NonFixingSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] .

gen:FixedByContinuityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport
    ] .

gen:FixedByConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:qualifiedValueShape gen:ConnectionBracketSupportShape ;
        sh:qualifiedMinCount 1
    ] .

gen:FixedByDemonstratedRestraintShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:qualifiedValueShape gen:DemonstratedRestraintSupportShape ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Demonstrated significant rotational restraint requires evidence."
    ] .

gen:FixedByTerminationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
        sh:message "A structural member terminating within the ice-strengthened area must have fixed support."
    ] .

gen:SimpleSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:simpleFrameSupport
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:NonFixingSupportShape
    ] .

</END_SHACL>
