gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node gen:QuantityMillimetreShape ;
        sh:or (
            [
                sh:minCount 1 ;
                sh:maxCount 1
            ]
            [
                sh:sparql [
                    sh:message "Plastic neutral axis height is required when local-frame area exceeds attached-plate-flange area." ;
                    sh:select """
                        PREFIX nltl: <https://w3id.org/nltl/vocab#>
                        PREFIX qudt: <http://qudt.org/schema/qudt/>
                        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                        SELECT $this
                        WHERE {
                            ?ship nltl:hasCalculationCase $this .
                            ?ship nltl:hasStructuralMember ?member .
                            ?member nltl:netLocalFrameFlangeArea ?area ;
                                    nltl:netAttachedShellPlateThickness ?thickness ;
                                    nltl:frameSpacing ?spacing .
                            ?area qudt:numericValue ?a .
                            ?thickness qudt:numericValue ?t .
                            ?spacing qudt:numericValue ?s .
                            FILTER (?a <= 10 * ?t * ?s)
                        }
                    """
                ]
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:or (
            [
                sh:minCount 1 ;
                sh:maxCount 1
            ]
            [
                sh:sparql [
                    sh:message "Lower interpolation endpoint is required for oblique framing." ;
                    sh:select """
                        PREFIX nltl: <https://w3id.org/nltl/vocab#>
                        PREFIX qudt: <http://qudt.org/schema/qudt/>
                        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                        SELECT $this
                        WHERE {
                            ?ship nltl:hasCalculationCase $this .
                            ?ship nltl:hasStructuralMember ?member .
                            ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                            ?localCase nltl:sectionCalculationCasePlating ?plating .
                            ?plating nltl:framingAngleOmega ?angle .
                            ?angle qudt:numericValue ?omega .
                            FILTER (?omega <= 20 || ?omega >= 70)
                        }
                    """
                ]
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:or (
            [
                sh:minCount 1 ;
                sh:maxCount 1
            ]
            [
                sh:sparql [
                    sh:message "Upper interpolation endpoint is required for oblique framing." ;
                    sh:select """
                        PREFIX nltl: <https://w3id.org/nltl/vocab#>
                        PREFIX qudt: <http://qudt.org/schema/qudt/>
                        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                        SELECT $this
                        WHERE {
                            ?ship nltl:hasCalculationCase $this .
                            ?ship nltl:hasStructuralMember ?member .
                            ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                            ?localCase nltl:sectionCalculationCasePlating ?plating .
                            ?plating nltl:framingAngleOmega ?angle .
                            ?angle qudt:numericValue ?omega .
                            FILTER (?omega <= 20 || ?omega >= 70)
                        }
                    """
                ]
            ]
        )
    ] .
