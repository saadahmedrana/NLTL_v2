<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node gs:I2_031_QuantityShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the patch load." ;
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must identify the plastic collapse mechanism produced by the midspan collapse load." ;
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_031_MemberShape ;
    ] .

gs:I2_031_MemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node gs:I2_031_LoadCaseShape ;
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gs:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gs:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node gs:I2_031_BottomLoadCaseShape ;
                sh:minCount 1 ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:transverseFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each applicable structural member, combined shear and bending demand must not exceed plastic strength, and plastic strength must equal the midspan load producing the plastic collapse mechanism within tolerance." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  ?ship nltl:hasStructuralMember $this ;
        nltl:plasticCollapseMechanism ?mechanism ;
        nltl:plasticStrength ?strengthQuantity ;
        nltl:midspanPlasticCollapseLoad ?midspanQuantity .
  $this nltl:hasStructuralMemberLoadCase ?loadCase ;
        nltl:plasticStrength ?memberStrengthQuantity ;
        nltl:midspanPlasticCollapseLoad ?memberMidspanQuantity .
  ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
  ?demandQuantity qudt:numericValue ?demandValue .
  ?memberStrengthQuantity qudt:numericValue ?strengthValue .
  ?memberMidspanQuantity qudt:numericValue ?midspanValue .
  FILTER (
    ?demandValue > ?strengthValue ||
    ABS(?strengthValue - ?midspanValue) >
      (ABS(?midspanValue) * 0.000001 + 0.000000001)
  )
}
""" ;
    ] .

gs:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gs:I2_031_QuantityShape ;
        sh:minCount 1 ;
    ] .

gs:I2_031_BottomLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gs:I2_031_QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gs:I2_031_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
