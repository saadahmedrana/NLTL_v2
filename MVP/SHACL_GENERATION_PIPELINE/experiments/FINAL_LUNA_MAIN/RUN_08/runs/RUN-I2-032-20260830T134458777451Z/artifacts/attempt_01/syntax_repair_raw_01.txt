@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    sh:class qudt:QuantityValue ;

        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_032_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityPressureShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityLengthMShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityLengthMShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityDimensionlessShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityAreaM2Shape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityLengthMShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityPressureShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "The actual net effective shear area must satisfy the I2.6.2 formula and be at least the calculated required shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressure ;
                      nltl:frameSpan ?frameSpan ;
                      nltl:loadPatchHeight ?patchHeight ;
                      nltl:peakPressureFactor ?peakFactor ;
                      nltl:shearArea ?actualArea ;
                      nltl:yieldStrength ?yieldStrength ;
                      nltl:hasStructuralMember ?member .

                ?pressure <http://qudt.org/schema/qudt/numericValue> ?pAvg .
                ?frameSpan <http://qudt.org/schema/qudt/numericValue> ?a .
                ?patchHeight <http://qudt.org/schema/qudt/numericValue> ?b .
                ?peakFactor <http://qudt.org/schema/qudt/numericValue> ?ppf .
                ?actualArea <http://qudt.org/schema/qudt/numericValue> ?aw .
                ?yieldStrength <http://qudt.org/schema/qudt/numericValue> ?sigmaY .

                ?member nltl:frameSpacing ?spacing ;
                        nltl:requiredTransverseFrameShearArea ?requiredArea ;
                        nltl:selectedHullAreaFactor ?afNode .

                ?spacing <http://qudt.org/schema/qudt/numericValue> ?s .
                ?requiredArea <http://qudt.org/schema/qudt/numericValue> ?atReported .
                ?afNode <http://qudt.org/schema/qudt/numericValue> ?af .

                BIND(IF(?a <= ?b, ?a, ?b) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?s * (?af * ?ppf * ?pAvg) / (0.577 * ?sigmaY)) AS ?atExpected)

                FILTER (
                    ABS(?atReported - ?atExpected) >
                    IF(ABS(?atExpected) * 0.000001 > 0.000000001,
                       ABS(?atExpected) * 0.000001,
                       0.000000001)
                    ||
                    (?aw * 10000 < ?atExpected)
                )
            }
        """
    ] .
