<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qudtunit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-034-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:in ( nltl:flatBarSection nltl:profileSection ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify one permitted frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-034-LoadCaseShape>
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2-034-PlasticStrengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node <urn:nltl:generated-shape:I2-034-MidspanLoadQuantityShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable longitudinal side frame must provide plastic strength and a midspan plastic-collapse load whose magnitudes define the same strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX qudtunit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:plasticStrength ?plasticNode ;
                          nltl:midspanPlasticCollapseLoad ?midspanNode .
                    ?plasticNode qudt:numericValue ?plasticValue ;
                                 qudt:unit qudtunit:KiloN .
                    ?midspanNode qudt:numericValue ?midspanValue ;
                                  qudt:unit qudtunit:KiloN .
                    FILTER (
                        datatype(?plasticValue) = xsd:decimal &&
                        datatype(?midspanValue) = xsd:decimal &&
                        ?plasticValue = ?midspanValue
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear-and-bending demand of every represented load case must not exceed the applicable member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:plasticStrength ?plasticNode ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .
                ?plasticNode qudt:numericValue ?plasticValue .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandNode .
                ?demandNode qudt:numericValue ?demandValue .
                FILTER (
                    datatype(?plasticValue) = xsd:decimal &&
                    datatype(?demandValue) = xsd:decimal &&
                    ?demandValue > ?plasticValue
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-034-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node <urn:nltl:generated-shape:I2-034-DemandQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case must provide a combined shear-and-bending demand."
    ] .

<urn:nltl:generated-shape:I2-034-DemandQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-034-PlasticStrengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudtunit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-034-MidspanLoadQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudtunit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
