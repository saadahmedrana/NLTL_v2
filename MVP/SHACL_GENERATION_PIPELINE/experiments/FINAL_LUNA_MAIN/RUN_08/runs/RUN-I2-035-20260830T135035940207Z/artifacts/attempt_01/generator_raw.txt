gen:I2_035_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityM2Shape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "Each structural member required shear area must equal the declared formula within tolerance, and the actual shear area must be at least the calculated required value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure/qudt:numericValue ?p ;
                      nltl:frameSpan/qudt:numericValue ?a ;
                      nltl:loadPatchHeight/qudt:numericValue ?b ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:shearArea/qudt:numericValue ?aw ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:hasStructuralMember ?member .

                ?member nltl:frameSpacing/qudt:numericValue ?s ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?al .

                BIND (?b / ?s AS ?bprime)
                BIND (IF(?bprime < 2,
                         ?b * (1 - 0.25 * ?bprime),
                         ?s) AS ?b2)
                BIND ((1 - 0.3 / ?bprime) * ?b2 AS ?b1)
                BIND (100 * 100 * (?af * ?ppf * ?p) * 0.5 * ?b1 * ?a /
                      (0.577 * ?sigma) AS ?expected)

                FILTER (
                    ABS(?al - ?expected) >
                    (ABS(?expected) * 0.000001 + 0.000001)
                    ||
                    (?aw * 10000) < ?expected
                )
            }
        """
    ] .
