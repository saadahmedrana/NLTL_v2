<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

[
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Ice-strengthened structure status must be recorded as true for this applicability branch."
            ] ;
            sh:property [
                sh:path nltl:gaugedThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/M>
                    ]
                ]
            ] ;
            sh:property [
                sh:path ( nltl:hasPlating nltl:iceLoadRequiredNetPlateThickness ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:steelRenewalRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "Steel renewal must be required when gauged thickness is less than ice-load required net plate thickness plus 0.5 mm." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:gaugedThickness ?gaugedNode ;
                              nltl:hasPlating ?plating ;
                              nltl:steelRenewalRequired ?renewal .
                        ?gaugedNode qudt:numericValue ?gauged .
                        ?plating nltl:iceLoadRequiredNetPlateThickness ?tnetNode .
                        ?tnetNode qudt:numericValue ?tnet .
                        FILTER (
                            ?gauged * 1000 < ?tnet + 0.5 &&
                            ?renewal != true
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    )
] .
<END_SHACL>
