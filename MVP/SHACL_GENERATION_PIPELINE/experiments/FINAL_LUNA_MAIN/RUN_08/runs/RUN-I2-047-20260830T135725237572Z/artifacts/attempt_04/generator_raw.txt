<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

[
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:netThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For an ice-strengthened structure, steelRenewalRequired must be true exactly when gaugedThickness is less than netThickness plus 0.0005 m." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:gaugedThickness ?gaugedNode ;
                              nltl:netThickness ?netNode ;
                              nltl:steelRenewalRequired ?renewal .
                        ?gaugedNode qudt:numericValue ?gauged ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?netNode qudt:numericValue ?net ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (
                            (?gauged < (?net + "0.0005"^^xsd:decimal) && ?renewal != true) ||
                            (?gauged >= (?net + "0.0005"^^xsd:decimal) && ?renewal != false)
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:steelRenewalRequired ;
                sh:hasValue false
            ]
        ]
    )
] .
<END_SHACL>
