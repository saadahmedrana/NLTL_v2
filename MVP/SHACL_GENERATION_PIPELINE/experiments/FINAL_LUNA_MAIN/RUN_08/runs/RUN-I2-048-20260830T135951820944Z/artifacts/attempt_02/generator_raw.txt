@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-CaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-CaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade case must identify exactly one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:message "Each steel-grade case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:message "Each steel-grade case must specify one supported material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:message "Each steel-grade case must specify one supported steel strength category."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade case must specify one as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:message "Each steel-grade case must specify exactly one supported actual hull structural steel grade."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The case has no exact applicable Table 8 required grade for its Polar Class, thickness, material class, and strength category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  ?ship nltl:hasSteelGradeRequirementCase $this ;
        nltl:polarClass ?pc .
  $this nltl:asBuiltPlateThickness ?thickness ;
        nltl:steelMaterialClass ?material ;
        nltl:steelStrengthCategory ?category .
  ?thickness qudt:numericValue ?tv ;
            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
  FILTER (
    (
      ?tv > 35 && ?tv <= 40 &&
      ?material = nltl:steelMaterialClassThree &&
      ?category = nltl:normalStrengthSteelCategory &&
      ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)
    )
    ||
    (
      ?tv > 40 && ?tv <= 45 &&
      ?material = nltl:steelMaterialClassThree &&
      ?category = nltl:normalStrengthSteelCategory &&
      ?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)
    )
    ||
    (
      ?tv > 45 && ?tv <= 50 &&
      ?material = nltl:steelMaterialClassThree &&
      ?category = nltl:normalStrengthSteelCategory &&
      ?pc IN (
        nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
        nltl:polarClassPc4, nltl:polarClassPc5
      )
    )
  )
  FILTER NOT EXISTS {
    $this nltl:requiredHullStructuralSteelGrade ?required .
    FILTER (
      (
        ?required = nltl:steelGradeB &&
        ?category = nltl:normalStrengthSteelCategory &&
        ?material IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo) &&
        (
          (?tv > 0 && ?tv <= 10) ||
          (?tv > 10 && ?tv <= 15 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 15 && ?tv <= 20 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 20 && ?tv <= 25 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 25 && ?tv <= 35 && ?material = nltl:steelMaterialClassOne)
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6,
          nltl:polarClassPc7
        )
      )
      ||
      (
        ?required = nltl:steelGradeAh &&
        ?category = nltl:highTensileSteelCategory &&
        ?material IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo) &&
        (
          (?tv > 0 && ?tv <= 10) ||
          (?tv > 10 && ?tv <= 15 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 15 && ?tv <= 20 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 20 && ?tv <= 25 && ?material = nltl:steelMaterialClassOne) ||
          (?tv > 25 && ?tv <= 35 && ?material = nltl:steelMaterialClassOne)
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6,
          nltl:polarClassPc7
        )
      )
      ||
      (
        ?required = nltl:steelGradeD &&
        ?category = nltl:normalStrengthSteelCategory &&
        (
          (?material = nltl:steelMaterialClassOne && ?tv > 10 && ?tv <= 35) ||
          (?material = nltl:steelMaterialClassTwo && ?tv > 10 && ?tv <= 25) ||
          (?material = nltl:steelMaterialClassThree && ?tv > 35 && ?tv <= 45)
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5
        )
      )
      ||
      (
        ?required = nltl:steelGradeDh &&
        ?category = nltl:highTensileSteelCategory &&
        (
          (?material = nltl:steelMaterialClassOne && ?tv > 10 && ?tv <= 35) ||
          (?material = nltl:steelMaterialClassTwo && ?tv > 10 && ?tv <= 25) ||
          (?material = nltl:steelMaterialClassThree && ?tv > 35 && ?tv <= 45)
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5
        )
      )
      ||
      (
        ?required = nltl:steelGradeE &&
        ?category = nltl:normalStrengthSteelCategory &&
        (
          (?material = nltl:steelMaterialClassThree && ?tv > 0 && ?tv <= 35) ||
          (?material IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo) && ?tv > 35 && ?tv <= 50) ||
          (?material = nltl:steelMaterialClassThree && ?tv > 35 && ?tv <= 50 && ?pc IN (nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6, nltl:polarClassPc7))
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6,
          nltl:polarClassPc7
        )
      )
      ||
      (
        ?required = nltl:steelGradeEh &&
        ?category = nltl:highTensileSteelCategory &&
        (
          (?material = nltl:steelMaterialClassThree && ?tv > 0 && ?tv <= 35) ||
          (?material IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo) && ?tv > 35 && ?tv <= 50) ||
          (?material = nltl:steelMaterialClassThree && ?tv > 35 && ?tv <= 50 && ?pc IN (nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6, nltl:polarClassPc7))
        ) &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5, nltl:polarClassPc6,
          nltl:polarClassPc7
        )
      )
      ||
      (
        ?required = nltl:steelGradeFh &&
        ?category = nltl:highTensileSteelCategory &&
        ?material = nltl:steelMaterialClassThree &&
        ?tv > 35 && ?tv <= 50 &&
        ?pc IN (
          nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
          nltl:polarClassPc4, nltl:polarClassPc5
        )
      )
    )
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual hull structural steel grade does not have a rank at least as high as the Table 8 required grade within the same strength category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  ?ship nltl:hasSteelGradeRequirementCase $this .
  $this nltl:steelStrengthCategory ?category ;
        nltl:actualHullStructuralSteelGrade ?actual ;
        nltl:requiredHullStructuralSteelGrade ?required .
  ?actual nltl:steelGradeOrderRank ?actualRank .
  ?required nltl:steelGradeOrderRank ?requiredRank .
  FILTER (
    (
      ?category = nltl:normalStrengthSteelCategory &&
      ?actual IN (
        nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE
      ) &&
      ?required IN (
        nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE
      )
    )
    ||
    (
      ?category = nltl:highTensileSteelCategory &&
      ?actual IN (
        nltl:steelGradeAh, nltl:steelGradeDh,
        nltl:steelGradeEh, nltl:steelGradeFh
      ) &&
      ?required IN (
        nltl:steelGradeAh, nltl:steelGradeDh,
        nltl:steelGradeEh, nltl:steelGradeFh
      )
    )
  )
  FILTER (?actualRank < ?requiredRank)
}
"""
    ] .
