nltl:SteelGradeRequirementCaseShape
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual steel grade must be in the same strength family as the required grade and must have rank at least as high as the required grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                FILTER(
                    (
                        ?actual IN (
                            nltl:steelGradeAh,
                            nltl:steelGradeDh,
                            nltl:steelGradeEh,
                            nltl:steelGradeFh
                        )
                        &&
                        ?required IN (
                            nltl:steelGradeB,
                            nltl:steelGradeD,
                            nltl:steelGradeE
                        )
                    )
                    ||
                    (
                        ?actual IN (
                            nltl:steelGradeB,
                            nltl:steelGradeD,
                            nltl:steelGradeE
                        )
                        &&
                        ?required IN (
                            nltl:steelGradeAh,
                            nltl:steelGradeDh,
                            nltl:steelGradeEh,
                            nltl:steelGradeFh
                        )
                    )
                    ||
                    (
                        ?actual IN (
                            nltl:steelGradeAh,
                            nltl:steelGradeDh,
                            nltl:steelGradeEh,
                            nltl:steelGradeFh
                        )
                        &&
                        ?required IN (
                            nltl:steelGradeAh,
                            nltl:steelGradeDh,
                            nltl:steelGradeEh,
                            nltl:steelGradeFh
                        )
                        &&
                        ?actualRank < ?requiredRank
                    )
                    ||
                    (
                        ?actual IN (
                            nltl:steelGradeB,
                            nltl:steelGradeD,
                            nltl:steelGradeE
                        )
