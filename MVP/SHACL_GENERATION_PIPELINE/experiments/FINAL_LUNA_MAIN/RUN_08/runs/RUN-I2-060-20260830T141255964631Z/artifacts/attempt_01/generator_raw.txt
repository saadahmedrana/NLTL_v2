<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_060_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The ship must record that loads carried by members in way of cut-outs do not cause instability."
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_060_StructuralMemberShape
    ] .

gen:I2_060_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCutout ;
        sh:class nltl:cutout
    ] ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A structural member linked to a cut-out must be recorded as being in the way of that cut-out." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCutout ?cutout .
                FILTER NOT EXISTS {
                    $this nltl:memberInWayOfCutout true .
                }
                FILTER NOT EXISTS {
                    $this nltl:memberInWayOfCutout false .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Where stiffening is necessary for a member in way of a cut-out, the ship must record structural stiffening." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCutout ?cutout .
                $this nltl:stiffeningNecessary true .
                ?ship nltl:hasStructuralMember $this .
                FILTER NOT EXISTS {
                    ?ship nltl:structuralStiffening ?stiffening .
                    FILTER (datatype(?stiffening) = xsd:string)
                }
            }
        """
    ] .
<END_SHACL>
