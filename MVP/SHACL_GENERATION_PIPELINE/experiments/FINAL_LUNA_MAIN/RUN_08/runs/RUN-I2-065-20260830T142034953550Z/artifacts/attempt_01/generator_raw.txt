<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-065-ComplexReadiness>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each nonlinear calculation case must provide the declared acceptance evidence, assessment result, required structural quantities, and prescribed quantity units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case a nltl:calculationCase ;
                    nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER (
                    NOT EXISTS {
                        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                        ?evidence a nltl:evidenceArtifact .
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:minorPermanentDeformationAssessmentSatisfied ?assessment .
                        FILTER (DATATYPE(?assessment) = xsd:boolean)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:fractureMargin ?fracture .
                        ?fracture qudt:numericValue ?fractureValue ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (DATATYPE(?fractureValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?fracture qudt:unit ?fractureUnit .
                            FILTER (?fractureUnit != qudt:UNITLESS)
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:majorBucklingMargin ?buckling .
                        ?buckling qudt:numericValue ?bucklingValue ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (DATATYPE(?bucklingValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?buckling qudt:unit ?bucklingUnit .
                            FILTER (?bucklingUnit != qudt:UNITLESS)
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldingMargin ?yielding .
                        ?yielding qudt:numericValue ?yieldingValue ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (DATATYPE(?yieldingValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?yielding qudt:unit ?yieldingUnit .
                            FILTER (?yieldingUnit != qudt:UNITLESS)
                        }
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentLateralDeformation ?lateral .
                        ?lateral qudt:numericValue ?lateralValue ;
                            qudt:unit qudt:MilliM .
                        FILTER (DATATYPE(?lateralValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?lateral qudt:unit ?lateralUnit .
                            FILTER (?lateralUnit != qudt:MilliM)
                        }
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                        ?outOfPlane qudt:numericValue ?outOfPlaneValue ;
                            qudt:unit qudt:MilliM .
                        FILTER (DATATYPE(?outOfPlaneValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?outOfPlane qudt:unit ?outOfPlaneUnit .
                            FILTER (?outOfPlaneUnit != qudt:MilliM)
                        }
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:relevantStructuralDimension ?dimension .
                        ?dimension qudt:numericValue ?dimensionValue ;
                            qudt:unit qudt:MilliM .
                        FILTER (DATATYPE(?dimensionValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?dimension qudt:unit ?dimensionUnit .
                            FILTER (?dimensionUnit != qudt:MilliM)
                        }
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
