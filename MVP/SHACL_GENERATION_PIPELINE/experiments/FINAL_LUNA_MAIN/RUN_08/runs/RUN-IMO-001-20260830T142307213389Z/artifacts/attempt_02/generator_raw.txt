<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide a design ice condition to determine applicability of the Category A rule."
    ] ;
    sh:sparql [
        sh:message "When the design ice condition is at least medium first-year ice with possible old-ice inclusions, the ship category must be Category A." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER (
                    EXISTS {
                        $this nltl:designIceCondition
                            nltl:mediumFirstYearIceWithPossibleOldIceInclusions .
                        FILTER NOT EXISTS {
                            $this nltl:shipCategory nltl:polarShipCategoryA .
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
