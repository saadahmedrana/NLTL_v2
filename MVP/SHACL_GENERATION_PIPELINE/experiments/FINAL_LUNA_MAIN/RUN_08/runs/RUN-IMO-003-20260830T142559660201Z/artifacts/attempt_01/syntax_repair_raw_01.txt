<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-003-ShipDesignIceConditionCategory>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition to determine the applicable category rule."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:in (
                    nltl:openWaterIceCondition
                    nltl:iceConditionLessSevereThanCategoryAAndB
                )
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1 ;
                sh:message "A ship designed for open water or ice conditions less severe than categories A and B shall have ship category C."
            ]
        ]
        [
            sh:sparql [
                sh:message "The declared design ice condition is an applicable condition and therefore requires ship category C." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:designIceCondition ?condition .
                        FILTER (
                            ?condition IN (
                                nltl:openWaterIceCondition,
                                nltl:iceConditionLessSevereThanCategoryAAndB
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
