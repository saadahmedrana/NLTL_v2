<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_003_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_003_Branch1
        gen:IMO_003_Branch2
    ) .

gen:IMO_003_Branch1
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        sh:message "A design ice condition equal to an applicable open-water or less-severe condition requires ship category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
            }
        """
    ] .

gen:IMO_003_Branch2
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:openWaterIceCondition
            nltl:iceConditionLessSevereThanCategoryAAndB
        )
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:polarShipCategoryC
    ] .
<END_SHACL>
