<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_004_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When the ice is sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m, its classification must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType ?origin ;
                      nltl:winterGrowthCount ?growthCount ;
                      nltl:iceThickness ?thickness .

                ?thickness qudt:numericValue ?thicknessValue ;
                           qudt:unit unit:M .

                FILTER (
                    ?origin = nltl:seaIceOrigin &&
                    ?growthCount <= 1 &&
                    ?thicknessValue >= 0.3 &&
                    ?thicknessValue <= 2.0
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:iceTypeClassification nltl:firstYearIce .
                    }
                    ||
                    EXISTS {
                        $this nltl:iceTypeClassification ?reportedType .
                        FILTER (?reportedType != nltl:firstYearIce)
                    }
                )
            }
        """
    ] .
<END_SHACL>
