<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_005_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A ship may use the ice condition 'ice free waters' only when total ice present is false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (STR(?condition) = "ice free waters")
                FILTER NOT EXISTS {
                    $this nltl:totalIcePresent ?icePresent .
                    FILTER (?icePresent = false)
                }
            }
        """
    ] .
<END_SHACL>
