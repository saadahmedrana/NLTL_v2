<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-006_MediumFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice type classification is medium first-year ice, ice thickness must be exactly one quantity in the inclusive 0.7–1.2 m range." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                {
                    FILTER NOT EXISTS {
                        $this nltl:iceThickness ?thickness .
                        ?thickness a qudt:QuantityValue ;
                                   qudt:numericValue ?value ;
                                   qudt:unit unit:M .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value >= "0.7"^^xsd:decimal &&
                            ?value <= "1.2"^^xsd:decimal
                        )
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness1, ?thickness2 .
                    FILTER (?thickness1 != ?thickness2)
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    ?thickness qudt:numericValue ?value1, ?value2 .
                    FILTER (?value1 != ?value2)
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    ?thickness qudt:unit ?unit1, ?unit2 .
                    FILTER (?unit1 != ?unit2)
                }
            }
        """
    ] .
<END_SHACL>
