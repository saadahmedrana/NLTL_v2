<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008_ThinFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:class nltl:iceTypeValue ;
        sh:message "A ship must provide an ice type classification to determine applicability of the thin first-year ice thickness range."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For thin first-year ice, ice thickness must be a decimal quantity in metres between 0.3 m and 0.7 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:thinFirstYearIce .

                FILTER (
                    NOT EXISTS {
                        $this nltl:iceThickness ?quantity .
                        ?quantity qudt:numericValue ?value ;
                                  qudt:unit unit:M .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value >= "0.3"^^xsd:decimal &&
                            ?value <= "0.7"^^xsd:decimal
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?quantity .
                        FILTER NOT EXISTS {
                            ?quantity qudt:numericValue ?value ;
                                      qudt:unit unit:M .
                            FILTER (
                                datatype(?value) = xsd:decimal &&
                                ?value >= "0.3"^^xsd:decimal &&
                                ?value <= "0.7"^^xsd:decimal
                            )
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:iceThickness ?quantity .
                        ?quantity qudt:unit ?unit1, ?unit2 .
                        FILTER (?unit1 != ?unit2)
                    }
                )
            }
        """
    ] .
<END_SHACL>
