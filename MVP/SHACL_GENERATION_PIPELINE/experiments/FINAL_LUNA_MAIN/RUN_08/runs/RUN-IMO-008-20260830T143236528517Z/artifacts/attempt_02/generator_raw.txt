<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

<urn:nltl:generated-shape:IMO-008_ThinFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as thin first-year ice must have an ice thickness quantity from 0.3 m through 0.7 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:thinFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?quantity .
                    ?quantity a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:M .
                    FILTER (?value >= 0.3 && ?value <= 0.7)
                }
            }
        """
    ] .
<END_SHACL>
