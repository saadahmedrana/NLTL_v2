<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-017-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "A Polar Ship Certificate form is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node <urn:nltl:generated-shape:IMO-017-SolasScheduleShape> ;
        sh:message "A SOLAS certificate schedule with certificate date records is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node <urn:nltl:generated-shape:IMO-017-SupplementShape> ;
        sh:message "The Polar Ship Certificate shall include a supplement with required equipment records."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "polarCertificateSupplementPresent shall be true."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:sparql [
        sh:message "Polar Ship Certificate dates and SOLAS certificate schedule dates shall match by controlled date category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:polarCertificateValidityDates ?date
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:surveyDates ?date
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:endorsementDates ?date
                    }
                }
                UNION
                {
                    $this nltl:polarCertificateValidityDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
                UNION
                {
                    $this nltl:surveyDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
                UNION
                {
                    $this nltl:endorsementDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                nltl:certificateScheduleDate ?date
                    }
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-017-SolasScheduleShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node <urn:nltl:generated-shape:IMO-017-DateRecordShape> ;
        sh:message "The SOLAS certificate schedule shall contain certificate schedule date records."
    ] .

<urn:nltl:generated-shape:IMO-017-DateRecordShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall have a permitted controlled date category."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record shall contain an xsd:date value."
    ] .

<urn:nltl:generated-shape:IMO-017-SupplementShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Certificate supplement shall record required equipment."
    ] .
<END_SHACL>
