<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:ApprovalRecordABShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "Approval must be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:or (
            [
                sh:pattern "acceptable.*Organization" ;
                sh:flags "i"
            ]
            [
                sh:pattern "equivalent.*safety" ;
                sh:flags "i"
            ]
        ) ;
        sh:minCount 1 ;
        sh:message "The approval standard must be acceptable to the Organization or offer an equivalent level of safety."
    ] .

gen:ApprovalRecordCStrengthenedShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "Approval must be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:pattern "acceptable.*adequate" ;
        sh:flags "i" ;
        sh:minCount 1 ;
        sh:message "The approval standard must be acceptable and adequate for the operating ice type and concentration."
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The operating ice type must be recorded for an ice-strengthened Category C ship."
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The operating ice concentration must be recorded as a unitless quantity."
    ] .

gen:CategoryABApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node gen:ApprovalRecordABShape
    ] .

gen:CategoryCStrengthenedApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:hasValue true ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node gen:ApprovalRecordCStrengthenedShape
    ] .

gen:CategoryCNotStrengthenedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:hasValue false ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] .

gen:IMO032ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A, Category B, or Category C."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-strengthening status must be recorded to determine Category C applicability."
    ] ;
    sh:or (
        gen:CategoryABApplicableShape
        gen:CategoryCStrengthenedApplicableShape
        gen:CategoryCNotStrengthenedShape
    ) .
<END_SHACL>
