@prefix gen: <urn:nltl:generated-shape:> .

@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO038Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Damage center position relative to maximum breadth must be provided as a boolean."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "760.0"^^xsd:decimal ;
                sh:maxInclusive "760.0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Damage extents must satisfy the applicable piecewise formulas, and the selected vertical damage position must be between the keel and 1.20 times the upper ice waterline draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .

                ?lengthNode qudt:numericValue ?length ;
                            qudt:unit qudt:M .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit qudt:M .
                ?longitudinalNode qudt:numericValue ?longitudinal ;
                                qudt:unit qudt:M .
                ?verticalNode qudt:numericValue ?vertical ;
                              qudt:unit qudt:M .
                ?positionNode qudt:numericValue ?position ;
                              qudt:unit qudt:M .

                FILTER (
                    DATATYPE(?forward) = xsd:boolean &&
                    DATATYPE(?length) = xsd:decimal &&
                    DATATYPE(?draught) = xsd:decimal &&
                    DATATYPE(?longitudinal) = xsd:decimal &&
                    DATATYPE(?vertical) = xsd:decimal &&
                    DATATYPE(?position) = xsd:decimal
                )

                BIND(
                    IF(
                        ?forward = true,
                        0.045 * ?length,
                        0.015 * ?length
                    ) AS ?expectedLongitudinal
                )
                BIND(
                    IF(
                        0.20 * ?draught < ?expectedLongitudinal,
                        0.20 * ?draught,
                        ?expectedLongitudinal
                    ) AS ?expectedVertical
                )

                FILTER(
                    ABS(?longitudinal - ?expectedLongitudinal) >
                        (0.000001 * ABS(?expectedLongitudinal) + 0.000000001)
                    ||
                    ABS(?vertical - ?expectedVertical) >
                        (0.000001 * ABS(?expectedVertical) + 0.000000001)
                    ||
                    ?position < 0
                    ||
                    ?position > 1.20 * ?draught
                )
            }
        """
    ] .
