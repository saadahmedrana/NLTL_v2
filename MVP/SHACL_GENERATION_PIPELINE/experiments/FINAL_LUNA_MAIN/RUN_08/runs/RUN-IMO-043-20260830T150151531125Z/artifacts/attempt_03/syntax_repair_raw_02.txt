<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltlgen:ShipMachineryInstallationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:class nltl:machineryInstallationItem ;
        sh:node nltlgen:MachineryInstallationShape
    ] .

nltlgen:MachineryInstallationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceAccretionHazard
            nltl:snowAccumulationHazard
            nltl:seawaterIceIngestionHazard
            nltl:liquidFreezingOrViscosityIncreaseHazard
            nltl:seawaterIntakeTemperatureHazard
            nltl:snowIngestionHazard
        )
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable hazard on a machinery installation must have protection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasApplicableHazard ?hazard .
                FILTER (
                    ?hazard IN (
                        nltl:iceAccretionHazard,
                        nltl:snowAccumulationHazard,
                        nltl:seawaterIceIngestionHazard,
                        nltl:liquidFreezingOrViscosityIncreaseHazard,
                        nltl:seawaterIntakeTemperatureHazard,
                        nltl:snowIngestionHazard
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:hasHazardProtectionEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] .
</END_SHACL>
