<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-047-Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship with an essential engine, combustion-air temperature control must be present and the combustion-air temperature must be within the manufacturer minimum and maximum limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?essentialEngine .
                {
                    FILTER NOT EXISTS {
                        $this nltl:combustionAirTemperatureControlPresent ?control .
                        FILTER (?control = "true"^^xsd:boolean)
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:combustionAirTemperature ?temperatureNode .
                        ?temperatureNode qudt:numericValue ?temperatureValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                        FILTER (DATATYPE(?temperatureValue) = xsd:decimal)

                        $this nltl:manufacturerMinAirTemperature ?minimumNode .
                        ?minimumNode qudt:numericValue ?minimumValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                        FILTER (DATATYPE(?minimumValue) = xsd:decimal)

                        $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                        ?maximumNode qudt:numericValue ?maximumValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                        FILTER (DATATYPE(?maximumValue) = xsd:decimal)

                        FILTER (
                            ?minimumValue <= ?temperatureValue &&
                            ?temperatureValue <= ?maximumValue
                        )
                    }
                }
            }
        """ ;
    ] .
<END_SHACL>
