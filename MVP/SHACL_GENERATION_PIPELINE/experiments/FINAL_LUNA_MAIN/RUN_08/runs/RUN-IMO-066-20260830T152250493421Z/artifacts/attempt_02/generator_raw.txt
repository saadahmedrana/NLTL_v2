<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_066_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide a single boolean ice-accretion applicability assessment."
    ] ;
    sh:or (
        gen:IMO_066_NotApplicableShape
        gen:IMO_066_ApplicableShape
    ) .

gen:IMO_066_NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:hasValue false
    ] .

gen:IMO_066_ApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "For an ice-accretion-exposed ship, every linked protected item in the six specified categories must have associated ice or snow removal or prevention means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:shipExposedToIceAccretion true .
                $this nltl:hasProtectedItem ?protectedItem .
                ?protectedItem nltl:protectedItemCategory ?category .
                FILTER (
                    ?category IN (
                        nltl:escapeRouteProtectedItem,
                        nltl:musterStationProtectedItem,
                        nltl:embarkationAreaProtectedItem,
                        nltl:survivalCraftProtectedItem,
                        nltl:launchingApplianceProtectedItem,
                        nltl:survivalCraftAccessProtectedItem
                    )
                )
                FILTER NOT EXISTS {
                    ?protectedItem nltl:hasIceSnowRemovalOrPreventionMeans ?means .
                }
            }
        """
    ] .
<END_SHACL>
