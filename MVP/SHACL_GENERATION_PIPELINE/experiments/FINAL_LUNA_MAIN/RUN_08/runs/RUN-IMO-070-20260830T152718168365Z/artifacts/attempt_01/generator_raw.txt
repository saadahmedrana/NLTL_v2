<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO070PassengerShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:passengerShip ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A passenger ship must report compatible sizing for the assigned personal survival equipment."
    ] ;
    sh:property [
        sh:path nltl:hasPersonOnBoardMember ;
        sh:node gen:IMO070PersonOnBoardMemberShape
    ] .

gen:IMO070PersonOnBoardMemberShape
    a sh:NodeShape ;
    sh:class nltl:personOnBoardMember ;
    sh:property [
        sh:path nltl:hasPersonalSurvivalEquipmentAssignment ;
        sh:minCount 1 ;
        sh:node gen:IMO070AssignmentShape ;
        sh:message "Each person on board must have an immersion suit or thermal protective aid assignment."
    ] .

gen:IMO070AssignmentShape
    a sh:NodeShape ;
    sh:class nltl:personalSurvivalEquipmentAssignment ;
    sh:property [
        sh:path nltl:assignedPersonalSurvivalEquipment ;
        sh:minCount 1 ;
        sh:node gen:IMO070EquipmentShape ;
        sh:message "Each personal survival equipment assignment must identify an immersion suit or thermal protective aid."
    ] .

gen:IMO070EquipmentShape
    a sh:NodeShape ;
    sh:class nltl:personalSurvivalEquipment ;
    sh:or (
        [
            sh:class nltl:immersionSuitEquipment
        ]
        [
            sh:class nltl:thermalProtectiveAidEquipment
        ]
    ) ;
    sh:message "Assigned personal survival equipment must be an immersion suit or a thermal protective aid."
.
<END_SHACL>
