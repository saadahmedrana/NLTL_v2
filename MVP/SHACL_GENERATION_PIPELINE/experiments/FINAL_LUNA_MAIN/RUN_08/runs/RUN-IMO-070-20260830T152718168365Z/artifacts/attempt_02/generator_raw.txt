<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO-070 a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each person on board of a passenger ship, exactly one personal survival equipment assignment identifying exactly one immersion suit or thermal protective aid is required, and size compatibility must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    EXISTS { $this rdf:type nltl:passengerShip }
                    ||
                    EXISTS {
                        $this nltl:shipType "passenger_ship"^^xsd:string
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:sizeCompatibilityStatus true
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        FILTER NOT EXISTS {
                            ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment1 .
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment2 .
                        FILTER (?assignment1 != ?assignment2)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                        FILTER NOT EXISTS {
                            ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                        ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment1 .
                        ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment2 .
                        FILTER (?equipment1 != ?equipment2)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                        ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment .
                        FILTER NOT EXISTS {
                            {
                                ?equipment rdf:type nltl:immersionSuitEquipment
                            }
                            UNION
                            {
                                ?equipment rdf:type nltl:thermalProtectiveAidEquipment
                            }
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
