<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltlgen:IMO_073_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboat ;
        sh:node nltlgen:IMO_073_LifeboatShape ;
        sh:message "Each linked lifeboat must have a permitted lifeboat type classification."
    ] .

nltlgen:IMO_073_LifeboatShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:lifeboatTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboatTypeValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:partiallyEnclosedLifeboat
            nltl:totallyEnclosedLifeboat
        ) ;
        sh:message "A lifeboat type must be partially enclosed or totally enclosed."
    ] .
<END_SHACL>
