<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO083ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO083ApplicableBranch
        gen:IMO083NotApplicableBranch
    ) .

gen:IMO083ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Ice-accretion applicability must be recorded as true or false."
    ] ;
    sh:property [
        sh:path nltl:hasRequiredNavigationOrCommunicationAntenna ;
        sh:minCount 1 ;
        sh:node gen:IMO083AntennaShape ;
        sh:message "When ice accretion is likely, at least one required navigation or communication antenna must be identified and each such antenna must have ice-accumulation prevention."
    ] .

gen:IMO083NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Ice-accretion applicability must be recorded as true or false."
    ] .

gen:IMO083AntennaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:antennaIceAccumulationPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each required navigation or communication antenna must have ice-accumulation prevention means."
    ] .
<END_SHACL>
