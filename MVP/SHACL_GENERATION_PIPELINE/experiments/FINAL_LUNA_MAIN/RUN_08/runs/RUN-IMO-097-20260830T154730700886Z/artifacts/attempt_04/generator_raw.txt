<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-097-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating applicability must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each lifeboat reached through the declared lifeboat path must record whether it is released when the ship operates in low air temperature." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipOperatesInLowAirTemperature true .
  {
    $this nltl:hasLifeboat ?craft .
  }
  UNION
  {
    $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
  }
  FILTER (
    NOT EXISTS {
      ?craft nltl:releasedRescueBoatOrLifeboat true .
    }
    &&
    NOT EXISTS {
      ?craft nltl:releasedRescueBoatOrLifeboat false .
    }
  )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every released rescue boat or lifeboat must have at least one ship-to-shore distress-alert device." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipOperatesInLowAirTemperature true .
  {
    $this nltl:hasLifeboat ?craft .
  }
  UNION
  {
    $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
  }
  ?craft nltl:releasedRescueBoatOrLifeboat true .
  FILTER NOT EXISTS {
    ?craft nltl:hasDistressAlertDevice ?device .
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every survival craft reached through the declared ship-to-survival-craft path must have the required communication devices." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:shipOperatesInLowAirTemperature true .
  $this nltl:hasSurvivalCraft ?craft .
  FILTER (
    NOT EXISTS {
      ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?distressDevice .
    }
    ||
    NOT EXISTS {
      ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
    }
    ||
    NOT EXISTS {
      ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
    }
  )
}
"""
    ] .
</BEGIN_SHACL>
