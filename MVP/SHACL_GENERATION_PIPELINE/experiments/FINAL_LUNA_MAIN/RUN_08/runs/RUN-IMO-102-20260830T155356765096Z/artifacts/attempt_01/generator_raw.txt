<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_102_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "The ship shall have exactly one controlled ice condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:message "The ship shall have exactly one controlled polar training ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO_102_CrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable crew member shall have a training record whose required level matches the recorded level and whose STCW qualification is valid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role .
                BIND(
                    IF(
                        ?ice = nltl:iceFreeIceCondition,
                        nltl:polarTrainingNotApplicable,
                        IF(
                            ?ice = nltl:openWaterIceCondition &&
                            ?shipType = nltl:otherShipPolarTrainingType,
                            nltl:polarTrainingNotApplicable,
                            IF(
                                ?ice = nltl:openWaterIceCondition,
                                nltl:basicPolarTrainingLevel,
                                IF(
                                    ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole,
                                    nltl:basicPolarTrainingLevel,
                                    nltl:advancedPolarTrainingLevel
                                )
                            )
                        )
                    )
                    AS ?required
                )
                FILTER (?required != nltl:polarTrainingNotApplicable)
                FILTER NOT EXISTS {
                    ?crew (
                        nltl:hasCrewTrainingRecord |
                        nltl:hasPolarTrainingRecord
                    ) ?record .
                    ?record nltl:requiredTrainingLevel ?required ;
                            nltl:trainingRecordLevel ?recordedLevel ;
                            nltl:stcwQualificationValid true .
                    FILTER (?recordedLevel = ?required)
                }
            }
        """
    ] .

gen:IMO_102_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:message "Each linked crew member shall have exactly one controlled polar training role."
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:node gen:IMO_102_TrainingRecordShape
    ] ;
    sh:property [
        sh:path nltl:hasPolarTrainingRecord ;
        sh:node gen:IMO_102_TrainingRecordShape
    ] .

gen:IMO_102_TrainingRecordShape
    a sh:NodeShape ;
    sh:class nltl:polarTrainingRecord ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:minCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A polar training record shall state a controlled required training level."
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:minCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A polar training record shall state a controlled recorded training level."
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable polar training record shall have a valid STCW qualification."
    ] .
<END_SHACL>
