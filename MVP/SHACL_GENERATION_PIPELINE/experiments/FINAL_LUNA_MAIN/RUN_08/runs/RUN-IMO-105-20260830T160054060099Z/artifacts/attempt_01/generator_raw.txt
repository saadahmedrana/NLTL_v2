<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_105_Arctic_Oil_Discharge_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must provide an operating area to determine applicability of the Arctic-waters discharge prohibition."
    ] ;
    sh:property [
        sh:path nltl:oilOrOilyMixtureDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:message "The oil or oily mixture discharge record must be boolean."
    ] ;
    sh:sparql [
        sh:message "In Arctic waters, discharge of oil or oily mixtures to the sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:operatingArea ?area ;
                      nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                FILTER (STR(?area) = "Arctic_waters")
                FILTER (?discharge = "false"^^xsd:boolean)
                FILTER NOT EXISTS {
                    ?this nltl:oilOrOilyMixtureDischargeToSea ?otherDischarge .
                    FILTER (?otherDischarge != "false"^^xsd:boolean)
                }
            }
        """
    ] .
<END_SHACL>
