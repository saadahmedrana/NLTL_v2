<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_105_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine applicability of the Arctic waters discharge prohibition."
    ] ;
    sh:property [
        sh:path nltl:oilOrOilyMixtureDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:message "Oil or oily mixture discharge to sea must be recorded as a boolean when represented."
    ] ;
    sh:sparql [
        sh:message "In Arctic waters, oil or oily mixture discharge to sea must be recorded and must be false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:operatingArea ?area .
    FILTER (STR(?area) = "Arctic_waters")
    FILTER (
        NOT EXISTS {
            $this nltl:oilOrOilyMixtureDischargeToSea ?discharge
        }
        ||
        EXISTS {
            $this nltl:oilOrOilyMixtureDischargeToSea ?dischargeValue .
            FILTER (?dischargeValue != false)
        }
    )
}
"""
    ] .
<END_SHACL>
