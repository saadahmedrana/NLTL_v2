<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_106_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For clean or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition applicability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable false .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?reported .
                    FILTER (?reported != false)
                }
            }
        """
    ] .
<END_SHACL>
