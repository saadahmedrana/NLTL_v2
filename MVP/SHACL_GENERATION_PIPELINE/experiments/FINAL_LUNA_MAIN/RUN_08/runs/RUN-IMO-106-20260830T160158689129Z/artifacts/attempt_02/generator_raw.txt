<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO_106_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:cleanBallastMaterial
            nltl:segregatedBallastMaterial
        ) ;
        sh:message "The discharged material classification must be clean ballast or segregated ballast."
    ] ;
    sh:sparql [
        sh:message "For clean or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition applicability must be explicitly false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?classification .
                FILTER (
                    ?classification = nltl:cleanBallastMaterial ||
                    ?classification = nltl:segregatedBallastMaterial
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable false
                }
            }
        """
    ] .
<END_SHACL>
