<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_106_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean or segregated ballast discharge, the oil-discharge prohibition applicability must be explicitly typed boolean false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:dischargedMaterialClassification ?material .
    FILTER (
        ?material = nltl:cleanBallastMaterial ||
        ?material = nltl:segregatedBallastMaterial
    )
    FILTER (
        NOT EXISTS {
            $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?validFalse .
            FILTER (
                datatype(?validFalse) = xsd:boolean &&
                ?validFalse = false
            )
        }
        ||
        EXISTS {
            $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?app .
            FILTER (
                !(datatype(?app) = xsd:boolean && ?app = false)
            )
        }
    )
}
""" ;
    ] .
<END_SHACL>
