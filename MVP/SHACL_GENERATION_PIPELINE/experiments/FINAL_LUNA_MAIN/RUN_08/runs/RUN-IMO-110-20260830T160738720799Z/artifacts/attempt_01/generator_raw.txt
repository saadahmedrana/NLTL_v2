<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable category A or B oil tanker constructed on or after 2017-01-01 with deadweight below 5,000 metric tonnes, both required cargo-tank protection statuses must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this rdf:type nltl:oilTanker ;
                    nltl:shipCategory ?category ;
                    nltl:shipType ?shipType ;
                    nltl:deadweight ?deadweight ;
                    nltl:constructionDate ?constructionDate .

                ?deadweight qudt:numericValue ?mass ;
                    qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?shipType = "oil_tanker")
                FILTER (
                    datatype(?mass) = xsd:decimal &&
                    ?mass < "5000"^^xsd:decimal
                )
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:wingTankProtectionStatus true
                    }
                )
            }
        """
    ] .
<END_SHACL>
