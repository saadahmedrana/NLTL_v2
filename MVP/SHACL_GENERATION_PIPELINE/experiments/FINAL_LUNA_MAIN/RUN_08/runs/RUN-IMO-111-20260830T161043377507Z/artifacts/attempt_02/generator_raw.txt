<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A, B, or C."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:IMO-111-NonApplicableShape>
        <urn:nltl:generated-shape:IMO-111-ApplicableShape>
    ) .

<urn:nltl:generated-shape:IMO-111-NonApplicableShape>
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A Category A or B ship constructed before 1 January 2017 is outside this requirement."
            ]
        ]
    ) .

<urn:nltl:generated-shape:IMO-111-ApplicableShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A Category A or B ship within this requirement must have a construction date on or after 1 January 2017."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented oil residue or oily bilge water holding tank must have a valid individual capacity in cubic metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasComponent ?tank .
  ?tank a ?tankClass .
  FILTER (?tankClass IN (nltl:oilResidueTank, nltl:oilyBilgeWaterHoldingTank))
  FILTER NOT EXISTS {
    ?tank nltl:individualTankCapacity ?capacityNode .
    ?capacityNode qudt:numericValue ?capacityValue ;
                  qudt:unit unit:M .
    FILTER (datatype(?capacityValue) = xsd:decimal)
  }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every represented applicable tank with an individual capacity greater than 30 m3 must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasComponent ?tank .
  ?tank a ?tankClass .
  FILTER (?tankClass IN (nltl:oilResidueTank, nltl:oilyBilgeWaterHoldingTank))
  ?tank nltl:individualTankCapacity ?capacityNode .
  ?capacityNode qudt:numericValue ?capacityValue ;
                qudt:unit unit:M .
  FILTER (datatype(?capacityValue) = xsd:decimal)
  FILTER (?capacityValue > 30)
  FILTER NOT EXISTS {
    ?tank nltl:tankOuterShellSeparation ?separationNode .
    ?separationNode qudt:numericValue ?separationValue ;
                    qudt:unit unit:M .
    FILTER (datatype(?separationValue) = xsd:decimal)
    FILTER (?separationValue >= 0.76)
  }
}
"""
    ] .
<END_SHACL>
