<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-115> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:comminutedAndDisinfected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage is comminuted and disinfected, MARPOL compliance, the distance exceeding 3 nautical miles, and the recorded distance from areas exceeding one tenth ice concentration are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType .
                $this nltl:comminutedAndDisinfected ?comminuted .
                FILTER(?dischargeType = "comminuted_and_disinfected")
                FILTER(?comminuted = true)

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus ?status ;
                          nltl:distanceToNearestIceShelfOrFastIce ?nearestDistance ;
                          nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance ;
                          nltl:nearbyIceConcentration ?nearbyConcentration ;
                          nltl:iceConcentration ?concentration .

                    FILTER(?status = true)

                    ?nearestDistance a qudt:QuantityValue ;
                                     qudt:numericValue ?nearestValue ;
                                     qudt:unit unit:MI_N .
                    FILTER(datatype(?nearestValue) = xsd:decimal)
                    FILTER(?nearestValue > 3)

                    ?areaDistance a qudt:QuantityValue ;
                                  qudt:numericValue ?areaValue ;
                                  qudt:unit unit:MI_N .
                    FILTER(datatype(?areaValue) = xsd:decimal)

                    ?nearbyConcentration a qudt:QuantityValue ;
                                         qudt:numericValue ?nearbyValue ;
                                         qudt:unit unit:UNITLESS .
                    FILTER(datatype(?nearbyValue) = xsd:decimal)
                    FILTER(?nearbyValue > 1)

                    ?concentration a qudt:QuantityValue ;
                                   qudt:numericValue ?concentrationValue ;
                                   qudt:unit unit:UNITLESS .
                    FILTER(datatype(?concentrationValue) = xsd:decimal)
                    FILTER(?concentrationValue > 1)
                }
            }
        """
    ] .
<END_SHACL>
