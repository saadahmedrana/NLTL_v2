@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:DischargeDistanceRecordShape
    a sh:NodeShape ;
    sh:class nltl:dischargeDistanceRecords ;
    sh:property [
        sh:path nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MI_N> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable discharge-distance record must record the distance to an area with ice concentration greater than one tenth in nautical miles."
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MI_N> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable discharge-distance record must record the distance to the nearest land, ice shelf, or fast ice in nautical miles."
    ] .

nltlgen:ApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:sewageDischargeUsesTreatmentPlant ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:sewageTreatmentPlantApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must have an approved sewage treatment plant."
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexIvRegulation11Point1Point2DischargeComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable ship must record compliance with MARPOL Annex IV regulation 11.1.2."
    ] ;
    sh:property [
        sh:path nltl:hasDischargeDistanceRecord ;
        sh:class nltl:dischargeDistanceRecords ;
        sh:node nltlgen:DischargeDistanceRecordShape ;
        sh:minCount 1 ;
        sh:message "An applicable ship must have a discharge-distance record containing both required distances."
    ] .

nltlgen:NonApplicableShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:sewageDischargeUsesTreatmentPlant ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

nltlgen:IMO117ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltlgen:ApplicableShipBranch
        nltlgen:NonApplicableShipBranch
    ) .
