<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-124> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Antarctic-area selector is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The MARPOL Annex V regulation 6.1 discharge status is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "For an Antarctic-area MARPOL Annex V regulation 6.1 discharge, the nearest-fast-ice distance must be at least 12 nautical miles, food waste must not be discharged onto ice, and a paired qualifying ice-concentration observation and distance must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?operatingArea ;
                       nltl:antarcticArea ?antarcticArea ;
                       nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .
                FILTER(?operatingArea = ?antarcticArea)
                FILTER(?discharge = true)
                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestFastIce ?fastIceDistance ;
                          nltl:foodWasteDischargeOntoIce ?foodWaste ;
                          nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance ;
                          nltl:iceConcentration ?iceConcentration ;
                          nltl:nearbyIceConcentration ?nearbyIceConcentration .

                    ?fastIceDistance a qudt:QuantityValue ;
                                     qudt:numericValue ?fastValue ;
                                     qudt:unit unit:MI_N .
                    FILTER(datatype(?fastValue) = xsd:decimal)
                    FILTER(?fastValue >= "12"^^xsd:decimal)

                    FILTER(datatype(?foodWaste) = xsd:boolean)
                    FILTER(?foodWaste = false)

                    ?areaDistance a qudt:QuantityValue ;
                                  qudt:numericValue ?areaDistanceValue ;
                                  qudt:unit unit:MI_N .
                    FILTER(datatype(?areaDistanceValue) = xsd:decimal)

                    ?iceConcentration a qudt:QuantityValue ;
                                      qudt:numericValue ?iceValue ;
                                      qudt:unit unit:UNITLESS .
                    FILTER(datatype(?iceValue) = xsd:decimal)
                    FILTER(?iceValue > "0.1"^^xsd:decimal)

                    ?nearbyIceConcentration a qudt:QuantityValue ;
                                             qudt:numericValue ?nearbyValue ;
                                             qudt:unit unit:UNITLESS .
                    FILTER(datatype(?nearbyValue) = xsd:decimal)
                    FILTER(?nearbyValue > "0.1"^^xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
