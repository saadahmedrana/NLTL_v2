<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_124_Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:or (
        <urn:nltl:generated-shape:IMO_124_ApplicableShape>
        <urn:nltl:generated-shape:IMO_124_NonApplicableShape>
    ) .

<urn:nltl:generated-shape:IMO_124_ApplicableShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:distanceToNearestFastIce ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "12"^^xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive "0.1"^^xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeOntoIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:sparql [
        sh:message "When the Antarctic-area MARPOL Annex V regulation 6.1 discharge condition applies, a distance and a qualifying ice-concentration observation must be represented on the same ship context." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?operatingArea ;
                      nltl:antarcticArea ?antarcticArea ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge ;
                      nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?distanceQuantity ;
                      nltl:iceConcentration ?concentrationQuantity .
                FILTER (
                    STR(?operatingArea) = STR(?antarcticArea)
                    && ?discharge = true
                )
                ?distanceQuantity qudt:numericValue ?distanceValue .
                ?concentrationQuantity qudt:numericValue ?concentrationValue .
                FILTER (?concentrationValue > "0.1"^^xsd:decimal)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The applicable Antarctic-area discharge condition must be established from the supplied area and discharge inputs." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?operatingArea ;
                      nltl:antarcticArea ?antarcticArea ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .
                FILTER (
                    !(STR(?operatingArea) = STR(?antarcticArea)
                    && ?discharge = true)
                )
            }
        """
    ] .

                )
            }
        """
    ] .
<END_SHACL>
