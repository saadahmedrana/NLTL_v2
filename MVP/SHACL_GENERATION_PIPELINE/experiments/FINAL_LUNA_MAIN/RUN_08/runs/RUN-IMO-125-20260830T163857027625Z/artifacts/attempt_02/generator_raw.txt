<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-125> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The polar-operation applicability value must be provided." ;
    ] ;
    sh:sparql [
        sh:message "When the ship operates in polar waters, the Garbage Record Book, Garbage Management Plan, and garbage placards must include polar-operation coverage." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:garbageRecordBookPolarCoverage true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:garbageManagementPlanPolarCoverage true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:garbagePlacardPolarCoverage true
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
