<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-002> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Chapter 9-1 applicability shall be recorded as a boolean." ;
    ] ;
    sh:sparql [
        sh:message "When Chapter 9-1 is marked applicable, the ship shall satisfy one permitted applicability branch: fishing vessel of at least 24 m length overall, pleasure yacht of at least 300 gross tonnage not engaged in trade, or cargo ship of at least 300 and below 500 gross tonnage." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter9Dash1Applicable ?app .
                FILTER (datatype(?app) = xsd:boolean)
                FILTER (?app = true)

                OPTIONAL {
                    $this nltl:shipType ?type .
                    FILTER (datatype(?type) = xsd:string)
                }

                OPTIONAL {
                    $this nltl:lengthOverall ?lengthNode .
                    ?lengthNode qudt:numericValue ?length ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?length) = xsd:decimal)
                }

                OPTIONAL {
                    $this nltl:grossTonnage ?grossNode .
                    ?grossNode qudt:numericValue ?gross ;
                               qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?gross) = xsd:decimal)
                }

                OPTIONAL {
                    $this nltl:engagedInTrade ?trade .
                    FILTER (datatype(?trade) = xsd:boolean)
                }

                FILTER (
                    !(
                        (
                            ?type = "fishing_vessel"
                            && bound(?length)
                            && ?length >= 24
                        )
                        ||
                        (
                            ?type = "pleasure_yacht"
                            && bound(?gross)
                            && ?gross >= 300
                            && bound(?trade)
                            && ?trade = false
                        )
                        ||
                        (
                            ?type = "cargo_ship"
                            && bound(?gross)
                            && ?gross >= 300
                            && ?gross < 500
                        )
                    )
                )
            }
        """ ;
    ] .
<END_SHACL>
