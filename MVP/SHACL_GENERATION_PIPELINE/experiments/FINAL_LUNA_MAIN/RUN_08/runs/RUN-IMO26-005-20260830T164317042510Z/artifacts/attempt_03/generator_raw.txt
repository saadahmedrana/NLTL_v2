<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_005_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:headingOrPositionSystem ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must represent at least one heading-reference or position-fixing system." ;
    ] ;
    sh:property [
        sh:path nltl:intendedArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must declare its intended area." ;
    ] ;
    sh:property [
        sh:path nltl:intendedAreaSuitabilityApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState ;
        sh:in ( nltl:evidenceStateApproved ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The intended-area suitability approval status must be Approved." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every represented heading-reference or position-fixing system must be covered by an Approved suitability status for the ship's declared intended area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:headingOrPositionSystem ?system .
                FILTER NOT EXISTS {
                    $this nltl:intendedArea ?area ;
                          nltl:intendedAreaSuitabilityApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """ ;
    ] .
<END_SHACL>
