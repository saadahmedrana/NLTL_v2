<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-012-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:IMO26-012-ApplicableShape>
        <urn:nltl:generated-shape:IMO26-012-NonApplicableShape>
    ) .

<urn:nltl:generated-shape:IMO26-012-ApplicableShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 500
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Applicable ships shall have a gross tonnage of at least 500."
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node <urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    ] ;
    sh:sparql [
        sh:message "An applicable ship shall have at least two distinct non-magnetic heading means, and each means shall be independent from another heading means of the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?means1 .
                        $this nltl:hasNonMagneticHeadingMeans ?means2 .
                        FILTER (?means1 != ?means2)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasNonMagneticHeadingMeans ?means .
                        FILTER NOT EXISTS {
                            ?means nltl:independentFromHeadingMeans ?other .
                            FILTER (?other != ?means)
                            $this nltl:hasNonMagneticHeadingMeans ?other .
                        }
                    }
                )
            }
        """
    ] .

<urn:nltl:generated-shape:IMO26-012-NonApplicableShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive 500
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ships below 500 gross tonnage are outside this requirement's applicability."
    ] .

<urn:nltl:generated-shape:IMO26-012-HeadingMeansShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to the main power source."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to the emergency power source."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:message "Each non-magnetic heading means shall identify an independent heading means."
    ] .
<END_SHACL>
