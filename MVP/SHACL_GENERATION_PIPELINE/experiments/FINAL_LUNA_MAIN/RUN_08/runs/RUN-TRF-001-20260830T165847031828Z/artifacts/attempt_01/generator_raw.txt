<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF001Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2021 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulatoryProvisionValue ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021 ;
        sh:minCount 1 ;
        sh:message "The 2021 section 1.8 Ice Classes provision must apply."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulatoryProvisionValue ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021 ;
        sh:minCount 1 ;
        sh:message "The 2021 chapter 2 Ice Class Draught provision must apply."
    ] ;
    sh:sparql [
        sh:message "Ships contracted for construction on or after 5 July 2021 must identify the 2021 Ice Class Regulations as the applicable edition." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionContractDate ?date .
    FILTER (?date >= "2021-07-05"^^xsd:date)
    FILTER (
        NOT EXISTS {
            $this nltl:applicableIceClassRegulationEdition nltl:iceClassRegulationEdition2021 .
        }
        ||
        EXISTS {
            $this nltl:applicableIceClassRegulationEdition ?edition .
            FILTER (?edition != nltl:iceClassRegulationEdition2021)
        }
    )
}
"""
    ] .
<END_SHACL>
