<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-003> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "A supplied construction contract date must be an xsd:date."
    ] ;
    sh:sparql [
        sh:message "Ships contracted for construction on or after 1 January 2012 and before 1 January 2019 must apply the 2010 Ice Class Regulations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (datatype(?date) = xsd:date)
                FILTER (
                    YEAR(?date) > 2012 ||
                    (YEAR(?date) = 2012 && MONTH(?date) > 1) ||
                    (YEAR(?date) = 2012 && MONTH(?date) = 1 && DAY(?date) >= 1)
                )
                FILTER (
                    YEAR(?date) < 2019 ||
                    (YEAR(?date) = 2019 && MONTH(?date) < 1)
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2010 .
                }
            }
        """
    ] .
<END_SHACL>
