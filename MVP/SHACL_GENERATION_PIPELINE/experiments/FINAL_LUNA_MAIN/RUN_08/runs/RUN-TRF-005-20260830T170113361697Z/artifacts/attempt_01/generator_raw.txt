<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2002 Ice Class Regulations apply."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine whether the 2002 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships constructed at the applicable stage on or after 1 September 2003 and contracted before 1 January 2010 must identify the 2002 Ice Class Regulations." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionStageDate ?stageDate ;
          nltl:constructionContractDate ?contractDate .
    FILTER (
        YEAR(?stageDate) > 2003 ||
        (YEAR(?stageDate) = 2003 && MONTH(?stageDate) > 9) ||
        (YEAR(?stageDate) = 2003 && MONTH(?stageDate) = 9 && DAY(?stageDate) >= 1)
    )
    FILTER (
        YEAR(?contractDate) < 2010 ||
        (YEAR(?contractDate) = 2010 && MONTH(?contractDate) = 1 && DAY(?contractDate) < 1)
    )
    FILTER (
        NOT EXISTS {
            $this nltl:applicableIceClassRegulationEdition
                nltl:iceClassRegulationEdition2002
        }
        ||
        EXISTS {
            $this nltl:applicableIceClassRegulationEdition ?edition .
            FILTER (?edition != nltl:iceClassRegulationEdition2002)
        }
    )
}
"""
    ] .
<END_SHACL>
