<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify the ice waterline or waterlines at which it is intended to operate in ice."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify an upper ice waterline."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must equal, in both directions, the envelope formed by the maximum vertical coordinate of the intended ice-operating waterline points at each longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

{
  SELECT DISTINCT ?this ?x (MAX(?intVertical) AS ?maximumIntendedVertical)
  WHERE {
    ?this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
    ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedLongitudinal .
    ?intendedLongitudinal qudt:numericValue ?x .
    ?intendedPoint nltl:verticalCoordinate ?intendedVertical .
    ?intendedVertical qudt:numericValue ?intVertical .
  }
  GROUP BY ?this ?x
}
{
  ?this nltl:hasUpperIceWaterline ?upperWaterline .
  ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
  ?upperPoint nltl:longitudinalPosition ?upperLongitudinal .
  ?upperLongitudinal qudt:numericValue ?x .
  ?upperPoint nltl:verticalCoordinate ?upperVertical .
  ?upperVertical qudt:numericValue ?upperValue .
  FILTER (?upperValue != ?maximumIntendedVertical)
}
UNION
{
  FILTER NOT EXISTS {
    ?this nltl:hasUpperIceWaterline ?upperWaterline2 .
    ?upperWaterline2 nltl:hasWaterlineProfilePoint ?upperPoint2 .
    ?upperPoint2 nltl:longitudinalPosition ?upperLongitudinal2 .
    ?upperLongitudinal2 qudt:numericValue ?x .
    ?upperPoint2 nltl:verticalCoordinate ?upperVertical2 .
    ?upperVertical2 qudt:numericValue ?upperValue2 .
    FILTER (?upperValue2 = ?maximumIntendedVertical)
  }
}
""" ;
    ] .

nltlgen:TRF_011_IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF_011_ProfilePointShape ;
        sh:message "Each represented ice waterline must have waterline profile points."
    ] .

nltlgen:TRF_011_ProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/M> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Each waterline profile point must provide one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/M> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Each waterline profile point must provide one vertical coordinate in metres."
    ] .
<END_SHACL>
