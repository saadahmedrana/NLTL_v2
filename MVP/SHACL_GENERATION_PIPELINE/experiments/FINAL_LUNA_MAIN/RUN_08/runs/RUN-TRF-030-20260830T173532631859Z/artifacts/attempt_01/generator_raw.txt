@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each direct analysis case must provide one load patch length in metres."
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:message "Each direct analysis case must provide one dimensionless ice load area factor ca."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each direct analysis case must provide one vertical load position in metres."
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each direct analysis case must provide one horizontal load position in metres."
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must confirm that the load position minimizes structural capacity."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must confirm evaluation of combined bending and shear."
    ] .

gen:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "The ship must provide one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "The ship must provide one load patch height in metres."
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether load length was determined from the structural arrangement."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "The ship must provide the externally computed applied ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:minCount 1 ;
        sh:node gen:DirectAnalysisCaseShape ;
        sh:message "The ship must provide at least one direct analysis case."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:minCount 2
            ]
        ]
    ) .
</END_SHACL>
