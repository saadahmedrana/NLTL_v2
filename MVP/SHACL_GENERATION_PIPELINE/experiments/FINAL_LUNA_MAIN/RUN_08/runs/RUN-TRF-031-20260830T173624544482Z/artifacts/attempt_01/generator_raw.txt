<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:StressQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:vonMisesYieldCriterion ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:node gen:StressQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "yieldShearStress must equal yieldPoint divided by the square root of 3 within a relative tolerance of 1e-6, and combinedBendingShearStress must be lower than yieldPoint." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:yieldPoint/qudt:numericValue ?yieldPoint ;
                      nltl:combinedBendingShearStress/qudt:numericValue ?combinedStress ;
                      nltl:yieldShearStress/qudt:numericValue ?yieldShearStress .
                BIND((?yieldPoint / 1.7320508075688772) AS ?expectedShear)
                BIND((ABS(?expectedShear) * 0.000001) AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000001,
                        ?relativeTolerance,
                        0.000000001) AS ?tolerance)
                FILTER (
                    ABS(?yieldShearStress - ?expectedShear) > ?tolerance
                    || ?combinedStress >= ?yieldPoint
                )
            }
        """
    ] .

gen:TRF_031_BeamTheoryShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:allowableShearStress ;
        sh:node gen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowableShearStress must be no larger than 0.9 times yieldPoint divided by the square root of 3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:yieldPoint/qudt:numericValue ?yieldPoint ;
                      nltl:allowableShearStress/qudt:numericValue ?allowableShearStress .
                BIND((0.9 * ?yieldPoint / 1.7320508075688772) AS ?maximumAllowable)
                FILTER (?allowableShearStress > ?maximumAllowable)
            }
        """
    ] .
<END_SHACL>
