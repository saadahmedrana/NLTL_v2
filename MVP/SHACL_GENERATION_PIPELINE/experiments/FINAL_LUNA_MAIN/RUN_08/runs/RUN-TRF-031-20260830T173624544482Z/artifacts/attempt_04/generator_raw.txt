<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF031StressQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:vonMisesYieldCriterion ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "von Mises" ;
        sh:message "The acceptance comparison must identify the von Mises yield criterion."
    ] ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node gen:TRF031StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node gen:TRF031StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node gen:TRF031StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:node gen:TRF031StressQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For the von Mises criterion, combined bending and shear stress must be lower than the yield point." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:vonMisesYieldCriterion "von Mises" ;
                      nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldPoint ?yieldNode .
                ?combinedNode qudt:numericValue ?combined .
                ?yieldNode qudt:numericValue ?yield .
                FILTER (?combined >= ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "yieldShearStress must equal yieldPoint divided by the square root of three within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldNode ;
                      nltl:yieldShearStress ?shearNode .
                ?yieldNode qudt:numericValue ?yield .
                ?shearNode qudt:numericValue ?reported .
                BIND (?yield / 1.7320508075688772 AS ?expected)
                FILTER (
                    (?expected >= 0 &&
                        (
                            (?reported - ?expected) > ((?expected * 0.000001) + 0.000000001) ||
                            (?expected - ?reported) > ((?expected * 0.000001) + 0.000000001)
                        )
                    ) ||
                    (?expected < 0 &&
                        (
                            (?reported - ?expected) > (((0 - ?expected) * 0.000001) + 0.000000001) ||
                            (?expected - ?reported) > (((0 - ?expected) * 0.000001) + 0.000000001)
                        )
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowableShearStress must not exceed 0.9 times yieldShearStress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:allowableShearStress ?allowableNode ;
                      nltl:yieldPoint ?yieldNode .
                ?allowableNode qudt:numericValue ?allowable .
                ?yieldNode qudt:numericValue ?yield .
                BIND ((0.9 * ?yield) / 1.7320508075688772 AS ?limit)
                FILTER (?allowable > ?limit)
            }
        """
    ] .
<END_SHACL>
