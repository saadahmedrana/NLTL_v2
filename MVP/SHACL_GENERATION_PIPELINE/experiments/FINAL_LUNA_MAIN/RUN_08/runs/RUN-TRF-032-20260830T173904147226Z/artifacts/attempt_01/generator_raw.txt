        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;

                sh:path qudt:numericValue ;

                sh:maxCount 1
            ]
        ] ;

        sh:message "When the regulation-required scantling is less than the classification-society-required scantling for a ship that is not ice strengthened, the selected design scantling must equal the classification-society-required scantling." ;

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                ?classificationQuantity qudt:numericValue ?classificationValue .
                ?regulationQuantity qudt:numericValue ?regulationValue .
                OPTIONAL {
                    $this nltl:selectedDesignScantling ?selectedQuantity .
                    ?selectedQuantity qudt:numericValue ?selectedValue .
                }
                FILTER (
                    ?iceStatus = false &&
                    ?regulationValue < ?classificationValue &&
                    (
                        !BOUND(?selectedValue) ||
                        ?selectedValue != ?classificationValue
                    )
                )
            }
        """
    ] .
