<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-032-ShipScantlingShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's ice strengthening status is required to determine applicability of the scantling rule."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The classification society required scantling is required for comparison."
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The scantling derived from the regulations is required for comparison."
    ] ;
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A selected design scantling must contain a decimal numeric value."
    ] ;
    sh:sparql [
        sh:message "For a ship that has not been ice strengthened, when the regulation-derived scantling is less than the classification society required scantling, the selected design scantling must use the classification society required value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus false .
                $this nltl:regulationRequiredScantling ?regulationQuantity .
                ?regulationQuantity qudt:numericValue ?regulationValue .
                $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                ?classificationQuantity qudt:numericValue ?classificationValue .
                FILTER (?regulationValue < ?classificationValue)
                FILTER NOT EXISTS {
                    $this nltl:selectedDesignScantling ?selectedQuantity .
                    ?selectedQuantity qudt:numericValue ?selectedValue .
                    FILTER (?selectedValue = ?classificationValue)
                }
            }
        """
    ] .
