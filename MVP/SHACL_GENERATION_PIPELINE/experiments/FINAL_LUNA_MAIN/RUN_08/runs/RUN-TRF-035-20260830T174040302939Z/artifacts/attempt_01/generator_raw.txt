<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-035-ShipRuleLengthShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:classificationSociety ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must identify the classification society whose rule length is used."
    ] ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship's rule length must be provided as a length quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The classification society rule length must be provided as a length quantity in metres."
    ] ;
    sh:sparql [
        sh:message "The ship's rule length must be the classification society rule length used for the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:ruleLength ?ruleQuantity .
                $this nltl:classificationSocietyRuleLength ?societyQuantity .
                ?ruleQuantity qudt:numericValue ?ruleValue ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .
                ?societyQuantity qudt:numericValue ?societyValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (?ruleValue != ?societyValue)
            }
        """
    ] .
<END_SHACL>
