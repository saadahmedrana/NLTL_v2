<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-036-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must link to exactly one ice class design parameter set."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice class value."
    ] .

<urn:nltl:generated-shape:TRF-036-ParameterSetShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The level ice thickness and design ice load height must conform to Table 4-1 for the linked ship ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                       nltl:iceClass ?iceClass .
                $this nltl:levelIceThickness ?levelQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity .
                ?levelQuantity qudt:numericValue ?levelValue ;
                               qudt:unit unit:M .
                ?heightQuantity qudt:numericValue ?heightValue ;
                                qudt:unit unit:M .
                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?levelValue <= "1.0"^^xsd:decimal &&
                        ?heightValue >= "0.35"^^xsd:decimal &&
                        ?heightValue <= "0.35"^^xsd:decimal
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIa &&
                        ?levelValue <= "0.8"^^xsd:decimal &&
                        ?heightValue >= "0.30"^^xsd:decimal &&
                        ?heightValue <= "0.30"^^xsd:decimal
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIb &&
                        ?levelValue <= "0.6"^^xsd:decimal &&
                        ?heightValue >= "0.25"^^xsd:decimal &&
                        ?heightValue <= "0.25"^^xsd:decimal
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIc &&
                        ?levelValue <= "0.4"^^xsd:decimal &&
                        ?heightValue >= "0.22"^^xsd:decimal &&
                        ?heightValue <= "0.22"^^xsd:decimal
                    )
                )
            }
        """
    ] .
<END_SHACL>
