<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF042_PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:TRF042_DesignIceLoadHeightQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each plating must specify one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:TRF042_SpacingQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each plating must specify one spacing in metres."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:node gen:TRF042_F2QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each plating must specify one longitudinal plating factor f2."
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042_CorrosionAdditionAlternativeShape ;
        sh:message "The corrosion and abrasion addition must normally be 2 mm, or be lower only with approved maintained special-surface-coating evidence."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:TRF042_CorrosionAdditionNormalShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:TRF042_CorrosionAdditionLowerShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:specialSurfaceCoatingMaintained ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:class nltl:evidenceState ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:evidenceStateApproved
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The longitudinal plating factor f2 must follow the applicable h/s formula and remain within the specified h/s range."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?fnum .

                BIND (?h / ?s AS ?ratio)
                BIND (
                    IF(
                        ?ratio <= 1,
                        (0.4 / ?ratio) + 0.6,
                        1.4 - (0.4 * ?ratio)
                    )
                    AS ?expected
                )
                BIND (
                    IF(
                        ?expected < 0,
                        0 - ?expected,
                        ?expected
                    )
                    AS ?absExpected
                )
                BIND (
                    IF(
                        (0.000001 * ?absExpected) > 0.000000000001,
                        0.000001 * ?absExpected,
                        0.000000000001
                    )
                    AS ?tolerance
                )

                FILTER (
                    ?ratio <= 0
                    || ?ratio > 1.8
                    || ?fnum < (?expected - ?tolerance)
                    || ?fnum > (?expected + ?tolerance)
                )
            }
        """
    ] .
