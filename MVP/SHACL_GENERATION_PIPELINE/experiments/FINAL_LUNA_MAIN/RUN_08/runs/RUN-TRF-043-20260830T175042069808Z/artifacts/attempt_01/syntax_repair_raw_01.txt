<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            gen:TRF043DeckStructureBoundaryConstraint
            gen:TRF043TankBoundaryPlatingConstraint
            gen:TRF043TankTopBoundaryConstraint
        ) ;
        sh:message "The termination case must identify a deck structure, tank boundary plating, or tank top as the adjacent boundary."
    ] ;
    sh:or (
        gen:TRF043AtMost250mmTerminationBranchShape
        gen:TRF043Above250mmTerminationBranchShape
    ) ;
    sh:message "The extension value must establish whether the 250 mm termination condition applies." .

gen:TRF043DeckStructureBoundaryConstraint
    sh:class nltl:deckStructure .

gen:TRF043TankBoundaryPlatingConstraint
    sh:class nltl:tankBoundaryPlating .

gen:TRF043TankTopBoundaryConstraint
    sh:class nltl:tankTop .

gen:TRF043AtMost250mmTerminationBranchShape
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043ExtensionAtMost250mmQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm."
    ] .

gen:TRF043Above250mmTerminationBranchShape
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043ExtensionAbove250mmQuantityShape
    ] .

gen:TRF043ExtensionAtMost250mmQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive "250"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:TRF043ExtensionAbove250mmQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive "250"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
