<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMmShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:FramingIceStrengtheningTerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:LengthMmShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination permission must be true when the extension is at most 250 mm and false when it exceeds 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension ;
                      nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ?permitted .
                ?extension qudt:numericValue ?value ;
                           qudt:unit qudt:MilliM .
                FILTER (
                    (?value <= 250 && ?permitted != true) ||
                    (?value > 250 && ?permitted = true)
                )
            }
        """
    ] .

gen:ShipFramingIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:FramingIceStrengtheningTerminationCaseShape
    ] .

gen:FrameIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node gen:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node gen:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hullRegion ;
                sh:hasValue nltl:bowRegion
            ] ;
            sh:property [
                sh:path nltl:reachesTankTopOrBelowFloorTop ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullRegion ;
                sh:hasValue nltl:midbodyRegion
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullRegion ;
                sh:hasValue nltl:sternRegion
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing vertical extents must match the applicable Table 4-6 ice-class and hull-region values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .
                ?above qudt:numericValue ?aboveValue ;
                       qudt:unit qudt:M .
                ?below qudt:numericValue ?belowValue ;
                       qudt:unit qudt:M .
                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?aboveValue != 1.2
                    ) ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?aboveValue != 1.0
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region IN (nltl:bowRegion, nltl:midbodyRegion) &&
                        ?belowValue != 2.0
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:sternRegion &&
                        ?belowValue != 1.6
                    ) ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:bowRegion &&
                        ?belowValue != 1.6
                    ) ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:midbodyRegion &&
                        ?belowValue != 1.3
                    ) ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:sternRegion &&
                        ?belowValue != 1.0
                    )
                )
            }
        """
    ] .

</END_SHACL>
