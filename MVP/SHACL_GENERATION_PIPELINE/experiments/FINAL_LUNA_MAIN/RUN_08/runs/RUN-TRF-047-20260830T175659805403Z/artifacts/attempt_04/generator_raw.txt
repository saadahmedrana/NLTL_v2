<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:AttachedSupportNodeShape
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:tankBoundaryPlating ]
        [ sh:class nltl:tankTop ]
        [ sh:class nltl:iceStringer ]
    ) .

gen:AttachedSupportPropertyShape
    a sh:PropertyShape ;
    sh:path (
        nltl:hasStrengthenedPart
        nltl:hasLowerEnd
        nltl:hasAttachedSupportingStructure
    ) ;
    sh:minCount 1 ;
    sh:node gen:AttachedSupportNodeShape ;
    sh:message "The lower end of the strengthened part must be attached to a permitted deck, tank plating, tank top, or ice stringer." .

gen:ExceptionalIntermediateBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The supporting structure must be at or below the lower limit of the ice belt for the exceptional intermediate-frame connection."
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The horizontal member must have the same scantlings as the main frames."
    ] ;
    sh:sparql [
        sh:message "An intermediate ice frame may use the exceptional connection only when its lower end terminates at a permitted below-ice-belt supporting structure and has a horizontal member connected to an adjacent main frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasStrengthenedPart ?strengthenedPart .
                    ?strengthenedPart nltl:hasLowerEnd ?lowerEnd .
                    ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?supportingStructure .
                    ?supportingStructure a ?supportingStructureType .
                    FILTER (
                        ?supportingStructureType IN (
                            nltl:deckStructure,
                            nltl:tankBoundaryPlating,
                            nltl:tankTop,
                            nltl:iceStringer
                        )
                    )
                    ?lowerEnd nltl:hasHorizontalConnectionMember ?horizontalMember .
                    ?horizontalMember nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                }
            }
        """
    ] .

gen:MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A main frame below the lower edge of the ice belt must be ice strengthened."
    ] ;
    sh:property gen:AttachedSupportPropertyShape .

gen:IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:or (
        gen:AttachedSupportPropertyShape
        gen:ExceptionalIntermediateBranch
    ) ;
    sh:message "The lower end of an intermediate ice frame must use the ordinary attached-supporting-structure arrangement or the permitted exceptional horizontal-member arrangement."
] .
<END_SHACL>
