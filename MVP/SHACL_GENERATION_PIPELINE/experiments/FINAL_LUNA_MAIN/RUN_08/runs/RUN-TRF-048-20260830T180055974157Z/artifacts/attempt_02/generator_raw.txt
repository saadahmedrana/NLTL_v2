    sh:sparql [
        sh:message "A continuous beam with brackets without significantly different boundary conditions requires frame moment factor m = 13.3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
