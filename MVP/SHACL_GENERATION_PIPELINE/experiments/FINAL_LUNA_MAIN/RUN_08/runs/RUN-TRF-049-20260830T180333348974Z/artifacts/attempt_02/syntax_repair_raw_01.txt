<urn:nltl:generated-shape:TRF049_FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:frameAttachment ;
                    sh:property [
                        sh:path nltl:effectiveAttachmentConfirmed ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:hasAttachedSupportingStructure ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:supportingStructure
                        ]
                    ]
                ]
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:MilliM
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A bracket thickness must be at least the frame web thickness in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:frameAttachment ;
            sh:property [
                sh:path nltl:hasConnectionBracket ;
                sh:minCount 1 ;
                sh:node <urn:nltl:generated-shape:TRF049_ConnectionBracketShape>
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each supporting web frame linked to a longitudinal frame must be attached through a frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSupportingWebFrame ?supportingWebFrame .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingWebFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each supporting bulkhead linked to a longitudinal frame must be attached through a frame attachment." ;
