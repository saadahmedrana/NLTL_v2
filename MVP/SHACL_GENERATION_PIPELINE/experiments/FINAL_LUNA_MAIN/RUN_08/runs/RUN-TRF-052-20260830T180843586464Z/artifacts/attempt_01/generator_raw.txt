            PREFIX nltl: <https://w3id.org/nltl/vocab#>

                $this nltl:frameAsymmetrical ?asym .
                $this nltl:frameWebAngleToShell ?angleNode .

                FILTER NOT EXISTS {
                    $this nltl:equivalentSupportStatusByDirectCalculation true .
                }
                FILTER NOT EXISTS {
                    $this nltl:antitrippingSupportSpacing ?spacingNode .
                    ?spacingNode qudt:numericValue ?spacing .

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                FILTER NOT EXISTS {
                    $this nltl:equivalentSupportStatusByDirectCalculation true .

                        (
                            NOT EXISTS {

                            }
                        )
                    ) ||

                        }
                    )
                )
            }
        """
