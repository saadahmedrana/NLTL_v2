<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceStringerQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must contain exactly one decimal numeric value."
    ] .

gen:IcePressureQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice pressure must use MPa as its unit."
    ] .

gen:DesignIceLoadHeightQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design ice load height must use metres as its unit."
    ] .

gen:SpanQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Span must use metres as its unit."
    ] .

gen:FrameMomentFactorQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame moment factor must be unitless."
    ] .

gen:YieldStrengthQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Yield strength must use MPa as its unit."
    ] .

gen:DesignLineLoadQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN-PER-M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design line load must use kN/m as its unit."
    ] .

gen:SectionModulusQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:CentiM3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Section modulus must use cm^3 as its unit."
    ] .

gen:RequiredShearAreaQuantityShape a sh:NodeShape ;
    sh:node gen:IceStringerQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Required shear area must use cm^2 as its unit."
    ] .

gen:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:locatedWithinIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice stringer must be explicitly recorded as located within the ice belt."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IcePressureQuantityShape ;
        sh:message "Ice pressure is required for the ice-belt stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DesignIceLoadHeightQuantityShape ;
        sh:message "Design ice load height is required for the ice-belt stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SpanQuantityShape ;
        sh:message "Stringer span is required for the ice-belt stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FrameMomentFactorQuantityShape ;
        sh:message "The frame moment factor is required for the ice-belt stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:YieldStrengthQuantityShape ;
        sh:message "Yield strength is required for the ice-belt stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DesignLineLoadQuantityShape ;
        sh:message "The calculated design line load is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:message "The calculated section modulus is required."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:RequiredShearAreaQuantityShape ;
        sh:message "The calculated required shear area is required."
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, and required shear area must satisfy the declared formulas within a relative tolerance of 1e-6, with an absolute tolerance of 1e-6 for expected values near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?strengthNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?modulusNode ;
                      nltl:requiredShearArea ?areaNode .

                ?pressureNode qudt:numericValue ?p .
                ?heightNode qudt:numericValue ?h .
                ?spanNode qudt:numericValue ?l .
                ?momentNode qudt:numericValue ?m .
                ?strengthNode qudt:numericValue ?sigma .
                ?lineNode qudt:numericValue ?lineReported .
                ?modulusNode qudt:numericValue ?zReported .
                ?areaNode qudt:numericValue ?aReported .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?lineExpected)
                BIND((0.9 * 1.8 * ?lineExpected * ?l * ?l / (?m * ?sigma) * 1000000) AS ?zExpected)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?lineExpected * ?l / (2 * ?sigma) * 10000) AS ?aExpected)

                FILTER(
                    (
                        (?lineExpected <= 0.001 &&
                         (?lineReported < (?lineExpected - 0.000001) ||
                          ?lineReported > (?lineExpected + 0.000001))) ||
                        (?lineExpected > 0.001 &&
                         (?lineReported < (?lineExpected * (1 - 0.000001)) ||
                          ?lineReported > (?lineExpected * (1 + 0.000001))))
                    ) ||
                    (
                        (?zExpected <= 0.001 &&
                         (?zReported < (?zExpected - 0.000001) ||
                          ?zReported > (?zExpected + 0.000001))) ||
                        (?zExpected > 0.001 &&
                         (?zReported < (?zExpected * (1 - 0.000001)) ||
                          ?zReported > (?zExpected * (1 + 0.000001))))
                    ) ||
                    (
                        (?aExpected <= 0.001 &&
                         (?aReported < (?aExpected - 0.000001) ||
                          ?aReported > (?aExpected + 0.000001))) ||
                        (?aExpected > 0.001 &&
                         (?aReported < (?aExpected * (1 - 0.000001)) ||
                          ?aReported > (?aExpected * (1 + 0.000001))))
                    )
                )
            }
        """
    ] .
<END_SHACL>
