    ] ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityKiloNPerM
    ] ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityM
    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityCentiM3
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityCentiM2
    ] ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:evidenceState
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityCentiM3
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF055_QuantityM
    ] ;
    sh:node gen:TRF055_LongHatchException ;
    sh:sparql [
        sh:message "The actual section modulus and shear area must meet their required values, and an approved very-long-hatch reduced line load must satisfy its permitted bounds." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:abreastOfHatch true ;
                      nltl:servesAsIceStringer true ;
                      nltl:actualSectionModulus/qudt:numericValue ?actualZ ;
                      nltl:requiredSectionModulus/qudt:numericValue ?requiredZ ;
                      nltl:actualShearArea/qudt:numericValue ?actualA ;
                      nltl:requiredShearArea/qudt:numericValue ?requiredA ;
                      nltl:veryLongHatchOpening ?long .
                FILTER (
                    ?actualZ < ?requiredZ ||
                    ?actualA < ?requiredA ||
                    (
                        ?long = true &&
                        EXISTS {
                            $this nltl:permittedReducedLineLoad/qudt:numericValue ?reduced ;
                                  nltl:reducedLineLoadApprovedByClassificationSociety true .
                            FILTER (
                                ?reduced < "0.10"^^xsd:decimal ||
                                ?reduced >= "0.15"^^xsd:decimal
                            )
                        }
                    )
                )
            }
        """
    ] .
