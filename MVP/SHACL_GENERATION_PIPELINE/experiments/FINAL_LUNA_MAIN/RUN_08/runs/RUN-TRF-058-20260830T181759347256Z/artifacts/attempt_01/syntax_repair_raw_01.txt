<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceLoadForce ;
                sh:node gen:ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:stringerOutsideIceBeltHeight ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:stringerSpan ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:adjustedIceLoadForce ;
                sh:node gen:ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within numerical tolerance" ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceLoadForce ?forceNode ;
                              nltl:stringerOutsideIceBeltHeight ?heightNode ;
                              nltl:stringerSpan ?spanNode ;
                              nltl:adjustedIceLoadForce ?adjustedNode .

                        ?forceNode qudt:numericValue ?force ;
                                   qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        ?heightNode qudt:numericValue ?height ;
                                    qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?spanNode qudt:numericValue ?span ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?adjustedNode qudt:numericValue ?adjusted ;
                                     qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                        BIND (?force * (1 - (?height / ?span)) AS ?expected)
                        BIND (ABS(?expected) * 0.000001 AS ?relativeTolerance)
                        BIND (
                            IF(
                                ?relativeTolerance > 0.000000001,
                                ?relativeTolerance,
                                0.000000001
                            )
                            AS ?tolerance
                        )

                        FILTER (ABS(?adjusted - ?expected) > ?tolerance)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
<END_SHACL>
