@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minCount 1 ;
                sh:in ( 2 3 )
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:sidePropellerStrengtheningEnvelope ;
                sh:node gen:TRF063EnvelopeShape ;
                sh:message "Each twin or triple screw ship shall provide a conforming side-propeller strengthening envelope."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minCount 1 ;
                sh:not [
                    sh:in ( 2 3 )
                ]
            ]
        ]
    ) .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:message "The strengthening envelope shall have the tank top as its boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063ExtentShape ;
        sh:message "The strengthening envelope shall extend at least 1.5 metres forward of the side propeller."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063ExtentShape ;
        sh:message "The strengthening envelope shall extend at least 1.5 metres aft of the side propeller."
    ] .

gen:TRF063ExtentShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
