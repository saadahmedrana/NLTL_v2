<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF063ApplicableShipShape
        gen:TRF063NonApplicableShipShape
    ) .

gen:TRF063ApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 3
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:node gen:TRF063EnvelopeShape ;
        sh:message "A twin- or triple-screw ship shall provide a side-propeller strengthening envelope."
    ] .

gen:TRF063NonApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:or (
            [
                sh:maxInclusive 1
            ]
            [
                sh:minInclusive 4
            ]
        ]
    ] .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:minCount 1 ;
        sh:class nltl:tankTop ;
        sh:message "The strengthening envelope shall have a tank-top boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063MetreExtentShape ;
        sh:message "The strengthening shall extend at least 1.5 metres forward of the side propeller."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063MetreExtentShape ;
        sh:message "The strengthening shall extend at least 1.5 metres aft of the side propeller."
    ] .

gen:TRF063MetreExtentShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 1.5
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
