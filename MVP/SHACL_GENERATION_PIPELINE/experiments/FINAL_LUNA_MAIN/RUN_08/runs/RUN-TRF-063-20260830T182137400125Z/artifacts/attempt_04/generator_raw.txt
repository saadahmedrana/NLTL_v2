<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF063TwinShipBranch
        gen:TRF063TripleShipBranch
        gen:TRF063NonApplicableLowPropellerCountBranch
        gen:TRF063NonApplicableHighPropellerCountBranch
    ) .

gen:TRF063TwinShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2 ;
        sh:message "A twin-screw ship must declare propellerCount 2."
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape ;
        sh:message "A twin-screw ship must link a conforming side-propeller strengthening envelope."
    ] .

gen:TRF063TripleShipBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 3 ;
        sh:maxInclusive 3 ;
        sh:message "A triple-screw ship must declare propellerCount 3."
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape ;
        sh:message "A triple-screw ship must link a conforming side-propeller strengthening envelope."
    ] .

gen:TRF063NonApplicableLowPropellerCountBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxInclusive 1
    ] .

gen:TRF063NonApplicableHighPropellerCountBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 4
    ] .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:sidePropellerStrengtheningEnvelope ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:minCount 1 ;
        sh:class nltl:tankTop ;
        sh:message "The strengthening envelope must identify a tank-top boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063MetreExtentShape ;
        sh:message "The forward strengthening extent must be at least 1.5 metres."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063MetreExtentShape ;
        sh:message "The aft strengthening extent must be at least 1.5 metres."
    ] .

gen:TRF063MetreExtentShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal ;
        sh:message "The extent value must be a decimal of at least 1.5."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The extent must use metres."
    ] .
<END_SHACL>
