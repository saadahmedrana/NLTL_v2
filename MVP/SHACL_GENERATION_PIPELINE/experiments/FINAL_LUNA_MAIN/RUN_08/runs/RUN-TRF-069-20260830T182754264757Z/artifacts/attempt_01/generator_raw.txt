<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF069Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted ice class." ;
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall identify its propeller type." ;
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall identify its pitch type." ;
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide the maximum ice torque in kN*m." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:propellerType ;
                sh:in ( "open" "ducted" )
            ] ;
            sh:property [
                sh:path nltl:pitchType ;
                sh:in ( "controllable" "fixed" )
            ] ;
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:normalOperationalConditions ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:radialIceEntryOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "An applicable design condition shall record the prescribed normal service-life load scope and each stated load-model exclusion." ;
            ]
        ]
        [
            sh:sparql [
                sh:message "If the ice-class, propeller-type, and pitch-type applicability conditions are not all met, the propulsion-machinery rule is non-applicable." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?iceClass ;
                              nltl:propellerType ?propellerType ;
                              nltl:pitchType ?pitchType .
                        FILTER (
                            !(
                                ?iceClass IN (
                                    nltl:iceClassIaSuper,
                                    nltl:iceClassIa,
                                    nltl:iceClassIb,
                                    nltl:iceClassIc
                                )
                            )
                            || !(?propellerType IN ("open", "ducted"))
                            || !(?pitchType IN ("controllable", "fixed"))
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
