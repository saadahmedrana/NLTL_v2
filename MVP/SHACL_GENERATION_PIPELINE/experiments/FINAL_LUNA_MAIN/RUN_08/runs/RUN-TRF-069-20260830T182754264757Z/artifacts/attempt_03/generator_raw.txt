<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:class nltl:designCondition
    ] ;
    sh:sparql [
        sh:message "For an applicable propulsion-machinery configuration, maximum ice torque and all required design-condition scope evidence must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                    && ?propellerType IN ("open", "ducted")
                    && ?pitchType IN ("controllable", "fixed")
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumIceTorque ?torque .
                        ?torque a qudt:QuantityValue ;
                                qudt:numericValue ?numericValue ;
                                qudt:unit unit:KiloN-M .
                        FILTER (datatype(?numericValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDesignCondition ?condition .
                        ?condition a nltl:designCondition ;
                                   nltl:normalOperationalConditions ?normalOperational ;
                                   nltl:normalServiceLifePropellerIceLoadCondition ?serviceLife ;
                                   nltl:fixedPitchPropellerReversalLoadIncluded ?reversal ;
                                   nltl:stoppedPropellerDraggingOutsideLoadModel ?dragging ;
                                   nltl:radialIceEntryOutsideLoadModel ?radial .

                        FILTER (
                            datatype(?normalOperational) = xsd:boolean
                            && ?normalOperational = true
                            && datatype(?serviceLife) = xsd:boolean
                            && ?serviceLife = true
                            && datatype(?reversal) = xsd:boolean
                            && ?reversal = true
                            && datatype(?dragging) = xsd:boolean
                            && ?dragging = true
                            && datatype(?radial) = xsd:boolean
                            && ?radial = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
