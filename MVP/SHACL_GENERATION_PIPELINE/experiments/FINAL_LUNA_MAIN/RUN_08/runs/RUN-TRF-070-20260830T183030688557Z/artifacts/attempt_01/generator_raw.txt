<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF070ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:or (
            [
                sh:hasValue "azimuthing"^^xsd:string
            ]
            [
                sh:hasValue "fixed"^^xsd:string
            ]
        ] ;
        sh:message "The ship shall identify an azimuthing or fixed thruster type."
    ] ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The thruster shall be identified as main propulsion."
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gen:TRF070ForceQuantityShape ;
        sh:message "A propeller/ice interaction load with a valid force quantity is required."
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gen:TRF070ForceQuantityShape ;
        sh:message "A thruster-body/ice interaction load with a valid force quantity is required."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node gen:TRF070PressureQuantityShape ;
        sh:message "A local ice pressure with a valid pressure quantity is required."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:node [
            sh:class nltl:designCondition ;
            sh:property [
                sh:path nltl:designConditionLocalStrengthCapacity ;
                sh:minCount 1 ;
                sh:node gen:TRF070ForceQuantityShape ;
                sh:message "Each design condition shall provide an externally assessed local-strength capacity."
            ]
        ] ;
        sh:message "The ship shall provide a linked design condition."
    ] ;
    sh:sparql [
        sh:message "The externally assessed local-strength capacity shall be at least the applicable local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasDesignCondition ?condition .
                ?condition nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacity .
                $this nltl:localIcePressure ?pressureQuantity .
                ?pressureQuantity qudt:numericValue ?pressure .
                FILTER (
                    datatype(?capacity) = xsd:decimal &&
                    datatype(?pressure) = xsd:decimal &&
                    ?capacity < ?pressure
                )
            }
        """
    ] .

gen:TRF070ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The force quantity shall contain a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The force quantity shall use kilonewtons."
    ] .

gen:TRF070PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The pressure quantity shall contain a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The pressure quantity shall use megapascals."
    ] .
<END_SHACL>
