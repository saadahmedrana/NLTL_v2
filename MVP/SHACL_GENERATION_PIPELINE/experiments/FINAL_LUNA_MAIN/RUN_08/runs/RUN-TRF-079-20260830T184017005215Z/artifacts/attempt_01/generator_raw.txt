<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF079StaticCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:propellerImmersionFunction ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:propellerShaftCentrelineDepth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:sparql [
        sh:message "The immersion function and submersion factor do not satisfy the specified formulae." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?iceQuantity ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Quantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:propellerImmersionFunction ?fQuantity ;
                      nltl:propellerShaftCentrelineDepth ?depthQuantity .

                ?iceQuantity qudt:numericValue ?hIce .
                ?k2Quantity qudt:numericValue ?k2 .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?fQuantity qudt:numericValue ?f .
                ?depthQuantity qudt:numericValue ?h0 .

                BIND(
                    IF(
                        ?diameter = 0,
                        0.0,
                        ((?h0 - ?hIce) / (?diameter / 2.0)) - 1.0
                    ) AS ?fCalculated
                )

                BIND(
                    IF(
                        ?f < 0,
                        0.8 - ?f,
                        IF(
                            ?f <= 1,
                            0.8 - (0.4 * ?f),
                            IF(
                                ?f <= 2.5,
                                0.6 - (0.2 * ?f),
                                0.1
                            )
                        )
                    ) AS ?k2Calculated
                )

                FILTER(
                    ?diameter = 0
                    ||
                    ABS(?f - ?fCalculated) >
                        ((0.000001 * ABS(?fCalculated)) + 0.000000001)
                    ||
                    ABS(?k2 - ?k2Calculated) >
                        ((0.000001 * ABS(?k2Calculated)) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
