<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-080> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one ice load cycle count." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one propeller blade count." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The effective ice load cycle count must use the unitless count unit." ;
            ]
        ] ;
        sh:message "Each ship must provide one effective ice load cycle count quantity." ;
    ] ;
    sh:sparql [
        sh:message "The effective ice load cycle count must equal the ice load cycle count multiplied by the propeller blade count, within a relative tolerance of 1e-6 plus an absolute fallback of 1e-7." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?ice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?resultNode .
                ?resultNode qudt:numericValue ?reported .
                BIND (xsd:decimal(?ice) * xsd:decimal(?blades) AS ?expected)
                FILTER (
                    ABS(?reported - ?expected) >
                    (ABS(?expected) * "0.000001"^^xsd:decimal +
                     "0.0000001"^^xsd:decimal)
                )
            }
        """
    ] .
<END_SHACL>
