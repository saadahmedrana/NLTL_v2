<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_080_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one ice load cycle count." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one propeller blade count." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
        ] ;
        sh:message "The effective ice load cycle count must be a single unitless decimal quantity." ;
    ] ;
    sh:sparql [
        sh:message "The effective ice load cycle count must equal iceLoadCycleCount multiplied by propellerBladeCount within an expected-result-scaled tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?resultNode .

                ?resultNode qudt:numericValue ?reported .

                BIND (xsd:decimal(?nice) * xsd:decimal(?blades) AS ?expected)

                FILTER (
                    (
                        ?expected >= 0 &&
                        (
                            ?reported < (?expected - ((?expected * 0.000001) + 0.000001)) ||
                            ?reported > (?expected + ((?expected * 0.000001) + 0.000001))
                        )
                    ) ||
                    (
                        ?expected < 0 &&
                        (
                            ?reported < (?expected - (((0 - ?expected) * 0.000001) + 0.000001)) ||
                            ?reported > (?expected + (((0 - ?expected) * 0.000001) + 0.000001))
                        )
                    )
                )
            }
        """ ;
    ] .
<END_SHACL>
