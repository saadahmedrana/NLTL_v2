        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce ?ffNode ;
                      nltl:backwardBladeForce ?fbNode ;
                      nltl:hydrodynamicBollardThrust ?hydroNode ;
                      nltl:forwardIceThrust ?tfNode ;
                      nltl:backwardIceThrust ?tbNode ;
                      nltl:forwardPropellerShaftDesignThrust ?fdrNode ;
                      nltl:backwardPropellerShaftDesignThrust ?bdrNode ;
                      nltl:propellerShaftDesignThrust ?drNode .

                ?ffNode qudt:numericValue ?ff ;

                ?hydroNode qudt:numericValue ?hydro ;

                ?tfNode qudt:numericValue ?tf ;
                       qudt:unit unit:KiloN .
                ?tbNode qudt:numericValue ?tb ;
                       qudt:unit unit:KiloN .
                ?fdrNode qudt:numericValue ?fdr ;

                ?bdrNode qudt:numericValue ?bdr ;
                        qudt:unit unit:KiloN .
                ?drNode qudt:numericValue ?dr ;
                       qudt:unit unit:KiloN .

                BIND ((1.1 * ?ff) AS ?expectedTf)
                BIND ((1.1 * ?fb) AS ?expectedTb)
                BIND ((?hydro + (2.2 * ?expectedTf)) AS ?forwardLoad)
                BIND ((1.5 * ?expectedTb) AS ?backwardLoad)
                BIND (IF(?forwardLoad >= ?backwardLoad, ?forwardLoad, ?backwardLoad) AS ?expectedDesign)

                BIND (IF(?tf >= ?expectedTf, ?tf - ?expectedTf, ?expectedTf - ?tf) AS ?diffTf)
                BIND (IF(?tb >= ?expectedTb, ?tb - ?expectedTb, ?expectedTb - ?tb) AS ?diffTb)
                BIND (IF(?fdr >= ?expectedDesign, ?fdr - ?expectedDesign, ?expectedDesign - ?fdr) AS ?diffFdr)
                BIND (IF(?bdr >= ?expectedDesign, ?bdr - ?expectedDesign, ?expectedDesign - ?bdr) AS ?diffBdr)
                BIND (IF(?dr >= ?expectedDesign, ?dr - ?expectedDesign, ?expectedDesign - ?dr) AS ?diffDr)

                BIND ((0.000001 * IF(?expectedTf >= 0, ?expectedTf, 0 - ?expectedTf) + 0.000001) AS ?tolTf)
                BIND ((0.000001 * IF(?expectedTb >= 0, ?expectedTb, 0 - ?expectedTb) + 0.000001) AS ?tolTb)
                BIND ((0.000001 * IF(?expectedDesign >= 0, ?expectedDesign, 0 - ?expectedDesign) + 0.000001) AS ?tolDesign)

                FILTER (
                    ?diffTf > ?tolTf ||
                    ?diffTb > ?tolTb ||
                    ?diffFdr > ?tolDesign ||
                    ?diffBdr > ?tolDesign ||
                    ?diffDr > ?tolDesign
                )
            }
        """
