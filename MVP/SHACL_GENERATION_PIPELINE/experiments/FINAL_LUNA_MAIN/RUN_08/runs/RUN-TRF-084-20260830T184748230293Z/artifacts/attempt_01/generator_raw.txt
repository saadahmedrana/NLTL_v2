<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single propeller type is required to determine applicability."
    ] ;
    sh:or (
        gen:TRF084CPKnownShape
        gen:TRF084CPUnknownShape
        gen:TRF084NonCPShape
    ) .

gen:TRF084CPKnownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:hasValue "CP" ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with known pitch, the MCR bollard pitch is required in metres."
    ] .

gen:TRF084CPUnknownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:hasValue "CP" ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrFreeRunning ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with unknown pitch, the free-running MCR pitch is required in metres."
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For a CP propeller with unknown pitch, the MCR bollard pitch is required in metres."
    ] ;
    sh:sparql [
        sh:message "For a CP propeller with unknown pitch, the MCR bollard pitch must equal 0.7 times the free-running MCR pitch within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerType "CP" ;
                      nltl:pitchValueKnown false ;
                      nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
                      nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                ?freeQuantity qudt:numericValue ?freeValue .
                ?bollardQuantity qudt:numericValue ?bollardValue .
                BIND(0.7 * ?freeValue AS ?expectedValue)
                BIND(
                    IF(
                        ABS(?expectedValue) * 0.000001 > 0.000000001,
                        ABS(?expectedValue) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?bollardValue - ?expectedValue) > ?tolerance)
            }
        """
    ] .

gen:TRF084NonCPShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This requirement is not applicable to non-CP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER(?propellerType = "CP")
            }
        """
    ] .
<END_SHACL>
