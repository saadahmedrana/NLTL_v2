<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF091ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091IceMillingSequenceCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF091PropulsionLoadCaseShape
    ] ;
    sh:sparql [
        sh:message "The ice-milling load sequence case used for propulsion-line strength evaluation must be the applicable propulsion load case and must not be designated for stalling analysis." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasIceMillingSequenceCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?case .
                    ?case nltl:loadCasePurpose nltl:propulsionLineStrengthEvaluationPurpose .
                    FILTER NOT EXISTS {
                        ?case nltl:loadCasePurpose nltl:stallingAnalysisPurposeValue .
                    }
                }
            }
        """
    ] .

gen:TRF091IceMillingSequenceCaseShape
    a sh:NodeShape .

gen:TRF091PropulsionLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCasePurposeValue ;
        sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The propulsion load case shall have propulsion-line strength evaluation as its purpose and shall not have stalling analysis as its purpose."
    ] .
<END_SHACL>
