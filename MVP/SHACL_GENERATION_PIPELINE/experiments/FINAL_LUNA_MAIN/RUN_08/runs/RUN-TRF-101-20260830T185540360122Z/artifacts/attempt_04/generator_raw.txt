<urn:nltl:generated-shape:TRF-101-FormulaAlternative> a sh:NodeShape ;
    sh:sparql [
        sh:message "The calculated blade failure load or reference strength does not satisfy the declared formula within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:approvedNonlinearBladeStressAnalysis ?analysis .
                }
                $this nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:bladeFailureUltimateLoad ?loadNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeMinimumYieldStrength ?yieldNode ;
                      nltl:bladeReferenceStrength1 ?refNode ;
                      nltl:bladeUltimateTensileStrength ?ultimateNode ;
                      nltl:propellerDiameter ?dNode .
                ?cNode qudt:numericValue ?c .
                ?rNode qudt:numericValue ?r .
                ?loadNode qudt:numericValue ?reportedLoad .
                ?tNode qudt:numericValue ?t .
                ?yieldNode qudt:numericValue ?yield .
                ?refNode qudt:numericValue ?reportedRef .
                ?ultimateNode qudt:numericValue ?ultimate .
                ?dNode qudt:numericValue ?D .
                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedRef)
                BIND ((300 * ?c * ?t * ?t * ?expectedRef) /
                      ((0.8 * ?D) - (2 * ?r)) AS ?expectedLoad)
                BIND ((MAX(?expectedRef, -?expectedRef) * 0.000001) + 0.000000001 AS ?refTolerance)
                BIND ((MAX(?expectedLoad, -?expectedLoad) * 0.000001) + 0.000000001 AS ?loadTolerance)
                FILTER (
                    ?reportedRef < (?expectedRef - ?refTolerance) ||
                    ?reportedRef > (?expectedRef + ?refTolerance) ||
                    ?reportedLoad < (?expectedLoad - ?loadTolerance) ||
                    ?reportedLoad > (?expectedLoad + ?loadTolerance)
                )
            }
        """
    ] .
