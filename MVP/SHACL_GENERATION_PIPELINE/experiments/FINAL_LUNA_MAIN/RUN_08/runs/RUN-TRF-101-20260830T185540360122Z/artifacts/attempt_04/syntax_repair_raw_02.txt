<urn:nltl:generated-shape:TRF-101-FormulaAlternative> a sh:NodeShape ;
    sh:sparql [
        sh:message "The calculated blade failure load or reference strength does not satisfy the declared formula within tolerance." ;
        sh:select "PREFIX nltl: <https://w3id.org/nltl/vocab#>\nPREFIX qudt: <http://qudt.org/schema/qudt/>\nPREFIX xsd: <http://www.w3.org/2001/XMLSchema#>\nSELECT $this\nWHERE {\n    FILTER NOT EXISTS {\n        $this nltl:approvedNonlinearBladeStressAnalysis ?analysis .\n    }\n    $this nltl:bladeChordLengthAtWeakestSection ?cNode ;\n          nltl:bladeCylindricalRootRadius ?rNode ;\n          nltl:bladeFailureUltimateLoad ?loadNode ;\n          nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;\n          nltl:bladeMinimumYieldStrength ?yieldNode ;\n          nltl:bladeReferenceStrength1 ?refNode ;\n          nltl:bladeUltimateTensileStrength ?ultimateNode ;\n          nltl:propellerDiameter ?dNode .\n    ?cNode qudt:numericValue ?c .\n    ?rNode qudt:numericValue ?r .\n    ?loadNode qudt:numericValue ?reportedLoad .\n    ?tNode qudt:numericValue ?t .\n    ?yieldNode qudt:numericValue ?yield .\n    ?refNode qudt:numericValue ?reportedRef .\n    ?ultimateNode qudt:numericValue ?ultimate .\n    ?dNode qudt:numericValue ?D .\n    BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedRef)\n    BIND ((300 * ?c * ?t * ?t * ?expectedRef) / ((0.8 * ?D) - (2 * ?r)) AS ?expectedLoad)\n    BIND ((IF(?expectedRef >= 0, ?expectedRef, -?expectedRef) * 0.000001) + 0.000000001 AS ?refTolerance)\n    BIND ((IF(?expectedLoad >= 0, ?expectedLoad, -?expectedLoad) * 0.000001) + 0.000000001 AS ?loadTolerance)\n    FILTER (\n        ?reportedRef < (?expectedRef - ?refTolerance) ||\n        ?reportedRef > (?expectedRef + ?refTolerance) ||\n        ?reportedLoad < (?expectedLoad - ?loadTolerance) ||\n        ?reportedLoad > (?expectedLoad + ?loadTolerance)\n    )\n}"
    ] .
