<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF104ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:pyramidStrengthPrinciple ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall record the pyramid strength principle for the propulsion line."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoss ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent
    ] ;
    sh:sparql [
        sh:message "When propeller blade loss is represented, every linked propeller shaft line component shall have significantDamageFromBladeLoss=false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerBladeLoss true ;
                      nltl:hasPropellerShaftLineComponent ?component .
                FILTER (
                    NOT EXISTS {
                        ?component nltl:significantDamageFromBladeLoss false .
                    }
                    ||
                    EXISTS {
                        ?component nltl:significantDamageFromBladeLoss ?damage .
                        FILTER (?damage != false)
                    }
                )
            }
        """
    ] .

gen:TRF104PropellerShaftLineComponentShape
    a sh:NodeShape ;
    sh:targetClass nltl:propellerShaftLineComponent ;
    sh:property [
        sh:path nltl:significantDamageFromBladeLoss ;
        sh:datatype xsd:boolean
    ] .
<END_SHACL>
