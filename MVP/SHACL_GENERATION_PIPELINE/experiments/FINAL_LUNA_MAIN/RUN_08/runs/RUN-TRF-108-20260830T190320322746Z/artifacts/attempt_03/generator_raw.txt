<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one SN-curve type selector."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:message "The ship must select one SN-curve type value."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact ;
        sh:message "A known SN-curve selection must have correspondence evidence."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A known SN-curve selection must identify the blade material property."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the SN-curve type is unknown, the two-slope SN curve must be selected." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:stressLifeCurveType ?curveType .
                FILTER (LCASE(STR(?curveType)) = "unknown")
                FILTER NOT EXISTS {
                    $this nltl:stressLifeCurveSelection nltl:twoSlopeStressLifeCurve .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the SN-curve type is known, correspondence evidence must be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:stressLifeCurveType ?curveType .
                FILTER (LCASE(STR(?curveType)) != "unknown")
                FILTER NOT EXISTS {
                    $this nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the SN-curve type is known, the selected curve must be associated with the blade material property." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:stressLifeCurveType ?curveType .
                FILTER (LCASE(STR(?curveType)) != "unknown")
                FILTER NOT EXISTS {
                    $this nltl:propellerBladeMaterialProperty ?materialProperty .
                }
            }
        """
    ] .
<END_SHACL>
