<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A stress-life curve type selector is required."
    ] ;
    sh:xone (
        gen:TRF108KnownCurveBranch
        gen:TRF108UnknownCurveBranch
    ) .

gen:TRF108KnownCurveBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The known-curve branch requires stressLifeCurveKnown to be true."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:message "A selected stress-life curve is required when the curve is known."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Blade material properties are required when the curve is known."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Correspondence evidence is required when the curve is known."
    ] .

gen:TRF108UnknownCurveBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "The unknown-curve branch requires stressLifeCurveKnown to be false."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:hasValue nltl:twoSlopeStressLifeCurve ;
        sh:message "The two-slope stress-life curve must be selected when the curve is unknown."
    ] .
<END_SHACL>
