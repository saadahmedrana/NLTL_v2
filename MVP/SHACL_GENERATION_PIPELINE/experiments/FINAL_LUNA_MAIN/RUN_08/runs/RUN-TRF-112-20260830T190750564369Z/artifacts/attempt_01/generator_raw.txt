<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF112ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node gen:TRF112ShaftComponentShape
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF112LoadCaseShape
    ] .

gen:TRF112LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:bladeFailureLoadCase ;
        sh:message "The propulsion load case must be identified as the total blade-failure load case."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ] ;
        sh:message "The applicable load case must provide the externally determined blade-failure ultimate load in kN."
    ] ;
    sh:property [
        sh:path nltl:combinedAxialBendingTorsionLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN>
            ]
        ] ;
        sh:message "The applicable load case must provide the externally determined combined axial, bending, and torsion load in kN."
    ] .

gen:TRF112ShaftComponentShape
    a sh:NodeShape ;
    sh:class nltl:propellerShaftLineComponent ;
    sh:property [
        sh:path nltl:bendingYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 1.0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ] ;
        sh:message "The externally calculated bending yield safety factor must be at least 1.0."
    ] ;
    sh:property [
        sh:path nltl:torsionalYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 1.0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ] ;
        sh:message "The externally calculated torsional yield safety factor must be at least 1.0."
    ] .
<END_SHACL>
