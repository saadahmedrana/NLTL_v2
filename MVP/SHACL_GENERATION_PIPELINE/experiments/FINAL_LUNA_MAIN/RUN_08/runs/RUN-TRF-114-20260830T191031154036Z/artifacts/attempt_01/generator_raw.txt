<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF114ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide a single ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF114ImpactCaseShape ;
        sh:message "The ship shall provide at least one applicable thruster ice impact load case."
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical impact case requires a single valid equivalent impact sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?value .
                    ?radius qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] .

gen:TRF114ImpactCaseShape
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadCaseIdentifier ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case shall reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case shall reference TRAFICOM Table 6-16 for the impact load case."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF114ThrusterBodyShape ;
        sh:message "The impact case shall identify the thruster body to which it applies."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The impact case shall provide the externally assessed thruster ice impact demand in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case shall provide evidence for the assessed loaded area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Non-hemispherical geometry shall include correspondence evidence."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The externally assessed thruster resistance capacity shall satisfy the impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:thrusterImpactCaseAppliesToBody ?body .
                ?body nltl:thrusterResistanceCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (datatype(?demandValue) = xsd:decimal)
                FILTER (datatype(?capacityValue) = xsd:decimal)
                FILTER (?capacityValue < ?demandValue)
            }
        """
    ] .

gen:TRF114ThrusterBodyShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thrusterResistanceCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The thruster body shall provide its externally assessed resistance capacity in kilonewtons."
    ] .
<END_SHACL>
