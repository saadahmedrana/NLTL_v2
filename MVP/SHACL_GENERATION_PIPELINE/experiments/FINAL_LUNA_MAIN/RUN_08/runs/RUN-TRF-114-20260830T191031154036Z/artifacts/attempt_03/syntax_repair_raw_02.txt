<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF114ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:class qudt:QuantityValue ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node gen:TRF114LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical contact case requires a valid equivalent impact sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification
                    nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?numericValue .
                    ?radius qudt:unit <http://qudt.org/vocab/unit/M> .
                }
            }
        """
    ] .

gen:TRF114LoadCaseShape a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:traficomTable6Dash3Reference
            nltl:iacsUrI2Table3
            nltl:iacsUrI2Table8
            nltl:traficomTable4Dash8
            nltl:traficomTable6Dash10
            nltl:traficomTable6Dash14
            nltl:traficomTable6Dash16Reference
            nltl:traficomTable6Dash6
            nltl:traficomTable6Dash7
            nltl:traficomTable6Dash8
            nltl:traficomTable6Dash9
        )
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:traficomTable6Dash16Reference
            nltl:iacsUrI2Table3
            nltl:iacsUrI2Table8
            nltl:traficomTable4Dash8
            nltl:traficomTable6Dash10
            nltl:traficomTable6Dash14
            nltl:traficomTable6Dash3Reference
            nltl:traficomTable6Dash6
            nltl:traficomTable6Dash7
            nltl:traficomTable6Dash8
            nltl:traficomTable6Dash9
        )
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:thrusterBody ;
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:hemisphericalContactGeometry
            nltl:nonHemisphericalContactGeometry
        )
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical contact case requires contact geometry correspondence evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:contactGeometryClassification
                    nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:contactGeometryCorrespondenceEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The externally assessed thruster resistance capacity must satisfy the thruster ice impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:thrusterImpactCaseAppliesToBody ?body .
                ?body nltl:thrusterResistanceCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?demandValue > ?capacityValue)
            }
        """
    ] .
<END_SHACL>
