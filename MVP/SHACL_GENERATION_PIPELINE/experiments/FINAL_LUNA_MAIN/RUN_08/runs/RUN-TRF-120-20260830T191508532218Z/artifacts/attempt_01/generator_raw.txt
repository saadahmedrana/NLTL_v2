<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "A damping value must be recorded for the thruster natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloGM
            ]
        ] ;
        sh:message "Water added mass must be supplied as a single decimal quantity in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:HZ
            ]
        ] ;
        sh:message "Thruster natural frequency must be supplied as a single decimal quantity in hertz."
    ] .

gs:ThrusterBodyShape
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloN-PER-M
            ]
        ] ;
        sh:message "Ship attachment stiffness must be supplied as a single decimal quantity in kilonewtons per metre."
    ] ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:qualifiedValueShape [
            sh:node gs:LongitudinalCaseShape
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedValueShapesDisjoint true ;
        sh:message "A longitudinal-direction natural-frequency calculation case is required."
    ] ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:qualifiedValueShape [
            sh:node gs:TransverseCaseShape
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedValueShapesDisjoint true ;
        sh:message "A transverse-direction natural-frequency calculation case is required."
    ] .

gs:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:longitudinalDirection
            nltl:transverseDirection
        ) ;
        sh:message "Each natural-frequency calculation case must identify a longitudinal or transverse vibration direction."
    ] .

gs:LongitudinalCaseShape
    a sh:NodeShape ;
    sh:node gs:CalculationCaseShape ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:longitudinalDirection
    ] .

gs:TransverseCaseShape
    a sh:NodeShape ;
    sh:node gs:CalculationCaseShape ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:transverseDirection
    ] .
<END_SHACL>
