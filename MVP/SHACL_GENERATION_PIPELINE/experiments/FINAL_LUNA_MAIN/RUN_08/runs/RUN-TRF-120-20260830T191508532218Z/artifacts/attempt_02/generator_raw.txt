<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF120_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide damping taken into account for thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the thruster global natural frequency in hertz."
    ] .

nltl:TRF120_ThrusterBodyShape a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-PER-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The thruster body must provide ship attachment stiffness in kilonewtons per metre."
    ] ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:minCount 1 ;
        sh:node nltl:TRF120_CalculationCaseShape ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:vibrationDirection ;
                sh:hasValue nltl:longitudinalDirection ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedValueShapesDisjoint true ;
        sh:message "The thruster body must include a longitudinal natural-frequency estimation case."
    ] ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:minCount 1 ;
        sh:node nltl:TRF120_CalculationCaseShape ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:vibrationDirection ;
                sh:hasValue nltl:transverseDirection ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedValueShapesDisjoint true ;
        sh:message "The thruster body must include a transverse natural-frequency estimation case."
    ] .

nltl:TRF120_CalculationCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:longitudinalDirection
            ]
            [
                sh:hasValue nltl:transverseDirection
            ]
        ) ;
        sh:message "Each natural-frequency calculation case must be identified as longitudinal or transverse."
    ] .
<END_SHACL>
