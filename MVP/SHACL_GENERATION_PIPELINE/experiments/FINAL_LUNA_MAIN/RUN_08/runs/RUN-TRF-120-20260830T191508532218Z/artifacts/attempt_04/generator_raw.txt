<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF120_NoEstimationCaseShape>
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?directCase .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?thrusterBody .
                    ?thrusterBody nltl:hasNaturalFrequencyCalculationCase ?componentCase .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF120_CalculationCaseShape>
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:longitudinalDirection
            nltl:transverseDirection
        ) ;
        sh:message "Each thruster natural-frequency calculation case must specify either longitudinal or transverse direction."
    ] .

<urn:nltl:generated-shape:TRF120_PopulatedShipShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:damping ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Damping must be recorded when a thruster natural-frequency estimation case is present."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloGM
            ]
        ] ;
        sh:message "Water added mass, with the canonical kilogram unit, must be recorded for each represented estimation structure."
    ] ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloN-PER-M
            ]
        ] ;
        sh:message "Ship attachment stiffness, with the canonical kilonewton-per-metre unit, must be recorded for each represented estimation structure."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:HZ
            ]
        ] ;
        sh:message "Thruster natural frequency, with the canonical hertz unit, must be recorded for each represented estimation structure."
    ] .

<urn:nltl:generated-shape:TRF120_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF120_NoEstimationCaseShape>
        <urn:nltl:generated-shape:TRF120_PopulatedShipShape>
    ) ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node <urn:nltl:generated-shape:TRF120_CalculationCaseShape>
    ] ;
    sh:property [
        sh:path (
            nltl:hasComponent
            nltl:hasNaturalFrequencyCalculationCase
        ) ;
        sh:node <urn:nltl:generated-shape:TRF120_CalculationCaseShape>
    ] .
<END_SHACL>
