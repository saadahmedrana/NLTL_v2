<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF123ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node gen:TRF123OccasionalForceLoadCaseShape ;
        sh:message "The ship must provide an occasional-force load case for the required readiness assessment."
    ] .

gen:TRF123OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The occasional-force load case must explicitly confirm that propeller blades are excluded."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The load case must provide the externally calculated component yield safety factor as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:node gen:TRF123ShipComponentShape ;
        sh:message "The occasional-force load case must identify each assessed component."
    ] ;
    sh:sparql [
        sh:message "The externally calculated occasional-force component stress, including the required safety margin, must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .
                $this nltl:componentYieldSafetyFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factor .
                ?component nltl:occasionalForceComponentStress ?stressQuantity2 .
                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?stress * ?factor > ?yield)
            }
        """
    ] .

gen:TRF123ShipComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each assessed component must provide its externally calculated occasional-force stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:class nltl:materialProperties ;
        sh:minCount 1 ;
        sh:node gen:TRF123MaterialPropertiesShape ;
        sh:message "Each assessed component must identify its material properties."
    ] .

gen:TRF123MaterialPropertiesShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The material properties must provide the component material yield strength in MPa."
    ] .
<END_SHACL>
