<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF130Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:inletChest ;
        sh:minCount 1 ;
        sh:qualifiedValueShape nltl:TRF130StandardChestShape ;
        sh:qualifiedMinCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:hasValue false ;
                sh:datatype xsd:boolean ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:class nltl:inletChest ;
                sh:qualifiedValueShape nltl:TRF130StandardChestShape ;
                sh:qualifiedMinCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:class nltl:inletChest ;
                sh:qualifiedValueShape nltl:TRF130AlternatingChestShape ;
                sh:qualifiedMinCount 2
            ]
        ]
    ) .

nltl:TRF130PowerQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF130AreaQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF130VolumeQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M3 ;
        sh:minCount 1 ;
        sh:maxCount 1

nltl:TRF130RatioQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF130StandardChestShape a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:hasValue "near_centreline_and_well_aft_if_possible" ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:TRF130RatioQuantityShape
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this .
                ?ship nltl:totalRelevantEnginePower ?powerQuantity .
                ?powerQuantity qudt:numericValue ?power .
                $this nltl:recommendedChestVolume ?volumeQuantity .
                ?volumeQuantity qudt:numericValue ?volume .
                FILTER (
                    ABS(?volume - (?power / 750)) >
                    (0.000001 * ABS(?power / 750) + 0.000001)
                )
            }
        """ ;
        sh:message "The recommended chest volume shall approximately equal total relevant engine power divided by 750."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:inletPipeArea ?inletQuantity .
                ?inletQuantity qudt:numericValue ?inletArea .
                $this nltl:strainerOpenArea ?strainerQuantity .
                ?strainerQuantity qudt:numericValue ?strainerArea .
                FILTER (?strainerArea < (4 * ?inletArea))
            }
        """ ;
        sh:message "The strainer open area shall be at least four times the inlet pipe area."
    ] .
