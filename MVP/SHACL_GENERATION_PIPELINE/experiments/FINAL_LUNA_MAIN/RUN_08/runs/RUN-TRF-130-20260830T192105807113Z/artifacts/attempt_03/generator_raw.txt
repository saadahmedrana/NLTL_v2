gen:CommonChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:node gen:RatioQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The strainer open area must be at least four times the inlet pipe area, and the declared open-area ratio must correspond to those areas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:strainerOpenArea ?openNode ;
                      nltl:inletPipeArea ?pipeNode ;
                      nltl:strainerOpenAreaRatio ?ratioNode .
                ?openNode qudt:numericValue ?open .
                ?pipeNode qudt:numericValue ?pipe .
                ?ratioNode qudt:numericValue ?ratio .
                BIND(?open / ?pipe AS ?expectedRatio)
                FILTER(
                    ?open < 4 * ?pipe ||
                    ?ratio < ?expectedRatio - (0.000001 * ?expectedRatio) - 0.000001 ||
                    ?ratio > ?expectedRatio + (0.000001 * ?expectedRatio) + 0.000001
                )
            }
        """
    ] .

gen:StandardChestShape
    a sh:NodeShape ;
    sh:node gen:CommonChestShape ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The recommended and actual sea-chest volumes must correspond, within tolerance, to total relevant engine power divided by 750." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:recommendedChestVolume ?recommendedNode ;
                      nltl:seaChestVolume ?seaNode .
                ?ship nltl:hasComponent $this ;
                      nltl:totalRelevantEnginePower ?powerNode .
                ?recommendedNode qudt:numericValue ?recommended .
                ?seaNode qudt:numericValue ?sea .
                ?powerNode qudt:numericValue ?power .
                BIND(?power / 750 AS ?expected)
                FILTER(
                    ?recommended < ?expected - (0.000001 * ?expected) - 0.000001 ||
                    ?recommended > ?expected + (0.000001 * ?expected) + 0.000001 ||
                    ?sea < ?expected - (0.000001 * ?expected) - 0.000001 ||
                    ?sea > ?expected + (0.000001 * ?expected) + 0.000001
                )
            }
        """
    ] .

        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
