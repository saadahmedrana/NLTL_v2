            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this .
                ?ship nltl:totalRelevantEnginePower ?powerNode .
                ?powerNode qudt:numericValue ?power .
                $this nltl:recommendedChestVolume ?volumeNode .
                ?volumeNode qudt:numericValue ?volume .
                $this nltl:strainerOpenArea ?openAreaNode .
                ?openAreaNode qudt:numericValue ?openArea .
                $this nltl:inletPipeArea ?pipeAreaNode .
                ?pipeAreaNode qudt:numericValue ?pipeArea .
                $this nltl:strainerOpenAreaRatio ?ratioNode .
                ?ratioNode qudt:numericValue ?ratio .
                BIND(?power / 750 AS ?expectedVolume)
                BIND(?openArea / ?pipeArea AS ?expectedRatio)
                FILTER(
                    ?volume >= ?expectedVolume - (0.000001 * ?expectedVolume + 0.000001) &&
                    ?volume <= ?expectedVolume + (0.000001 * ?expectedVolume + 0.000001)
                )
                FILTER(?openArea >= 4 * ?pipeArea)
                FILTER(?ratio >= 4)
                FILTER(
                    ?ratio >= ?expectedRatio - (0.000001 * ?expectedRatio + 0.000001) &&
                    ?ratio <= ?expectedRatio + (0.000001 * ?expectedRatio + 0.000001)
                )
            }
        """
