<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Average ice pressure must be supplied as a single MPa quantity."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node gen:DegreeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Buttock angle must be supplied as a single degree quantity."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node gen:AspectRatioQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Load patch aspect ratio must equal 3.6 and use a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Non-bow load patch height must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Non-bow load patch length must be supplied as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node gen:DegreeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Normal frame angle must be supplied as a single degree quantity."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability, when represented, must be a single boolean."
    ] ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string ;
        sh:maxCount 1 ;
        sh:message "Hull area, when represented, must be a single string value."
    ] ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:node gen:I2_009_HullAreaAssignmentShape ;
        sh:minCount 1 ;
        sh:message "A hull-area assignment for the other ice-strengthened area must be supplied."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When bow-form applicability is true, buttock angle must be greater than 0 degrees and less than 80 degrees, and normal frame angle must be greater than 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability true ;
                      nltl:buttockAngle ?gamma ;
                      nltl:normalFrameAngle ?beta .
                ?gamma qudt:numericValue ?gammaValue .
                ?beta qudt:numericValue ?betaValue .
                FILTER (
                    ?gammaValue <= 0 ||
                    ?gammaValue >= 80 ||
                    ?betaValue <= 10
                )
            }
        """
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AspectRatioQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 3.6 ;
        sh:maxInclusive 3.6
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_009_HullAreaAssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:in ( nltl:otherIceStrengthenedAreaValue ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The hull-area assignment must identify the other ice-strengthened area value."
    ] .
<END_SHACL>
