    sh:sparql [
        sh:message "The peak pressure factor lookup case must be the same table lookup case, reference IACS UR I2 Table 3, and provide lookup selection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasPeakPressureFactorLookupCase ?case .
                $this nltl:hasTableLookupCase ?case .
                FILTER NOT EXISTS {
                    ?case nltl:tableReference nltl:iacsUrI2Table3 .
                }
                UNION

                    FILTER NOT EXISTS {

                        ?evidence a nltl:evidenceArtifact .
                    }
                }
            }
        """
    ] ;
    sh:sparql [

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                ?height qudt:numericValue ?heightValue .
                ?length qudt:numericValue ?lengthValue .
                BIND ((?forceValue / (1000 * ?heightValue * ?lengthValue)) AS ?expectedValue)
                BIND (1e-9 + (1e-6 * ABS(?expectedValue)) AS ?tolerance)
                FILTER (ABS(?averageValue - ?expectedValue) > ?tolerance)
            }
        """
