<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_022_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:node g:I2_022_PlatedShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable plating requires average ice pressure, peak pressure factor, yield strength, and load patch height on the ship, each as a decimal quantity with its canonical unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:platingHullAreaClassification ?area .
                FILTER (
                    ?area IN (
                        nltl:bowIntermediateBottomHullArea,
                        nltl:midbodyBottomHullArea,
                        nltl:sternBottomHullArea
                    )
                )
                ?plating nltl:framingAngleOmega ?angle .
                ?angle qudt:numericValue ?angleValue ;
                       qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (datatype(?angleValue) = xsd:decimal)
                FILTER (?angleValue >= 70)

                FILTER (
                    NOT EXISTS {
                        $this nltl:averageIcePressure ?averagePressure .
                        ?averagePressure qudt:numericValue ?averageValue ;
                                          qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?averageValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:peakPressureFactor ?peakFactor .
                        ?peakFactor qudt:numericValue ?peakValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?peakValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yield .
                        ?yield qudt:numericValue ?yieldValue ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?yieldValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:loadPatchHeight ?patchHeight .
                        ?patchHeight qudt:numericValue ?heightValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?heightValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .

g:I2_022_PlatedShape
    a sh:NodeShape ;
    sh:or (
        g:I2_022_ApplicablePlatingShape
        g:I2_022_NonApplicablePlatingShape
    ) .

g:I2_022_ApplicablePlatingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:platingHullAreaClassification ;
        sh:hasValue nltl:bowIntermediateBottomHullArea ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 70 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:platingSupportedByStructuralMember ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:frameSpacing ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;

                    ]

            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1

                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;

            sh:class qudt:QuantityValue ;

        ]
    ] .
