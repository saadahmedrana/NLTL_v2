nltl:GeneratedI2_030CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node nltl:GeneratedI2_030InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node nltl:GeneratedI2_030InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Interpolation endpoints are required when the local-frame area exceeds the attached-plate-flange area or when the framing angle is oblique." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this .
                {
                    ?ship nltl:hasStructuralMember ?member .
                    ?member nltl:netLocalFrameFlangeArea ?areaNode .
                    ?areaNode qudt:numericValue ?area .
                    ?member nltl:netAttachedShellPlateThickness ?thicknessNode .
                    ?thicknessNode qudt:numericValue ?thickness .
                    ?member nltl:frameSpacing ?spacingNode .
                    ?spacingNode qudt:numericValue ?spacing .
                    FILTER (100 * ?area > 1000 * ?thickness * ?spacing)
                }
                UNION
                {
                    ?ship nltl:hasStructuralMember ?member2 .
                    ?member2 nltl:hasLocalFrameSectionCalculationCase ?localCase .
                    ?localCase nltl:sectionCalculationCasePlating ?plating .
                    ?plating nltl:framingAngleOmega ?angleNode .
                    ?angleNode qudt:numericValue ?angle .
                    FILTER (?angle > 20 && ?angle < 70)
                }
                FILTER NOT EXISTS {
                    $this nltl:interpolationLowerEndpoint ?lower .
                    ?lower a nltl:interpolationPoint .
                    $this nltl:interpolationUpperEndpoint ?upper .
                    ?upper a nltl:interpolationPoint .
                }
            }
        """
    ] .
