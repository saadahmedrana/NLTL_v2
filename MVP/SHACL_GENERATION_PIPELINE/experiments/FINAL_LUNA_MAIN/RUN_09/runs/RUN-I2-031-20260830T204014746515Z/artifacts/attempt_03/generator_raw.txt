<urn:nltl:generated-shape:I2_031_StructuralMemberShape>
    a sh:NodeShape ;
    sh:or (
        <urn:nltl:generated-shape:I2_031_BottomMemberShape>
        <urn:nltl:generated-shape:I2_031_SideMemberShape>
    ) ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:in (
            nltl:bottomStructureLocation
            nltl:sideStructureLocation
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2_031_LoadCaseShape>
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node <urn:nltl:generated-shape:I2_031_QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must provide its midspan plastic-collapse load in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2_031_QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must provide one plastic strength in kN."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand must not exceed plastic strength, and plastic strength must equal the midspan plastic-collapse load within the stated tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:midspanPlasticCollapseLoad ?midspanQuantity ;
                      nltl:plasticStrength ?strengthQuantity ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .

                ?midspanQuantity qudt:numericValue ?midspanValue .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .

                FILTER (
                    ?demandValue > ?strengthValue
                    ||
                    (
                        (?midspanValue >= 0 && ?midspanValue <= 0.000000000001 &&
                         (?strengthValue < ?midspanValue - 0.000001 ||
                          ?strengthValue > ?midspanValue + 0.000001))
                        ||
                        (?midspanValue >= 0 && ?midspanValue > 0.000000000001 &&
                         (?strengthValue < ?midspanValue - (0.000001 * ?midspanValue) ||
                          ?strengthValue > ?midspanValue + (0.000001 * ?midspanValue)))
                        ||
                        (?midspanValue < 0 && (0 - ?midspanValue) <= 0.000000000001 &&
                         (?strengthValue < ?midspanValue - 0.000001 ||
                          ?strengthValue > ?midspanValue + 0.000001))
                        ||
                        (?midspanValue < 0 && (0 - ?midspanValue) > 0.000000000001 &&
                         (?strengthValue < ?midspanValue - (0.000001 * (0 - ?midspanValue)) ||
                          ?strengthValue > ?midspanValue + (0.000001 * (0 - ?midspanValue))))
                    )
                )
            }
        """
    ] .
