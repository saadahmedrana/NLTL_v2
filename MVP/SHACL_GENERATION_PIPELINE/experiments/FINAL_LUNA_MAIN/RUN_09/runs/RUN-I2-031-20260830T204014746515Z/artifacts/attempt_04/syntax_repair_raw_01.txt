@prefix gen: <urn:nltl:generated-shape:> .

@prefix qudt: <http://qudt.org/schema/qudt/> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the patch load in kN."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:bottomStructureLocation
                    nltl:sideStructureLocation
                ) ;
                sh:message "Each applicable structural member must identify a bottom or side structure location."
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:structuralLocation ;
                        sh:hasValue nltl:bottomStructureLocation ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:hasStructuralMemberLoadCase ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path nltl:combinedShearAndBendingDemand ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ] ;
                            sh:property [
                                sh:path nltl:loadPatchOrientation ;
                                sh:hasValue nltl:loadPatchHeightParallelToFrame ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:message "A bottom-structure load case must apply the patch-load dimension parallel to the frame."
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:structuralLocation ;
                        sh:hasValue nltl:sideStructureLocation ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:transverseFrameOrientation ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                        sh:message "A side structural member must be identified as transverse."
                    ] ;
                    sh:property [
                        sh:path nltl:hasStructuralMemberLoadCase ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path nltl:combinedShearAndBendingDemand ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:datatype xsd:decimal ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1
                                    ]
                                ]
                            ]
                        ]
                    ]
                ]
            ) ;
            sh:property [
                sh:path nltl:midspanPlasticCollapseLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The member must provide its midspan plastic-collapse load in kN."
            ] ;
            sh:property [
                sh:path nltl:plasticStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The member must provide its plastic strength in kN."
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "For every structural-member load case, combined shear and bending demand must not exceed plastic strength, and plastic strength must match the midspan plastic-collapse load within a relative tolerance of 1e-6 or an absolute tolerance of 1e-9 near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:plasticStrength ?strengthNode .
                        ?strengthNode qudt:numericValue ?strength .
                        $this nltl:midspanPlasticCollapseLoad ?midspanNode .
                        ?midspanNode qudt:numericValue ?midspan .
                        $this nltl:hasStructuralMemberLoadCase ?loadCase .
                        ?loadCase nltl:combinedShearAndBendingDemand ?demandNode .
                        ?demandNode qudt:numericValue ?demand .
                        FILTER (
                            ?demand > ?strength
                            ||
                            (
                                ?midspan > 0 &&
                                (
                                    ?strength < (?midspan - (?midspan * 0.000001)) ||
                                    ?strength > (?midspan + (?midspan * 0.000001))
                                )
                            )
                            ||
                            (
                                ?midspan <= 0 &&
                                ?midspan >= 0 &&
                                (
                                    ?strength < (?midspan - 0.000000001) ||
                                    ?strength > (?midspan + 0.000000001)
                                )
                            )
                            ||
                            (
                                ?midspan < 0 &&
                                (
                                    ?strength < (?midspan - ((0 - ?midspan) * 0.000001)) ||
                                    ?strength > (?midspan + ((0 - ?midspan) * 0.000001))
                                )
                            )
                        )
                    }
                """
            ]
        ]
    ] .
