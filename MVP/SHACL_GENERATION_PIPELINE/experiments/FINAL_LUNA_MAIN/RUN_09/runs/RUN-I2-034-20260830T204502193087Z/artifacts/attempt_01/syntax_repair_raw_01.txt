<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:I2_034_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:nodeKind sh:IRI
    ] .

nltlgen:I2_034_StructuralMemberShape a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMember ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state whether it is longitudinally oriented."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:structuralLocationValue ;
        sh:message "Each represented structural member must identify its structural location."
    ] ;
    sh:or (
        [
            sh:path nltl:plasticStrength ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:node nltlgen:I2_034_KiloNewtonQuantityShape
        ]
        [
            sh:path nltl:midspanPlasticCollapseLoad ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:node nltlgen:I2_034_KiloNewtonQuantityShape
        ]
    ) ;
    sh:message "A represented structural member must provide either its plastic strength or its midspan plastic collapse load." ;
    sh:sparql [
        sh:message "For a longitudinal frame in a side structure, the combined shear-and-bending demand must not exceed the plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:hasStructuralMemberLoadCase ?case .
                ?case nltl:combinedShearAndBendingDemand ?demandNode .
                ?demandNode <http://qudt.org/schema/qudt/numericValue> ?demand .
                $this nltl:plasticStrength ?strengthNode .
                ?strengthNode <http://qudt.org/schema/qudt/numericValue> ?strength .
                FILTER (?demand > ?strength)
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a longitudinal frame in a side structure, the combined shear-and-bending demand must not exceed the midspan plastic collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:hasStructuralMemberLoadCase ?case .
                ?case nltl:combinedShearAndBendingDemand ?demandNode .
                ?demandNode <http://qudt.org/schema/qudt/numericValue> ?demand .
                $this nltl:midspanPlasticCollapseLoad ?strengthNode .
                ?strengthNode <http://qudt.org/schema/qudt/numericValue> ?strength .
                FILTER (?demand > ?strength)
            }
        """
    ] .

nltlgen:I2_034_LoadCaseShape a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMemberLoadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:I2_034_KiloNewtonQuantityShape ;
        sh:message "Each represented load case must provide its combined shear-and-bending demand."
    ] .

nltlgen:I2_034_KiloNewtonQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must contain one decimal numeric value."
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use the kN unit."
    ] .
<END_SHACL>
