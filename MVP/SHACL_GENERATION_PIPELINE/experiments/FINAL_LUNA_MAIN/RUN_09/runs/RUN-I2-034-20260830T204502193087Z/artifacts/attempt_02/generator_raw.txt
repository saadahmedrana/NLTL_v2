<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-034-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type using a canonical frame profile type value."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each longitudinal local frame in a side structure, combined shear-and-bending demand must not exceed the member plastic strength or midspan plastic collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:longitudinalFrameOrientation true ;
                        nltl:structuralLocation nltl:sideStructureLocation ;
                        nltl:plasticStrength ?strength ;
                        nltl:midspanPlasticCollapseLoad ?collapseLoad .
                ?strength qudt:numericValue ?strengthValue .
                ?collapseLoad qudt:numericValue ?collapseValue .
                ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                FILTER (
                    ?demandValue > ?strengthValue ||
                    ?demandValue > ?collapseValue
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-034-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must state whether it is longitudinally oriented."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must identify its structural location."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape> ;
        sh:message "The structural member must provide its plastic strength in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape> ;
        sh:message "The structural member must provide its midspan plastic collapse load in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-034-LoadCaseShape>
    ] .

<urn:nltl:generated-shape:I2-034-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-034-QuantityShape> ;
        sh:message "The load case must provide the combined shear-and-bending demand."
    ] .

<urn:nltl:generated-shape:I2-034-QuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-034-KiloNewtonQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
