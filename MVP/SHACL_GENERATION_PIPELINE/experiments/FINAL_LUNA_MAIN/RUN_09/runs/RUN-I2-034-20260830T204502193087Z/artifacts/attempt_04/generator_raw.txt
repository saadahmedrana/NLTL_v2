<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A force quantity must have exactly one KiloN unit."
    ] .

gen:LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case must provide exactly one combined shear-and-bending demand."
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The structural member must be identified as a longitudinal frame."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must be located in the side structure."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:LoadCaseShape ;
        sh:message "Each represented structural-member load case must satisfy the load-case constraint."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member must provide exactly one midspan plastic-collapse load."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member must provide exactly one plastic-strength value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every combined shear-and-bending demand must not exceed plastic strength, and plastic strength must equal the midspan plastic-collapse load within tolerance." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:longitudinalFrameOrientation true ;
          nltl:structuralLocation nltl:sideStructureLocation ;
          nltl:midspanPlasticCollapseLoad ?midspanQuantity ;
          nltl:plasticStrength ?strengthQuantity ;
          nltl:hasStructuralMemberLoadCase ?loadCase .

    ?midspanQuantity qudt:numericValue ?mv ;
                     qudt:unit <http://qudt.org/vocab/unit/KiloN> .

    ?strengthQuantity qudt:numericValue ?pv ;
                     qudt:unit <http://qudt.org/vocab/unit/KiloN> .

    ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
    ?demandQuantity qudt:numericValue ?dv ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloN> .

    FILTER (
        ?dv > ?pv ||
        ABS(?pv - ?mv) > (0.000001 * ABS(?mv) + 0.000000001)
    )
}
"""
    ] .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one canonical frame-profile-type relationship."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:message "Each represented structural member must satisfy the longitudinal side-frame requirement."
    ] .
<END_SHACL>
