<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:GaugedThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:NetThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:SteelRenewalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:node nltlgen:GaugedThicknessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:node nltlgen:NetThicknessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:iceStrengthenedStructureStatus ) ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasComponent nltl:iceStrengthenedStructureStatus ) ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For an ice-strengthened structure, steel renewal is required when gauged thickness is less than net thickness plus 0.0005 m." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:gaugedThickness ?gaugedNode ;
                              nltl:netThickness ?netNode ;
                              nltl:steelRenewalRequired ?renewal .
                        ?gaugedNode <http://qudt.org/schema/qudt/numericValue> ?gauged .
                        ?netNode <http://qudt.org/schema/qudt/numericValue> ?net .
                        FILTER (?gauged < (?net + 0.0005))
                        FILTER (?renewal != true)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:hasComponent nltl:iceStrengthenedStructureStatus ) ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:steelRenewalRequired ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .

<END_SHACL>
