<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "true"^^xsd:boolean ;
        sh:message "The ship must record that direct calculation is used."
    ] ;
    sh:sparql [
        sh:message "Every load carrying stringer or web frame linked to the ship must form part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member rdf:type ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage rdf:type nltl:grillageSystem .
                }
            }
        """
    ] .
<END_SHACL>
