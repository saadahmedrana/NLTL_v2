<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-064-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/schema/qudt/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one yield strength value in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-064-CalculationCaseShape> ;
        sh:message "The ship shall provide a calculation case for the applicable structural calculation."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-064-StructuralMemberShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal web shear stress shall be less than yield strength divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?shearQuantity .
                ?shearQuantity qudt:numericValue ?shearValue .
                FILTER (?shearValue >= (?yieldValue / 1.7320508075688772))
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal flange von Mises stress shall be less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldValue .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stressValue .
                FILTER (?stressValue >= (1.15 * ?yieldValue))
            }
        """
    ] .

<urn:nltl:generated-shape:I2-064-CalculationCaseShape>

        sh:hasValue nltl:linearCalculationMethodValue ;
        sh:message "The calculation case shall identify the linear calculation method."
    ] .

        sh:minCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberWeb ;

        sh:node <urn:nltl:generated-shape:I2-064-StructuralMemberWebShape>

    sh:property [
        sh:path nltl:hasStructuralMemberFlange ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-064-StructuralMemberFlangeShape>
    ] .
