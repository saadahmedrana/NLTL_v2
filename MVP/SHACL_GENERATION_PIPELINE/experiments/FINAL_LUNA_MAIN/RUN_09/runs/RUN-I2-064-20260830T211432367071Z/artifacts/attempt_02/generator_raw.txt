<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-064-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:I2-064-LinearBranch>
        <urn:nltl:generated-shape:I2-064-NonlinearBranch>
    ) .

<urn:nltl:generated-shape:I2-064-LinearBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-064-LinearCalculationCaseShape>
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-064-YieldStrengthShape>
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-064-StructuralMemberShape>
    ] ;
    sh:sparql [
        sh:message "For a linear calculation, nominal web shear stress must be less than yield strength divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                ?yieldQuantity qudt:unit unit:MegaPA .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?shearQuantity .
                ?shearQuantity qudt:numericValue ?shear .
                ?shearQuantity qudt:unit unit:MegaPA .
                FILTER (?shear >= (?yield / 1.7320508075688772))
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a linear calculation, nominal flange von Mises stress must be less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                ?yieldQuantity qudt:unit unit:MegaPA .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .
                ?stressQuantity qudt:unit unit:MegaPA .
                FILTER (?stress >= (1.15 * ?yield))
            }
        """
    ] .
