<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:class nltl:calculationMethodValue ;
        sh:or [
            rdf:first [
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            rdf:rest [
                rdf:first [
                    sh:hasValue nltl:directCalculationMethodValue
                ] ;
                rdf:rest [
                    rdf:first [
                        sh:hasValue nltl:linearCalculationMethodValue
                    ] ;
                    rdf:rest [
                        rdf:first [
                            sh:hasValue nltl:prescribedAnalyticalProcedure
                        ] ;
                        rdf:rest rdf:nil
                    ]
                ]
            ]
        ] ;
        sh:message "Each calculation case must identify an allowed calculation method."
    ] .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:CalculationCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every nonlinear calculation case must provide the declared structural inputs, externally assessed deformation result, satisfied assessment result, and nonlinear-analysis acceptance evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER NOT EXISTS {
                    $this nltl:fractureMargin ?fracture .
                    ?fracture qudt:numericValue ?fractureValue ;
                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?fractureValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:majorBucklingMargin ?buckling .
                    ?buckling qudt:numericValue ?bucklingValue ;
                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?bucklingValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:yieldingMargin ?yielding .
                    ?yielding qudt:numericValue ?yieldingValue ;
                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?yieldingValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    ?case nltl:permanentLateralDeformation ?lateral .
                    ?lateral qudt:numericValue ?lateralValue ;
                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (datatype(?lateralValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    ?case nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                    ?outOfPlane qudt:numericValue ?outOfPlaneValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (datatype(?outOfPlaneValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    ?case nltl:relevantStructuralDimension ?dimension .
                    ?dimension qudt:numericValue ?dimensionValue ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (datatype(?dimensionValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    ?case nltl:minorPermanentDeformationAssessmentSatisfied ?satisfied .
                    FILTER (datatype(?satisfied) = xsd:boolean && ?satisfied = true)
                }

                FILTER NOT EXISTS {
                    ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                    ?evidence rdf:type nltl:evidenceArtifact .
                }
            }
        """
    ] .
<END_SHACL>
