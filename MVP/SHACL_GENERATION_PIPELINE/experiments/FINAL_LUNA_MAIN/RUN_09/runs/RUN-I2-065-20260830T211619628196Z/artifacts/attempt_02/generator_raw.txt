<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bucklingResponse ;
        sh:datatype xsd:string ;
        sh:message "Buckling response must be a string when supplied."
    ] ;
    sh:property [
        sh:path nltl:plasticDeformation ;
        sh:datatype xsd:string ;
        sh:message "Plastic deformation must be a string when supplied."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:hasValue nltl:nonlinearCalculationMethodValue
                    ] ;
                    sh:property [
                        sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                        sh:class nltl:evidenceArtifact ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:permanentLateralDeformation ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:permanentOutOfPlaneDeformation ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:relevantStructuralDimension ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:in (
                            nltl:directCalculationMethodValue
                            nltl:linearCalculationMethodValue
                            nltl:prescribedAnalyticalProcedure
                        )
                    ]
                ]
            )
        ]
    ] .

gs:NonlinearCaptureShape a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A nonlinear calculation case must be associated with ship-level buckling-response and plastic-deformation capture." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
                ?ship nltl:hasCalculationCase $this .
                FILTER (
                    NOT EXISTS {
                        ?ship nltl:bucklingResponse ?buckling .
                        FILTER (isLiteral(?buckling) && datatype(?buckling) = xsd:string)
                    }
                    ||
                    NOT EXISTS {
                        ?ship nltl:plasticDeformation ?plastic .
                        FILTER (isLiteral(?plastic) && datatype(?plastic) = xsd:string)
                    }
                )
            }
        """
    ] .
<END_SHACL>
