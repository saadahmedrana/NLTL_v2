<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-067-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:strengthContinuityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship with structural connections must provide one boolean strength-continuity status."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Strength continuity must be ensured at all structural connections." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:structuralConnection ?connection .
                FILTER NOT EXISTS {
                    $this nltl:strengthContinuityStatus true .
                }
            }
        """
    ] .
<END_SHACL>
