<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO002Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for at least thin first-year ice with possible old-ice inclusions and declared as not Category A must have Category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?designCondition ;
                      nltl:shipCategory ?category .
                FILTER (
                    ?designCondition = nltl:thinFirstYearIceWithPossibleOldIceInclusions
                    && ?category != nltl:polarShipCategoryA
                    && ?category != nltl:polarShipCategoryB
                )
            }
        """ ;
    ] .
<END_SHACL>
