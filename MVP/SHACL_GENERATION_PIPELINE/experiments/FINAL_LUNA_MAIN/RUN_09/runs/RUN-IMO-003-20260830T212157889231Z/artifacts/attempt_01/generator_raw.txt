<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-003-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its design ice condition to determine applicability of the Category C rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship designed for open water or ice conditions less severe than categories A and B, ship category must be Category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryC .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, no ship category other than Category C is permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
                $this nltl:shipCategory ?category .
                FILTER (?category != nltl:polarShipCategoryC)
            }
        """
    ] .
<END_SHACL>
