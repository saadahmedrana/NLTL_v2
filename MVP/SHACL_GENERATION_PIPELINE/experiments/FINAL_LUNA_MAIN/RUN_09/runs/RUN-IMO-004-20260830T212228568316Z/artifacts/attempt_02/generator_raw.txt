<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_004_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When the ice is sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m, the thickness must use metres and the ice type classification must be exactly first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                    nltl:winterGrowthCount ?growth ;
                    nltl:iceThickness ?thickness .

                ?thickness qudt:numericValue ?value .

                FILTER(DATATYPE(?growth) = xsd:integer)
                FILTER(DATATYPE(?value) = xsd:decimal)
                FILTER(?growth <= 1)
                FILTER(?value >= 0.3 && ?value <= 2.0)

                FILTER(
                    NOT EXISTS {
                        ?thickness qudt:unit unit:M
                    }
                    ||
                    EXISTS {
                        ?thickness qudt:unit ?thicknessUnit .
                        FILTER(?thicknessUnit != unit:M)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceTypeClassification nltl:firstYearIce
                    }
                    ||
                    EXISTS {
                        $this nltl:iceTypeClassification ?classification .
                        FILTER(?classification != nltl:firstYearIce)
                    }
                )
            }
        """
    ] .
<END_SHACL>
