<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-005-ice-free-condition>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship whose ice condition is 'ice free waters' must have totalIcePresent exactly false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "ice free waters"^^xsd:string)
                FILTER (
                    NOT EXISTS {
                        $this nltl:totalIcePresent ?falseValue .
                        FILTER (?falseValue = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:totalIcePresent ?trueValue .
                        FILTER (?trueValue = true)
                    }
                )
            }
        """
    ] .
<END_SHACL>
