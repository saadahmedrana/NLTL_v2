<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "An ice condition classification is required to determine whether the open-water condition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For the open-water ice condition, sea ice concentration must be less than 1/10 and no ice of land origin may be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:seaIceConcentration ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit qudt:UNITLESS .
                    $this nltl:iceOfLandOriginPresent ?landOriginIce .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value < 1 &&
                        datatype(?landOriginIce) = xsd:boolean &&
                        ?landOriginIce = false
                    )
                }
            }
        """
    ] .
<END_SHACL>
