<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007-OpenWaterIceConditionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:hasValue nltl:openWaterIceCondition
    ] ;
    sh:property [
        sh:path nltl:iceOfLandOriginPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:seaIceConcentration ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open water, sea ice concentration must be less than 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:seaIceConcentration ?quantity .
                    ?quantity qudt:numericValue ?value .
                    FILTER (?value < 1)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-007-NonOpenWaterIceConditionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice condition classification must identify a non-open-water condition for this applicability branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?classification .
                FILTER (?classification != nltl:openWaterIceCondition)
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-007-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:IMO-007-OpenWaterIceConditionShape>
        <urn:nltl:generated-shape:IMO-007-NonOpenWaterIceConditionShape>
    ) .
<END_SHACL>
