<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008-Static>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:sparql [
                sh:message "A ship classified as thin first-year ice must provide the applicable ice thickness." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        FILTER EXISTS {
                            $this nltl:iceTypeClassification nltl:thinFirstYearIce .
                        }
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:iceThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "0.3"^^xsd:decimal ;
                        sh:maxInclusive "0.7"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "A ship not classified as thin first-year ice is not subject to the thin first-year ice thickness range." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:iceTypeClassification nltl:thinFirstYearIce .
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
