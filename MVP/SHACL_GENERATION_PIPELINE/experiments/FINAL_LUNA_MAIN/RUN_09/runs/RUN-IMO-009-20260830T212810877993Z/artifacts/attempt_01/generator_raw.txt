@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO009BergyWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, land-origin ice concentration must be a dimensionless decimal less than 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition "bergy waters" .
                FILTER (
                    NOT EXISTS {
                        $this nltl:landOriginIceConcentration ?q .
                    }
                    ||
                    EXISTS {
                        $this nltl:landOriginIceConcentration ?q .
                        FILTER NOT EXISTS {
                            ?q a qudt:QuantityValue ;
                               qudt:numericValue ?v ;
                               qudt:unit unit:UNITLESS .
                            FILTER (datatype(?v) = xsd:decimal && ?v < 1)
                        }
                    }
                )
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, total ice concentration must be a dimensionless decimal no greater than 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition "bergy waters" .
                FILTER (
                    NOT EXISTS {
                        $this nltl:totalIceConcentration ?q .
                    }
                    ||
                    EXISTS {
                        $this nltl:totalIceConcentration ?q .
                        FILTER NOT EXISTS {
                            ?q a qudt:QuantityValue ;
                               qudt:numericValue ?v ;
                               qudt:unit unit:UNITLESS .
                            FILTER (datatype(?v) = xsd:decimal && ?v <= 1)
                        }
                    }
                )
            }
        """ ;
    ] .
