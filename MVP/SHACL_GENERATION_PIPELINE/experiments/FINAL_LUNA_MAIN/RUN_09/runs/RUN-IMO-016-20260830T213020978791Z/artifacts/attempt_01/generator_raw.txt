<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:node gs:IMO_016_CertificateFormShape ;
        sh:message "The ship shall have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:in ( "English" "French" "Spanish" )
            ]
        ]
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:approvedTranslationPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "A translation into English, French, or Spanish is required when the certificate language is another language."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A certificate language other than English, French, or Spanish requires an approved translation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (
                    ?language NOT IN ("English", "French", "Spanish")
                )
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent true .
                }
            }
        """
    ] .

gs:IMO_016_CertificateFormShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:class nltl:certificateFormModelValue ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate form shall correspond to the Appendix I model."
    ] .
<END_SHACL>
