<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO037ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryC ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasLoadingConditionCase ;
                sh:node gen:IMO037LoadingConditionCaseShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable loading condition must have a residual stability factor within 1e-6 of 1.0 or a confirmed applicable alternative-instrument residual stability outcome." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasLoadingConditionCase ?case .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                FILTER NOT EXISTS {
                    ?case nltl:residualStabilityFactorSI ?quantity .
                    ?quantity qudt:numericValue ?reportedValue ;
                              qudt:unit qudt:UNITLESS .
                    FILTER (
                        datatype(?reportedValue) = xsd:decimal &&
                        ABS(?reportedValue - 1.0) <= 0.000001
                    )
                }

                FILTER NOT EXISTS {
                    ?case nltl:alternativeInstrumentApplicable true ;
                          nltl:alternativeInstrumentResidualStabilityStatus true .
                }
            }
        """
    ] .

gen:IMO037LoadingConditionCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each loading condition case must identify the loading condition."
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Each loading condition case must provide result evidence."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:residualStabilityFactorSI ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path ( nltl:residualStabilityFactorSI qudt:unit ) ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path ( nltl:residualStabilityFactorSI qudt:numericValue ) ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:alternativeInstrumentApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
</END_SHACL>
