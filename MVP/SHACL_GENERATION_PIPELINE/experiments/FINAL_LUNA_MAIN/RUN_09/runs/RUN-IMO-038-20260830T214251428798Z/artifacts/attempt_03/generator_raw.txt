<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_038_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Damage-center position relative to maximum breadth is required."
    ] ;

    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Upper ice waterline length is required in metres."
    ] ;

    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Upper ice waterline draught is required in metres."
    ] ;

    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Longitudinal damage extent is required in metres."
    ] ;

    sh:property [
        sh:path nltl:transversePenetration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 760 ;
                sh:maxInclusive 760
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Transverse penetration must be exactly 760 mm."
    ] ;

    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Vertical damage extent is required in metres."
    ] ;

    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Selected vertical damage position is required in metres."
    ] ;

    sh:sparql [
        sh:message "Longitudinal damage extent must equal 4.5% or 1.5% of upper ice waterline length according to the damage-center position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:longitudinalDamageExtent ?extentNode .

                ?lengthNode qudt:numericValue ?length .
                ?extentNode qudt:numericValue ?actual .

                BIND(
                    IF(
                        ?forward = true,
                        0.045 * ?length,
                        0.015 * ?length
                    )
                    AS ?expected
                )

                FILTER (
                    ?actual <
                    (?expected - ((?expected * 0.000001) + 0.000000001))
                    ||
                    ?actual >
                    (?expected + ((?expected * 0.000001) + 0.000000001))
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Vertical damage extent must equal the lesser of 20% of upper ice waterline draught and longitudinal damage extent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode .

                ?draughtNode qudt:numericValue ?draught .
                ?longitudinalNode qudt:numericValue ?longitudinal .
                ?verticalNode qudt:numericValue ?actual .

                BIND(
                    IF(
                        (0.20 * ?draught) < ?longitudinal,
                        (0.20 * ?draught),
                        ?longitudinal
                    )
                    AS ?expected
                )

                FILTER (
                    ?actual <
                    (?expected - ((?expected * 0.000001) + 0.000000001))
                    ||
                    ?actual >
                    (?expected + ((?expected * 0.000001) + 0.000000001))
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Selected vertical damage position must be between the keel and 1.20 times the upper ice waterline draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .

                ?draughtNode qudt:numericValue ?draught .
                ?positionNode qudt:numericValue ?position .

                FILTER (
                    ?position < 0
                    ||
                    ?position > (1.20 * ?draught)
                )
            }
        """
    ] .
<END_SHACL>
