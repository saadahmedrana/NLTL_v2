<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_041_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Opening actuation type is required to determine whether the hydraulic-fluid protection condition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships with hydraulically operated hatches or doors shall provide means to prevent freezing or excessive viscosity of hydraulic fluids." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuationType .
                FILTER (?actuationType = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent true .
                }
            }
        """
    ] .
</END_SHACL>
