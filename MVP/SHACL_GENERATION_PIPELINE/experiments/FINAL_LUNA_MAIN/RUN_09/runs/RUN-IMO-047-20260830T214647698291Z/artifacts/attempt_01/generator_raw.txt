<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control shall be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:message "The combustion-air temperature shall be provided in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion shall be provided in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion shall be provided in degrees Celsius."
    ] ;
    sh:sparql [
        sh:message "The combustion-air temperature shall be within the manufacturer's minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperature ;
                      nltl:manufacturerMinAirTemperature ?minimum ;
                      nltl:manufacturerMaxAirTemperature ?maximum .
                ?temperature qudt:numericValue ?temperatureValue .
                ?minimum qudt:numericValue ?minimumValue .
                ?maximum qudt:numericValue ?maximumValue .
                FILTER (
                    ?temperatureValue < ?minimumValue ||
                    ?temperatureValue > ?maximumValue
                )
            }
        """
    ] .

gen:IMO_047_TemperatureQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
    ] .
<END_SHACL>
