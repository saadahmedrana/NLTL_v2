<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TemperatureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IMO_047_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:minCount 1 ;
        sh:message "At least one essential engine must be recorded."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:TemperatureQuantityShape ;
        sh:message "A combustion-air temperature in degrees Celsius is required."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:TemperatureQuantityShape ;
        sh:message "A manufacturer minimum air temperature in degrees Celsius is required."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:TemperatureQuantityShape ;
        sh:message "A manufacturer maximum air temperature in degrees Celsius is required."
    ] ;
    sh:sparql [
        sh:message "For every essential engine, combustion-air temperature must be within the manufacturer limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?engine .
                $this nltl:combustionAirTemperature ?temperatureQuantity .
                $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .
                ?temperatureQuantity qudt:numericValue ?temperature .
                ?minimumQuantity qudt:numericValue ?minimum .
                ?maximumQuantity qudt:numericValue ?maximum .
                FILTER (
                    ?temperature < ?minimum ||
                    ?temperature > ?maximum
                )
            }
        """
    ] .
<END_SHACL>
