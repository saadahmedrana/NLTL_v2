<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO048Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented exposed machinery or foundation material requires approved material evidence, an authorized approving authority, an accepted or equivalent-safety standard, and a polar service temperature basis." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:exposedMachineryOrFoundationMaterial ?material .

                FILTER NOT EXISTS {
                    $this nltl:materialApprovalStatus nltl:evidenceStateApproved
                }

                FILTER EXISTS {
                    $this nltl:materialApprovalStatus ?status .
                    FILTER (?status != nltl:evidenceStateApproved)
                }

                FILTER NOT EXISTS {
                    $this nltl:approvingAuthority ?authority .
                    FILTER (
                        ?authority IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                }

                FILTER EXISTS {
                    $this nltl:approvingAuthority ?authority .
                    FILTER (
                        ?authority NOT IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:approvalStandard ?standard .
                    FILTER (
                        ?standard IN (
                            "standard acceptable to the Organization",
                            "another standard offering an equivalent level of safety"
                        )
                    )
                }

                FILTER EXISTS {
                    $this nltl:approvalStandard ?standard .
                    FILTER (
                        ?standard NOT IN (
                            "standard acceptable to the Organization",
                            "another standard offering an equivalent level of safety"
                        )
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:polarServiceTemperature ?temperature .
                    ?temperature rdf:type qudt:QuantityValue .
                    ?temperature qudt:numericValue ?numericValue .
                    ?temperature qudt:unit <http://qudt.org/vocab/unit/DEG_C> .
                    FILTER (datatype(?numericValue) = xsd:decimal)
                }
            }
        """ ;
    ] .
<END_SHACL>
