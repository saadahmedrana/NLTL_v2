<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-048> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:exposedMachineryOrFoundationMaterial ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:polarServiceTemperature ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each exposed machinery or foundation material must have approved status, an authorized approving authority, an accepted or equivalent-safety standard, and a polar service temperature basis." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:exposedMachineryOrFoundationMaterial ?material .
                FILTER (
                    NOT EXISTS {
                        $this nltl:materialApprovalStatus nltl:evidenceStateApproved
                    }
                    ||
                    EXISTS {
                        $this nltl:materialApprovalStatus ?status .
                        FILTER (?status != nltl:evidenceStateApproved)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:approvingAuthority ?authority .
                        FILTER (
                            datatype(?authority) = xsd:string &&
                            STR(?authority) IN (
                                "Administration",
                                "recognized organization accepted by the Administration"
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:approvingAuthority ?authority .
                        FILTER (
                            datatype(?authority) != xsd:string ||
                            STR(?authority) NOT IN (
                                "Administration",
                                "recognized organization accepted by the Administration"
                            )
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:approvalStandard ?standard .
                        FILTER (
                            datatype(?standard) = xsd:string &&
                            STR(?standard) IN (
                                "standard acceptable to the Organization",
                                "another standard offering an equivalent level of safety"
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:approvalStandard ?standard .
                        FILTER (
                            datatype(?standard) != xsd:string ||
                            STR(?standard) NOT IN (
                                "standard acceptable to the Organization",
                                "another standard offering an equivalent level of safety"
                            )
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:polarServiceTemperature ?pst .
                        ?pst a qudt:QuantityValue ;
                             qudt:numericValue ?numericValue ;
                             qudt:unit unit:DEG_C .
                        FILTER (datatype(?numericValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:polarServiceTemperature ?pst .
                        FILTER NOT EXISTS {
                            ?pst a qudt:QuantityValue ;
                                 qudt:numericValue ?numericValue ;
                                 qudt:unit unit:DEG_C .
                            FILTER (datatype(?numericValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:polarServiceTemperature ?pst .
                        ?pst qudt:unit ?temperatureUnit .
                        FILTER (?temperatureUnit != unit:DEG_C)
                    }
                    ||
                    EXISTS {
                        $this nltl:polarServiceTemperature ?pst .
                        ?pst qudt:numericValue ?numericValue .
                        FILTER (datatype(?numericValue) != xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
