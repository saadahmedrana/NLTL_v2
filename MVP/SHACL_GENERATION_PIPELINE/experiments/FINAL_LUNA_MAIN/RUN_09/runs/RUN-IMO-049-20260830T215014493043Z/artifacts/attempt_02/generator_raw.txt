<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO049ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must identify the applicable machinery component type." ;
    ] ;
    sh:or (
        nltl:IMO049CategoryABCase
        nltl:IMO049CategoryCIceStrengthenedCase
        nltl:IMO049CategoryCNotIceStrengthenedCase
    ) .

nltl:IMO049CategoryABCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryA ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryB ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node nltl:IMO049ComponentABShape ;
    ] .

nltl:IMO049CategoryCIceStrengthenedCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node nltl:IMO049ComponentCShape ;
    ] .

nltl:IMO049CategoryCNotIceStrengthenedCase a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
    ] .

nltl:IMO049ComponentABShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node nltl:IMO049ApprovalABShape ;
        sh:message "Each applicable machinery component must have an approved scantling approval record." ;
    ] .

nltl:IMO049ComponentCShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node nltl:IMO049ApprovalCShape ;
        sh:message "Each applicable ice-strengthened category C machinery component must have an approved scantling approval record with operating ice context." ;
    ] .

nltl:IMO049ApprovalABShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
    ] .

nltl:IMO049ApprovalCShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:hasValue "acceptable standard adequate for operating ice type and concentration" ;
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:node nltl:IMO049UnitlessQuantityShape ;
    ] .

nltl:IMO049UnitlessQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
