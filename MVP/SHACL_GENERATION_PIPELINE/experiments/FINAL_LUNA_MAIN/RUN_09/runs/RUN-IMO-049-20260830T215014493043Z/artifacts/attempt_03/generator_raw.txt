        [ sh:and ( gen:IMO049InventoryShape gen:IMO049CategoryCNotIceStrengthenedShape ) ]
    ) .

    sh:property [

        sh:minCount 1 ;
        sh:message "The ship shall identify the applicable machinery component inventory."
    ] .

    sh:property [

        sh:node gen:IMO049ABComponentShape ;
        sh:message "Each applicable component of a category A ship shall have qualifying approval evidence."
    ] .

gen:IMO049CategoryBShape

        sh:path nltl:hasComponent ;
        sh:node gen:IMO049ABComponentShape ;
        sh:message "Each applicable component of a category B ship shall have qualifying approval evidence."
    ] .

    sh:property [

        sh:minCount 1

        sh:path nltl:shipIceStrengthened ;

gen:IMO049ABComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO049ABApprovalShape ;
        sh:message "An applicable machinery component shall have an approval record."
    ] .

gen:IMO049CComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO049CApprovalShape ;
        sh:message "An applicable machinery component shall have an approval record."
    ] .

gen:IMO049ABApprovalShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status shall be Approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "Approval shall be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval shall use an Organization-acceptable or equivalent-safety standard."
    ] .

gen:IMO049CApprovalShape
    a sh:NodeShape ;

        sh:path nltl:scantlingApprovalStatus ;

        sh:message "The scantling approval status shall be Approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "Approval shall be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;

            "acceptable standard adequate for operating ice type and concentration"

    ] ;
    sh:property [

                sh:maxCount 1

        ] ;
        sh:message "Category C approval shall record operating ice concentration as a unitless quantity."
    ] .
