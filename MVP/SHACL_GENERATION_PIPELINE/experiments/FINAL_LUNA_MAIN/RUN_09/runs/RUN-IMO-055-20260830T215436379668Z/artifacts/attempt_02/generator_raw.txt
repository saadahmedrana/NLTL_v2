<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-055> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:message "Valve exposure values must be boolean."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:message "Valve type values must be strings."
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:message "Ice accretion protection values must be boolean."
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean ;
        sh:message "Continuous accessibility values must be boolean."
    ] ;
    sh:sparql [
        sh:message "A ship must not contain contradictory boolean values for the valve exposure, ice accretion protection, or continuous accessibility facts." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this ?property ?value1 .
                $this ?property ?value2 .
                FILTER (
                    ?property IN (
                        nltl:valveExposed,
                        nltl:iceAccretionProtectionPresent,
                        nltl:continuouslyAccessible
                    )
                )
                FILTER (?value1 != ?value2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "An exposed isolating, pressure, or vacuum valve must have ice accretion protection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:valveExposed true ;
                      nltl:valveType ?valveType .
                FILTER (?valveType IN ("isolating", "pressure", "vacuum"))
                FILTER NOT EXISTS {
                    $this nltl:iceAccretionProtectionPresent true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An exposed isolating, pressure, or vacuum valve must remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:valveExposed true ;
                      nltl:valveType ?valveType .
                FILTER (?valveType IN ("isolating", "pressure", "vacuum"))
                FILTER NOT EXISTS {
                    $this nltl:continuouslyAccessible true .
                }
            }
        """
    ] .
<END_SHACL>
