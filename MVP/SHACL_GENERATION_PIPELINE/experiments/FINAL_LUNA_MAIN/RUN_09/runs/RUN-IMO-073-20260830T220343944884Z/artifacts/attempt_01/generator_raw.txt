    sh:targetClass nltl:ship ;

        sh:node <urn:nltl:generated-shape:IMO-073-LifeboatShape> ;
        sh:message "Each linked lifeboat must have a permitted lifeboat type."
    ] .

    sh:property [
        sh:path nltl:lifeboatTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboatTypeValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:partiallyEnclosedLifeboat
            nltl:totallyEnclosedLifeboat
        ) ;
        sh:message "The lifeboat type must be partially enclosed or totally enclosed."
    ] .
