<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_077_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The additional-equipment applicability indicator must be provided exactly once."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:personsCapacityRequirement ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:message "The persons capacity requirement is required when additional equipment is carried in the survival craft."
            ] ;
            sh:property [
                sh:path nltl:additionalEquipmentCapacityRequirement ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:message "The additional equipment capacity requirement is required when additional equipment is carried in the survival craft."
            ] ;
            sh:property [
                sh:path nltl:survivalCraftAvailableCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:message "The available survival-craft capacity is required when additional equipment is carried in the survival craft."
            ] ;
            sh:property [
                sh:path nltl:launchingApplianceCapacity ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloGM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Launching-appliance capacity must be provided as a kilogram quantity."
            ] ;
            sh:property [
                sh:path nltl:loadedSurvivalCraftRequirement ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloGM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The loaded survival-craft requirement must be provided as a kilogram quantity."
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:personsCapacityRequirement ?persons .
                        $this nltl:additionalEquipmentCapacityRequirement ?additional .
                        $this nltl:survivalCraftAvailableCapacity ?available .
                        FILTER (
                            datatype(?persons) = xsd:integer &&
                            datatype(?additional) = xsd:integer &&
                            datatype(?available) = xsd:integer &&
                            ?available < (?persons + ?additional)
                        )
                    }
                """ ;
                sh:message "Survival-craft available capacity must be at least the persons capacity requirement plus the additional equipment capacity requirement."
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX unit: <http://qudt.org/vocab/unit/>

                    SELECT $this
                    WHERE {
                        $this nltl:launchingApplianceCapacity ?launchNode .
                        ?launchNode qudt:numericValue ?launchCapacity ;
                                    qudt:unit unit:KiloGM .
                        $this nltl:loadedSurvivalCraftRequirement ?loadedNode .
                        ?loadedNode qudt:numericValue ?loadedRequirement ;
                                   qudt:unit unit:KiloGM .
                        FILTER (?launchCapacity < ?loadedRequirement)
                    }
                """ ;
                sh:message "Launching-appliance capacity must be at least the loaded survival-craft requirement."
            ]
        ]
        [
            sh:property [
                sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
