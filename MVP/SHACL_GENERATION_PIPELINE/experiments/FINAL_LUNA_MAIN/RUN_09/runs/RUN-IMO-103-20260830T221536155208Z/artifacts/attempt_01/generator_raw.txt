    ] .

        sh:datatype xsd:boolean ;

    sh:property [
        sh:path nltl:administrationPermissionStatus ;
        sh:datatype xsd:boolean ;

        sh:maxCount 1 ;
        sh:hasValue true ;

    ] ;
    sh:property [
        sh:path nltl:stcwConventionRegulationIi2ComplianceStatus ;

        sh:minCount 1 ;

                sh:datatype xsd:decimal ;

                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;

                sh:maxCount 1
            ]

        sh:message "Designated officers shall meet the applicable basic-training requirements for the ship type and operating ice condition." ;

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipType ?shipType ;
                      nltl:iceCondition ?iceCondition ;
                      nltl:iceConcentration ?quantity .
                ?quantity qudt:numericValue ?concentration .

                FILTER (
                    (
                        ?shipType IN ("passenger ship", "tanker")
                        && ?iceCondition NOT IN ("open waters", "bergy waters")
                    )
                    ||
                    (
                        ?shipType = "cargo ship other than tanker"
                        && ?concentration > "0.2"^^xsd:decimal
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:designatedOfficerBasicTrainingStatus true .
                }
            }
        """
    ] .
