<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_105_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:property [
        sh:path nltl:oilOrOilyMixtureDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:message "Oil or oily mixture discharge status must be a boolean when represented."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, oil or oily mixture discharge to sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters")
                FILTER NOT EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea
                        "false"^^xsd:boolean .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, no oil or oily mixture discharge to sea may be recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters")
                $this nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                FILTER (?discharge != "false"^^xsd:boolean)
            }
        """
    ] .
<END_SHACL>
