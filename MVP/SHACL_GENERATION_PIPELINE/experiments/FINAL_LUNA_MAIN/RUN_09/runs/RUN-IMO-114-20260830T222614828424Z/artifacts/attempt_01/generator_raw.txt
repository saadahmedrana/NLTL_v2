<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_114_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ship category must be specified as Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Construction date must be specified as a single date."
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether the carried noxious liquid substance requires a type 3 tank must be specified."
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Cargo tank type must be specified as a single string value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, where the carried noxious liquid substance requires a type 3 tank and the cargo tank type is 3, Administration approval and the applicable polar-operation certificate entry are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipCategory ?category .
    $this nltl:constructionDate ?constructionDate .
    $this nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 .
    $this nltl:cargoTankType ?cargoTankType .

    FILTER (
        (?category = nltl:polarShipCategoryA ||
         ?category = nltl:polarShipCategoryB) &&
        ?constructionDate >= "2017-01-01"^^xsd:date &&
        ?requiresType3 = true &&
        ?cargoTankType = "3"^^xsd:string
    )

    FILTER NOT EXISTS {
        $this nltl:administrationCarriageApprovalStatus nltl:evidenceStateApproved .
    }

    FILTER NOT EXISTS {
        $this nltl:applicableCertificatePolarOperationEntryPresent true .
    }

    FILTER NOT EXISTS {
        $this nltl:administrationCarriageApprovalStatus ?otherApproval .
        FILTER (?otherApproval != nltl:evidenceStateApproved)
    }

    FILTER NOT EXISTS {
        $this nltl:applicableCertificatePolarOperationEntryPresent ?otherEntry .
        FILTER (?otherEntry != true)
    }
}
""" ;
    ] .
<END_SHACL>
