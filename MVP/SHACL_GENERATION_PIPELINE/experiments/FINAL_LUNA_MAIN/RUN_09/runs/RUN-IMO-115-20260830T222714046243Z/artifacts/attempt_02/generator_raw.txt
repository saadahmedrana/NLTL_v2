gs:IMO_115_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage discharge is comminuted and disinfected, MARPOL compliance, the strict distance threshold, and the applicable ice-concentration distance record are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType .
                FILTER (STR(?dischargeType) = "comminuted_and_disinfected")

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus true
                }

                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestIceShelfOrFastIce ?nearestDistance .
                    ?nearestDistance qudt:numericValue ?nearestValue ;
                                     qudt:unit unit:MI_N .
                    FILTER (
                        DATATYPE(?nearestValue) = xsd:decimal &&
                        ?nearestValue > 3
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                    ?areaDistance qudt:numericValue ?areaDistanceValue ;
                                  qudt:unit unit:MI_N .
                    FILTER (DATATYPE(?areaDistanceValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:iceConcentration ?ice .
                    ?ice qudt:numericValue ?iceValue ;
                         qudt:unit unit:UNITLESS .
                    FILTER (
                        DATATYPE(?iceValue) = xsd:decimal &&
                        ?iceValue > 0.1
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:nearbyIceConcentration ?nearbyIce .
                    ?nearbyIce qudt:numericValue ?nearbyValue ;
                              qudt:unit unit:UNITLESS .
                    FILTER (
                        DATATYPE(?nearbyValue) = xsd:decimal &&
                        ?nearbyValue > 0.1
                    )
                }
            }
        """
    ] ;
