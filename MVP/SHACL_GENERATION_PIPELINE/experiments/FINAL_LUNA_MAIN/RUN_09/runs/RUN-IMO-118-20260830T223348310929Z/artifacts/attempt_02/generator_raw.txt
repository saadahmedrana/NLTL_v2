<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_118_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability of the sewage-discharge rule."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:message "Sewage discharge status must be a boolean when provided."
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:message "Paragraph 4.2.1.3 compliance status must be a boolean when provided."
    ] ;
    sh:sparql [
        sh:message "For a ship constructed on or after 1 January 2017, Category A, Category B, or passenger-ship evidence is required to establish applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?d .
                FILTER (
                    YEAR(?d) > 2017 ||
                    (
                        YEAR(?d) = 2017 &&
                        (
                            MONTH(?d) > 1 ||
                            (MONTH(?d) = 1 && DAY(?d) >= 1)
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:shipCategory ?category .
                    FILTER (
                        ?category IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        )
                    )
                }
                FILTER NOT EXISTS {
                    $this a nltl:passengerShip .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Applicable ships must not discharge sewage into the sea unless paragraph 4.2.1.3 compliance is recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?d .
                FILTER (
                    YEAR(?d) > 2017 ||
                    (
                        YEAR(?d) = 2017 &&
                        (
                            MONTH(?d) > 1 ||
                            (MONTH(?d) = 1 && DAY(?d) >= 1)
                        )
                    )
                )
                FILTER (
                    EXISTS {
                        $this nltl:shipCategory ?category .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this a nltl:passengerShip .
                    }
                )
                FILTER NOT EXISTS {
                    $this nltl:sewageDischargeToSea "false"^^xsd:boolean .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus
                        "true"^^xsd:boolean .
                }
            }
        """
    ] .
<END_SHACL>
