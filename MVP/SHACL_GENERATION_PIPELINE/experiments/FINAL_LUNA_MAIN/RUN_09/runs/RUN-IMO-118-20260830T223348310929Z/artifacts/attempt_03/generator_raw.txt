<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-118> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability of the sewage-discharge rule."
    ] ;
    sh:sparql [
        sh:message "A ship constructed on or after 2017-01-01 must be Category A, Category B, or typed as a passenger ship for this rule's applicability to be determined." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS {
                    $this nltl:shipCategory ?category .
                    FILTER (?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    ))
                }
                FILTER NOT EXISTS {
                    $this a nltl:passengerShip .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Applicable sewage discharge to sea is prohibited unless compliance with Polar Code paragraph 4.2.1.3 is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)
                {
                    $this nltl:shipCategory ?category .
                    FILTER (?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    ))
                }
                UNION
                {
                    $this a nltl:passengerShip .
                }
                $this nltl:sewageDischargeToSea true .
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus true .
                }
            }
        """
    ] .
<END_SHACL>
