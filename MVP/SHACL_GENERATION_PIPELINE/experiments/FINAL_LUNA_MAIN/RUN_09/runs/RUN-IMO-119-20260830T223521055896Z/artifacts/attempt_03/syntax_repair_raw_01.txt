<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO119Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship category is required to determine applicability of the sewage-discharge rule."
    ] ;
    sh:property [
        sh:path nltl:extendedPeriodOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Extended-period operation information is required to determine applicability of the sewage-discharge rule."
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Sewage-discharge information is required to determine applicability of the sewage-discharge rule."
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice concentration must be supplied as a dimensionless quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship operating at ice concentration above 1/10 for an extended period with sewage discharge, an approved sewage treatment plant, approved MARPOL certification, approved regulation 9.1.1 or 9.2.1 certification, and Administration discharge approval are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )

                $this nltl:iceConcentration ?iceQuantity .
                ?iceQuantity qudt:numericValue ?iceValue .
                ?iceQuantity qudt:unit unit:UNITLESS .
                FILTER ( ?iceValue > 1 )

                $this nltl:extendedPeriodOperation true .
                $this nltl:sewageDischargeOccurs true .

                FILTER NOT EXISTS {
                    $this nltl:approvedSewageTreatmentPlantUsed true .
                }
                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation9Point1Point1Or9Point2Point1CertificationStatus
                        nltl:evidenceStateApproved .
                }
                FILTER NOT EXISTS {
                    $this nltl:marpolCertificationStatus
                        nltl:evidenceStateApproved .
                }
                FILTER NOT EXISTS {
                    $this nltl:administrationDischargeApprovalStatus
                        nltl:evidenceStateApproved .
                }
            }
        """
    ] .
</BEGIN_SHACL>
