<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO124Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Antarctic MARPOL Annex V regulation 6.1 discharges, distance to the nearest fast ice must be at least 12 nautical miles." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .
                FILTER (STR(?area) = "Antarctic_area")
                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestFastIce ?distance .
                    ?distance qudt:numericValue ?value ;
                              qudt:unit unit:MI_N .
                    FILTER (
                        DATATYPE(?value) = xsd:decimal &&
                        ?value >= "12"^^xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Antarctic MARPOL Annex V regulation 6.1 discharges, the distance from areas with ice concentration exceeding one tenth must be recorded in nautical miles." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .
                FILTER (STR(?area) = "Antarctic_area")
                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?distance .
                    ?distance qudt:numericValue ?value ;
                              qudt:unit unit:MI_N .
                    FILTER (DATATYPE(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Antarctic MARPOL Annex V regulation 6.1 discharges, the relevant nearby ice concentration must exceed one tenth and be recorded as a dimensionless quantity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .
                FILTER (STR(?area) = "Antarctic_area")
                FILTER NOT EXISTS {
                    $this nltl:nearbyIceConcentration ?concentration .
                    ?concentration qudt:numericValue ?value ;
                                   qudt:unit unit:UNITLESS .
                    FILTER (
                        DATATYPE(?value) = xsd:decimal &&
                        ?value > "0.1"^^xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Food waste must not be discharged onto ice in the applicable Antarctic case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .
                FILTER (STR(?area) = "Antarctic_area")
                FILTER NOT EXISTS {
                    $this nltl:foodWasteDischargeOntoIce false .
                }
            }
        """
    ] .
<END_SHACL>
