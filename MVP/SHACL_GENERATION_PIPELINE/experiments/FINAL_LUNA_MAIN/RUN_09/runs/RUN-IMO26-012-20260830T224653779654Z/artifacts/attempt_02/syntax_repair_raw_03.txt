<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_012_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 500
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Gross tonnage must be a single decimal QuantityValue in UNITLESS units and be at least 500."
    ] ;
    sh:sparql [
        sh:message "Ships of at least 500 gross tonnage must have at least two distinct non-magnetic heading means; each must be typed, connected to both power sources, and independently related to another distinct heading means on the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:grossTonnage ?grossNode .
                ?grossNode qudt:numericValue ?grossValue ;
                    qudt:unit qudt:UNITLESS .
                FILTER (datatype(?grossValue) = xsd:decimal)
                FILTER (?grossValue >= 500)

                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?missingMeans .
                    FILTER NOT EXISTS {
                        ?missingMeans a nltl:nonMagneticHeadingMeans ;
                            nltl:connectedToMainPower ?mainPower ;
                            nltl:connectedToEmergencyPower ?emergencyPower ;
                            nltl:independentFromHeadingMeans ?otherMeans .
                        ?mainPower a nltl:benchmarkEntity .
                        ?emergencyPower a nltl:benchmarkEntity .
                        $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                        ?otherMeans a nltl:nonMagneticHeadingMeans .
                        FILTER (?otherMeans != ?missingMeans)
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?firstMeans .
                    $this nltl:hasNonMagneticHeadingMeans ?secondMeans .
                    FILTER (?firstMeans != ?secondMeans)
                }
            }
        }
        """
    ] .
</END_SHACL>
