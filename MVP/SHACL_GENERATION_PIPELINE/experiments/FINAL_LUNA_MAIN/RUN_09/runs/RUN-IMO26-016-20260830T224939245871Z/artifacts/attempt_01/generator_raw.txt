<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-016> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Operating status in polar waters is required to determine Chapter 11-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Ship type is required to determine Chapter 11-1 applicability."
    ] ;
    sh:sparql [
        sh:message "For a ship operating in polar waters, Chapter 11-1 applicability and the required ship particulars do not satisfy the applicable threshold rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters true ;
                      nltl:shipType ?type .

                FILTER (
                    (
                        STR(?type) = "fishing_vessel" &&
                        (
                            NOT EXISTS {
                                $this nltl:lengthOverall ?lengthValue .
                                ?lengthValue qudt:numericValue ?length ;
                                             qudt:unit <http://qudt.org/vocab/unit/M> .
                                FILTER (datatype(?length) = xsd:decimal && ?length >= 24)
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:lengthOverall ?lengthValue .
                                    ?lengthValue qudt:numericValue ?length ;
                                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                                    FILTER (datatype(?length) = xsd:decimal && ?length >= 24)
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                    ||
                    (
                        STR(?type) = "pleasure_yacht" &&
                        (
                            NOT EXISTS {
                                $this nltl:grossTonnage ?tonnageValue .
                                ?tonnageValue qudt:numericValue ?tonnage ;
                                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                FILTER (
                                    datatype(?tonnage) = xsd:decimal &&
                                    ?tonnage >= 300
                                )
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:engagedInTrade false
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:grossTonnage ?tonnageValue .
                                    ?tonnageValue qudt:numericValue ?tonnage ;
                                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                    FILTER (
                                        datatype(?tonnage) = xsd:decimal &&
                                        ?tonnage >= 300
                                    )
                                }
                                &&
                                EXISTS {
                                    $this nltl:engagedInTrade false
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                    ||
                    (
                        STR(?type) = "cargo_ship" &&
                        (
                            NOT EXISTS {
                                $this nltl:grossTonnage ?tonnageValue .
                                ?tonnageValue qudt:numericValue ?tonnage ;
                                              qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                FILTER (
                                    datatype(?tonnage) = xsd:decimal &&
                                    ?tonnage >= 300 &&
                                    ?tonnage < 500
                                )
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:grossTonnage ?tonnageValue .
                                    ?tonnageValue qudt:numericValue ?tonnage ;
                                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                    FILTER (
                                        datatype(?tonnage) = xsd:decimal &&
                                        ?tonnage >= 300 &&
                                        ?tonnage < 500
                                    )
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
