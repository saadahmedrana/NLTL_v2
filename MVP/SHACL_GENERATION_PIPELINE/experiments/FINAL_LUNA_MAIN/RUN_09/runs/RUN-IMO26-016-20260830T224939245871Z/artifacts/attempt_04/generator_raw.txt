<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must state whether it operates in polar waters."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Chapter 11-1 applicability must be recorded as a single boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Chapter 11-1 applicability does not match the ship's polar-water status, type, length, gross tonnage, and trade status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters ?polar ;
                      nltl:polarCodeChapter11Dash1Applicable ?app .

                {
                    FILTER (?polar = true && ?app = true)
                    FILTER NOT EXISTS {
                        $this rdf:type nltl:fishingVessel ;
                              nltl:lengthOverall ?lengthValue .
                        ?lengthValue qudt:numericValue ?length ;
                                      qudt:unit qudt:M .
                        FILTER (?length >= 24)
                    }
                    FILTER NOT EXISTS {
                        $this rdf:type nltl:pleasureYacht ;
                              nltl:grossTonnage ?tonnageValue ;
                              nltl:engagedInTrade false .
                        ?tonnageValue qudt:numericValue ?tonnage ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (?tonnage >= 300)
                    }
                    FILTER NOT EXISTS {
                        $this rdf:type nltl:cargoShip ;
                              nltl:grossTonnage ?tonnageValue .
                        ?tonnageValue qudt:numericValue ?tonnage ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (?tonnage >= 300 && ?tonnage < 500)
                    }
                }

                UNION

                {
                    FILTER (?polar = true && ?app = false)
                    {
                        $this rdf:type nltl:fishingVessel ;
                              nltl:lengthOverall ?lengthValue .
                        ?lengthValue qudt:numericValue ?length ;
                                      qudt:unit qudt:M .
                        FILTER (?length >= 24)
                    }
                    UNION
                    {
                        $this rdf:type nltl:pleasureYacht ;
                              nltl:grossTonnage ?tonnageValue ;
                              nltl:engagedInTrade false .
                        ?tonnageValue qudt:numericValue ?tonnage ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (?tonnage >= 300)
                    }
                    UNION
                    {
                        $this rdf:type nltl:cargoShip ;
                              nltl:grossTonnage ?tonnageValue .
                        ?tonnageValue qudt:numericValue ?tonnage ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (?tonnage >= 300 && ?tonnage < 500)
                    }
                }

                UNION

                {
                    FILTER (?polar = false && ?app = true)
                }
            }
        """ ;
    ] .
<END_SHACL>
