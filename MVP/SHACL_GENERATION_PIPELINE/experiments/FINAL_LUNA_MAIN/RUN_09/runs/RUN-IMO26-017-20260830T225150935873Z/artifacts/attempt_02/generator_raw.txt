<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_017_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:intendedVoyageHazard ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:voyagePlanPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:voyagePlanCoverage ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        gen:IMO26_017_ApplicableBranch
        gen:IMO26_017_NotApplicableBranch
    ) .

gen:IMO26_017_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:voyagePlanPresent ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "When Polar Code chapter 11-1 is applicable, the voyage plan must be present."
    ] ;
    sh:sparql [
        sh:message "When Polar Code chapter 11-1 is applicable, voyage plan presence must not contain a non-true value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:voyagePlanPresent ?plan .
                FILTER (?plan != true)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every intended voyage hazard must be covered by the voyage plan." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:intendedVoyageHazard ?hazard .
                FILTER NOT EXISTS {
                    $this nltl:voyagePlanCoverage true .
                }
                FILTER NOT EXISTS {
                    $this nltl:voyagePlanCoverage ?coverage .
                    FILTER (?coverage != true)
                }
            }
        """
    ] .

gen:IMO26_017_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:hasValue false
    ] .
<END_SHACL>
