<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_017_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:intendedVoyageHazard ;
        sh:datatype xsd:string ;
        sh:message "Each intended voyage hazard value must be a string."
    ] ;
    sh:or (
        gen:IMO26_017_ApplicableBranch
        gen:IMO26_017_InapplicableBranch
    ) .

gen:IMO26_017_ApplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Chapter 11-1 applicability must be recorded as true for the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:voyagePlanPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "An applicable ship must record that a voyage plan is present."
    ] ;
    sh:sparql [
        sh:message "Chapter 11-1 applicability must not contain a false value in the applicable branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter11Dash1Applicable false .
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ship must not contain a false voyage-plan-presence value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:voyagePlanPresent false .
            }
        """
    ] ;
    sh:sparql [
        sh:message "When an intended voyage hazard exists, voyage-plan coverage must include a true value and no non-true value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:intendedVoyageHazard ?hazard .
                FILTER (
                    NOT EXISTS {
                        $this nltl:voyagePlanCoverage true .
                    }
                    ||
                    EXISTS {
                        $this nltl:voyagePlanCoverage ?coverage .
                        FILTER (?coverage != true)
                    }
                )
            }
        """
    ] .

gen:IMO26_017_InapplicableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "Chapter 11-1 applicability must be recorded as false for the inapplicable branch."
    ] ;
    sh:sparql [
        sh:message "Chapter 11-1 applicability must not contain a true value in the inapplicable branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter11Dash1Applicable true .
            }
        """
    ] .
<END_SHACL>
