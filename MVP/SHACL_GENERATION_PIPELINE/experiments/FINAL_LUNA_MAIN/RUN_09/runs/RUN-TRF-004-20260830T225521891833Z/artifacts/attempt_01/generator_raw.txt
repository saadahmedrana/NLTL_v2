<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF004Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2008 Ice Class Regulations apply."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?date .
                        FILTER (YEAR(?date) >= 2010 && YEAR(?date) < 2012)
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRegulationEdition2008 ;
                sh:minCount 1 ;
                sh:message "Ships contracted from 1 January 2010 through 31 December 2011 must identify the 2008 Ice Class Regulations as applicable."
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?date .
                        FILTER (YEAR(?date) < 2010 || YEAR(?date) >= 2012)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
