<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of TRF-007."
    ] ;
    sh:or (
        gen:TRF007NonApplicableByIceClassShape
        gen:TRF007ApplicableShape
        gen:TRF007NonApplicableByConstructionDateShape
    ) .

gen:TRF007NonApplicableByIceClassShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF007ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide its construction stage date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide its delivery date."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
        sh:minCount 1 ;
        sh:message "An applicable IA or IA Super ship must provide the assessment date."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "For an IA or IA Super ship, the keel or similar construction stage must be before 1 September 2003 for this conditional rule to apply." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:constructionStageDate ?constructionDate .
                FILTER ( ! ( ?constructionDate < "2003-09-01"^^xsd:date ) )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The applicable ship must comply no later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:deliveryDate ?deliveryDate .
                ?this nltl:assessmentDate ?assessmentDate .
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        &&
                        (
                            MONTH(?assessmentDate) > 1
                            ||
                            (
                                MONTH(?assessmentDate) = 1
                                &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:TRF007NonApplicableByConstructionDateShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
        sh:minCount 1 ;
        sh:message "An IA or IA Super ship must provide its construction stage date to determine applicability."
    ] ;
    sh:sparql [
        sh:message "The conditional TRF-007 requirement applies when the construction stage date is before 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:constructionStageDate ?constructionDate .
                FILTER ( ?constructionDate < "2003-09-01"^^xsd:date )
            }
        """
    ] .
<END_SHACL>
