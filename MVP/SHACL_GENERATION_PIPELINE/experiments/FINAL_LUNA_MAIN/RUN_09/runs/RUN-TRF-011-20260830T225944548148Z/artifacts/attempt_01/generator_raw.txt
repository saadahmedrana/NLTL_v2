<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal, pointwise, the envelope of the highest intended ice-operating waterline points." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                {
                    {
                        SELECT ?x (MAX(?intendedVertical) AS ?maximumVertical)
                        WHERE {
                            $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
                            ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                            ?intendedPoint nltl:longitudinalPosition ?intendedPosition .
                            ?intendedPosition qudt:numericValue ?xValue .
                            BIND(?xValue AS ?x)
                            ?intendedPoint nltl:verticalCoordinate ?intendedCoordinate .
                            ?intendedCoordinate qudt:numericValue ?intendedVertical .
                        }
                        GROUP BY ?x
                    }
                    OPTIONAL {
                        $this nltl:hasUpperIceWaterline ?upperWaterline .
                        ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                        ?upperPoint nltl:longitudinalPosition ?upperPosition .
                        ?upperPosition qudt:numericValue ?upperX .
                        FILTER(?upperX = ?x)
                        ?upperPoint nltl:verticalCoordinate ?upperCoordinate .
                        ?upperCoordinate qudt:numericValue ?upperVertical .
                    }
                    FILTER(!BOUND(?upperVertical) || ?upperVertical != ?maximumVertical)
                }
                UNION
                {
                    $this nltl:hasUpperIceWaterline ?extraUpperWaterline .
                    ?extraUpperWaterline nltl:hasWaterlineProfilePoint ?extraUpperPoint .
                    ?extraUpperPoint nltl:longitudinalPosition ?extraUpperPosition .
                    ?extraUpperPosition qudt:numericValue ?extraX .
                    OPTIONAL {
                        SELECT ?x (MAX(?intendedVerticalForX) AS ?maximumVerticalForX)
                        WHERE {
                            $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterlineForX .
                            ?intendedWaterlineForX nltl:hasWaterlineProfilePoint ?intendedPointForX .
                            ?intendedPointForX nltl:longitudinalPosition ?intendedPositionForX .
                            ?intendedPositionForX qudt:numericValue ?x .
                            ?intendedPointForX nltl:verticalCoordinate ?intendedCoordinateForX .
                            ?intendedCoordinateForX qudt:numericValue ?intendedVerticalForX .
                        }
                        GROUP BY ?x
                    }
                    FILTER(!BOUND(?maximumVerticalForX))
                }
            }
        """
    ] .
