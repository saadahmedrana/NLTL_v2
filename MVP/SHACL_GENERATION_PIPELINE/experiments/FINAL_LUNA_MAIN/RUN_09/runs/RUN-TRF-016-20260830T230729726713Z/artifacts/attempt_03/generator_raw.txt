gs:TRF-016-ApplicableNavigatingInIce
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node gs:TRF-016-LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node gs:TRF-016-LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node gs:TRF-016-LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node gs:TRF-016-LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node gs:TRF-016-UIWLWaterline ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node gs:TRF-016-QualifyingBallastTank
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, draught amidships must be at least the LIWL draught amidships." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#draughtAmidships> ?draughtQ ;
                      <https://w3id.org/nltl/vocab#liwlDraughtAmidships> ?liwlQ .
                ?draughtQ <http://qudt.org/schema/qudt/numericValue> ?draught .
                ?liwlQ <http://qudt.org/schema/qudt/numericValue> ?liwl .
                FILTER (?draught < ?liwl)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Forward draught must meet the UIWL displacement and level-ice-thickness threshold." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#forwardDraught> ?forwardQ ;
                      <https://w3id.org/nltl/vocab#levelIceThickness> ?iceQ ;
                      <https://w3id.org/nltl/vocab#hasUpperIceWaterline> ?uiwl .
                ?forwardQ <http://qudt.org/schema/qudt/numericValue> ?forward .
                ?iceQ <http://qudt.org/schema/qudt/numericValue> ?ice .
                ?uiwl <https://w3id.org/nltl/vocab#waterlineDisplacement> ?dispQ .
                ?dispQ <http://qudt.org/schema/qudt/numericValue> ?displacement .
                FILTER NOT EXISTS {
                    $this <https://w3id.org/nltl/vocab#hasUpperIceWaterline> ?otherUiwl .
                    ?otherUiwl <https://w3id.org/nltl/vocab#waterlineDisplacement> ?otherDispQ .
                    ?otherDispQ <http://qudt.org/schema/qudt/numericValue> ?otherDisplacement .
                    FILTER (?otherDisplacement > ?displacement)
                }
                BIND (IF ((2 + (0.00025 * ?displacement)) * ?ice < (4 * ?ice),
                          (2 + (0.00025 * ?displacement)) * ?ice,
                          4 * ?ice) AS ?expected)
                BIND (IF (ABS(?expected) * 0.000001 > 0.000000001,
                          ABS(?expected) * 0.000001,
                          0.000000001) AS ?tolerance)
                FILTER (?forward < (?expected - ?tolerance))
            }
        """
    ] .
