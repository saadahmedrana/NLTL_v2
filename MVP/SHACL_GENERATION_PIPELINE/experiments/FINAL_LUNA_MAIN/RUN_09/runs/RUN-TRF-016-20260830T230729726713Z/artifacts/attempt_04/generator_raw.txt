g:TRF-016-Navigating a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node g:TRF-016-LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node g:TRF-016-LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node g:TRF-016-LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node g:TRF-016-LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node g:TRF-016-UIWL ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node g:TRF-016-BallastTank
    ] ;
    sh:sparql [
        sh:message "Draught amidships must be at least the LIWL draught amidships when navigating in ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:draughtAmidships ?draughtNode ;
                      nltl:liwlDraughtAmidships ?liwlNode .
                ?draughtNode qudt:numericValue ?draught ;
                             qudt:unit unit:M .
                ?liwlNode qudt:numericValue ?liwl ;
                            qudt:unit unit:M .
                FILTER (?draught < ?liwl)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Forward draught must meet the level-ice-thickness threshold based on the greatest-displacement upper ice waterline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasUpperIceWaterline ?uiwl ;
                      nltl:levelIceThickness ?iceNode ;
                      nltl:forwardDraught ?forwardNode .
                ?uiwl nltl:waterlineDisplacement ?displacementNode .
                ?displacementNode qudt:numericValue ?displacement ;
                                  qudt:unit unit:TON_Metric .
                ?iceNode qudt:numericValue ?iceThickness ;
                         qudt:unit unit:M .
                ?forwardNode qudt:numericValue ?forward ;
                             qudt:unit unit:M .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?otherUiwl .
                    ?otherUiwl nltl:waterlineDisplacement ?otherDisplacementNode .
                    ?otherDisplacementNode qudt:numericValue ?otherDisplacement ;
                                            qudt:unit unit:TON_Metric .
                    FILTER (?otherDisplacement > ?displacement)
                }
                BIND ((2 + (0.00025 * ?displacement)) * ?iceThickness AS ?uncapped)
                BIND (4 * ?iceThickness AS ?cap)
                BIND (IF(?uncapped < ?cap, ?uncapped, ?cap) AS ?expected)
                BIND ((0.000001 * ?expected) + 0.000000000001 AS ?tolerance)
                FILTER (?forward + ?tolerance < ?expected)
            }
        """
    ] .
