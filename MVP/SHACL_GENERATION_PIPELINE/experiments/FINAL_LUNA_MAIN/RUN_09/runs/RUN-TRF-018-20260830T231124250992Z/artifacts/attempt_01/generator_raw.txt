@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:CalculatedRequiredPowerPropertyShape
    a sh:PropertyShape ;
    sh:path nltl:calculatedRequiredPower ;
    sh:nodeKind sh:IRI ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue qudt:KiloW ;
            sh:nodeKind sh:IRI ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] .

    sh:path nltl:maximumContinuousRatingPower ;
    sh:nodeKind sh:IRI ;

    sh:node [
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;

gen:TRF018ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceClass must identify one permitted Finnish-Swedish ice class."
    ] ;
    sh:or (
        gen:IceClassIaSuperBranch
        gen:IceClassIaBranch
        gen:IceClassIbBranch
        gen:IceClassIcBranch
        gen:IceClassIiBranch
        gen:IceClassIiiBranch
    ) ;
    sh:sparql [
        sh:message "For ice class IA Super, maximumContinuousRatingPower must be at least the greater of calculatedRequiredPower and 2800 kW; for IA, IB, or IC it must be at least the greater of calculatedRequiredPower and 1000 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?maximumNode .
                ?calculatedNode qudt:numericValue ?calculatedPower ;
                                qudt:unit qudt:KiloW .
                ?maximumNode qudt:numericValue ?maximumPower ;
                             qudt:unit qudt:KiloW .
                FILTER (
                    (
                        ?ice = nltl:iceClassIaSuper &&
                        (?maximumPower < ?calculatedPower || ?maximumPower < 2800.0)
                    )
                    ||
                    (
                        (
                            ?ice = nltl:iceClassIa ||
                            ?ice = nltl:iceClassIb ||
                            ?ice = nltl:iceClassIc
                        ) &&
                        (?maximumPower < ?calculatedPower || ?maximumPower < 1000.0)
                    )
                )
            }
        """
    ] .
