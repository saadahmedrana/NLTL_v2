<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship with ice-class regulation 1985 applicable and ice class IB or IC, maximum continuous rating power must be at least the required power under the 1985 rules, with both quantities expressed in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:applicableIceClassRegulationEdition
                      nltl:iceClassRuleEdition1985 ;
                      nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?maximumQuantity ;
                          nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                    ?maximumQuantity qudt:numericValue ?maximumValue ;
                                     qudt:unit qudt:KiloW .
                    ?requiredQuantity qudt:numericValue ?requiredValue ;
                                    qudt:unit qudt:KiloW .
                    FILTER (
                        datatype(?maximumValue) = xsd:decimal &&
                        datatype(?requiredValue) = xsd:decimal &&
                        ?maximumValue >= ?requiredValue
                    )
                }
            }
        """
    ] .
<END_SHACL>
