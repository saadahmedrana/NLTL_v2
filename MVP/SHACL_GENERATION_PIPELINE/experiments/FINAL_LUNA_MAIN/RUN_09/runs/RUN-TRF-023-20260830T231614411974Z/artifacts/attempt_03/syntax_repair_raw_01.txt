@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:MaximumContinuousRatingPowerPropertyShape
    sh:path nltl:maximumContinuousRatingPower ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        sh:class qudt:QuantityValue ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] .

gs:RequiredPowerUnder1985RulesPropertyShape
    sh:path nltl:requiredPowerUnder1985Rules ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        sh:class qudt:QuantityValue ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ] .

gs:TRF023ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:not [
                    sh:in (
                        nltl:iceClassIb
                        nltl:iceClassIc
                    )
                ]
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:in (
                            nltl:iceClassIb
                            nltl:iceClassIc
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:applicableIceClassRegulationEdition ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:not [
                            sh:hasValue nltl:iceClassRuleEdition1985
                        ]
                    ]
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:in (
                            nltl:iceClassIb
                            nltl:iceClassIc
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:applicableIceClassRegulationEdition ;
                        sh:hasValue nltl:iceClassRuleEdition1985 ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property gs:MaximumContinuousRatingPowerPropertyShape
                ]
                [
                    sh:property gs:RequiredPowerUnder1985RulesPropertyShape
                ]
            )
        ]
    ) ;
    sh:message "For ice class IB or IC under the 1985 ice class rules, the ship must provide maximum continuous rating power and required power in kW, with maximum power at least the required power." .

gs:TRF023PowerComparisonShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be at least the required power under the 1985 rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )
                $this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRuleEdition1985 .
                $this nltl:maximumContinuousRatingPower ?maximumQuantity .
                ?maximumQuantity qudt:numericValue ?maximumValue .
                $this nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                ?requiredQuantity qudt:numericValue ?requiredValue .
                FILTER ( ?maximumValue < ?requiredValue )
            }
        """
    ] .
