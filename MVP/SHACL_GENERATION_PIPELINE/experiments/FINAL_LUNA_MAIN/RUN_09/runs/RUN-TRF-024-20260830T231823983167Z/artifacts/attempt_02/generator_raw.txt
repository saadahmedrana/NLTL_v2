<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF024ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1
    ] ;
    sh:or (
        gs:TRF024ApplicableShape
        gs:TRF024NonApplicableIceClassShape
        gs:TRF024NonApplicableConstructionShape
    ) .

gs:TRF024ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .

                FILTER (
                    (
                        YEAR(?constructionDate) < 2003
                        ||
                        (
                            YEAR(?constructionDate) = 2003
                            && MONTH(?constructionDate) < 9
                        )
                    )
                    &&
                    (
                        YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
                        ||
                        (
                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                            && MONTH(?assessmentDate) = 1
                            && DAY(?assessmentDate) = 1
                        )
                    )
                )
            }
        """ ;
        sh:message "For ice class IA Super or IA with construction before 1 September 2003, the assessment date must be no later than 1 January in the year twenty years after delivery."
    ] .

gs:TRF024NonApplicableIceClassShape
    a sh:NodeShape ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass NOT IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
            }
        """ ;
        sh:message "This requirement is not applicable to ice classes other than IA Super or IA."
    ] .

gs:TRF024NonApplicableConstructionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?constructionDate .

                FILTER (
                    YEAR(?constructionDate) > 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) >= 9
                    )
                )
            }
        """ ;
        sh:message "The deadline condition is not applicable when construction began on or after 1 September 2003."
    ] .
<END_SHACL>
