<urn:nltl:generated-shape:TRF-028> a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:hasEvidence ?e .
                ?e a ?evidenceType .
                FILTER (
                    ?evidenceType IN (
                        nltl:alternativeCalculationEvidence,
                        nltl:modelTestEvidence
                    )
                )
                FILTER (
                    EXISTS { ?this nltl:engineOutputCoefficientKe ?ke }

                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:hasEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:or (
            [
                sh:node [
                    sh:class nltl:alternativeCalculationEvidence
                ]
            ]
            [
                sh:node [
                    sh:class nltl:modelTestEvidence
                ]
            ]
        ] ;
        sh:message "The ship using the alternative Ke or RCH method must provide alternative-calculation or model-test evidence."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:brashIceChannelResistanceRch ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/N>
                    ]
                ]
            ]
        ]

        sh:path nltl:approvalStatus ;

        ) ;
        sh:message "Approval or revocation status must use a permitted evidence state."
    ] ;

        sh:message "A ship with approvalStatus true must have approvalRevocationStatus evidenceStateApproved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:approvalStatus true ;

            }
        """
    ] .
