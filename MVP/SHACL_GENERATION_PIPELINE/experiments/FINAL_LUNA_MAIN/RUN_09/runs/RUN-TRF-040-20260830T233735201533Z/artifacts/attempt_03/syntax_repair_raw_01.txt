<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF040Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:designOperationSpeedInIce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design operation speed in ice shall be provided as a single knot quantity."
    ] ;
    sh:property [
        sh:path nltl:openWaterServiceSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Open-water service speed shall be provided as a single knot quantity."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Main-frame spacing shall be provided as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Shell thickness shall be provided as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sidescuttles shall not be situated in the ice belt."
    ] ;
    sh:property [
        sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The weather-deck position shall be stated to select the applicable ice-belt strength branch."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bulwarkStrengthStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:freeingPortStrengthStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For ice class IA Super, forefoot shell strengthening status shall equal bow-region equivalence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER NOT EXISTS {
                    $this nltl:forefootShellStrengtheningStatus ?forefoot .
                    $this nltl:bowRegionEquivalent ?bow .
                    FILTER (?forefoot = ?bow)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For IA or IA Super ships with open-water service speed of at least 18 knots, upper-bow strengthening status shall equal midbody equivalence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )
                $this nltl:openWaterServiceSpeed ?speedQuantity .
                ?speedQuantity qudt:numericValue ?speed .
                FILTER (?speed >= "18"^^xsd:decimal)
                FILTER NOT EXISTS {
                    $this nltl:upperBowIceBeltStrengtheningStatus ?upperBow .
                    $this nltl:midbodyEquivalent ?midbody .
                    FILTER (?upperBow = ?midbody)
                }
            }
        """
    ] .
<END_SHACL>
