<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape ;
        sh:message "Each linked framing ice-strengthening termination case must satisfy the applicable termination rule."
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043ExtensionQuantityShape ;
        sh:message "A termination case must state the extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:message "The termination boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:or (
        [
            sh:and (
                [
                    sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                    sh:node [
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:maxInclusive 250
                        ]
                    ]
                ]
                [
                    sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
                [
                    sh:path nltl:terminationAdjacentBoundary ;
                    sh:or (
                        [ sh:class nltl:deckStructure ]
                        [ sh:class nltl:tankBoundaryPlating ]
                        [ sh:class nltl:tankTop ]
                    )
                ]
            )
        ]
        [
            sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
            sh:node [
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:minExclusive 250
                ]
            ]
        ]
    ) ;
    sh:message "When the extension is at most 250 mm, termination at the adjacent deck, tank boundary plating, or tank top must be expressly permitted." .

gen:TRF043ExtensionQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:TRF043FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:message "The frame must identify one of the ice classes covered by Table 4-6."
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:message "The frame must identify its hull region for the Table 4-6 lookup."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MetreQuantityShape ;
        sh:message "The frame must state the required strengthening extent above the upper ice waterline."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF043MetreQuantityShape ;
        sh:message "The frame must state the required strengthening extent below the lower ice waterline."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:and (
                [
                    sh:path nltl:upperBowIceBeltRequired ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
                [
                    sh:path nltl:reachesUpperBowIceBeltTop ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
            )
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame strengthening extents do not satisfy the Table 4-6 values for its ice class and hull region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveNode ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowNode .

                ?aboveNode qudt:numericValue ?above .
                ?belowNode qudt:numericValue ?below .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:bowRegion &&
                        (
                            ?above < 1.2 ||
                            ?below < 2.0 ||
                            NOT EXISTS {
                                $this nltl:reachesTankTopOrBelowFloorTop true
                            }
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:midbodyRegion &&
                        (
                            ?above < 1.2 ||
                            ?below < 2.0
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:sternRegion &&
                        (
                            ?above < 1.2 ||
                            ?below < 1.6
                        )
                    )
                    ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        ?region = nltl:bowRegion &&
                        (
                            ?above < 1.0 ||
                            ?below < 1.6
                        )
                    )
                    ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        ?region = nltl:midbodyRegion &&
                        (
                            ?above < 1.0 ||
                            ?below < 1.3
                        )
                    )
                    ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        ?region = nltl:sternRegion &&
                        (
                            ?above < 1.0 ||
                            ?below < 1.0
                        )
                    )
                )
            }
        """
    ] .

gen:TRF043MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
