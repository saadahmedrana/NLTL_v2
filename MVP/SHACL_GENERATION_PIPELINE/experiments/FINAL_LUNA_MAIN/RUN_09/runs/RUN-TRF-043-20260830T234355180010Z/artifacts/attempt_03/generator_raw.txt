<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

<urn:nltl:generated-shape:TRF-043-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:minCount 1 ;
        sh:message "Each ship must identify its framing ice-strengthening termination cases."
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node <urn:nltl:generated-shape:TRF-043-LengthMillimetresShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A termination case must record one extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                        sh:node <urn:nltl:generated-shape:TRF-043-AtMost250MillimetresShape>
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:terminationAdjacentBoundary ;
                        sh:or (
                            [ sh:class nltl:deckStructure ]
                            [ sh:class nltl:tankBoundaryPlating ]
                            [ sh:class nltl:tankTop ]
                        ) ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node <urn:nltl:generated-shape:TRF-043-GreaterThan250MillimetresShape>
            ]
        ]
    ) ;
    sh:message "If the extension is no more than 250 mm, termination at the adjacent deck, tank boundary, or tank top must be permitted." .

<urn:nltl:generated-shape:TRF-043-LengthMillimetresShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .

<urn:nltl:generated-shape:TRF-043-AtMost250MillimetresShape>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The extension must be no more than 250 mm for the permitted termination branch." ;
        sh:select """
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this qudt:numericValue ?value .
                FILTER ( ?value > "250"^^xsd:decimal )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-043-GreaterThan250MillimetresShape>
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The non-applicable termination branch requires an extension greater than 250 mm." ;
        sh:select """
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this qudt:numericValue ?value .
                FILTER ( ?value <= "250"^^xsd:decimal )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-043-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:class nltl:hullRegionValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-043-LengthMetresShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF-043-LengthMetresShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:upperBowIceBeltRequired ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:reachesUpperBowIceBeltTop ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The framing vertical extents must satisfy the Table 4-6 values for the ice class and hull region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?hullRegion ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below ;
                      nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .

                ?above qudt:numericValue ?aboveValue .
                ?below qudt:numericValue ?belowValue .

                FILTER (
                    !(
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?aboveValue >= "1.2"^^xsd:decimal &&
                            (
                                (
                                    ?hullRegion = nltl:bowRegion &&
                                    ?reachesTankTop = true
                                )
                                ||
                                (
                                    ?hullRegion = nltl:midbodyRegion &&
                                    ?belowValue >= "2.0"^^xsd:decimal
                                )
                                ||
                                (
                                    ?hullRegion = nltl:sternRegion &&
                                    ?belowValue >= "1.6"^^xsd:decimal
                                )
                            )
                        )
                        ||
                        (
                            ?iceClass IN (
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            ?aboveValue >= "1.0"^^xsd:decimal &&
                            (
                                (
                                    ?hullRegion = nltl:bowRegion &&
                                    ?belowValue >= "1.6"^^xsd:decimal
                                )
                                ||
                                (
                                    ?hullRegion = nltl:midbodyRegion &&
                                    ?belowValue >= "1.3"^^xsd:decimal
                                )
                                ||
                                (
                                    ?hullRegion = nltl:sternRegion &&
                                    ?belowValue >= "1.0"^^xsd:decimal
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-043-LengthMetresShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
