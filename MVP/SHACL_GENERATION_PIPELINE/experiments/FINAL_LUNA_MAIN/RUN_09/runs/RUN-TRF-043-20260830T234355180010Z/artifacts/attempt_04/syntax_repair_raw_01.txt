<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ExtensionAtMost250MillimetresShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 0 ;
        sh:maxInclusive 250
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ExtensionAbove250MillimetresShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 250
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TerminationCaseShape
    a sh:NodeShape ;
    sh:class nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:LengthMillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                        sh:node gen:ExtensionAtMost250MillimetresShape ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:terminationAdjacentBoundary ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:or (
                                [ sh:class nltl:deckStructure ]
                                [ sh:class nltl:tankBoundaryPlating ]
                                [ sh:class nltl:tankTop ]
                            )
                        ]
                    ]
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                        sh:node gen:ExtensionAbove250MillimetresShape ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            )
        ]
    ) .

gen:FrameShape
    a sh:NodeShape ;
    sh:class nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:upperBowIceBeltRequired ;
                        sh:hasValue false
                    ]
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:upperBowIceBeltRequired ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:reachesUpperBowIceBeltTop ;
                        sh:hasValue true
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing ice-strengthening vertical extents do not satisfy Table 4-6 for the frame ice class and hull region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?hullRegion ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .

                ?above qudt:numericValue ?aboveValue ;
                       qudt:unit unit:M .
                ?below qudt:numericValue ?belowValue ;
                       qudt:unit unit:M .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?aboveValue >= 1.2 &&
                        (
                            (
                                ?hullRegion = nltl:bowRegion &&
                                EXISTS {
                                    $this nltl:reachesTankTopOrBelowFloorTop true
                                }
                            ) ||
                            (
                                ?hullRegion = nltl:midbodyRegion &&
                                ?belowValue >= 2.0
                            ) ||
                            (
                                ?hullRegion = nltl:sternRegion &&
                                ?belowValue >= 1.6
                            )
                        )
                    ) ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        ?aboveValue >= 1.0 &&
                        (
                            (
                                ?hullRegion = nltl:bowRegion &&
                                ?belowValue >= 1.6
                            ) ||
                            (
                                ?hullRegion = nltl:midbodyRegion &&
                                ?belowValue >= 1.3
                            ) ||
                            (
                                ?hullRegion = nltl:sternRegion &&
                                ?belowValue >= 1.0
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TerminationCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node gen:FrameShape
    ] .

gen:FrameTargetShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:node gen:FrameShape .
<END_SHACL>
