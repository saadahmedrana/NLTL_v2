@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-047> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:class nltl:frameStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A strengthened part is required for the frame."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:class nltl:frameEnd ;
        sh:minCount 1 ;
        sh:message "The transverse frame must have a lower end."
    ] ;
    sh:property [
        sh:path ( nltl:hasLowerEnd nltl:hasAttachedSupportingStructure ) ;
        sh:or (
            [
                sh:class nltl:deckStructure
            ]
            [
                sh:class nltl:tankBoundaryPlating
            ]
            [
                sh:class nltl:tankTop
            ]
            [
                sh:class nltl:iceStringer
            ]
        ) ;
        sh:minCount 1 ;
        sh:message "The lower end must be attached to a deck, tank boundary plating, tank top, or ice stringer."
    ] ;
    sh:sparql [
        sh:message "A horizontal connection from a lower end must connect to an adjacent main frame, use the same scantlings as the main frame, and be represented only where the supporting structure is at or below the ice-belt lower limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasLowerEnd ?lowerEnd .
                ?lowerEnd nltl:hasHorizontalConnectionMember ?member .
                FILTER NOT EXISTS {
                    $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit "true"^^xsd:boolean .
                }
                FILTER NOT EXISTS {
                    ?member nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                }
                FILTER NOT EXISTS {
                    $this nltl:sameScantlingsAsMainFrame "true"^^xsd:boolean .
                }
            }
        """
    ] .

</END_SHACL>
