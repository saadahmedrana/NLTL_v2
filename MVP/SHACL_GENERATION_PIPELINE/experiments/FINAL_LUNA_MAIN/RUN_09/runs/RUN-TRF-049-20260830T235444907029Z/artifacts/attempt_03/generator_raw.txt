<urn:nltl:generated-shape:TRF-049-IceLongitudinalFrame>
    a sh:NodeShape ;
    sh:class nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A frame within the ice-strengthened area must have confirmed effective attachment."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-049-FrameWebThickness> ;
        sh:message "The frame web thickness is required for bracket thickness verification."
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-049-LongitudinalAttachment> ;
        sh:message "A longitudinal frame must provide attachment evidence for its supporting structures."
    ] ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead linked to a longitudinal frame must correspond to a frame attachment and reciprocal supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supportingStructure .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every longitudinal frame attachment must identify a supporting structure with a reciprocal supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A bracket must not be thinner than the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                    qudt:unit unit:MilliM .
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                    qudt:unit unit:MilliM .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .
