<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF049FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [ sh:node gen:TRF049InIceFrameShape ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .

gen:TRF049InIceFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF049FrameWebThicknessValueShape
    ] ;
    sh:or (
        [ sh:node gen:TRF049LongitudinalFrameShape ]
        [ sh:node gen:TRF049TransverseFrameShape ]
    ) .

gen:TRF049LongitudinalFrameShape
    a sh:NodeShape ;
    sh:class nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gen:TRF049LongitudinalAttachmentShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked supporting web frame or bulkhead must correspond to a frame attachment and reciprocal supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                {
                    ?this nltl:hasSupportingWebFrame ?supportingStructure .
                }
                UNION
                {
                    ?this nltl:hasSupportingBulkhead ?supportingStructure .
                }
                FILTER NOT EXISTS {
                    ?this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] .

gen:TRF049TransverseFrameShape
    a sh:NodeShape ;
    sh:class nltl:transverseFrame ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gen:TRF049AttachmentShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [ sh:node gen:TRF049TerminatingTransverseFrameShape ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked supporting structure must correspond to a frame attachment and reciprocal supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                {
                    ?this nltl:hasSupportingWebFrame ?supportingStructure .
                }
                UNION
                {
                    ?this nltl:hasSupportingBulkhead ?supportingStructure .
                }
                FILTER NOT EXISTS {
                    ?this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] .

gen:TRF049TerminatingTransverseFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gen:TRF049AttachmentShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A transverse frame terminating at a deck or stringer must have a bracket or similar attachment construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                FILTER NOT EXISTS {
                    ?this nltl:hasFrameAttachment ?attachment .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] .

gen:TRF049AttachmentShape
    a sh:NodeShape ;
    sh:class nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each frame attachment must have a reciprocal supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this nltl:hasAttachedSupportingStructure ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?this .
                }
            }
        """
    ] .

gen:TRF049LongitudinalAttachmentShape
    a sh:NodeShape ;
    sh:node gen:TRF049AttachmentShape ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:minCount 1 ;
        sh:node gen:TRF049ConnectionBracketShape
    ] .

gen:TRF049ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF049BracketThicknessValueShape
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bracket thickness must be at least the thickness of the linked frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?frame nltl:frameWebThickness ?frameThickness .
                ?frame nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?this .
                ?this nltl:bracketThickness ?bracketThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (
                    DATATYPE(?frameValue) = xsd:decimal &&
                    DATATYPE(?bracketValue) = xsd:decimal &&
                    ?bracketValue < ?frameValue
                )
            }
        """
    ] .

gen:TRF049FrameWebThicknessValueShape

    sh:property [

    sh:property [

        sh:minCount 1 ;

        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:TRF049BracketThicknessValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:TRF049SupportingStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:supportingStructure ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each supporting-structure attachment must point back to the corresponding frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this nltl:hasSupportingStructureAttachment ?attachment .
                FILTER NOT EXISTS {
                    ?attachment nltl:hasAttachedSupportingStructure ?this .
                }
            }
        """
    ] .
