gen:TRF052FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must specify one permitted ice class."
    ] ;
    sh:or (
        gen:TRF052DirectArrangementShape
        gen:TRF052EquivalentSupportShape
    ) .

gen:TRF052EquivalentSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An alternative arrangement requires demonstrated equivalent support by direct calculation."
    ] .

gen:TRF052DirectArrangementShape
    a sh:NodeShape ;
    sh:or (
        gen:TRF052AsymmetricalSpacingShape
        gen:TRF052AngledSpacingShape
        gen:TRF052NoSpacingTriggerShape
    ) ;
    sh:or (
        gen:TRF052LongSpanScopeShape
        gen:TRF052IaSuperScopeShape
        gen:TRF052IaScopeShape
        gen:TRF052IbIcScopeShape
    ) .

gen:TRF052AsymmetricalSpacingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:node gen:TRF052SpacingQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "For an asymmetrical frame, antitripping support spacing must not exceed 1300 mm."
    ] .

gen:TRF052AngledSpacingShape
    a sh:NodeShape ;

        sh:path nltl:frameWebAngleToShell ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:node gen:TRF052SpacingQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "For a frame with web angle below 90 degrees to the shell, antitripping support spacing must not exceed 1300 mm."
    ] .

gen:TRF052NoSpacingTriggerShape
