<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF053_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one dimensionless frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one calculated design line load in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one calculated section modulus in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one calculated required shear area in cm2."
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area must satisfy the prescribed ice-stringer calculations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?sectionNode ;
                      nltl:requiredShearArea ?areaNode .

                ?pressureNode qudt:numericValue ?pressure ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?heightNode qudt:numericValue ?height ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spanNode qudt:numericValue ?span ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                ?momentNode qudt:numericValue ?moment ;
                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?yieldNode qudt:numericValue ?yield ;
                          qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?lineNode qudt:numericValue ?reportedLine ;
                         qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                ?sectionNode qudt:numericValue ?reportedSection ;
                            qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?areaNode qudt:numericValue ?reportedArea ;
                         qudt:unit <http://qudt.org/vocab/unit/CentiM2> .

                BIND(IF((?pressure * ?height) < 0.15,
                        0.15,
                        (?pressure * ?height)) AS ?lineMN)
                BIND(?lineMN * 1000 AS ?expectedLine)
                BIND(0.9 * 1.8 * ?lineMN * ?span * ?span /
                     (?moment * ?yield) * 1000000 AS ?expectedSection)
                BIND(1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?lineMN * ?span /
                     (2 * ?yield) * 10000 AS ?expectedArea)

                FILTER(
                    ABS(?reportedLine - ?expectedLine) >
                    (0.000001 * ABS(?expectedLine) + 0.000000001)
                    ||
                    ABS(?reportedSection - ?expectedSection) >
                    (0.000001 * ABS(?expectedSection) + 0.000000001)
                    ||
                    ABS(?reportedArea - ?expectedArea) >
                    (0.000001 * ABS(?expectedArea) + 0.000000001)
                )
            }
        """
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
