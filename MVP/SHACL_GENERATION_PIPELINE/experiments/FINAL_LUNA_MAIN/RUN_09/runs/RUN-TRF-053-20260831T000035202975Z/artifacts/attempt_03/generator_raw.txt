            PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    ?loadReported > ?loadExpected + ?loadTolerance ||
                    ?loadReported < ?loadExpected - ?loadTolerance ||
                    ?sectionReported > ?sectionExpected + ?sectionTolerance ||
                    ?sectionReported < ?sectionExpected - ?sectionTolerance ||
                    ?areaReported > ?areaExpected + ?areaTolerance ||
                    ?areaReported < ?areaExpected - ?areaTolerance
                )
