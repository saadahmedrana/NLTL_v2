@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    ] ;
    sh:property [

nltlgen:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LineLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:AreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "designLineLoad, outsideIceBeltFactor, sectionModulus, and requiredShearArea must satisfy the TRF-054 formulas within a relative tolerance of 1e-6 plus an absolute tolerance of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this
                    nltl:icePressure ?pNode ;
                    nltl:designIceLoadHeight ?hNode ;
                    nltl:span ?lNode ;
                    nltl:frameMomentFactorM ?mNode ;
                    nltl:yieldStrength ?sigmaNode ;
                    nltl:distanceToIceBelt ?hsNode ;
                    nltl:stringerSpan ?lsNode ;
                    nltl:designLineLoad ?lineNode ;
                    nltl:outsideIceBeltFactor ?factorNode ;
                    nltl:sectionModulus ?zNode ;
                    nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?lineNode qudt:numericValue ?line .
                ?factorNode qudt:numericValue ?factor .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND (?p * ?h AS ?rawLineLoad)
                BIND (IF(?rawLineLoad < 0.15, 0.15, ?rawLineLoad) AS ?designLineLoadMNPerM)
                BIND (?designLineLoadMNPerM * 1000 AS ?expectedLine)
                BIND (1 - (?hs / ?ls) AS ?expectedFactor)
                BIND (0.80 * 1.8 * ?designLineLoadMNPerM * ?l * ?l /
                      (?m * ?sigma) * ?expectedFactor * 1000000 AS ?expectedZ)
                BIND (1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?designLineLoadMNPerM * ?l / (2 * ?sigma) *
                      ?expectedFactor * 10000 AS ?expectedA)

                FILTER (
                    ABS(?line - ?expectedLine) >
                        (ABS(?expectedLine) * 0.000001 + 0.000000001)
                    ||
                    ABS(?factor - ?expectedFactor) >
                        (ABS(?expectedFactor) * 0.000001 + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                        (ABS(?expectedZ) * 0.000001 + 0.000000001)
                    ||
                    ABS(?a - ?expectedA) >
                        (ABS(?expectedA) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
