            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:distanceToAdjacentIceStringer ?lsNode ;
                      nltl:designLineLoad ?qNode ;
                      nltl:outsideIceBeltFactor ?fNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?qNode qudt:numericValue ?qObserved .
                ?fNode qudt:numericValue ?fObserved .
                ?zNode qudt:numericValue ?zObserved .
                ?aNode qudt:numericValue ?aObserved .

                BIND (?p * ?h AS ?product)
                BIND (IF(?product < 0.15, 0.15, ?product) AS ?lineLoadMNPerM)
                BIND (?lineLoadMNPerM * 1000 AS ?qExpected)
                BIND (1 - (?hs / ?ls) AS ?fExpected)
                BIND (0.80 * 1.8 * ?lineLoadMNPerM * ?l * ?l /
                      (?m * ?sigma) * ?fExpected * 1000000 AS ?zExpected)
                BIND (1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?lineLoadMNPerM * ?l /
                      (2 * ?sigma) * ?fExpected * 10000 AS ?aExpected)

                FILTER (
                    ABS(?qObserved - ?qExpected) >
                    0.000001 * IF(ABS(?qExpected) > 1, ABS(?qExpected), 1)
                    ||
                    ABS(?fObserved - ?fExpected) >
                    0.000001 * IF(ABS(?fExpected) > 1, ABS(?fExpected), 1)
                    ||
                    ABS(?zObserved - ?zExpected) >
                    0.000001 * IF(ABS(?zExpected) > 1, ABS(?zExpected), 1)
                    ||
                    ABS(?aObserved - ?aExpected) >
                    0.000001 * IF(ABS(?aExpected) > 1, ABS(?aExpected), 1)
                )
            }
        """
