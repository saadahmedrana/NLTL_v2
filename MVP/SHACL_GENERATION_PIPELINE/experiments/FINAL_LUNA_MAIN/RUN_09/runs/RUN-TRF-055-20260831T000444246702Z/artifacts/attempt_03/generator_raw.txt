gen:TRF055NarrowDeckStripShape
    a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:or (
        gen:NotApplicableByHatchShape
        gen:NotApplicableByIceStringerShape
        gen:ApplicableOrdinaryHatchShape
        gen:ApplicableVeryLongHatchShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable narrow deck strip, the actual section modulus and shear area shall meet their required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:abreastOfHatch true ;
                      nltl:servesAsIceStringer true ;
                      nltl:actualSectionModulus ?actualSectionModulusNode ;
                      nltl:requiredSectionModulus ?requiredSectionModulusNode ;
                      nltl:actualShearArea ?actualShearAreaNode ;
                      nltl:requiredShearArea ?requiredShearAreaNode .

                ?actualSectionModulusNode qudt:numericValue ?actualSectionModulus .
                ?requiredSectionModulusNode qudt:numericValue ?requiredSectionModulus .
                ?actualShearAreaNode qudt:numericValue ?actualShearArea .
                ?requiredShearAreaNode qudt:numericValue ?requiredShearArea .

                FILTER (
                    ?actualSectionModulus < ?requiredSectionModulus ||
                    ?actualShearArea < ?requiredShearArea
                )
            }
        """
    ] .
