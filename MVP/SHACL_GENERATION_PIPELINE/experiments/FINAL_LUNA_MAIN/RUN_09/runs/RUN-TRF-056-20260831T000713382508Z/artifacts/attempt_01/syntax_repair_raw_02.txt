<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-056-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node [
            sh:class nltl:weatherdeckHatch ;
            sh:property [
                sh:path nltl:hatchOpeningLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a weatherdeck hatch longer than half the ship breadth, ship side deflection and both hatch cover and hatch fitting design evidence must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipBreadth ?breadthValue .
                ?breadthValue qudt:numericValue ?breadth .
                ?breadthValue qudt:unit <http://qudt.org/vocab/unit/M> .
                $this nltl:hasWeatherdeckHatch ?hatch .
                ?hatch nltl:hatchOpeningLength ?lengthValue .
                ?lengthValue qudt:numericValue ?length .
                ?lengthValue qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER ( ?length > ( ?breadth / 2 ) )
                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflectionValue .
                        ?deflectionValue qudt:numericValue ?deflection .
                        ?deflectionValue qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                        ?coverEvidence a nltl:hatchCoverDesignEvidence .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                        ?fittingEvidence a nltl:hatchFittingDesignEvidence .
                    }
                )
            }
        """
    ] .
</END_SHACL>
