<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF-057-WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A web frame must have one ice pressure value in MPa." ;
        sh:node [
            sh:class q:QuantityValue ;
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A web frame must have one ice load height value in metres." ;
        sh:node [
            sh:class q:QuantityValue ;
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A web frame must have one web frame spacing value in metres." ;
        sh:node [
            sh:class q:QuantityValue ;
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A web frame must have one calculated ice load value in kN." ;
        sh:node [
            sh:class q:QuantityValue ;
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "webFrameIceLoad must equal 1.8 times the greater of icePressure multiplied by iceLoadHeight and 0.15 MN/m, multiplied by webFrameSpacing; the comparison allows a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:iceLoadHeight ?heightNode ;
                      nltl:webFrameSpacing ?spacingNode ;
                      nltl:webFrameIceLoad ?loadNode .

                ?pressureNode q:numericValue ?pressure ;
                              q:unit unit:MegaPA .
                ?heightNode q:numericValue ?height ;
                            q:unit unit:M .
                ?spacingNode q:numericValue ?spacing ;
                             q:unit unit:M .
                ?loadNode q:numericValue ?actualLoad ;
                           q:unit unit:KiloN .

                BIND((?pressure * ?height) AS ?pressureHeight)
                BIND(IF(?pressureHeight < 0.15, 0.15, ?pressureHeight) AS ?effectiveLineLoad)
                BIND((1.8 * ?effectiveLineLoad * ?spacing * 1000) AS ?expectedLoad)
                BIND(ABS(?expectedLoad) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000001, ?relativeTolerance, 0.000000001) AS ?tolerance)

                FILTER(ABS(?actualLoad - ?expectedLoad) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
