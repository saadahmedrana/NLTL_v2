<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF066Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF066ApplicableShape
        gen:TRF066NonApplicableShape
    ) .

gen:TRF066ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
        ) ;
        sh:message "For ice class IA or IA Super, the ice class must be recorded."
    ] ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:or (
            gen:TRF066PracticableShape
            gen:TRF066NotPracticableShape
        ) ;
        sh:message "For ice class IA or IA Super, ice-knife practicability must be recorded."
    ] .

gen:TRF066NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The recorded ice class must be a permitted non-applicable class for this requirement."
    ] .

gen:TRF066PracticableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasRudderProtectionArrangement ;
        sh:minCount 1 ;
        sh:node gen:TRF066KnifeOrEquivalentArrangementShape ;
        sh:message "When an ice knife is practicable, the rudder must have an ice knife extending below the LIWL or an equivalent protection means."
    ] .

gen:TRF066NotPracticableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasRudderProtectionArrangement ;
        sh:minCount 1 ;
        sh:node gen:TRF066EquivalentArrangementShape ;
        sh:message "When an ice knife is not practicable, an equivalent rudder protection means is required."
    ] .

gen:TRF066KnifeOrEquivalentArrangementShape
    a sh:NodeShape ;
    sh:or (
        gen:TRF066IceKnifeArrangementShape
        gen:TRF066EquivalentArrangementShape
    ) .

gen:TRF066IceKnifeArrangementShape
    a sh:NodeShape ;
    sh:class nltl:rudderProtectionArrangement ;
    sh:class nltl:iceKnife ;
    sh:property [
        sh:path nltl:extendsBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ice knife must extend below the lower ice waterline."
    ] .

gen:TRF066EquivalentArrangementShape
    a sh:NodeShape ;
    sh:class nltl:rudderProtectionArrangement ;
    sh:class nltl:equivalentRudderProtectionMeans .
<END_SHACL>
