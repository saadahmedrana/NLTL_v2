<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF069DesignConditionShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The design condition shall identify normal operational conditions."
    ] ;
    sh:property [
        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The design condition shall cover normal service-life propeller and ice loads."
    ] ;
    sh:property [
        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The design condition shall include fixed-pitch propeller reversal loads."
    ] ;
    sh:property [
        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The design condition shall record stopped-propeller dragging as outside the prescribed load model."
    ] ;
    sh:property [
        sh:path nltl:radialIceEntryOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The design condition shall record radial ice entry into a turned azimuthing thruster as outside the prescribed load model."
    ] .

gen:TRF069ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:message "The applicable propulsion-machinery rule requires ice class IA Super, IA, IB, or IC."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:message "The applicable propulsion-machinery rule requires an open or ducted propeller."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:message "The applicable propulsion-machinery rule requires a controllable-pitch or fixed-pitch design."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The applicable propulsion-machinery rule requires the maximum ice torque in kN*m."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:node gen:TRF069DesignConditionShape ;
        sh:minCount 1 ;
        sh:message "The applicable propulsion-machinery rule requires the prescribed design-load conditions."
    ] .

gen:TRF069NonApplicableShape a sh:NodeShape ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    {
                        $this nltl:iceClass ?value .
                        FILTER (?value NOT IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ))
                    }
                    UNION
                    {
                        $this nltl:propellerType ?value .
                        FILTER (?value NOT IN ("open", "ducted"))
                    }
                    UNION
                    {
                        $this nltl:pitchType ?value .
                        FILTER (?value NOT IN ("controllable", "fixed"))
                    }
                }
            }
        """ ;
        sh:message "The propulsion-machinery rule is non-applicable because at least one selector is outside the regulated combination."
    ] .

gen:TRF069Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF069ApplicableShape
        gen:TRF069NonApplicableShape
    ) .
<END_SHACL>
