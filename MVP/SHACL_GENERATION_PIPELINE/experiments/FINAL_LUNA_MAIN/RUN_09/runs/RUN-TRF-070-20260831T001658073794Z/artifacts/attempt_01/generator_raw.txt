<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-070> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an azimuthing or fixed main-propulsion thruster, required ice-load inputs, a normal-operational design condition, and a capacity meeting local ice pressure must be documented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:mainPropulsionThruster ?main ;
                       nltl:thrusterType ?type .

                FILTER (datatype(?main) = xsd:boolean)
                FILTER (?main = true)
                FILTER (datatype(?type) = xsd:string)
                FILTER (STR(?type) IN ("azimuthing", "fixed"))

                FILTER (
                    NOT EXISTS {
                        $this nltl:propellerIceInteractionLoad ?propellerLoad .
                        ?propellerLoad qudt:numericValue ?propellerValue ;
                                       qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?propellerValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterBodyIceInteractionLoad ?bodyLoad .
                        ?bodyLoad qudt:numericValue ?bodyValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?bodyValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:localIcePressure ?pressure .
                        ?pressure qudt:numericValue ?pressureValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?pressureValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDesignCondition ?condition .
                        ?condition a nltl:designCondition ;
                                   nltl:normalOperationalConditions ?normal ;
                                   nltl:designConditionLocalStrengthCapacity ?capacity .
                        FILTER (datatype(?normal) = xsd:boolean)
                        FILTER (?normal = true)
                        ?capacity qudt:numericValue ?capacityValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?capacityValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:localIcePressure ?pressure .
                        ?pressure qudt:numericValue ?pressureValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?pressureValue) = xsd:decimal)

                        $this nltl:hasDesignCondition ?condition .
                        ?condition a nltl:designCondition ;
                                   nltl:normalOperationalConditions true ;
                                   nltl:designConditionLocalStrengthCapacity ?capacity .
                        ?capacity qudt:numericValue ?capacityValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?capacityValue) = xsd:decimal)
                        FILTER (?capacityValue >= ?pressureValue)
                    }
                )
            }
        """
    ] .
<END_SHACL>
