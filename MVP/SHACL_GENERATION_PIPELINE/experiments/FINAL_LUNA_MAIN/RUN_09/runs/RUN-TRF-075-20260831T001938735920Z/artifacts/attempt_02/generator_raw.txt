<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:TRF075ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:TRF075LoadCaseShape ;
        sh:message "Each propeller blade load case must conform to the required blade, ice-block, and force-choice constraints."
    ] .

gen:TRF075LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseBlade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:propellerBlade ;
        sh:message "A propeller blade load case must link to exactly one propeller blade."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseIceBlock ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceBlock ;
        sh:message "A propeller blade load case must link to exactly one ice block."
    ] ;
    sh:xone (
        [
            sh:path nltl:backwardBladeForce ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:node [
                sh:class qudt:QuantityValue ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue unit:KiloN
                ]
            ] ;
            sh:message "The load case must provide exactly one backward blade force in kilonewtons."
        ]
        [
            sh:path nltl:forwardBladeForce ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:node [
                sh:class qudt:QuantityValue ;
                sh:property [
                    sh:path qudt:numericValue ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:property [
                    sh:path qudt:unit ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue unit:KiloN
                ]
            ] ;
            sh:message "The load case must provide exactly one forward blade force in kilonewtons."
        ]
    ) .
<END_SHACL>
