<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_076_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:message "When provided, propeller type must be an xsd:string."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:TRF_076_LoadCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An FP propeller must provide a boolean propeller reversing capability value to select the applicable load-case requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (STR(?type) = "FP")
                FILTER NOT EXISTS {
                    $this nltl:propellerReversingCapability ?reversing .
                    FILTER (datatype(?reversing) = xsd:boolean)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "CP and applicable FP propellers must cover load cases 1 through 4; reversing FP propellers must also cover load case 5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (STR(?type) = "CP" || STR(?type) = "FP")

                FILTER (
                    STR(?type) = "CP"
                    ||
                    EXISTS {
                        $this nltl:propellerReversingCapability ?reversing .
                        FILTER (
                            datatype(?reversing) = xsd:boolean
                            && ?reversing = true
                        )
                    }
                )

                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case1 .
                    ?case1 nltl:propellerBladeLoadCaseNumber ?number1 .
                    FILTER (?number1 = 1)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case2 .
                    ?case2 nltl:propellerBladeLoadCaseNumber ?number2 .
                    FILTER (?number2 = 2)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case3 .
                    ?case3 nltl:propellerBladeLoadCaseNumber ?number3 .
                    FILTER (?number3 = 3)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case4 .
                    ?case4 nltl:propellerBladeLoadCaseNumber ?number4 .
                    FILTER (?number4 = 4)
                }

                FILTER (
                    STR(?type) != "FP"
                    ||
                    NOT EXISTS {
                        $this nltl:propellerReversingCapability ?reversing5 .
                        FILTER (
                            datatype(?reversing5) = xsd:boolean
                            && ?reversing5 = true
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case5 .
                        ?case5 nltl:propellerBladeLoadCaseNumber ?number5 .
                        FILTER (?number5 = 5)
                    }
                )
            }
        """
    ] .

gen:TRF_076_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented propeller blade load case must have exactly one integer load case number."
    ] .
<END_SHACL>
