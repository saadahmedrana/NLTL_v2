    ] ;

        sh:node gen:TRF078DimensionlessQuantityShape ;

    ] ;
    sh:property [

    sh:sparql [

            SELECT $this
            WHERE {
                {

                }

                    $this nltl:hasTableLookupCase ?case .

                    }
                }
            }

    ] ;
    sh:sparql [
        sh:message "Each Table 6-7 lookup case must be ship-linked, typed, reference Table 6-7, and provide one propeller-location selection key and one selected location factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    $this nltl:hasTableLookupCase ?case .
                    ?case nltl:tableReference nltl:traficomTable6Dash7 .
                    FILTER NOT EXISTS { ?case a nltl:tableLookupCase . }

                }

        sh:message "The applicable Table 6-6 and Table 6-7 lookup cases must be unique among ship-linked valid cases, and iceLoadCycleCount must equal k1*k2*k3*N_class*n_n within an expected-result-scaled tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?reported ;
                      nltl:hasTableLookupCase ?case6 ;
                      nltl:hasTableLookupCase ?case7 .

                ?k2Node qudt:numericValue ?k2 ;
                        qudt:unit unit:UNITLESS .
                ?k3Node qudt:numericValue ?k3 ;
                        qudt:unit unit:UNITLESS .

                        qudt:unit unit:UNITLESS .

                ?case6 a nltl:tableLookupCase ;
                        nltl:tableReference nltl:traficomTable6Dash6 ;

                        nltl:selectedIceClassCycleCount ?nclassNode .

                    nltl:iceClassIa,
                    nltl:iceClassIaSuper,
                    nltl:iceClassIb,
                    nltl:iceClassIc,
                    nltl:iceClassIi,
                    nltl:iceClassIii
                ))
                ?nclassNode qudt:numericValue ?nclass ;
                            qudt:unit unit:UNITLESS .

                ?case7 a nltl:tableLookupCase ;
                        nltl:tableReference nltl:traficomTable6Dash7 ;
                        nltl:tableLookupPropellerLocation ?location ;
                        nltl:selectedPropellerLocationFactor ?k1Node .
                ?location a nltl:propellerLocationValue .
                ?k1Node qudt:numericValue ?k1 ;
                        qudt:unit unit:UNITLESS .

                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?other6 .
                    ?other6 a nltl:tableLookupCase ;
                            nltl:tableReference nltl:traficomTable6Dash6 ;
                            nltl:iceClass ?otherIceClass ;
                            nltl:selectedIceClassCycleCount ?otherNclassNode .
                    ?otherIceClass a nltl:iceClassValue .
                    FILTER (?otherIceClass IN (
                        nltl:iceClassIa,
                        nltl:iceClassIaSuper,
                        nltl:iceClassIb,
                        nltl:iceClassIc,
                        nltl:iceClassIi,
                        nltl:iceClassIii
                    ))
                    ?otherNclassNode qudt:numericValue ?otherNclass ;
                                        qudt:unit unit:UNITLESS .
                    FILTER (?other6 != ?case6)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?other7 .
                    ?other7 a nltl:tableLookupCase ;
                            nltl:tableReference nltl:traficomTable6Dash7 ;
                            nltl:tableLookupPropellerLocation ?otherLocation ;
                            nltl:selectedPropellerLocationFactor ?otherK1Node .

                    ?otherK1Node qudt:numericValue ?otherK1 ;
                                 qudt:unit unit:UNITLESS .
                    FILTER (?other7 != ?case7)
                }

                BIND ((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (
                    ?reported < (?expected - (0.000001 * ?expected)) ||
                    ?reported > (?expected + (0.000001 * ?expected))
                )
            }
        """
