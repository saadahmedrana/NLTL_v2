<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one ice load cycle count." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one propeller blade count." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The ship must provide one unitless effective ice load cycle count." ;
    ] ;
    sh:sparql [
        sh:message "effectiveIceLoadCycleCount must equal iceLoadCycleCount multiplied by propellerBladeCount, within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?cycles ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?effectiveNode .
                ?effectiveNode qudt:numericValue ?reported .
                BIND ((?cycles * ?blades) AS ?expected)
                FILTER (
                    ABS(?reported - ?expected) >
                    (0.000001 * ABS(?expected) + 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
