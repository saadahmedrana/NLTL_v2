<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF081QuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The force quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The force quantity must use exactly one kN unit."
    ] .

gen:TRF081Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one forward blade force in kN."
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one backward blade force in kN."
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one hydrodynamic bollard thrust in kN."
    ] ;
    sh:property [
        sh:path nltl:forwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one forward ice thrust in kN."
    ] ;
    sh:property [
        sh:path nltl:backwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one backward ice thrust in kN."
    ] ;
    sh:property [
        sh:path nltl:forwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one forward propeller shaft design thrust in kN."
    ] ;
    sh:property [
        sh:path nltl:backwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one backward propeller shaft design thrust in kN."
    ] ;
    sh:property [
        sh:path nltl:propellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF081QuantityValueShape ;
        sh:message "Each ship must provide one propeller shaft design thrust in kN."
    ] ;
    sh:sparql [
        sh:message "Ice and design thrusts must satisfy the TRF-081 formulas. Derived decimal results are checked with relative tolerance 1e-6 plus absolute tolerance 1e-9 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce ?ffQ ;
                      nltl:backwardBladeForce ?fbQ ;
                      nltl:hydrodynamicBollardThrust ?tQ ;
                      nltl:forwardIceThrust ?tfQ ;
                      nltl:backwardIceThrust ?tbQ ;
                      nltl:forwardPropellerShaftDesignThrust ?forwardDesignQ ;
                      nltl:backwardPropellerShaftDesignThrust ?backwardDesignQ ;
                      nltl:propellerShaftDesignThrust ?designQ .

                ?ffQ qudt:numericValue ?ff .
                ?fbQ qudt:numericValue ?fb .
                ?tQ qudt:numericValue ?t .
                ?tfQ qudt:numericValue ?tf .
                ?tbQ qudt:numericValue ?tb .
                ?forwardDesignQ qudt:numericValue ?forwardDesign .
                ?backwardDesignQ qudt:numericValue ?backwardDesign .
                ?designQ qudt:numericValue ?design .

                BIND((1.1 * ?ff) AS ?tfExpected)
                BIND((1.1 * ?fb) AS ?tbExpected)
                BIND((?t + (2.2 * ?tfExpected)) AS ?forwardExpected)
                BIND((1.5 * ?tbExpected) AS ?backwardExpected)
                BIND(IF(?forwardExpected >= ?backwardExpected,
                        ?forwardExpected,
                        ?backwardExpected) AS ?designExpected)

                FILTER (
                    ABS(?tf - ?tfExpected) >
                        ((ABS(?tfExpected) * 0.000001) + 0.000000001)
                    ||
                    ABS(?tb - ?tbExpected) >
                        ((ABS(?tbExpected) * 0.000001) + 0.000000001)
                    ||
                    ABS(?forwardDesign - ?forwardExpected) >
                        ((ABS(?forwardExpected) * 0.000001) + 0.000000001)
                    ||
                    ABS(?backwardDesign - ?backwardExpected) >
                        ((ABS(?backwardExpected) * 0.000001) + 0.000000001)
                    ||
                    ABS(?design - ?designExpected) >
                        ((ABS(?designExpected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
