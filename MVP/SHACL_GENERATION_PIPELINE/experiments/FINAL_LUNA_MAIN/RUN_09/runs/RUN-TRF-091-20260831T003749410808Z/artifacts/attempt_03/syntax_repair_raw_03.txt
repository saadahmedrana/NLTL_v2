<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-091-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each propeller ice-milling load sequence case must be the propulsion load case used for strength evaluation and must not be designated for stalling analysis." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasIceMillingSequenceCase ?iceCase .
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?iceCase .
                    ?iceCase nltl:loadCasePurpose
                        nltl:propulsionLineStrengthEvaluationPurpose .
                    FILTER NOT EXISTS {
                        ?iceCase nltl:loadCasePurpose
                            nltl:stallingAnalysisPurposeValue .
                    }
                }
            }
        """
    ] .
</END_SHACL>
