    a sh:NodeShape ;
    sh:node nltlg:ShipCalculationInputsShape ;
    sh:property [
        sh:path nltl:approvedNonlinearBladeStressAnalysis ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:approvalRecord ;
        sh:message "An approved nonlinear blade stress analysis record is required."
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;

        sh:message "An approved-analysis case must include a canonical load case."
    ] .

nltlg:CalculatedBranch
    a sh:NodeShape ;
    sh:node nltlg:ShipCalculationInputsShape ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:class nltl:loadCase ;
        sh:node nltlg:LoadCaseShape ;
        sh:message "A calculated blade-failure load must include a canonical load case."
    ] ;
    sh:sparql [
        sh:message "The calculated reference strength and blade failure ultimate load do not satisfy the declared formula within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection/qudt:numericValue ?c ;
                      nltl:bladeCylindricalRootRadius/qudt:numericValue ?r ;
                      nltl:bladeFailureUltimateLoad/qudt:numericValue ?observedLoad ;
                      nltl:bladeMaximumThicknessAtWeakestSection/qudt:numericValue ?t ;
                      nltl:bladeMinimumYieldStrength/qudt:numericValue ?yield ;
                      nltl:bladeReferenceStrength1/qudt:numericValue ?observedRef ;
                      nltl:bladeUltimateTensileStrength/qudt:numericValue ?ultimate ;
                      nltl:propellerDiameter/qudt:numericValue ?diameter .

                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedRef)
                BIND ((300.0 * ?c * ?t * ?t * ?expectedRef) /
                      ((0.8 * ?diameter) - (2.0 * ?r)) AS ?expectedLoad)

                FILTER (
                    (?expectedRef >= 0.0 &&
                     (?observedRef < (?expectedRef - ((?expectedRef * 0.000001) + 0.000000001)) ||
                      ?observedRef > (?expectedRef + ((?expectedRef * 0.000001) + 0.000000001))))
                    ||
                    (?expectedRef < 0.0 &&
                     (?observedRef < (?expectedRef - ((-?expectedRef * 0.000001) + 0.000000001)) ||
                      ?observedRef > (?expectedRef + ((-?expectedRef * 0.000001) + 0.000000001))))
                    ||
                    (?expectedLoad >= 0.0 &&
                     (?observedLoad < (?expectedLoad - ((?expectedLoad * 0.000001) + 0.000000001)) ||
                      ?observedLoad > (?expectedLoad + ((?expectedLoad * 0.000001) + 0.000000001))))
                    ||
                    (?expectedLoad < 0.0 &&
                     (?observedLoad < (?expectedLoad - ((-?expectedLoad * 0.000001) + 0.000000001)) ||
                      ?observedLoad > (?expectedLoad + ((-?expectedLoad * 0.000001) + 0.000000001))))
                )
            }
        """
    ] .

nltlg:BladeFailureRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltlg:ApprovedAnalysisBranch
        nltlg:CalculatedBranch
    ) ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade failure status must be recorded."
    ] ;
    sh:sparql [
        sh:message "Blade failure status must be true exactly when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset/qudt:numericValue ?tipOffset ;
                      nltl:propellerDiameter/qudt:numericValue ?diameter ;
                      nltl:bladeFailureStatus ?status .

                FILTER (
                    (?tipOffset > (0.10 * ?diameter) && ?status != true) ||
                    (?tipOffset <= (0.10 * ?diameter) && ?status != false)
                )
            }
        """
    ] .
