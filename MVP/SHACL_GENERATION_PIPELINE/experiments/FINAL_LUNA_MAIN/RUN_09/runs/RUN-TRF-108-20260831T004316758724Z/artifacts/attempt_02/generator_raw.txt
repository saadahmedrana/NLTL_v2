@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-108-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF-108-KnownCurveBranch>
        <urn:nltl:generated-shape:TRF-108-UnknownCurveBranch>
    ) .

<urn:nltl:generated-shape:TRF-108-KnownCurveBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "The selected SN curve, blade material property, and correspondence evidence must be represented for the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:stressLifeCurveSelection ?curve ;
                      nltl:propellerBladeMaterialProperty ?material ;
                      nltl:stressLifeCurveSelectionCorrespondenceEvidence ?evidence .
                ?evidence a nltl:evidenceArtifact .
            }
        """
    ] .
