<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ExtremeIceForceQuantityShape ;
        sh:message "The ship shall provide the externally calculated extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:node [
            sh:or (
                gen:TRF111ExtremeOperationalLoadCaseShape
                gen:TRF111FatigueLoadCaseShape
                gen:TRF111BladeFailureLoadCaseShape
            )
        ] ;
        sh:message "Each propulsion load case shall identify an applicable shaft-line load-case type and satisfy its associated readiness check."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:node gen:TRF111ShaftLineComponentShape
    ] .

gen:ExtremeIceForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SafetyFactorQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF111ExtremeOperationalLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:hasValue nltl:extremeOperationalLoadCase ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF111YieldFactorAtLeast130Shape ;
        sh:message "The extreme operational load case shall have a component yield safety factor of at least 1.3."
    ] .

gen:TRF111FatigueLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:hasValue nltl:fatigueLoadCase ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF111BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:hasValue nltl:bladeFailureLoadCase ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF111YieldFactorAtLeast100Shape ;
        sh:message "The blade failure load case shall have a component yield safety factor of at least 1.0."
    ] .

gen:TRF111ShaftLineComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:fatigueSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF111FatigueFactorAtLeast150Shape ;
        sh:message "Each propeller shaft line component shall have a fatigue safety factor of at least 1.5."
    ] .

gen:TRF111YieldFactorAtLeast130Shape
    a sh:NodeShape ;
    sh:node gen:SafetyFactorQuantityShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 1.3
    ] .

gen:TRF111YieldFactorAtLeast100Shape
    a sh:NodeShape ;
    sh:node gen:SafetyFactorQuantityShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 1.0
    ] .

gen:TRF111FatigueFactorAtLeast150Shape
    a sh:NodeShape ;
    sh:node gen:SafetyFactorQuantityShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 1.5
    ] .
<END_SHACL>
