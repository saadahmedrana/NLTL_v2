<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF113Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:steeringMechanism
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must identify a steering mechanism component."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:thrusterFitting
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must identify the thruster unit fitting component."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape [
            sh:class nltl:thrusterBody
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must identify the thruster body component."
    ] ;
    sh:property [
        sh:path nltl:bladePlasticBending ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must record the plastic blade-bending design condition."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeOrientation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must record the propeller blade orientation producing the maximum component load."
    ] ;
    sh:property [
        sh:path nltl:thrusterBladeLossDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "The ship must provide evidence for the blade-loss design assessment."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The blade-loss assessment must link the recorded maximum-load orientation to a propulsion load case confirmed for the maximum component load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:propellerBladeOrientation ?orientation .
                    $this nltl:hasPropulsionLoadCase ?case .
                    ?case a nltl:loadCase .
                    ?case nltl:maximumComponentLoadBladeOrientationConfirmed ?confirmed .
                    FILTER ( ?confirmed = true )
                }
            }
        """
    ] .
</END_SHACL>
