<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF123ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node gs:TRF123OccasionalForceLoadCaseShape ;
        sh:message "The ship must provide an occasional-force load case for the required analysis."
    ] .

gs:TRF123OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The occasional-force analysis case must confirm that propeller blades are excluded."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF123SafetyFactorQuantityShape ;
        sh:message "The occasional-force analysis case must provide the externally calculated component yield safety factor."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:node gs:TRF123AssessedComponentShape ;
        sh:message "The occasional-force analysis case must identify each assessed component."
    ] ;
    sh:sparql [
        sh:message "The externally calculated occasional-force component stress, including the declared safety factor, must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:componentYieldSafetyFactor ?safetyFactorQuantity .
                ?safetyFactorQuantity qudt:numericValue ?safetyFactor .

                $this nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .

                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yieldStrength .

                FILTER (
                    ?stress * ?safetyFactor > ?yieldStrength
                )
            }
        """
    ] .

gs:TRF123AssessedComponentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF123StressQuantityShape ;
        sh:message "Each assessed component must provide its externally calculated occasional-force stress."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:class nltl:materialProperties ;
        sh:minCount 1 ;
        sh:node gs:TRF123MaterialPropertiesShape ;
        sh:message "Each assessed component must identify its material properties."
    ] .

gs:TRF123MaterialPropertiesShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF123YieldStrengthQuantityShape ;
        sh:message "The component material properties must provide the material yield strength."
    ] .

gs:TRF123SafetyFactorQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF123YieldStrengthQuantityShape
