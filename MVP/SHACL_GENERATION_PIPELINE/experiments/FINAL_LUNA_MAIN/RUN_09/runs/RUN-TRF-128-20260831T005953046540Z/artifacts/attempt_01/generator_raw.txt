<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF128Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF128IaSuperReversal
        gen:TRF128NoReversal
        gen:TRF128OtherIceClassReversal
    ) .

gen:TRF128IaSuperReversal
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "0.5"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR
            ]
        ]
    ] ;
    sh:message "An IA Super ship requiring propulsion engine reversal for astern must have compressor charge time at most 0.5 hour." .

gen:TRF128NoReversal
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "1"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR
            ]
        ]
    ] ;
    sh:message "A ship not requiring propulsion engine reversal for astern must have compressor charge time at most 1 hour." .

gen:TRF128OtherIceClassReversal
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:not [
            sh:hasValue nltl:iceClassIaSuper
        ]
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "1"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR
            ]
        ]
    ] ;
    sh:message "A non-IA Super ship must have compressor charge time at most 1 hour, including when propulsion engine reversal is required for astern." .
<END_SHACL>
