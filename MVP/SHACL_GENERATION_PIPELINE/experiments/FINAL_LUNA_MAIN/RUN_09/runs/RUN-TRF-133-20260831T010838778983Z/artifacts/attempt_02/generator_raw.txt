<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice-class value."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:class nltl:hullStructure ;
        sh:node gen:TRF133WarningTriangleShape ;
        sh:minCount 1 ;
        sh:message "The ship must have a conforming warning triangle marking."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:class nltl:hullStructure ;
        sh:node gen:TRF133IceClassDraughtMarkingShape ;
        sh:minCount 1 ;
        sh:message "The ship must have a conforming ice-class draught marking."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Timber load line applicability must be explicitly recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:timberLoadLineVerticalReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:loadLineRingCentreReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "The ice-class draught mark must use the reference point selected by the timber load line applicability value." .

    sh:property [

        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "300"^^xsd:decimal ;
                sh:maxInclusive "300"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each warning triangle side must be 300 mm."
    ] ;
    sh:property [
        sh:path nltl:verticalOffsetAboveSummerFreshWaterLoadLine ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "1000"^^xsd:decimal ;
                sh:maxInclusive "1000"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1

            sh:property [

                sh:hasValue unit:MilliM ;

gen:TRF133IceClassDraughtMarkingShape
    a sh:NodeShape ;
    sh:class nltl:iceClassDraughtMarking ;
    sh:property [
        sh:path nltl:draughtMarkAftOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "540"^^xsd:decimal ;
                sh:maxInclusive "540"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ice-class draught mark must be 540 mm abaft its applicable reference point."
    ] ;
    sh:property [
        sh:path nltl:draughtMarkAftReferencePoint ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:loadLineRingCentreReference
            nltl:timberLoadLineVerticalReference
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The draught mark must identify a permitted reference point."
    ] ;
    sh:property [
        sh:path nltl:letterDimensionsEqualLoadLineMark ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "All ice-class draught-mark letters must have the load-line mark dimensions."
    ] .
