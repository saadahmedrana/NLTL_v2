<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_018_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:minCount 1 ;
        sh:message "The ship must identify its design ice load patch."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPeakPressureFactorLookupCase ;
        sh:minCount 1 ;
        sh:node gen:I2_018_LookupCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:I2_018_LookupCaseShape
    ] ;
    sh:sparql [
        sh:message "The peak-pressure lookup case and table lookup case must be the same shared case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasPeakPressureFactorLookupCase ?peakCase ;
                      nltl:hasTableLookupCase ?tableCase .
                FILTER (?peakCase != ?tableCase)
            }
        """
    ] ;
    sh:sparql [
        sh:message "averageIcePressure must equal iceForce divided by the load-patch area, converted from kN/m2 to MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceForce ?forceQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:loadPatchLength ?lengthQuantity ;
                      nltl:averageIcePressure ?pressureQuantity .
                ?forceQuantity qudt:numericValue ?force .
                ?heightQuantity qudt:numericValue ?height .
                ?lengthQuantity qudt:numericValue ?length .
                ?pressureQuantity qudt:numericValue ?pressure .
                FILTER (?height != 0 && ?length != 0)
                BIND ((?force / (?height * ?length * 1000)) AS ?expected)
                BIND (((?expected * 0.000001) + 0.000000000001) AS ?tolerance)
                FILTER (
                    ?pressure < (?expected - ?tolerance) ||
                    ?pressure > (?expected + ?tolerance)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "peakPressureFactor must match the selected factor from the shared Table 3 lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor ?factorQuantity ;
                      nltl:hasPeakPressureFactorLookupCase ?lookupCase .
                ?lookupCase nltl:selectedPeakPressureFactor ?selectedQuantity .
                ?factorQuantity qudt:numericValue ?factor .
                ?selectedQuantity qudt:numericValue ?selected .
                BIND (((?selected * 0.000001) + 0.000000000001) AS ?tolerance)
                FILTER (
                    ?factor < (?selected - ?tolerance) ||
                    ?factor > (?selected + ?tolerance)
                )
            }
        """
    ] .

    sh:property [

        sh:maxCount 1 ;
        sh:in (
            nltl:bottomStructureFrameCategory
            nltl:longitudinallyFramedPlatingCategory
            nltl:stringerSideLongitudinalOrWebFrameCategory
            nltl:transverseFrameWithLoadDistributingStringersCategory
            nltl:transverseFrameWithoutLoadDistributingStringersCategory
            nltl:transverselyFramedPlatingCategory
        )
    ] ;
    sh:property [
        sh:path nltl:selectedPeakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iacsUrI2Table3
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] .
</END_SHACL>
