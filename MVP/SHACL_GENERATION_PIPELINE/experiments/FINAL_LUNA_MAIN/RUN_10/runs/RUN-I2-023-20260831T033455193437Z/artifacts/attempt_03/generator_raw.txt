gen:NonApplicablePlatingShape
    a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AngleQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The plating is applicable only when framing angle Omega is at most 20 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:framingAngleOmega ?omegaNode .
                ?omegaNode qudt:numericValue ?omega .
                FILTER ( ?omega <= 20 )
            }
        """
    ] .

gen:ApplicableBranchBGreaterEqualSShape
    a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:platingSupportedByStructuralMember ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMmQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For applicable plating, the b >= s calculation branch requires Omega <= 20 degrees and load patch height b >= frame spacing s." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasPlating $this .
                $this nltl:framingAngleOmega ?omegaNode ;
                      nltl:platingSupportedByStructuralMember ?member .
                ?omegaNode qudt:numericValue ?omega .
                ?member nltl:frameSpacing ?sNode .
                ?sNode qudt:numericValue ?s .
                ?ship nltl:loadPatchHeight ?bNode .
                ?bNode qudt:numericValue ?b .
                FILTER ( ?omega > 20 || ?b < ?s )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The applicable b >= s branch is missing one or more required ship-owned formula operands." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasPlating $this .
                FILTER (
                    NOT EXISTS {
                        ?ship nltl:averageIcePressure ?p .
                        ?p qudt:numericValue ?v .
                        ?p qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    }
                    ||
                    NOT EXISTS {
                        ?ship nltl:frameSpan ?p .
                        ?p qudt:numericValue ?v .
                        ?p qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                    ||
                    NOT EXISTS {
                        ?ship nltl:loadPatchHeight ?p .
                        ?p qudt:numericValue ?v .
                        ?p qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                    ||
                    NOT EXISTS {
                        ?ship nltl:peakPressureFactor ?p .
                        ?p qudt:numericValue ?v .
                        ?p qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    }
                    ||
                    NOT EXISTS {
                        ?ship nltl:yieldStrength ?p .
                        ?p qudt:numericValue ?v .
                        ?p qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    }
                )
            }
        """
    ] .

    sh:targetClass nltl:ship ;

        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:node [
