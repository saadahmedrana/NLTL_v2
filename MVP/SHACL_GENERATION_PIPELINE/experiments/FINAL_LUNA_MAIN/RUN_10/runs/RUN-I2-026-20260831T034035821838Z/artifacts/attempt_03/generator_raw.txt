<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node gen:I2_026_MemberSupportShape ;
        sh:message "Each structural member must provide a member support with an explicit support-condition selection."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Continuity through the support must be explicitly recorded."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Termination within the ice-strengthened area must be explicitly recorded."
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:maxCount 1 ;
        sh:message "Rotational-restraint evidence must reference an evidence artifact."
    ] ;
    sh:sparql [
        sh:message "Support fixity must be fixed when termination, continuity, a connection bracket, or demonstrated rotational restraint with evidence applies; otherwise it must be simple, and evidence cannot be present without demonstrated restraint." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:frameSupportCondition ?condition .
                FILTER (
                    (
                        ?condition = nltl:fixedFrameSupport
                        &&
                        !(
                            EXISTS {
                                $this nltl:terminatesWithinIceStrengthenedArea true
                            }
                            ||
                            EXISTS {
                                $this nltl:continuousThroughSupport true
                            }
                            ||
                            EXISTS {
                                ?support nltl:connectionBracketPresent true
                            }
                            ||
                            (
                                EXISTS {
                                    ?support nltl:demonstratedRotationalRestraint true
                                }
                                &&
                                EXISTS {
                                    $this nltl:significantRotationalRestraintEvidence ?evidence
                                }
                            )
                        )
                    )
                    ||
                    (
                        ?condition = nltl:simpleFrameSupport
                        &&
                        (
                            EXISTS {
                                $this nltl:terminatesWithinIceStrengthenedArea true
                            }
                            ||
                            EXISTS {
                                $this nltl:continuousThroughSupport true
                            }
                            ||
                            EXISTS {
                                ?support nltl:connectionBracketPresent true
                            }
                            ||
                            EXISTS {
                                $this nltl:significantRotationalRestraintEvidence ?evidence
                            }
                            ||
                            EXISTS {
                                ?support nltl:demonstratedRotationalRestraint true
                            }
                        )
                    )
                )
            }
        """
    ] .

gen:I2_026_MemberSupportShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The support condition must be explicitly selected as fixed or simple."
    ] ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Connection-bracket presence must be explicitly recorded."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Demonstrated rotational restraint must be explicitly recorded."
    ] .
<END_SHACL>
