<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gs:CalculationCaseShape
    ] .

gs:CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each calculation case shall record that attached shell plating is included in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each calculation case shall record that attached shell plating is excluded from the shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Each calculation case shall explicitly represent whether flange material is included in the shear area."
    ] ;
    sh:or (
        gs:FlangeMaterialExcludedShape
        gs:FlangeMaterialIncludedShape
    ) .

gs:FlangeMaterialExcludedShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] .

gs:FlangeMaterialIncludedShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:class nltl:structuralMember ;
        sh:node gs:FittedStructuralMemberShape ;
        sh:message "When flange material is included in the shear area, the calculation case shall identify a structural member with a fitted flange."
    ] .

gs:FittedStructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The structural member shall record that its flange is fitted when flange material is included in the shear area."
    ] .
<END_SHACL>
