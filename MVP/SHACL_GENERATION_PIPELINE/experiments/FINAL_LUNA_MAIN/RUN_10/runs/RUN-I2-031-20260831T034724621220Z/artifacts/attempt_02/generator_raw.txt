<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node gen:I2_031_ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the patch load in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall identify the plastic collapse mechanism used to define plastic strength."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_031_ApplicableMemberShape ;
        sh:message "Each linked applicable structural member shall satisfy the local-frame applicability and strength constraints."
    ] .

gen:I2_031_ApplicableMemberShape
    a sh:NodeShape ;
    sh:or (
        gen:I2_031_BottomMemberBranch
        gen:I2_031_SideTransverseMemberBranch
    ) ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:I2_031_ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable member shall provide its plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:I2_031_ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable member shall provide the midspan plastic-collapse load in kN."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand shall not exceed plastic strength, and plastic strength shall equal the expected midspan plastic-collapse load within a relative tolerance of 1e-6 with a 1e-9 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strength ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                $this nltl:midspanPlasticCollapseLoad ?midspanQuantity .
                ?midspanQuantity qudt:numericValue ?midspan ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                FILTER (
                    ?demand > ?strength ||
                    ABS(?strength - ?midspan) >
                    (ABS(?midspan) * "0.000001"^^xsd:decimal + "0.000000001"^^xsd:decimal)
                )
            }
        """
    ] .

gen:I2_031_BottomMemberBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:bottomStructureLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A bottom applicable member shall identify bottom structure location."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_031_BottomLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "A bottom applicable member shall have a load case with the required patch-load orientation."
    ] .

gen:I2_031_SideTransverseMemberBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A side applicable member shall identify side structure location."
    ] ;
    sh:property [
        sh:path nltl:transverseFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A side applicable member shall be identified as a transverse local frame."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_031_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "A side transverse local frame shall have a load case."
    ] .

gen:I2_031_BottomLoadCaseShape
    a sh:NodeShape ;
    sh:node gen:I2_031_LoadCaseShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "For bottom structure, the patch-load dimension b shall be parallel with the frame direction."
    ] .

gen:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_031_ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case shall provide combined shear and bending demand in kN."
    ] .

gen:I2_031_ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
