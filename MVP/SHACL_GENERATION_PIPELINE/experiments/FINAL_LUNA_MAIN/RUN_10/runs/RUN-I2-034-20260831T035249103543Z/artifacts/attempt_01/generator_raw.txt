<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node nltl:I2_034_StructuralMemberShape
    ] .

nltl:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "A non-longitudinal structural member is not applicable to the longitudinal side-frame requirement."
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The structural member must identify itself as longitudinal to establish applicability."
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:sideStructureLocation ;
                sh:message "An applicable longitudinal member must be located in the side structure."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node nltl:I2_034_LoadCaseShape
            ] ;
            sh:sparql [
                sh:message "Each represented load case for an applicable member requires a valid plastic-strength or midspan-plastic-collapse-load input." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasStructuralMemberLoadCase ?loadCase .
                        FILTER (
                            !EXISTS {
                                $this nltl:plasticStrength ?strength .
                                ?strength qudt:numericValue ?strengthValue ;
                                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                                FILTER (datatype(?strengthValue) = xsd:decimal)
                            }
                            &&
                            !EXISTS {
                                $this nltl:midspanPlasticCollapseLoad ?collapseLoad .
                                ?collapseLoad qudt:numericValue ?collapseValue ;
                                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                                FILTER (datatype(?collapseValue) = xsd:decimal)
                            }
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The combined shear-and-bending demand must not exceed the member plastic strength or midspan plastic-collapse load." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasStructuralMemberLoadCase ?loadCase .
                        ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                        ?demand qudt:numericValue ?demandValue .
                        FILTER (datatype(?demandValue) = xsd:decimal)
                        FILTER (
                            EXISTS {
                                $this nltl:plasticStrength ?strength .
                                ?strength qudt:numericValue ?strengthValue ;
                                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                                FILTER (datatype(?strengthValue) = xsd:decimal)
                                FILTER (?demandValue > ?strengthValue)
                            }
                            &&
                            EXISTS {
                                $this nltl:midspanPlasticCollapseLoad ?collapseLoad .
                                ?collapseLoad qudt:numericValue ?collapseValue ;
                                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                                FILTER (datatype(?collapseValue) = xsd:decimal)
                                FILTER (?demandValue > ?collapseValue)
                            }
                        )
                    }
                """
            ]
        )
    ] .

nltl:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:I2_034_DemandQuantityShape ;
        sh:message "Each represented load case must provide one combined shear-and-bending demand quantity."
    ] .

nltl:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:I2_034_StructuralMemberResultShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:plasticStrength ;
                sh:class qudt:QuantityValue ;
                sh:maxCount 1 ;
                sh:node nltl:I2_034_KiloNewtonQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:midspanPlasticCollapseLoad ;
                sh:class qudt:QuantityValue ;
                sh:maxCount 1 ;
                sh:node nltl:I2_034_KiloNewtonQuantityShape
            ]
        ]
    ) .

nltl:I2_034_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .
<END_SHACL>
