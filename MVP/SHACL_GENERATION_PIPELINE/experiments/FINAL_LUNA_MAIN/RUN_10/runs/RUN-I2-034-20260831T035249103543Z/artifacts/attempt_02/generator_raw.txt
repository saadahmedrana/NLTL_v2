<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_StructuralMemberShape
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state whether it is longitudinal."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state its structural location."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:plasticStrength ;
                        sh:node gen:I2_034_KiloNewtonQuantityShape ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:midspanPlasticCollapseLoad ;
                        sh:node gen:I2_034_KiloNewtonQuantityShape ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ) ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node gen:I2_034_LoadCaseShape
            ]
        ]
    ) .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_034_DemandQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case for an applicable member must provide its combined shear-and-bending demand."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear-and-bending demand must not exceed the represented plastic strength or midspan plastic-collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                {
                    $this ^nltl:hasStructuralMemberLoadCase ?member .
                    ?member nltl:plasticStrength ?capacity .
                    ?capacity qudt:numericValue ?capacityValue .
                    FILTER(?demandValue > ?capacityValue)
                }
                UNION
                {
                    $this ^nltl:hasStructuralMemberLoadCase ?member .
                    ?member nltl:midspanPlasticCollapseLoad ?capacity .
                    ?capacity qudt:numericValue ?capacityValue .
                    FILTER(?demandValue > ?capacityValue)
                }
            }
        """
    ] .

gen:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_034_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
