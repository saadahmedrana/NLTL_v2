<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship shall declare one frame profile type."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_StructuralMemberShape
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member shall declare whether it is longitudinal."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member shall declare its structural location."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue false
            ]
        ]
        gen:I2_034_NonApplicableLocationShape
        gen:I2_034_PlasticStrengthBranchShape
        gen:I2_034_MidspanCollapseLoadBranchShape
    ) .

gen:I2_034_NonApplicableLocationShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "This branch applies only to longitudinal members outside side structures." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation .
            }
        """
    ] .

gen:I2_034_PlasticStrengthBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:I2_034_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable member shall provide exactly one plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:maxCount 0
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_034_LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "The combined shear-and-bending demand shall not exceed the selected plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase ;
                      nltl:plasticStrength ?capacity .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?demand > ?capacityValue)
            }
        """
    ] .

gen:I2_034_MidspanCollapseLoadBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:I2_034_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable member shall provide exactly one midspan plastic collapse load in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:maxCount 0
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_034_LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "The combined shear-and-bending demand shall not exceed the selected midspan plastic collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase ;
                      nltl:midspanPlasticCollapseLoad ?capacity .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?demand > ?capacityValue)
            }
        """
    ] .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_034_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case shall provide one combined shear-and-bending demand in kN."
    ] .

gen:I2_034_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
