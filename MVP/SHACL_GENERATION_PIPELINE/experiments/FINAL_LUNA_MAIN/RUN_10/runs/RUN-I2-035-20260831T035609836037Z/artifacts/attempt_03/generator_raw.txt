        sh:message "The reported required longitudinal frame shear area must equal the calculated value within a relative tolerance of 0.000001, with an absolute tolerance floor of 0.000001 cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpacing/qudt:numericValue ?s ;
                      nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                      nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reportedAL .
                ?ship nltl:hasStructuralMember $this ;
                       nltl:averageIcePressure/qudt:numericValue ?pavg ;
                       nltl:frameSpan/qudt:numericValue ?b ;
                       nltl:peakPressureFactor/qudt:numericValue ?ppfs ;
                       nltl:yieldStrength/qudt:numericValue ?sigmaY .
                FILTER (?s != 0)
                BIND (?b / ?s AS ?bprime)
                BIND (1 - (0.3 / ?bprime) AS ?ko)
                BIND (IF(?bprime < 2, ?b * (1 - (0.25 * ?bprime)), ?s) AS ?b2)
                BIND (?ko * ?b2 AS ?b1)
                BIND ((100 * 100 * (?af * ?ppfs * ?pavg) * 0.5 * ?b1 * ?b) / (0.577 * ?sigmaY) AS ?expectedAL)
                BIND (IF(?expectedAL >= 0, ?expectedAL * 0.000001, -?expectedAL * 0.000001) AS ?relativeTolerance)
                BIND (IF(?relativeTolerance >= 0.000001, ?relativeTolerance, 0.000001) AS ?tolerance)
                FILTER (
                    ?reportedAL < (?expectedAL - ?tolerance)
                    || ?reportedAL > (?expectedAL + ?tolerance)
                )
            }
        """
    ] .
