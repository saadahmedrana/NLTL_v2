<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-037-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:minCount 1 ;
        sh:message "The ship must identify the ice load patch used for the structural design."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:or (
                [ sh:class nltl:webFrame ]
                [ sh:class nltl:loadCarryingStringer ]
            ) ;
            sh:property [
                sh:path nltl:memberCapacity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Each web frame or load-carrying stringer must have one member capacity expressed in kN."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:combinedShearBendingEffect ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:loadPatchApplicationLocation ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Each web frame or load-carrying stringer must have a load case recording the combined bending-and-shear effect and the load-patch application location."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:node [
            sh:property [
                sh:path nltl:loadPatchApplicationLocation ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:memberCapacityMinimizationConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "The ice-load-patch design case must confirm that the selected application location minimises member capacity under combined bending and shear."
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The ice-load-patch design case must be connected to each applicable structural-member load case through the same load-patch application location." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER (?memberType IN (nltl:webFrame, nltl:loadCarryingStringer))
                ?loadCase nltl:combinedShearBendingEffect ?effect ;
                          nltl:loadPatchApplicationLocation ?loadLocation .
                $this nltl:hasIceLoadPatchDesignCase ?designCase .
                ?designCase nltl:loadPatchApplicationLocation ?designLocation ;
                            nltl:memberCapacityMinimizationConfirmed true .
                FILTER (STR(?loadLocation) = STR(?designLocation))
            }
        """
    ] .
</END_SHACL>
