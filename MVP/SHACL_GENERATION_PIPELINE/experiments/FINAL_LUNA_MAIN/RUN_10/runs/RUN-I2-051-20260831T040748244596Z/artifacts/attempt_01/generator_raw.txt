    ] ;
    sh:sparql [
        sh:message "For each positive-shear calculation case, the design vertical ice shear force must equal the longitudinal distribution factor multiplied by the bow force, with MegaNewton-to-kiloNewton conversion." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:hasCalculationCase ?case .

                ?fibQuantity <http://qudt.org/schema/qudt/numericValue> ?fib .
                ?fiQuantity <http://qudt.org/schema/qudt/numericValue> ?fi .
                ?cfQuantity <http://qudt.org/schema/qudt/numericValue> ?cf .
                ?luiQuantity <http://qudt.org/schema/qudt/numericValue> ?lui .
                ?case nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:shearForceDirection <https://w3id.org/nltl/vocab#positiveShearForce> .
                ?positionQuantity <http://qudt.org/schema/qudt/numericValue> ?position .

                BIND(?cf * ?fib * 1000 AS ?expected)
                FILTER (
                    (
                        ?position >= 0 &&
                        ?position <= (0.6 * ?lui) &&
                        ?cf != 0
                    ) ||
                    (
                        ?position >= (0.9 * ?lui) &&
                        ?position <= ?lui &&
                        ?cf != 1
                    ) ||
                    ABS(?fi - ?expected) >
                    (0.000001 * ABS(?expected) + 0.000001)
                )
            }
        """
    ] .
