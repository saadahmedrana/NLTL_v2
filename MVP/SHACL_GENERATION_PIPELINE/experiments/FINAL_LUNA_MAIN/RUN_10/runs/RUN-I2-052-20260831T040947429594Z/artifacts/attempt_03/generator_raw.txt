<urn:nltl:generated-shape:I2-052-CalculationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node <urn:nltl:generated-shape:I2-052-IntermediatePointShape>
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node <urn:nltl:generated-shape:I2-052-EndpointShape>
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node <urn:nltl:generated-shape:I2-052-EndpointShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The intermediate interpolation coordinate must lie strictly between the lower and upper endpoint coordinates in the same calculation case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasInterpolationPoint ?mid .
                $this nltl:interpolationLowerEndpoint ?lower .
                $this nltl:interpolationUpperEndpoint ?upper .

                ?mid nltl:interpolationCoordinate ?midCoordinate .
                ?lower nltl:interpolationCoordinate ?lowerCoordinate .
                ?upper nltl:interpolationCoordinate ?upperCoordinate .

                ?midCoordinate qudt:numericValue ?midValue .
                ?lowerCoordinate qudt:numericValue ?lowerValue .
                ?upperCoordinate qudt:numericValue ?upperValue .

                FILTER (
                    DATATYPE(?midValue) = xsd:decimal &&
                    DATATYPE(?lowerValue) = xsd:decimal &&
                    DATATYPE(?upperValue) = xsd:decimal &&
                    !(
                        ?lowerValue < ?midValue &&
                        ?midValue < ?upperValue
                    )
                )
            }
        """
    ] .
