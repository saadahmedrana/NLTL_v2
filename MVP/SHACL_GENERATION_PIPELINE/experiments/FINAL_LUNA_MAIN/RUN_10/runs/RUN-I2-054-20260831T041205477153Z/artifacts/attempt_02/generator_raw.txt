    sh:sparql [
        sh:message "The calculation-case and hull-girder bending-moment paths must identify the same case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?case .
                    $this nltl:hasHullGirderBendingMomentCalculationCase ?case .
                }
            }
        """
    ] .
