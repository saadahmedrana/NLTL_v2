<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        sh:message "A member in way of a cutout must not be unstable, and required stiffening must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout ;
                        nltl:memberInWayOfCutout true ;
                        nltl:stiffeningNecessary ?necessary .
                FILTER (
                    NOT EXISTS {
                        $this nltl:memberInstability false
                    }
                    || EXISTS {
                        $this nltl:memberInstability true
                    }
                    || (
                        ?necessary = true
                        && NOT EXISTS {
                            $this nltl:structuralStiffening ?stiffening
                        }
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMember ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A structural member linked to a cutout must provide the applicability and stiffening-necessity indicators." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasCutout ?cutout .
                FILTER (
                    NOT EXISTS {
                        $this nltl:memberInWayOfCutout ?inWay
                    }
                    || NOT EXISTS {
                        $this nltl:stiffeningNecessary ?necessary
                    }
                )
            }
        """
    ] .
<END_SHACL>
