<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A scoped calculation case must be linked from a ship and assess a shell-plating or local-frame hull structure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?ship rdf:type nltl:ship ;
                          nltl:hasCalculationCase $this .
                    $this nltl:calculationCaseAssessedHullStructure ?structure .
                    ?structure rdf:type ?scopeType .
                    FILTER (?scopeType IN (
                        nltl:plating,
                        nltl:structuralMember
                    ))
                }
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A scoped calculation case must use the prescribed analytical procedure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?structure .
                ?structure rdf:type ?scopeType .
                FILTER (?scopeType IN (
                    nltl:plating,
                    nltl:structuralMember
                ))
                FILTER NOT EXISTS {
                    $this nltl:calculationMethod
                        nltl:prescribedAnalyticalProcedure .
                }
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Direct calculation must not be used as an alternative to the prescribed analytical procedure on the same scoped calculation case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?structure .
                ?structure rdf:type ?scopeType .
                FILTER (?scopeType IN (
                    nltl:plating,
                    nltl:structuralMember
                ))
                $this nltl:calculationMethod
                    nltl:directCalculationMethodValue .
            }
        """
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The assessed hull structure must have a single thickness quantity with a decimal numeric value in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?structure .
                ?structure rdf:type ?scopeType .
                FILTER (?scopeType IN (
                    nltl:plating,
                    nltl:structuralMember
                ))
                FILTER (
                    NOT EXISTS {
                        ?structure nltl:thickness ?quantity .
                        ?quantity qudt:numericValue ?value ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?value) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?structure nltl:thickness ?quantity1, ?quantity2 .
                        FILTER (?quantity1 != ?quantity2)
                    }
                )
            }
        """
    ] .
<END_SHACL>
