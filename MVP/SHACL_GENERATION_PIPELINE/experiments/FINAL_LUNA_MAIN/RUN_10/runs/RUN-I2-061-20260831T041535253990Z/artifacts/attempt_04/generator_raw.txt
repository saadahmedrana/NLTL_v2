<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltl:GeneratedShape_I2_061_CalculationCase a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable calculation case shall use the prescribed analytical procedure, shall not use direct calculation as an alternative, and shall provide hull-structure thickness in metres as a decimal quantity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:calculationCaseAssessedHullStructure ?hull .
                ?hull rdf:type ?scopeType .
                FILTER (?scopeType IN (nltl:plating, nltl:structuralMember))
                FILTER (
                    NOT EXISTS {
                        $this nltl:calculationMethod nltl:prescribedAnalyticalProcedure
                    }
                    ||
                    EXISTS {
                        $this nltl:calculationMethod nltl:directCalculationMethodValue
                    }
                    ||
                    NOT EXISTS {
                        ?hull nltl:thickness ?quantity .
                        ?quantity qudt:numericValue ?numericValue .
                        ?quantity qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?numericValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .

nltl:GeneratedShape_I2_061_Ship a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node nltl:GeneratedShape_I2_061_CalculationCase
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable calculation case linked to the ship shall use the prescribed analytical procedure, shall not use direct calculation as an alternative, and shall provide hull-structure thickness in metres as a decimal quantity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationCaseAssessedHullStructure ?hull .
                ?hull rdf:type ?scopeType .
                FILTER (?scopeType IN (nltl:plating, nltl:structuralMember))
                FILTER (
                    NOT EXISTS {
                        ?case nltl:calculationMethod nltl:prescribedAnalyticalProcedure
                    }
                    ||
                    EXISTS {
                        ?case nltl:calculationMethod nltl:directCalculationMethodValue
                    }
                    ||
                    NOT EXISTS {
                        ?hull nltl:thickness ?quantity .
                        ?quantity qudt:numericValue ?numericValue .
                        ?quantity qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?numericValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
