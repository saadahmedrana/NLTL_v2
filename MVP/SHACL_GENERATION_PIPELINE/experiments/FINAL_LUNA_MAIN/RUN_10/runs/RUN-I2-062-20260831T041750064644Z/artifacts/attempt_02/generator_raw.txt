<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-062-DirectCalculationReadinessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each load-carrying stringer or web frame linked to the ship and identified as part of a grillage system, directCalculationUsed must be recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this rdf:type nltl:ship ;
                      nltl:hasStructuralMember ?member .

                ?member rdf:type ?memberType ;
                        nltl:partOfGrillageSystem ?grillage .

                ?grillage rdf:type nltl:grillageSystem .

                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:directCalculationUsed ?used .
                    FILTER (
                        datatype(?used) = xsd:boolean &&
                        ?used = "true"^^xsd:boolean
                    )
                }
            }
        """
    ] .
<END_SHACL>
