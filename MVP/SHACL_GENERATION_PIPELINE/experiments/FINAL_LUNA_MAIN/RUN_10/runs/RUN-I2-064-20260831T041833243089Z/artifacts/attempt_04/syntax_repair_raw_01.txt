            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

        sh:message "For a linear calculation, web shear stress must be less than yield strength divided by sqrt(3), and flange von Mises stress must be less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                $this nltl:yieldStrength ?yieldNode .
                ?yieldNode qudt:numericValue ?yield .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?shearNode .
                ?shearNode qudt:numericValue ?shear .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?flangeNode .
                ?flangeNode qudt:numericValue ?vonMises .
                FILTER (
                    ?shear >= (?yield / 1.7320508075688772)
                    ||
                    ?vonMises >= (1.15 * ?yield)
                )
            }
        """
