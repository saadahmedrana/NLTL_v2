<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-065-ComplexReadinessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "A ship calculation case using the nonlinear calculation method must provide the declared nonlinear-analysis evidence, all required margin and deformation inputs with the specified units and decimal datatypes, and a satisfied minor permanent deformation assessment." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

    FILTER NOT EXISTS {
        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
        ?evidence a nltl:evidenceArtifact .
    }
    FILTER NOT EXISTS {
        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
        FILTER NOT EXISTS {
            ?evidence a nltl:evidenceArtifact .
        }
    }

    FILTER NOT EXISTS {
        $this nltl:fractureMargin ?fractureMargin .
        ?fractureMargin a qudt:QuantityValue ;
            qudt:numericValue ?fractureNumber ;
            qudt:unit qudt:UNITLESS .
        FILTER(datatype(?fractureNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        $this nltl:fractureMargin ?fractureMargin .
        FILTER NOT EXISTS {
            ?fractureMargin a qudt:QuantityValue ;
                qudt:numericValue ?fractureNumber ;
                qudt:unit qudt:UNITLESS .
            FILTER(datatype(?fractureNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        $this nltl:majorBucklingMargin ?majorBucklingMargin .
        ?majorBucklingMargin a qudt:QuantityValue ;
            qudt:numericValue ?majorBucklingNumber ;
            qudt:unit qudt:UNITLESS .
        FILTER(datatype(?majorBucklingNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        $this nltl:majorBucklingMargin ?majorBucklingMargin .
        FILTER NOT EXISTS {
            ?majorBucklingMargin a qudt:QuantityValue ;
                qudt:numericValue ?majorBucklingNumber ;
                qudt:unit qudt:UNITLESS .
            FILTER(datatype(?majorBucklingNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        ?case nltl:permanentLateralDeformation ?lateralDeformation .
        ?lateralDeformation a qudt:QuantityValue ;
            qudt:numericValue ?lateralNumber ;
            qudt:unit qudt:MilliM .
        FILTER(datatype(?lateralNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        ?case nltl:permanentLateralDeformation ?lateralDeformation .
        FILTER NOT EXISTS {
            ?lateralDeformation a qudt:QuantityValue ;
                qudt:numericValue ?lateralNumber ;
                qudt:unit qudt:MilliM .
            FILTER(datatype(?lateralNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlaneDeformation .
        ?outOfPlaneDeformation a qudt:QuantityValue ;
            qudt:numericValue ?outOfPlaneNumber ;
            qudt:unit qudt:MilliM .
        FILTER(datatype(?outOfPlaneNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlaneDeformation .
        FILTER NOT EXISTS {
            ?outOfPlaneDeformation a qudt:QuantityValue ;
                qudt:numericValue ?outOfPlaneNumber ;
                qudt:unit qudt:MilliM .
            FILTER(datatype(?outOfPlaneNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        ?case nltl:relevantStructuralDimension ?structuralDimension .
        ?structuralDimension a qudt:QuantityValue ;
            qudt:numericValue ?dimensionNumber ;
            qudt:unit qudt:MilliM .
        FILTER(datatype(?dimensionNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        ?case nltl:relevantStructuralDimension ?structuralDimension .
        FILTER NOT EXISTS {
            ?structuralDimension a qudt:QuantityValue ;
                qudt:numericValue ?dimensionNumber ;
                qudt:unit qudt:MilliM .
            FILTER(datatype(?dimensionNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        $this nltl:yieldingMargin ?yieldingMargin .
        ?yieldingMargin a qudt:QuantityValue ;
            qudt:numericValue ?yieldingNumber ;
            qudt:unit qudt:UNITLESS .
        FILTER(datatype(?yieldingNumber) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        $this nltl:yieldingMargin ?yieldingMargin .
        FILTER NOT EXISTS {
            ?yieldingMargin a qudt:QuantityValue ;
                qudt:numericValue ?yieldingNumber ;
                qudt:unit qudt:UNITLESS .
            FILTER(datatype(?yieldingNumber) = xsd:decimal)
        }
    }

    FILTER NOT EXISTS {
        ?case nltl:minorPermanentDeformationAssessmentSatisfied ?assessment .
        FILTER(datatype(?assessment) = xsd:boolean && ?assessment = true)
    }
    FILTER NOT EXISTS {
        ?case nltl:minorPermanentDeformationAssessmentSatisfied ?assessment .
        FILTER(!(datatype(?assessment) = xsd:boolean && ?assessment = true))
    }
}
""" ;
    ] .
<END_SHACL>
