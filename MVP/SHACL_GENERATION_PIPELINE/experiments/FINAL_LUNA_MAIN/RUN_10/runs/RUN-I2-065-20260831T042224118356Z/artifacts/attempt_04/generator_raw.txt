<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node [
            sh:class nltl:calculationCase
        ]
    ] .

gen:I2_065_CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:class nltl:calculationMethodValue ;
        sh:minCount 1 ;
        sh:message "Each represented calculation case must identify the calculation method used to select applicability."
    ] ;
    sh:sparql [
        sh:message "A nonlinear calculation case must provide nonlinear-analysis acceptance evidence, the required ship-owned margins, the required case-owned deformation and dimension values in millimetres, and a true minor permanent deformation assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this .
                $this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                OPTIONAL {
                    $this nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }

                OPTIONAL {
                    ?ship nltl:fractureMargin ?fractureMargin .
                    ?fractureMargin qudt:numericValue ?fractureNumeric ;
                                    qudt:unit qudt:UNITLESS .
                }

                OPTIONAL {
                    ?ship nltl:majorBucklingMargin ?majorBucklingMargin .
                    ?majorBucklingMargin qudt:numericValue ?majorBucklingNumeric ;
                                         qudt:unit qudt:UNITLESS .
                }

                OPTIONAL {
                    $this nltl:permanentLateralDeformation ?lateralDeformation .
                    ?lateralDeformation qudt:numericValue ?lateralNumeric ;
                                        qudt:unit qudt:MilliM .
                }

                OPTIONAL {
                    $this nltl:permanentOutOfPlaneDeformation ?outOfPlaneDeformation .
                    ?outOfPlaneDeformation qudt:numericValue ?outOfPlaneNumeric ;
                                           qudt:unit qudt:MilliM .
                }

                OPTIONAL {
                    $this nltl:relevantStructuralDimension ?structuralDimension .
                    ?structuralDimension qudt:numericValue ?dimensionNumeric ;
                                         qudt:unit qudt:MilliM .
                }

                OPTIONAL {
                    ?ship nltl:yieldingMargin ?yieldingMargin .
                    ?yieldingMargin qudt:numericValue ?yieldingNumeric ;
                                    qudt:unit qudt:UNITLESS .
                }

                OPTIONAL {
                    $this nltl:minorPermanentDeformationAssessmentSatisfied ?assessment .
                }

                FILTER (
                    !BOUND(?evidence) ||
                    !BOUND(?fractureNumeric) ||
                    datatype(?fractureNumeric) != xsd:decimal ||
                    !BOUND(?fractureMargin) ||
                    !BOUND(?majorBucklingNumeric) ||
                    datatype(?majorBucklingNumeric) != xsd:decimal ||
                    !BOUND(?majorBucklingMargin) ||
                    !BOUND(?lateralNumeric) ||
                    datatype(?lateralNumeric) != xsd:decimal ||
                    !BOUND(?lateralDeformation) ||
                    !BOUND(?outOfPlaneNumeric) ||
                    datatype(?outOfPlaneNumeric) != xsd:decimal ||
                    !BOUND(?outOfPlaneDeformation) ||
                    !BOUND(?dimensionNumeric) ||
                    datatype(?dimensionNumeric) != xsd:decimal ||
                    !BOUND(?structuralDimension) ||
                    !BOUND(?yieldingNumeric) ||
                    datatype(?yieldingNumeric) != xsd:decimal ||
                    !BOUND(?yieldingMargin) ||
                    !BOUND(?assessment) ||
                    datatype(?assessment) != xsd:boolean ||
                    ?assessment != true
                )
            }
        """
    ] .
<END_SHACL>
