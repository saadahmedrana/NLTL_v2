<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-066> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasWeld ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        sh:message "Each linked weld must record whether it is within an ice-strengthened area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasWeld ?weld .
                FILTER NOT EXISTS {
                    ?weld nltl:withinIceStrengthenedArea ?scope .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Welding within an ice-strengthened area must be of the double continuous type." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasWeld ?weld .
                ?weld nltl:withinIceStrengthenedArea ?scope .
                FILTER ( ?scope = true || ?scope = "true"^^xsd:boolean )
                FILTER NOT EXISTS {
                    ?weld nltl:weldType nltl:doubleContinuousWeld .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-066-weld-scope> a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasWeld ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] .

<END_SHACL>
