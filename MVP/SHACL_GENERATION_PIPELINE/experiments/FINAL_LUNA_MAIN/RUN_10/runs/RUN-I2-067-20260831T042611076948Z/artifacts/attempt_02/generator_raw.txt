<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_067_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:strengthContinuityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that continuity of strength is ensured."
    ] .

gen:I2_067_StructuralConnectionShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:structuralConnection ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Continuity of strength must be ensured at every structural connection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                ?ship nltl:structuralConnection $this .
                FILTER NOT EXISTS {
                    ?ship nltl:strengthContinuityStatus ?status .
                    FILTER (?status = true)
                }
            }
        """
    ] .
<END_SHACL>
