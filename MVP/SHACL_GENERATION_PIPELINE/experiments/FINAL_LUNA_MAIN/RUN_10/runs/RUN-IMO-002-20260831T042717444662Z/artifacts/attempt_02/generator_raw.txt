<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_002_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its Polar Code category."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for at least thin first-year ice with possible old-ice inclusions and not in category A must be category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions .
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category != nltl:polarShipCategoryA &&
                    ?category != nltl:polarShipCategoryB
                )
            }
        """
    ] .
<END_SHACL>
