<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship has sea ice origin, no more than one winter of growth, and ice thickness from 0.3 m through 2.0 m, its ice type classification must be first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin .
    $this nltl:winterGrowthCount ?growthCount .
    $this nltl:iceThickness ?thicknessNode .
    ?thicknessNode qudt:numericValue ?thickness .
    ?thicknessNode qudt:unit <http://qudt.org/vocab/unit/M> .
    FILTER(
        datatype(?growthCount) = xsd:integer &&
        ?growthCount <= 1 &&
        datatype(?thickness) = xsd:decimal &&
        ?thickness >= 0.3 &&
        ?thickness <= 2.0
    )
    FILTER(
        NOT EXISTS {
            $this nltl:iceTypeClassification nltl:firstYearIce .
        }
        ||
        EXISTS {
            $this nltl:iceTypeClassification ?classification .
            FILTER(?classification != nltl:firstYearIce)
        }
    )
}
"""
    ] .
<END_SHACL>
