<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-005-IceCondition>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An ice condition of 'ice free waters' requires totalIcePresent to be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "ice free waters")
                FILTER NOT EXISTS {
                    $this nltl:totalIcePresent ?present .
                    FILTER (?present = false)
                }
            }
        """
    ] .
<END_SHACL>
