                $this nltl:iceCondition ?condition ;
                      nltl:landOriginIceConcentration ?quantity .
                ?quantity qudt:numericValue ?value ;
                          qudt:unit ?quantityUnit .
                FILTER (STR(?condition) = "bergy waters")

                    datatype(?value) != xsd:decimal ||

            }
