<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:polarShipCertificateForm
        ] ;
        sh:message "A ship shall have a Polar Ship Certificate form."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:node gen:IMO017SolasScheduleShape ;
        sh:message "A ship shall have a SOLAS certificate schedule with certificate date records."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:node gen:IMO017SupplementShape ;
        sh:message "A ship shall have a Polar Certificate supplement with required equipment records."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement shall be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates shall be provided."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates shall be provided."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates shall be provided."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate validity dates, survey dates, and endorsements shall match the corresponding SOLAS certificate schedule dates." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                {
                    $this nltl:polarCertificateValidityDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:polarCertificateValidityDates ?date .
                    }
                }
                UNION
                {
                    $this nltl:surveyDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:surveyDates ?date .
                    }
                }
                UNION
                {
                    $this nltl:endorsementDates ?date .
                    FILTER NOT EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }
                UNION
                {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER NOT EXISTS {
                        $this nltl:endorsementDates ?date .
                    }
                }
            }
        """
    ] .

gen:IMO017SolasScheduleShape
    a sh:NodeShape ;
    sh:class nltl:solasCertificateSchedule ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO017DateRecordShape ;
        sh:message "The SOLAS certificate schedule shall contain certificate schedule date records."
    ] .

gen:IMO017DateRecordShape
    a sh:NodeShape ;
    sh:class nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each certificate schedule date record shall contain one xsd:date value."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall have a validity, survey, or endorsement category."
    ] .

gen:IMO017SupplementShape
    a sh:NodeShape ;
    sh:class nltl:polarCertificateSupplement ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:requiredEquipmentRecords
        ] ;
        sh:message "The Polar Certificate supplement shall contain required equipment records."
    ] .
<END_SHACL>
