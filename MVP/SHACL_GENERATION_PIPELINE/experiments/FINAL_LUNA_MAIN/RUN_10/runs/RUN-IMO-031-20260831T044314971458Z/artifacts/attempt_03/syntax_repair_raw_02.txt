<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_031_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:approvalRecord ;
            sh:class nltl:benchmarkEntity ;
            sh:property [
                sh:path nltl:exposedStructureMaterial ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each approval record must identify an exposed-structure material."
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class nltl:evidenceState ;
                    sh:hasValue nltl:evidenceStateApproved
                ] ;
                sh:message "Each exposed-structure material must have approved approval evidence."
            ] ;
            sh:property [
                sh:path nltl:approvingAuthority ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    "Administration"
                    "recognized organization accepted by the Administration"
                ) ;
                sh:message "The approving authority must be the Administration or a recognized organization accepted by it."
            ] ;
            sh:property [
                sh:path nltl:approvalStandard ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    "standard acceptable to the Organization"
                    "another standard offering an equivalent level of safety"
                ) ;
                sh:message "The approval standard must be acceptable to the Organization or offer an equivalent level of safety."
            ] ;
            sh:property [
                sh:path nltl:polarServiceTemperature ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG_C ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Each approval record must state the polar service temperature basis in degrees Celsius."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each approval record's polar service temperature basis must match the ship's polar service temperature." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:polarServiceTemperature ?shipTemperature .
                $this nltl:hasEvidence ?approvalRecord .
                ?approvalRecord nltl:polarServiceTemperature ?recordTemperature .
                ?shipTemperature qudt:numericValue ?shipValue .
                ?recordTemperature qudt:numericValue ?recordValue .
                FILTER (?shipValue != ?recordValue)
            }
        """
    ] .
</END_SHACL>
