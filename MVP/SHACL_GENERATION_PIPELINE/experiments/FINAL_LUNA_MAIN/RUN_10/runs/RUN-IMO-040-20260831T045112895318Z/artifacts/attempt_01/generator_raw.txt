<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_040_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_040_ApplicableBranch
        gen:IMO_040_NotApplicableBranch
    ) .

gen:IMO_040_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "iceAccretionLikely must be true for the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:hasRelevantHatchOrDoor ;
        sh:node [
            sh:property [
                sh:path nltl:hasIceOrSnowRemovalOrPreventionMeans ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:message "Each relevant hatch or door must have associated means to remove or prevent ice and snow accretion."
            ]
        ]
    ] .

gen:IMO_040_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "iceAccretionLikely must be false for the non-applicable branch."
    ] .
