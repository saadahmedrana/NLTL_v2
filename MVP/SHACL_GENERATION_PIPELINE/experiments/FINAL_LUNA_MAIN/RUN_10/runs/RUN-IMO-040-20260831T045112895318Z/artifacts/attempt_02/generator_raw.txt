<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-040-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice accretion is likely, every relevant hatch or door must have an associated ice or snow removal or prevention means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceAccretionLikely true .
                $this nltl:hasRelevantHatchOrDoor ?hatch .
                FILTER NOT EXISTS {
                    ?hatch nltl:hasIceOrSnowRemovalOrPreventionMeans ?means .
                    ?means a nltl:iceSnowRemovalOrPreventionMeans .
                }
            }
        """ ;
    ] .
</urn:nltl:generated-shape:IMO-040-ShipShape>

<urn:nltl:generated-shape:IMO-040-HatchOrDoorMeansShape>
    a sh:NodeShape ;
    sh:targetClass nltl:relevantHatchOrDoorItem ;
    sh:property [
        sh:path nltl:hasIceOrSnowRemovalOrPreventionMeans ;
        sh:class nltl:iceSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:message "A relevant hatch or door must have an associated ice or snow removal or prevention means."
    ] .
</urn:nltl:generated-shape:IMO-040-HatchOrDoorMeansShape>
<END_SHACL>
