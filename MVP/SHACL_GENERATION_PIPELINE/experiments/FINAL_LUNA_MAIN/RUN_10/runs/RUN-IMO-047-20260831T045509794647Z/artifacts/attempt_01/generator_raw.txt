<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A temperature quantity must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A temperature quantity must use degrees Celsius."
    ] .

gen:IMO047Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:minCount 1 ;
        sh:message "The ship must identify its essential engine or engines for this requirement."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:node gen:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The combustion-air temperature must be supplied in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:node gen:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:node gen:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combustion-air temperature must be within the manufacturer's minimum and maximum air-temperature criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureNode ;
                      nltl:manufacturerMinAirTemperature ?minimumNode ;
                      nltl:manufacturerMaxAirTemperature ?maximumNode .

                ?temperatureNode qudt:numericValue ?temperature .
                ?minimumNode qudt:numericValue ?minimum .
                ?maximumNode qudt:numericValue ?maximum .

                FILTER (?temperature < ?minimum || ?temperature > ?maximum)
            }
        """
    ] .
<END_SHACL>
