<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:IMO049ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:machineryComponent
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable Category A or B machinery component must have an approved approval record with an authorized authority and an acceptable or equivalent-safety standard." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:hasComponent ?component .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )
                ?component a nltl:machineryComponent ;
                           nltl:machineryComponentType ?componentType .
                FILTER (
                    DATATYPE(?componentType) = xsd:string &&
                    STR(?componentType) IN (
                        "propeller blade",
                        "propulsion line",
                        "steering equipment",
                        "appendage"
                    )
                )
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard ?standard .
                    FILTER (
                        DATATYPE(?authority) = xsd:string &&
                        STR(?authority) IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                    FILTER (
                        DATATYPE(?standard) = xsd:string &&
                        STR(?standard) IN (
                            "standard acceptable to the Organization",
                            "another standard offering an equivalent level of safety"
                        )
                    )
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable ice-strengthened Category C machinery component must have approved ice-context evidence, an authorized authority, and a standard adequate for the operating ice type and concentration." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:shipCategory nltl:polarShipCategoryC ;
                      nltl:shipIceStrengthened true ;
                      nltl:hasComponent ?component .
                ?component a nltl:machineryComponent ;
                           nltl:machineryComponentType ?componentType .
                FILTER (
                    DATATYPE(?componentType) = xsd:string &&
                    STR(?componentType) IN (
                        "propeller blade",
                        "propulsion line",
                        "steering equipment",
                        "appendage"
                    )
                )
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard ?standard ;
                            nltl:operatingIceType ?iceType ;
                            nltl:operatingIceConcentration ?concentration .
                    FILTER (
                        DATATYPE(?authority) = xsd:string &&
                        STR(?authority) IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                    FILTER (
                        DATATYPE(?standard) = xsd:string &&
                        STR(?standard) = "acceptable standard adequate for operating ice type and concentration"
                    )
                    FILTER (DATATYPE(?iceType) = xsd:string)
                    ?concentration qudt:numericValue ?numericValue ;
                                   qudt:unit unit:UNITLESS .
                    FILTER (DATATYPE(?numericValue) = xsd:decimal)
                }
            }
        """ ;
    ] .

gen:IMO049MachineryComponentShape
    a sh:NodeShape ;
    sh:targetClass nltl:machineryComponent ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:class nltl:approvalRecord
    ] .
<END_SHACL>
