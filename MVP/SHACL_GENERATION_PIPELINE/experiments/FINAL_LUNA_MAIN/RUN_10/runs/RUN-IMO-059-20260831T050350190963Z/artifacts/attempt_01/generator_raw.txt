<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_059_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFirefighterOutfit ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:firefighterOutfitItem ;
        sh:node gen:IMO_059_FirefighterOutfitShape ;
        sh:message "Each firefighter outfit must be stored in a warm location."
    ] .

gen:IMO_059_FirefighterOutfitShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStorageLocation ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:physicalStorageLocation ;
        sh:node gen:IMO_059_StorageLocationShape ;
        sh:message "Each firefighter outfit must have a storage location."
    ] .

gen:IMO_059_StorageLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:storageTemperatureClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:storageTemperatureClassValue ;
        sh:hasValue nltl:warmStorageLocation ;
        sh:message "The storage location must be classified as warm."
    ] .
<END_SHACL>
