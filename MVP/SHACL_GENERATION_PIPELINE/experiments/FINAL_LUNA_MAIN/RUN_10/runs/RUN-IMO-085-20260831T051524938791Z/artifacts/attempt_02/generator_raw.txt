<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-085-BridgeWingProtection> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Category A or B ships constructed on or after 2017-01-01, bridge wings must be enclosed or the bridge-wing protection design status must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )

                $this nltl:constructionDate ?constructionDate .
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (
                    YEAR(?constructionDate) > 2017
                    ||
                    (
                        YEAR(?constructionDate) = 2017
                        && MONTH(?constructionDate) >= 1
                        && DAY(?constructionDate) >= 1
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:bridgeWingsEnclosed true
                }
                FILTER NOT EXISTS {
                    $this nltl:bridgeWingProtectionDesignStatus true
                }
            }
        """
    ] .
<END_SHACL>
