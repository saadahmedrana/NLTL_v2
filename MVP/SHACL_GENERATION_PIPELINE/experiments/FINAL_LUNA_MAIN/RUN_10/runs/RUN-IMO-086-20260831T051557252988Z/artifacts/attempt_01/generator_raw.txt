<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO086ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gen:IMO086HeadingMeansShape ;
        sh:message "The ship must have at least two non-magnetic heading means, each with the required power connections and independence relationship."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means must be independent from another distinct non-magnetic heading means linked to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    FILTER ( ?otherMeans != ?means )
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                }
            }
        """
    ] .

gen:IMO086HeadingMeansShape
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:nodeKind sh:IRI ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means must be connected to the ship's main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means must be connected to the ship's emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:message "Each non-magnetic heading means must declare an independent heading means."
    ] .
<END_SHACL>
