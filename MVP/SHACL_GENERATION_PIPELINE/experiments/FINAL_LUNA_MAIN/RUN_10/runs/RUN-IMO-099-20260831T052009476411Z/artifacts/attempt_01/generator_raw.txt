<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qudt-unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_099_QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must contain exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt-unit:DAY ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The duration quantity must use the day unit."
    ] .

gen:IMO_099_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that a battery-life management procedure is present."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that the procedure is implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:node gen:IMO_099_QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide communication-device availability in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:node gen:IMO_099_QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the maximum expected rescue time in days."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Communication-device availability must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX qudt-unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availability .
                ?availability qudt:numericValue ?availableValue ;
                              qudt:unit qudt-unit:DAY .
                $this nltl:maximumExpectedRescueTime ?rescueTime .
                ?rescueTime qudt:numericValue ?rescueValue ;
                            qudt:unit qudt-unit:DAY .
                FILTER (?availableValue < ?rescueValue)
            }
        """
    ] .
<END_SHACL>
