<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_110_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single valid construction date is required."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single deadweight quantity in metric tonnes is required."
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The entire cargo tank length double-bottom protection status must be recorded as a single boolean."
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The wing-tank protection status must be recorded as a single boolean."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 tonnes, both required cargo-tank protection statuses must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))

                $this rdf:type nltl:oilTanker .

                $this nltl:constructionDate ?constructionDate .
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                $this nltl:deadweight ?deadweight .
                ?deadweight qudt:numericValue ?deadweightValue ;
                            qudt:unit unit:TON_Metric .
                FILTER (?deadweightValue < 5000)

                $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?doubleBottomStatus .
                $this nltl:wingTankProtectionStatus ?wingTankStatus .
                FILTER (
                    ?doubleBottomStatus != true ||
                    ?wingTankStatus != true
                )
            }
        """
    ] .
<END_SHACL>
