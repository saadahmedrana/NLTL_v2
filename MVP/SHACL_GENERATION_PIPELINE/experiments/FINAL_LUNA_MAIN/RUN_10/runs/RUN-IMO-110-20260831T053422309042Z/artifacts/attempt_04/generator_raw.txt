<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship type must be recorded as a single string value."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:message "The construction date must be on or after 1 January 2017."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxExclusive "5000"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship deadweight must be a single metric-tonne quantity below 5,000 tonnes."
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The entire cargo tank length must have compliant double-bottom protection."
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The wing tanks or spaces must have compliant protection."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B oil tanker of less than 5,000 tonnes deadweight constructed on or after 1 January 2017, both required cargo-tank protection statuses must be exactly true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?constructionDate ;
                      rdf:type nltl:oilTanker .

                ?deadweight qudt:numericValue ?mass ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?shipType = "oil_tanker"
                    && ?mass < 5000
                    && ?constructionDate >= "2017-01-01"^^xsd:date
                )

                FILTER NOT EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true .
                }

                FILTER NOT EXISTS {
                    $this nltl:wingTankProtectionStatus true .
                }
            }
        """
    ] .
<END_SHACL>
