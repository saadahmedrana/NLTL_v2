<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one Category A or Category B ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date on or after 2017-01-01."
    ] ;
    sh:sparql [
        sh:message "Each oil residue tank or oily bilge water holding tank must have an individual capacity with a decimal value in cubic metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:hasComponent ?tank .
                ?tank a ?tankClass .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER (?tankClass IN (
                    nltl:oilResidueTank,
                    nltl:oilyBilgeWaterHoldingTank
                ))
                FILTER NOT EXISTS {
                    ?tank nltl:individualTankCapacity ?capacityNode .
                    ?capacityNode qudt:numericValue ?capacityValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/M3> .
                    FILTER (datatype(?capacityValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every applicable tank with an individual capacity greater than 30 m3 must be separated from the outer shell by at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:hasComponent ?tank .
                ?tank a ?tankClass ;
                      nltl:individualTankCapacity ?capacityNode .
                ?capacityNode qudt:numericValue ?capacityValue ;
                              qudt:unit <http://qudt.org/vocab/unit/M3> .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER (?tankClass IN (
                    nltl:oilResidueTank,
                    nltl:oilyBilgeWaterHoldingTank
                ))
                FILTER (datatype(?capacityValue) = xsd:decimal)
                FILTER (?capacityValue > 30)
                FILTER NOT EXISTS {
                    ?tank nltl:tankOuterShellSeparation ?separationNode .
                    ?separationNode qudt:numericValue ?separationValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?separationValue) = xsd:decimal)
                    FILTER (?separationValue >= 0.76)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-111-OilResidueTankShape>
    a sh:NodeShape ;
    sh:targetClass nltl:oilResidueTank ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M3> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

<urn:nltl:generated-shape:IMO-111-OilyBilgeWaterHoldingTankShape>
    a sh:NodeShape ;
    sh:targetClass nltl:oilyBilgeWaterHoldingTank ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M3> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
