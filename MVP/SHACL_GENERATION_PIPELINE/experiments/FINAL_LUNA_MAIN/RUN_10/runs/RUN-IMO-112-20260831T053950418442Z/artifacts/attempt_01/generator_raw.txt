<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_112_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine whether the Arctic-waters discharge prohibition applies."
    ] ;
    sh:sparql [
        sh:message "In Arctic waters, discharge of noxious liquid substances or mixtures containing them must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters")
                FILTER (
                    NOT EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea false
                    }
                    ||
                    EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                        FILTER (?discharge != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
