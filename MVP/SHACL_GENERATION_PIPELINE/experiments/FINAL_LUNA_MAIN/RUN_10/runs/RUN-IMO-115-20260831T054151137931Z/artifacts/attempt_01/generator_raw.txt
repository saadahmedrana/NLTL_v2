<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-115> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage is identified as comminuted and disinfected, the MARPOL compliance status, strict distance threshold, ice-concentration condition, and related distance record must be present and conforming." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    EXISTS {
                        $this nltl:sewageDischargeType "comminuted_and_disinfected"
                    }
                    ||
                    EXISTS {
                        $this nltl:comminutedAndDisinfected true
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:sewageDischargeType ?dischargeType
                        FILTER (
                            datatype(?dischargeType) = xsd:string
                            && ?dischargeType = "comminuted_and_disinfected"
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:comminutedAndDisinfected true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestIceShelfOrFastIce ?nearestDistance
                        ?nearestDistance a qudt:QuantityValue ;
                            qudt:numericValue ?nearestValue ;
                            qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal
                            && ?nearestValue > "3"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearbyConcentration
                        ?nearbyConcentration a qudt:QuantityValue ;
                            qudt:numericValue ?nearbyValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?nearbyValue) = xsd:decimal
                            && ?nearbyValue > "1"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance
                        ?areaDistance a qudt:QuantityValue ;
                            qudt:numericValue ?areaDistanceValue ;
                            qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?areaDistanceValue) = xsd:decimal
                        )
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
