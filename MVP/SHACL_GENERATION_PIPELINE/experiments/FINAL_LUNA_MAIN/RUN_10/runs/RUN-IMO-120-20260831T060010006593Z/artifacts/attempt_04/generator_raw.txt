    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters"^^xsd:string)
            }
        """ ;
        sh:message "The non-applicable branch requires an operating area other than Arctic_waters."
    ] .
