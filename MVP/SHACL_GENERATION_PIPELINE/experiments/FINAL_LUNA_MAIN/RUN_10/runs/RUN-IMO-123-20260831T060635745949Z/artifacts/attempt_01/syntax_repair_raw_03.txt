<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO123_DischargeRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:distanceToAreaWithIceConcentrationGreaterThanOneTenth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MI_N>
            ]
        ] ;
        sh:message "Each discharge-distance record must record the distance from an area with ice concentration greater than one tenth in nautical miles."
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 12
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MI_N>
            ]
        ] ;
        sh:message "The distance to the nearest land, ice shelf, or fast ice must be at least 12 nautical miles."
    ] .

gen:IMO123_CargoResidueDischargeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDischargeDistanceRecord ;
        sh:node gen:IMO123_DischargeRecordShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Cargo-residue discharge records are permitted only when the residue, voyage, reception-facility, Arctic-route, and ice-concentration conditions are satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasDischargeDistanceRecord ?record .
                FILTER NOT EXISTS {
                    $this nltl:residueNotRecoverableByCommonMethods true ;
                          nltl:shipEnRoute true ;
                          nltl:harmfulSubstancePresent false ;
                          nltl:departureAndDestinationWithinArctic true ;
                          nltl:routeRemainsWithinArctic true ;
                          nltl:adequateReceptionFacilitiesAvailable false ;
                          nltl:nearbyIceConcentration ?concentrationNode .
                    ?concentrationNode qudt:numericValue ?concentration ;
                                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (
                        DATATYPE(?concentration) = xsd:decimal &&
                        ?concentration > 0.1
                    )
                }
            }
        """
    ] .
</END_SHACL>
