<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-005-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:headingOrPositionSystem ;
        sh:datatype xsd:string ;
        sh:message "Each heading-reference or position-fixing system must be represented as a string."
    ] ;
    sh:property [
        sh:path nltl:intendedArea ;
        sh:datatype xsd:string ;
        sh:message "Each represented intended area must be a string."
    ] ;
    sh:property [
        sh:path nltl:intendedAreaSuitabilityApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Any represented intended-area suitability approval status must be Approved."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supplied heading-reference or position-fixing system must have an intended area and an Approved suitability status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:ship ;
                      nltl:headingOrPositionSystem ?system .
                FILTER NOT EXISTS {
                    $this nltl:intendedArea ?area ;
                          nltl:intendedAreaSuitabilityApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
