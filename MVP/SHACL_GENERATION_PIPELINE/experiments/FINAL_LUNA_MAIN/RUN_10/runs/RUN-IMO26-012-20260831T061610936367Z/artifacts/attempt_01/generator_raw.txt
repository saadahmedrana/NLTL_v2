<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node <urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    ] .

<urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall identify an independent non-magnetic heading means."
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means shall be independent from another distinct heading means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?ship nltl:hasNonMagneticHeadingMeans $this .
                    $this nltl:independentFromHeadingMeans ?other .
                    ?ship nltl:hasNonMagneticHeadingMeans ?other .
                    FILTER (?other != $this)
                }
            }
        """
    ] .
<END_SHACL>
