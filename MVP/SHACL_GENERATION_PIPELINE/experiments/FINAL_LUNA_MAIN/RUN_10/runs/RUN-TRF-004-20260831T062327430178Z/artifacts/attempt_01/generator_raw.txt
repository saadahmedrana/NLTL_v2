<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2008 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        sh:message "For a ship contracted for construction in 2010 or 2011, the applicable Ice Class Regulation edition must be the 2008 edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (datatype(?date) = xsd:date)
                FILTER (
                    YEAR(?date) = 2010 ||
                    YEAR(?date) = 2011
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2008 .
                }
            }
        """
    ] .
<END_SHACL>
