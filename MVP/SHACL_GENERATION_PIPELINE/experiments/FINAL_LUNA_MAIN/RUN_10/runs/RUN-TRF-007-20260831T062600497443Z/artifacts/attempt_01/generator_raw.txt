<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-007-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of TRF-007."
    ] ;
    sh:sparql [
        sh:message "For an IA Super or IA ship whose construction stage was before 1 September 2003, construction stage date, delivery date, assessment date, and a permitted 2017 compliance method are required; the assessment date must not be later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:constructionStageDate ?constructionDate .
                        FILTER (DATATYPE(?constructionDate) = xsd:date)
                    }
                    ||
                    (
                        EXISTS {
                            $this nltl:constructionStageDate ?constructionDate .
                            FILTER (
                                DATATYPE(?constructionDate) = xsd:date &&
                                (
                                    YEAR(?constructionDate) < 2003 ||
                                    (
                                        YEAR(?constructionDate) = 2003 &&
                                        (
                                            MONTH(?constructionDate) < 9 ||
                                            (
                                                MONTH(?constructionDate) = 9 &&
                                                DAY(?constructionDate) < 1
                                            )
                                        )
                                    )
                                )
                            )
                        }
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:deliveryDate ?deliveryDate .
                                FILTER (DATATYPE(?deliveryDate) = xsd:date)
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:assessmentDate ?assessmentDate .
                                FILTER (DATATYPE(?assessmentDate) = xsd:date)
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:engineOutputComplianceMethod ?method .
                                FILTER (
                                    ?method = nltl:traficom2017Section3Point2Point2Method ||
                                    ?method = nltl:traficom2017Section3Point2Point4Method
                                )
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:deliveryDate ?deliveryDate ;
                                      nltl:assessmentDate ?assessmentDate .
                                FILTER (
                                    DATATYPE(?deliveryDate) = xsd:date &&
                                    DATATYPE(?assessmentDate) = xsd:date &&
                                    (
                                        YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                                        (
                                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                            MONTH(?assessmentDate) = 1 &&
                                            DAY(?assessmentDate) <= 1
                                        )
                                    )
                                )
                            }
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
