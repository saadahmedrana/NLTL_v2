<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:maxCount 1 ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:constructionStageDate ;
                        sh:datatype xsd:date ;
                        sh:minCount 1 ;
                        sh:maxExclusive "2003-09-01"^^xsd:date
                    ] ;
                    sh:property [
                        sh:path nltl:deliveryDate ;
                        sh:datatype xsd:date ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:assessmentDate ;
                        sh:datatype xsd:date ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:engineOutputComplianceMethod ;
                        sh:nodeKind sh:IRI ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:in (
                            nltl:traficom2017Section3Point2Point2Method
                            nltl:traficom2017Section3Point2Point4Method
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:constructionStageDate ;
                        sh:datatype xsd:date ;
                        sh:minCount 1 ;
                        sh:maxInclusive "2003-09-01"^^xsd:date
                    ]
                ]
            )
        ]
        [
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "A ship with ice class IA or IA Super must satisfy the applicable TRF-007 date and compliance conditions." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?ice .
                        FILTER (
                            ?ice IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA or IA Super ship constructed before 1 September 2003, the 2017 Ice Class Regulations requirement is overdue." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:iceClass ?ice ;
                      nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate ;
                      nltl:engineOutputComplianceMethod ?method .
                FILTER (
                    ?ice IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    datatype(?deliveryDate) = xsd:date &&
                    datatype(?assessmentDate) = xsd:date
                )
                FILTER (
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        MONTH(?constructionDate) < 9
                    )
                )
                FILTER (
                    ?method IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        (
                            MONTH(?assessmentDate) > 1 ||
                            (
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
