<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-009> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IA or IA Super must provide a valid construction stage date to establish applicability of the rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate .
                    FILTER (datatype(?constructionDate) = xsd:date)
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IA or IA Super constructed before 1 September 2003 must have a permitted 2017 compliance method and an assessment date no later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .

                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIaSuper
                ))
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        MONTH(?constructionDate) < 9
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate ;
                          nltl:assessmentDate ?assessmentDate ;
                          nltl:engineOutputComplianceMethod ?method .

                    FILTER (
                        datatype(?deliveryDate) = xsd:date &&
                        datatype(?assessmentDate) = xsd:date &&
                        ?method IN (
                            nltl:traficom2017Section3Point2Point2Method,
                            nltl:traficom2017Section3Point2Point4Method
                        ) &&
                        (
                            YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) <= 1
                            )
                        )
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
