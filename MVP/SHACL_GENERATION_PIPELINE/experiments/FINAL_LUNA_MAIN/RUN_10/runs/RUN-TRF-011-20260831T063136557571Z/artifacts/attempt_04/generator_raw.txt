<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "At each longitudinal position, the UIWL vertical coordinate must equal the maximum intended ice-operating waterline coordinate within a tolerance of 1e-6 relative to the expected maximum, with a 1e-9 absolute fallback; both waterlines must have corresponding points." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                {
                    SELECT ?this ?x ?expected ?actualMin ?actualMax ?uiCount ?intCount
                    WHERE {
                        {
                            SELECT ?this ?x
                                   (MAX(?iv) AS ?expected)
                                   (MIN(?uv) AS ?actualMin)
                                   (MAX(?uv) AS ?actualMax)
                                   (COUNT(?up) AS ?uiCount)
                                   (COUNT(?ip) AS ?intCount)
                            WHERE {
                                {
                                    ?this nltl:hasUpperIceWaterline ?uiwl .
                                    ?uiwl nltl:hasWaterlineProfilePoint ?up .
                                    ?up nltl:longitudinalPosition ?ux ;
                                        nltl:verticalCoordinate ?uq .
                                    ?uq qudt:numericValue ?uv .
                                    BIND(?ux AS ?x)
                                }
                                UNION
                                {
                                    ?this nltl:hasIntendedIceOperatingWaterline ?iwl .
                                    ?iwl nltl:hasWaterlineProfilePoint ?ip .
                                    ?ip nltl:longitudinalPosition ?ix ;
                                        nltl:verticalCoordinate ?iq .
                                    ?iq qudt:numericValue ?iv .
                                    BIND(?ix AS ?x)
                                }
                            }
                            GROUP BY ?this ?x
                        }
                    }
                    GROUP BY ?this ?x ?expected ?actualMin ?actualMax ?uiCount ?intCount
                }
                FILTER(
                    ?uiCount = 0 ||
                    ?intCount = 0 ||
                    !BOUND(?expected) ||
                    !BOUND(?actualMin) ||
                    !BOUND(?actualMax) ||
                    ?actualMin < (?expected - (ABS(?expected) * 0.000001 + 0.000000001)) ||
                    ?actualMax > (?expected + (ABS(?expected) * 0.000001 + 0.000000001))
                )
            }
        """
    ] .
