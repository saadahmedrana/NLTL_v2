<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF013ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF013IceStrengthenedCase
        gen:TRF013NotIceStrengthenedCase
    ) .

gen:TRF013NotIceStrengthenedCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF013IceStrengthenedCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node gen:TRF013LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an ice-strengthened ship, each operating draught must be between its corresponding minimum and maximum ice-class draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingForeDraught ?operatingForeQuantity .
                ?operatingForeQuantity qudt:numericValue ?operatingFore .

                $this nltl:maximumIceClassDraughtFore ?maximumForeQuantity .
                ?maximumForeQuantity qudt:numericValue ?maximumFore .

                $this nltl:minimumIceClassDraughtFore ?minimumForeQuantity .
                ?minimumForeQuantity qudt:numericValue ?minimumFore .

                $this nltl:operatingAftDraught ?operatingAftQuantity .
                ?operatingAftQuantity qudt:numericValue ?operatingAft .

                $this nltl:maximumIceClassDraughtAft ?maximumAftQuantity .
                ?maximumAftQuantity qudt:numericValue ?maximumAft .

                $this nltl:minimumIceClassDraughtAft ?minimumAftQuantity .
                ?minimumAftQuantity qudt:numericValue ?minimumAft .

                FILTER (
                    ?operatingFore < ?minimumFore ||
                    ?operatingFore > ?maximumFore ||
                    ?operatingAft < ?minimumAft ||
                    ?operatingAft > ?maximumAft
                )
            }
        """
    ] .

gen:TRF013LengthQuantity
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
