<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF014_LengthMetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF014_ClassCertificateShape
    a sh:NodeShape ;
    sh:class nltl:classCertificate ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAmidships ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAmidships ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF014_PostCutoffShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2007-07-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF014_PreCutoffMarkedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxExclusive "2007-07-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF014_PreCutoffDockingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxExclusive "2007-07-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:firstScheduledDryDockingDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF014_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node nltl:TRF014_LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasClassCertificate ;
        sh:node nltl:TRF014_ClassCertificateShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:class nltl:iceDraughtRestrictionDocument ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        nltl:TRF014_PostCutoffShape
        nltl:TRF014_PreCutoffMarkedShape
        nltl:TRF014_PreCutoffDockingShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required ice draught marking is missing or is not supported by the applicable date and draught condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:assessmentDate ?assessmentDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
                      nltl:upperIceWaterlineDraught ?uiwlQuantity .

                ?summerQuantity qudt:numericValue ?summerDraught .
                ?uiwlQuantity qudt:numericValue ?uiwlDraught .

                FILTER (
                    (
                        ?constructionDate >= "2007-07-01"^^xsd:date
                        && ?summerDraught > ?uiwlDraught
                        && NOT EXISTS {
                            $this nltl:warningTrianglePresent true ;
                                  nltl:iceClassDraughtMarkPresent true ;
                                  nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity .
                        }
                    )
                    ||
                    (
                        ?constructionDate < "2007-07-01"^^xsd:date
                        && ?summerDraught > ?uiwlDraught
                        && NOT EXISTS {
                            $this nltl:warningTrianglePresent true ;
                                  nltl:iceClassDraughtMarkPresent true ;
                                  nltl:iceClassDraughtMarkDraughtAmidships ?completeMarkQuantity .
                        }
                        && (
                            NOT EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
                            }
                            ||
                            EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
                                FILTER (?dryDockingDate <= ?assessmentDate)
                                FILTER NOT EXISTS {
                                    $this nltl:warningTrianglePresent true ;
                                          nltl:iceClassDraughtMarkPresent true ;
                                          nltl:iceClassDraughtMarkDraughtAmidships ?dueMarkQuantity .
                                }
                            }
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
