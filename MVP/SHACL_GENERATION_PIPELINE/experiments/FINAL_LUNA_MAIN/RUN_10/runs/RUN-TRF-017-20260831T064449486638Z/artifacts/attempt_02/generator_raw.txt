@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF017Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        <urn:nltl:generated-shape:TRF017RestrictedBranch>
        <urn:nltl:generated-shape:TRF017UnrestrictedBranch>
    ) .

<urn:nltl:generated-shape:TRF017RestrictedBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:restrictedPropulsionOutput ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must equal the restricted propulsion output." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?pQuantity .
                ?pQuantity qudt:numericValue ?pValue ;
                           qudt:unit unit:KiloW .
                $this nltl:restrictedPropulsionOutput ?rQuantity .
                ?rQuantity qudt:numericValue ?rValue ;
                           qudt:unit unit:KiloW .
                FILTER (?pValue != ?rValue)
            }
        """
    ] .

<urn:nltl:generated-shape:TRF017UnrestrictedBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:additionalPropulsionPower ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must equal the total continuous propulsion output, including any additional propulsion power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?pQuantity .
                ?pQuantity qudt:numericValue ?pValue ;
                           qudt:unit unit:KiloW .
                $this nltl:propulsionMachineryContinuousOutput ?mQuantity .
                ?mQuantity qudt:numericValue ?mValue ;
                           qudt:unit unit:KiloW .
                OPTIONAL {
                    $this nltl:additionalPropulsionPower ?aQuantity .
                    ?aQuantity qudt:numericValue ?aValue ;
                               qudt:unit unit:KiloW .
                }
                BIND (?mValue + COALESCE(?aValue, 0) AS ?expectedValue)
                FILTER (
                    ABS(?pValue - ?expectedValue) >
                    (ABS(?expectedValue) * 0.000001 + 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
