<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-018-PowerQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-018-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:calculatedRequiredPower ;
                sh:node <urn:nltl:generated-shape:TRF-018-PowerQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node <urn:nltl:generated-shape:TRF-018-PowerQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:maxCount 0
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximum continuous rating power must be at least the calculated required power and 1000 kW; for IA Super, it must be at least the calculated required power and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?mcrNode .

                ?calculatedNode qudt:numericValue ?calculatedPower ;
                                qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?mcrNode qudt:numericValue ?mcrPower ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloW> .

                FILTER (
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        )
                        &&
                        (
                            ?mcrPower < ?calculatedPower ||
                            ?mcrPower < 1000
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper
                        &&
                        (
                            ?mcrPower < ?calculatedPower ||
                            ?mcrPower < 2800
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
