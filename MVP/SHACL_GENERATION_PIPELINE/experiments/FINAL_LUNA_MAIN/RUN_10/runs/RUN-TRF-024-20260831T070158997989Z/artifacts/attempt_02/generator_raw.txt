<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-024-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF-024-ApplicableShape>
        <urn:nltl:generated-shape:TRF-024-NonApplicableIaShape>
    ) .

<urn:nltl:generated-shape:TRF-024-ApplicableShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        ) ;
        sh:minCount 1 ;
        sh:message "Applicable ship must have ice class IA Super or IA."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ship must have one construction stage date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ship must have one delivery date."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ship must have one assessment date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For IA Super or pre-1-September-2003 IA ships, the assessment date must not be later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                FILTER (
                    ?ice = nltl:iceClassIaSuper ||
                    (
                        ?ice = nltl:iceClassIa &&
                        (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        MONTH(?assessmentDate) = 1 &&
                        DAY(?assessmentDate) <= 1
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-024-NonApplicableIaShape> a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIa ;
        sh:minCount 1 ;
        sh:message "The non-applicable alternative is limited to ice class IA."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The IA applicability alternative requires one construction stage date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The non-applicable IA alternative requires construction stage on or after 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIa ;
                      nltl:constructionStageDate ?constructionDate .
                FILTER (
                    YEAR(?constructionDate) > 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        (
                            MONTH(?constructionDate) > 9 ||
                            (
                                MONTH(?constructionDate) = 9 &&
                                DAY(?constructionDate) >= 1
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
