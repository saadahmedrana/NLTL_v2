<urn:nltl:generated-shape:TRF-024-DeadlineCheck> a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The assessment date must be no later than 1 January in the year twenty years after the delivery year." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                BIND((YEAR(?deliveryDate) + 20) AS ?deadlineYear)
                FILTER (
                    YEAR(?assessmentDate) < ?deadlineYear
                    ||
                    (
                        YEAR(?assessmentDate) = ?deadlineYear
                        &&
                        MONTH(?assessmentDate) = 1
                        &&
                        DAY(?assessmentDate) <= 1
                    )
                )
            }
        """
    ] .
