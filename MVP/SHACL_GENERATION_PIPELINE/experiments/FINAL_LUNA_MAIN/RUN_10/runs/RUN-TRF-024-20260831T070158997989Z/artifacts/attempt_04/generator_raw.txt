<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF024Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF024ApplicableIaSuperShape
        gen:TRF024ApplicableIaShape
        gen:TRF024NonApplicableIaSuperShape
        gen:TRF024NonApplicableIaShape
        gen:TRF024OtherIceClassShape
    ) .

gen:TRF024ApplicableIaSuperShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper ;
        sh:minCount 1 ;
        sh:message "Ice class IA Super is required for this applicability branch."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:maxExclusive "2003-09-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction-stage date before 1 September 2003 is required for this branch."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A delivery date is required to establish the twenty-year deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An assessment date is required to evaluate the deadline."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                BIND(YEAR(?deliveryDate) + 20 AS ?deadlineYear)
                FILTER(
                    YEAR(?assessmentDate) > ?deadlineYear ||
                    (
                        YEAR(?assessmentDate) = ?deadlineYear &&
                        (
                            MONTH(?assessmentDate) > 1 ||
                            (
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    )
                )
            }
        """ ;
        sh:message "For an IA Super ship constructed before 1 September 2003, section 3.2.2 compliance was assessed after the 1 January twenty-year deadline."
    ] .

gen:TRF024NonApplicableIaSuperShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper ;

    sh:property [
