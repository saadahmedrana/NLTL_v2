gs:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:node gs:PressureQuantityShape ;
        sh:message "The ship must provide the externally determined ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape ;
        sh:message "The ship must provide the load patch height in metres."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape ;
        sh:message "The ship must provide the spacing input needed for position checks."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape ;
        sh:message "The ship must provide the span input needed for position checks."
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether load length was determined from the structural arrangement."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:minCount 1 ;
        sh:node gs:PressureQuantityShape ;
        sh:message "The externally computed applied ice pressure must be recorded in MPa."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:minCount 2 ;
                sh:class nltl:directAnalysisCase ;
                sh:node gs:TRF030DirectAnalysisCaseShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:minCount 2 ;
                sh:class nltl:directAnalysisCase ;
                sh:node gs:TRF030DirectAnalysisCaseWithLengthShape
            ]
        ]
    ) .

    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must record confirmation of the capacity-minimising load position."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must record that combined bending and shear were evaluated."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:class nltl:loadPositionTypeValue ;
        sh:minCount 1 ;
        sh:node gs:TRF030VerticalPositionTypeShape ;
        sh:message "Each direct analysis case must identify its vertical load position semantics."
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:class nltl:loadPositionTypeValue ;
        sh:minCount 1 ;
        sh:node gs:TRF030HorizontalPositionTypeShape ;

    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;

gs:TRF030DirectAnalysisCaseWithLengthShape
    a sh:NodeShape ;
    sh:and (
        gs:TRF030DirectAnalysisCaseShape
        [
            sh:property [
                sh:path nltl:loadPatchLength ;
                sh:minCount 1 ;
                sh:node gs:LengthQuantityShape ;

            ]
        ]
    ) .

gs:TRF030VerticalPositionTypeShape

    sh:or (
        [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition

        ]
        [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineReferencePosition ;
                sh:minCount 1 ;
                sh:node gs:LengthQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ] ;
            sh:property [
                sh:path nltl:lowerIceWaterlineReferencePosition ;
                sh:minCount 1 ;
                sh:node gs:LengthQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineReferencePosition ;
                sh:minCount 1 ;
                sh:node gs:LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:lowerIceWaterlineReferencePosition ;
                sh:minCount 1 ;

            ]
        ]

gs:TRF030HorizontalPositionTypeShape
