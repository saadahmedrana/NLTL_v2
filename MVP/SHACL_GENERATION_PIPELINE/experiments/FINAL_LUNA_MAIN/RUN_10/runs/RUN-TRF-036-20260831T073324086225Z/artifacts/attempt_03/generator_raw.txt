<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

<urn:nltl:generated-shape:TRF-036-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice-class design parameter set."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:message "The ship must specify one of the permitted Finnish-Swedish ice classes."
    ] .

<urn:nltl:generated-shape:TRF-036-ParameterSetShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue unit:M ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Level ice thickness must be a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue unit:M ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Design ice load height must be a single metre quantity."
    ] ;
    sh:sparql [
        sh:message "The level ice thickness and design ice load height do not conform to the Table 4-1 values for the ship's ice class." ;
        sh:select """
            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                      nltl:iceClass ?iceClass .
                $this nltl:levelIceThickness ?hiQuantity ;
                      nltl:designIceLoadHeight ?hQuantity .
                ?hiQuantity qudt:numericValue ?hi ;
                            qudt:unit unit:M .
                ?hQuantity qudt:numericValue ?h ;
                            qudt:unit unit:M .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper &&
                     (?hi > 1.0 || ?h < 0.35 || ?h > 0.35))
                    ||
                    (?iceClass = nltl:iceClassIa &&
                     (?hi > 0.8 || ?h < 0.30 || ?h > 0.30))
                    ||
                    (?iceClass = nltl:iceClassIb &&
                     (?hi > 0.6 || ?h < 0.25 || ?h > 0.25))
                    ||
                    (?iceClass = nltl:iceClassIc &&
                     (?hi > 0.4 || ?h < 0.22 || ?h > 0.22))
                )
            }
        """
    ] .
<END_SHACL>
