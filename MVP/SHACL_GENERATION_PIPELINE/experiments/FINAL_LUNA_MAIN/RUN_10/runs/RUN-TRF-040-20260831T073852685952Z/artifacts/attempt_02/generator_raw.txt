    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have one permitted Finnish-Swedish ice class."
    ] ;

    sh:property [
        sh:path nltl:openWaterServiceSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The open-water service speed shall be provided in knots."
    ] ;

    sh:property [
        sh:path nltl:designOperationSpeedInIce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The specified frame spacing input shall be provided in metres."
    ] ;

    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sidescuttles shall not be situated in the ice belt."
    ] ;

    sh:property [
        sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The weather-deck position shall be specified to determine the bulwark and freeing-port condition."
    ] ;

    sh:sparql [
        sh:message "For ice class IA Super, forefoot strengthening shall be equivalent to the bow region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER NOT EXISTS {
                    $this nltl:forefootShellStrengtheningStatus ?forefoot ;
                          nltl:bowRegionEquivalent ?bow .
                    FILTER (?forefoot = ?bow)
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "For IA Super or IA ships with open-water service speed of at least 18 knots, upper-bow ice-belt strengthening shall be equivalent to the midbody region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:openWaterServiceSpeed ?speed .
                ?speed qudt:numericValue ?speedValue ;
                       qudt:unit unit:KN .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper ||
                     ?iceClass = nltl:iceClassIa) &&
                    ?speedValue >= "18"^^xsd:decimal
                )
                FILTER NOT EXISTS {
                    $this nltl:upperBowIceBeltStrengtheningStatus ?upperBow ;
                          nltl:midbodyEquivalent ?midbody .
                    FILTER (?upperBow = ?midbody)
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "When the weather deck is below the upper ice-belt limit, bulwark and freeing-port strength shall meet the ice-belt shell requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this WHERE {
                $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                FILTER NOT EXISTS {
                    $this nltl:bulwarkStrengthStatus true ;
                          nltl:freeingPortStrengthStatus true .
                }
            }
        """
    ] .
