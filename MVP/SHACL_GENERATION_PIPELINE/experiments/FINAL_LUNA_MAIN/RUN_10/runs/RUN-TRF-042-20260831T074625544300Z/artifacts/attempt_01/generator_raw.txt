            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive 0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:UNITLESS
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxInclusive 2
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;

            ]

    ] ;
    sh:property [

        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must satisfy the applicable h/s formula within the specified tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .
                ?hNode qudt:numericValue ?h ;
                       qudt:unit qudt:M .
                ?sNode qudt:numericValue ?s ;
                       qudt:unit qudt:M .
                ?fNode qudt:numericValue ?actual ;
                       qudt:unit qudt:UNITLESS .
                FILTER (?s > 0)
                BIND (?h / ?s AS ?ratio)
                BIND (0.4 / ?ratio + 0.6 AS ?expectedLowRatio)
                BIND (1.4 - 0.4 * ?ratio AS ?expectedHighRatio)
                BIND (
                    IF(
                        ?ratio <= 1,
                        ?expectedLowRatio,
                        ?expectedHighRatio
                    ) AS ?expected
                )
                BIND (
                    IF(
                        1e-6 * ABS(?expected) > 1e-12,
                        1e-6 * ABS(?expected),
                        1e-12
                    ) AS ?tolerance
                )
                FILTER (
                    ?ratio <= 0 ||
                    ?ratio > 1.8 ||
                    ABS(?actual - ?expected) > ?tolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A corrosion and abrasion addition below 2 mm requires a maintained special surface coating and approved material evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc ;
                        qudt:unit qudt:MilliM .
                FILTER (?tc < 2)
                FILTER NOT EXISTS {
                    $this nltl:specialSurfaceCoatingMaintained ?coating ;
                          nltl:materialApprovalStatus nltl:evidenceStateApproved .
                    FILTER (?coating = true)
                }
            }
        """
    ] .
