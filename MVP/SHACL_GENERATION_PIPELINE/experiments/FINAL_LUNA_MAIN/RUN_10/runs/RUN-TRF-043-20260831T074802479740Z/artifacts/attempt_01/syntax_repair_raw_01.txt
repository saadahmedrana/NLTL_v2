<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasFramingIceStrengtheningTerminationCase ;
    sh:class nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:or (
            [
                sh:class nltl:deckStructure ;
            ]
            [
                sh:class nltl:tankBoundaryPlating ;
            ]
            [
                sh:class nltl:tankTop ;
            ]
        ) ;
        sh:minCount 1 ;
        sh:message "Each framing ice-strengthening termination case must identify a deck structure, tank boundary plating, or tank top as the adjacent boundary."
    ] ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each termination case must provide one extension quantity in millimetres."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    a sh:NodeShape ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:maxInclusive 250
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    a sh:NodeShape ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minExclusive 250
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "Termination at the adjacent boundary is permitted only when the extension beyond that boundary is no more than 250 mm."
    .
<END_SHACL>
