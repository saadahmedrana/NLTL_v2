<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-043-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node <urn:nltl:generated-shape:TRF-043-TerminationCaseShape> ;
        sh:message "Each framing ice-strengthening termination case must conform to the applicable termination constraints."
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The termination case must record its extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:message "The termination boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive 0 ;
                        sh:maxInclusive 250
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minExclusive 250
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "Where the extension is no more than 250 mm, termination at the adjacent boundary must be permitted." .

<urn:nltl:generated-shape:TRF-043-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:message "The frame must specify a permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:message "The frame must specify a bow, midbody, or stern hull region."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The frame must record its vertical extent above the upper ice waterline in metres."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The frame must record its vertical extent below the lower ice waterline in metres."
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue false ;
            ]
            [
                sh:hasValue true ;
            ]
        ) ;
        sh:message "The frame must specify whether an upper bow ice belt is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame vertical extents do not satisfy the applicable Table 4-6 ice-class and hull-region values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER ( NOT EXISTS {
                    $this nltl:iceClass ?iceClass ;
                          nltl:hullRegion ?region ;
                          nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                          nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .
                    ?above qudt:numericValue ?aboveValue .
                    ?below qudt:numericValue ?belowValue .

                    FILTER (
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?aboveValue >= 1.2 && ?aboveValue <= 1.2 &&
                            (
                                (
                                    ?region = nltl:bowRegion &&
                                    ?belowValue >= 2.0 && ?belowValue <= 2.0 &&
                                    $this nltl:reachesTankTopOrBelowFloorTop true
                                ) ||
                                (
                                    ?region = nltl:midbodyRegion &&
                                    ?belowValue >= 2.0 && ?belowValue <= 2.0
                                ) ||
                                (
                                    ?region = nltl:sternRegion &&
                                    ?belowValue >= 1.6 && ?belowValue <= 1.6
                                )
                            )
                        ) ||
                        (
                            ?iceClass IN (
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            ?aboveValue >= 1.0 && ?aboveValue <= 1.0 &&
                            (
                                (
                                    ?region = nltl:bowRegion &&
                                    ?belowValue >= 1.6 && ?belowValue <= 1.6
                                ) ||
                                (
                                    ?region = nltl:midbodyRegion &&
                                    ?belowValue >= 1.3 && ?belowValue <= 1.3
                                ) ||
                                (
                                    ?region = nltl:sternRegion &&
                                    ?belowValue >= 1.0 && ?belowValue <= 1.0
                                )
                            )
                        )
                    )
                } )
            }
        """
    ] .
<END_SHACL>
