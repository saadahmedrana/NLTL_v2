<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF045FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        gs:TRF045OrdinaryScantlingsNotUsed
        gs:TRF045OrdinaryScantlingsUsed
    ) ;
    sh:message "Ordinary frame scantlings may be used only when less than 15% of the frame span is within the ice-strengthening zone." .

gs:TRF045OrdinaryScantlingsNotUsed
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gs:TRF045OrdinaryScantlingsUsed
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSpanWithinIceStrengtheningZone ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When ordinary frame scantlings are used, the frame span within the ice-strengthening zone must be less than 15% of the frame span." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true ;
                      nltl:frameSpanWithinIceStrengtheningZone ?withinQuantity ;
                      nltl:span ?spanQuantity .
                ?withinQuantity qudt:numericValue ?withinValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?spanQuantity qudt:numericValue ?spanValue ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    !(?withinValue < (0.15 * ?spanValue))
                )
            }
        """
    ] .
<END_SHACL>
