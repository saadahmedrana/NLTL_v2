<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF047PermittedSupportingStructureShape
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:tankBoundaryPlating ]
        [ sh:class nltl:tankTop ]
        [ sh:class nltl:iceStringer ]
    ) .

gs:TRF047AttachedLowerEndShape
    a sh:NodeShape ;
    sh:class nltl:frameEnd ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node gs:TRF047PermittedSupportingStructureShape ;
        sh:message "The lower end must be attached to a permitted deck, tank plating, tank top, or ice stringer."
    ] .

gs:TRF047AlternativeLowerEndShape
    a sh:NodeShape ;
    sh:class nltl:frameEnd ;
    sh:property [
        sh:path nltl:hasTerminationBelowSupportingStructure ;
        sh:minCount 1 ;
        sh:node gs:TRF047PermittedSupportingStructureShape ;
        sh:message "The intermediate frame termination must be below a permitted supporting structure."
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:horizontalConnectionMember ;
            sh:property [
                sh:path nltl:connectsToAdjacentMainFrame ;
                sh:minCount 1 ;
                sh:class nltl:mainFrame ;
                sh:message "The horizontal connection member must connect to an adjacent main frame."
            ]
        ] ;
        sh:message "The permitted alternative requires a horizontal connection member."
    ] .

gs:TRF047MainFrameShape
    a sh:NodeShape ;
    sh:class nltl:mainFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:class nltl:frameStrengthenedPart ;
        sh:message "A main frame must have a strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:node gs:TRF047AttachedLowerEndShape ;
        sh:message "The lower end of a main frame strengthened part must be attached to a permitted supporting structure."
    ] ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Main frames below the lower edge of the ice belt must be ice strengthened."
    ] .

gs:TRF047IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:class nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:class nltl:frameStrengthenedPart ;
        sh:message "An intermediate ice frame must have a strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                gs:TRF047AttachedLowerEndShape
                gs:TRF047AlternativeLowerEndShape
            )
        ] ;
        sh:message "The lower end of an intermediate ice frame must be attached to a permitted supporting structure or satisfy the permitted horizontal-member alternative."
    ] ;
    sh:property [
        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The supporting structure for the horizontal-member alternative must be at or below the lower limit of the ice belt."
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The horizontal connection member must have the same scantlings as the main frames."
    ] .

gs:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        gs:TRF047MainFrameShape
        gs:TRF047IntermediateIceFrameShape
    ) ;
    sh:message "The frame must conform as a main frame or an intermediate ice frame under clause 4.4.2.3." .
<END_SHACL>
