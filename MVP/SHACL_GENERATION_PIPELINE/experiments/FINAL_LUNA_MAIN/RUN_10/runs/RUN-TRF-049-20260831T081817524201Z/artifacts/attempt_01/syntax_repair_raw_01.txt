<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF049FrameShape a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or [
            rdf:first [
                sh:hasValue true ;
                sh:property [
                    sh:path nltl:effectiveAttachmentConfirmed ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ] ;
                sh:property [
                    sh:path nltl:hasFrameAttachment ;
                    sh:minCount 1 ;
                    sh:node gen:TRF049AttachmentShape
                ]
            ] ;
            rdf:rest [
                rdf:first [
                    sh:hasValue false
                ] ;
                rdf:rest rdf:nil
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node gen:TRF049AttachmentShape
    ] ;
    sh:or [
        rdf:first [
            sh:class nltl:longitudinalFrame ;
            sh:property [
                sh:path nltl:hasSupportingWebFrame ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ] ;
            sh:property [
                sh:path nltl:hasSupportingBulkhead ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node gen:TRF049FittedAttachmentShape
            ]
        ] ;
        rdf:rest [
            rdf:first [
                sh:class nltl:transverseFrame ;
                sh:property [
                    sh:path nltl:terminatesAtDeckOrIceStringer ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:or [
                        rdf:first [
                            sh:hasValue true ;
                            sh:property [
                                sh:path nltl:hasFrameAttachment ;
                                sh:minCount 1 ;
                                sh:node gen:TRF049FittedAttachmentShape
                            ]
                        ] ;
                        rdf:rest [
                            rdf:first [
                                sh:hasValue false
                            ] ;
                            rdf:rest rdf:nil
                        ]
                    ]
                ]
            ] ;
            rdf:rest rdf:nil
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each installed bracket must be at least as thick as the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameThickness ;
                      nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThickness .

                ?frameThickness qudt:numericValue ?frameValue .
                ?bracketThickness qudt:numericValue ?bracketValue .

                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

gen:TRF049AttachmentShape a sh:NodeShape ;
    sh:class nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] .

gen:TRF049FittedAttachmentShape a sh:NodeShape ;
    sh:class nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:or [
        rdf:first [
            sh:property [
                sh:path nltl:hasConnectionBracket ;
                sh:minCount 1 ;
                sh:node gen:TRF049BracketShape
            ]
        ] ;
        rdf:rest [
            rdf:first [
                sh:property [
                    sh:path nltl:similarAttachmentConstructionPresent ;
                    sh:datatype xsd:boolean ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:hasValue true
                ]
            ] ;
            rdf:rest rdf:nil
        ]
    ] .

gen:TRF049BracketShape a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF049BracketThicknessShape
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:TRF049BracketThicknessShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
