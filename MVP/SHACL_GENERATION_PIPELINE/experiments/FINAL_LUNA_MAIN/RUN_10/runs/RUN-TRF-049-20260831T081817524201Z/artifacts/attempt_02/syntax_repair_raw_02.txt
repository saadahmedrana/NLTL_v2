                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    WHERE {
                        $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                        FILTER NOT EXISTS {
                            ?support nltl:hasSupportingStructureAttachment ?attachment .

    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [

                sh:datatype xsd:boolean ;
                sh:minCount 1 ;

                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "A bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?frameThickness .
                    ?frameThickness qudt:numericValue ?frameValue .
                    ?frameThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?bracket nltl:bracketThickness ?bracketThickness .
                    ?bracketThickness qudt:numericValue ?bracketValue .
                    ?bracketThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (?bracketValue >= ?frameValue)
                }
            }
        """
    ] .
