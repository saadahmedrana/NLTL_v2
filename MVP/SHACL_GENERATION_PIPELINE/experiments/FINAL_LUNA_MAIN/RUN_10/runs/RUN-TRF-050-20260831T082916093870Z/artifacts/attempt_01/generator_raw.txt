<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF050_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord ;
        sh:minCount 1 ;
        sh:message "Each frame must have a frame attachment record."
    ] .

gen:TRF050_FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must be attached to the shell by a double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Shell plate butt crossing must be recorded to determine whether the scalloping prohibition applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Scalloping is not allowed when the frame does not cross a shell plate butt."
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue true
            ]
        ]
    ) .
<END_SHACL>
