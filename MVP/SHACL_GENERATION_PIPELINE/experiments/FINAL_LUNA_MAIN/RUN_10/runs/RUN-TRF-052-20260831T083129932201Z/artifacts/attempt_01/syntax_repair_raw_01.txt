<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-052-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Frame asymmetry status is required to determine antitripping support applicability."
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Frame web angle to shell is required in degrees."
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Frame span is required in metres."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A permitted Finnish-Swedish ice-class value is required."
    ] ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Equivalent-support calculation status is required to select the alternative arrangement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame does not satisfy the applicable antitripping support spacing or regional scope requirement." ;
        sh:select """PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:equivalentSupportStatusByDirectCalculation ?equivalent .
    FILTER (?equivalent != true)

    {
        $this nltl:frameAsymmetrical ?asymmetrical ;
              nltl:frameWebAngleToShell ?angleNode .
        ?angleNode qudt:numericValue ?angle .
        FILTER (
            ?asymmetrical = true ||
            (datatype(?angle) = xsd:decimal && ?angle < 90)
        )
        FILTER NOT EXISTS {
            $this nltl:antitrippingSupportSpacing ?spacingNode .
            ?spacingNode qudt:numericValue ?spacing ;
                         qudt:unit <http://qudt.org/vocab/unit/MilliM> .
            FILTER (
                datatype(?spacing) = xsd:decimal &&
                ?spacing <= 1300
            )
        }
    }
    UNION
    {
        $this nltl:frameSpan ?spanNode .
        ?spanNode qudt:numericValue ?span .
        FILTER (
            datatype(?span) = xsd:decimal &&
            ?span > 4
        )
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
            }
        }
    }
    UNION
    {
        $this nltl:frameSpan ?spanNode ;
              nltl:iceClass ?iceClass .
        ?spanNode qudt:numericValue ?span .
        FILTER (
            datatype(?span) = xsd:decimal &&
            ?span <= 4 &&
            ?iceClass = nltl:iceClassIaSuper
        )
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
            }
        }
    }
    UNION
    {
        $this nltl:frameSpan ?spanNode ;
              nltl:iceClass nltl:iceClassIa .
        ?spanNode qudt:numericValue ?span .
        FILTER (
            datatype(?span) = xsd:decimal &&
            ?span <= 4
        )
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
            }
        }
        UNION
        {
            FILTER NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
            }
        }
    }
    UNION
    {
        $this nltl:frameSpan ?spanNode ;
              nltl:iceClass ?iceClass .
        ?spanNode qudt:numericValue ?span .
        FILTER (
            datatype(?span) = xsd:decimal &&
            ?span <= 4 &&
            (
                ?iceClass = nltl:iceClassIb ||
                ?iceClass = nltl:iceClassIc
            )
        )
        FILTER NOT EXISTS {
            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
        }
    }
}"""
    ] .
</END_SHACL>
