<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF052FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame does not satisfy the applicable antitripping support spacing or scope requirement." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    OPTIONAL {
        $this nltl:equivalentSupportStatusByDirectCalculation ?status
    }
    OPTIONAL {
        $this nltl:frameAsymmetrical ?asym
    }
    OPTIONAL {
        $this nltl:frameWebAngleToShell ?angleNode .
        ?angleNode qudt:numericValue ?angleValue .
        ?angleNode qudt:unit <http://qudt.org/vocab/unit/DEG>
    }
    OPTIONAL {
        $this nltl:frameSpan ?spanNode .
        ?spanNode qudt:numericValue ?spanValue .
        ?spanNode qudt:unit <http://qudt.org/vocab/unit/M>
    }
    OPTIONAL {
        $this nltl:antitrippingSupportSpacing ?spacingNode .
        ?spacingNode qudt:numericValue ?spacingValue .
        ?spacingNode qudt:unit <http://qudt.org/vocab/unit/MilliM>
    }
    OPTIONAL {
        $this nltl:iceClass ?ice
    }
    FILTER NOT EXISTS {
        $this nltl:equivalentSupportStatusByDirectCalculation true
    }
    FILTER (
        !BOUND(?asym) ||
        DATATYPE(?asym) != xsd:boolean ||
        !BOUND(?angleValue) ||
        DATATYPE(?angleValue) != xsd:decimal ||
        !BOUND(?spanValue) ||
        DATATYPE(?spanValue) != xsd:decimal ||
        !BOUND(?spacingValue) ||
        DATATYPE(?spacingValue) != xsd:decimal ||
        !BOUND(?ice) ||
        ?ice NOT IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc,
            nltl:iceClassIi,
            nltl:iceClassIii
        ) ||
        (
            (
                ?asym = true ||
                ?angleValue < 90
            ) &&
            ?spacingValue > 1300
        ) ||
        (
            ?spanValue <= 4 &&
            ?ice = nltl:iceClassIa &&
            (
                NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                } ||
                NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                }
            )
        ) ||
        (
            ?spanValue <= 4 &&
            (
                ?ice = nltl:iceClassIb ||
                ?ice = nltl:iceClassIc
            ) &&
            NOT EXISTS {
                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
            }
        )
    )
}
"""
    ] .
<END_SHACL>
