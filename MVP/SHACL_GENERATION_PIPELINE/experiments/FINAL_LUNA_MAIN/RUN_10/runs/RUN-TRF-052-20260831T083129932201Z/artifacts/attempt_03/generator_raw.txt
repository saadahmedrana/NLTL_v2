<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF052FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame does not satisfy the applicable antitripping support spacing or regional coverage requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:equivalentSupportStatusByDirectCalculation ?status .
                    FILTER (?status = true)
                }

                $this nltl:frameAsymmetrical ?asymmetrical .
                $this nltl:frameWebAngleToShell ?angleNode .
                ?angleNode qudt:numericValue ?angleValue .

                $this nltl:frameSpan ?spanNode .
                ?spanNode qudt:numericValue ?spanValue .

                $this nltl:iceClass ?iceClass .

                FILTER (
                    (
                        (
                            (?asymmetrical = true) ||
                            (?angleValue < 90)
                        )
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:antitrippingSupportSpacing ?spacingNode .
                                ?spacingNode qudt:numericValue ?spacingValue ;
                                             qudt:unit qudt:MilliM .
                                FILTER (
                                    datatype(?spacingValue) = xsd:decimal &&
                                    ?spacingValue <= 1300
                                )
                            }
                            ||
                            EXISTS {
                                $this nltl:antitrippingSupportSpacing ?spacingNode1 .
                                $this nltl:antitrippingSupportSpacing ?spacingNode2 .
                                FILTER (?spacingNode1 != ?spacingNode2)
                            }
                        )
                    )
                    ||
                    (
                        ?spanValue > 4 &&
                        (
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
                            }
                        )
                    )
                    ||
                    (
                        ?spanValue <= 4 &&
                        ?iceClass = nltl:iceClassIaSuper &&
                        (
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
                            }
                        )
                    )
                    ||
                    (
                        ?spanValue <= 4 &&
                        ?iceClass = nltl:iceClassIa &&
                        (
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                            }
                        )
                    )
                    ||
                    (
                        ?spanValue <= 4 &&
                        (
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        NOT EXISTS {
                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                        }
                    )
                )
            }
        """
    ] .

</END_SHACL>
