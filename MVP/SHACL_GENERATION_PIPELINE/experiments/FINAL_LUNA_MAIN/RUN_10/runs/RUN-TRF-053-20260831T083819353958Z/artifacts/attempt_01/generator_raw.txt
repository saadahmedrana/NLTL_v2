<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF053IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:IcePressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:DesignIceLoadHeightQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:SpanQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:FrameMomentFactorQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one dimensionless frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:YieldStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:DesignLineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated design line load in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated section modulus in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:RequiredShearAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated required shear area in cm2."
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area must satisfy the declared ice-stringer formulas; calculated decimal results are checked with a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:designLineLoad ?loadNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?loadNode qudt:numericValue ?reportedLoad .
                ?zNode qudt:numericValue ?reportedZ .
                ?aNode qudt:numericValue ?reportedA .

                BIND((?p * ?h) AS ?rawLoad)
                BIND(IF(?rawLoad > 0.15, ?rawLoad, 0.15) AS ?expectedLoad)
                BIND((0.9 * 1.8 * ?expectedLoad * ?l * ?l / (?m * ?sigma) * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?expectedLoad * ?l / (2 * ?sigma) * 10000) AS ?expectedA)

                FILTER(
                    ABS(?reportedLoad - ?expectedLoad) >
                    IF(ABS(?expectedLoad) * 0.000001 > 0.000000001,
                       ABS(?expectedLoad) * 0.000001,
                       0.000000001)
                    ||
                    ABS(?reportedZ - ?expectedZ) >
                    IF(ABS(?expectedZ) * 0.000001 > 0.000000001,
                       ABS(?expectedZ) * 0.000001,
                       0.000000001)
                    ||
                    ABS(?reportedA - ?expectedA) >
                    IF(ABS(?expectedA) * 0.000001 > 0.000000001,
                       ABS(?expectedA) * 0.000001,
                       0.000000001)
                )
            }
        """
    ] .
