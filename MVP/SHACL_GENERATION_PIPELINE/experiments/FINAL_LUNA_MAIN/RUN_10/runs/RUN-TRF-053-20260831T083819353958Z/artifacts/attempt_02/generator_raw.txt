<urn:nltl:generated-shape:TRF-053-IceStringerShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-IcePressureValueShape>

    sh:property [
        sh:path nltl:designIceLoadHeight ;

    sh:property [
        sh:path nltl:frameMomentFactorM ;

        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-FrameMomentFactorValueShape>
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-YieldStrengthValueShape>
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-DesignLineLoadValueShape>
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-SectionModulusValueShape>
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-053-RequiredShearAreaValueShape>
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, and required shear area must satisfy the declared TRF-053 formulas; design line load is reported in kN/m while the formula operands use MN/m. Formula-result tolerance is relative 1e-6 with an absolute fallback scale of 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:designLineLoad ?loadNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p ;
                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?hNode qudt:numericValue ?h ;
                       qudt:unit <http://qudt.org/vocab/unit/M> .
                ?lNode qudt:numericValue ?l ;
                       qudt:unit <http://qudt.org/vocab/unit/M> .
                ?mNode qudt:numericValue ?m ;
                       qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?sigmaNode qudt:numericValue ?sigma ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?loadNode qudt:numericValue ?reportedLoad ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                ?zNode qudt:numericValue ?reportedZ ;
                       qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?aNode qudt:numericValue ?reportedA ;
                       qudt:unit <http://qudt.org/vocab/unit/CentiM2> .

                BIND(?p * ?h AS ?product)
                BIND(IF(?product < 0.15, 0.15, ?product) AS ?expectedLoadMNPerM)
                BIND(1000 * ?expectedLoadMNPerM AS ?expectedLoadKNPerM)
                BIND(0.9 * 1.8 * ?expectedLoadMNPerM * ?l * ?l /
                     (?m * ?sigma) * 1000000 AS ?expectedZ)
                BIND(1.7320508075688772 * 0.9 * 1.8 * 1.2 *
                     ?expectedLoadMNPerM * ?l /
                     (2 * ?sigma) * 10000 AS ?expectedA)

                FILTER(
                    ABS(?reportedLoad - ?expectedLoadKNPerM) >
                    0.000001 * IF(
                        ABS(?expectedLoadKNPerM) > 1,
                        ABS(?expectedLoadKNPerM),
                        1
                    )
                    ||
                    ABS(?reportedZ - ?expectedZ) >
                    0.000001 * IF(
                        ABS(?expectedZ) > 1,
                        ABS(?expectedZ),
                        1
                    )
                    ||
                    ABS(?reportedA - ?expectedA) >
                    0.000001 * IF(
                        ABS(?expectedA) > 1,
                        ABS(?expectedA),
                        1
                    )
                )
            }
        """
    ] .
