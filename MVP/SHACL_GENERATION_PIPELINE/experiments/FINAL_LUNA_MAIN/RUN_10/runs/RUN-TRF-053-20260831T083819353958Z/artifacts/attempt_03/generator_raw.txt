gen:TRF053IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:QuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DesignIceLoadHeightQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SpanQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FrameMomentFactorQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:YieldStrengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DesignLineLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:RequiredShearAreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, and required shear area must satisfy the TRF-053 formulas; formula-result comparisons use a 1e-6 relative tolerance plus a 1e-9 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this
                    nltl:icePressure ?pNode ;
                    nltl:designIceLoadHeight ?hNode ;
                    nltl:span ?lNode ;
                    nltl:frameMomentFactorM ?mNode ;
                    nltl:yieldStrength ?sigmaNode ;
                    nltl:designLineLoad ?loadNode ;
                    nltl:sectionModulus ?zNode ;
                    nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?loadNode qudt:numericValue ?reportedLoad .
                ?zNode qudt:numericValue ?reportedZ .
                ?aNode qudt:numericValue ?reportedA .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?expectedLoad)
                BIND(1000 * ?expectedLoad AS ?expectedReportedLoad)
                BIND((0.9 * 1.8 * ?expectedLoad * ?l * ?l /
                      (?m * ?sigma)) * 1000000 AS ?expectedZ)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 *
                      ?expectedLoad * ?l / (2 * ?sigma)) * 10000 AS ?expectedA)

                BIND((0.000001 * IF(?expectedReportedLoad < 0,
                      -?expectedReportedLoad, ?expectedReportedLoad)) + 0.000000001 AS ?loadTolerance)
                BIND((0.000001 * IF(?expectedZ < 0, -?expectedZ, ?expectedZ)) + 0.000000001 AS ?zTolerance)
                BIND((0.000001 * IF(?expectedA < 0, -?expectedA, ?expectedA)) + 0.000000001 AS ?aTolerance)

                BIND(?reportedLoad - ?expectedReportedLoad AS ?loadDifference)
                BIND(?reportedZ - ?expectedZ AS ?zDifference)
                BIND(?reportedA - ?expectedA AS ?aDifference)

                FILTER(
                    (?loadDifference > ?loadTolerance) ||
                    (?loadDifference < -?loadTolerance) ||
                    (?zDifference > ?zTolerance) ||
                    (?zDifference < -?zTolerance) ||
                    (?aDifference > ?aTolerance) ||
                    (?aDifference < -?aTolerance)
                )
            }
        """
    ] .
