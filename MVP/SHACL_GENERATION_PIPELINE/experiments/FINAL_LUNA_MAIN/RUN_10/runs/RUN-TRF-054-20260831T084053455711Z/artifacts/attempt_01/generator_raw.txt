    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054YieldStrengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054ForcePerLengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF054AreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "The design line load, outside-ice-belt factor, section modulus, and required shear area must match the declared calculation within a relative tolerance of 1e-6, with an absolute tolerance of 1e-9 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:distanceToIceBelt ?iceBeltDistanceNode ;
                      nltl:stringerSpan ?stringerSpanNode ;
                      nltl:designLineLoad ?lineLoadNode ;
                      nltl:outsideIceBeltFactor ?outsideFactorNode ;
                      nltl:sectionModulus ?sectionModulusNode ;
                      nltl:requiredShearArea ?shearAreaNode .

                ?pressureNode qudt:numericValue ?pressure .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?momentNode qudt:numericValue ?moment .
                ?yieldNode qudt:numericValue ?yield .
                ?iceBeltDistanceNode qudt:numericValue ?iceBeltDistance .
                ?stringerSpanNode qudt:numericValue ?stringerSpan .
                ?lineLoadNode qudt:numericValue ?lineLoad .
                ?outsideFactorNode qudt:numericValue ?outsideFactor .
                ?sectionModulusNode qudt:numericValue ?sectionModulus .
                ?shearAreaNode qudt:numericValue ?shearArea .

                BIND(IF((?pressure * ?height) < 0.15,
                        0.15,
                        (?pressure * ?height)) AS ?expectedLineLoad)
                BIND((1.0 - (?iceBeltDistance / ?stringerSpan)) AS ?expectedOutsideFactor)
                BIND((0.80 * 1.8 * ?expectedLineLoad * ?stringerSpan * ?stringerSpan /
                      (?moment * ?yield) * ?expectedOutsideFactor * 1000000.0)
                      AS ?expectedSectionModulus)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?expectedLineLoad *
                      ?stringerSpan / (2.0 * ?yield) * ?expectedOutsideFactor * 10000.0)
                      AS ?expectedShearArea)

                BIND(IF(ABS(?expectedLineLoad) > 0.000001,
                        ABS(?expectedLineLoad) * 0.000001,
                        0.000000001) AS ?lineTolerance)
                BIND(IF(ABS(?expectedOutsideFactor) > 0.000001,
                        ABS(?expectedOutsideFactor) * 0.000001,
                        0.000000001) AS ?outsideTolerance)
                BIND(IF(ABS(?expectedSectionModulus) > 0.000001,
                        ABS(?expectedSectionModulus) * 0.000001,
                        0.000000001) AS ?sectionTolerance)
                BIND(IF(ABS(?expectedShearArea) > 0.000001,
                        ABS(?expectedShearArea) * 0.000001,
                        0.000000001) AS ?areaTolerance)

                FILTER(
                    ABS(?lineLoad - ?expectedLineLoad) > ?lineTolerance ||
                    ABS(?outsideFactor - ?expectedOutsideFactor) > ?outsideTolerance ||
                    ABS(?sectionModulus - ?expectedSectionModulus) > ?sectionTolerance ||
                    ABS(?shearArea - ?expectedShearArea) > ?areaTolerance
                )
            }
        """
    ] .
