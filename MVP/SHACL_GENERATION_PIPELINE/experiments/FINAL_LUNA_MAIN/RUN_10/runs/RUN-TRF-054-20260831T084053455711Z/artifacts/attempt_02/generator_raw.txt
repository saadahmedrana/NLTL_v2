<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ForcePerLengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceStringerCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;

    sh:property [
        sh:path nltl:locatedOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice stringer must state whether it is located outside the ice belt."
    ] ;

    sh:property [
        sh:path nltl:supportsIceStrengthenedFrames ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice stringer must state whether it supports ice-strengthened frames."
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:icePressure ;
                sh:node gen:PressureQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designIceLoadHeight ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:span ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:stringerSpan ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:PressureQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToIceBelt ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToAdjacentIceStringer ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designLineLoad ;
                sh:node gen:ForcePerLengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:outsideIceBeltFactor ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:sectionModulus ;
                sh:node gen:SectionModulusQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredShearArea ;
                sh:node gen:AreaQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;

    sh:sparql [
        sh:message "For an ice stringer outside the ice belt supporting ice-strengthened frames, the declared loads, factor, section modulus, and shear area must satisfy the specified calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true ;
                      nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:span/qudt:numericValue ?span ;
                      nltl:stringerSpan/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hs ;
                      nltl:distanceToAdjacentIceStringer/qudt:numericValue ?ls ;
                      nltl:designLineLoad/qudt:numericValue ?reportedLoad ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?reportedFactor ;
                      nltl:sectionModulus/qudt:numericValue ?reportedZ ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedA .

                BIND(
                    IF((?p * ?h) < 0.15, 0.15, (?p * ?h))
                    AS ?loadMNPerM
                )
                BIND((?loadMNPerM * 1000) AS ?expectedLoad)
                BIND((1 - (?hs / ?ls)) AS ?expectedFactor)
                BIND(
                    (0.80 * 1.8 * ?loadMNPerM * ?l * ?l /
                     (?m * ?sigma) * (1 - (?hs / ?ls)) * 1000000)
                    AS ?expectedZ
                )
                BIND(
                    (1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                     ?loadMNPerM * ?l / (2 * ?sigma) *
                     (1 - (?hs / ?ls)) * 10000)
                    AS ?expectedA
                )

                FILTER(
                    (?span != ?l) ||
                    (
                        (
                            ?expectedLoad > 0.001 &&
                            (
                                ?reportedLoad < (?expectedLoad * 0.999999) ||
                                ?reportedLoad > (?expectedLoad * 1.000001)
                            )
                        ) ||
                        (
                            ?expectedLoad <= 0.001 &&
                            ?expectedLoad >= -0.001 &&
                            (
                                ?reportedLoad < (?expectedLoad - 0.000000001) ||
                                ?reportedLoad > (?expectedLoad + 0.000000001)
                            )
                        )
                    ) ||
                    (
                        (
                            ?expectedFactor > 0.001 &&
                            (
                                ?reportedFactor < (?expectedFactor * 0.999999) ||
                                ?reportedFactor > (?expectedFactor * 1.000001)
                            )
                        ) ||
                        (
                            ?expectedFactor <= 0.001 &&
                            ?expectedFactor >= -0.001 &&
                            (
                                ?reportedFactor < (?expectedFactor - 0.000000001) ||
                                ?reportedFactor > (?expectedFactor + 0.000000001)
                            )
                        )
                    ) ||
                    (
                        (
                            ?expectedZ > 0.001 &&
                            (
                                ?reportedZ < (?expectedZ * 0.999999) ||
                                ?reportedZ > (?expectedZ * 1.000001)
                            )
                        ) ||
                        (
                            ?expectedZ <= 0.001 &&
                            ?expectedZ >= -0.001 &&
                            (
                                ?reportedZ < (?expectedZ - 0.000000001) ||
                                ?reportedZ > (?expectedZ + 0.000000001)
                            )
                        )
                    ) ||
                    (
                        (
                            ?expectedA > 0.001 &&
                            (
                                ?reportedA < (?expectedA * 0.999999) ||
                                ?reportedA > (?expectedA * 1.000001)
                            )
                        ) ||
                        (
                            ?expectedA <= 0.001 &&
                            ?expectedA >= -0.001 &&
                            (
                                ?reportedA < (?expectedA - 0.000000001) ||
                                ?reportedA > (?expectedA + 0.000000001)
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
