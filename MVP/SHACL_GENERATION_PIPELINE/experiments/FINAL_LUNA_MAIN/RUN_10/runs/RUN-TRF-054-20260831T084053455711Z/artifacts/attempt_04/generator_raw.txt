nltlg:IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:locatedOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice stringer must be located outside the ice belt."
    ] ;
    sh:property [
        sh:path nltl:supportsIceStrengthenedFrames ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice stringer must support ice-strengthened frames."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltlg:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltlg:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltlg:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node nltlg:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltlg:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltlg:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node nltlg:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToAdjacentIceStringer ;
        sh:node nltlg:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltlg:KiloNewtonPerMeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node nltlg:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltlg:CentiMeterCubedQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltlg:CentiMeterSquaredQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable ice stringer, design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy the declared calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true ;
                      nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:span/qudt:numericValue ?span ;
                      nltl:stringerSpan/qudt:numericValue ?stringerSpan ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hs ;
                      nltl:distanceToAdjacentIceStringer/qudt:numericValue ?ls ;
                      nltl:designLineLoad/qudt:numericValue ?reportedLoad ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?reportedFactor ;
                      nltl:sectionModulus/qudt:numericValue ?reportedZ ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedA .

                FILTER (?span = ?stringerSpan)

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?loadMNPerM)
                BIND(?loadMNPerM * 1000 AS ?expectedLoad)
                BIND(1 - (?hs / ?ls) AS ?expectedFactor)
                BIND(0.80 * 1.8 * ?loadMNPerM * (?stringerSpan * ?stringerSpan) /
                     (?m * ?sigma) * ?expectedFactor * 1000000 AS ?expectedZ)
                BIND(1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?loadMNPerM *
                     ?stringerSpan / (2 * ?sigma) * ?expectedFactor * 10000 AS ?expectedA)

                FILTER (
                    (?reportedLoad < ?expectedLoad - 0.000001 * ?expectedLoad) ||
                    (?reportedLoad > ?expectedLoad + 0.000001 * ?expectedLoad) ||
                    (?expectedFactor >= 1 &&
                     (?reportedFactor < ?expectedFactor - 0.000001 * ?expectedFactor ||
                      ?reportedFactor > ?expectedFactor + 0.000001 * ?expectedFactor)) ||
                    (?expectedFactor < 1 && ?expectedFactor > -1 &&
                     (?reportedFactor < ?expectedFactor - 0.000001 ||
                      ?reportedFactor > ?expectedFactor + 0.000001)) ||
                    (?expectedFactor <= -1 &&
                     (?reportedFactor < ?expectedFactor - 0.000001 * (-?expectedFactor) ||
                      ?reportedFactor > ?expectedFactor + 0.000001 * (-?expectedFactor))) ||
                    (?expectedZ >= 1 &&
                     (?reportedZ < ?expectedZ - 0.000001 * ?expectedZ ||
                      ?reportedZ > ?expectedZ + 0.000001 * ?expectedZ)) ||
                    (?expectedZ < 1 && ?expectedZ > -1 &&
                     (?reportedZ < ?expectedZ - 0.000001 ||
                      ?reportedZ > ?expectedZ + 0.000001)) ||
                    (?expectedZ <= -1 &&
                     (?reportedZ < ?expectedZ - 0.000001 * (-?expectedZ) ||
                      ?reportedZ > ?expectedZ + 0.000001 * (-?expectedZ))) ||
                    (?expectedA >= 1 &&
                     (?reportedA < ?expectedA - 0.000001 * ?expectedA ||
                      ?reportedA > ?expectedA + 0.000001 * ?expectedA)) ||
                    (?expectedA < 1 && ?expectedA > -1 &&
                     (?reportedA < ?expectedA - 0.000001 ||
                      ?reportedA > ?expectedA + 0.000001)) ||
                    (?expectedA <= -1 &&
                     (?reportedA < ?expectedA - 0.000001 * (-?expectedA) ||
                      ?reportedA > ?expectedA + 0.000001 * (-?expectedA)))
                )
            }
        """
    ] .
