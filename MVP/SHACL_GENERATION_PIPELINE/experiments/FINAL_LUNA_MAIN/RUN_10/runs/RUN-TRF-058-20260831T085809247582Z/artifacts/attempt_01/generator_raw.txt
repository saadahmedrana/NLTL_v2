gs:TRF058OutsideIceBeltCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:supportedStringerOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The applicability selector must state that the supported stringer is outside the ice belt."
    ] ;
    sh:property [
        sh:path nltl:iceLoadForce ;
        sh:node gs:QuantityKiloNewtonShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice load force is required when the stringer is outside the ice belt."
    ] ;
    sh:property [
        sh:path nltl:stringerOutsideIceBeltHeight ;
        sh:node gs:QuantityMetreShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The stringer height outside the ice belt is required when applicable."
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gs:QuantityMetreShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The stringer span is required when the stringer is outside the ice belt."
    ] ;
    sh:property [
        sh:path nltl:adjustedIceLoadForce ;
        sh:node gs:QuantityKiloNewtonShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The adjusted ice load force is required when the stringer is outside the ice belt."
    ] ;
    sh:sparql [
        sh:message "Adjusted ice load force must equal ice load force multiplied by (1 - stringer outside ice belt height / stringer span), within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadForce ?forceNode ;
                      nltl:stringerOutsideIceBeltHeight ?heightNode ;
                      nltl:stringerSpan ?spanNode ;
                      nltl:adjustedIceLoadForce ?adjustedNode .

                ?forceNode qudt:numericValue ?force .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?adjustedNode qudt:numericValue ?adjusted .

                FILTER (?span != 0)
                BIND (?force * (1 - (?height / ?span)) AS ?expected)
                FILTER (
                    ABS(?adjusted - ?expected) > 0.000000001 &&
                    ABS(?adjusted - ?expected) > (ABS(?expected) * 0.000001)
                )
            }
        """
    ] .
