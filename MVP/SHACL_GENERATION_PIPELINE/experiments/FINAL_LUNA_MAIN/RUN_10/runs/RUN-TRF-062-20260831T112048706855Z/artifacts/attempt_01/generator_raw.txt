<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-062-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:minimumClearance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "The ship must specify one minimum clearance in metres."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeTipHullClearance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "The ship must specify one propeller blade tip to hull clearance in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The propeller blade tip to hull clearance must be no less than the minimum clearance h0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:minimumClearance ?minimumNode .
                ?minimumNode qudt:numericValue ?minimumValue .
                $this nltl:propellerBladeTipHullClearance ?clearanceNode .
                ?clearanceNode qudt:numericValue ?clearanceValue .
                FILTER (
                    !(xsd:decimal(?clearanceValue) >= xsd:decimal(?minimumValue))
                )
            }
        """
    ] .
<END_SHACL>
