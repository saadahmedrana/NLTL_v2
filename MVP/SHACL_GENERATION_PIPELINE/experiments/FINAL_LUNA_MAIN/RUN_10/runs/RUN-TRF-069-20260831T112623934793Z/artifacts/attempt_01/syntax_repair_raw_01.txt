<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF-069-StaticShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:message "The ship shall declare an applicable Finnish-Swedish ice class."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:in ( "open" "ducted" ) ;
        sh:minCount 1 ;
        sh:message "The ship shall declare an open or ducted propeller type."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:in ( "controllable" "fixed" ) ;
        sh:minCount 1 ;
        sh:message "The ship shall declare a controllable or fixed pitch type."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The maximum ice torque shall be provided as a decimal quantity in kN*m."
    ] ;
    sh:sparql [
        sh:message "For an applicable propulsion-machinery configuration, the design condition shall record normal service-life operation, fixed-pitch reversal loads, and the prescribed exclusions for stopped-propeller dragging and radial ice entry." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
                FILTER (?propellerType IN ("open", "ducted"))
                FILTER (?pitchType IN ("controllable", "fixed"))

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?condition .
                    ?condition a nltl:designCondition ;
                               nltl:normalOperationalConditions ?normal ;
                               nltl:normalServiceLifePropellerIceLoadCondition ?serviceLife ;
                               nltl:fixedPitchPropellerReversalLoadIncluded ?reversal ;
                               nltl:stoppedPropellerDraggingOutsideLoadModel ?dragging ;
                               nltl:radialIceEntryOutsideLoadModel ?radial .

                    FILTER (
                        datatype(?normal) = xsd:boolean &&
                        ?normal = true &&
                        datatype(?serviceLife) = xsd:boolean &&
                        ?serviceLife = true &&
                        datatype(?reversal) = xsd:boolean &&
                        ?reversal = true &&
                        datatype(?dragging) = xsd:boolean &&
                        ?dragging = true &&
                        datatype(?radial) = xsd:boolean &&
                        ?radial = true
                    )
                }
            }
        """
    ] .
<END_SHACL>
