@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF070Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "An applicable ship must identify main propulsion thrusters."
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:datatype xsd:string ;
                sh:in ( "azimuthing" "fixed" ) ;
                sh:message "An applicable main propulsion thruster must be azimuthing or fixed."
            ] ;
            sh:property [
                sh:path nltl:propellerIceInteractionLoad ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide a propeller/ice interaction load in kN."
            ] ;
            sh:property [
                sh:path nltl:thrusterBodyIceInteractionLoad ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide a thruster-body ice interaction load in kN."
            ] ;
            sh:property [
                sh:path nltl:localIcePressure ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide local ice pressure in MPa."
            ] ;
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:designCondition ;
                    sh:property [
                        sh:path nltl:normalOperationalConditions ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:message "The design condition must represent normal operational conditions."
                    ] ;
                    sh:property [
                        sh:path nltl:designConditionExpectedLoad ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:message "The design condition must provide the expected thruster-body load in kN."
                    ] ;
                    sh:property [
                        sh:path nltl:designConditionLocalStrengthCapacity ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:message "The design condition must provide externally assessed local-strength capacity in kN."
                    ]
                ] ;
                sh:message "The applicable ship must provide a design condition with the required thruster-body load and capacity."
            ] ;
            sh:sparql [
                sh:message "The local-strength capacity must be at least the applicable local ice pressure." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:localIcePressure ?pressureQuantity .
                        ?pressureQuantity qudt:numericValue ?pressure .
                        $this nltl:hasDesignCondition ?condition .
                        ?condition nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                        ?capacityQuantity qudt:numericValue ?capacity .
                        FILTER (?capacity < ?pressure)
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "Ships outside the azimuthing or fixed main-propulsion-thruster scope are not subject to this requirement." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:mainPropulsionThruster ?mainThruster .
                        $this nltl:thrusterType ?type .
                        FILTER (
                            ?mainThruster = false ||
                            ?type NOT IN ("azimuthing", "fixed")
                        )
                    }
                """
            ]
        ]
    ) .
