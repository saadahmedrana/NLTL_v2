<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_076_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in ( "CP" "FP" ) ;
        sh:message "A ship must declare propeller type CP or FP."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:minCount 4 ;
        sh:node [
            sh:class nltl:propellerBladeLoadCase ;
            sh:property [
                sh:path nltl:propellerBladeLoadCaseNumber ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:message "Each propeller blade load case must have an integer case number."
            ]
        ] ;
        sh:message "At least four propeller blade load cases must be represented."
    ] ;
    sh:or (
        gen:TRF_076_CPBranch
        gen:TRF_076_FPNonReversingBranch
        gen:TRF_076_FPReversingBranch
    ) .

gen:TRF_076_CPBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:hasValue "CP"
    ] ;
    sh:sparql [
        sh:message "For a CP propeller, propeller blade load cases 1 through 4 must be covered." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case1 .
                    ?case1 nltl:propellerBladeLoadCaseNumber ?number1 .
                    FILTER(?number1 = 1)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case2 .
                    ?case2 nltl:propellerBladeLoadCaseNumber ?number2 .
                    FILTER(?number2 = 2)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case3 .
                    ?case3 nltl:propellerBladeLoadCaseNumber ?number3 .
                    FILTER(?number3 = 3)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case4 .
                    ?case4 nltl:propellerBladeLoadCaseNumber ?number4 .
                    FILTER(?number4 = 4)
                }
            }
        """
    ] .
