<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-084_ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a CP propeller, pitchValueKnown must be present as a boolean." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propellerType ?propellerType .
    FILTER (?propellerType = "CP"^^xsd:string)
    FILTER NOT EXISTS {
        $this nltl:pitchValueKnown ?known .
        FILTER (datatype(?known) = xsd:boolean)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "For a CP propeller whose pitch value is not known, propellerPitchAtMcrBollard must equal 0.7 times propellerPitchAtMcrFreeRunning, with metre units." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propellerType ?propellerType ;
          nltl:pitchValueKnown ?known .
    FILTER (?propellerType = "CP"^^xsd:string)
    FILTER (?known = false)

    FILTER NOT EXISTS {
        $this nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
              nltl:propellerPitchAtMcrBollard ?bollardQuantity .
        ?freeQuantity qudt:numericValue ?freeValue ;
                      qudt:unit unit:M .
        ?bollardQuantity qudt:numericValue ?bollardValue ;
                         qudt:unit unit:M .
        FILTER (datatype(?freeValue) = xsd:decimal)
        FILTER (datatype(?bollardValue) = xsd:decimal)
        BIND ((7.0 / 10.0) * ?freeValue AS ?expectedValue)
        FILTER (
            ABS(?bollardValue - ?expectedValue)
            <= (ABS(?expectedValue) * 0.000001 + 0.000001)
        )
    }
}
"""
    ] .
<END_SHACL>
