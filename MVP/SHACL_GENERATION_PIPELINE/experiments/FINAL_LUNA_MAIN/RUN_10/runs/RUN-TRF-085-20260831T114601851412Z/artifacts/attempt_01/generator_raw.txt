<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the propeller diameter in metres."
    ] ;
    sh:or (
        gen:TRF085KnownSpeedBranch
        gen:TRF085TableLookupBranch
    ) .

gen:TRF085KnownSpeedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A known bollard propeller speed at maximum continuous rating must be provided in revolutions per minute."
    ] .

gen:TRF085TableLookupBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:tableLookupCase ;
            sh:property [
                sh:path nltl:tableLookupApplied ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:lookupInputQuantity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:lookupResultQuantity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:lookupSelectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ] ;
            sh:property [
                sh:path nltl:tableReference ;
                sh:minCount 1 ;
                sh:hasValue nltl:traficomTable6Dash9
            ]
        ] ;
        sh:message "When the bollard propeller speed is not known, a Table 6-9 lookup case with input, result, table reference, and selection evidence is required."
    ] ;
    sh:sparql [
        sh:message "The Table 6-9 lookup input must equal the ship's propeller diameter and its result must equal the selected bollard propeller speed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameter .
                ?diameter qudt:numericValue ?diameterValue .
                $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed .
                ?speed qudt:numericValue ?speedValue .
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:lookupInputQuantity ?input .
                ?input qudt:numericValue ?inputValue .
                ?case nltl:lookupResultQuantity ?result .
                ?result qudt:numericValue ?resultValue .
                FILTER (?diameterValue != ?inputValue || ?speedValue != ?resultValue)
            }
        """
    ] .
<END_SHACL>
