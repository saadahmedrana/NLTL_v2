    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;

        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;

            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:REV-PER-MIN
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;

                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:REV-PER-MIN
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:tableLookupCase ;
                    sh:property [
                        sh:path nltl:tableLookupApplied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:lookupInputQuantity ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:lookupResultQuantity ;

                            sh:property [
                                sh:path qudt:numericValue ;

                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [

                    ] ;
                    sh:property [
                        sh:path nltl:tableReference ;
                        sh:minCount 1 ;

                        sh:hasValue nltl:traficomTable6Dash9
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The Table 6-9 lookup input must equal propeller diameter and the lookup result must equal the reported bollard propeller speed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bollardPropellerSpeedKnown false .
                $this nltl:propellerDiameter ?diameterQuantity .
                ?diameterQuantity qudt:numericValue ?diameter ;
                    qudt:unit qudt:M .
                $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speedQuantity .
                ?speedQuantity qudt:numericValue ?speed ;
                    qudt:unit qudt:REV-PER-MIN .
                $this nltl:hasTableLookupCase ?case .
                ?case nltl:lookupInputQuantity ?inputQuantity ;
                    nltl:lookupResultQuantity ?resultQuantity .
                ?inputQuantity qudt:numericValue ?input .
                ?resultQuantity qudt:numericValue ?result .
                FILTER (?input != ?diameter || ?result != ?speed)
            }
        """
    ] .
