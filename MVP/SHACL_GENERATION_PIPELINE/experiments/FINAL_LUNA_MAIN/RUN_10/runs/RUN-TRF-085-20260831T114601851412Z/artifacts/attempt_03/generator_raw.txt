<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall state whether the bollard propeller speed is known."
    ] ;
    sh:or (
        gen:TRF085KnownSpeedBranch
        gen:TRF085TableLookupBranch
    ) .

gen:TRF085KnownSpeedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:hasValue true ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085SpeedQuantityShape ;
        sh:message "A known bollard propeller speed shall be reported."
    ] .

gen:TRF085TableLookupBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:hasValue false ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085DiameterQuantityShape ;
        sh:message "Propeller diameter is required for the Table 6-9 selection."
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085SpeedQuantityShape ;
        sh:message "The table-lookup alternative shall report the bollard propeller speed."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:TRF085TableLookupCaseShape ;
        sh:message "A Table 6-9 lookup case is required when the bollard speed is not known."
    ] ;
    sh:sparql [
        sh:message "The Table 6-9 lookup input shall equal propeller diameter and its result shall equal the reported bollard propeller speed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameter ;
                      nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed ;
                      nltl:hasTableLookupCase ?case .

                ?diameter qudt:numericValue ?diameterValue .
                ?speed qudt:numericValue ?speedValue .
                ?case nltl:lookupInputQuantity ?input ;
                      nltl:lookupResultQuantity ?result .
                ?input qudt:numericValue ?inputValue .
                ?result qudt:numericValue ?resultValue .

                FILTER (
                    (?inputValue != ?diameterValue) ||
                    (?resultValue != ?speedValue)
                )
            }
        """
    ] .

gen:TRF085TableLookupCaseShape a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The lookup case shall record that the table lookup was applied."
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash9 ;
        sh:message "The lookup case shall reference TRAFICOM Table 6-9."
    ] .

gen:TRF085DiameterQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:TRF085SpeedQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
    ] .

gen:TRF085LookupQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
