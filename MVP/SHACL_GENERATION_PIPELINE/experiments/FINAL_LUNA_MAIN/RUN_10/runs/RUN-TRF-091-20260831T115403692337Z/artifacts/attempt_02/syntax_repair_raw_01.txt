<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF091ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
                sh:minCount 1 ;
                sh:message "An ice-milling load case shall be used for propulsion-line strength evaluation."
            ] ;
            sh:not [
                sh:property [
                    sh:path nltl:loadCasePurpose ;
                    sh:hasValue nltl:stallingAnalysisPurposeValue
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each supplied ice-milling sequence case shall also be linked as the propulsion load case." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:hasIceMillingSequenceCase ?iceCase .
    FILTER NOT EXISTS {
        $this nltl:hasPropulsionLoadCase ?iceCase .
    }
}
"""
    ] .
</END_SHACL>
