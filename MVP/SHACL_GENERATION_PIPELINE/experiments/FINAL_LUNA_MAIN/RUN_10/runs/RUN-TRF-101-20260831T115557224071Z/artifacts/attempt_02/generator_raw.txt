<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF101Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Propeller diameter must be supplied as a metre quantity."
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade tip offset must be supplied as a metre quantity."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:node gs:TRF101LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "A load case with the required application radius and weakest-direction confirmation is required."
    ] ;
    sh:or (
        gs:TRF101ApprovedAnalysisBranch
        gs:TRF101CalculationBranch
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset ?tipQuantity ;
                      nltl:propellerDiameter ?diameterQuantity .
                ?tipQuantity qudt:numericValue ?tipOffset .
                ?diameterQuantity qudt:numericValue ?diameter .
                FILTER (
                    ?tipOffset > (0.10 * ?diameter)
                    && NOT EXISTS {
                        $this nltl:bladeFailureStatus true
                    }
                )
            }
        """
    ] .

gs:TRF101LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 0.8 ;
                sh:maxInclusive 0.8
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The blade failure load must be applied at a radius ratio of 0.8."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The blade failure load must be confirmed in the weakest direction."
    ] .

gs:TRF101ApprovedAnalysisBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:approvedNonlinearBladeStressAnalysis ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An approved nonlinear blade stress analysis is required when the direct calculation is not used."
    ] .

gs:TRF101CalculationBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The supplied reference strength and blade failure ultimate load must satisfy the declared calculation within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?cQuantity ;
                      nltl:bladeCylindricalRootRadius ?rQuantity ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tQuantity ;
                      nltl:bladeMinimumYieldStrength ?yieldQuantity ;
                      nltl:bladeUltimateTensileStrength ?ultimateQuantity ;
                      nltl:bladeReferenceStrength1 ?referenceQuantity ;
                      nltl:bladeFailureUltimateLoad ?loadQuantity ;
                      nltl:propellerDiameter ?diameterQuantity .
                ?cQuantity qudt:numericValue ?c .
                ?rQuantity qudt:numericValue ?r .
                ?tQuantity qudt:numericValue ?t .
                ?yieldQuantity qudt:numericValue ?yield .
                ?ultimateQuantity qudt:numericValue ?ultimate .
                ?referenceQuantity qudt:numericValue ?reference .
                ?loadQuantity qudt:numericValue ?load .
                ?diameterQuantity qudt:numericValue ?diameter .

                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedReference)
                BIND ((300.0 * ?c * ?t * ?t * ?reference) /
                      ((0.8 * ?diameter) - (2.0 * ?r)) AS ?expectedLoad)

                FILTER (
                    (
                        (?expectedReference >= 1.0
                         && (?reference < (?expectedReference - ((?expectedReference * 0.000001) + 0.000000001))
                             || ?reference > (?expectedReference + ((?expectedReference * 0.000001) + 0.000000001))))
                        ||
                        (?expectedReference <= -1.0
                         && (?reference < (?expectedReference - ((-1.0 * ?expectedReference * 0.000001) + 0.000000001))
                             || ?reference > (?expectedReference + ((-1.0 * ?expectedReference * 0.000001) + 0.000000001))))
                        ||
                        (?expectedReference > -1.0
                         && ?expectedReference < 1.0
                         && (?reference < (?expectedReference - 0.000001)
                             || ?reference > (?expectedReference + 0.000001)))
                    )
                    ||
                    (
                        (?expectedLoad >= 1.0
                         && (?load < (?expectedLoad - ((?expectedLoad * 0.000001) + 0.000000001))
                             || ?load > (?expectedLoad + ((?expectedLoad * 0.000001) + 0.000000001))))
                        ||
                        (?expectedLoad <= -1.0
                         && (?load < (?expectedLoad - ((-1.0 * ?expectedLoad * 0.000001) + 0.000000001))
                             || ?load > (?expectedLoad + ((-1.0 * ?expectedLoad * 0.000001) + 0.000000001))))
                        ||
                        (?expectedLoad > -1.0
                         && ?expectedLoad < 1.0
                         && (?load < (?expectedLoad - 0.000001)
                             || ?load > (?expectedLoad + 0.000001)))
                    )
                    ||
                    ((0.8 * ?diameter) - (2.0 * ?r) = 0.0)
                )
            }
        """
    ] .
<END_SHACL>
