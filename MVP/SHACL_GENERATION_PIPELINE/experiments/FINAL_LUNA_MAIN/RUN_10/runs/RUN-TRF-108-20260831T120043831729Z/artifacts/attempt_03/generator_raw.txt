<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must state whether the SN-curve is known."
    ] ;
    sh:or (
        gen:TRF108KnownShape
        gen:TRF108UnknownShape
    ) .

gen:TRF108KnownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The known SN-curve branch requires stressLifeCurveKnown to be true."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A known SN-curve must correspond to recorded blade material properties."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A known SN-curve must have one selected curve type."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Evidence must document correspondence between the selected SN-curve and blade material properties."
    ] .

gen:TRF108UnknownShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The unknown SN-curve branch requires stressLifeCurveKnown to be false."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:hasValue nltl:twoSlopeStressLifeCurve ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "If the SN-curve is unknown, the two-slope SN curve must be selected."
    ] .
<END_SHACL>
