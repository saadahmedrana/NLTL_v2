<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:message "The ship must provide one extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:ShaftLineLoadCaseTypeValueShape ;
                sh:message "Each propulsion load case must declare one applicable shaft line load case type."
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:UnitlessQuantityShape ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minInclusive 1.3
                            ]
                        ] ;
                        sh:message "Extreme operational load cases require a component yield safety factor of at least 1.3."
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase
                    ] ;
                    sh:property [
                        sh:path (
                            nltl:hasPropellerShaftLineComponent
                            nltl:hasComponent
                        ) ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:shipComponent ;
                            sh:property [
                                sh:path nltl:fatigueSafetyFactor ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node gen:UnitlessQuantityShape ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minInclusive 1.5
                                    ]
                                ] ;
                                sh:message "Fatigue load cases require a fatigue safety factor of at least 1.5 on each linked ship component."
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:UnitlessQuantityShape ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minInclusive 1.0
                            ]
                        ] ;
                        sh:message "Blade failure load cases require a component yield safety factor of at least 1.0."
                    ]
                ]
            )
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:node [
                    sh:class nltl:shipComponent ;
                    sh:property [
                        sh:path nltl:fatigueSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:UnitlessQuantityShape ;
                        sh:message "Each linked ship component must provide one fatigue safety factor."
                    ]
                ]
            ]
        ]
    ] .

gen:ShaftLineLoadCaseTypeValueShape
    a sh:NodeShape ;
    sh:class nltl:shaftLineLoadCaseTypeValue .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
