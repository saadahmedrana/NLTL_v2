<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ] ;
        sh:message "The ship must identify its propeller shaft line component."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class nltl:shaftLineLoadCaseTypeValue
                ]
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class nltl:shaftLineLoadCaseTypeValue
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.3 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class nltl:shaftLineLoadCaseTypeValue
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class nltl:shaftLineLoadCaseTypeValue
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.0 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            )
        ] ;
        sh:message "Each propulsion load case must be an applicable extreme-operational, fatigue, or blade-failure case with the required externally calculated result."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A fatigue load case requires a linked ship component with a fatigue safety factor of at least 1.5, expressed as a dimensionless quantity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor qudt:numericValue ?value .
                    ?factor qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?value) = xsd:decimal)
                    FILTER (?value >= 1.5)
                }
            }
        """
    ] .
<END_SHACL>
