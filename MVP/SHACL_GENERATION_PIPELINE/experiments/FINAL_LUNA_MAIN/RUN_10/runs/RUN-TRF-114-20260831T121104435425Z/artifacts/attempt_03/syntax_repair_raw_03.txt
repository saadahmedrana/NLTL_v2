<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-114-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide one ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node <urn:nltl:generated-shape:TRF-114-LoadCaseShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A non-hemispherical contact geometry case shall have an equivalent impact sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?value .
                    ?radius qudt:unit <http://qudt.org/vocab/unit/M> .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-114-LoadCaseShape>
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:message "The impact case shall reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:message "The impact case shall reference TRAFICOM Table 6-16 for the impact load case."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:thrusterBody ;
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:hemisphericalContactGeometry
            nltl:nonHemisphericalContactGeometry
        )
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case shall include evidence for the loaded area."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "A non-hemispherical case shall include contact-geometry correspondence evidence."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed thruster resistance capacity shall satisfy the impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:thrusterImpactCaseAppliesToBody ?body .
                ?body nltl:thrusterResistanceCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER ( ?capacityValue < ?demandValue )
            }
        """
    ] .
</END_SHACL>
