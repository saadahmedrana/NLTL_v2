<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF123ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node gen:TRF123OccasionalForceCaseShape
    ] .

gen:TRF123OccasionalForceCaseShape
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF123ExcludedCaseShape
        gen:TRF123AssessedCaseShape
    ) ;
    sh:sparql [
        sh:message "For every assessed component, occasional-force stress multiplied by the yield safety factor must not exceed the component material yield strength, allowing a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerBladeExcludedFromOccasionalForceScope false .
                $this nltl:occasionalForceCaseAssessedComponent ?component .
                $this nltl:componentYieldSafetyFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factor .

                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .

                ?component nltl:hasMaterialProperties ?material .
                ?material nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .

                FILTER (
                    (?stress * ?factor) >
                    (
                        ?yield
                        + 0.000001 * IF(
                            (?stress * ?factor) < 0,
                            - (?stress * ?factor),
                            (?stress * ?factor)
                        )
                        + 0.000000001
                    )
                )
            }
        """
    ] .

gen:TRF123ExcludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:hasValue true
    ] .

gen:TRF123AssessedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:minCount 1 ;
        sh:node gen:TRF123DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:minCount 1 ;
        sh:node gen:TRF123AssessedComponentShape
    ] .

gen:TRF123AssessedComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:minCount 1 ;
        sh:node gen:TRF123PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node gen:TRF123MaterialPropertiesShape
    ] .

gen:TRF123MaterialPropertiesShape
    a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:node gen:TRF123PressureQuantityShape
    ] .

gen:TRF123PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF123DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
