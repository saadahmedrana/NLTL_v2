    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "At least one inlet chest must satisfy the standard cooling-water chest arrangement and calculations." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:hasComponent ?ch .
                        ?ch a nltl:inletChest ;
                            nltl:seaInletLocation "near centreline and well aft if possible" ;
                            nltl:iceAccumulationSpaceAboveInlet true ;
                            nltl:fullCapacityDischargePipeConnected true ;
                            nltl:recommendedChestVolume ?rv ;
                            nltl:strainerOpenArea ?sa ;
                            nltl:inletPipeArea ?ia .

                        ?rv qudt:numericValue ?rvn ;
                            qudt:unit unit:M3 .
                        ?sa qudt:numericValue ?san ;
                            qudt:unit unit:M2 .
                        ?ia qudt:numericValue ?ian ;
                            qudt:unit unit:M2 .
                        $this nltl:totalRelevantEnginePower ?tp .
                        ?tp qudt:numericValue ?tpn ;
                            qudt:unit unit:KiloW .

                        FILTER (
                            ABS(?rvn - (?tpn / 750)) <=
                            (0.000001 * ABS(?tpn / 750) + 0.000000001)
                        )
                        FILTER (?san >= 4 * ?ian)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "When the standard volume and height requirements cannot be met, at least two alternating inlet chests must be provided." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    SELECT $this
                    WHERE {
                        $this nltl:hasComponent ?ch1 .
                        $this nltl:hasComponent ?ch2 .
                        ?ch1 a nltl:inletChest ;
                             nltl:alternatingCoolingWaterIntakeAndDischarge true .
                        ?ch2 a nltl:inletChest ;
                             nltl:alternatingCoolingWaterIntakeAndDischarge true .
                        FILTER (?ch1 != ?ch2)
                    }
                """
            ]
        ]
    ) .

<urn:nltl:generated-shape:TRF-130-CalculationShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each inlet chest must satisfy the declared volume approximation and strainer-area comparison." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?ch ;
                      nltl:totalRelevantEnginePower ?tp .
                ?ch a nltl:inletChest ;
                    nltl:recommendedChestVolume ?rv ;
                    nltl:strainerOpenArea ?sa ;
                    nltl:inletPipeArea ?ia .
                ?tp qudt:numericValue ?tpn ;
                    qudt:unit unit:KiloW .
                ?rv qudt:numericValue ?rvn ;
                    qudt:unit unit:M3 .
                ?sa qudt:numericValue ?san ;
                    qudt:unit unit:M2 .
                ?ia qudt:numericValue ?ian ;
                    qudt:unit unit:M2 .
                FILTER (
                    ABS(?rvn - (?tpn / 750)) >
                    (0.000001 * ABS(?tpn / 750) + 0.000000001)
                    || ?san < 4 * ?ian
                )
            }
        """
    ] .
