                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

        ]
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Recommended chest volume must approximately equal total relevant engine power divided by 750." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this ;
                      nltl:totalRelevantEnginePower ?totalPower .
                $this nltl:recommendedChestVolume ?recommended .
                ?totalPower qudt:numericValue ?totalValue .
                ?recommended qudt:numericValue ?recommendedValue .
                FILTER (
                    (?recommendedValue - (?totalValue / 750)) >
                    ((?totalValue / 750) * 0.000001 + 0.000000001)
                    ||
                    ((?totalValue / 750) - ?recommendedValue) >
                    ((?totalValue / 750) * 0.000001 + 0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Strainer open area must be at least four times the inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:strainerOpenArea ?openArea ;
                      nltl:inletPipeArea ?pipeArea .
                ?openArea qudt:numericValue ?openValue .
                ?pipeArea qudt:numericValue ?pipeValue .
                FILTER (?openValue < (4 * ?pipeValue))
            }
        """
    ] .
