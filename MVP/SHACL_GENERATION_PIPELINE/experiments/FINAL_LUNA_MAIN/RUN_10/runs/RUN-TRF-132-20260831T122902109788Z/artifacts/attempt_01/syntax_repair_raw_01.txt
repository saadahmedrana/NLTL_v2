<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:generatedShape:TRF132EngineOutput> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship ice class must be specified using a permitted ice-class value."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The engine output must be at least f1*f2*f3*(f4*Delta + P0), and must be at least 740 kW for ice classes IB and IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:engineOutputFormulaBasePowerP0 ?p0Node ;
                      nltl:engineOutputFormulaDisplacement ?deltaNode ;
                      nltl:engineOutputFormulaFactorF1 ?f1Node ;
                      nltl:engineOutputFormulaFactorF2 ?f2Node ;
                      nltl:engineOutputFormulaFactorF3 ?f3Node ;
                      nltl:engineOutputFormulaFactorF4 ?f4Node ;
                      nltl:maximumContinuousRatingPower ?outputNode .

                ?p0Node qudt:numericValue ?p0 .
                ?deltaNode qudt:numericValue ?delta .
                ?f1Node qudt:numericValue ?f1 .
                ?f2Node qudt:numericValue ?f2 .
                ?f3Node qudt:numericValue ?f3 .
                ?f4Node qudt:numericValue ?f4 .
                ?outputNode qudt:numericValue ?output .

                FILTER (
                    ?output < (?f1 * ?f2 * ?f3 * ((?f4 * ?delta) + ?p0))
                    ||
                    (
                        (
                            EXISTS {
                                $this nltl:iceClass nltl:iceClassIb .
                            }
                            ||
                            EXISTS {
                                $this nltl:iceClass nltl:iceClassIc .
                            }
                        )
                        && ?output < 740
                    )
                )
            }
        """
    ] .
