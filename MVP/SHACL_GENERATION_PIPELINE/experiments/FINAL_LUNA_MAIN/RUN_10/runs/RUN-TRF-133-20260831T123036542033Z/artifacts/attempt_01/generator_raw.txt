<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted Finnish-Swedish ice class."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Timber load line mark applicability must be explicitly recorded."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:node nltlgen:WarningTriangleShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one conforming warning triangle marking."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node nltlgen:IceClassDraughtMarkingTimberShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:hasValue false ;
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node nltlgen:IceClassDraughtMarkingLoadLineShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ]
    ) ;
    sh:message "The ice-class draught-mark reference must match the explicitly recorded timber load line applicability."

nltlgen:IceClassDraughtMarkingBaseShape
    a sh:NodeShape ;
    sh:class nltl:iceClassDraughtMarking ;
    sh:class nltl:hullStructure ;
    sh:property [
        sh:path nltl:draughtMarkAftOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "540"^^xsd:decimal ;
                sh:maxInclusive "540"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ice-class draught mark must be 540 mm abaft its applicable reference."
    ] ;
    sh:property [
        sh:path nltl:markingPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "5"^^xsd:decimal ;
                sh:maxInclusive "8"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The marking plate thickness must be between 5 and 8 mm."
    ] ;
    sh:property [
        sh:path nltl:markingWeldedToShipSide ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reflectingMarkingColour ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:redReflectingMarkingColour
            nltl:yellowReflectingMarkingColour
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The marking must have one permitted red or yellow reflecting colour."
    ] ;
    sh:property [
        sh:path nltl:markingPlainlyVisibleInIceConditions ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:letterDimensionsEqualLoadLineMark ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:IceClassDraughtMarkingLoadLineShape
    a sh:NodeShape ;
    sh:and (
        nltlgen:IceClassDraughtMarkingBaseShape
        [
            sh:property [
                sh:path nltl:draughtMarkAftReferencePoint ;
                sh:hasValue nltl:loadLineRingCentreReference ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .

nltlgen:IceClassDraughtMarkingTimberShape
    a sh:NodeShape ;
    sh:and (
        nltlgen:IceClassDraughtMarkingBaseShape
        [
            sh:property [
                sh:path nltl:draughtMarkAftReferencePoint ;
                sh:hasValue nltl:timberLoadLineVerticalReference ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .

    sh:property [
        sh:path nltl:verticalOffsetAboveSummerFreshWaterLoadLine ;
        sh:minCount 1 ;

            sh:property [
                sh:path qudt:numericValue ;

                sh:minInclusive "1000"^^xsd:decimal ;

            sh:property [

                sh:datatype xsd:decimal ;
                sh:minInclusive "300"^^xsd:decimal ;
                sh:maxInclusive "300"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1

                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The warning triangle sides must be 300 mm long."
    ] ;
    sh:property [
        sh:path nltl:markingPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "5"^^xsd:decimal ;
                sh:maxInclusive "8"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:markingWeldedToShipSide ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reflectingMarkingColour ;

            nltl:redReflectingMarkingColour
            nltl:yellowReflectingMarkingColour
        ) ;
        sh:minCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:markingPlainlyVisibleInIceConditions ;
        sh:datatype xsd:boolean ;
