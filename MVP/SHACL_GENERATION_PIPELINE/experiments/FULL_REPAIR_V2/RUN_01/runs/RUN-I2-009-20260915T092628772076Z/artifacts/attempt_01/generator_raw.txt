<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlg:QuantityValueShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must contain one decimal numeric value."
    ] .

nltlg:AverageIcePressureShape a sh:NodeShape ;
    sh:node nltlg:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Average ice pressure must use MPa."
    ] .

nltlg:AngleShape a sh:NodeShape ;
    sh:node nltlg:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The angle must use degrees."
    ] .

nltlg:AspectRatioShape a sh:NodeShape ;
    sh:node nltlg:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load patch aspect ratio must be dimensionless."
    ] ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 3.6 ;
        sh:maxInclusive 3.6 ;
        sh:message "The fixed load patch aspect ratio must be 3.6."
    ] .

nltlg:LengthShape a sh:NodeShape ;
    sh:node nltlg:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length must use metres."
    ] .

nltlg:HullAreaAssignmentShape a sh:NodeShape ;
    sh:class nltl:hullAreaAssignment ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:class nltl:hullAreaValue ;
        sh:hasValue nltl:otherIceStrengthenedArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assignment must identify the other ice-strengthened area."
    ] .

nltlg:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:node nltlg:HullAreaAssignmentShape ;
        sh:minCount 1 ;
        sh:message "A ship must provide a hull-area assignment for the other ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide its hull area."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node nltlg:AverageIcePressureShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide its average ice pressure."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node nltlg:AspectRatioShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must use the required fixed load patch aspect ratio."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node nltlg:LengthShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide the non-bow load patch height."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node nltlg:LengthShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide the non-bow load patch length."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node nltlg:AngleShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide the buttock angle at the stem."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node nltlg:AngleShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide the normal frame angle."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability must be explicitly recorded."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow-form applicability must be true exactly when the buttock angle is greater than 0 and less than 80 degrees and the normal frame angle is greater than 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability ?app .
                $this nltl:buttockAngle ?gammaNode .
                $this nltl:normalFrameAngle ?betaNode .
                ?gammaNode qudt:numericValue ?gamma .
                ?betaNode qudt:numericValue ?beta .
                FILTER (
                    (?app = true &&
                     NOT (?gamma > 0 && ?gamma < 80 && ?beta > 10))
                    ||
                    (?app = false &&
                     (?gamma > 0 && ?gamma < 80 && ?beta > 10))
                )
            }
        """
    ] .
<END_SHACL>
