<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-026-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-026-StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2-026-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural member must state whether it is continuous through its support."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural member must state whether it terminates within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-026-MemberSupportShape> ;
        sh:message "Each structural member must identify its member support."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The support condition must be fixed for continuous members, members with a connection bracket, members with demonstrated significant rotational restraint, and members terminating within an ice-strengthened area; otherwise it must be simple." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>

SELECT $this
WHERE {
  $this nltl:hasMemberSupport ?support .
  ?support nltl:frameSupportCondition ?condition .
  FILTER (
    (
      ?condition = nltl:fixedFrameSupport &&
      NOT EXISTS { $this nltl:continuousThroughSupport true } &&
      NOT EXISTS { ?support nltl:connectionBracketPresent true } &&
      NOT EXISTS { ?support nltl:demonstratedRotationalRestraint true }
    )
    ||
    (
      ?condition = nltl:simpleFrameSupport &&
      (
        EXISTS { $this nltl:continuousThroughSupport true } ||
        EXISTS { ?support nltl:connectionBracketPresent true } ||
        EXISTS { ?support nltl:demonstratedRotationalRestraint true }
      )
    )
    ||
    (
      ?condition != nltl:fixedFrameSupport &&
      ?condition != nltl:simpleFrameSupport
    )
    ||
    (
      EXISTS { $this nltl:terminatesWithinIceStrengthenedArea true } &&
      ?condition != nltl:fixedFrameSupport
    )
  )
}
"""
    ] .

<urn:nltl:generated-shape:I2-026-MemberSupportShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each member support must specify either fixed frame support or simple frame support."
    ] ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each member support must state whether a connection bracket is present."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each member support must state whether significant rotational restraint is demonstrated."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Evidence is required when significant rotational restraint is demonstrated."
            ]
        ]
    ) .

<END_SHACL>
