<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_StructuralMemberShape ;
        sh:message "Each represented structural member must provide applicability information and satisfy the side-structure load-case constraint when applicable."
    ] ;
    sh:property [
        sh:path nltl:frameProfileType
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:sparql [
                    sh:message "A non-side structural member must have a structural location other than sideStructureLocation." ;
                    sh:select """
                        PREFIX nltl: <https://w3id.org/nltl/vocab#>

                        SELECT $this
                        WHERE {
                            $this nltl:structuralLocation ?location .
                            FILTER (?location = nltl:sideStructureLocation)
                        }
                    """
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:sideStructureLocation
            ] ;
            sh:property [
                sh:path nltl:plasticStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:I2_034_KiloNewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:midspanPlasticCollapseLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:I2_034_KiloNewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node gen:I2_034_LoadCaseShape ;
                sh:message "Each represented side-structure load case must include the demand and plastic-strength quantities and must not exceed the plastic strength."
            ] ;
            sh:sparql [
                sh:message "Combined shear-and-bending demand must not exceed the member plastic strength or midspan plastic-collapse load." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:plasticStrength ?strengthQuantity .
                        ?strengthQuantity qudt:numericValue ?strengthValue ;
                                           qudt:unit unit:KiloN .

                        $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                        ?collapseQuantity qudt:numericValue ?collapseValue ;
                                           qudt:unit unit:KiloN .

                        $this nltl:hasStructuralMemberLoadCase ?loadCase .
                        ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                        ?demandQuantity qudt:numericValue ?demandValue ;
                                        qudt:unit unit:KiloN .

                        FILTER (
                            ?demandValue > ?strengthValue ||
                            ?demandValue > ?collapseValue
                        )
                    }
                """
            ]
        ]
    ) .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_034_KiloNewtonQuantityShape
    ] .

gen:I2_034_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .
<END_SHACL>
