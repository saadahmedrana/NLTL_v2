@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-055-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:MegaPA
            ]
        ] ;
        sh:message "A single design stress in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:MegaPA
            ]
        ] ;
        sh:message "A single permissible stress in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;

    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Design stress must not exceed permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:designStress ?design .
                ?design qudt:numericValue ?designValue ;
                        qudt:unit unit:MegaPA .
                $this nltl:permissibleStress ?permissible .
                ?permissible qudt:numericValue ?permissibleValue ;
                             qudt:unit unit:MegaPA .
                FILTER (?designValue > ?permissibleValue)
            }
        """
    ] .
