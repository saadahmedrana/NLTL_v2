<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-060-MemberShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member in way of a cut-out must not be unstable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                        nltl:memberInstability true .
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Structural stiffening must be recorded where stiffening is necessary." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true ;
                        nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (datatype(?stiffening) = xsd:string)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-MemberShape>
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasCutout ;
                sh:class nltl:cutout ;
                sh:minCount 1 ;
                sh:message "An applicable member must identify its cut-out."
            ] ;
            sh:property [
                sh:path nltl:memberInstability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false ;
                sh:message "An applicable member must be recorded as not unstable."
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:message "An applicable member must state whether stiffening is necessary."
            ]
        ]
    ) .

<END_SHACL>
