<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gs:CalculationCaseShape ;
        sh:message "Each represented calculation case must follow the prescribed analytical procedure."
    ] .

gs:CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gs:HullStructureShape ;
        sh:minCount 1 ;
        sh:message "A calculation case must identify its assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationCaseAssessedHullStructure ;
                sh:node gs:ApplicableHullStructureShape
            ] ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:prescribedAnalyticalProcedure ) ;
                sh:minCount 1 ;
                sh:message "Direct calculation must not replace the prescribed analytical procedure for shell-plating or local-frame requirements governed by I2.4, I2.6, or I2.7."
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationCaseAssessedHullStructure ;
                sh:not [
                    sh:node gs:ApplicableHullStructureShape
                ]
            ]
        ]
    ) .

gs:ApplicableHullStructureShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:plating
        ]
        [
            sh:class nltl:structuralMember
        ]
    ) .

gs:HullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node gs:ThicknessQuantityShape ;
        sh:minCount 1 ;
        sh:message "The assessed hull structure must provide its thickness."
    ] .

gs:ThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
