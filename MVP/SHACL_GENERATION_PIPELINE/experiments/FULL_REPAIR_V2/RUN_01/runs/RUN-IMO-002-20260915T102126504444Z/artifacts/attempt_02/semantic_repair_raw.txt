<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO002ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition." ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its Polar Code category." ;
    ] ;
    sh:sparql [
        sh:message "A ship designed to operate in at least thin first-year ice with possible old-ice inclusions, and not declared as category A, must be category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?designIceCondition .
                VALUES ?designIceCondition {
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions
                }
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryA .
                }
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryB .
                }
            }
        """ ;
    ] .
<END_SHACL>
