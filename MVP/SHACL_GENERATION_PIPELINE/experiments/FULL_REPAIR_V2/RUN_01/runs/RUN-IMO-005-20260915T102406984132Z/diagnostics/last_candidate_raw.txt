<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-005-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#string>
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype <http://www.w3.org/2001/XMLSchema#boolean>
    ] ;
    sh:sparql [
        sh:message "An ice condition of 'ice free waters' requires totalIcePresent to be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (STR(?condition) = "ice free waters")
                FILTER NOT EXISTS {
                    $this nltl:totalIcePresent false .
                }
            }
        """
    ] .
<END_SHACL>
