<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_007_OpenWaterDefinitionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "A ship shall record an ice condition classification."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ice condition is open water, sea ice concentration shall be a decimal less than 1 with unitless units, and ice of land origin shall be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?value ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value < 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        FILTER NOT EXISTS {
                            ?concentration qudt:numericValue ?value .
                            FILTER (
                                datatype(?value) = xsd:decimal &&
                                ?value < 1
                            )
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        FILTER NOT EXISTS {
                            ?concentration qudt:unit qudt:UNITLESS .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:unit ?unit .
                        FILTER (?unit != qudt:UNITLESS)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent ?present .
                        FILTER (
                            datatype(?present) = xsd:boolean &&
                            ?present = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?present .
                        FILTER NOT EXISTS {
                            FILTER (
                                datatype(?present) = xsd:boolean &&
                                ?present = false
                            )
                        }
                    }
                )
            }
        """
    ] .

<END_SHACL>
