<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node [
            sh:class nltl:iceTypeValue
        ] ;
        sh:message "A ship must have one ice type classification."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Any recorded ice thickness must be a single metre-valued decimal quantity."
    ] ;
    sh:sparql [
        sh:message "For thin first-year ice, ice thickness must be between 0.3 m and 0.7 m inclusive." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceTypeClassification nltl:thinFirstYearIce .
    FILTER NOT EXISTS {
        $this nltl:iceThickness ?thickness .
        ?thickness a qudt:QuantityValue ;
                   qudt:numericValue ?value ;
                   qudt:unit unit:M .
        FILTER (
            datatype(?value) = xsd:decimal &&
            ?value >= "0.3"^^xsd:decimal &&
            ?value <= "0.7"^^xsd:decimal
        )
    }
}
"""
    ] .
<END_SHACL>
