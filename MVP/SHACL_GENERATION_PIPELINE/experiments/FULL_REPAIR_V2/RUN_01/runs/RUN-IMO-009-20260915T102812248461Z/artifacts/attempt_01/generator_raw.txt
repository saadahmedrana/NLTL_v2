<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO009BergyWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An ice condition is required to determine whether the bergy-waters limits apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, land-origin ice concentration must be a dimensionless quantity less than 1/10." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER (STR(?condition) = "bergy waters")
    FILTER NOT EXISTS {
        $this nltl:landOriginIceConcentration ?quantity .
        ?quantity a qudt:QuantityValue ;
            qudt:numericValue ?value ;
            qudt:unit unit:UNITLESS .
        FILTER (datatype(?value) = xsd:decimal && ?value < 1)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, total ice concentration must be a dimensionless quantity no greater than 1/10." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER (STR(?condition) = "bergy waters")
    FILTER NOT EXISTS {
        $this nltl:totalIceConcentration ?quantity .
        ?quantity a qudt:QuantityValue ;
            qudt:numericValue ?value ;
            qudt:unit unit:UNITLESS .
        FILTER (datatype(?value) = xsd:decimal && ?value <= 1)
    }
}
"""
    ] .
<END_SHACL>
