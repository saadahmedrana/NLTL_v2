<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO032ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO032CategoryABBranch
        gen:IMO032CategoryCStrengthenedBranch
        gen:IMO032CategoryCNotStrengthenedBranch
    ) .

gen:IMO032CategoryABBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1 ;
        sh:message "A category A or B ship shall identify its ship category."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO032ApprovalABShape ;
        sh:message "A category A or B ship shall have an approved scantling approval record."
    ] .

gen:IMO032CategoryCStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
        sh:message "The applicable ship category shall be category C."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "An applicable category C ship shall identify that it is ice strengthened."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO032ApprovalCShape ;
        sh:message "An ice-strengthened category C ship shall have an approval record covering its operating ice conditions."
    ] .

gen:IMO032CategoryCNotStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
    ] .

gen:IMO032ApprovalABShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status shall be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The scantling approval shall be by the Administration or an accepted recognized organization."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval standard shall be acceptable to the Organization or offer an equivalent level of safety."
    ] .

gen:IMO032ApprovalCShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status shall be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The scantling approval shall be by the Administration or an accepted recognized organization."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "acceptable standard adequate for operating ice type and concentration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval standard shall be adequate for the operating ice type and concentration."
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The category C approval record shall specify the operating ice type."
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IMO032IceConcentrationQuantityShape ;
        sh:message "The category C approval record shall specify the operating ice concentration as a unitless quantity."
    ] .

gen:IMO032IceConcentrationQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Operating ice concentration shall have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Operating ice concentration shall use the unitless unit."
    ] .
<END_SHACL>
