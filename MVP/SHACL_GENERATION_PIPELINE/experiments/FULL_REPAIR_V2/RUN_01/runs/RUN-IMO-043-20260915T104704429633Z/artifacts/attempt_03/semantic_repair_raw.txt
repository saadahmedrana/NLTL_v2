<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO043_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:class nltl:machineryInstallationItem ;
        sh:node gen:IMO043_MachineryInstallationShape ;
        sh:message "Each linked machinery installation must satisfy the applicable hazard coverage constraints."
    ] .

gen:IMO043_MachineryInstallationShape
    a sh:NodeShape ;
    sh:node gen:IMO043_HazardCoverageShape ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:class nltl:associatedEquipmentItem ;
        sh:node gen:IMO043_AssociatedEquipmentShape ;
        sh:message "The machinery installation must identify associated equipment when such equipment is represented."
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:message "The machinery installation must provide hazard-protection evidence when an applicable hazard is declared."
    ] .

gen:IMO043_AssociatedEquipmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasAssociatedEquipment
        ] ;
        sh:class nltl:machineryInstallationItem ;
        sh:node gen:IMO043_HazardCoverageShape ;
        sh:message "Each associated equipment item must satisfy the applicable hazard coverage constraints through its owning machinery installation."
    ] .

gen:IMO043_HazardCoverageShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:iceAccretionHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when ice accretion is an applicable hazard." ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:snowAccumulationHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when snow accumulation is an applicable hazard." ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:seawaterIceIngestionHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when seawater ice ingestion is an applicable hazard." ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:liquidFreezingOrViscosityIncreaseHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when liquid freezing or viscosity increase is an applicable hazard." ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:seawaterIntakeTemperatureHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when seawater intake temperature is an applicable hazard." ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasApplicableHazard ;
                    sh:hasValue nltl:snowIngestionHazard
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasHazardProtectionEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact
            ]
        ]
    ) ;
    sh:message "Protection coverage evidence is required when snow ingestion is an applicable hazard." .
<END_SHACL>
