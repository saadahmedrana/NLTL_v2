<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO047ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:minCount 1 ;
        sh:class nltl:essentialEngine ;
        sh:message "The ship must identify at least one essential engine."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:node gen:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The combustion-air temperature must be supplied in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:node gen:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:node gen:IMO047TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combustion-air temperature must be within the engine manufacturer's minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:combustionAirTemperatureControlPresent true .
                    $this nltl:combustionAirTemperature ?temperatureQuantity .
                    $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                    $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .

                    ?temperatureQuantity
                        qudt:numericValue ?temperatureValue ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                    ?minimumQuantity
                        qudt:numericValue ?minimumValue ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                    ?maximumQuantity
                        qudt:numericValue ?maximumValue ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                    FILTER (
                        datatype(?temperatureValue) = xsd:decimal &&
                        datatype(?minimumValue) = xsd:decimal &&
                        datatype(?maximumValue) = xsd:decimal &&
                        ?temperatureValue >= ?minimumValue &&
                        ?temperatureValue <= ?maximumValue
                    )
                }
            }
        """
    ] .

gen:IMO047TemperatureQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
