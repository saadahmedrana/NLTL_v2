<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_057_StaticShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Every linked fire pump must have a containing compartment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER NOT EXISTS {
                    ?pump nltl:hasContainingCompartment ?compartment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each containing compartment of a linked fire pump must be maintained above 0 degrees Celsius." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                ?pump nltl:hasContainingCompartment ?compartment .
                FILTER NOT EXISTS {
                    ?compartment a nltl:compartment ;
                        nltl:maintainedTemperature ?quantity .
                    ?quantity a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:DEG_C .
                    FILTER (
                        ?value > 0
                    )
                }
            }
        """
    ] .
<END_SHACL>
