<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-064> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "The ship must record the number of persons on board."
    ] ;
    sh:property [
        sh:path nltl:assignedThermalProtection ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Thermal-protection assignment must be recorded for every person on board."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every person on board must have an assigned thermal-protection equipment assignment." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:personOnBoard ?personCount .
                OPTIONAL {
                    $this nltl:assignedThermalProtection ?assignment .
                    FILTER ( ?assignment = true )
                }
            }
            GROUP BY $this ?personCount
            HAVING ( COUNT(?assignment) < ?personCount )
        """
    ] ;
    sh:property [
        sh:path nltl:intendedVoyage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The intended voyage must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:anticipatedCold ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated cold must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:anticipatedWind ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated wind must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:immersionPotential ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The applicable potential for immersion in polar water must be recorded."
    ] ;
    sh:property [
        sh:path nltl:thermalProtectionApprovalStatus ;
        sh:minCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:class nltl:evidenceState ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:intendedVoyage ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "The approval evidence must cover the intended voyage."
            ] ;
            sh:property [
                sh:path nltl:anticipatedCold ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "The approval evidence must cover anticipated cold."
            ] ;
            sh:property [
                sh:path nltl:anticipatedWind ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:message "The approval evidence must cover anticipated wind."
            ] ;
            sh:property [
                sh:path nltl:immersionPotential ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:message "The approval evidence must cover applicable polar-water immersion."
            ]
        ] ;
        sh:message "Thermal-protection adequacy approval must have approved status and cover the intended voyage, anticipated cold and wind, and applicable polar-water immersion."
    ] .
<END_SHACL>
