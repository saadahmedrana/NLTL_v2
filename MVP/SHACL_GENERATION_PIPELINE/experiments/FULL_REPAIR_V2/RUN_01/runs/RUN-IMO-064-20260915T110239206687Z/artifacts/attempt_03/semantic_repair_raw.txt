<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-064> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "The ship must record the number of persons on board."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every person on board must have an assigned thermal-protection equipment assignment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:personOnBoard ?personCount .

                {
                    SELECT $this (COUNT(DISTINCT ?person) AS ?assignedPersonCount)
                    WHERE {
                        ?assignment a rdf:Statement ;
                                    rdf:predicate nltl:assignedThermalProtection ;
                                    rdf:object true ;
                                    rdf:subject ?person .
                    }
                    GROUP BY $this
                }

                FILTER (?assignedPersonCount < ?personCount)
            }
        """
    ] ;
    sh:property [
        sh:path nltl:intendedVoyage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The intended voyage must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:anticipatedCold ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated cold must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:anticipatedWind ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Anticipated wind must be recorded for thermal-protection adequacy."
    ] ;
    sh:property [
        sh:path nltl:immersionPotential ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The applicable potential for immersion in polar water must be recorded."
    ] ;
    sh:property [
        sh:path nltl:thermalProtectionApprovalStatus ;
        sh:minCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:class nltl:evidenceState ;
        sh:message "Thermal-protection adequacy approval must have approved status and cover the intended voyage, anticipated cold and wind, and applicable polar-water immersion."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The approved thermal-protection adequacy evidence must be ship-specific and cover the ship's intended voyage, anticipated cold and wind, and applicable polar-water immersion." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:thermalProtectionApprovalStatus nltl:evidenceStateApproved ;
                      nltl:intendedVoyage ?shipVoyage ;
                      nltl:anticipatedCold ?shipCold ;
                      nltl:anticipatedWind ?shipWind ;
                      nltl:immersionPotential ?shipImmersion .

                FILTER NOT EXISTS {
                    ?evidence a rdf:Statement ;
                              rdf:subject $this ;
                              rdf:predicate nltl:thermalProtectionApprovalStatus ;
                              rdf:object nltl:evidenceStateApproved ;
                              nltl:intendedVoyage ?evidenceVoyage ;
                              nltl:anticipatedCold ?evidenceCold ;
                              nltl:anticipatedWind ?evidenceWind ;
                              nltl:immersionPotential ?evidenceImmersion .

                    FILTER (
                        ?evidenceVoyage = ?shipVoyage &&
                        ?evidenceCold = ?shipCold &&
                        ?evidenceWind = ?shipWind &&
                        ?evidenceImmersion = ?shipImmersion
                    )
                }
            }
        """
    ] .
<END_SHACL>
