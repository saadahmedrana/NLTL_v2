@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_068_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's ice-covered-water operating condition must be recorded as a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether direct evacuation onto ice is applicable must be recorded as a single boolean value."
    ] ;
    sh:or (
        gen:IMO_068_IceCoveredWatersBranch
        gen:IMO_068_DirectOntoIceBranch
        gen:IMO_068_NeitherApplicableBranch
    ) .

gen:IMO_068_IceCoveredWatersBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:safeEvacuationMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Safe evacuation means must be present when operating in ice-covered waters."
    ] ;
    sh:property [
        sh:path nltl:survivalEquipmentDeploymentApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Survival equipment deployment must have approved status when operating in ice-covered waters."
    ] .

gen:IMO_068_DirectOntoIceBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:safeEvacuationMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Safe evacuation means must be present when direct evacuation onto ice is applicable."
    ] ;
    sh:property [
        sh:path nltl:survivalEquipmentDeploymentApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Survival equipment deployment must have approved status when direct evacuation onto ice is applicable."
    ] .

gen:IMO_068_NeitherApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:hasValue false
    ] .
