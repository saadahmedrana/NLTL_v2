<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_083_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_083_ApplicableBranch
        gen:IMO_083_NotApplicableBranch
    ) .

gen:IMO_083_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Ice-accretion applicability must be recorded as true for the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:hasRequiredNavigationOrCommunicationAntenna ;
        sh:node gen:IMO_083_AntennaShape ;
        sh:message "Each required navigation or communication antenna must provide ice-prevention evidence when ice accretion is likely."
    ] .

gen:IMO_083_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "Ice-accretion applicability must be recorded as false for the non-applicable branch."
    ] .

gen:IMO_083_AntennaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:antennaIceAccumulationPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "A required navigation or communication antenna must have means to prevent ice accumulation."
    ] .
<END_SHACL>
