<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:IMO_099_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall have a battery life management procedure."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery life management procedure shall be implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Communication device availability shall be recorded in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The maximum expected rescue time shall be recorded in days."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Communication equipment availability shall be at least the maximum expected time of rescue." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availabilityNode .
                ?availabilityNode qudt:numericValue ?availability .
                $this nltl:maximumExpectedRescueTime ?rescueNode .
                ?rescueNode qudt:numericValue ?rescueTime .
                FILTER (?availability < ?rescueTime)
            }
        """
    ] .
<END_SHACL>
