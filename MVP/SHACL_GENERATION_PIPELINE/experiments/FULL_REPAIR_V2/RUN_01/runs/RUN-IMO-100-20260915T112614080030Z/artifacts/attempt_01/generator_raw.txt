<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_100_VoyagePlanHazardCoverageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:voyagePlanPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that a voyage plan is present."
    ] ;
    sh:property [
        sh:path nltl:identifiedIntendedVoyageHazard ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:voyagePlanHazardCoverage ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every identified intended voyage hazard must be addressed by the voyage plan." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:identifiedIntendedVoyageHazard ?hazard .
                FILTER NOT EXISTS {
                    $this nltl:voyagePlanHazardCoverage true .
                }
            }
        """
    ] .
<END_SHACL>
