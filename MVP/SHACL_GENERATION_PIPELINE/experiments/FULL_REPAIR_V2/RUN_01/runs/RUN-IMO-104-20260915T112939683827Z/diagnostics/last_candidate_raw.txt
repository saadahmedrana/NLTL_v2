<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-104-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization status."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:node <urn:nltl:generated-shape:IMO-104-CrewMemberShape>
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node <urn:nltl:generated-shape:IMO-104-CrewMemberShape>
    ] ;
    sh:sparql [
        sh:message "Each crew member shall have a familiarization record for every PWOM procedure or equipment item relevant to the crew member's assigned duties." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCrewMemberInventory ?crewMember .
                ?crewMember nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crewMember nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-104-CrewMemberShape>
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-AssignedDutyShape> ;
        sh:message "Each crew member shall have an assigned duty from which relevant PWOM items can be identified."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-FamiliarizationRecordShape> ;
        sh:message "Each crew member shall have a PWOM familiarization record."
    ] .

<urn:nltl:generated-shape:IMO-104-AssignedDutyShape>
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-ManualItemShape> ;
        sh:message "Each assigned duty shall identify the relevant PWOM procedure or equipment items."
    ] .

<urn:nltl:generated-shape:IMO-104-FamiliarizationRecordShape>
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-ManualItemShape> ;
        sh:message "Each familiarization record shall identify the PWOM item covered."
    ] .

<urn:nltl:generated-shape:IMO-104-ManualItemShape>
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .
<END_SHACL>
