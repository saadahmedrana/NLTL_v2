<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_106_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:dischargedMaterialTypeValue ;
        sh:message "A ship must provide one discharged material classification."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean ballast or segregated ballast discharge, paragraph 1.1.1 must be recorded as not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material = nltl:cleanBallastMaterial ||
                    ?material = nltl:segregatedBallastMaterial
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                    FILTER (?applicable = false)
                }
            }
        """
    ] .
<END_SHACL>
