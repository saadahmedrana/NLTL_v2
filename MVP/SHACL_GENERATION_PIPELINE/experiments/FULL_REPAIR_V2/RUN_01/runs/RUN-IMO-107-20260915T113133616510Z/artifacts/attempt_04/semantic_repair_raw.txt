<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO107PolarOperationCoverageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:shipOperatesInPolarWaters ;
                    sh:hasValue true
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipOperatesInPolarWaters ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "The ship's polar-water applicability must be explicitly true when polar-operation coverage is required."
            ] ;
            sh:property [
                sh:path nltl:oilRecordBookPolarCoverage ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "When the ship operates in polar waters, polar operation must be covered in the applicable Oil Record Book."
            ] ;
            sh:property [
                sh:path nltl:manualPolarCoverage ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "When the ship operates in polar waters, polar operation must be covered in the applicable manuals."
            ] ;
            sh:property [
                sh:path nltl:pollutionEmergencyPlanPolarCoverage ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "When the ship operates in polar waters, polar operation must be covered in the applicable oil or marine pollution emergency plan."
            ]
        ]
    ) .
<END_SHACL>
