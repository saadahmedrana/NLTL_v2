<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110-StaticShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one permitted polar ship category."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one ship type."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one construction date."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide one deadweight quantity in metric tonnes."
    ] ;
    sh:sparql [
        sh:message "An applicable Category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 tonnes must have both specified cargo-tank protection statuses compliant." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?constructionDate ;
                      a nltl:oilTanker .

                ?deadweight qudt:numericValue ?deadweightValue ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?shipType = "oil_tanker"^^xsd:string
                    && ?deadweightValue < 5000
                    && ?constructionDate >= "2017-01-01"^^xsd:date
                )

                FILTER NOT EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true .
                }

                FILTER NOT EXISTS {
                    $this nltl:wingTankProtectionStatus true .
                }
            }
        """
    ] .
</END_SHACL>
