<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_111_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:message "The ship category must be Category A, B, or C."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Category A or B ships must have a construction date."
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, each specified oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:tankType ?tankType .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (
                    ?tankType = "oil_residue_tank" ||
                    ?tankType = "oily_bilge_water_holding_tank"
                )

                FILTER NOT EXISTS {
                    $this nltl:individualTankCapacity ?capacity ;
                          nltl:tankOuterShellSeparation ?separation .

                    ?capacity qudt:numericValue ?capacityValue ;
                              qudt:unit unit:M3 .
                    ?separation qudt:numericValue ?separationValue ;
                                 qudt:unit unit:M .

                    FILTER (?capacityValue > "30"^^xsd:decimal)
                    FILTER (?separationValue >= "0.76"^^xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
