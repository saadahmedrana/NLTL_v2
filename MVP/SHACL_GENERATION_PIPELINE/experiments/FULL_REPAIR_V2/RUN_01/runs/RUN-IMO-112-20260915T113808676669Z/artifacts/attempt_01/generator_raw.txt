<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO112Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must specify its operating area to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:or (
        gen:IMO112NonArcticBranch
        gen:IMO112ArcticBranch
    ) .

gen:IMO112NonArcticBranch a sh:NodeShape ;
    sh:sparql [
        sh:message "The NLS discharge prohibition applies when the operating area is Arctic waters." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters"^^xsd:string)
            }
        """
    ] .

gen:IMO112ArcticBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "Discharge of noxious liquid substances or mixtures containing them must be recorded as false in Arctic waters."
    ] .
<END_SHACL>
