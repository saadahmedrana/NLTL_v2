<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When operating area is Arctic_waters and food waste is discharged to sea, the nearest land, ice-shelf, or fast-ice distance must be at least 12 nautical miles, and a recorded distance from an area with ice concentration above one tenth must be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .
                FILTER (
                    STR(?area) = "Arctic_waters" &&
                    ?discharge = true
                )
                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest ;
                          nltl:nearbyIceConcentration ?nearby ;
                          nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recorded .

                    ?nearest a qudt:QuantityValue ;
                             qudt:numericValue ?nearestValue ;
                             qudt:unit unit:MI_N .
                    ?nearby a qudt:QuantityValue ;
                            qudt:numericValue ?nearbyValue ;
                            qudt:unit unit:UNITLESS .
                    ?recorded a qudt:QuantityValue ;
                             qudt:numericValue ?recordedValue ;
                             qudt:unit unit:MI_N .

                    FILTER (
                        datatype(?nearestValue) = xsd:decimal &&
                        datatype(?nearbyValue) = xsd:decimal &&
                        datatype(?recordedValue) = xsd:decimal &&
                        ?nearestValue >= 12 &&
                        ?nearbyValue > 1
                    )
                }
            }
        """
    ] .
<END_SHACL>
