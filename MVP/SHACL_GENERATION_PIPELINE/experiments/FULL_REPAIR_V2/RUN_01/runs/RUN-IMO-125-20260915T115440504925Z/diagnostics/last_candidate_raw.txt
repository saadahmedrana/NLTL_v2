<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO125ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO125NonPolarBranch
        gen:IMO125PolarBranch
    ) ;
    sh:message "The ship must identify whether it operates in polar waters and, when applicable, record polar-operation coverage in the Garbage Record Book, Garbage Management Plan, and garbage placards." .

gen:IMO125NonPolarBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:message "shipOperatesInPolarWaters must not be recorded as true for the non-applicable branch."
    ] ;
    sh:not [
        a sh:NodeShape ;
        sh:property [
            sh:path nltl:shipOperatesInPolarWaters ;
            sh:datatype xsd:boolean ;
            sh:hasValue true
        ]
    ] .

gen:IMO125PolarBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "shipOperatesInPolarWaters must be recorded as true for the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:garbageRecordBookPolarCoverage ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Polar-operation coverage must be recorded in the applicable Garbage Record Book."
    ] ;
    sh:property [
        sh:path nltl:garbageManagementPlanPolarCoverage ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Polar-operation coverage must be recorded in the applicable Garbage Management Plan."
    ] ;
    sh:property [
        sh:path nltl:garbagePlacardPolarCoverage ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Polar-operation coverage must be recorded in the applicable garbage placards."
    ] .
<END_SHACL>
