<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-002-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship type is required to determine Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Chapter 9-1 applicability must be recorded as a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Chapter 9-1 applicability must equal the applicable branch for the ship type and required dimensions, tonnage, and trade status." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipType ?type .
    $this nltl:polarCodeChapter9Dash1Applicable ?app .
    FILTER (
        (
            ?type = "fishing_vessel" &&
            (
                !EXISTS {
                    $this nltl:lengthOverall ?lengthValue .
                    ?lengthValue qudt:numericValue ?lengthNumber ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?lengthNumber) = xsd:decimal)
                }
                ||
                (
                    EXISTS {
                        $this nltl:lengthOverall ?lengthValue2 .
                        ?lengthValue2 qudt:numericValue ?lengthNumber2 ;
                                   qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?lengthNumber2) = xsd:decimal)
                    }
                    &&
                    (
                        (
                            EXISTS {
                                $this nltl:lengthOverall ?lengthValue3 .
                                ?lengthValue3 qudt:numericValue ?lengthNumber3 ;
                                           qudt:unit <http://qudt.org/vocab/unit/M> .
                                FILTER (datatype(?lengthNumber3) = xsd:decimal && ?lengthNumber3 >= 24)
                            }
                            && ?app != true
                        )
                        ||
                        (
                            !EXISTS {
                                $this nltl:lengthOverall ?lengthValue4 .
                                ?lengthValue4 qudt:numericValue ?lengthNumber4 ;
                                           qudt:unit <http://qudt.org/vocab/unit/M> .
                                FILTER (datatype(?lengthNumber4) = xsd:decimal && ?lengthNumber4 >= 24)
                            }
                            && ?app != false
                        )
                    )
                )
            )
        )
        ||
        (
            ?type = "pleasure_yacht" &&
            (
                !EXISTS {
                    $this nltl:grossTonnage ?grossValue .
                    ?grossValue qudt:numericValue ?grossNumber ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?grossNumber) = xsd:decimal)
                }
                ||
                !EXISTS {
                    $this nltl:engagedInTrade ?trade .
                    FILTER (datatype(?trade) = xsd:boolean)
                }
                ||
                (
                    EXISTS {
                        $this nltl:grossTonnage ?grossValue2 .
                        ?grossValue2 qudt:numericValue ?grossNumber2 ;
                                     qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?grossNumber2) = xsd:decimal)
                    }
                    &&
                    EXISTS {
                        $this nltl:engagedInTrade ?trade2 .
                        FILTER (datatype(?trade2) = xsd:boolean)
                    }
                    &&
                    (
                        (
                            EXISTS {
                                $this nltl:grossTonnage ?grossValue3 .
                                ?grossValue3 qudt:numericValue ?grossNumber3 ;
                                             qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                FILTER (datatype(?grossNumber3) = xsd:decimal && ?grossNumber3 >= 300)
                            }
                            &&
                            EXISTS {
                                $this nltl:engagedInTrade ?trade3 .
                                FILTER (datatype(?trade3) = xsd:boolean && ?trade3 = false)
                            }
                            && ?app != true
                        )
                        ||
                        (
                            !(
                                EXISTS {
                                    $this nltl:grossTonnage ?grossValue4 .
                                    ?grossValue4 qudt:numericValue ?grossNumber4 ;
                                                 qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                    FILTER (datatype(?grossNumber4) = xsd:decimal && ?grossNumber4 >= 300)
                                }
                                &&
                                EXISTS {
                                    $this nltl:engagedInTrade ?trade4 .
                                    FILTER (datatype(?trade4) = xsd:boolean && ?trade4 = false)
                                }
                            )
                            && ?app != false
                        )
                    )
                )
            )
        )
        ||
        (
            ?type = "cargo_ship" &&
            (
                !EXISTS {
                    $this nltl:grossTonnage ?cargoGrossValue .
                    ?cargoGrossValue qudt:numericValue ?cargoGrossNumber ;
                                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (datatype(?cargoGrossNumber) = xsd:decimal)
                }
                ||
                (
                    EXISTS {
                        $this nltl:grossTonnage ?cargoGrossValue2 .
                        ?cargoGrossValue2 qudt:numericValue ?cargoGrossNumber2 ;
                                       qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?cargoGrossNumber2) = xsd:decimal)
                    }
                    &&
                    (
                        (
                            EXISTS {
                                $this nltl:grossTonnage ?cargoGrossValue3 .
                                ?cargoGrossValue3 qudt:numericValue ?cargoGrossNumber3 ;
                                                   qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                FILTER (datatype(?cargoGrossNumber3) = xsd:decimal &&
                                        ?cargoGrossNumber3 >= 300 &&
                                        ?cargoGrossNumber3 < 500)
                            }
                            && ?app != true
                        )
                        ||
                        (
                            !EXISTS {
                                $this nltl:grossTonnage ?cargoGrossValue4 .
                                ?cargoGrossValue4 qudt:numericValue ?cargoGrossNumber4 ;
                                                   qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                                FILTER (datatype(?cargoGrossNumber4) = xsd:decimal &&
                                        ?cargoGrossNumber4 >= 300 &&
                                        ?cargoGrossNumber4 < 500)
                            }
                            && ?app != false
                        )
                    )
                )
            )
        )
        ||
        (
            ?type NOT IN ("fishing_vessel", "pleasure_yacht", "cargo_ship") &&
            ?app != false
        )
    )
}
"""
    ] .
<END_SHACL>
