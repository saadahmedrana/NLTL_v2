<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF005Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:message "An ice class is required as an input for this requirement."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRegulationEdition2002 ;
                sh:minCount 1 ;
                sh:message "Ships meeting the stated dates shall identify the 2002 Ice Class Regulations as the applicable edition."
            ] ;
            sh:sparql [
                sh:message "The 2002 Ice Class Regulations applicability branch requires a construction stage on or after 1 September 2003 and a contract before 1 January 2010." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?contract ;
                              nltl:constructionStageDate ?stage .
                        FILTER (
                            datatype(?contract) = xsd:date &&
                            datatype(?stage) = xsd:date
                        )
                        FILTER (
                            !(
                                (
                                    YEAR(?stage) > 2003 ||
                                    (
                                        YEAR(?stage) = 2003 &&
                                        (
                                            MONTH(?stage) > 9 ||
                                            (MONTH(?stage) = 9 && DAY(?stage) >= 1)
                                        )
                                    )
                                ) &&
                                (
                                    YEAR(?contract) < 2010 ||
                                    (
                                        YEAR(?contract) = 2010 &&
                                        MONTH(?contract) = 1 &&
                                        DAY(?contract) < 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
            ] ;
            sh:sparql [
                sh:message "The 2002 Ice Class Regulations shall not be selected when the stated construction dates do not meet their applicability condition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?contract ;
                              nltl:constructionStageDate ?stage .
                        FILTER (
                            datatype(?contract) = xsd:date &&
                            datatype(?stage) = xsd:date
                        )
                        FILTER (
                            !(
                                (
                                    YEAR(?stage) > 2003 ||
                                    (
                                        YEAR(?stage) = 2003 &&
                                        (
                                            MONTH(?stage) > 9 ||
                                            (MONTH(?stage) = 9 && DAY(?stage) >= 1)
                                        )
                                    )
                                ) &&
                                (
                                    YEAR(?contract) < 2010 ||
                                    (
                                        YEAR(?contract) = 2010 &&
                                        MONTH(?contract) = 1 &&
                                        DAY(?contract) < 1
                                    )
                                )
                            )
                        )
                        FILTER EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                  nltl:iceClassRegulationEdition2002 .
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
