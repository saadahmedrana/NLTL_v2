<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:message "Assessment date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "Construction stage date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:message "Delivery date must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA Super or IA ship constructed before 1 September 2003, the ship must have a permitted 2017 compliance method and the assessment date must be no later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )

                FILTER (
                    YEAR(?constructionDate) < 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        &&
                        (
                            MONTH(?constructionDate) < 9
                            ||
                            (
                                MONTH(?constructionDate) = 9
                                &&
                                DAY(?constructionDate) < 1
                            )
                        )
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate ;
                          nltl:assessmentDate ?assessmentDate ;
                          nltl:engineOutputComplianceMethod ?method .

                    FILTER (
                        ?method IN (
                            nltl:traficom2017Section3Point2Point2Method,
                            nltl:traficom2017Section3Point2Point4Method
                        )
                    )

                    FILTER (
                        YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
                        ||
                        (
                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                            &&
                            (
                                MONTH(?assessmentDate) < 1
                                ||
                                (
                                    MONTH(?assessmentDate) = 1
                                    &&
                                    DAY(?assessmentDate) <= 1
                                )
                            )
                        )
                    )
                }
            }
        """
    ] .
<END_SHACL>
