<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one ice class for applicability assessment."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one construction stage date for applicability assessment."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one delivery date for the twenty-year deadline calculation."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one assessment date for deadline evaluation."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:message "The conditional requirement is applicable only to IA Super or IA ships constructed before 1 September 2003." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the Ice Class Regulations of 2017." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionDate .
                        FILTER (
                            !(
                                ?iceClass IN (
                                    nltl:iceClassIaSuper,
                                    nltl:iceClassIa
                                )
                                &&
                                (
                                    YEAR(?constructionDate) < 2003 ||
                                    (
                                        YEAR(?constructionDate) = 2003 &&
                                        (
                                            MONTH(?constructionDate) < 9 ||
                                            (
                                                MONTH(?constructionDate) = 9 &&
                                                DAY(?constructionDate) < 1
                                            )
                                        )
                                    )
                                )
                            )
                        )
                    }
                """
            ]
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:engineOutputComplianceMethodValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "The compliance method must be section 3.2.2 or section 3.2.4 of the Ice Class Regulations of 2017."
            ] ;
            sh:sparql [
                sh:message "The required compliance must be achieved no later than 1 January in the year twenty years after delivery."
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                (
                                    MONTH(?assessmentDate) > 1 ||
                                    (
                                        MONTH(?assessmentDate) = 1 &&
                                        DAY(?assessmentDate) > 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
