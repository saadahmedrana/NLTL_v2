<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape> ;
        sh:message "The ship must identify its intended ice-operating waterline and its profile points."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape> ;
        sh:message "The ship must identify an upper ice waterline and its profile points."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal, pointwise, the envelope of the highest intended ice-operating waterline points at each longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                {
                    {
                        SELECT $this ?x (MAX(?iy) AS ?maxY)
                        WHERE {
                            $this nltl:hasIntendedIceOperatingWaterline ?iw .
                            ?iw nltl:hasWaterlineProfilePoint ?ip .
                            ?ip nltl:longitudinalPosition ?ixq .
                            ?ixq <http://qudt.org/schema/qudt/numericValue> ?x .
                            ?ip nltl:verticalCoordinate ?iyq .
                            ?iyq <http://qudt.org/schema/qudt/numericValue> ?iy .
                        }
                        GROUP BY $this ?x
                    }
                    $this nltl:hasUpperIceWaterline ?uw .
                    ?uw nltl:hasWaterlineProfilePoint ?up .
                    ?up nltl:longitudinalPosition ?uxq .
                    ?uxq <http://qudt.org/schema/qudt/numericValue> ?x .
                    ?up nltl:verticalCoordinate ?uyq .
                    ?uyq <http://qudt.org/schema/qudt/numericValue> ?uy .
                    FILTER (?uy != ?maxY)
                }
                UNION
                {
                    $this nltl:hasIntendedIceOperatingWaterline ?iw2 .
                    ?iw2 nltl:hasWaterlineProfilePoint ?ip2 .
                    ?ip2 nltl:longitudinalPosition ?ixq2 .
                    ?ixq2 <http://qudt.org/schema/qudt/numericValue> ?ix2 .
                    FILTER NOT EXISTS {
                        $this nltl:hasUpperIceWaterline ?uw2 .
                        ?uw2 nltl:hasWaterlineProfilePoint ?up2 .
                        ?up2 nltl:longitudinalPosition ?uxq2 .
                        ?uxq2 <http://qudt.org/schema/qudt/numericValue> ?ux2 .
                        FILTER (?ux2 = ?ix2)
                    }
                }
                UNION
                {
                    $this nltl:hasUpperIceWaterline ?uw3 .
                    ?uw3 nltl:hasWaterlineProfilePoint ?up3 .
                    ?up3 nltl:longitudinalPosition ?uxq3 .
                    ?uxq3 <http://qudt.org/schema/qudt/numericValue> ?ux3 .
                    FILTER NOT EXISTS {
                        $this nltl:hasIntendedIceOperatingWaterline ?iw3 .
                        ?iw3 nltl:hasWaterlineProfilePoint ?ip3 .
                        ?ip3 nltl:longitudinalPosition ?ixq3 .
                        ?ixq3 <http://qudt.org/schema/qudt/numericValue> ?ix3 .
                        FILTER (?ix3 = ?ux3)
                    }
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-ProfilePointShape> ;
        sh:message "Each ice waterline must contain profile points."
    ] .

<urn:nltl:generated-shape:TRF-011-ProfilePointShape>
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape> ;
        sh:message "Each waterline profile point must have one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape> ;
        sh:message "Each waterline profile point must have one vertical coordinate in metres."
    ] .

<urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
