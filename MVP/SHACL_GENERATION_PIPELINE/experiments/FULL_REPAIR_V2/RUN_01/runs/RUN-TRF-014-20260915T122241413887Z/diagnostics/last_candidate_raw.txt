gen:TRF014NotApplicableBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The marking alternatives apply only when the summer load line in fresh water is higher than the upper ice waterline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT ?this
            WHERE {
                ?this nltl:summerLoadLineFreshWaterDraught ?summerDraught ;
                      nltl:upperIceWaterlineDraught ?uiwlDraught .
                ?summerDraught qudt:numericValue ?summerValue .
                ?uiwlDraught qudt:numericValue ?uiwlValue .
                FILTER (?summerValue > ?uiwlValue)
            }
        """
    ] .

gen:TRF014MarkingPropertiesShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice-class draught mark amidships must equal the certificate maximum ice-class draught amidships." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT ?this
            WHERE {
                ?this nltl:hasClassCertificate ?certificate ;
                      nltl:iceClassDraughtMarkDraughtAmidships ?mark .
                ?mark qudt:numericValue ?markValue ;
                      qudt:unit unit:M .
                ?certificate nltl:maximumIceClassDraughtAmidships ?certificateDraught .
                ?certificateDraught qudt:numericValue ?certificateValue ;
                                   qudt:unit unit:M .
                FILTER (?markValue != ?certificateValue)
            }
        """
    ] .

gen:TRF014PostCutoffMarkingBranch
    a sh:NodeShape ;
    sh:and ( gen:TRF014MarkingPropertiesShape ) ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships built on or after 1 July 2007 shall have the required ice marking evidence when the summer load line in fresh water is higher than the upper ice waterline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                FILTER NOT EXISTS {
                    ?this nltl:constructionDate ?constructionDate ;
                          nltl:summerLoadLineFreshWaterDraught ?summerDraught ;
                          nltl:upperIceWaterlineDraught ?uiwlDraught ;
                          nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?mark .
                    ?summerDraught qudt:numericValue ?summerValue .
                    ?uiwlDraught qudt:numericValue ?uiwlValue .
                    FILTER (
                        ?constructionDate >= "2007-07-01"^^xsd:date &&
                        ?summerValue > ?uiwlValue
                    )
                }
            }
        """
    ] .

gen:TRF014PreCutoffDeferredBranch
    a sh:NodeShape ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A pre-cutoff ship without complete marking evidence shall provide a first scheduled dry docking date later than both 1 July 2007 and the assessment date when the summer load line in fresh water is higher than the upper ice waterline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                FILTER NOT EXISTS {
                    ?this nltl:constructionDate ?constructionDate ;
                          nltl:summerLoadLineFreshWaterDraught ?summerDraught ;
                          nltl:upperIceWaterlineDraught ?uiwlDraught ;
                          nltl:firstScheduledDryDockingDate ?dryDockingDate ;
                          nltl:assessmentDate ?assessmentDate .
                    ?summerDraught qudt:numericValue ?summerValue .
                    ?uiwlDraught qudt:numericValue ?uiwlValue .
                    FILTER (
                        ?constructionDate < "2007-07-01"^^xsd:date &&
                        ?summerValue > ?uiwlValue &&
                        ?dryDockingDate > ?assessmentDate &&
                        ?dryDockingDate > "2007-07-01"^^xsd:date

                }
            }
        """
    ] .
