<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MassQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:BallastTankShape
    a sh:NodeShape ;
    sh:targetClass nltl:ballastTank ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NavigatingInIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:IceNavigationApplicableShape
        gen:IceNavigationNotApplicableShape
    ) .

gen:IceNavigationNotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceNavigationApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node gen:MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:node gen:BallastTankShape
    ] ;
    sh:sparql [
        sh:message "Draught amidships must be at least the LIWL draught amidships, and forward draught must meet the level-ice-thickness displacement limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:draughtAmidships ?draughtQ ;
                      nltl:liwlDraughtAmidships ?liwlQ ;
                      nltl:forwardDraught ?forwardQ ;
                      nltl:displacementAtUpperIceWaterline ?displacementQ ;
                      nltl:levelIceThickness ?iceQ .

                ?draughtQ qudt:numericValue ?draught .
                ?liwlQ qudt:numericValue ?liwl .
                ?forwardQ qudt:numericValue ?forward .
                ?displacementQ qudt:numericValue ?displacement .
                ?iceQ qudt:numericValue ?ice .

                BIND ((2 + (0.00025 * ?displacement)) * ?ice AS ?calculatedLimit)
                BIND (4 * ?ice AS ?maximumLimit)
                BIND (IF(?calculatedLimit < ?maximumLimit,
                         ?calculatedLimit,
                         ?maximumLimit) AS ?forwardLimit)

                FILTER (?draught < ?liwl || ?forward < ?forwardLimit)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ballast tank situated above the LIWL and needed to reach the LIWL must have freezing prevention." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasBallastTank ?tank .
                ?tank nltl:situatedAboveLowerIceWaterline true ;
                      nltl:usedToReachLowerIceWaterline true .

                FILTER NOT EXISTS {
                    ?tank nltl:freezingPreventionPresent true
                }
            }
        """
    ] .
</END_SHACL>
