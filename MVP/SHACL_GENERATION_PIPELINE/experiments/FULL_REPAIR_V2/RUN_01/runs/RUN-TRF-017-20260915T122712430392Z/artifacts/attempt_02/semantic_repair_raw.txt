<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnrestrictedPropulsionOutputBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must equal the unrestricted continuous propulsion machinery output plus any additional propulsion power sources when no restriction applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?p .
                ?p qudt:numericValue ?pv .
                $this nltl:propulsionMachineryContinuousOutput ?c .
                ?c qudt:numericValue ?cv .
                OPTIONAL {
                    $this nltl:additionalPropulsionPower ?a .
                    ?a qudt:numericValue ?av .
                }
                BIND (COALESCE(?av, 0) AS ?additional)
                FILTER (?pv != (?cv + ?additional))
            }
        """
    ] .

gen:RestrictedPropulsionOutputBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:restrictedPropulsionOutput ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must equal the restricted propulsion output when a restriction applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?p .
                ?p qudt:numericValue ?pv .
                $this nltl:restrictedPropulsionOutput ?r .
                ?r qudt:numericValue ?rv .
                FILTER (?pv != ?rv)
            }
        """
    ] .

gen:TRF_017_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:additionalPropulsionPower ;
        sh:node gen:PowerQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:UnrestrictedPropulsionOutputBranch
        gen:RestrictedPropulsionOutputBranch
    ) .
<END_SHACL>
