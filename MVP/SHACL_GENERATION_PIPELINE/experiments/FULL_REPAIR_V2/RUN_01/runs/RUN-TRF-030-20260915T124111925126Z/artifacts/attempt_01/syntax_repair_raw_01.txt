<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:PressureQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:LengthQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:DimensionlessQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:DirectAnalysisCaseReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case shall confirm the capacity-minimizing load position."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case shall confirm evaluation of combined bending and shear."
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node nltl:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case shall provide its load patch length."
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node nltl:DimensionlessQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case shall provide the corresponding load area factor ca."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node nltl:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node nltl:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-030-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltl:PressureQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the ice pressure input in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node nltl:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the load patch height."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node nltl:PressureQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the externally calculated applied ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall state whether load length was determined from the arrangement."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:node nltl:DirectAnalysisCaseReadinessShape ;
        sh:minCount 2 ;
        sh:message "The ship shall provide several paired direct analysis cases."
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:minCount 2 ;
                sh:node nltl:DirectAnalysisCaseReadinessShape ;
                sh:message "When load length is not determined from the arrangement, several cases with corresponding load lengths and ca values are required."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The direct analysis cases shall include the prescribed vertical positions and a mid-span or spacing-centred horizontal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasDirectAnalysisCase ?upperCase .
                        ?upperCase nltl:verticalLoadPositionType
                            nltl:upperIceWaterlineCenteredPosition .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDirectAnalysisCase ?halfCase .
                        ?halfCase nltl:verticalLoadPositionType
                            nltl:halfLoadHeightBelowLowerIceWaterlinePosition .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDirectAnalysisCase ?intermediateCase .
                        ?intermediateCase nltl:verticalLoadPositionType
                            nltl:intermediateVerticalPosition .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasDirectAnalysisCase ?midSpanCase .
                        ?midSpanCase nltl:horizontalLoadPositionType
                            nltl:midSpanOrSpacingCenteredPosition .
                    }
                )
            }
        """
    ] .
<END_SHACL>
