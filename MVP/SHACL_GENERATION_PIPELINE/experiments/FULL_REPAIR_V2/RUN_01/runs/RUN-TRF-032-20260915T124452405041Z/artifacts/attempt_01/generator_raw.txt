<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF032Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gs:TRF032IceStrengthenedBranch
        gs:TRF032NotIceStrengthenedBranch
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship is not ice strengthened and the regulation-required scantling is less than the classification-society-required scantling, the selected design scantling must equal the classification-society-required scantling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:regulationRequiredScantling ?regulationScantling ;
                      nltl:classificationSocietyRequiredScantling ?classificationScantling .

                ?regulationScantling qudt:numericValue ?regulationValue .
                ?classificationScantling qudt:numericValue ?classificationValue .

                FILTER (
                    datatype(?regulationValue) = xsd:decimal &&
                    datatype(?classificationValue) = xsd:decimal &&
                    ?regulationValue < ?classificationValue
                )

                FILTER NOT EXISTS {
                    $this nltl:selectedDesignScantling ?selectedScantling .
                    ?selectedScantling qudt:numericValue ?selectedValue .
                    FILTER (
                        datatype(?selectedValue) = xsd:decimal &&
                        ?selectedValue = ?classificationValue
                    )
                }
            }
        """
    ] .

gs:TRF032IceStrengthenedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF032NotIceStrengthenedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
