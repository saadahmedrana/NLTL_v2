<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF046FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A frame shall identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF046UpperEndShape ;
        sh:message "A frame shall identify an upper end attached to an eligible supporting structure."
    ] .

gen:TRF046UpperEndShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasAttachedSupportingStructure ;
                sh:minCount 1 ;
                sh:or (
                    [ sh:class nltl:deckStructure ]
                    [ sh:class nltl:tankBoundaryPlating ]
                    [ sh:class nltl:iceStringer ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasTerminationAboveSupportingStructure ;
                sh:minCount 1 ;
                sh:or (
                    [ sh:class nltl:deckStructure ]
                    [ sh:class nltl:tankBoundaryPlating ]
                    [ sh:class nltl:iceStringer ]
                )
            ]
        ]
    ) ;
    sh:message "The upper end shall be attached to a deck, tank boundary plating, or ice stringer." .
<END_SHACL>
