<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A main frame or intermediate ice frame must have a strengthened part." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a ?frameKind .
                FILTER ( ?frameKind IN (nltl:mainFrame, nltl:intermediateIceFrame) )
                FILTER NOT EXISTS {
                    $this nltl:hasStrengthenedPart ?strengthenedPart .
                }
            }
        """
    ] .

gs:TRF047MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A main frame below the lower edge of the ice belt must be ice strengthened."
    ] .

gs:TRF047TransverseFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:class nltl:frameEnd ;
        sh:minCount 1 ;
        sh:message "A transverse frame must have a represented lower end."
    ] .

gs:TRF047FrameEndShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameEnd ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:tankTop ]
                [ sh:class nltl:iceStringer ]
            )
        ] ;
        sh:message "The lower end must be attached to a deck, tank boundary plating, tank top, or ice stringer."
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:node [
            sh:class nltl:horizontalConnectionMember ;
            sh:property [
                sh:path nltl:connectsToAdjacentMainFrame ;
                sh:class nltl:mainFrame ;
                sh:minCount 1 ;
                sh:message "A horizontal connection member must connect to an adjacent main frame."
            ]
        ]
    ] .

gs:TRF047HorizontalConnectionMemberShape
    a sh:NodeShape ;
    sh:targetClass nltl:horizontalConnectionMember ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:class nltl:mainFrame ;
        sh:minCount 1 ;
        sh:message "A horizontal connection member must connect to an adjacent main frame."
    ] .
<END_SHACL>
