<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_052_FrameSupportShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame lacks required applicability data, exceeds the permitted antitripping support spacing, or lacks the required antitripping support region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:frameAsymmetrical ?asym .
                        FILTER (?asym = true || ?asym = false)
                    }
                    ||
                    (
                        EXISTS {
                            $this nltl:frameAsymmetrical false
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:frameWebAngleToShell ?angle .
                            ?angle qudt:numericValue ?av ;
                                   qudt:unit qudt:DEG .
                            FILTER (xsd:decimal(?av) < 90)
                        }
                    )
                    ||
                    NOT EXISTS {
                        $this nltl:frameSpan ?span .
                        ?span qudt:numericValue ?spanValue ;
                              qudt:unit qudt:M .
                        FILTER (xsd:decimal(?spanValue) >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceClass ?ice .
                        FILTER (
                            ?ice = nltl:iceClassIaSuper ||
                            ?ice = nltl:iceClassIa ||
                            ?ice = nltl:iceClassIb ||
                            ?ice = nltl:iceClassIc ||
                            ?ice = nltl:iceClassIi ||
                            ?ice = nltl:iceClassIii
                        )
                    }
                    ||
                    (
                        (
                            EXISTS { $this nltl:frameAsymmetrical true }
                            ||
                            EXISTS {
                                $this nltl:frameAsymmetrical false ;
                                     nltl:frameWebAngleToShell ?angle2 .
                                ?angle2 qudt:numericValue ?av2 ;
                                        qudt:unit qudt:DEG .
                                FILTER (xsd:decimal(?av2) < 90)
                            }
                        )
                        &&
                        NOT EXISTS {
                            $this nltl:equivalentSupportStatusByDirectCalculation true
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:antitrippingSupportSpacing ?spacing .
                            ?spacing qudt:numericValue ?spacingValue ;
                                     qudt:unit qudt:MilliM .
                            FILTER (xsd:decimal(?spacingValue) <= 1300)
                        }
                    )
                    ||
                    (
                        NOT EXISTS {
                            $this nltl:equivalentSupportStatusByDirectCalculation true
                        }
                        &&
                        EXISTS {
                            $this nltl:frameSpan ?span2 .
                            ?span2 qudt:numericValue ?spanValue2 ;
                                  qudt:unit qudt:M .
                            FILTER (xsd:decimal(?spanValue2) <= 4)
                        }
                        &&
                        EXISTS { $this nltl:iceClass nltl:iceClassIa }
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                            }
                        )
                    )
                    ||
                    (
                        NOT EXISTS {
                            $this nltl:equivalentSupportStatusByDirectCalculation true
                        }
                        &&
                        EXISTS {
                            $this nltl:frameSpan ?span3 .
                            ?span3 qudt:numericValue ?spanValue3 ;
                                  qudt:unit qudt:M .
                            FILTER (xsd:decimal(?spanValue3) <= 4)
                        }
                        &&
                        (
                            EXISTS { $this nltl:iceClass nltl:iceClassIb }
                            ||
                            EXISTS { $this nltl:iceClass nltl:iceClassIc }
                        )
                        &&
                        NOT EXISTS {
                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                        }
                    )
                )
            }
        """
    ] .
</END_SHACL>
