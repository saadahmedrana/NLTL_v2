<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF053IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:IcePressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:DesignIceLoadHeightQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:SpanQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:FrameMomentFactorQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one dimensionless frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:YieldStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:DesignLineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated design line load in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated section modulus in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:RequiredShearAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide exactly one calculated required shear area in cm2."
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area must satisfy the TRF-053 formulae; derived decimal results are checked with a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:designLineLoad ?lineLoadNode ;
                      nltl:sectionModulus ?sectionModulusNode ;
                      nltl:requiredShearArea ?shearAreaNode .

                ?pressureNode qudt:numericValue ?pressure .
                ?heightNode qudt:numericValue ?height .
                ?spanNode qudt:numericValue ?span .
                ?momentNode qudt:numericValue ?moment .
                ?yieldNode qudt:numericValue ?yieldStrength .
                ?lineLoadNode qudt:numericValue ?lineLoad .
                ?sectionModulusNode qudt:numericValue ?sectionModulus .
                ?shearAreaNode qudt:numericValue ?shearArea .

                BIND(IF((?pressure * ?height) < xsd:decimal("0.15"),
                        xsd:decimal("0.15"),
                        (?pressure * ?height)) AS ?designLineLoadMNPerM)

                BIND((?designLineLoadMNPerM * xsd:decimal("1000")) AS ?expectedLineLoad)
                BIND((xsd:decimal("0.9") * xsd:decimal("1.8") *
                      ?designLineLoadMNPerM * ?span * ?span /
                      (?moment * ?yieldStrength) * xsd:decimal("1000000"))
                     AS ?expectedSectionModulus)
                BIND((xsd:decimal("1.7320508075688772") * xsd:decimal("0.9") *
                      xsd:decimal("1.8") * xsd:decimal("1.2") *
                      ?designLineLoadMNPerM * ?span /
                      (xsd:decimal("2") * ?yieldStrength) *
                      xsd:decimal("10000"))
                     AS ?expectedShearArea)

                FILTER(
                    ABS(?lineLoad - ?expectedLineLoad) >
                    IF(ABS(?expectedLineLoad) * xsd:decimal("0.000001") >
                       xsd:decimal("0.000000001"),
                       ABS(?expectedLineLoad) * xsd:decimal("0.000001"),
                       xsd:decimal("0.000000001"))
                    ||
                    ABS(?sectionModulus - ?expectedSectionModulus) >
                    IF(ABS(?expectedSectionModulus) * xsd:decimal("0.000001") >
                       xsd:decimal("0.000000001"),
                       ABS(?expectedSectionModulus) * xsd:decimal("0.000001"),
                       xsd:decimal("0.000000001"))
                    ||
                    ABS(?shearArea - ?expectedShearArea) >
                    IF(ABS(?expectedShearArea) * xsd:decimal("0.000001") >
                       xsd:decimal("0.000000001"),
                       ABS(?expectedShearArea) * xsd:decimal("0.000001"),
                       xsd:decimal("0.000000001"))
                )
            }
        """
    ] .
