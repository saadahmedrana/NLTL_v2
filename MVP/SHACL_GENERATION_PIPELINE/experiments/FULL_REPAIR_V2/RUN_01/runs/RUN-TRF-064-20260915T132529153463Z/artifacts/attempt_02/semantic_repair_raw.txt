<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF064ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasSidePropellerShafting ;
        sh:class nltl:sidePropellerShaftingAssembly ;
        sh:minCount 1 ;
        sh:message "A ship shall identify its side propeller shafting." ;
    ] ;
    sh:property [
        sh:path nltl:hasSternTube ;
        sh:class nltl:sternTube ;
        sh:minCount 1 ;
        sh:message "A ship shall identify its stern tube." ;
    ] ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasDetachedStrut ;
                    sh:minCount 1 ;
                ]
            ] ;
            sh:property [
                sh:path ( nltl:hasSidePropellerShafting nltl:enclosedByPlatedBossing ) ;
                sh:class nltl:platedBossingStructure ;
                sh:minCount 1 ;
                sh:message "Side propeller shafting shall be enclosed within plated bossing when detached struts are not used." ;
            ] ;
            sh:property [
                sh:path ( nltl:hasSternTube nltl:enclosedByPlatedBossing ) ;
                sh:class nltl:platedBossingStructure ;
                sh:minCount 1 ;
                sh:message "The stern tube shall be enclosed within plated bossing when detached struts are not used." ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:hasDetachedStrut ;
                sh:node gen:TRF064DetachedStrutShape ;
                sh:minCount 1 ;
                sh:message "When detached struts are used, design, strength, and hull-attachment evidence shall be provided." ;
            ]
        ]
    ) .

gen:TRF064DetachedStrutShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:class nltl:strutDesignEvidence ;
        sh:minCount 1 ;
        sh:message "Detached strut design evidence is required." ;
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:class nltl:strutStrengthEvidence ;
        sh:minCount 1 ;
        sh:message "Detached strut strength evidence is required." ;
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:class nltl:strutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:message "Detached strut hull-attachment evidence is required." ;
    ] .
<END_SHACL>
