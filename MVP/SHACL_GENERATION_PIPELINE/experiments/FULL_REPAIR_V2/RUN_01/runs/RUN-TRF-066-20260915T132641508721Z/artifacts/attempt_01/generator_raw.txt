gen:TRF066Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF066IaPracticableShape
        gen:TRF066IaNotPracticableShape
        gen:TRF066IaSuperPracticableShape
        gen:TRF066IaSuperNotPracticableShape
        gen:TRF066OtherIceClassShape
    ) ;
    sh:message "The ship's ice class and the applicable rudder protection arrangement must conform to the requirement." .

gen:TRF066IaPracticableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:datatype nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iceClassIa
    ] ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasRudderProtectionArrangement ;
        sh:minCount 1 ;
        sh:node gen:TRF066IceKnifeShape
    ] .

    sh:property [

gen:TRF066OtherIceClassShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:datatype nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF066IceKnifeShape
    a sh:NodeShape ;
    sh:class nltl:iceKnife ;
    sh:property [
        sh:path nltl:extendsBelowLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:message "An applicable ice knife must extend below the lower ice waterline." .

gen:TRF066EquivalentMeansShape
    a sh:NodeShape ;
    sh:class nltl:equivalentRudderProtectionMeans ;
    sh:message "When an ice knife is not practicable, an equivalent rudder protection means is required." .
