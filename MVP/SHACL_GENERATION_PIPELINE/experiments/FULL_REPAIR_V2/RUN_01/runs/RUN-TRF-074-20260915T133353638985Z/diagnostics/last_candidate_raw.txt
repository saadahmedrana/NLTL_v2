<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal quantity value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use metres as its unit."
    ] .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship ice class is required to determine applicability of the propeller submersion rule."
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:node gen:TRF074QuantityShape ;
        sh:minCount 1 ;
        sh:message "The ballast-condition propeller top depth is required."
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:node gen:TRF074QuantityShape ;
        sh:minCount 1 ;
        sh:message "The required propeller submersion depth hi is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, when the ballast-condition propeller top depth is less than hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                ?depthNode qudt:numericValue ?depth .
                ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .

                $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                ?hiNode qudt:numericValue ?hi .
                ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    datatype(?depth) = xsd:decimal &&
                    datatype(?hi) = xsd:decimal &&
                    ?depth < ?hi
                )

                FILTER NOT EXISTS {
                    $this nltl:propulsionSystemDesignedForIceClass
                        nltl:iceClassIa
                }
            }
        """
    ] .
</END_SHACL>
