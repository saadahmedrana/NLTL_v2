<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_076_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide one propeller type for load-case applicability." ;
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:TRF_076_LoadCaseShape ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "CP and FP propellers must cover load cases 1 through 4." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    $this nltl:propellerType ?type .
                    FILTER(?type IN ("CP", "FP"))
                    FILTER NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case .
                        ?case nltl:propellerBladeLoadCaseNumber ?number .
                        FILTER(?number = 1)
                    }
                }
                UNION
                {
                    $this nltl:propellerType ?type .
                    FILTER(?type IN ("CP", "FP"))
                    FILTER NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case .
                        ?case nltl:propellerBladeLoadCaseNumber ?number .
                        FILTER(?number = 2)
                    }
                }
                UNION
                {
                    $this nltl:propellerType ?type .
                    FILTER(?type IN ("CP", "FP"))
                    FILTER NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case .
                        ?case nltl:propellerBladeLoadCaseNumber ?number .
                        FILTER(?number = 3)
                    }
                }
                UNION
                {
                    $this nltl:propellerType ?type .
                    FILTER(?type IN ("CP", "FP"))
                    FILTER NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case .
                        ?case nltl:propellerBladeLoadCaseNumber ?number .
                        FILTER(?number = 4)
                    }
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an FP propeller with reversing capability, load case 5 must also be covered." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType "FP" .
                $this nltl:propellerReversingCapability true .
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER(?number = 5)
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An FP propeller must provide a boolean reversing-capability value to determine whether load case 5 applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType "FP" .
                FILTER NOT EXISTS {
                    $this nltl:propellerReversingCapability ?reversing .
                    FILTER(
                        datatype(?reversing) = xsd:boolean &&
                        (?reversing = true || ?reversing = false)
                    )
                }
            }
        """ ;
    ] .

gen:TRF_076_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked propeller blade load case must have one integer case number." ;
    ] .

</END_SHACL>
