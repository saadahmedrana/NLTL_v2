<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF078ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 2 ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash6 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedMaxCount 1 ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash7 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:qualifiedMaxCount 1 ;
        sh:message "The ship must provide exactly one linked lookup case for each of Tables 6-6 and 6-7."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:node gen:TRF078DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:node gen:TRF078DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:node gen:TRF078DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 multiplied by k2, k3, N_class, and n_n." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?classCase ;
                      nltl:hasTableLookupCase ?locationCase ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?reported .

                ?classCase nltl:tableReference nltl:traficomTable6Dash6 ;
                            nltl:selectedIceClassCycleCount ?nclassNode .
                ?locationCase nltl:tableReference nltl:traficomTable6Dash7 ;
                               nltl:selectedPropellerLocationFactor ?k1Node .

                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nnNode qudt:numericValue ?nn .
                ?nclassNode qudt:numericValue ?nclass .
                ?k1Node qudt:numericValue ?k1 .

                BIND ((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (xsd:decimal(?reported) != ?expected)
            }
        """
    ] .

gen:TRF078DimensionlessQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078SelectedIceClassCycleCountShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:selectedIceClassCycleCount ;
        sh:node gen:TRF078DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078SelectedPropellerLocationFactorShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:selectedPropellerLocationFactor ;
        sh:node gen:TRF078DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078LookupCaseShape a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasTableLookupCase ;
    sh:class nltl:tableLookupCase ;
    sh:or (
        [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash6 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:selectedIceClassCycleCount ;
                sh:node gen:TRF078DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash7 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:tableLookupPropellerLocation ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:propellerLocationValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:selectedPropellerLocationFactor ;
                sh:node gen:TRF078DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
<END_SHACL>
