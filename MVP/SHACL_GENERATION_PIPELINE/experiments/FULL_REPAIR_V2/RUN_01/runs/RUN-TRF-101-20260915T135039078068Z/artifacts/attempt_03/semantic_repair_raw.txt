                      nltl:bladeFailureUltimateLoad/qudt:numericValue ?load ;

                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedReference)
                BIND ((300 * ?c * ?t * ?t * ?expectedReference) /
                      ((0.8 * ?diameter) - (2 * ?r)) AS ?expectedLoad)

                    ABS(?reference - ?expectedReference) >
                    ((1e-6 * ABS(?expectedReference)) + 1e-9)

                    ABS(?load - ?expectedLoad) >
                    ((1e-6 * ABS(?expectedLoad)) + 1e-9)
                )
            }
