<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-015> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:node [
            sh:class nltl:bowSubregionCalculationCase ;
            sh:property [
                sh:path nltl:bowFormApplicabilityClassification ;
                sh:class nltl:bowFormApplicabilityValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable I2.3.1(v) bow subregion case must contain the declared calculation inputs, lookup factors, and externally computed bow shape coefficient with the required quantity units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasBowSubregionCalculationCase ?case .
                ?case nltl:bowFormApplicabilityClassification nltl:i231vBowFormApplicability .

                FILTER NOT EXISTS {
                    $this nltl:upperIceWaterlineLengthLUI ?lui .
                    ?lui qudt:numericValue ?luiValue ;
                         qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?luiValue a xsd:decimal .

                    $this nltl:betaIPrime ?betaPrime .
                    ?betaPrime qudt:numericValue ?betaValue ;
                               qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?betaValue a xsd:decimal .

                    ?case nltl:bowSubregionMidLengthPosition ?midPosition .
                    ?midPosition qudt:numericValue ?midValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?midValue a xsd:decimal .

                    ?case nltl:bowSubregionWaterlineAngle ?angle .
                    ?angle qudt:numericValue ?angleValue ;
                           qudt:unit <http://qudt.org/vocab/unit/DEG> .
                    ?angleValue a xsd:decimal .

                    ?case nltl:bowShapeCoefficient ?shapeCoefficient .
                    ?shapeCoefficient qudt:numericValue ?shapeValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?shapeValue a xsd:decimal .

                    ?case nltl:hasFailureClassFactorLookupCase ?lookupCase .
                    ?lookupCase nltl:flexuralFailureClassFactor ?flexuralFactor .
                    ?flexuralFactor qudt:numericValue ?flexuralValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?flexuralValue a xsd:decimal .

                    ?lookupCase nltl:crushingFailureClassFactor ?crushingFactor .
                    ?crushingFactor qudt:numericValue ?crushingValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?crushingValue a xsd:decimal .
                }
            }
        """
    ] .
<END_SHACL>
