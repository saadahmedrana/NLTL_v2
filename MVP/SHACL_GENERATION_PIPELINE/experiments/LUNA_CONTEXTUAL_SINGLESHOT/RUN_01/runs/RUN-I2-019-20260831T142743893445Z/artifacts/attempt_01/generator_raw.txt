<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-019> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:node [
                    sh:class nltl:hullAreaValue ;
                    sh:property [
                        sh:path nltl:hullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each structural member spanning one or more hull areas must use the largest spanned hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasSpannedHullArea ?area .

                {
                    SELECT ?member (MAX(?factorNumber) AS ?maximumFactor)
                    WHERE {
                        ?member nltl:hasSpannedHullArea ?areaForMaximum .
                        ?areaForMaximum nltl:hullAreaFactor ?factorNode .
                        ?factorNode a qudt:QuantityValue ;
                            qudt:numericValue ?factorNumber ;
                            qudt:unit unit:UNITLESS .
                        FILTER (datatype(?factorNumber) = xsd:decimal)
                    }
                    GROUP BY ?member
                }

                FILTER (
                    NOT EXISTS {
                        ?member nltl:selectedHullAreaFactor ?selectedNode
                    }
                    ||
                    EXISTS {
                        ?member nltl:selectedHullAreaFactor ?invalidSelectedNode .
                        FILTER NOT EXISTS {
                            ?invalidSelectedNode a qudt:QuantityValue ;
                                qudt:numericValue ?invalidNumber ;
                                qudt:unit unit:UNITLESS .
                            FILTER (datatype(?invalidNumber) = xsd:decimal)
                        }
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:selectedHullAreaFactor ?selectedNodeForComparison .
                        ?selectedNodeForComparison a qudt:QuantityValue ;
                            qudt:numericValue ?selectedNumber ;
                            qudt:unit unit:UNITLESS .
                        FILTER (datatype(?selectedNumber) = xsd:decimal)
                        FILTER (
                            ABS(?selectedNumber - ?maximumFactor)
                            <= IF(
                                ABS(?maximumFactor) * 0.000001 > 0.000000001,
                                ABS(?maximumFactor) * 0.000001,
                                0.000000001
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
