<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-021-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-021-PlatingShape>
    ] .

<urn:nltl:generated-shape:I2-021-PlatingShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAllowance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Thickness must equal the ice-load required net plate thickness plus the corrosion and abrasion allowance; values are compared after converting millimetres to metres within a relative tolerance of 1e-6 with a 1e-9 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadRequiredNetPlateThickness ?netQuantity ;
                      nltl:corrosionAbrasionAllowance ?allowanceQuantity ;
                      nltl:thickness ?thicknessQuantity .
                ?netQuantity qudt:numericValue ?net ;
                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?allowanceQuantity qudt:numericValue ?allowance ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                BIND((?net + ?allowance) / 1000 AS ?expected)
                BIND(ABS(?thickness - ?expected) AS ?difference)
                FILTER(
                    ?difference > (ABS(?expected) * 0.000001) &&
                    ?difference > 0.000000001
                )
            }
        """
    ] .
<END_SHACL>
