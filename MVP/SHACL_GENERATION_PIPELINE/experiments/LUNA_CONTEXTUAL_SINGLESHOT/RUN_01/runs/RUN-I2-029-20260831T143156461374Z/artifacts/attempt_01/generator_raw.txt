<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_029_CalculationCaseShape
    ] .

gen:I2_029_CalculationCaseShape a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must include attached shell plating in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must exclude attached shell plating from the shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must explicitly represent whether flange material is included in the shear area."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:minCount 1 ;
        sh:node gen:I2_029_StructuralMemberShape ;
        sh:message "The calculation case must link to its evaluated structural member."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When flange material is included in the shear area, every linked structural member must record that a flange is fitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:flangeMaterialIncludedInShearArea true .
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                FILTER NOT EXISTS {
                    ?member nltl:flangeFitted true .
                }
            }
        """
    ] .

gen:I2_029_StructuralMemberShape a sh:NodeShape ;
    sh:class nltl:structuralMember .
<END_SHACL>
