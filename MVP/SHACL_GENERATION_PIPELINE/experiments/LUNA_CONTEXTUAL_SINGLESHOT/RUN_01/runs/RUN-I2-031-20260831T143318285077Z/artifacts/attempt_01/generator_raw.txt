<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_031_StructuralMemberShape ;
        sh:message "Each linked structural member must provide the required I2-031 structural-member data."
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:datatype nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the frame profile type."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the patch load in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the plastic collapse mechanism used to define plastic strength."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand must not exceed the plastic strength of the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?member nltl:plasticStrength ?strength .
                ?demand qudt:numericValue ?demandValue ;
                        qudt:unit unit:KiloN .
                ?strength qudt:numericValue ?strengthValue ;
                          qudt:unit unit:KiloN .
                FILTER (?demandValue > ?strengthValue)
            }
        """
    ] .

gen:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_031_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The structural member must provide a load case for the I2-031 assessment."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:bottomStructureLocation
            ]
            [
                sh:hasValue nltl:sideStructureLocation
            ]
        ) ;
        sh:message "The structural member location must be bottom structure or side structure."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must provide the midspan load defining plastic collapse."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must provide its plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:transverseFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must provide the combined shear and bending demand in kN."
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:message "For bottom-structure loading, patch-load dimension b must be parallel with the frame direction."
    ] .

gen:I2_031_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
