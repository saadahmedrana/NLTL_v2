            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this ;
                       nltl:averageIcePressure ?pNode ;
                       nltl:frameSpan ?aNode ;
                       nltl:loadPatchHeight ?bNode ;
                       nltl:peakPressureFactor ?ppfNode ;
                       nltl:shearArea ?awNode ;
                       nltl:yieldStrength ?syNode .

                $this nltl:frameSpacing ?sNode ;
                      nltl:selectedHullAreaFactor ?afNode ;
                      nltl:requiredTransverseFrameShearArea ?atNode .

                ?pNode qudt:numericValue ?p .
                ?aNode qudt:numericValue ?a .
                ?bNode qudt:numericValue ?b .
                ?ppfNode qudt:numericValue ?ppf .
                ?awNode qudt:numericValue ?aw .
                ?syNode qudt:numericValue ?sy .
                ?sNode qudt:numericValue ?s .
                ?afNode qudt:numericValue ?af .
                ?atNode qudt:numericValue ?at .

                BIND(IF(?a <= ?b, ?a, ?b) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?s * ?af * ?ppf * ?p) / (0.577 * ?sy) AS ?expected)
                BIND(IF(ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001) AS ?tolerance)

                FILTER(
                    ABS(?at - ?expected) > ?tolerance ||
                    (?aw * 10000) < ?expected
                )
            }
        """
    ] .
