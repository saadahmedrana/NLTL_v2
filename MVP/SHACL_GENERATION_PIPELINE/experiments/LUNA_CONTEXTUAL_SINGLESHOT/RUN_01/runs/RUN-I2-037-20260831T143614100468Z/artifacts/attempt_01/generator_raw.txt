<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node nltl:I2_037_StructuralMemberShape ;
        sh:message "Each linked structural member must have the required load case and member capacity." ;
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:node nltl:I2_037_DesignCaseShape ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice load patch design case confirming the required capacity minimization." ;
    ] .

nltl:I2_037_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node nltl:I2_037_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "Each structural member must have a load case with the required bending-and-shear effect and application location." ;
    ] ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
        ] ;
    ] .

nltl:I2_037_LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The load case must identify the combined bending-and-shear effect." ;
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The load case must identify the location where the ice load patch is applied." ;
    ] .

nltl:I2_037_DesignCaseShape
    a sh:NodeShape ;
    sh:class nltl:iceLoadPatchDesignCase ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice load patch design case must confirm that member capacity under combined bending and shear is minimised." ;
    ] .
<END_SHACL>
