    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:I2_040_YieldStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

        sh:path nltl:hasStructuralMember ;

    ] .

        sh:maxCount 1 ;
        sh:message "The structural member shall provide exactly one web-height quantity in mm."
    ] ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:node gen:I2_040_NetWebThicknessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member shall provide exactly one net-web-thickness quantity in mm."
    ] ;
    sh:property [
        sh:path nltl:structuralSectionType ;

            nltl:teeSection

        sh:minCount 1 ;
        sh:message "The structural member shall identify whether it is a flat-bar, bulb, tee, or angle section."
