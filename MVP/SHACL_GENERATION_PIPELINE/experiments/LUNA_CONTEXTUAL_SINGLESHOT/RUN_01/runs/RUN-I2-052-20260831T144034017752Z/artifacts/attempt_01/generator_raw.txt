<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A quantity value must provide a decimal numeric value."
    ] .

gen:LengthQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The longitudinal position must provide a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The longitudinal position must use metres."
    ] .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The distribution factor must provide a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The distribution factor must be dimensionless."
    ] .

gen:InterpolationPointShape a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node gen:QuantityShape ;
        sh:message "Each interpolation point must provide an interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node gen:QuantityShape ;
        sh:message "Each interpolation point must provide an interpolation point value."
    ] .

gen:CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:message "Each calculation case must contain interpolation points."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:message "Each calculation case must identify a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:message "Each calculation case must identify an upper interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "Each calculation case must provide the hull girder longitudinal position from aft."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "Each calculation case must provide the externally determined interpolated longitudinal distribution factor."
    ] .

gen:I2052ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide a calculation case for the linear interpolation determination."
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must provide the externally determined interpolated value."
    ] .
<END_SHACL>
