<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_062_ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember
        ]
    ] ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:message "directCalculationUsed must be a boolean when recorded."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every load carrying stringer or web frame linked to the ship must form part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage a nltl:grillageSystem .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship must record that direct calculation was used when it has a load carrying stringer or web frame forming part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType ;
                        nltl:partOfGrillageSystem ?grillage .
                ?grillage a nltl:grillageSystem .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:directCalculationUsed ?used .
                    FILTER (?used = true)
                }
            }
        """
    ] .
<END_SHACL>
