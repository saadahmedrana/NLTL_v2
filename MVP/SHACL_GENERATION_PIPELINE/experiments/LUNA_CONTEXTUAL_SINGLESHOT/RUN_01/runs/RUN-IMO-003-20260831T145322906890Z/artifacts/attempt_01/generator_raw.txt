gs:IMO003ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:openWaterIceCondition
            nltl:iceConditionLessSevereThanCategoryAAndB
        ) ;
        sh:message "A ship shall declare a permitted design ice condition for this rule."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:openWaterIceCondition ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in ( nltl:polarShipCategoryC ) ;
                sh:message "A ship designed for open water shall have Polar Code category C."
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:iceConditionLessSevereThanCategoryAAndB ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in ( nltl:polarShipCategoryC ) ;
                sh:message "A ship designed for ice conditions less severe than categories A and B shall have Polar Code category C."
            ]
        ]
    ) .
