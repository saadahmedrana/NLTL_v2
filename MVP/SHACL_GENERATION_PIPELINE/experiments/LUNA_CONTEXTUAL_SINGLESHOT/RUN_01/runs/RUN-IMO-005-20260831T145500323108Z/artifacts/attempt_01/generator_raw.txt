<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_005_IceFreeWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship identified as being in ice free waters must record totalIcePresent as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "ice free waters")
                FILTER NOT EXISTS {
                    $this nltl:totalIcePresent ?presence .
                    FILTER (?presence = false)
                }
            }
        """
    ] .
<END_SHACL>
