<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO006MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceTypeValue
        ] ;
        sh:message "Each ship must provide an ice type classification." 
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice thickness must be a decimal QUDT quantity expressed in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For medium first-year ice, ice thickness must be between 0.7 m and 1.2 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?thickness .
                    ?thickness qudt:numericValue ?value ;
                               qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.7"^^xsd:decimal &&
                        ?value <= "1.2"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
