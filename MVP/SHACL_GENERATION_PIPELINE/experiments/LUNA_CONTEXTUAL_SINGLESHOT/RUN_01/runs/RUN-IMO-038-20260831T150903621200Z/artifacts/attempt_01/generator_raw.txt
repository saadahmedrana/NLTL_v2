            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?forward ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .

                ?draughtNode qudt:numericValue ?draught .
                ?longitudinalNode qudt:numericValue ?longitudinal .
                ?verticalNode qudt:numericValue ?vertical .
                ?positionNode qudt:numericValue ?position .

                BIND(IF(?forward = true,
                        ?length * 0.045,
                        ?length * 0.015) AS ?expectedLongitudinal)
                BIND(IF(?draught * 0.20 < ?expectedLongitudinal,
                        ?draught * 0.20,
                        ?expectedLongitudinal) AS ?expectedVertical)
                BIND((?expectedLongitudinal * 0.000001) + 0.000000001 AS ?longitudinalTolerance)
                BIND((?expectedVertical * 0.000001) + 0.000000001 AS ?verticalTolerance)

                FILTER (
                    ?longitudinal < (?expectedLongitudinal - ?longitudinalTolerance) ||
                    ?longitudinal > (?expectedLongitudinal + ?longitudinalTolerance) ||
                    ?vertical < (?expectedVertical - ?verticalTolerance) ||
                    ?vertical > (?expectedVertical + ?verticalTolerance) ||
                    ?position < 0.0 ||
                    ?position > (?draught * 1.20)
                )
            }
        """
