<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO066ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO066NotExposedBranch
        gen:IMO066ExposedBranch
    ) .

gen:IMO066NotExposedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The ship's ice-accretion applicability must be recorded as false when this conditional requirement is not applicable."
    ] .

gen:IMO066ExposedBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship's ice-accretion applicability must be recorded as true when this conditional requirement applies."
    ] ;
    sh:property [
        sh:path nltl:hasProtectedItem ;
        sh:minCount 1 ;
        sh:node gen:IMO066ProtectedItemShape ;
        sh:message "For a ship exposed to ice accretion, protected-item inventory and associated ice or snow removal or prevention means are required."
    ] .

gen:IMO066ProtectedItemShape a sh:NodeShape ;
    sh:class nltl:protectedItem ;
    sh:property [
        sh:path nltl:protectedItemCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:escapeRouteProtectedItem
            nltl:musterStationProtectedItem
            nltl:embarkationAreaProtectedItem
            nltl:survivalCraftProtectedItem
            nltl:launchingApplianceProtectedItem
            nltl:survivalCraftAccessProtectedItem
        ) ;
        sh:message "Each protected item must identify one permitted escape, muster, embarkation, survival-craft, launching-appliance, or survival-craft-access category."
    ] ;
    sh:property [
        sh:path nltl:hasIceSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceSnowRemovalOrPreventionMeans
        ] ;
        sh:message "Each protected item must have associated means to remove or prevent ice and snow accretion."
    ] .
<END_SHACL>
