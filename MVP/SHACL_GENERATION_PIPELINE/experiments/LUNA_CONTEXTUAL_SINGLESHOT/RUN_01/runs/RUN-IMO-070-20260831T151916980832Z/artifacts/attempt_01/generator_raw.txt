<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-070> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each person on board a passenger ship must have an assigned immersion suit or thermal protective aid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            SELECT $this WHERE {
                $this a nltl:passengerShip ;
                      nltl:hasPersonOnBoardMember ?person .
                FILTER NOT EXISTS {
                    ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                    ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment .
                    ?equipment rdf:type ?equipmentType .
                    FILTER (?equipmentType IN (
                        nltl:immersionSuitEquipment,
                        nltl:thermalProtectiveAidEquipment
                    ))
                }
            }
        """ ;
    ] ;
    sh:sparql [
        sh:message "A passenger ship must record that the assigned survival equipment is properly sized." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this WHERE {
                $this a nltl:passengerShip .
                {
                    FILTER NOT EXISTS {
                        $this nltl:sizeCompatibilityStatus ?status .
                    }
                }
                UNION
                {
                    $this nltl:sizeCompatibilityStatus ?status .
                    FILTER (?status != true)
                }
            }
        """ ;
    ] .
<END_SHACL>
