<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_086_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:node gen:IMO_086_HeadingMeansShape ;
        sh:message "The ship shall have at least two non-magnetic means to determine and display heading."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means shall be independent from another distinct non-magnetic heading means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    FILTER (?otherMeans != ?means)
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                }
            }
        """
    ] .

gen:IMO_086_HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's main source of power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's emergency source of power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:message "Each non-magnetic heading means shall identify an independent heading means."
    ] .
<END_SHACL>
