<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_095_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:searchAndRescueCommunicationCapability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall declare search and rescue communication capability."
    ] ;
    sh:property [
        sh:path nltl:rescueCoordinationCentreVoiceOrData ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall provide voice or data communications with relevant rescue coordination centres."
    ] ;
    sh:property [
        sh:path nltl:supportedAircraftVoiceFrequency ;
        sh:minCount 2 ;
        sh:node gen:IMO_095_FrequencyQuantityShape ;
        sh:message "The ship shall support aircraft voice communications on both 121.5 MHz and 123.1 MHz."
    ] ;
    sh:sparql [
        sh:message "The supported aircraft voice frequencies shall include 121.5 MHz." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:supportedAircraftVoiceFrequency ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:MegaHZ .
                    FILTER(?value = 121.5)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The supported aircraft voice frequencies shall include 123.1 MHz." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:supportedAircraftVoiceFrequency ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:MegaHZ .
                    FILTER(?value = 123.1)
                }
            }
        """
    ] .

gen:IMO_095_FrequencyQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaHZ
    ] .
<END_SHACL>
