<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO098OtherSurvivalCraftShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasLocationSignalDevice ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one location-signal device."
    ] ;
    sh:property [
        sh:path nltl:hasOnSceneCommunicationDevice ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one on-scene transmit/receive communication device."
    ] .

gen:IMO098ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:nodeKind sh:IRI ;
        sh:node gen:IMO098OtherSurvivalCraftShape
    ] .

gen:IMO098NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:hasValue false
    ] .

gen:IMO098ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's low-air-temperature operating status must be recorded as a single boolean value."
    ] ;
    sh:or (
        gen:IMO098ApplicableShape
        gen:IMO098NotApplicableShape
    ) .
<END_SHACL>
