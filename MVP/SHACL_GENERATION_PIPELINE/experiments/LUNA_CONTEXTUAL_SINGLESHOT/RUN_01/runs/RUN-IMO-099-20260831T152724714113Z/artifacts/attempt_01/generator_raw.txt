<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-099-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery-life management procedure must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery-life management procedure must be recorded as implemented and active."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-099-AvailabilityQuantityShape> ;
        sh:message "Communication device availability must be provided in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-099-RescueTimeQuantityShape> ;
        sh:message "The maximum expected rescue time must be provided in days."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Communication device availability must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availabilityQuantity .
                ?availabilityQuantity qudt:numericValue ?availability .
                ?availabilityQuantity qudt:unit <http://qudt.org/vocab/unit/DAY> .

                $this nltl:maximumExpectedRescueTime ?rescueQuantity .
                ?rescueQuantity qudt:numericValue ?rescueTime .
                ?rescueQuantity qudt:unit <http://qudt.org/vocab/unit/DAY> .

                FILTER(?availability < ?rescueTime)
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-099-AvailabilityQuantityShape> a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DAY>
    ] .

<urn:nltl:generated-shape:IMO-099-RescueTimeQuantityShape> a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DAY>
    ] .
<END_SHACL>
