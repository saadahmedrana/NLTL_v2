<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-104-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization status."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node <urn:nltl:generated-shape:IMO-104-CrewMemberShape>
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:node <urn:nltl:generated-shape:IMO-104-CrewMemberShape>
    ] ;
    sh:sparql [
        sh:message "Each assigned duty's relevant PWOM procedure or equipment must be represented in the crew member's familiarization records." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-104-CrewMemberShape> a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:node <urn:nltl:generated-shape:IMO-104-AssignedDutyShape>
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-FamiliarizationRecordShape> ;
        sh:message "Each crew member must have a PWOM familiarization record."
    ] .

<urn:nltl:generated-shape:IMO-104-AssignedDutyShape> a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-ManualItemShape> ;
        sh:message "Each represented assigned duty must identify its relevant PWOM procedure or equipment."
    ] .

<urn:nltl:generated-shape:IMO-104-FamiliarizationRecordShape> a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-104-ManualItemShape> ;
        sh:message "A familiarization record must identify the PWOM procedure or equipment covered."
    ] .

<urn:nltl:generated-shape:IMO-104-ManualItemShape> a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem ;
    sh:property [
        sh:path nltl:relevantPolarWaterOperationalManualProcedureOrEquipment ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
