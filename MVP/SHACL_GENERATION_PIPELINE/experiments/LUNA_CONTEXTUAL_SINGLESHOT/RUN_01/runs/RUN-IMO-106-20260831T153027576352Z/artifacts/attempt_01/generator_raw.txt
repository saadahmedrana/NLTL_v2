<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO106ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:message "A discharged material classification is required to determine whether the paragraph 1.1.1 prohibition applies."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean ;
        sh:message "The paragraph 1.1.1 applicability indicator must be a boolean."
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast, paragraph 1.1.1 oil-discharge prohibition applicability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material = nltl:cleanBallastMaterial ||
                    ?material = nltl:segregatedBallastMaterial
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicableFalse .
                        FILTER (?applicableFalse = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicableOther .
                        FILTER (?applicableOther != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
