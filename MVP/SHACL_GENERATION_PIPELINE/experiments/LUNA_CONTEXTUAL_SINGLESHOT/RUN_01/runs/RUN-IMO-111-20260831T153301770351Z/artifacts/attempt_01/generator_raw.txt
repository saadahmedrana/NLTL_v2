<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_111_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship constructed on or after 1 January 2017, each oil residue or oily bilge water holding tank with capacity greater than 30 m3 must be separated from the outer shell by at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipCategory ?category .
    FILTER (?category IN (
        nltl:polarShipCategoryA,
        nltl:polarShipCategoryB
    )) .

    $this nltl:constructionDate ?constructionDate .
    FILTER (?constructionDate >= "2017-01-01"^^xsd:date) .

    $this nltl:tankType ?tankType .
    FILTER (?tankType IN (
        "oil_residue_tank",
        "oily_bilge_water_holding_tank"
    )) .

    $this nltl:individualTankCapacity ?capacity .
    ?capacity qudt:numericValue ?capacityValue ;
              qudt:unit unit:M3 .
    FILTER (?capacityValue > 30) .

    FILTER NOT EXISTS {
        $this nltl:tankOuterShellSeparation ?separation .
        ?separation qudt:numericValue ?separationValue ;
                    qudt:unit unit:M .
        FILTER (?separationValue >= 0.76)
    }
}
"""
    ] .

</END_SHACL>
