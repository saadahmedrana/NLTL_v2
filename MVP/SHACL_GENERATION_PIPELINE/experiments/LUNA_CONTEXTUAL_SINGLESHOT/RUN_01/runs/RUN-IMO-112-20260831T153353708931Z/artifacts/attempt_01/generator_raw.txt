<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_112_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An operating area is required to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of noxious liquid substances or mixtures containing them must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (?area = "Arctic_waters")
                FILTER NOT EXISTS {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea false .
                }
            }
        """
    ] .
<END_SHACL>
