<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:arcticWaters ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When operating in Arctic_waters and discharging food waste to sea, the ship must provide a distance of at least 12 nautical miles, ice-concentration data above 1/10, and the recorded distance from that area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea true .

                FILTER (?area = "Arctic_waters")

                FILTER (
                    NOT EXISTS {
                        $this nltl:distanceToNearestLandIceShelfOrFastIce ?distance .
                        ?distance qudt:numericValue ?distanceValue ;
                                  qudt:unit unit:MI_N .
                        FILTER (
                            datatype(?distanceValue) = xsd:decimal &&
                            ?distanceValue >= "12"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearby .
                        ?nearby qudt:numericValue ?nearbyValue ;
                                qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?nearbyValue) = xsd:decimal &&
                            ?nearbyValue > "0.1"^^xsd:decimal
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceConcentration ?ice .
                        ?ice qudt:numericValue ?iceValue ;
                             qudt:unit unit:UNITLESS .
                        FILTER (datatype(?iceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?recordedDistance .
                        ?recordedDistance qudt:numericValue ?recordedValue ;
                                           qudt:unit unit:MI_N .
                        FILTER (datatype(?recordedValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
