<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_124_StaticShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestFastIce ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an Antarctic-area MARPOL Annex V regulation 6.1 discharge, the nearest-fast-ice distance must be at least 12 nautical miles, the distance from an area with ice concentration above 1/10 must be recorded, and food waste must not be discharged onto ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?this
WHERE {
    ?this nltl:operatingArea ?operatingArea ;
          nltl:antarcticArea ?antarcticArea ;
          nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .
    FILTER (
        ?operatingArea = "Antarctic_area"^^xsd:string &&
        ?antarcticArea = "Antarctic_area"^^xsd:string &&
        ?discharge = true
    )
    FILTER NOT EXISTS {
        ?this nltl:distanceToNearestFastIce ?nearestFastIce ;
              nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?iceAreaDistance ;
              nltl:nearbyIceConcentration ?nearbyConcentration ;
              nltl:foodWasteDischargeOntoIce false .

        ?nearestFastIce qudt:numericValue ?nearestValue ;
                        qudt:unit unit:MI_N .
        FILTER (xsd:decimal(?nearestValue) >= 12)

        ?nearbyConcentration qudt:numericValue ?nearbyValue ;
                             qudt:unit unit:UNITLESS .
        FILTER (xsd:decimal(?nearbyValue) > 1)

        ?iceAreaDistance qudt:numericValue ?iceAreaDistanceValue ;
                         qudt:unit unit:MI_N .
        FILTER (xsd:decimal(?iceAreaDistanceValue) >= 0)
    }
}
"""
    ] .
</END_SHACL>
