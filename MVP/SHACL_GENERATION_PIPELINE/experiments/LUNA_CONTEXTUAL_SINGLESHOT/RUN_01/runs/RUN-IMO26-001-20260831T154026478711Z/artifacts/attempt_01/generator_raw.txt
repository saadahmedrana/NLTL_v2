<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-001> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "SOLAS chapter I certification status is required to establish Polar Code chapter 9 applicability."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Polar Code chapter 9 may apply only to a ship certified in accordance with SOLAS chapter I." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter9Applicable ?applicable .
                FILTER ( ?applicable = true )
                FILTER NOT EXISTS {
                    $this nltl:solasChapterICertified ?certified .
                    FILTER ( ?certified = true )
                }
            }
        """
    ] .
<END_SHACL>
