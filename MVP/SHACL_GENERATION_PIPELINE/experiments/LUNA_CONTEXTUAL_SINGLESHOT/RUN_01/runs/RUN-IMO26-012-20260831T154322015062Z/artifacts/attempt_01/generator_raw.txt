<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012-ship> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node [
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means must be connected to main power."
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means must be connected to emergency power."
            ] ;
            sh:property [
                sh:path nltl:independentFromHeadingMeans ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:nonMagneticHeadingMeans ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means must identify an independent heading means."
            ] ;
            sh:sparql [
                sh:message "Each non-magnetic heading means must be independent from another heading means belonging to the same ship." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            ?ship nltl:hasNonMagneticHeadingMeans $this .
                            $this nltl:independentFromHeadingMeans ?other .
                            FILTER (?other != $this)
                            ?ship nltl:hasNonMagneticHeadingMeans ?other .
                        }
                    }
                """
            ]
        ]
    ] .
<END_SHACL>
