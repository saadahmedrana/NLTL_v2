<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "The construction contract date must be an xsd:date when supplied."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulatoryProvisionValue ;
        sh:in (
            nltl:iceClassSection1Point8Provision2021
            nltl:iceClassDraughtChapter2Provision2021
        ) ;
        sh:minCount 2 ;
        sh:maxCount 2 ;
        sh:message "Both the 2021 section 1.8 Ice Classes provision and chapter 2 Ice Class Draught provision must apply."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue
    ] ;
    sh:sparql [
        sh:message "The 2021 Ice Class Regulations must be identified for ships contracted for construction on or after 5 July 2021, and must not be identified for ships contracted before that date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    (
                        ?date >= "2021-07-05"^^xsd:date
                        && NOT EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRegulationEdition2021 .
                        }
                    )
                    ||
                    (
                        ?date < "2021-07-05"^^xsd:date
                        && EXISTS {
                            $this nltl:applicableIceClassRegulationEdition
                                nltl:iceClassRegulationEdition2021 .
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
