<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF004Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine whether the 2008 Ice Class Regulations apply."
    ] ;
    sh:sparql [
        sh:message "Ships contracted for construction on or after 1 January 2010 but before 1 January 2012 shall identify the 2008 Ice Class Regulations as the applicable regulation edition." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionContractDate ?date .
    BIND(YEAR(?date) AS ?year)
    BIND(MONTH(?date) AS ?month)
    BIND(DAY(?date) AS ?day)
    FILTER(
        (?year > 2010 ||
         (?year = 2010 && (?month > 1 || (?month = 1 && ?day >= 1)))) &&
        (?year < 2012 ||
         (?year = 2012 && (?month < 1 || (?month = 1 && ?day < 1))))
    )
    FILTER NOT EXISTS {
        $this nltl:applicableIceClassRegulationEdition
              nltl:iceClassRegulationEdition2008
    }
}
""" ;
    ] .
<END_SHACL>
