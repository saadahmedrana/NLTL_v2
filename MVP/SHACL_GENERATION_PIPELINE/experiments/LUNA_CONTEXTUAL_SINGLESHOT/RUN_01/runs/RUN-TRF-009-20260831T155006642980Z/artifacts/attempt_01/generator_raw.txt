<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of TRF-009."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A ship must provide its construction stage date to determine applicability of TRF-009."
    ] ;
    sh:or (
        gen:TRF009ApplicableShape
        gen:TRF009NotApplicableShape
    ) .

gen:TRF009ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An applicable ship must provide its delivery date for the TRF-009 deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An applicable ship must provide an assessment date for the TRF-009 deadline."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "TRF-009 applies to an IA Super or IA ship constructed before 1 September 2003, and the ship must not be assessed before 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                       nltl:constructionStageDate ?construction ;
                       nltl:deliveryDate ?delivery ;
                       nltl:assessmentDate ?assessment .
                FILTER (
                    ?ice IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    YEAR(?construction) < 2003 ||
                    (
                        YEAR(?construction) = 2003 &&
                        (
                            MONTH(?construction) < 9 ||
                            (
                                MONTH(?construction) = 9 &&
                                DAY(?construction) < 1
                            )
                        )
                    )
                )
                FILTER (
                    YEAR(?assessment) > YEAR(?delivery) + 20 ||
                    (
                        YEAR(?assessment) = YEAR(?delivery) + 20 &&
                        (
                            MONTH(?assessment) > 1 ||
                            (
                                MONTH(?assessment) = 1 &&
                                DAY(?assessment) >= 1
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:TRF009NotApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The ship is neither IA Super or IA constructed before 1 September 2003, so the conditional TRF-009 obligation is not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                       nltl:constructionStageDate ?construction .
                FILTER (
                    NOT (
                        ?ice IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa
                        )
                        &&
                        (
                            YEAR(?construction) < 2003 ||
                            (
                                YEAR(?construction) = 2003 &&
                                (
                                    MONTH(?construction) < 9 ||
                                    (
                                        MONTH(?construction) = 9 &&
                                        DAY(?construction) < 1
                                    )
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
