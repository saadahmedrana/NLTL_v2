<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For a ship of ice class IB or IC subject to the 1985 ice class regulations, maximum continuous rating power must be at least the required power under the 1985 rules, with both quantities expressed in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRegulationEdition2002 ;
                      nltl:maximumContinuousRatingPower ?mcr ;
                      nltl:requiredPowerUnder1985Rules ?requiredPower .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?validMcr .
                        ?validMcr qudt:numericValue ?mcrValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        FILTER (datatype(?mcrValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPowerUnder1985Rules ?validRequiredPower .
                        ?validRequiredPower qudt:numericValue ?requiredValue ;
                                             qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        FILTER (datatype(?requiredValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?validMcr ;
                              nltl:requiredPowerUnder1985Rules ?validRequiredPower .
                        ?validMcr qudt:numericValue ?mcrValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        ?validRequiredPower qudt:numericValue ?requiredValue ;
                                            qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        FILTER (
                            datatype(?mcrValue) = xsd:decimal &&
                            datatype(?requiredValue) = xsd:decimal &&
                            ?mcrValue >= ?requiredValue
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
