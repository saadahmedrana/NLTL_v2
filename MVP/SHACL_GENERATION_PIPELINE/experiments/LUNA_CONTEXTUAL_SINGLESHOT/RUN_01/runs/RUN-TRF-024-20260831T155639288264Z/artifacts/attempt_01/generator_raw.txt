<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF024Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1 ;
                sh:message "A non-IA or non-IA Super ice class must be identified."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Construction stage date is required to determine applicability."
            ] ;
            sh:sparql [
                sh:message "For a non-IA or non-IA Super ship, the construction stage date must not be before 1 September 2003." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate .
                        FILTER (?stageDate >= "2003-09-01"^^xsd:date)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                ) ;
                sh:minCount 1 ;
                sh:message "The ship's ice class must be IA Super or IA for this applicability branch."
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Construction stage date is required to determine applicability."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Delivery date is required to determine the twenty-year deadline."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Assessment date is required to verify the compliance deadline."
            ] ;
            sh:sparql [
                sh:message "For an IA Super or IA ship constructed before 1 September 2003, compliance must be established no later than 1 January of the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (?stageDate < "2003-09-01"^^xsd:date)
                        FILTER (
                            YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                && MONTH(?assessmentDate) = 1
                                && DAY(?assessmentDate) = 1
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
