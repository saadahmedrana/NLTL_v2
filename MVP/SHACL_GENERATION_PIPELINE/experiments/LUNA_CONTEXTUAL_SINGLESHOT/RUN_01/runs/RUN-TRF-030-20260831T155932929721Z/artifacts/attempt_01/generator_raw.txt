<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the ice pressure input in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the load patch height in metres."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the externally determined applied ice pressure in MPa."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:class nltl:directAnalysisCase ;
                sh:node gen:DirectAnalysisCaseShape ;
                sh:minCount 2 ;
                sh:message "At least several direct analysis cases are required, including vertical and horizontal load positions."
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:class nltl:directAnalysisCase ;
                sh:node gen:ArrangementDependentAnalysisCaseShape ;
                sh:minCount 2 ;
                sh:message "When load length is not determined from the arrangement, several paired load-length and ca analysis cases are required."
            ]
        ]
    ) .

gen:DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:minCount 1 ;
        sh:or (
            [ sh:hasValue nltl:upperIceWaterlineCenteredPosition ]
            [ sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition ]
            [ sh:hasValue nltl:intermediateVerticalPosition ]
        )
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:minCount 1 ;
        sh:or (
            [ sh:hasValue nltl:midSpanOrSpacingCenteredPosition ]
            [ sh:hasValue nltl:otherHorizontalPosition ]
            [ sh:hasValue nltl:upperIceWaterlineCenteredPosition ]
        )
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] .

gen:ArrangementDependentAnalysisCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
