<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        gen:TRF031BeamTheoryBranch
        gen:TRF031NonBeamTheoryBranch
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be lower than the yield point, and yield shear stress must equal yield point divided by sqrt(3) within a relative tolerance of 1e-6 with a small absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldNode ;
                      nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldShearStress ?shearNode .
                ?yieldNode qudt:numericValue ?yield .
                ?combinedNode qudt:numericValue ?combined .
                ?shearNode qudt:numericValue ?shear .
                FILTER (?combined >= ?yield)
                FILTER (
                    ABS(?shear - (?yield / 1.7320508075688772935))
                    >
                    (ABS(?yield / 1.7320508075688772935) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .

gen:TRF031BeamTheoryBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:allowableShearStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must be no larger than 0.9 times yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed ?beam ;
                      nltl:allowableShearStress ?allowableNode ;
                      nltl:yieldShearStress ?yieldShearNode .
                ?allowableNode qudt:numericValue ?allowable .
                ?yieldShearNode qudt:numericValue ?yieldShear .
                FILTER (?beam = "true"^^xsd:boolean)
                FILTER (?allowable > (0.9 * ?yieldShear))
            }
        """
    ] .

gen:TRF031NonBeamTheoryBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .
</END_SHACL>
