<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-035-ShipRuleLengthShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall have one rule length quantity expressed in metres."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall have one classification-society rule length quantity expressed in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length shall equal the rule length used by the classification society." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ruleLength ?ruleLengthQuantity .
                ?ruleLengthQuantity qudt:numericValue ?ruleLengthValue .
                $this nltl:classificationSocietyRuleLength ?societyRuleLengthQuantity .
                ?societyRuleLengthQuantity qudt:numericValue ?societyRuleLengthValue .
                FILTER (?ruleLengthValue != ?societyRuleLengthValue)
            }
        """
    ] .
<END_SHACL>
