<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF_050_FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:class nltl:weldTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "Each frame attachment record shall specify a double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each frame attachment record shall state whether a shell plate butt is crossed."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "Scalloping is not allowed when the frame attachment does not cross a shell plate butt."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each frame attachment record shall be linked from a frame by frameAttachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?frame nltl:frameAttachment $this .
                }
            }
        """
    ] .
<END_SHACL>
