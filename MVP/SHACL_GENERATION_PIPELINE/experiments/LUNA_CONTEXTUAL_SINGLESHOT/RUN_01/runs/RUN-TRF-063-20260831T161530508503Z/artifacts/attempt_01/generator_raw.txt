<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:class nltl:sidePropellerStrengtheningEnvelope ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:message "Each side-propeller strengthening envelope shall identify a tank top boundary."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The forward strengthening extent shall be at least 1.5 metres."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The aft strengthening extent shall be at least 1.5 metres."
    ] .

gen:TRF063ForwardExtentThreshold
    a sh:NodeShape ;
    sh:targetClass nltl:sidePropellerStrengtheningEnvelope ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive "1.5"^^xsd:decimal
            ]
        ]
    ] .

gen:TRF063AftExtentThreshold
    a sh:NodeShape ;
    sh:targetClass nltl:sidePropellerStrengtheningEnvelope ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive "1.5"^^xsd:decimal
            ]
        ]
    ] .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide a single propeller count to establish applicability."
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:node gen:TRF063EnvelopeShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A twin- or triple-screw ship shall provide a side-propeller strengthening envelope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count .
                FILTER (?count IN (2, 3))
                FILTER NOT EXISTS {
                    $this nltl:hasSidePropellerStrengtheningEnvelope ?envelope .
                }
            }
        """
    ] .
</END_SHACL>
