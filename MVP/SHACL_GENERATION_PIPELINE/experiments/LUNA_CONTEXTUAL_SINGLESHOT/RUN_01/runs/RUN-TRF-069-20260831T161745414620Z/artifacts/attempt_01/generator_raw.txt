<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF-069 a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable propulsion-machinery configuration, the normal service-life propeller ice-load scope and all stated load-model exclusions shall be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
                FILTER (?propellerType IN ("open", "ducted"))
                FILTER (?pitchType IN ("controllable", "fixed"))

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?condition .
                    ?condition a nltl:designCondition ;
                               nltl:normalOperationalConditions true ;
                               nltl:normalServiceLifePropellerIceLoadCondition true ;
                               nltl:fixedPitchPropellerReversalLoadIncluded true ;
                               nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                               nltl:radialIceEntryOutsideLoadModel true .
                }
            }
        """
    ] .
</BEGIN_SHACL>
