<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-001> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one construction contract date to determine UR I2 Rev.4 applicability."
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must record exactly one UR I2 Rev.4 applicability value."
    ] ;
    sh:sparql [
        sh:message "UR I2 Rev.4 applicability must be true for ships contracted for construction on or after 1 January 2021 and false otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date ;
                      nltl:urI2RevisionApplicability ?applicability .
                FILTER (
                    (
                        (
                            YEAR(?date) > 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) > 1) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 1 && DAY(?date) >= 1)
                        ) &&
                        ?applicability != true
                    ) ||
                    (
                        (
                            YEAR(?date) < 2021 ||
                            (YEAR(?date) = 2021 && MONTH(?date) < 1) ||
                            (YEAR(?date) = 2021 && MONTH(?date) = 1 && DAY(?date) < 1)
                        ) &&
                        ?applicability != false
                    )
                )
            }
        """
    ] .
<END_SHACL>
