<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:I2_011_ApplicableShape
        gen:I2_011_BulbousBowAbsentShape
        gen:I2_011_NonPolarClassShape
    ) .

gen:I2_011_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship subject to the PC6 or PC7 bulbous-bow rule must have exactly one applicable polar class."
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship subject to the applicable rule must record that a bulbous bow is present."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The I2.3.2.1(iii) bow design force must be supplied as a single force quantity in MN."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The I2.3.2.1(iv) bow design force must be supplied as a single force quantity in MN."
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The shape coefficient must be supplied as a single dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 1.3 ;
                sh:maxInclusive 1.3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "For the lower-bound calculation, the load-patch aspect ratio must be exactly 1.3 and dimensionless."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The bow design force must be supplied as a single force quantity in kN."
    ] ;
    sh:sparql [
        sh:message "The bow design force must equal, within numerical tolerance, the greater of the I2.3.2.1(iii) and I2.3.2.1(iv) force values, converted from MN to kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode ;
                      nltl:bowDesignForce ?resultNode .

                ?iiiNode qudt:numericValue ?iii ;
                         qudt:unit qudt:MegaN .
                ?ivNode qudt:numericValue ?iv ;
                        qudt:unit qudt:MegaN .
                ?resultNode qudt:numericValue ?result ;
                            qudt:unit qudt:KiloN .

                BIND(IF(?iii >= ?iv, ?iii, ?iv) * 1000.0 AS ?expected)

                FILTER (
                    (?expected <= 0.000001 &&
                        (?result < ?expected - 0.000001 ||
                         ?result > ?expected + 0.000001))
                    ||
                    (?expected > 0.000001 &&
                        (?result < ?expected * 0.999999 ||
                         ?result > ?expected * 1.000001))
                )
            }
        """
    ] .

gen:I2_011_BulbousBowAbsentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_011_NonPolarClassShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
