<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_014_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The bow region must be divided into four sub-regions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:node gen:I2_014_BowSubregionCaseShape ;
        sh:message "Four bow subregion calculation cases, each with the required calculation evidence and reported values, are required."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_ForceQuantityShape ;
        sh:message "A selected maximum bow force is required."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_LineLoadQuantityShape ;
        sh:message "A selected maximum bow line load is required."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_PressureQuantityShape ;
        sh:message "A selected maximum bow pressure is required."
    ] ;
    sh:sparql [
        sh:message "The selected maximum bow force, line load, and pressure must equal the respective maxima reported for the four bow subregion calculation cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this
                    nltl:selectedMaximumBowForce ?selectedForceNode ;
                    nltl:selectedMaximumBowLineLoad ?selectedLineLoadNode ;
                    nltl:selectedMaximumBowPressure ?selectedPressureNode .

                ?selectedForceNode qudt:numericValue ?selectedForce .
                ?selectedLineLoadNode qudt:numericValue ?selectedLineLoad .
                ?selectedPressureNode qudt:numericValue ?selectedPressure .

                FILTER (
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?forceCase .
                        ?forceCase nltl:bowSubregionCalculatedForce ?forceNode .
                        ?forceNode qudt:numericValue ?forceValue .
                        FILTER NOT EXISTS {
                            $this nltl:hasBowSubregionCalculationCase ?greaterForceCase .
                            ?greaterForceCase nltl:bowSubregionCalculatedForce ?greaterForceNode .
                            ?greaterForceNode qudt:numericValue ?greaterForce .
                            FILTER (?greaterForce > ?forceValue)
                        }
                        FILTER (?selectedForce != ?forceValue)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?lineLoadCase .
                        ?lineLoadCase nltl:bowSubregionCalculatedLineLoad ?lineLoadNode .
                        ?lineLoadNode qudt:numericValue ?lineLoadValue .
                        FILTER NOT EXISTS {
                            $this nltl:hasBowSubregionCalculationCase ?greaterLineLoadCase .
                            ?greaterLineLoadCase nltl:bowSubregionCalculatedLineLoad ?greaterLineLoadNode .
                            ?greaterLineLoadNode qudt:numericValue ?greaterLineLoad .
                            FILTER (?greaterLineLoad > ?lineLoadValue)
                        }
                        FILTER (?selectedLineLoad != ?lineLoadValue)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?pressureCase .
                        ?pressureCase nltl:bowSubregionCalculatedPressure ?pressureNode .
                        ?pressureNode qudt:numericValue ?pressureValue .
                        FILTER NOT EXISTS {
                            $this nltl:hasBowSubregionCalculationCase ?greaterPressureCase .
                            ?greaterPressureCase nltl:bowSubregionCalculatedPressure ?greaterPressureNode .
                            ?greaterPressureNode qudt:numericValue ?greaterPressure .
                            FILTER (?greaterPressure > ?pressureValue)
                        }
                        FILTER (?selectedPressure != ?pressureValue)
                    }
                )
            }
        """
    ] .

gen:I2_014_BowSubregionCaseShape
    a sh:NodeShape ;
    sh:class nltl:bowSubregionCalculationCase ;
    sh:property [
        sh:path nltl:bowSubregionCalculationEvidence ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "Each bow subregion calculation case must have calculation evidence."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionMidLengthPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_LengthQuantityShape ;
        sh:message "Each bow subregion calculation case must report its mid-length position."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_DimensionlessQuantityShape ;
        sh:message "Each bow subregion calculation case must report its load patch aspect ratio."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCalculatedForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_ForceQuantityShape ;
        sh:message "Each bow subregion calculation case must report force."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCalculatedLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_LineLoadQuantityShape ;
        sh:message "Each bow subregion calculation case must report line load."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCalculatedPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_PressureQuantityShape ;
        sh:message "Each bow subregion calculation case must report pressure."
    ] .
