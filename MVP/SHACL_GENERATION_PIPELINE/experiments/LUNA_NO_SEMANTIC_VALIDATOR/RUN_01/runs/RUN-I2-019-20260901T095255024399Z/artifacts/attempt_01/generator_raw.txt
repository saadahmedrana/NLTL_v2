<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberShape
    ] .

gs:StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:node gs:HullAreaValueShape
    ] ;
    sh:sparql [
        sh:message "A structural member spanning one or more hull areas must have a selected hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSpannedHullArea ?area .
                FILTER NOT EXISTS {
                    $this nltl:selectedHullAreaFactor ?selectedNode .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected hull area factor must equal the largest hull area factor among the hull areas spanned by the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedHullAreaFactor ?selectedNode .
                ?selectedNode qudt:numericValue ?selectedValue .
                $this nltl:hasSpannedHullArea ?area .
                ?area nltl:hullAreaFactor ?factorNode .
                ?factorNode qudt:numericValue ?factorValue .
            }
            GROUP BY $this ?selectedValue
            HAVING (?selectedValue != MAX(?factorValue))
        """
    ] .

gs:HullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:HullAreaFactorQuantityShape
    ] .

gs:HullAreaFactorQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:SelectedHullAreaFactorQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
