gen:I2_022_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_022_PressureQuantityShape ;
        sh:message "The ship shall provide exactly one average ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_022_DimensionlessQuantityShape ;
        sh:message "The ship shall provide exactly one peak pressure factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_022_PressureQuantityShape ;
        sh:message "The ship shall provide exactly one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_022_LengthQuantityShape ;
        sh:message "The ship shall provide exactly one load patch height in metres."
    ] ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:plating ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:framingAngleOmega ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:I2_022_ApplicableAngleQuantityShape
                    ] ;
                    sh:property [
                        sh:path nltl:platingHullAreaClassification ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:in (
                            nltl:bowIntermediateBottomHullArea
                            nltl:midbodyBottomHullArea
                            nltl:sternBottomHullArea
                        )
                    ] ;
                    sh:property [
                        sh:path nltl:iceLoadRequiredNetPlateThickness ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:I2_022_LengthMillimetreQuantityShape ;
                        sh:message "Applicable plating shall provide the externally calculated net plate thickness in millimetres."
                    ] ;
                    sh:property [
                        sh:path nltl:platingSupportedByStructuralMember ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:structuralMember ;
                            sh:property [
                                sh:path nltl:frameSpacing ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node gen:I2_022_LengthQuantityShape
                            ] ;
                            sh:property [
                                sh:path nltl:selectedHullAreaFactor ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:node gen:I2_022_DimensionlessQuantityShape
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:framingAngleOmega ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gen:I2_022_NonApplicableAngleQuantityShape
                    ]
                ]
            )
        ]
    ] .
