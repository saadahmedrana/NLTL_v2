<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_029_CalculationCaseShape ;
        sh:message "Each calculation case must satisfy the I2.5.6 section-modulus and shear-area constraints."
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be included in the section modulus calculation."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be excluded from the shear-area calculation."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Flange-material inclusion in the shear area must be explicitly represented."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If flange material is included in the shear area, the linked structural member must be fitted with a flange." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:flangeMaterialIncludedInShearArea true .
                FILTER NOT EXISTS {
                    $this nltl:sectionCalculationCaseStructuralMember ?member .
                    ?member a nltl:structuralMember .
                    ?member nltl:flangeFitted true .
                }
            }
        """
    ] .
</END_SHACL>
