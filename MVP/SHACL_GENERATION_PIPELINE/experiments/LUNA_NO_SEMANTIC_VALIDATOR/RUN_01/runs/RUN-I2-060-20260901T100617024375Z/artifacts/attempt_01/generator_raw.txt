<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_060_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_060_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member in way of a cutout must not be unstable." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    ?member nltl:memberInWayOfCutout true .
    FILTER (
        NOT EXISTS {
            $this nltl:memberInstability ?state .
            FILTER (?state = false)
        }
        || EXISTS {
            $this nltl:memberInstability true
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where stiffening is necessary for a member in way of a cutout, structural stiffening must be recorded." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    ?member nltl:memberInWayOfCutout true .
    ?member nltl:stiffeningNecessary true .
    FILTER NOT EXISTS {
        $this nltl:structuralStiffening ?stiffening .
        FILTER (datatype(?stiffening) = xsd:string)
    }
}
"""
    ] .

gen:I2_060_StructuralMemberShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasCutout ;
                sh:minCount 1 ;
                sh:class nltl:cutout
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean
    ] .
<END_SHACL>
