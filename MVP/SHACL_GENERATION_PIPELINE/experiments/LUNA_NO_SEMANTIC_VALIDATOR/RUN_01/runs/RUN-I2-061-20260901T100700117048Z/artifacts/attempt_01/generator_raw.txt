<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node gen:CalculationCaseShape ;
        sh:message "The ship must identify a calculation case for the prescribed hull-structure procedure."
    ] .

gen:CalculationCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must use the prescribed analytical procedure and must not use direct calculation as an alternative."
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:node gen:AssessedHullStructureShape ;
        sh:message "The calculation case must identify its assessed hull structure."
    ] .

gen:AssessedHullStructureShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:ThicknessQuantityShape ;
        sh:message "The assessed hull structure must provide its thickness as a quantity value."
    ] .

gen:ThicknessQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Thickness must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness must be expressed in metres."
    ] .
<END_SHACL>
