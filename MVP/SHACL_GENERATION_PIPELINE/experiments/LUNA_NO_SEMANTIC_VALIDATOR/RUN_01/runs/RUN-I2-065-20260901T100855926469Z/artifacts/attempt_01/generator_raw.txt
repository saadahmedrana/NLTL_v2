<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue nltl:nonlinearCalculationMethodValue
                    ] ;
                    sh:property [
                        sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                        sh:class nltl:evidenceArtifact ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:permanentLateralDeformation ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:MilliM ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:permanentOutOfPlaneDeformation ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:MilliM ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:relevantStructuralDimension ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:MilliM ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue nltl:linearCalculationMethodValue
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue nltl:directCalculationMethodValue
                    ]
                ]
            )
        ]
    ] ;

    sh:property [
        sh:path nltl:fractureMargin ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:majorBucklingMargin ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:yieldingMargin ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:sparql [
        sh:message "A ship with a nonlinear calculation case must provide fracture, major buckling, and yielding margins." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
                FILTER (
                    NOT EXISTS { $this nltl:fractureMargin ?fractureMargin }
                    || NOT EXISTS { $this nltl:majorBucklingMargin ?majorBucklingMargin }
                    || NOT EXISTS { $this nltl:yieldingMargin ?yieldingMargin }
                )
            }
        """
    ] .
<END_SHACL>
