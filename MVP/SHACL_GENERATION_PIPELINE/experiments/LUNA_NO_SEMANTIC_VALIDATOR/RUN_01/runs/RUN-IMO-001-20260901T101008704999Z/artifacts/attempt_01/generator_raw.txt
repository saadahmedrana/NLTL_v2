<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-001-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its design ice condition to determine applicability of the Category A rule."
    ] ;
    sh:sparql [
        sh:message "When the design ice condition is at least medium first-year ice with possible old-ice inclusions, the ship category must be Category A and must not include another category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition
                    nltl:mediumFirstYearIceWithPossibleOldIceInclusions .
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipCategory nltl:polarShipCategoryA .
                    }
                    ||
                    EXISTS {
                        $this nltl:shipCategory ?category .
                        FILTER (?category != nltl:polarShipCategoryA)
                    }
                )
            }
        """
    ] .
<END_SHACL>
