<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO007Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice condition classification to determine whether the open-water condition applies."
    ] ;
    sh:sparql [
        sh:message "For an open-water ice condition, sea ice concentration must be less than 1/10 with unitless quantity data, and ice of land origin must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?value ;
                                        qudt:unit qudt:UNITLESS .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value < 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentrationNode .
                        FILTER NOT EXISTS {
                            ?concentrationNode qudt:numericValue ?concentrationValue ;
                                                qudt:unit qudt:UNITLESS .
                            FILTER (
                                datatype(?concentrationValue) = xsd:decimal &&
                                ?concentrationValue < 1
                            )
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent false .
                        FILTER (datatype(false) = xsd:boolean)
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?landOriginValue .
                        FILTER NOT EXISTS {
                            FILTER (
                                datatype(?landOriginValue) = xsd:boolean &&
                                ?landOriginValue = false
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
