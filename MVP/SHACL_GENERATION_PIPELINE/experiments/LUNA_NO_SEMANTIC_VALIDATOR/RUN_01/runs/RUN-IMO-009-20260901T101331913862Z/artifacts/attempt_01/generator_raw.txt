<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-009-BergyWaters> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "A ship must provide an ice condition to determine whether the bergy-waters limits apply."
    ] ;
    sh:sparql [
        sh:message "For bergy waters, land-origin ice concentration must be less than 1/10 and total ice concentration must not exceed 1/10, with dimensionless quantity values." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER (STR(?condition) = "bergy waters")
    FILTER NOT EXISTS {
        $this nltl:landOriginIceConcentration ?landConcentration .
        ?landConcentration a qudt:QuantityValue ;
            qudt:numericValue ?landValue ;
            qudt:unit unit:UNITLESS .
        FILTER (datatype(?landValue) = xsd:decimal)
        FILTER (?landValue < 1)

        $this nltl:totalIceConcentration ?totalConcentration .
        ?totalConcentration a qudt:QuantityValue ;
            qudt:numericValue ?totalValue ;
            qudt:unit unit:UNITLESS .
        FILTER (datatype(?totalValue) = xsd:decimal)
        FILTER (?totalValue <= 1)
    }
}
"""
    ] .
<END_SHACL>
