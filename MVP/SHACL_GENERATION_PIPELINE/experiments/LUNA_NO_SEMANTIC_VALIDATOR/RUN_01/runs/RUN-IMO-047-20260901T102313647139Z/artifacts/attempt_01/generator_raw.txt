<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO_047_TemperatureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A combustion-air temperature quantity must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
        sh:message "A combustion-air temperature quantity must use degrees Celsius."
    ] .

nltlgen:IMO_047_EssentialEngineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An essential engine must provide combustion-air temperature control."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:node nltlgen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An essential engine must record its combustion-air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:node nltlgen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An essential engine must provide its manufacturer minimum combustion-air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:node nltlgen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An essential engine must provide its manufacturer maximum combustion-air temperature."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion-air temperature must be within the essential engine manufacturer's temperature limits." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:combustionAirTemperature/qudt:numericValue ?temperature .
    $this nltl:manufacturerMinAirTemperature/qudt:numericValue ?minimum .
    $this nltl:manufacturerMaxAirTemperature/qudt:numericValue ?maximum .
    FILTER (?temperature < ?minimum || ?temperature > ?maximum)
}
"""
    ] .

nltlgen:IMO_047_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:node nltlgen:IMO_047_EssentialEngineShape ;
        sh:message "Every linked essential engine must comply with the combustion-air temperature control and manufacturer criteria."
    ] .
<END_SHACL>
