<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-048> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:exposedMachineryOrFoundationMaterial ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState
    ] ;
    sh:property [
        sh:path nltl:polarServiceTemperature ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each exposed machinery or foundation material requires an approved material status, an authorized approving authority, an authorized approval standard, and a polar service temperature basis." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this a nltl:ship ;
          nltl:exposedMachineryOrFoundationMaterial ?material .
    FILTER NOT EXISTS {
        $this nltl:approvingAuthority ?authority ;
              nltl:approvalStandard ?standard ;
              nltl:materialApprovalStatus nltl:evidenceStateApproved ;
              nltl:polarServiceTemperature ?temperature .
        FILTER (
            ?authority IN (
                "Administration"^^xsd:string,
                "recognized organization accepted by the Administration"^^xsd:string
            )
        )
        FILTER (
            ?standard IN (
                "standard acceptable to the Organization"^^xsd:string,
                "another standard offering an equivalent level of safety"^^xsd:string
            )
        )
        ?temperature qudt:numericValue ?numericValue ;
                     qudt:unit unit:DEG_C .
        FILTER (datatype(?numericValue) = xsd:decimal)
    }
}
"""
    ] .
<END_SHACL>
