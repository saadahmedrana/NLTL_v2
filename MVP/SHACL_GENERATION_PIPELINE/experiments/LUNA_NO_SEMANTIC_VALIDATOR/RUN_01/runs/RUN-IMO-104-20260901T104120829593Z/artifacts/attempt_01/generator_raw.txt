<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each crew member's assigned-duty-relevant PWOM item shall have a corresponding familiarization record item." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasCrewMemberInventory ?crewMember .
                ?crewMember nltl:hasAssignedDuty ?assignedDuty .
                ?assignedDuty nltl:hasRelevantPolarWaterOperationalManualItem ?manualItem .
                FILTER NOT EXISTS {
                    ?crewMember nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?manualItem .
                }
            }
        """
    ] .

gen:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasCrewMemberInventory ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:minCount 1 ;
        sh:message "Each crew member shall have a PWOM familiarization record."
    ] .

gen:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasAssignedDuty ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem
    ] .

gen:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasFamiliarizationRecord ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem
    ] .
<END_SHACL>
