<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:IMO_115_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A sewage discharge type is required to determine applicability of the conditional rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When sewage discharge is comminuted and disinfected, MARPOL Annex IV regulation 11.1.1 compliance, the distance to nearest ice shelf or fast ice, nearby ice concentration, and the distinct distance from areas exceeding 1/10 ice concentration must be recorded and conform to the specified conditions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType .
                FILTER (STR(?dischargeType) = "comminuted_and_disinfected")

                {
                    FILTER NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus true
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:distanceToNearestIceShelfOrFastIce ?nearestDistance .
                        ?nearestDistance rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?nearestValue ;
                            qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER (
                            datatype(?nearestValue) = xsd:decimal &&
                            ?nearestValue > "3"^^xsd:decimal
                        )
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearbyConcentration .
                        ?nearbyConcentration rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?nearbyValue ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?nearbyValue) = xsd:decimal &&
                            ?nearbyValue > "0.1"^^xsd:decimal
                        )
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?areaDistanceValue ;
                            qudt:unit <http://qudt.org/vocab/unit/MI_N> .
                        FILTER (datatype(?areaDistanceValue) = xsd:decimal)
                    }
                }
            }
        """
    ] .
<END_SHACL>
