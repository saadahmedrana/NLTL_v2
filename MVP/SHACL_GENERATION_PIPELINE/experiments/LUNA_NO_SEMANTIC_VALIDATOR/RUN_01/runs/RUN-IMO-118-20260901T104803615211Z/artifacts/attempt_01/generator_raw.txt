<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:generated-shape-IMO-118> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability of the sewage-discharge prohibition."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:message "A valid polar ship category is required to determine applicability of the sewage-discharge prohibition."
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship constructed on or after 1 January 2017 that is Category A, Category B, or a passenger ship, sewage discharge to sea must be false unless compliance with paragraph 4.2.1.3 is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )
                {
                    $this nltl:shipCategory ?category .
                    FILTER (
                        ?category = nltl:polarShipCategoryA ||
                        ?category = nltl:polarShipCategoryB
                    )
                }
                UNION
                {
                    $this rdf:type nltl:passengerShip .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ?compliance .
                    FILTER (?compliance = true)
                }
                FILTER NOT EXISTS {
                    $this nltl:sewageDischargeToSea ?discharge .
                    FILTER (?discharge = false)
                }
            }
        """
    ] .
<END_SHACL>
