<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO26_009_ApplicableBranch
        gen:IMO26_009_NotApplicableBranch
    ) .

gen:IMO26_009_ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "When ice accretion is likely, the applicability value must be true."
    ] ;
    sh:property [
        sh:path nltl:hasRequiredNavigationOrCommunicationAntenna ;
        sh:minCount 1 ;
        sh:node gen:IMO26_009_AntennaShape ;
        sh:message "When ice accretion is likely, at least one required navigation or communication antenna must be represented and each represented antenna must have ice accumulation prevention."
    ] .

gen:IMO26_009_NotApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "When ice accretion is not likely, the applicability value must be false."
    ] .

gen:IMO26_009_AntennaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:antennaIceAccumulationPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each required navigation or communication antenna must have ice accumulation prevention."
    ] .
<END_SHACL>
