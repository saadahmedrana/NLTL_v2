@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_016_Chapter11_1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship operating in polar waters, Chapter 11-1 applicability requires valid ship type and branch inputs, and must be true for a qualifying fishing vessel, pleasure yacht, or cargo ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInPolarWaters ?polar .
                FILTER(?polar = true)

                FILTER(
                    NOT EXISTS {
                        $this nltl:shipType ?type .
                        FILTER(
                            datatype(?type) = xsd:string &&
                            STR(?type) IN ("fishing_vessel", "pleasure_yacht", "cargo_ship")
                        )
                    }
                    ||
                    (
                        EXISTS {
                            $this nltl:shipType ?typeFishing .
                            FILTER(
                                datatype(?typeFishing) = xsd:string &&
                                STR(?typeFishing) = "fishing_vessel"
                            )
                        }
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:lengthOverall ?lengthQuantity .
                                ?lengthQuantity qudt:numericValue ?lengthValue ;
                                                  qudt:unit qudt:M .
                                FILTER(datatype(?lengthValue) = xsd:decimal)
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:lengthOverall ?lengthQuantity2 .
                                    ?lengthQuantity2 qudt:numericValue ?lengthValue2 ;
                                                     qudt:unit qudt:M .
                                    FILTER(
                                        datatype(?lengthValue2) = xsd:decimal &&
                                        ?lengthValue2 >= 24
                                    )
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                    ||
                    (
                        EXISTS {
                            $this nltl:shipType ?typeYacht .
                            FILTER(
                                datatype(?typeYacht) = xsd:string &&
                                STR(?typeYacht) = "pleasure_yacht"
                            )
                        }
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:grossTonnage ?grossQuantity .
                                ?grossQuantity qudt:numericValue ?grossValue ;
                                                qudt:unit qudt:UNITLESS .
                                FILTER(datatype(?grossValue) = xsd:decimal)
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:engagedInTrade ?trade .
                                FILTER(datatype(?trade) = xsd:boolean)
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:grossTonnage ?grossQuantity2 .
                                    ?grossQuantity2 qudt:numericValue ?grossValue2 ;
                                                     qudt:unit qudt:UNITLESS .
                                    FILTER(
                                        datatype(?grossValue2) = xsd:decimal &&
                                        ?grossValue2 >= 300
                                    )
                                }
                                &&
                                EXISTS {
                                    $this nltl:engagedInTrade false
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                    ||
                    (
                        EXISTS {
                            $this nltl:shipType ?typeCargo .
                            FILTER(
                                datatype(?typeCargo) = xsd:string &&
                                STR(?typeCargo) = "cargo_ship"
                            )
                        }
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:grossTonnage ?grossQuantity3 .
                                ?grossQuantity3 qudt:numericValue ?grossValue3 ;
                                                 qudt:unit qudt:UNITLESS .
                                FILTER(datatype(?grossValue3) = xsd:decimal)
                            }
                            ||
                            (
                                EXISTS {
                                    $this nltl:grossTonnage ?grossQuantity4 .
                                    ?grossQuantity4 qudt:numericValue ?grossValue4 ;
                                                     qudt:unit qudt:UNITLESS .
                                    FILTER(
                                        datatype(?grossValue4) = xsd:decimal &&
                                        ?grossValue4 >= 300 &&
                                        ?grossValue4 < 500
                                    )
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:polarCodeChapter11Dash1Applicable true
                                }
                            )
                        )
                    )
                )
            }
        """
    ] .
