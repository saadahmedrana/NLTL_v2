<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF015Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:intendedRouteSeaWaterSalinity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SalinityQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadingCalculationSeaWaterSalinity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SalinityQuantityShape
    ] ;
    sh:or (
        gen:TRF015IceNavigationShape
        gen:TRF015NonIceNavigationShape
    ) ;
    sh:sparql [
        sh:message "The loading-calculation sea-water salinity must equal the intended-route sea-water salinity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:intendedRouteSeaWaterSalinity ?intended ;
                      nltl:loadingCalculationSeaWaterSalinity ?loading .
                ?intended qudt:numericValue ?intendedValue ;
                          qudt:unit nltl:gramPerKilogramSalinityUnit .
                ?loading qudt:numericValue ?loadingValue ;
                         qudt:unit nltl:gramPerKilogramSalinityUnit .
                FILTER (
                    datatype(?intendedValue) = xsd:decimal &&
                    datatype(?loadingValue) = xsd:decimal &&
                    ?intendedValue != ?loadingValue
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, the operating draught and trim must not exceed their permitted limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true ;
                      nltl:operatingDraught ?operatingDraught ;
                      nltl:upperIceWaterlineDraught ?upperDraught ;
                      nltl:operatingTrim ?operatingTrim ;
                      nltl:maximumPermittedIceTrim ?maximumTrim .

                ?operatingDraught qudt:numericValue ?operatingDraughtValue ;
                                  qudt:unit unit:M .
                ?upperDraught qudt:numericValue ?upperDraughtValue ;
                              qudt:unit unit:M .
                ?operatingTrim qudt:numericValue ?operatingTrimValue ;
                               qudt:unit unit:M .
                ?maximumTrim qudt:numericValue ?maximumTrimValue ;
                             qudt:unit unit:M .

                FILTER (
                    datatype(?operatingDraughtValue) = xsd:decimal &&
                    datatype(?upperDraughtValue) = xsd:decimal &&
                    datatype(?operatingTrimValue) = xsd:decimal &&
                    datatype(?maximumTrimValue) = xsd:decimal &&
                    (
                        ?operatingDraughtValue > ?upperDraughtValue ||
                        ?operatingTrimValue > ?maximumTrimValue
                    )
                )
            }
        """
    ] .

gen:SalinityQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:gramPerKilogramSalinityUnit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MeterQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF015IceNavigationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:operatingTrim ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumPermittedIceTrim ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] .

gen:TRF015NonIceNavigationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
