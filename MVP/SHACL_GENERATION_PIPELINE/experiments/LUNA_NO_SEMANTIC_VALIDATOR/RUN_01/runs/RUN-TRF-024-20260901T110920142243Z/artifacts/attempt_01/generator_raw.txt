<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-024> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must identify its ice class to assess the IA or IA Super applicability of this requirement."
    ] ;
    sh:sparql [
        sh:message "For a ship retaining ice class IA or IA Super with construction started before 1 September 2003, construction-stage, delivery, and assessment dates must be valid dates, and assessment must occur no later than 1 January in the year twenty years after delivery." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceClass ?ice .
    FILTER (
        ?ice IN (
            nltl:iceClassIa,
            nltl:iceClassIaSuper
        )
    )

    {
        FILTER NOT EXISTS {
            $this nltl:constructionStageDate ?constructionDate .
            FILTER (datatype(?constructionDate) = xsd:date)
        }
    }
    UNION
    {
        $this nltl:constructionStageDate ?constructionDate .
        FILTER (datatype(?constructionDate) != xsd:date)
    }
    UNION
    {
        FILTER NOT EXISTS {
            $this nltl:deliveryDate ?deliveryDate .
            FILTER (datatype(?deliveryDate) = xsd:date)
        }
    }
    UNION
    {
        $this nltl:deliveryDate ?deliveryDate .
        FILTER (datatype(?deliveryDate) != xsd:date)
    }
    UNION
    {
        FILTER NOT EXISTS {
            $this nltl:assessmentDate ?assessmentDate .
            FILTER (datatype(?assessmentDate) = xsd:date)
        }
    }
    UNION
    {
        $this nltl:assessmentDate ?assessmentDate .
        FILTER (datatype(?assessmentDate) != xsd:date)
    }
    UNION
    {
        $this nltl:constructionStageDate ?constructionDate .
        $this nltl:deliveryDate ?deliveryDate .
        $this nltl:assessmentDate ?assessmentDate .
        FILTER (datatype(?constructionDate) = xsd:date)
        FILTER (datatype(?deliveryDate) = xsd:date)
        FILTER (datatype(?assessmentDate) = xsd:date)
        FILTER (?constructionDate < "2003-09-01"^^xsd:date)
        FILTER (
            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
            ||
            (
                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                &&
                (
                    MONTH(?assessmentDate) > 1
                    ||
                    (
                        MONTH(?assessmentDate) = 1
                        &&
                        DAY(?assessmentDate) > 1
                    )
                )
            )
        )
    }
}
"""
    ] .
<END_SHACL>
