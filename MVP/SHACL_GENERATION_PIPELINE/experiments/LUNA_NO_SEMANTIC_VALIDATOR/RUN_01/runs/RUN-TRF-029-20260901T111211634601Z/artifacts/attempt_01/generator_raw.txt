<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF029DirectAnalysisShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF029DirectAnalysisUsedBranch
        gen:TRF029DirectAnalysisNotUsedBranch
    ) .

gen:TRF029DirectAnalysisUsedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:directAnalysisUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "directAnalysisUsed must be true for the direct-analysis branch."
    ] ;
    sh:property [
        sh:path nltl:prescribedProcedureApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "Prescribed procedures must be marked not applicable when direct analysis is used."
    ] ;
    sh:property [
        sh:path nltl:directAnalysisApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:minCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Direct analysis must have approved status."
    ] ;
    sh:property [
        sh:path nltl:structuralArrangement ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A structural arrangement is required when direct analysis is used."
    ] .

gen:TRF029DirectAnalysisNotUsedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:directAnalysisUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "directAnalysisUsed must be false for the prescribed-procedure branch."
    ] .
<END_SHACL>
