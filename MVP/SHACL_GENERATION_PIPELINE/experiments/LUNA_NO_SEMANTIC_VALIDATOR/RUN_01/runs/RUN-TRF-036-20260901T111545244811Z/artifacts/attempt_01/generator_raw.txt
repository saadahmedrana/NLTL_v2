<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF036_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:message "The ship must have one applicable ice class from Table 4-1."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:node nltlgen:TRF036_ParameterSetShape ;
        sh:message "The ship must provide an ice-class design parameter set."
    ] .

nltlgen:TRF036_ParameterSetShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Level ice thickness must be a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design ice load height must be a single metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The level ice thickness and design ice load height must match the Table 4-1 values for the ship's ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                       nltl:iceClass ?class .
                $this nltl:levelIceThickness ?hiQuantity ;
                      nltl:designIceLoadHeight ?hQuantity .
                ?hiQuantity qudt:numericValue ?hi ;
                            qudt:unit unit:M .
                ?hQuantity qudt:numericValue ?h ;
                           qudt:unit unit:M .
                FILTER (
                    (?class = nltl:iceClassIaSuper &&
                     (?hi > 1.0 || ?h != 0.35)) ||
                    (?class = nltl:iceClassIa &&
                     (?hi > 0.8 || ?h != 0.30)) ||
                    (?class = nltl:iceClassIb &&
                     (?hi > 0.6 || ?h != 0.25)) ||
                    (?class = nltl:iceClassIc &&
                     (?hi > 0.4 || ?h != 0.22))
                )
            }
        """
    ] .
<END_SHACL>
