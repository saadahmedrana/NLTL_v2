@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-049-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameWebThickness ;
                sh:node <urn:nltl:generated-shape:TRF-049-LengthQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A frame in the ice-strengthened area must provide its web thickness."
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:node <urn:nltl:generated-shape:TRF-049-AttachmentShape> ;
                sh:minCount 1 ;
                sh:message "A frame in the ice-strengthened area must be effectively attached to its supporting structures."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A connection bracket must have at least the same thickness as the frame web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                $this nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                    qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                    qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (
                    datatype(?frameValue) = xsd:decimal &&
                    datatype(?bracketValue) = xsd:decimal &&
                    ?bracketValue < ?frameValue
                )
            }
        """
    ] .
