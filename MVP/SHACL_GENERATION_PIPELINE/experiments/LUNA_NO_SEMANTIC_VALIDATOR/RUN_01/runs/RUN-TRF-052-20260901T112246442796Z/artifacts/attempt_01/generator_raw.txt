<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF052FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        gen:TRF052AlternativeArrangementShape
        gen:TRF052StandardArrangementShape
    ) .

gen:TRF052AlternativeArrangementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:TRF052StandardArrangementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF052AngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF052SpanQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:frameAsymmetrical ?asym .
                        $this nltl:frameWebAngleToShell ?angleNode .
                        ?angleNode qudt:numericValue ?angle .
                        FILTER ( ! (?asym = true || ?angle < 90) )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:antitrippingSupportSpacing ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF052SpacingQuantityShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The antitripping support regions required for the frame span and ice class are incomplete." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:frameSpan ?spanNode .
                ?spanNode qudt:numericValue ?span .
                $this nltl:iceClass ?ice .

                FILTER (
                    (
                        ?span > 4 &&
                        !(
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    ) ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIaSuper &&
                        !(
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:sternRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue }
                        )
                    ) ||
                    (
                        ?span <= 4 &&
                        ?ice = nltl:iceClassIa &&
                        !(
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:bowRegion } &&
                            EXISTS { $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion }
                        )
                    ) ||
                    (
                        ?span <= 4 &&
                        (?ice = nltl:iceClassIb || ?ice = nltl:iceClassIc) &&
                        !EXISTS {
                            $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                        }
                    )
                )
            }
        """
    ] .

gen:TRF052SpacingQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive 1300
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

gen:TRF052SpanQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:TRF052AngleQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG>
    ] .
<END_SHACL>
