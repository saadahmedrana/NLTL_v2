<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CentiMetreCubedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CentiMetreSquaredQuantityShape
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area must satisfy the specified ice-stringer formulas within a relative tolerance of 1e-6 and an absolute tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentFactorNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:designLineLoad ?lineLoadNode ;
                      nltl:sectionModulus ?sectionModulusNode ;
                      nltl:requiredShearArea ?shearAreaNode .

                ?pressureNode qudt:numericValue ?p .
                ?heightNode qudt:numericValue ?h .
                ?spanNode qudt:numericValue ?l .
                ?momentFactorNode qudt:numericValue ?m .
                ?yieldNode qudt:numericValue ?sigma .
                ?lineLoadNode qudt:numericValue ?reportedLine .
                ?sectionModulusNode qudt:numericValue ?reportedZ .
                ?shearAreaNode qudt:numericValue ?reportedA .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?lineMNPerM)
                BIND((?lineMNPerM * 1000) AS ?expectedLine)
                BIND((0.9 * 1.8 * ?lineMNPerM * ?l * ?l / (?m * ?sigma) * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?lineMNPerM * ?l / (2 * ?sigma) * 10000) AS ?expectedA)

                FILTER(
                    ABS(?reportedLine - ?expectedLine) > (ABS(?expectedLine) * 0.000001 + 0.000001)
                    || ABS(?reportedZ - ?expectedZ) > (ABS(?expectedZ) * 0.000001 + 0.000001)
                    || ABS(?reportedA - ?expectedA) > (ABS(?expectedA) * 0.000001 + 0.000001)
                )
            }
        """
    ] .
