@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "icePressure is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "designIceLoadHeight is required in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "span is required in metres."
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gen:LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "stringerSpan is required in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "frameMomentFactorM is required as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "yieldStrength is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gen:LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "distanceToIceBelt is required in metres."
    ] ;
    sh:property [
        sh:path nltl:distanceToAdjacentIceStringer ;
        sh:node gen:LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "distanceToAdjacentIceStringer is required in metres."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:ForcePerLengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "designLineLoad is required in kN/m."
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gen:DimensionlessQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "outsideIceBeltFactor is required as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "sectionModulus is required in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:AreaQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "requiredShearArea is required in cm2."
    ] ;
    sh:sparql [
        sh:message "designLineLoad, outsideIceBeltFactor, sectionModulus, and requiredShearArea must satisfy the TRF-054 formulas; derived decimal results use a relative tolerance of 1e-6 with a 1e-12 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this
                    nltl:icePressure ?pressureNode ;
                    nltl:designIceLoadHeight ?heightNode ;
                    nltl:span ?spanNode ;
                    nltl:stringerSpan ?stringerSpanNode ;
                    nltl:frameMomentFactorM ?momentNode ;
                    nltl:yieldStrength ?yieldNode ;
                    nltl:distanceToIceBelt ?iceBeltNode ;
                    nltl:distanceToAdjacentIceStringer ?adjacentNode ;
                    nltl:designLineLoad ?lineLoadNode ;
                    nltl:outsideIceBeltFactor ?outsideFactorNode ;
                    nltl:sectionModulus ?sectionNode ;
                    nltl:requiredShearArea ?areaNode .

                ?pressureNode qudt:numericValue ?p .
                ?heightNode qudt:numericValue ?h .
                ?spanNode qudt:numericValue ?span .
                ?stringerSpanNode qudt:numericValue ?l .
                ?momentNode qudt:numericValue ?m .
                ?yieldNode qudt:numericValue ?sigma .
                ?iceBeltNode qudt:numericValue ?hs .
                ?adjacentNode qudt:numericValue ?ls .
                ?lineLoadNode qudt:numericValue ?reportedLine .
                ?outsideFactorNode qudt:numericValue ?reportedFactor .
                ?sectionNode qudt:numericValue ?reportedZ .
                ?areaNode qudt:numericValue ?reportedA .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?line)
                BIND((1.0 - (?hs / ?ls)) AS ?outsideFactor)
                BIND((0.80 * 1.8 * ?line * ?l * ?l / (?m * ?sigma) * ?outsideFactor * 1000000.0) AS ?z)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?line * ?l / (2.0 * ?sigma) * ?outsideFactor * 10000.0) AS ?a)

                FILTER(
                    ABS(?reportedLine - ?line) >
                    (0.000001 * IF(ABS(?line) > 0.000000000001, ABS(?line), 0.000000000001))
                    ||
                    ABS(?reportedFactor - ?outsideFactor) >
                    (0.000001 * IF(ABS(?outsideFactor) > 0.000000000001, ABS(?outsideFactor), 0.000000000001))
                    ||
                    ABS(?reportedZ - ?z) >
                    (0.000001 * IF(ABS(?z) > 0.000000000001, ABS(?z), 0.000000000001))
                    ||
                    ABS(?reportedA - ?a) >
                    (0.000001 * IF(ABS(?a) > 0.000000000001, ABS(?a), 0.000000000001))
                )
            }
        """
    ] .
