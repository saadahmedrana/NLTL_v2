<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF056ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node [
            sh:class nltl:weatherdeckHatch ;
            sh:property [
                sh:path nltl:hatchOpeningLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a weatherdeck hatch longer than half the ship breadth, ship-side deflection and both hatch-cover and hatch-fitting design evidence must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipBreadth ?breadthQuantity ;
                      nltl:hasWeatherdeckHatch ?hatch .
                ?breadthQuantity qudt:numericValue ?breadthValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (?openingValue > (?breadthValue / 2))

                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflectionQuantity .
                        ?deflectionQuantity qudt:numericValue ?deflectionValue ;
                                             qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                        ?coverEvidence rdf:type nltl:hatchCoverDesignEvidence .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                        ?fittingEvidence rdf:type nltl:hatchFittingDesignEvidence .
                    }
                )
            }
        """
    ] .
<END_SHACL>
