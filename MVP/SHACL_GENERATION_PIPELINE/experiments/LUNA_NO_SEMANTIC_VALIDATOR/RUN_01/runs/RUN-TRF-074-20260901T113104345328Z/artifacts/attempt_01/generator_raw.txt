<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must declare an ice class to determine applicability of the propeller submersion rule."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:minCount 1 ;
                sh:node gen:TRF074LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:minCount 1 ;
                sh:node gen:TRF074LengthQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:minCount 1 ;
                sh:node gen:TRF074LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:minCount 1 ;
                sh:node gen:TRF074LengthQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, when the ballast-condition propeller top depth is below hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))

                $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
                ?depthQuantity qudt:numericValue ?depth .
                ?depthQuantity qudt:unit <http://qudt.org/vocab/unit/M> .

                $this nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
                ?hiQuantity qudt:numericValue ?hi .
                ?hiQuantity qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    datatype(?depth) = xsd:decimal &&
                    datatype(?hi) = xsd:decimal &&
                    ?depth < ?hi
                )

                FILTER NOT EXISTS {
                    $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
                }
            }
        """
    ] .

gen:TRF074LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
