<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF078QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF078TableLookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash6
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:selectedIceClassCycleCount ;
                sh:node nltl:TRF078QuantityValueShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:tableReference ;
                sh:hasValue nltl:traficomTable6Dash7
            ] ;
            sh:property [
                sh:path nltl:tableLookupPropellerLocation ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:propellerLocationValue ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:selectedPropellerLocationFactor ;
                sh:node nltl:TRF078QuantityValueShape ;
                sh:minCount 1
            ]
        ]
    ) .

nltl:TRF078ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node nltl:TRF078TableLookupCaseShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:node nltl:TRF078QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "Both Table 6-6 and Table 6-7 lookup cases are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?case6 .
                        ?case6 nltl:tableReference nltl:traficomTable6Dash6 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?case7 .
                        ?case7 nltl:tableReference nltl:traficomTable6Dash7 .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "iceLoadCycleCount must equal k1 multiplied by k2, k3, N_class, and n_n within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?case6 ;
                      nltl:hasTableLookupCase ?case7 ;
                      nltl:iceLoadCycleCoefficientK2/qudt:numericValue ?k2 ;
                      nltl:iceLoadCycleCoefficientK3/qudt:numericValue ?k3 ;
                      nltl:propellerBladeLoadSpectrumCountFactor/qudt:numericValue ?nn ;
                      nltl:iceLoadCycleCount ?reported .
                ?case6 nltl:tableReference nltl:traficomTable6Dash6 ;
                        nltl:selectedIceClassCycleCount/qudt:numericValue ?nclass .
                ?case7 nltl:tableReference nltl:traficomTable6Dash7 ;
                        nltl:selectedPropellerLocationFactor/qudt:numericValue ?k1 .
                BIND ((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (
                    ABS(xsd:decimal(?reported) - ?expected)
                    >
                    (ABS(?expected) * 0.000001 + 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
