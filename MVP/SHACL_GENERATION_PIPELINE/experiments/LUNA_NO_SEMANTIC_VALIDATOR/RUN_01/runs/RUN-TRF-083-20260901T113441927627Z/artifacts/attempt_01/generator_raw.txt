<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF083Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:rotationalSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Rotational speed known must be recorded to select the applicable bollard-speed branch."
    ] ;
    sh:or (
        gen:TRF083KnownSpeedBranch
        gen:TRF083LookupBranch
    ) .

gen:TRF083KnownSpeedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:rotationalSpeedKnown ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "When bollard propeller speed is known, the MCR bollard propeller speed must be supplied in revolutions per minute."
    ] .

gen:TRF083LookupBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:rotationalSpeedKnown ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:TRF083TableLookupCaseShape ;
        sh:message "When bollard propeller speed is unknown, a Table 6-9 lookup case must be supplied."
    ] .

gen:TRF083TableLookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Table 6-9 lookup case must be marked as applied."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:traficomTable6Dash9 ;
        sh:message "The unknown bollard propeller speed must reference TRAFICOM Table 6-9."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "The Table 6-9 selection must have supporting evidence."
    ] .
<END_SHACL>
