<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF091-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:message "The ship must identify the ice milling load sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF091-PropulsionLoadCaseShape> ;
        sh:message "The ship must identify a propulsion load case for the ice milling load sequence."
    ] .

<urn:nltl:generated-shape:TRF091-PropulsionLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCasePurposeValue ;
        sh:in ( nltl:propulsionLineStrengthEvaluationPurpose ) ;
        sh:minCount 1 ;
        sh:message "The ice milling load sequence propulsion load case must be used for propulsion line strength evaluation and not for stalling analysis."
    ] .
<END_SHACL>
