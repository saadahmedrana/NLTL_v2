<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF_108_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the SN-curve is known."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must record the SN-curve type."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must record the propeller blade material properties used to select the SN-curve."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:message "The ship must provide the selected SN-curve."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Evidence must show that the selected SN-curve corresponds to the blade material properties."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:class nltl:stressLifeCurveTypeValue
            ]
        ]
        [
            sh:property [
                sh:path nltl:stressLifeCurveKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:stressLifeCurveSelection ;
                sh:hasValue nltl:twoSlopeStressLifeCurve
            ]
        ]
    ) .
<END_SHACL>
