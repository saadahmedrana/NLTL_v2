<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the damping taken into account for thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass as a single decimal quantity in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the thruster global natural frequency as a single decimal quantity in hertz."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked thruster body must provide ship attachment stiffness as a single decimal quantity in kN/m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                FILTER (
                    NOT EXISTS {
                        ?body nltl:shipAttachmentStiffness ?q .
                    }
                    ||
                    EXISTS {
                        ?body nltl:shipAttachmentStiffness ?q .
                        FILTER NOT EXISTS {
                            ?q a qudt:QuantityValue ;
                               qudt:numericValue ?v ;
                               qudt:unit unit:KiloN-PER-M .
                            FILTER (datatype(?v) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        ?body nltl:shipAttachmentStiffness ?q1, ?q2 .
                        FILTER (?q1 != ?q2)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked thruster body must have a calculation case for every represented global natural-frequency direction, and each case must identify exactly one longitudinal or transverse direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                FILTER (
                    NOT EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                    }
                    ||
                    EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                        FILTER NOT EXISTS {
                            ?case a nltl:calculationCase ;
                                  nltl:vibrationDirection ?direction .
                            FILTER (
                                ?direction IN (
                                    nltl:longitudinalDirection,
                                    nltl:transverseDirection
                                )
                            )
                        }
                    }
                    ||
                    EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                        ?case nltl:vibrationDirection ?direction1, ?direction2 .
                        FILTER (?direction1 != ?direction2)
                    }
                    ||
                    EXISTS {
                        ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                        ?case nltl:vibrationDirection ?direction .
                        FILTER (
                            ?direction NOT IN (
                                nltl:longitudinalDirection,
                                nltl:transverseDirection
                            )
                        )
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
