gen:I2_011_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:I2_011_NonApplicablePolarCase
        gen:I2_011_NonApplicableBulbCase
        gen:I2_011_ApplicableCase
    ) ;
    sh:sparql [
        sh:message "For an applicable PC6 or PC7 bulbous-bow ship, the bow design force must not be less than either declared method force." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:bulbousBowPresent ?bulbousBow ;
                      nltl:bowDesignForce ?resultNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode .

                FILTER (?polarClass IN (
                    nltl:polarClassPc6,
                    nltl:polarClassPc7
                ))
                FILTER (?bulbousBow = true)

                ?resultNode qudt:numericValue ?result .
                ?iiiNode qudt:numericValue ?iii .
                ?ivNode qudt:numericValue ?iv .

                FILTER (
                    ?result < (?iii * 1000) ||
                    ?result < (?iv * 1000)
                )
            }
        """
    ] .
