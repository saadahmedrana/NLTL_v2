gen:I2_014_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the bow waterline length."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The bow region must be divided into four subregions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:node gen:BowSubregionCalculationCaseShape ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:message "The ship must have exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:node gen:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowForce must equal the maximum calculated force among the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                ?case nltl:bowSubregionCalculatedForce ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseValue .
            }
            GROUP BY $this ?selectedValue
            HAVING (MAX(?caseValue) != ?selectedValue)
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowLineLoad must equal the maximum calculated line load among the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowLineLoad ?selectedQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                ?case nltl:bowSubregionCalculatedLineLoad ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseValue .
            }
            GROUP BY $this ?selectedValue
            HAVING (MAX(?caseValue) != ?selectedValue)
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowPressure must equal the maximum calculated pressure among the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowPressure ?selectedQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                ?case nltl:bowSubregionCalculatedPressure ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseValue .
            }
            GROUP BY $this ?selectedValue
            HAVING (MAX(?caseValue) != ?selectedValue)
        """
    ] .
