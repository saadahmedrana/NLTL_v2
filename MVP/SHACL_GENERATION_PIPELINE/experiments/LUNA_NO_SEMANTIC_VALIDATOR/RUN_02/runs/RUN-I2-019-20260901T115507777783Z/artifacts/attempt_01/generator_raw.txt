<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-019-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:node [
                    sh:class nltl:hullAreaValue ;
                    sh:property [
                        sh:path nltl:hullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each structural member spanning one or more hull areas must have a selected hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasSpannedHullArea ?area .
                FILTER NOT EXISTS {
                    ?member nltl:selectedHullAreaFactor ?selectedNode .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected hull area factor must equal the largest hull area factor among the hull areas spanned by the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                {
                    SELECT ?member (MAX(?factorValue) AS ?maximumFactor)
                    WHERE {
                        ?member nltl:hasSpannedHullArea ?area .
                        ?area nltl:hullAreaFactor ?factorNode .
                        ?factorNode qudt:numericValue ?factorValue .
                    }
                    GROUP BY ?member
                }
                ?member nltl:selectedHullAreaFactor ?selectedNode .
                ?selectedNode qudt:numericValue ?selectedValue .
                FILTER (
                    ABS(?selectedValue - ?maximumFactor) >
                    (ABS(?maximumFactor) * "0.000001"^^xsd:decimal + "0.000000001"^^xsd:decimal)
                )
            }
        """
    ] .
<END_SHACL>
