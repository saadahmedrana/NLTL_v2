<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:message "Each linked structural member must provide support applicability information and a permitted support condition."
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node gen:MemberSupportShape ;
        sh:message "A structural member must identify the support used to establish its support condition."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Continuous-through-support applicability must be recorded as a single boolean."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Termination within the ice-strengthened area must be recorded as a single boolean."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport )
        ] ;
        sh:message "The frame support condition must be either fixed support or simple support."
    ] ;
    sh:or (
        gen:FixedByContinuityShape
        gen:FixedByTerminationShape
        gen:FixedByBracketShape
        gen:FixedByRestraintShape
        gen:SimpleSupportShape
    ) .

gen:MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Connection-bracket applicability must be recorded as a single boolean."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Demonstrated rotational-restraint applicability must be recorded as a single boolean."
    ] ;
    sh:or (
        gen:RestraintDemonstratedShape
        gen:RestraintNotDemonstratedShape
    ) .

gen:RestraintDemonstratedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "Demonstrated significant rotational restraint requires supporting evidence."
    ] .

gen:RestraintNotDemonstratedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:hasValue false
    ] .

gen:FixedByContinuityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:hasValue nltl:fixedFrameSupport
        ]
    ] ;
    sh:message "A member continuous through its support requires fixed support."

gen:FixedByTerminationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:hasValue nltl:fixedFrameSupport
        ]
    ] ;
    sh:message "A member terminating within the ice-strengthened area requires fixed support."

gen:FixedByBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node [
            sh:property [
                sh:path nltl:connectionBracketPresent ;
                sh:hasValue true
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:hasValue nltl:fixedFrameSupport
        ]
    ] ;
    sh:message "A member attached with a connection bracket requires fixed support."

gen:FixedByRestraintShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node [
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:hasValue true
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:hasValue nltl:fixedFrameSupport
        ]
    ] ;
    sh:message "Demonstrated significant rotational restraint requires fixed support."

gen:SimpleSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node [
            sh:property [
                sh:path nltl:connectionBracketPresent ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:hasValue false
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:node [
            sh:hasValue nltl:simpleFrameSupport
        ]
    ] ;
    sh:message "Where continuity, a connection bracket, demonstrated rotational restraint, and ice-area termination are absent, simple support must be assumed."

</END_SHACL>
