<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node nltlgen:I2_029_CalculationCaseShape
    ] .

nltlgen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must include attached shell plating in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must exclude attached shell plating from the shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The calculation case must explicitly represent whether flange material is included in the shear area."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:structuralMember
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When flange material is included in the shear area, the linked structural member must exist and have a fitted flange." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:flangeMaterialIncludedInShearArea true .
    FILTER (
        NOT EXISTS {
            $this nltl:sectionCalculationCaseStructuralMember ?anyMember .
        }
        ||
        EXISTS {
            $this nltl:sectionCalculationCaseStructuralMember ?badMember .
            FILTER NOT EXISTS {
                ?badMember nltl:flangeFitted true .
            }
        }
    )
}
"""
    ] .
<END_SHACL>
