<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_051_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_051_CalculationCaseShape
    ] ;
    sh:sparql [
        sh:message "For a positive-shear calculation case in the specified longitudinal intervals, the distribution factor and design vertical ice shear force must satisfy FI = Cf * FIB; calculated FI is compared with a 1e-6 relative tolerance plus a 1e-9 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:hasCalculationCase ?case .

                ?fibQuantity qudt:numericValue ?fib .
                ?fiQuantity qudt:numericValue ?fi .
                ?cfQuantity qudt:numericValue ?cf .
                ?luiQuantity qudt:numericValue ?lui .

                ?case nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:shearForceDirection nltl:positiveShearForce .
                ?positionQuantity qudt:numericValue ?position .

                FILTER (
                    (
                        ?position >= 0 &&
                        ?position <= (0.6 * ?lui) &&
                        (
                            ?cf != 0 ||
                            ABS(?fi - (0 * ?fib)) >
                            (0.000001 * ABS(0 * ?fib) + 0.000000001)
                        )
                    )
                    ||
                    (
                        ?position >= (0.9 * ?lui) &&
                        ?position <= ?lui &&
                        (
                            ?cf != 1 ||
                            ABS(?fi - (1 * ?fib)) >
                            (0.000001 * ABS(1 * ?fib) + 0.000000001)
                        )
                    )
                )
            }
        """
    ] .

gen:I2_051_CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:positiveShearForce
    ] .
<END_SHACL>
