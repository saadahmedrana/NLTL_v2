<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:message "Each represented calculation case must be linked from the ship and identify its assessed hull structure."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gen:AssessedHullStructureShape ;
        sh:minCount 1 ;
        sh:message "A calculation case must identify the assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "The prescribed analytical procedure must be used for the calculation case."
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:calculationMethod ;
            sh:hasValue nltl:directCalculationMethodValue
        ]
    ] .

gen:AssessedHullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The assessed hull structure must provide thickness as a decimal length quantity in metres."
    ] .
<END_SHACL>
