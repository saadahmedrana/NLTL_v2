<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_062_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship record must confirm that the required direct calculation was used."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked load carrying stringer or web frame must be recorded as forming part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberClass .
                FILTER (
                    ?memberClass = nltl:loadCarryingStringer ||
                    ?memberClass = nltl:webFrame
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage a nltl:grillageSystem .
                }
            }
        """
    ] .
</BEGIN_SHACL>
