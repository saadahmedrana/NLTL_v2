<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_003_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:openWaterIceCondition
            nltl:iceConditionLessSevereThanCategoryAAndB
        ) ;
        sh:message "A ship design ice condition must be declared as open water or less severe than categories A and B."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:message "A ship designed for open water or less severe ice conditions than categories A and B shall have Polar Ship Category C."
    ] .
<END_SHACL>
