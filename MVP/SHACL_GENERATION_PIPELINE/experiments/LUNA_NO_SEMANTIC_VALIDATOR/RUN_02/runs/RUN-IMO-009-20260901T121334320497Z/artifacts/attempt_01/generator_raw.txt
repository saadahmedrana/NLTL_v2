<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:IMO_009_StaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "A ship must specify its ice condition."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, land-origin ice concentration must be less than 1/10 and total ice concentration must not exceed 1/10, using dimensionless QUDT quantities." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "bergy waters"^^xsd:string)

                FILTER (
                    NOT EXISTS {
                        $this nltl:landOriginIceConcentration ?land .
                        ?land a qudt:QuantityValue ;
                              qudt:numericValue ?landValue ;
                              qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?landValue) = xsd:decimal &&
                            ?landValue < 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:landOriginIceConcentration ?badLand .
                        FILTER NOT EXISTS {
                            ?badLand a qudt:QuantityValue ;
                                     qudt:numericValue ?badLandValue ;
                                     qudt:unit unit:UNITLESS .
                            FILTER (
                                datatype(?badLandValue) = xsd:decimal &&
                                ?badLandValue < 1
                            )
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:totalIceConcentration ?total .
                        ?total a qudt:QuantityValue ;
                              qudt:numericValue ?totalValue ;
                              qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?totalValue) = xsd:decimal &&
                            ?totalValue <= 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:totalIceConcentration ?badTotal .
                        FILTER NOT EXISTS {
                            ?badTotal a qudt:QuantityValue ;
                                      qudt:numericValue ?badTotalValue ;
                                      qudt:unit unit:UNITLESS .
                            FILTER (
                                datatype(?badTotalValue) = xsd:decimal &&
                                ?badTotalValue <= 1
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
