<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO016ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:node gen:IMO016CertificateShape ;
        sh:message "The ship must have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The certificate language must be recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:in ( "English" "French" "Spanish" )
            ]
        ]
        [
            sh:sparql [
                sh:message "When the certificate language is not English, French, or Spanish, an approved translation must be present." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:certificateLanguage ?language .
                        FILTER (STR(?language) NOT IN ("English", "French", "Spanish"))
                        $this nltl:approvedTranslationPresent ?translation .
                        FILTER (
                            ?translation = true &&
                            datatype(?translation) = xsd:boolean
                        )
                    }
                """
            ]
        )
    ] .

gen:IMO016CertificateShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate form must correspond to the Appendix I model."
    ] .
<END_SHACL>
