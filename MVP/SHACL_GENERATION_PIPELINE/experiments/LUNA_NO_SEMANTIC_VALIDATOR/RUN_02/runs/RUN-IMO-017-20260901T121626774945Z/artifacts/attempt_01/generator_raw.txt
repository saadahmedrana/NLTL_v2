<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "A Polar Ship Certificate form is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasScheduleShape ;
        sh:message "A SOLAS certificate schedule with certificate date records is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017SupplementShape ;
        sh:message "A Polar Certificate supplement with required equipment records is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate validity, survey, and endorsement dates must match the corresponding SOLAS certificate schedule date records." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarCertificateValidityDates ?validityDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    FILTER NOT EXISTS {
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?scheduleDate .
                        FILTER (?scheduleDate = ?validityDate)
                    }
                }
                $this nltl:surveyDates ?surveyDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule2 .
                    FILTER NOT EXISTS {
                        ?schedule2 nltl:hasCertificateScheduleDateRecord ?record2 .
                        ?record2 nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                 nltl:certificateScheduleDate ?scheduleDate2 .
                        FILTER (?scheduleDate2 = ?surveyDate)
                    }
                }
                $this nltl:endorsementDates ?endorsementDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule3 .
                    FILTER NOT EXISTS {
                        ?schedule3 nltl:hasCertificateScheduleDateRecord ?record3 .
                        ?record3 nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                 nltl:certificateScheduleDate ?scheduleDate3 .
                        FILTER (?scheduleDate3 = ?endorsementDate)
                    }
                }
            }
        """
    ] .

gen:IMO017SolasScheduleShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017DateRecordShape
    ] .

gen:IMO017DateRecordShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record must contain a certificate schedule date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record must have a permitted date category."
    ] .

gen:IMO017SupplementShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Certificate supplement must record required equipment."
    ] .
<END_SHACL>
