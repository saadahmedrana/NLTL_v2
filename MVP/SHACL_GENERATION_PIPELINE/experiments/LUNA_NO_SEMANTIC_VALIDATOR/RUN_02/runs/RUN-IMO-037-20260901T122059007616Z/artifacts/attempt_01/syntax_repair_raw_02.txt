<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO037ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category is required to determine applicability of the residual stability requirement."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date is required to determine applicability of the residual stability requirement."
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionCase ;
        sh:class nltl:loadingConditionCase
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship constructed on or after 1 January 2017, every linked loading condition case must contain the required result inputs, result evidence, and either residual stability factor s_i equal to 1 or a compliant permitted alternative-instrument result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)

                $this nltl:hasLoadingConditionCase ?case .

                OPTIONAL {
                    ?case a nltl:loadingConditionCase .
                }
                OPTIONAL {
                    ?case nltl:loadingConditionIdentifier ?identifier .
                }
                OPTIONAL {
                    ?case nltl:alternativeInstrumentApplicable ?alternativeApplicable .
                }
                OPTIONAL {
                    ?case nltl:residualStabilityFactorSI ?factorQuantity .
                    OPTIONAL {
                        ?factorQuantity qudt:numericValue ?factorValue .
                    }
                    OPTIONAL {
                        ?factorQuantity qudt:unit ?factorUnit .
                    }
                }
                OPTIONAL {
                    ?case nltl:alternativeInstrumentResidualStabilityStatus ?alternativeStatus .
                }
                OPTIONAL {
                    ?case nltl:hasLoadingConditionResultEvidence ?evidence .
                    OPTIONAL {
                        ?evidence a nltl:evidenceArtifact .
                    }
                }

                FILTER (
                    !BOUND(?case)
                    || !BOUND(?identifier)
                    || DATATYPE(?identifier) != xsd:string
                    || !BOUND(?alternativeApplicable)
                    || DATATYPE(?alternativeApplicable) != xsd:boolean
                    || !BOUND(?factorQuantity)
                    || !BOUND(?factorValue)
                    || DATATYPE(?factorValue) != xsd:decimal
                    || !BOUND(?factorUnit)
                    || ?factorUnit != <http://qudt.org/vocab/unit/UNITLESS>
                    || !BOUND(?evidence)
                    || !BOUND(?evidence)
                    || (
                        ?alternativeApplicable = false
                        && (
                            ?factorValue < 1
                            || ?factorValue > 1
                        )
                    )
                    || (
                        ?alternativeApplicable = true
                        && (
                            !BOUND(?alternativeStatus)
                            || DATATYPE(?alternativeStatus) != xsd:boolean
                            || ?alternativeStatus != true
                        )
                        && (
                            ?factorValue < 1
                            || ?factorValue > 1
                        )
                    )
                )
            }
        """
    ] .

</END_SHACL>
