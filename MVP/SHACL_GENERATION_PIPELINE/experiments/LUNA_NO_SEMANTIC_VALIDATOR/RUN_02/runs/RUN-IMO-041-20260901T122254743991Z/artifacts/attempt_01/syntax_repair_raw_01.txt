<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_041_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each ship must provide an opening actuation type to determine applicability of the hydraulic-fluid protection requirement."
    ] ;
    sh:or (
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:openingActuationType ;
                sh:hasValue "hydraulic"^^xsd:string
            ] ;
            sh:property [
                sh:path nltl:hydraulicFluidFreezingOrViscosityControlPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:message "Ships with hydraulically operated hatches or doors must provide freezing or excessive-viscosity control for hydraulic fluids."
            ]
        ]
        [
            a sh:NodeShape ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:openingActuationType ?actuation .
                        FILTER NOT EXISTS {
                            $this nltl:openingActuationType "hydraulic"^^xsd:string .
                        }
                    }
                """ ;
                sh:message "The non-hydraulic actuation alternative applies only when no hydraulic opening actuation type is recorded."
            ] ;
        ]
    ) .
<END_SHACL>
