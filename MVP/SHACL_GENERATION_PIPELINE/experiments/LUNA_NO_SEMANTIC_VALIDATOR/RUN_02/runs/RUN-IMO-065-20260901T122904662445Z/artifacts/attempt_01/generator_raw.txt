<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qv: <http://qudt.org/schema/qudt/> .
@prefix qudtunit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-065-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:abandonmentEnvironment ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An assessed abandonment environment is required."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-065-TimeQuantityShape> ;
        sh:message "Maximum expected rescue time is required in days."
    ] ;
    sh:property [
        sh:path nltl:survivalSupportDuration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-065-TimeQuantityShape> ;
        sh:message "Survival support duration is required in days."
    ] ;
    sh:property [
        sh:path nltl:habitableEnvironmentPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall provide a habitable environment."
    ] ;
    sh:property [
        sh:path nltl:weatherProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall protect persons from cold, wind, and sun."
    ] ;
    sh:property [
        sh:path nltl:accommodationSpacePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall provide accommodation space."
    ] ;
    sh:property [
        sh:path nltl:sustenanceMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall provide means of sustenance."
    ] ;
    sh:property [
        sh:path nltl:safeAccessExitPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall provide safe access and exit points."
    ] ;
    sh:property [
        sh:path nltl:rescueCommunicationPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Survival resources shall provide communication with rescue assets."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Survival support duration shall be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qv: <http://qudt.org/schema/qudt/>
            PREFIX qudtunit: <http://qudt.org/vocab/unit/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:survivalSupportDuration ?supportQuantity .
                ?supportQuantity qv:numericValue ?supportValue ;
                                 qv:unit qudtunit:DAY .
                $this nltl:maximumExpectedRescueTime ?rescueQuantity .
                ?rescueQuantity qv:numericValue ?rescueValue ;
                                qv:unit qudtunit:DAY .
                FILTER (?supportValue < ?rescueValue)
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-065-TimeQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qv:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qv:unit ;
        sh:hasValue qudtunit:DAY ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
