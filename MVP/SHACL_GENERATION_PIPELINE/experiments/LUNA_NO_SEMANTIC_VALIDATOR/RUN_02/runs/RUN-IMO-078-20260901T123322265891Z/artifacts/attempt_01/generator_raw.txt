<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO078ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO078CrewMemberShape
    ] ;
    sh:property [
        sh:path nltl:hasPassenger ;
        sh:node gen:IMO078PassengerShape
    ] .

gen:IMO078CrewMemberShape a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:personalAndGroupSurvivalEquipmentTrainingCompleted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each linked crew member must have completed personal and group survival equipment training."
    ] .

gen:IMO078PassengerShape a sh:NodeShape ;
    sh:class nltl:passenger ;
    sh:property [
        sh:path nltl:personalSurvivalEquipmentInstructionCompleted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Each linked passenger must have completed personal survival equipment instruction."
    ] .
<END_SHACL>
