<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO085ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one permitted ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one construction date."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:bridgeWingsEnclosed ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bridgeWingProtectionDesignStatus ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            )
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shipCategory ;
                        sh:hasValue nltl:polarShipCategoryC
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:constructionDate ;
                        sh:maxInclusive "2016-12-31"^^xsd:date
                    ]
                ]
            )
        ]
    ) ;
    sh:message "Category A or B ships constructed on or after 1 January 2017 must have enclosed bridge wings or approved bridge-wing protection design." .
<END_SHACL>
