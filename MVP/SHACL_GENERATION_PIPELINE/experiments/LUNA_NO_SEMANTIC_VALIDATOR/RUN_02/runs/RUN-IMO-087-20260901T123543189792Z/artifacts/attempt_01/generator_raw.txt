<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_087_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DEG
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For ships proceeding to latitudes over 80 degrees, at least one GNSS compass or equivalent must be recorded and it must be connected to both main and emergency power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plannedOrActualLatitude ?latitudeNode .
                ?latitudeNode qudt:numericValue ?latitude .
                ?latitudeNode qudt:unit unit:DEG .
                FILTER (?latitude > "80"^^xsd:decimal)

                FILTER NOT EXISTS {
                    $this nltl:gnssCompassOrEquivalentCount ?count .
                    FILTER (
                        datatype(?count) = xsd:integer &&
                        ?count >= 1
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToMainPower ?mainPower .
                    FILTER (isIRI(?mainPower))
                }

                FILTER NOT EXISTS {
                    $this nltl:connectedToEmergencyPower ?emergencyPower .
                    FILTER (isIRI(?emergencyPower))
                }
            }
        """
    ] .
<END_SHACL>
