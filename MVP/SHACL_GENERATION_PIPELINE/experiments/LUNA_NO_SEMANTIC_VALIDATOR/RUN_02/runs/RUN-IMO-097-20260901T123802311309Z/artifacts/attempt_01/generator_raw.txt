gen:IMO097ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (

            sh:property [

            sh:sparql [
                sh:message "Every required rescue boat or lifeboat must carry at least one distress alert device when the ship operates in low air temperature." ;

                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .

                            ?craft nltl:hasDistressAlertDevice ?device .

                    }

                sh:message "Every survival craft must carry at least one on-scene communication device when the ship operates in low air temperature." ;

                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                        $this nltl:hasSurvivalCraft ?craft .

                """
            ]
