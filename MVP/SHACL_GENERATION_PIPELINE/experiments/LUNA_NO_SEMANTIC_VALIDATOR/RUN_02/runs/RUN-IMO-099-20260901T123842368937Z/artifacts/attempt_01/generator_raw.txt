<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_099_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must have a battery-life management procedure."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery-life management procedure must be implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CommunicationDeviceAvailabilityQuantityShape ;
        sh:message "Communication-device availability must be provided in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MaximumExpectedRescueTimeQuantityShape ;
        sh:message "The maximum expected rescue time must be provided in days."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Communication-device availability must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availability ;
                      nltl:maximumExpectedRescueTime ?rescueTime .
                ?availability qudt:numericValue ?availabilityValue ;
                              qudt:unit <http://qudt.org/vocab/unit/DAY> .
                ?rescueTime qudt:numericValue ?rescueValue ;
                             qudt:unit <http://qudt.org/vocab/unit/DAY> .
                FILTER (
                    datatype(?availabilityValue) = xsd:decimal &&
                    datatype(?rescueValue) = xsd:decimal &&
                    ?availabilityValue < ?rescueValue
                )
            }
        """
    ] .

gen:CommunicationDeviceAvailabilityQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DAY> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MaximumExpectedRescueTimeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DAY> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
