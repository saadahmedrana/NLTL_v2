<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO102ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled ice condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:class nltl:polarTrainingShipTypeValue ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled polar training ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO102CrewMemberShape
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:class nltl:polarTrainingCrewRoleValue ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented crew member must have exactly one controlled polar training role."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable crew member must have a matching required training level, polar training record, and valid STCW qualification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                      nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role .

                BIND(
                    IF(
                        ?ice = nltl:iceFreeIceCondition,
                        nltl:polarTrainingNotApplicable,
                        IF(
                            ?ice = nltl:openWaterIceCondition &&
                            ?shipType = nltl:otherShipPolarTrainingType,
                            nltl:polarTrainingNotApplicable,
                            IF(
                                ?ice = nltl:openWaterIceCondition,
                                nltl:basicPolarTrainingLevel,
                                IF(
                                    ?role = nltl:masterPolarTrainingRole ||
                                    ?role = nltl:chiefMatePolarTrainingRole,
                                    nltl:advancedPolarTrainingLevel,
                                    nltl:basicPolarTrainingLevel
                                )
                            )
                        )
                    ) AS ?expectedLevel
                )

                FILTER(?expectedLevel != nltl:polarTrainingNotApplicable)

                FILTER NOT EXISTS {
                    $this nltl:hasPolarTrainingRecord ?evidenceRecord .
                }

                FILTER NOT EXISTS {
                    $this nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?requiredLevel ;
                            nltl:trainingRecordLevel ?recordLevel ;
                            nltl:stcwQualificationValid ?valid .
                    FILTER(?requiredLevel = ?expectedLevel)
                    FILTER(?recordLevel = ?expectedLevel)
                    FILTER(?valid = true)
                }
            }
        """
    ] .

</END_SHACL>
