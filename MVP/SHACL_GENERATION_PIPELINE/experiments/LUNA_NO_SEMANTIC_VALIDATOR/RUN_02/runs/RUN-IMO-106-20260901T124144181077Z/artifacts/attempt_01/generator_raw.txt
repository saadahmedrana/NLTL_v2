<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-106-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:message "A ship must provide the discharged material classification."
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition applicability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                    FILTER (?applicable = false)
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?invalidApplicable .
                    FILTER (?invalidApplicable != false)
                }
            }
        """
    ] .
<END_SHACL>
