@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_114_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:cargoTankType ;
                sh:hasValue "3"
            ] ;
            sh:property [
                sh:path nltl:administrationCarriageApprovalStatus ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:applicableCertificatePolarOperationEntryPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:sparql [
                sh:message "The conditional approval and certificate-entry requirements are not satisfied for an applicable ship." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:shipCategory ?category ;
                              nltl:constructionDate ?date ;
                              nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 ;
                              nltl:cargoTankType ?tankType .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                            && ?date >= "2017-01-01"^^xsd:date
                            && ?requiresType3 = true
                            && ?tankType = "3"
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
