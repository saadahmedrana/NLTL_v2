<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO119Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path nltl:extendedPeriodOperation ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeOccurs ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:approvedSewageTreatmentPlantUsed ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:administrationDischargeApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState ;
        sh:in ( nltl:evidenceStateApproved )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, the approved sewage treatment plant and Administration discharge approval must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:iceConcentration ?quantity ;
                      nltl:extendedPeriodOperation true ;
                      nltl:sewageDischargeOccurs true .
                ?quantity qudt:numericValue ?ice .
                FILTER(?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER(datatype(?ice) = xsd:decimal)
                FILTER(?ice > 1)
                FILTER NOT EXISTS {
                    $this nltl:approvedSewageTreatmentPlantUsed true .
                }
                FILTER NOT EXISTS {
                    $this nltl:administrationDischargeApprovalStatus
                        nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
