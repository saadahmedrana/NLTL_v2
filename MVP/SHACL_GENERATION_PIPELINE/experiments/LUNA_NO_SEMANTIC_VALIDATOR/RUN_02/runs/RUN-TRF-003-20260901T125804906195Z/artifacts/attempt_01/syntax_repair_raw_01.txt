<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF003Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single construction contract date is required to determine applicability of the 2010 Ice Class Regulations."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:iceClassRegulationEdition2010 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Ships contracted for construction from 1 January 2012 to before 1 January 2019 must identify the 2010 Ice Class Regulations."
            ] ;
            sh:sparql [
                sh:message "The construction contract date is not within the applicability period for the 2010 Ice Class Regulations." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?d .
                        FILTER (
                            YEAR(?d) > 2012 ||
                            (YEAR(?d) = 2012 && MONTH(?d) > 1) ||
                            (YEAR(?d) = 2012 && MONTH(?d) = 1 && DAY(?d) >= 1)
                        )
                        FILTER (
                            YEAR(?d) < 2019
                        )
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "The construction contract date must be before 1 January 2012 or on or after 1 January 2019 when the 2010 Ice Class Regulations are not applicable." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionContractDate ?d .
                        FILTER (
                            YEAR(?d) < 2012 ||
                            YEAR(?d) >= 2019
                        )
                    }
                """
            ]
        ]
    ) .
</END_SHACL>
