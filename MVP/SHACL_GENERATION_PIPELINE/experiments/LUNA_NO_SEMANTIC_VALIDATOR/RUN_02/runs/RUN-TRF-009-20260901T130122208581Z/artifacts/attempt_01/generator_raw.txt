<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-009> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:message "The ship is not ice class IA or IA Super, so the conditional deadline rule is not applicable."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                ) ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "Construction stage date is required to determine applicability of the conditional rule."
            ] ;
            sh:sparql [
                sh:message "For an IA or IA Super ship whose construction stage was before 1 September 2003, the required 2017 Ice Class Regulations compliance method and deadline evidence must be satisfied." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?ice ;
                              nltl:constructionStageDate ?constructionDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER ( ?ice IN (nltl:iceClassIa, nltl:iceClassIaSuper) )
                        FILTER NOT EXISTS {
                            $this nltl:iceClass ?otherIce .
                            FILTER ( ?otherIce NOT IN (nltl:iceClassIa, nltl:iceClassIaSuper) )
                        }
                        FILTER (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )
                        FILTER NOT EXISTS {
                            $this nltl:engineOutputComplianceMethod ?method .
                            FILTER (
                                ?method IN (
                                    nltl:traficom2017Section3Point2Point2Method,
                                    nltl:traficom2017Section3Point2Point4Method
                                )
                            )
                        }
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                ) ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "The construction stage date must be on or after 1 September 2003 for the conditional rule to be non-applicable." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?ice ;
                              nltl:constructionStageDate ?constructionDate .
                        FILTER ( ?ice IN (nltl:iceClassIa, nltl:iceClassIaSuper) )
                        FILTER (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        )
    ] .
<END_SHACL>
