<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF017Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Maximum continuous rating power P must be provided as a single quantity in kW."
    ] ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Continuous propulsion machinery output must be provided as a single quantity in kW."
    ] ;
    sh:or (
        gen:TRF017RestrictionAppliesShape
        gen:TRF017RestrictionDoesNotApplyShape
    ) ;
    sh:sparql [
        sh:message "Maximum continuous rating power P must equal the maximum continuous output of the propulsion machinery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?pNode .
                ?pNode qudt:numericValue ?p .

                $this nltl:propulsionMachineryContinuousOutput ?cNode .
                ?cNode qudt:numericValue ?c .

                FILTER (?p != ?c)
            }
        """
    ] ;
    sh:sparql [
        sh:message "When propulsion output is restricted, P must equal the restricted propulsion output." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propulsionOutputRestrictionApplies true ;
                      nltl:maximumContinuousRatingPower ?pNode ;
                      nltl:restrictedPropulsionOutput ?rNode .

                ?pNode qudt:numericValue ?p .
                ?rNode qudt:numericValue ?r .

                FILTER (?p != ?r)
            }
        """
    ] .

gen:TRF017RestrictionAppliesShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The restriction-applicability indicator must be true for the restricted-output case."
    ] ;
    sh:property [
        sh:path nltl:restrictedPropulsionOutput ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Restricted propulsion output must be provided as a single quantity in kW when a restriction applies."
    ] .

gen:TRF017RestrictionDoesNotApplyShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The restriction-applicability indicator must be false for the unrestricted-output case."
    ] .
<END_SHACL>
