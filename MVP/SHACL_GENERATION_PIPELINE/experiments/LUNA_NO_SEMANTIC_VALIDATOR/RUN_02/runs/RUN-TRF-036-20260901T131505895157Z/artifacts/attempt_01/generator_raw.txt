<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF036ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one applicable Finnish-Swedish ice class from Table 4-1."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:node [
            sh:class nltl:iceClassDesignParameterSet ;
            sh:property [
                sh:path nltl:levelIceThickness ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:designIceLoadHeight ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

<urn:nltl:generated-shape:TRF036ParameterSetShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The parameter set must provide level ice thickness as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The parameter set must provide design ice load height as a single metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Level ice thickness and design ice load height must match the Table 4-1 values for the linked ship ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                       nltl:iceClass ?iceClass .
                $this nltl:levelIceThickness ?levelQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity .
                ?levelQuantity qudt:numericValue ?levelValue ;
                               qudt:unit unit:M .
                ?heightQuantity qudt:numericValue ?heightValue ;
                                qudt:unit unit:M .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper &&
                     ?levelValue = 1.0 &&
                     ?heightValue = 0.35)
                    ||
                    (?iceClass = nltl:iceClassIa &&
                     ?levelValue = 0.8 &&
                     ?heightValue = 0.30)
                    ||
                    (?iceClass = nltl:iceClassIb &&
                     ?levelValue = 0.6 &&
                     ?heightValue = 0.25)
                    ||
                    (?iceClass = nltl:iceClassIc &&
                     ?levelValue = 0.4 &&
                     ?heightValue = 0.22)
                )
            }
        """
    ] .
<END_SHACL>
