                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design ice load height is required as a metre quantity."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive 0
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Spacing is required as a positive metre quantity."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The longitudinal plating factor f2 is required as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:datatype qudt:QuantityValue ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;

        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The reported longitudinal plating factor f2 must follow the applicable h/s formula within the specified tolerance, with 0 <= h/s <= 1.8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hQuantity ;
                      nltl:spacing ?sQuantity ;
                      nltl:longitudinalPlatingFactorF2 ?fQuantity .
                ?hQuantity qudt:numericValue ?h .
                ?sQuantity qudt:numericValue ?s .
                ?fQuantity qudt:numericValue ?actual .
                BIND(?h / ?s AS ?ratio)
                BIND(
                    IF(
                        ?ratio <= 1,
                        0.4 / ?ratio + 0.6,
                        1.4 - 0.4 * ?ratio
                    ) AS ?expected
                )
                FILTER(
                    ?ratio < 0 ||
                    ?ratio > 1.8 ||
                    ABS(?actual - ?expected) >
                    IF(
                        1e-6 * ABS(?expected) > 1e-12,
                        1e-6 * ABS(?expected),
                        1e-12
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A corrosion and abrasion addition below 2 mm requires a maintained special surface coating and approved material evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?additionQuantity .
                ?additionQuantity qudt:numericValue ?addition .
                FILTER(?addition < 2)
                FILTER NOT EXISTS {
                    $this nltl:specialSurfaceCoatingMaintained true ;
                          nltl:materialApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] .
