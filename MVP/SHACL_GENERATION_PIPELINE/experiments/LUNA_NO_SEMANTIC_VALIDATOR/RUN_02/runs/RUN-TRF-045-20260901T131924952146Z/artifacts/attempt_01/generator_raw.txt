<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-045_FrameSpanApplicabilityShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A frame span quantity in metres is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:frameSpanWithinIceStrengtheningZone ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The frame span within the ice-strengthening zone in metres is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary frame scantlings shall be used when less than 15 percent of the frame span is within the ice-strengthening zone." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:span ?spanQuantity .
                ?spanQuantity qudt:numericValue ?spanValue .
                $this nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
                ?zoneQuantity qudt:numericValue ?zoneValue .
                FILTER (
                    ?zoneValue < (?spanValue * "0.15"^^xsd:decimal)
                )
                FILTER NOT EXISTS {
                    $this nltl:ordinaryFrameScantlingsUsed ?ordinaryValue .
                    FILTER ( ?ordinaryValue = true )
                }
            }
        """
    ] .
<END_SHACL>
