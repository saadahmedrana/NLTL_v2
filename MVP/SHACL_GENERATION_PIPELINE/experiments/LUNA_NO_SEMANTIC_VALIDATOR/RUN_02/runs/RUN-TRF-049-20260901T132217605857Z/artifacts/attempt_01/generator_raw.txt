<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF049QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

g:TRF049BracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node g:TRF049QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An installed connection bracket must have a single thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened against buckling."
    ] .

g:TRF049FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame attachment must be effectively attached."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node g:TRF049BracketShape
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
    ] .

g:TRF049SupportingStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:minCount 1 ;
        sh:node g:TRF049SupportingAttachmentShape ;
        sh:message "Each linked supporting structure must have a frame attachment."
    ] .

g:TRF049SupportingAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The attachment to the supporting structure must be effective."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:minCount 1 ;
        sh:node g:TRF049BracketShape ;
        sh:message "A longitudinal frame must be attached to each supporting web frame or bulkhead by a bracket."
    ] .

g:TRF049WithinShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node g:TRF049FrameAttachmentShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:node g:TRF049SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:node g:TRF049SupportingStructureShape
    ] .

g:TRF049OutsideShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

g:TRF049FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-strengthened-area applicability of the frame must be recorded."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node g:TRF049QuantityShape
    ] ;
    sh:or (
        g:TRF049WithinShape
        g:TRF049OutsideShape
    ) ;
    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have a connection bracket or similar construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      a nltl:transverseFrame ;
                      nltl:terminatesAtDeckOrIceStringer true ;
                      nltl:hasFrameAttachment ?attachment .
                FILTER NOT EXISTS {
                    ?attachment nltl:hasConnectionBracket ?bracket .
                }
                FILTER NOT EXISTS {
                    ?attachment nltl:similarAttachmentConstructionPresent true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every installed connection bracket must be at least as thick as the frame web plate, with both thicknesses represented in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:hasFrameAttachment|nltl:hasSupportingBulkhead/nltl:hasSupportingStructureAttachment|nltl:hasSupportingWebFrame/nltl:hasSupportingStructureAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?frameQuantity .
                    ?frameQuantity qudt:numericValue ?frameValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?bracket nltl:bracketThickness ?bracketQuantity .
                    ?bracketQuantity qudt:numericValue ?bracketValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER(datatype(?frameValue) = xsd:decimal)
                    FILTER(datatype(?bracketValue) = xsd:decimal)
                    FILTER(?bracketValue >= ?frameValue)
                }
            }
        """
    ] .
<END_SHACL>
