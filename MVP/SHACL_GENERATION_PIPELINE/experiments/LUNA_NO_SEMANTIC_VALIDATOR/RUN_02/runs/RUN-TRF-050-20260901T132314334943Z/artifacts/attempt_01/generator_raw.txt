<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord ;
        sh:minCount 1 ;
        sh:node gen:FrameAttachmentRecordShape ;
        sh:message "Each frame shall have a frame attachment record." ;
    ] .

gen:FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:class nltl:weldTypeValue ;
        sh:minCount 1 ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "The frame shall be attached to the shell by a double continuous weld." ;
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Shell plate butt crossing applicability must be recorded." ;
    ] ;
    sh:or (
        [
            sh:path nltl:shellPlateButtCrossing ;
            sh:hasValue true ;
        ]
        [
            sh:path nltl:shellPlateButtCrossing ;
            sh:hasValue false ;
        ]
        [
            sh:path nltl:scallopingPresent ;
            sh:datatype xsd:boolean ;
            sh:minCount 1 ;
            sh:hasValue false ;
            sh:message "Scalloping is not allowed when no shell plate butt is crossed." ;
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a shell plate butt is crossed, scalloping may be present; otherwise scalloping must be absent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shellPlateButtCrossing ?crossing .
                FILTER (?crossing = "false"^^xsd:boolean)
                FILTER (
                    NOT EXISTS {
                        $this nltl:scallopingPresent "false"^^xsd:boolean .
                    }
                    ||
                    EXISTS {
                        $this nltl:scallopingPresent ?scalloping .
                        FILTER (?scalloping != "false"^^xsd:boolean)
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
