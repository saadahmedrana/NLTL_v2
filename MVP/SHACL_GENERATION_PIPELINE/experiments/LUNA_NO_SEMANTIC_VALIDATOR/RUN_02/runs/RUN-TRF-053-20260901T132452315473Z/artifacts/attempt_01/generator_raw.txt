@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [

        sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M> ;

nltlg:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltlg:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one ice pressure in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one design ice load height in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltlg:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one span in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltlg:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one frame moment factor."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltlg:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must provide one yield strength in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltlg:ForcePerLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design line load must be provided in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltlg:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The section modulus must be provided in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltlg:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The required shear area must be provided in cm2."
    ] ;
    sh:sparql [
        sh:message "The design line load, section modulus, and required shear area must satisfy the specified ice-stringer formulae." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sNode ;
                      nltl:designLineLoad ?qNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sNode qudt:numericValue ?sigma .
                ?qNode qudt:numericValue ?qReported .
                ?zNode qudt:numericValue ?zReported .
                ?aNode qudt:numericValue ?aReported .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?qMNPerM)
                BIND((?qMNPerM * 1000) AS ?qExpected)
                BIND((0.9 * 1.8 * ?qMNPerM * ?l * ?l / (?m * ?sigma) * 1000000) AS ?zExpected)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?qMNPerM * ?l / (2 * ?sigma) * 10000) AS ?aExpected)

                FILTER (
                    ABS(?qReported - ?qExpected) >
                    (0.000001 * ABS(?qExpected) + 0.000000001)
                    ||
                    ABS(?zReported - ?zExpected) >
                    (0.000001 * ABS(?zExpected) + 0.000000001)
                    ||
                    ABS(?aReported - ?aExpected) >
                    (0.000001 * ABS(?aExpected) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
