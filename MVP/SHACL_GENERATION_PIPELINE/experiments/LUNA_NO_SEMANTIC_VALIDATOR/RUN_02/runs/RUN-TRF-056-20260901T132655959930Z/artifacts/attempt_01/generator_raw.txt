<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-056-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node [
            sh:class nltl:weatherdeckHatch ;
            sh:property [
                sh:path nltl:hatchOpeningLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:M
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a weatherdeck hatch longer than half the ship breadth, ship-side deflection and both hatch-cover and hatch-fitting design evidence must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                $this nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingLength .
                FILTER (
                    datatype(?breadth) = xsd:decimal &&
                    datatype(?openingLength) = xsd:decimal &&
                    ?openingLength > (?breadth / 2)
                )
                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflectionQuantity .
                        ?deflectionQuantity qudt:numericValue ?deflection .
                        ?deflectionQuantity qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?deflection) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                        ?coverEvidence a nltl:hatchCoverDesignEvidence .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                        ?fittingEvidence a nltl:hatchFittingDesignEvidence .
                    }
                )
            }
        """
    ] .
</urn:nltl:generated-shape:TRF-056-ShipShape>
<END_SHACL>
