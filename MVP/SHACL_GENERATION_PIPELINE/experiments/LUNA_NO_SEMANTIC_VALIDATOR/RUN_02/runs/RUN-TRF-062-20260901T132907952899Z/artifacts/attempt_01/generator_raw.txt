<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_062_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:minimumClearance ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "A minimum clearance value in metres is required."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeTipHullClearance ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ] ;
        sh:message "A propeller blade tip to hull clearance value in metres is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The minimum propeller blade tip to hull clearance must be no less than the minimum clearance."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerBladeTipHullClearance ?actualQuantity .
                ?actualQuantity qudt:numericValue ?actualValue .
                $this nltl:minimumClearance ?minimumQuantity .
                ?minimumQuantity qudt:numericValue ?minimumValue .
                FILTER (
                    DATATYPE(?actualValue) = xsd:decimal &&
                    DATATYPE(?minimumValue) = xsd:decimal &&
                    ?actualValue < ?minimumValue
                )
            }
        """
    ] .
<END_SHACL>
