<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF_063_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "A propeller count is required to determine whether the twin- or triple-screw applicability condition applies."
    ] ;
    sh:sparql [
        sh:message "For a twin- or triple-screw ship, every side-propeller strengthening envelope must identify a tank top boundary and provide forward and aft extents of at least 1.5 m in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count .
                FILTER (?count = 2 || ?count = 3)
                {
                    FILTER NOT EXISTS {
                        $this nltl:hasSidePropellerStrengtheningEnvelope ?anyEnvelope .
                    }
                }
                UNION
                {
                    $this nltl:hasSidePropellerStrengtheningEnvelope ?envelope .
                    FILTER NOT EXISTS {
                        ?envelope nltl:strengtheningEnvelopeBoundary ?boundary .
                        ?boundary a nltl:tankTop .
                        ?envelope nltl:strengtheningForwardExtent ?forwardQuantity .
                        ?forwardQuantity qudt:numericValue ?forwardValue ;
                                           qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (
                            datatype(?forwardValue) = xsd:decimal &&
                            ?forwardValue >= "1.5"^^xsd:decimal
                        )
                        ?envelope nltl:strengtheningAftExtent ?aftQuantity .
                        ?aftQuantity qudt:numericValue ?aftValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (
                            datatype(?aftValue) = xsd:decimal &&
                            ?aftValue >= "1.5"^^xsd:decimal
                        )
                    }
                }
            }
        """
    ] .
</BEGIN_SHACL>
