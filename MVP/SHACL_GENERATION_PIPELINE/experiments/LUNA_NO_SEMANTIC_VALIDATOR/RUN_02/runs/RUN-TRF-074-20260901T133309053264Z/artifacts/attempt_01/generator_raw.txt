<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must identify a permitted ice class."
    ] ;
    sh:or (
        gen:TRF074NonApplicableShape
        gen:TRF074SufficientDepthShape
        gen:TRF074InsufficientDepthShape
    ) .

gen:TRF074NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF074SufficientDepthShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IB and IC, the ballast-condition propeller top depth must be at least hi, or the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                FILTER NOT EXISTS {
                    ?this nltl:ballastConditionPropellerTopDepth ?depthNode .
                    ?depthNode qudt:numericValue ?depth .
                    ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                    ?hiNode qudt:numericValue ?hi .
                    ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (?depth >= ?hi)
                }
            }
        """
    ] .

gen:TRF074InsufficientDepthShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemDesignedForIceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iceClassIa
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IB and IC, when the ballast-condition propeller top depth is less than hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                FILTER NOT EXISTS {
                    ?this nltl:ballastConditionPropellerTopDepth ?depthNode .
                    ?depthNode qudt:numericValue ?depth .
                    ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                    ?hiNode qudt:numericValue ?hi .
                    ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (?depth < ?hi)
                }
            }
        """
    ] .
