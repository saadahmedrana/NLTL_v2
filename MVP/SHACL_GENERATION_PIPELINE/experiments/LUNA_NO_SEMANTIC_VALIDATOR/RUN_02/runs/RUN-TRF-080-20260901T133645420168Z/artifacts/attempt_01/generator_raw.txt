<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "iceLoadCycleCount is required as an integer input." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "propellerBladeCount is required as an integer input." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue qudt:UNITLESS ;
            ]
        ] ;
        sh:message "effectiveIceLoadCycleCount must provide a decimal unitless quantity value." ;
    ] ;
    sh:sparql [
        sh:message "effectiveIceLoadCycleCount must equal iceLoadCycleCount multiplied by propellerBladeCount within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?resultNode .
                ?resultNode qudt:numericValue ?actual .
                BIND((xsd:decimal(?nice) * xsd:decimal(?blades)) AS ?expected)
                FILTER(
                    ABS(xsd:decimal(?actual) - ?expected)
                    >
                    (0.000001 * ABS(?expected) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
