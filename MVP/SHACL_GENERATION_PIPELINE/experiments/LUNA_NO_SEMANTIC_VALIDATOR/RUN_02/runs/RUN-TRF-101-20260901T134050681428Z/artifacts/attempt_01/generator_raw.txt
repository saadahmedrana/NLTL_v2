<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BladeFailureRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node gen:BladeFailureLoadCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedNonlinearBladeStressAnalysis ;
                sh:minCount 1 ;
                sh:class nltl:approvalRecord
            ]
        ]
        [
            sh:node gen:BladeFailureFormulaShape
        ]
    ) ;
    sh:sparql [
        sh:message "bladeFailureStatus must be true exactly when bladeTipOffset exceeds 10 percent of propellerDiameter" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?dNode ;
                      nltl:bladeTipOffset ?tNode ;
                      nltl:bladeFailureStatus ?status .
                ?dNode qudt:numericValue ?dValue .
                ?tNode qudt:numericValue ?tValue .
                FILTER (
                    (?tValue > (0.10 * ?dValue) && ?status != true) ||
                    (?tValue <= (0.10 * ?dValue) && ?status != false)
                )
            }
        """
    ] .

        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1

gen:BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive 0.8 ;
                sh:maxInclusive 0.8
            ]
        ]
    ] .

gen:BladeFailureFormulaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "bladeReferenceStrength1 and bladeFailureUltimateLoad must satisfy the declared blade-failure calculation within numerical tolerance" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:propellerDiameter ?dNode ;
                      nltl:bladeMinimumYieldStrength ?yNode ;
                      nltl:bladeUltimateTensileStrength ?uNode ;
                      nltl:bladeReferenceStrength1 ?refNode ;
                      nltl:bladeFailureUltimateLoad ?fNode .
                ?cNode qudt:numericValue ?c .
                ?tNode qudt:numericValue ?t .
                ?rNode qudt:numericValue ?r .
                ?dNode qudt:numericValue ?d .
                ?yNode qudt:numericValue ?y .
                ?uNode qudt:numericValue ?u .
                ?refNode qudt:numericValue ?ref .
                ?fNode qudt:numericValue ?f .
                BIND ((0.6 * ?y) + (0.4 * ?u) AS ?expectedRef)
                BIND ((300 * ?c * ?t * ?t * ?expectedRef) /
                      ((0.8 * ?d) - (2 * ?r)) AS ?expectedF)
                FILTER (
                    ((ABS(?ref - ?expectedRef) >
                      ((ABS(?expectedRef) * 0.000001) + 0.000000001))) ||
                    ((ABS(?f - ?expectedF) >
                      ((ABS(?expectedF) * 0.000001) + 0.000000001)))
                )
            }
        """
    ] .
<END_SHACL>
