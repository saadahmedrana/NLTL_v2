            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bladeExpandedAreaRatio/qudt:numericValue ?ear ;

                      nltl:leadingEdgeChordPortionAtZeroPointEightRadius/qudt:numericValue ?le ;

                BIND((4 * ?ear / ?z) AS ?ratio)
                BIND((0.7 * (1 - (?ratio * ?ratio * ?ratio))) AS ?rawFactor)
                BIND(IF(?rawFactor < 0.3, 0.3, ?rawFactor) AS ?expectedFactor)
                BIND(IF(?le > (0.8 * ?te), ?le, (0.8 * ?te)) AS ?arm)
                BIND((?arm * ?expectedFactor * ?force) AS ?expectedTorque)
                FILTER(
                    ABS(?factor - ?expectedFactor) >
                    (ABS(?expectedFactor) * 0.000001 + 0.000001)
                )
                FILTER(
                    ABS(?bladeTorque - ?expectedTorque) >
                    (ABS(?expectedTorque) * 0.000001 + 0.000001)
                )
                FILTER(
                    ABS(?maximumTorque - ?expectedTorque) >
                    (ABS(?expectedTorque) * 0.000001 + 0.000001)
                )
            }
        """
