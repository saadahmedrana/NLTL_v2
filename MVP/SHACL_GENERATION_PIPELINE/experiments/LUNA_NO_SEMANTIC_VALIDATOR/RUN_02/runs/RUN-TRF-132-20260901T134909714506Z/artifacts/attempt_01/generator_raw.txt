<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF132Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify an ice class for engine-output applicability." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula base power P_0 must be supplied as a single kW quantity." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula displacement Delta must be supplied as a single decimal quantity." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula factor f_1 must be supplied as a single dimensionless quantity." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula factor f_2 must be supplied as a single dimensionless quantity." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula factor f_3 must be supplied as a single dimensionless quantity." ;
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The formula factor f_4 must be supplied as a single decimal quantity." ;
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The engine output must be supplied as a single kW quantity." ;
    ] ;
    sh:sparql [
        sh:message "The engine output must be at least f_1*f_2*f_3*(f_4*Delta + P_0), within a relative tolerance of 1e-6 with an absolute fallback of 1e-6 kW, and at least 740 kW for ice classes IB and IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:engineOutputFormulaBasePowerP0 ?p0Node ;
                      nltl:engineOutputFormulaDisplacement ?deltaNode ;
                      nltl:engineOutputFormulaFactorF1 ?f1Node ;
                      nltl:engineOutputFormulaFactorF2 ?f2Node ;
                      nltl:engineOutputFormulaFactorF3 ?f3Node ;
                      nltl:engineOutputFormulaFactorF4 ?f4Node ;
                      nltl:maximumContinuousRatingPower ?outputNode .

                ?p0Node qudt:numericValue ?p0 .
                ?deltaNode qudt:numericValue ?delta .
                ?f1Node qudt:numericValue ?f1 .
                ?f2Node qudt:numericValue ?f2 .
                ?f3Node qudt:numericValue ?f3 .
                ?f4Node qudt:numericValue ?f4 .
                ?outputNode qudt:numericValue ?output .

                BIND (?f1 * ?f2 * ?f3 * ((?f4 * ?delta) + ?p0) AS ?expected)
                FILTER (
                    (?output + (
                        IF (ABS(?expected) * 0.000001 > 0.000001,
                            ABS(?expected) * 0.000001,
                            0.000001)
                    ) < ?expected)
                    ||
                    (
                        (?ice = nltl:iceClassIb || ?ice = nltl:iceClassIc)
                        && ?output < 740
                    )
                )
            }
        """
    ] .
<END_SHACL>
