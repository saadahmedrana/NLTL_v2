        sh:maxCount 1 ;

    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;

        sh:maxCount 1 ;

    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:node gen:TRF133WarningTriangleMarkingShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one conforming warning-triangle marking."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;

                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:node [
                    sh:property [
                        sh:path nltl:draughtMarkAftReferencePoint ;
                        sh:hasValue nltl:loadLineRingCentreReference ;

                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ) .
