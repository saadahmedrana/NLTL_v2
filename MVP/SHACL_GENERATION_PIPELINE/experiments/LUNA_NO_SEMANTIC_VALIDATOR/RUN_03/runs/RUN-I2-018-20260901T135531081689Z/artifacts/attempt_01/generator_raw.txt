<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_018_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:class nltl:iceLoadPatch ;
        sh:minCount 1 ;
        sh:message "The ship must identify its design ice load patch."
    ] ;

    sh:property [
        sh:path nltl:hasPeakPressureFactorLookupCase ;
        sh:class nltl:peakPressureFactorLookupCase ;
        sh:minCount 1 ;
        sh:node gen:I2_018_PeakPressureFactorLookupCaseShape ;
        sh:message "The ship must provide a peak pressure factor lookup case."
    ] ;

    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:I2_018_TableLookupCaseShape ;
        sh:message "The ship must provide the cited table lookup case and selection evidence."
    ] ;

    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_AverageIcePressureShape ;
        sh:message "Average ice pressure must be reported in MPa."
    ] ;

    sh:property [
        sh:path nltl:iceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_IceForceShape ;
        sh:message "Ice force must be provided in kN."
    ] ;

    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_LengthShape ;
        sh:message "Load patch height must be provided in metres."
    ] ;

    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_LengthShape ;
        sh:message "Load patch length must be provided in metres."
    ] ;

    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_PeakPressureFactorShape ;
        sh:message "The ship must report the selected peak pressure factor."
    ] ;

    sh:sparql [
        sh:message "Average ice pressure must equal ice force converted from kN to MN divided by load patch height and length, and the reported peak pressure factor must equal the selected table factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?avgNode ;
                      nltl:iceForce ?forceNode ;
                      nltl:loadPatchHeight ?heightNode ;
                      nltl:loadPatchLength ?lengthNode ;
                      nltl:peakPressureFactor ?shipFactorNode ;
                      nltl:hasPeakPressureFactorLookupCase ?lookupCase .

                ?avgNode qudt:numericValue ?actualAverage .
                ?forceNode qudt:numericValue ?force .
                ?heightNode qudt:numericValue ?height .
                ?lengthNode qudt:numericValue ?length .
                ?shipFactorNode qudt:numericValue ?shipFactor .
                ?lookupCase nltl:selectedPeakPressureFactor ?selectedFactorNode .
                ?selectedFactorNode qudt:numericValue ?selectedFactor .

                BIND ((?force / 1000) / (?height * ?length) AS ?expectedAverage)

                FILTER (
                    ABS(?actualAverage - ?expectedAverage) >
                    (0.000001 * ABS(?expectedAverage) + 0.000000001)
                    ||
                    ?shipFactor != ?selectedFactor
                )
            }
        """
    ] .

gen:I2_018_AverageIcePressureShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_018_IceForceShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_018_LengthShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_018_PeakPressureFactorShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_018_PeakPressureFactorLookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:localizedStructuralMemberCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bottomStructureFrameCategory
            nltl:longitudinallyFramedPlatingCategory
            nltl:stringerSideLongitudinalOrWebFrameCategory
            nltl:transverseFrameWithLoadDistributingStringersCategory
            nltl:transverseFrameWithoutLoadDistributingStringersCategory
            nltl:transverselyFramedPlatingCategory
        ) ;
        sh:message "The localized structural member category must be one of the permitted canonical categories."
    ] ;
    sh:property [
        sh:path nltl:selectedPeakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_018_PeakPressureFactorShape ;
        sh:message "The lookup case must provide the selected peak pressure factor."
    ] .

gen:I2_018_TableLookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:iacsUrI2Table3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The peak pressure factor must reference IACS UR I2 Table 3."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The table selection must include lookup selection evidence."
    ] .
<END_SHACL>
