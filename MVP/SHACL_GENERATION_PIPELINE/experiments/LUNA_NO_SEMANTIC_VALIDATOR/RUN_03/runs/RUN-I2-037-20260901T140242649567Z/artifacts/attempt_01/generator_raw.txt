gen:I2_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_037_StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "Each ship shall identify the structural members covered by the ice-load design requirement."
    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatchDesignCase ;
        sh:class nltl:iceLoadPatchDesignCase ;
        sh:node gen:I2_037_IceLoadPatchDesignCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship shall provide an ice-load-patch design case for this requirement."
    ] .

gen:I2_037_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:class nltl:loadCase ;
        sh:node gen:I2_037_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "Each structural member shall have a load case representing the required ice-load application."
    ] ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each structural member shall provide its capacity as a single force quantity in kN."
    ] .

gen:I2_037_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member load case shall identify the combined bending-and-shear effect used for the capacity assessment."
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member load case shall identify the location where the ice load patch is applied."
    ] .

gen:I2_037_IceLoadPatchDesignCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-load-patch design case shall confirm that the selected application location minimizes member capacity under combined bending and shear."
    ] .
