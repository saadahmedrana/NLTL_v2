<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:I2-055:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qudtunit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:StressQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudtunit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:StrengthCriteriaShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:node nltlgen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide one design stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:node nltlgen:StressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide one permissible stress in MPa."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The strength criterion status must be true."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Design stress must not exceed permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designStress ?designQuantity .
                ?designQuantity qudt:numericValue ?designValue ;
                    qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                $this nltl:permissibleStress ?permissibleQuantity .
                ?permissibleQuantity qudt:numericValue ?permissibleValue ;
                    qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER ( ?designValue > ?permissibleValue )
            }
        """
    ] .
<END_SHACL>
