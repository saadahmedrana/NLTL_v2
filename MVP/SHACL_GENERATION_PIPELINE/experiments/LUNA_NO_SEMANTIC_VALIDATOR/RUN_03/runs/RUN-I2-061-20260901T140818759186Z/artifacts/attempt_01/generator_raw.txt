<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:message "Each represented calculation case must identify its assessed hull structure and prescribed analytical procedure."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:HullStructureShape ;
        sh:message "A calculation case must identify exactly one assessed hull structure."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The prescribed analytical procedure must be used; direct calculations must not replace it."
    ] .

gen:HullStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:hullStructure ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ThicknessQuantityShape ;
        sh:message "The assessed hull structure must provide its thickness as a metre quantity."
    ] .

gen:ThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
