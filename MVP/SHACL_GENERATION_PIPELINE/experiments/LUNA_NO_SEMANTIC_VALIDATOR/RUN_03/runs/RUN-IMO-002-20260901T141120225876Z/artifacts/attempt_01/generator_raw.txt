<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-002-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its design ice condition to determine applicability of the category rule."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare its Polar Code category to determine and satisfy the category rule."
    ] ;
    sh:sparql [
        sh:message "A ship designed to operate in at least thin first-year ice with possible old-ice inclusions and not declared as category A must be declared category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?iceCondition .
                $this nltl:shipCategory ?category .
                FILTER (
                    ?iceCondition = nltl:thinFirstYearIceWithPossibleOldIceInclusions
                    && ?category != nltl:polarShipCategoryA
                    && ?category != nltl:polarShipCategoryB
                )
            }
        """
    ] .
<END_SHACL>
