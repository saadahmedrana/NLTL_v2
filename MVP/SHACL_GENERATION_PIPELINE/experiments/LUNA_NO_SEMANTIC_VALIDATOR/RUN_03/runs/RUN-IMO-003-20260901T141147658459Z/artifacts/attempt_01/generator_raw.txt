<BEGIN_SHACL>
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:IMO003ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must declare a design ice condition."
    ] ;
    sh:or (
        gs:IMO003OpenWaterBranch
        gs:IMO003LessSevereBranch
        gs:IMO003NonApplicableBranch
    ) .

gs:IMO003OpenWaterBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:hasValue nltl:openWaterIceCondition
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:message "A ship designed for open water must have polar ship category C."
    ] .

gs:IMO003LessSevereBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:hasValue nltl:iceConditionLessSevereThanCategoryAAndB
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:message "A ship designed for ice conditions less severe than categories A and B must have polar ship category C."
    ] .

gs:IMO003NonApplicableBranch
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The design ice condition is applicable to the category C rule and requires category C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:openWaterIceCondition,
                        nltl:iceConditionLessSevereThanCategoryAAndB
                    )
                )
            }
        """
    ] .
<END_SHACL>
