<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship has sea-ice origin, no more than one winter growth, and ice thickness from 0.3 m through 2.0 m, its ice type classification must be first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
           nltl:winterGrowthCount ?growth ;
           nltl:iceThickness ?thicknessNode .
    ?thicknessNode qudt:numericValue ?thickness ;
                   qudt:unit unit:M .
    FILTER (datatype(?growth) = xsd:integer)
    FILTER (?growth <= 1)
    FILTER (datatype(?thickness) = xsd:decimal)
    FILTER (?thickness >= "0.3"^^xsd:decimal &&
            ?thickness <= "2.0"^^xsd:decimal)
    FILTER (
        NOT EXISTS {
            $this nltl:iceTypeClassification nltl:firstYearIce .
        }
        ||
        EXISTS {
            $this nltl:iceTypeClassification ?classification .
            FILTER (?classification != nltl:firstYearIce)
        }
    )
}
""" ;
    ] .
<END_SHACL>
