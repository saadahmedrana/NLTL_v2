<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_016_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:node gen:IMO_016_CertificateShape
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record the language used for the Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:approvedTranslationPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When the certificate language is not English, French, or Spanish, an approved translation must be recorded as present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (
                    STR(?language) NOT IN ("English", "French", "Spanish")
                )
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent ?translation .
                    FILTER (?translation = true)
                }
            }
        """
    ] .

gen:IMO_016_CertificateShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:certificateFormModelValue ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate must use the Appendix I model."
    ] .
<END_SHACL>
