<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The polar certificate supplement presence indicator must be true."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Endorsement dates are required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate form is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gs:IMO017SupplementShape ;
        sh:minCount 1 ;
        sh:message "A Polar Certificate supplement with required equipment records is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gs:IMO017SolasScheduleShape ;
        sh:minCount 1 ;
        sh:message "A SOLAS certificate schedule with certificate date records is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each SOLAS certificate schedule date must match the corresponding Polar Ship Certificate date category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSolasCertificateSchedule ?schedule .
                ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                ?record nltl:certificateScheduleDate ?date ;
                        nltl:certificateScheduleDateType ?type .
                FILTER (
                    (?type = nltl:validityDateCategory &&
                     NOT EXISTS {
                         $this nltl:polarCertificateValidityDates ?date
                     })
                    ||
                    (?type = nltl:surveyDateCategory &&
                     NOT EXISTS {
                         $this nltl:surveyDates ?date
                     })
                    ||
                    (?type = nltl:endorsementDateCategory &&
                     NOT EXISTS {
                         $this nltl:endorsementDates ?date
                     })
                )
            }
        """
    ] .

gs:IMO017SupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Certificate supplement must record required equipment."
    ] .

gs:IMO017SolasScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gs:IMO017DateRecordShape ;
        sh:minCount 1 ;
        sh:message "The SOLAS certificate schedule must contain certificate date records."
    ] .

gs:IMO017DateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A certificate schedule date record must contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:minCount 1 ;
        sh:message "A certificate schedule date record must identify a permitted date category."
    ] .
<END_SHACL>
