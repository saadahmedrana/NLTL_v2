<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_040_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_040_ApplicableShape
        gen:IMO_040_NotApplicableShape
    ) .

gen:IMO_040_ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must identify whether ice accretion is likely."
    ] ;
    sh:property [
        sh:path nltl:hasRelevantHatchOrDoor ;
        sh:node gen:IMO_040_HatchOrDoorShape
    ] .

gen:IMO_040_NotApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "The ship must identify whether ice accretion is likely."
    ] .

gen:IMO_040_HatchOrDoorShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasIceOrSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:message "Each relevant hatch or door must have associated means to remove or prevent ice and snow accretion."
    ] .
<END_SHACL>
