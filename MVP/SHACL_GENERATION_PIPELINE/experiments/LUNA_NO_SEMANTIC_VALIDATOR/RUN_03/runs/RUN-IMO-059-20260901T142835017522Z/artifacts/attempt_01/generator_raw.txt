<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO059ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFirefighterOutfit ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:firefighterOutfitItem ;
        sh:node gen:IMO059FirefighterOutfitShape ;
        sh:message "Each firefighter outfit must be stored at a warm-class location."
    ] .

gen:IMO059FirefighterOutfitShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStorageLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:physicalStorageLocation ;
        sh:node gen:IMO059StorageLocationShape ;
        sh:minCount 1 ;
        sh:message "Each firefighter outfit must have a storage location."
    ] .

gen:IMO059StorageLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:storageTemperatureClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:storageTemperatureClassValue ;
        sh:hasValue nltl:warmStorageLocation ;
        sh:minCount 1 ;
        sh:message "The storage location for a firefighter outfit must be classified as warm."
    ] .
<END_SHACL>
