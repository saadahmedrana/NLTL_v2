<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO_070_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each person on board a passenger ship must have one assigned, properly sized immersion suit or thermal protective aid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:passengerShip .
                FILTER (
                    NOT EXISTS {
                        $this nltl:sizeCompatibilityStatus ?status .
                        FILTER (
                            datatype(?status) = xsd:boolean &&
                            ?status = true
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:hasPersonOnBoardMember ?person .
                        FILTER NOT EXISTS {
                            ?person a nltl:personOnBoardMember ;
                                nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                            ?assignment a nltl:personalSurvivalEquipmentAssignment ;
                                nltl:assignedPersonalSurvivalEquipment ?equipment .
                            ?equipment a ?equipmentType .
                            FILTER (
                                ?equipmentType IN (
                                    nltl:immersionSuitEquipment,
                                    nltl:thermalProtectiveAidEquipment
                                )
                            )
                        }
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
