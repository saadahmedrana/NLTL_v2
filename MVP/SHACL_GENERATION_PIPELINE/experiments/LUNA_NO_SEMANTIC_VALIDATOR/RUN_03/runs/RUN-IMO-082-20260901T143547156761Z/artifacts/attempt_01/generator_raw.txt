<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_082_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The ship shall record compliance with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship shall provide its bridge configuration to determine whether a clear view astern is required."
    ] ;
    sh:sparql [
        sh:message "When the bridge configuration requires an astern view, the ship shall record a compliant clear view astern status." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>

SELECT $this
WHERE {
    $this nltl:bridgeConfigurationType
        nltl:asternViewRequiredBridgeConfiguration .
    FILTER NOT EXISTS {
        $this nltl:clearViewAsternStatus ?status .
        FILTER ( ?status = true )
    }
}
"""
    ] .
<END_SHACL>
