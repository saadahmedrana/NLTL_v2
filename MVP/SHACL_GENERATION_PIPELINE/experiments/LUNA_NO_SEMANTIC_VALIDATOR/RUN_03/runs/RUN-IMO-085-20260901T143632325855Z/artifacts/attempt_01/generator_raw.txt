<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO085CategoryCBranch
        gen:IMO085Pre2017Branch
        gen:IMO085EnclosedBranch
        gen:IMO085ProtectedBranch
    ) .

gen:IMO085CategoryAB a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:datatype nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] .

gen:IMO085CategoryC a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:datatype nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO085DateAny a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] .

gen:IMO085DatePre2017 a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2016-12-31"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO085Date2017OrLater a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO085Enclosed a sh:NodeShape ;
    sh:property [
        sh:path nltl:bridgeWingsEnclosed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO085Protected a sh:NodeShape ;
    sh:property [
        sh:path nltl:bridgeWingProtectionDesignStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IMO085CategoryCBranch a sh:NodeShape ;
    sh:and (
        gen:IMO085CategoryC
        gen:IMO085DateAny
    ) .

gen:IMO085Pre2017Branch a sh:NodeShape ;
    sh:and (
        gen:IMO085CategoryAB
        gen:IMO085DatePre2017
    ) .

gen:IMO085EnclosedBranch a sh:NodeShape ;
    sh:and (
        gen:IMO085CategoryAB
        gen:IMO085Date2017OrLater
        gen:IMO085Enclosed
    ) .

gen:IMO085ProtectedBranch a sh:NodeShape ;
    sh:and (
        gen:IMO085CategoryAB
        gen:IMO085Date2017OrLater
        gen:IMO085Protected
    ) .
<END_SHACL>
