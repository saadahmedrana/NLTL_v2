<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-099-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must have a battery-life management procedure."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery-life management procedure must be implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-099-DayQuantityShape> ;
        sh:message "Communication device availability must be recorded in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-099-DayQuantityShape> ;
        sh:message "The maximum expected rescue time must be recorded in days."
    ] ;
    sh:sparql [
        sh:message "Communication equipment availability must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availability .
                ?availability qudt:numericValue ?availabilityValue .
                $this nltl:maximumExpectedRescueTime ?rescueTime .
                ?rescueTime qudt:numericValue ?rescueValue .
                FILTER (?availabilityValue < ?rescueValue)
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-099-DayQuantityShape> a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DAY>
    ] .
<END_SHACL>
