<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:IMO109Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:cargoTankCarriesOil ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a Category A or B non-oil-tanker ship constructed on or after 1 January 2017 whose cargo tanks carry oil, cargo tank outer shell separation must be at least 0.76 m in unit:M." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true .

                FILTER ( ?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ) )
                FILTER ( ?shipType != "oil_tanker" )
                FILTER NOT EXISTS {
                    $this a nltl:oilTanker
                }
                FILTER ( ?constructionDate >= "2017-01-01"^^xsd:date )

                FILTER (
                    NOT EXISTS {
                        $this nltl:cargoTankOuterShellSeparation ?quantity .
                        ?quantity qudt:numericValue ?value ;
                                  qudt:unit unit:M .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value >= "0.76"^^xsd:decimal
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:cargoTankOuterShellSeparation ?quantity .
                        FILTER NOT EXISTS {
                            ?quantity qudt:numericValue ?value ;
                                      qudt:unit unit:M .
                            FILTER (
                                datatype(?value) = xsd:decimal &&
                                ?value >= "0.76"^^xsd:decimal
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
