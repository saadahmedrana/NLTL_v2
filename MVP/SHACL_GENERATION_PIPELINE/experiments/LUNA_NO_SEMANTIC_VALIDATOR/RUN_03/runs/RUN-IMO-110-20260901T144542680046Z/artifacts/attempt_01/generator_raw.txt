<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one valid polar ship category."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one ship type."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must have one deadweight quantity expressed in metric tonnes."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date."
    ] ;
    sh:sparql [
        sh:message "For a Category A or B oil tanker of less than 5,000 tonnes constructed on or after 1 January 2017, the required double-bottom and wing-tank protection statuses must both be compliant." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipCategory ?category .
    FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
    $this nltl:shipType ?shipType .
    FILTER (?shipType = "oil_tanker")
    $this nltl:deadweight ?deadweight .
    ?deadweight qudt:numericValue ?deadweightValue .
    ?deadweight qudt:unit unit:TON_Metric .
    FILTER (datatype(?deadweightValue) = xsd:decimal && ?deadweightValue < 5000.0)
    $this nltl:constructionDate ?constructionDate .
    FILTER (
        YEAR(?constructionDate) > 2017 ||
        (YEAR(?constructionDate) = 2017 && MONTH(?constructionDate) > 1) ||
        (YEAR(?constructionDate) = 2017 && MONTH(?constructionDate) = 1 && DAY(?constructionDate) >= 1)
    )
    FILTER (
        NOT EXISTS {
            $this a nltl:oilTanker .
        } ||
        NOT EXISTS {
            $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus "true"^^xsd:boolean .
        } ||
        EXISTS {
            $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus "false"^^xsd:boolean .
        } ||
        NOT EXISTS {
            $this nltl:wingTankProtectionStatus "true"^^xsd:boolean .
        } ||
        EXISTS {
            $this nltl:wingTankProtectionStatus "false"^^xsd:boolean .
        }
    )
}
"""
    ] .
<END_SHACL>
