<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-DirectStatic> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Category A or B ships constructed on or after 2017-01-01, each oil residue tank or oily bilge water holding tank with individual capacity greater than 30 m3 must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))

                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )

                $this nltl:tankType ?tankType .
                FILTER (?tankType IN (
                    "oil_residue_tank"^^xsd:string,
                    "oily_bilge_water_holding_tank"^^xsd:string
                ))

                $this nltl:individualTankCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue ;
                          qudt:unit unit:M3 .
                FILTER (
                    datatype(?capacityValue) = xsd:decimal &&
                    ?capacityValue > 30
                )

                FILTER NOT EXISTS {
                    $this nltl:tankOuterShellSeparation ?separation .
                    ?separation qudt:numericValue ?separationValue ;
                                qudt:unit unit:M .
                    FILTER (
                        datatype(?separationValue) = xsd:decimal &&
                        ?separationValue >= 0.76
                    )
                }
            }
        """
    ] .
<END_SHACL>
