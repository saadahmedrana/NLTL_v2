<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-112> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Each ship must provide an operating area to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:sparql [
        sh:message "In Arctic waters, discharge of noxious liquid substances or mixtures containing them into the sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                {
                    $this nltl:operatingArea ?area .
                    FILTER (STR(?area) = "Arctic_waters")
                    FILTER (
                        NOT EXISTS {
                            $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?ok .
                            FILTER (
                                datatype(?ok) = xsd:boolean
                                && ?ok = false
                            )
                        }
                    )
                }
                UNION
                {
                    $this nltl:operatingArea ?area .
                    FILTER (STR(?area) = "Arctic_waters")
                    FILTER (
                        EXISTS {
                            $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?bad .
                            FILTER (
                                !(datatype(?bad) = xsd:boolean && ?bad = false)
                            )
                        }
                    )
                }
            }
        """
    ] .
<END_SHACL>
