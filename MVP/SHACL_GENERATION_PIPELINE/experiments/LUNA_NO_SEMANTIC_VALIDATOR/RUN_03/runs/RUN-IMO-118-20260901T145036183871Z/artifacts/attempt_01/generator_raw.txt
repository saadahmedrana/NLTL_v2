<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-118-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one construction date."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one valid polar ship category."
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "The sewage paragraph 4.2.1.3 compliance status must be a single boolean value when provided."
    ] ;
    sh:sparql [
        sh:message "For a ship constructed on or after 1 January 2017 that is Category A, Category B, or a passenger ship, sewage discharge to sea must be false unless paragraph 4.2.1.3 compliance is true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )
                {
                    $this nltl:shipCategory ?category .
                    FILTER (
                        ?category IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        )
                    )
                }
                UNION
                {
                    $this a nltl:passengerShip .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus
                        "true"^^xsd:boolean .
                }
                FILTER NOT EXISTS {
                    $this nltl:sewageDischargeToSea
                        "false"^^xsd:boolean .
                }
            }
        """
    ] .
<END_SHACL>
