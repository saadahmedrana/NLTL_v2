@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_013_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO26_013_ApplicableShape
        gen:IMO26_013_NonApplicableShape
    ) ;
    sh:message "The latitude condition and its applicable equipment and power-connection requirements must be satisfied." .

gen:IMO26_013_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive "80"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Latitude must be greater than 80 degrees for this applicable branch."
    ] ;
    sh:property [
        sh:path nltl:gnssCompassOrEquivalentCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 1 ;
        sh:message "At least one GNSS compass or equivalent is required above 80 degrees latitude."
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:node [
            sh:class nltl:benchmarkEntity
        ] ;
        sh:minCount 1 ;
        sh:message "The GNSS compass or equivalent must be connected to the ship's main source of power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:node [
            sh:class nltl:benchmarkEntity
        ] ;
        sh:minCount 1 ;
        sh:message "The GNSS compass or equivalent must be connected to the ship's emergency source of power."
    ] .

    a sh:NodeShape ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxInclusive "80"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Latitude must be 80 degrees or less for this non-applicable branch."
    ] .
