<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF004Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2008 Ice Class Regulations."
    ] ;
    sh:or (
        gen:TRF004ApplicableShape
        gen:TRF004NonApplicableShape
    ) .

gen:TRF004ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2010-01-01"^^xsd:date ;
        sh:maxExclusive "2012-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iceClassRegulationEdition2008 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ships contracted between 1 January 2010 inclusive and 1 January 2012 exclusive must identify the 2008 Ice Class Regulations as the applicable edition."
    ] .

gen:TRF004NonApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "The construction contract date must be outside the applicability period of the 2008 Ice Class Regulations for the non-applicable branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    YEAR(?date) = 2010 ||
                    YEAR(?date) = 2011
                )
            }
        """
    ] .
<END_SHACL>
