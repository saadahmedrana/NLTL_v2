<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-005-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF-005-ApplicableBranch>
        <urn:nltl:generated-shape:TRF-005-NonApplicableBranch>
    ) .

<urn:nltl:generated-shape:TRF-005-ApplicableBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:in ( nltl:iceClassRegulationEdition2002 ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ships in the specified date range shall use the 2002 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "The 2002 Ice Class Regulations apply when the construction stage date is on or after 1 September 2003 and the contract date is before 1 January 2010." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate ;
                      nltl:constructionStageDate ?stageDate .
                FILTER (datatype(?contractDate) = xsd:date)
                FILTER (datatype(?stageDate) = xsd:date)
                BIND(YEAR(?stageDate) AS ?stageYear)
                BIND(MONTH(?stageDate) AS ?stageMonth)
                BIND(DAY(?stageDate) AS ?stageDay)
                BIND(YEAR(?contractDate) AS ?contractYear)
                BIND(MONTH(?contractDate) AS ?contractMonth)
                BIND(DAY(?contractDate) AS ?contractDay)
                FILTER (
                    !(
                        (
                            ?stageYear > 2003 ||
                            (
                                ?stageYear = 2003 &&
                                (
                                    ?stageMonth > 9 ||
                                    (?stageMonth = 9 && ?stageDay >= 1)
                                )
                            )
                        ) &&
                        (
                            ?contractYear < 2010 ||
                            (
                                ?contractYear = 2010 &&
                                (
                                    ?contractMonth < 1 ||
                                    (?contractMonth = 1 && ?contractDay < 1)
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .
