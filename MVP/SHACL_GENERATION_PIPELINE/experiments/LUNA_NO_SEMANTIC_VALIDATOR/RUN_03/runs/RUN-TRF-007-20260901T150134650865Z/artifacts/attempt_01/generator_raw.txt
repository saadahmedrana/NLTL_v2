<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall have exactly one Finnish-Swedish ice class."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction stage date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The delivery date is required to determine the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessment date is required to evaluate the compliance deadline."
    ] ;
    sh:or (
        gen:TRF007NonApplicableShape
        gen:TRF007ApplicableShape
    ) .

gen:TRF007NonApplicableShape a sh:NodeShape ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .
                FILTER (
                    ?iceClass NOT IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                    ||
                    YEAR(?constructionDate) > 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        &&
                        (
                            MONTH(?constructionDate) > 9
                            ||
                            (
                                MONTH(?constructionDate) = 9
                                &&
                                DAY(?constructionDate) >= 1
                            )
                        )
                    )
                )
            }
        """ ;
        sh:message "The conditional requirement is non-applicable because the ship is not IA Super or IA, or its construction stage was not before 1 September 2003."
    ] .

gen:TRF007ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship shall identify exactly one permitted 2017 Ice Class Regulations compliance method."
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    YEAR(?constructionDate) < 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        &&
                        (
                            MONTH(?constructionDate) < 9
                            ||
                            (
                                MONTH(?constructionDate) = 9
                                &&
                                DAY(?constructionDate) < 1
                            )
                        )
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        &&
                        MONTH(?assessmentDate) > 1
                    )
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        &&
                        MONTH(?assessmentDate) = 1
                        &&
                        DAY(?assessmentDate) > 1
                    )
                )
            }
        """ ;
        sh:message "The applicable ship has exceeded the deadline of 1 January in the year when twenty years have elapsed since delivery."
    ] .
<END_SHACL>
