<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-009-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A ship must have a permitted Finnish-Swedish ice class to determine applicability of TRF-009."
    ] ;
    sh:sparql [
        sh:message "For an applicable IA Super or IA ship, construction stage date, delivery date, assessment date, and a permitted 2017 compliance method are required, and the assessment date must not be later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )

                FILTER (
                    NOT EXISTS {
                        ?this nltl:constructionStageDate ?constructionDate .
                        FILTER (datatype(?constructionDate) = xsd:date)
                    }
                    ||
                    EXISTS {
                        ?this nltl:constructionStageDate ?constructionDate .
                        FILTER (datatype(?constructionDate) = xsd:date)
                        FILTER (
                            YEAR(?constructionDate) < 2003 ||
                            (
                                YEAR(?constructionDate) = 2003 &&
                                (
                                    MONTH(?constructionDate) < 9 ||
                                    (
                                        MONTH(?constructionDate) = 9 &&
                                        DAY(?constructionDate) < 1
                                    )
                                )
                            )
                        )

                        FILTER NOT EXISTS {
                            ?this nltl:deliveryDate ?deliveryDate ;
                                  nltl:assessmentDate ?assessmentDate ;
                                  nltl:engineOutputComplianceMethod ?method .

                            FILTER (datatype(?deliveryDate) = xsd:date)
                            FILTER (datatype(?assessmentDate) = xsd:date)
                            FILTER (
                                ?method = nltl:traficom2017Section3Point2Point2Method ||
                                ?method = nltl:traficom2017Section3Point2Point4Method
                            )
                            FILTER (
                                YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                                (
                                    YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                    MONTH(?assessmentDate) = 1 &&
                                    DAY(?assessmentDate) = 1
                                )
                            )
                        }
                    }
                )
            }
        """
    ] .
</urn:nltl:generated-shape:TRF-009-ShipShape>
<END_SHACL>
