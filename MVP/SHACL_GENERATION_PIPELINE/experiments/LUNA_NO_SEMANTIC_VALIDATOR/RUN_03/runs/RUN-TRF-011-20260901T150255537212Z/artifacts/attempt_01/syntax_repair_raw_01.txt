<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_011_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify its intended ice-operating waterline and its profile points."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gs:TRF_011_IceWaterlineShape ;
        sh:message "The ship must identify an upper ice waterline and its profile points."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must contain, at every longitudinal position, the maximum vertical coordinate among the intended ice-operating waterline points, and must not contain points at other positions." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
  {
    {
      SELECT ?x (MAX(?yv) AS ?maxY)
      WHERE {
        $this nltl:hasIntendedIceOperatingWaterline ?intended .
        ?intended nltl:hasWaterlineProfilePoint ?ip .
        ?ip nltl:longitudinalPosition ?xq ;
            nltl:verticalCoordinate ?yq .
        ?xq qudt:numericValue ?x .
        ?yq qudt:numericValue ?yv .
      }
      GROUP BY ?x
    }
    OPTIONAL {
      $this nltl:hasUpperIceWaterline ?upper .
      ?upper nltl:hasWaterlineProfilePoint ?up .
      ?up nltl:longitudinalPosition ?uxq ;
          nltl:verticalCoordinate ?uyq .
      ?uxq qudt:numericValue ?ux .
      ?uyq qudt:numericValue ?upY .
      FILTER(?ux = ?x)
    }
    FILTER(!BOUND(?upY) || ?upY != ?maxY)
  }
  UNION
  {
    $this nltl:hasUpperIceWaterline ?upper2 .
    ?upper2 nltl:hasWaterlineProfilePoint ?up2 .
    ?up2 nltl:longitudinalPosition ?uxq2 .
    ?uxq2 qudt:numericValue ?ux2 .
    FILTER NOT EXISTS {
      $this nltl:hasIntendedIceOperatingWaterline ?intended2 .
      ?intended2 nltl:hasWaterlineProfilePoint ?ip2 .
      ?ip2 nltl:longitudinalPosition ?ixq2 .
      ?ixq2 qudt:numericValue ?ix2 .
      FILTER(?ix2 = ?ux2)
    }
  }
}
""" ;
    ] .

gs:TRF_011_IceWaterlineShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gs:TRF_011_ProfilePointShape
    ] .

gs:TRF_011_ProfilePointShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF_011_LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF_011_LengthQuantityShape
    ] .

gs:TRF_011_LengthQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
