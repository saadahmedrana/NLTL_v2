gen:TRF014SummerNotHigherShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The summer load line fresh-water draught must not be higher than the UIWL for the non-marking branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:summerLoadLineFreshWaterDraught ?summer .
                ?summer qudt:numericValue ?summerValue .
                $this nltl:upperIceWaterlineDraught ?uiwl .
                ?uiwl qudt:numericValue ?uiwlValue .
                FILTER (?summerValue > ?uiwlValue)
            }
        """
    ] .

gen:TRF014SummerHigherWithMarkingShape

gen:TRF014DockingAfterAssessmentShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The first scheduled dry docking date must be after the assessment date when markings are not yet due." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:firstScheduledDryDockingDate ?dockingDate .
                $this nltl:assessmentDate ?assessmentDate .
                FILTER (?dockingDate <= ?assessmentDate)
            }
        """
    ] .

gen:TRF014DockingDueShape
    a sh:NodeShape ;
    sh:node gen:TRF014MarkingShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The markings must be present when the first scheduled dry docking date is on or before the assessment date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:firstScheduledDryDockingDate ?dockingDate .
                $this nltl:assessmentDate ?assessmentDate .
                FILTER (?dockingDate > ?assessmentDate)
            }
        """
    ] .
