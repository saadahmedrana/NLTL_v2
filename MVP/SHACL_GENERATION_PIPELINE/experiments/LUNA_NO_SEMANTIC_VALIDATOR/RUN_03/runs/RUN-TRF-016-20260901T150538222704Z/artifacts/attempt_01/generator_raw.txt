        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;

        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

                sh:minCount 1 ;

            sh:property [
                sh:path nltl:usedToReachLowerIceWaterline ;
                sh:datatype xsd:boolean ;

nltlg:NotNavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:NavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node nltlg:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node nltlg:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node nltlg:MassQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node nltlg:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node nltlg:QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:class nltl:ballastTank ;
        sh:node nltlg:BallastTankShape
    ] ;
    sh:sparql [
        sh:message "Draught amidships must be at least the LIWL draught amidships." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:draughtAmidships ?draughtNode ;
                      nltl:liwlDraughtAmidships ?liwlNode .
                ?draughtNode qudt:numericValue ?draught .
                ?liwlNode qudt:numericValue ?liwl .
                FILTER (?draught < ?liwl)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Forward draught must meet the displacement and level-ice-thickness formula threshold." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:forwardDraught ?forwardNode ;
                      nltl:displacementAtUpperIceWaterline ?displacementNode ;
                      nltl:levelIceThickness ?thicknessNode .
                ?forwardNode qudt:numericValue ?forward .
                ?displacementNode qudt:numericValue ?displacement .
                ?thicknessNode qudt:numericValue ?thickness .
                FILTER (
                    ?forward < ((2 + (0.00025 * ?displacement)) * ?thickness)
                    &&
                    ?forward < (4 * ?thickness)
                )
            }
        """
    ] .
