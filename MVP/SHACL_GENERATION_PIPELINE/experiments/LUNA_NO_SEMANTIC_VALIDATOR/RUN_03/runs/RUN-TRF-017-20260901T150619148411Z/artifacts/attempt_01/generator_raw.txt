@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_017 a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:property [
                    sh:path nltl:propulsionOutputRestrictionApplies ;
                    sh:hasValue false ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ]
            ]
            [
                sh:property [
                    sh:path nltl:propulsionOutputRestrictionApplies ;
                    sh:hasValue true ;
                    sh:minCount 1 ;
                    sh:maxCount 1
                ] ;
                sh:property [
                    sh:path nltl:restrictedPropulsionOutput ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node [
                        sh:class qudt:QuantityValue ;
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue unit:KiloW ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must equal the continuous propulsion machinery output, or the restricted propulsion output when a restriction applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?maximumPower ;
                      nltl:propulsionOutputRestrictionApplies ?restricted .

                ?maximumPower qudt:numericValue ?maximumValue .

                {
                    FILTER (?restricted = false)
                    $this nltl:propulsionMachineryContinuousOutput ?expectedPower .
                }
                UNION
                {
                    FILTER (?restricted = true)
                    $this nltl:restrictedPropulsionOutput ?expectedPower .
                }

                ?expectedPower qudt:numericValue ?expectedValue .
                FILTER (?maximumValue != ?expectedValue)
            }
        """
    ] .
