<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF024ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall state one permitted ice class."
    ] ;
    sh:or (
        gen:TRF024NonTargetIceClassShape
        gen:TRF024TargetIceClassAfterConstructionCutoffShape
        gen:TRF024ApplicableShape
    ) .

gen:TRF024NonTargetIceClassShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF024TargetIceClassAfterConstructionCutoffShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction-stage date is required to determine applicability of the twenty-year rule."
    ] ;
    sh:sparql [
        sh:message "For ice class IA Super or IA, the keel must not have been laid or the ship must not have reached a similar construction stage before 1 September 2003 for this non-applicable branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                $this nltl:constructionStageDate ?constructionDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (DATATYPE(?constructionDate) = xsd:date)
                FILTER (
                    YEAR(?constructionDate) < 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) < 9
                    )
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) = 9
                        && DAY(?constructionDate) < 1
                    )
                )
            }
        """
    ] .

gen:TRF024ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction-stage date is required to determine applicability of the twenty-year rule."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The delivery date is required to determine the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessment date is required to verify the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:hasValue nltl:traficom2017Section3Point2Point2Method ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall provide the compliance method for the requirements in section 3.2.2."
    ] ;
    sh:sparql [
        sh:message "For an applicable ship of ice class IA Super or IA, section 3.2.2 compliance must be established no later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                $this nltl:constructionStageDate ?constructionDate .
                $this nltl:deliveryDate ?deliveryDate .
                $this nltl:assessmentDate ?assessmentDate .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    DATATYPE(?constructionDate) = xsd:date
                    && DATATYPE(?deliveryDate) = xsd:date
                    && DATATYPE(?assessmentDate) = xsd:date
                )
                FILTER (
                    YEAR(?constructionDate) < 2003
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) < 9
                    )
                    ||
                    (
                        YEAR(?constructionDate) = 2003
                        && MONTH(?constructionDate) = 9
                        && DAY(?constructionDate) < 1
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        && MONTH(?assessmentDate) > 1
                    )
                    ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                        && MONTH(?assessmentDate) = 1
                        && DAY(?assessmentDate) > 1
                    )
                )
            }
        """
    ] .
<END_SHACL>
