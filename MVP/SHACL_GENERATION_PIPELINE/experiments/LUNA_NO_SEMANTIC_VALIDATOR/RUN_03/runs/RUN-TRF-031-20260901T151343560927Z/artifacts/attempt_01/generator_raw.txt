                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint/qudt:numericValue ?yieldPoint ;
                      nltl:combinedBendingShearStress/qudt:numericValue ?combined ;
                      nltl:yieldShearStress/qudt:numericValue ?yieldShear .
                BIND ((?yieldPoint / 1.7320508075688772) AS ?expected)
                BIND (
                    IF(
                        (ABS(?expected) * 0.000001) > 0.000000001,
                        (ABS(?expected) * 0.000001),
                        0.000000001
                    ) AS ?tolerance
                )
                FILTER (?combined >= ?yieldPoint || ABS(?yieldShear - ?expected) > ?tolerance)
            }
        """
