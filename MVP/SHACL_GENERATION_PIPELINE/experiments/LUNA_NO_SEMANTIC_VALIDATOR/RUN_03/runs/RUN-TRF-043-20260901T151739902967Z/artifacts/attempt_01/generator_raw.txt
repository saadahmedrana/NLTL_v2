<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:class nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:TRF043ExtensionQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each framing ice-strengthening termination case must record its extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:node gen:TRF043AdjacentBoundaryShape ;
        sh:message "Each framing ice-strengthening termination case must identify a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
                ?quantity qudt:numericValue ?extension ;
                         qudt:unit qudt:MilliM .
                FILTER (?extension <= 250)
                FILTER NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true
                }
            }
        """
    ] .

gen:TRF043ExtensionQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF043AdjacentBoundaryShape
    a sh:NodeShape ;
    sh:or (
        gen:TRF043DeckStructureShape
        gen:TRF043TankBoundaryPlatingShape
        gen:TRF043TankTopShape
    ) .

gen:TRF043DeckStructureShape
    a sh:NodeShape ;
    sh:class nltl:deckStructure .

gen:TRF043TankBoundaryPlatingShape
    a sh:NodeShape ;
    sh:class nltl:tankBoundaryPlating .

gen:TRF043TankTopShape
    a sh:NodeShape ;
    sh:class nltl:tankTop .
<END_SHACL>
