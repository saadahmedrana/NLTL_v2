    sh:sparql [
        sh:message "frameBoundaryConditionFactorM0 must match the selected Table 4-7 frame boundary condition" ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionType ?condition ;
                      nltl:frameBoundaryConditionFactorM0 ?m0Node .
                ?m0Node qudt:numericValue ?m0 .
                FILTER (
                    (?condition = nltl:bulkCarrierTopWingTankFrameCondition &&
                     (?m0 < 7 || ?m0 > 7)) ||
                    (?condition = nltl:singleDeckTankTopToMainDeckFrameCondition &&
                     (?m0 < 6 || ?m0 > 6)) ||
                    (?condition = nltl:multiDeckOrStringerContinuousFrameCondition &&
                     (?m0 < 5.7 || ?m0 > 5.7)) ||
                    (?condition = nltl:twoDeckFrameCondition &&
                     (?m0 < 5 || ?m0 > 5))

            }
        """

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this

                      nltl:span ?lNode ;
                      nltl:frameBoundaryConditionFactorM0 ?m0Node ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:frameMomentFactorMt ?mtNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?sNode qudt:numericValue ?s .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?m0Node qudt:numericValue ?m0 .
                ?sigmaNode qudt:numericValue ?sigma .
                ?mtNode qudt:numericValue ?mt .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                FILTER (?l != 0)
                BIND ((7 * ?m0) / (7 - (5 * ?h / ?l)) AS ?expectedMt)
                BIND ((?p * ?s * ?h * ?l / (?expectedMt * ?sigma)) * 1000000 AS ?expectedZ)
                BIND ((1.7320508075688772 * 1.2 * ?p * ?h * ?s / (2 * ?sigma)) * 10000 AS ?expectedA)

                FILTER (
                    ABS(?mt - ?expectedMt) >
                    IF(ABS(?expectedMt) * 0.000001 > 0.000001,
                       ABS(?expectedMt) * 0.000001,
                       0.000001)
                    ||
                    ABS(?z - ?expectedZ) >
                    IF(ABS(?expectedZ) * 0.000001 > 0.000001,
                       ABS(?expectedZ) * 0.000001,
                       0.000001)
                    ||
                    ABS(?a - ?expectedA) >
                    IF(ABS(?expectedA) * 0.000001 > 0.000001,
                       ABS(?expectedA) * 0.000001,
                       0.000001)
                )
            }
        """
