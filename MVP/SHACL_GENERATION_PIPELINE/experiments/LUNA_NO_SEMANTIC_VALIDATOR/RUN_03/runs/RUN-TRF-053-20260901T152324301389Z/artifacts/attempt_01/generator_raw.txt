<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:DecimalQuantityValue a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Quantity value must contain exactly one decimal numeric value."
    ] .

gen:LengthQuantityValue a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Length quantity must use metres."
    ] .

gen:ForcePerLengthQuantityValue a sh:NodeShape ;
    sh:property [

        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design line load quantity must use kilonewtons per metre."
    ] .

    ] .

gen:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice pressure is required for the stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design ice load height is required for the stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stringer span is required for the stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Frame moment factor m is required for the stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Yield strength is required for the stringer calculation."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:ForcePerLengthQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Calculated design line load is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Calculated section modulus is required."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:AreaQuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Calculated required shear area is required."
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, or required shear area does not satisfy the declared formula; calculated decimal results use a relative tolerance of 1e-6 with an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:designLineLoad ?dlNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?dlNode qudt:numericValue ?dl .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?dlExpected)
                BIND((0.9 * 1.8 * ?dlExpected * ?l * ?l / (?m * ?sigma) * 1000000) AS ?zExpected)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?dlExpected * ?l / (2 * ?sigma) * 10000) AS ?aExpected)

                BIND(IF(ABS(?dlExpected) * 0.000001 > 0.000000001, ABS(?dlExpected) * 0.000001, 0.000000001) AS ?dlTolerance)
                BIND(IF(ABS(?zExpected) * 0.000001 > 0.000000001, ABS(?zExpected) * 0.000001, 0.000000001) AS ?zTolerance)
                BIND(IF(ABS(?aExpected) * 0.000001 > 0.000000001, ABS(?aExpected) * 0.000001, 0.000000001) AS ?aTolerance)

                FILTER(
                    (?dl - ?dlExpected > ?dlTolerance) ||
                    (?dlExpected - ?dl > ?dlTolerance) ||
                    (?z - ?zExpected > ?zTolerance) ||
                    (?zExpected - ?z > ?zTolerance) ||
                    (?a - ?aExpected > ?aTolerance) ||
                    (?aExpected - ?a > ?aTolerance)
                )
            }
        """
    ] .
<END_SHACL>
