<urn:nltl:generated-shape:TRF-055-StandardScantling>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:actualSectionModulus ;
        sh:node <urn:nltl:generated-shape:TRF-055-SectionModulusQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:node <urn:nltl:generated-shape:TRF-055-SectionModulusQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualShearArea ;
        sh:node <urn:nltl:generated-shape:TRF-055-ShearAreaQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node <urn:nltl:generated-shape:TRF-055-ShearAreaQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Actual section modulus and actual shear area must meet their respective required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualZ .
                $this nltl:requiredSectionModulus ?requiredZ .
                $this nltl:actualShearArea ?actualA .
                $this nltl:requiredShearArea ?requiredA .
                ?actualZ qudt:numericValue ?actualZValue .
                ?requiredZ qudt:numericValue ?requiredZValue .
                ?actualA qudt:numericValue ?actualAValue .
                ?requiredA qudt:numericValue ?requiredAValue .
                FILTER (?actualZValue < ?requiredZValue || ?actualAValue < ?requiredAValue)
            }
        """
    ] .
