<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:WeatherdeckHatchShape a sh:NodeShape ;
    sh:class nltl:weatherdeckHatch ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSideDeflection ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasHatchCoverDesignEvidence ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:hatchCoverDesignEvidence
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHatchFittingDesignEvidence ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:hatchFittingDesignEvidence
        ]
    ] .

gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node gen:WeatherdeckHatchShape
    ] ;
    sh:sparql [
        sh:message "Each linked weatherdeck hatch requires hatch opening length and the ship requires breadth to establish the applicability condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:hatchOpeningLength ?opening .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:shipBreadth ?breadth .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a weatherdeck hatch longer than half the ship breadth, ship-side deflection and both hatch design evidence records shall be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                ?hatch nltl:hatchOpeningLength ?opening .
                ?opening qudt:numericValue ?openingLength .
                $this nltl:shipBreadth ?breadth .
                ?breadth qudt:numericValue ?shipBreadthValue .
                FILTER (?openingLength > (?shipBreadthValue / 2))
                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflection .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                    }
                )
            }
        """
    ] .
<END_SHACL>
