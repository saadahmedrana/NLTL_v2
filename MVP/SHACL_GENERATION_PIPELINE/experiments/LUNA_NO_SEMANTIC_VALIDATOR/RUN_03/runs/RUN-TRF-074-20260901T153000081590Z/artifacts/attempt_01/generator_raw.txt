<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF074_ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must identify its ice class so the conditional propeller-submersion rule can be evaluated."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IB and IC, ballast propeller depth and required submersion depth hi must be present as decimal metre quantities, with no conflicting quantity unit or datatype." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER (
                    NOT EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        ?depthNode qudt:numericValue ?depthValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?depthValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        ?hiNode qudt:numericValue ?hiValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?hiValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?badDepthNode .
                        FILTER NOT EXISTS {
                            ?badDepthNode qudt:numericValue ?badDepthValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?badDepthValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:requiredPropellerSubmersionDepthHi ?badHiNode .
                        FILTER NOT EXISTS {
                            ?badHiNode qudt:numericValue ?badHiValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?badHiValue) = xsd:decimal)
                        }
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ballast-condition propeller top depth is less than required submersion depth hi for ice class IB or IC, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                ?depthNode qudt:numericValue ?depthValue ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                ?hiNode qudt:numericValue ?hiValue ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (?depthValue < ?hiValue)
                FILTER NOT EXISTS {
                    $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa
                }
            }
        """
    ] .
</urn:nltl:generated-shape:TRF074_ShipShape>
<END_SHACL>
