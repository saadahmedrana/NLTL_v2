<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF078ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TRF078LookupCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide applicable Table 6-6 and Table 6-7 lookup cases."
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:node gen:TRF078UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:node gen:TRF078UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:node gen:TRF078UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide exactly one integer N_ice load-cycle count."
    ] ;
    sh:sparql [
        sh:message "The ship must have one Table 6-6 lookup case and one Table 6-7 lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?table6Case .
                        ?table6Case nltl:tableReference nltl:traficomTable6Dash6 .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasTableLookupCase ?table7Case .
                        ?table7Case nltl:tableReference nltl:traficomTable6Dash7 .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "N_ice must equal k1 multiplied by k2, k3, N_class, and n_n." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?nnNode ;
                      nltl:iceLoadCycleCount ?reported ;
                      nltl:hasTableLookupCase ?table6Case ;
                      nltl:hasTableLookupCase ?table7Case .

                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?nnNode qudt:numericValue ?nn .
                ?table6Case nltl:tableReference nltl:traficomTable6Dash6 ;
                            nltl:selectedIceClassCycleCount ?nclassNode .
                ?table7Case nltl:tableReference nltl:traficomTable6Dash7 ;
                            nltl:selectedPropellerLocationFactor ?k1Node .
                ?nclassNode qudt:numericValue ?nclass .
                ?k1Node qudt:numericValue ?k1 .

                BIND ((?k1 * ?k2 * ?k3 * ?nclass * ?nn) AS ?expected)
                FILTER (
                    ABS(xsd:decimal(?reported) - ?expected)
                    >
                    (ABS(?expected) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .

gen:TRF078LookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:or (
        gen:TRF078Table6LookupCaseShape
        gen:TRF078Table7LookupCaseShape
    ) .

gen:TRF078Table6LookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash6 ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedIceClassCycleCount ;
        sh:node gen:TRF078UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078Table7LookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash7 ;
        sh:class nltl:tableReferenceValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:tableLookupPropellerLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:propellerLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedPropellerLocationFactor ;
        sh:node gen:TRF078UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF078UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
