g:TRF081ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "Ice thrust and propeller shaft design thrust results must satisfy the declared formulas within a relative tolerance of 1e-6, with an absolute tolerance of 1e-6 kN near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce/qudt:numericValue ?ff ;
                      nltl:backwardBladeForce/qudt:numericValue ?fb ;
                      nltl:hydrodynamicBollardThrust/qudt:numericValue ?t ;
                      nltl:forwardIceThrust/qudt:numericValue ?tf ;
                      nltl:backwardIceThrust/qudt:numericValue ?tb ;
                      nltl:forwardPropellerShaftDesignThrust/qudt:numericValue ?fr ;
                      nltl:backwardPropellerShaftDesignThrust/qudt:numericValue ?br ;
                      nltl:propellerShaftDesignThrust/qudt:numericValue ?dr .
                BIND ("1.1"^^xsd:decimal * ?ff AS ?tfExpected)
                BIND ("1.1"^^xsd:decimal * ?fb AS ?tbExpected)
                BIND (?t + ("2.2"^^xsd:decimal * ?tfExpected) AS ?forwardExpected)
                BIND ("1.5"^^xsd:decimal * ?tbExpected AS ?backwardExpected)
                BIND (IF(?forwardExpected >= ?backwardExpected, ?forwardExpected, ?backwardExpected) AS ?designExpected)
                FILTER (
                    ABS(?tf - ?tfExpected) >
                        IF(ABS(?tfExpected) > "1"^^xsd:decimal,
                           ABS(?tfExpected) * "0.000001"^^xsd:decimal,
                           "0.000001"^^xsd:decimal)
                    ||
                    ABS(?tb - ?tbExpected) >
                        IF(ABS(?tbExpected) > "1"^^xsd:decimal,
                           ABS(?tbExpected) * "0.000001"^^xsd:decimal,
                           "0.000001"^^xsd:decimal)
                    ||
                    ABS(?fr - ?forwardExpected) >
                        IF(ABS(?forwardExpected) > "1"^^xsd:decimal,
                           ABS(?forwardExpected) * "0.000001"^^xsd:decimal,
                           "0.000001"^^xsd:decimal)
                    ||
                    ABS(?br - ?backwardExpected) >
                        IF(ABS(?backwardExpected) > "1"^^xsd:decimal,
                           ABS(?backwardExpected) * "0.000001"^^xsd:decimal,
                           "0.000001"^^xsd:decimal)
                    ||
                    ABS(?dr - ?designExpected) >
                        IF(ABS(?designExpected) > "1"^^xsd:decimal,
                           ABS(?designExpected) * "0.000001"^^xsd:decimal,
                           "0.000001"^^xsd:decimal)
                )
            }
        """
    ] .
