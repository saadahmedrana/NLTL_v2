@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF086Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one propeller type to determine applicability of the CP-propeller rule."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerType "CP"^^xsd:string .
                    }
                """ ;
                sh:message "The non-CP applicability branch cannot be used for a CP propeller."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller with known pitch, pitchValueKnown must be true."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "A CP propeller with known pitch must provide its MCR bollard-condition pitch in metres."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller with unknown pitch, pitchValueKnown must be false."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "A CP propeller with unknown pitch must provide its MCR free-running pitch in metres."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "A CP propeller with unknown pitch must provide its MCR bollard-condition pitch in metres."
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerType "CP"^^xsd:string ;
                              nltl:pitchValueKnown false ;
                              nltl:propellerPitchAtMcrFreeRunning ?free ;
                              nltl:propellerPitchAtMcrBollard ?bollard .
                        ?free qudt:numericValue ?freeValue ;
                              qudt:unit unit:M .
                        ?bollard qudt:numericValue ?bollardValue ;
                                 qudt:unit unit:M .
                        BIND((0.7 * ?freeValue) AS ?expected)
                        FILTER(
                            ABS(?bollardValue - ?expected) >
                            ((ABS(?expected) * 0.000001) + 0.000000001)
                        )
                    }
                """ ;
                sh:message "For an unknown CP-propeller pitch, the MCR bollard pitch must equal 0.7 times the MCR free-running pitch within a relative tolerance of 1e-6 plus a small absolute fallback."
            ]
        ]
    ) .
