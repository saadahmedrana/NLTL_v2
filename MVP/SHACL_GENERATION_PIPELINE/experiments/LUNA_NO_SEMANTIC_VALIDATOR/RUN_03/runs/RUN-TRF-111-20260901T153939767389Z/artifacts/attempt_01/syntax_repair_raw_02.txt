<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one extreme ice force in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:extremeOperationalLoadCase
                    nltl:fatigueLoadCase
                    nltl:bladeFailureLoadCase
                )
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The load-case component yield safety factor does not meet the applicable minimum." ;
                sh:select '''
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shaftLineLoadCaseType ?caseType .
                        FILTER (
                            ?caseType = nltl:extremeOperationalLoadCase ||
                            ?caseType = nltl:fatigueLoadCase ||
                            ?caseType = nltl:bladeFailureLoadCase
                        )
                        FILTER NOT EXISTS {
                            $this nltl:componentYieldSafetyFactor ?factor .
                            ?factor qudt:numericValue ?value ;
                                    qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                            FILTER (
                                datatype(?value) = xsd:decimal &&
                                (
                                    (?caseType = nltl:extremeOperationalLoadCase && ?value >= 1.3) ||
                                    (?caseType = nltl:fatigueLoadCase && ?value >= 1.5) ||
                                    (?caseType = nltl:bladeFailureLoadCase && ?value >= 1.0)
                                )
                            )
                        }
                    }
                '''
            ]
        ] ;
        sh:message "The ship must provide an applicable propulsion load case."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:message "The ship must identify the applicable propeller shaft line components."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a fatigue load case applies, each ship component must provide a fatigue safety factor of at least 1.5, dimensionless." ;
        sh:select '''
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasComponent ?component .
                FILTER (
                    NOT EXISTS {
                        ?component nltl:fatigueSafetyFactor ?factor .
                        ?factor qudt:numericValue ?value ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value >= 1.5
                        )
                    } ||
                    EXISTS {
                        ?component nltl:fatigueSafetyFactor ?invalidFactor .
                        FILTER NOT EXISTS {
                            ?invalidFactor qudt:numericValue ?invalidValue ;
                                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                            FILTER (datatype(?invalidValue) = xsd:decimal)
                        }
                    }
                )
            }
        '''
    ] .
<END_SHACL>
