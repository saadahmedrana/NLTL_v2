<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF120ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented thruster natural-frequency calculation case must declare a longitudinal or transverse vibration direction, and the ship and thruster body must provide valid natural-frequency, water-added-mass, and ship-attachment-stiffness quantities." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?body ?case
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .

                FILTER (
                    (
                        NOT EXISTS {
                            ?case nltl:vibrationDirection nltl:longitudinalDirection
                        }
                        &&
                        NOT EXISTS {
                            ?case nltl:vibrationDirection nltl:transverseDirection
                        }
                    )
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency .
                        ?frequency a qudt:QuantityValue .
                        ?frequency qudt:numericValue ?frequencyValue .
                        ?frequency qudt:unit unit:HZ .
                        FILTER(datatype(?frequencyValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:waterAddedMass ?addedMass .
                        ?addedMass a qudt:QuantityValue .
                        ?addedMass qudt:numericValue ?addedMassValue .
                        ?addedMass qudt:unit unit:KiloGM .
                        FILTER(datatype(?addedMassValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?body nltl:shipAttachmentStiffness ?stiffness .
                        ?stiffness a qudt:QuantityValue .
                        ?stiffness qudt:numericValue ?stiffnessValue .
                        ?stiffness qudt:unit unit:KiloN-PER-M .
                        FILTER(datatype(?stiffnessValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
</END_SHACL>
