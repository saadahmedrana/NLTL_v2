<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A design ice force calculation method must be recorded."
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:polarClassPc6
            ]
            [
                sh:hasValue nltl:polarClassPc7
            ]
        ) ;
        sh:message "The ship must be assigned Polar Class PC6 or PC7."
    ] ;
    sh:or (
        gen:I2_010_VerticalSidedBowFormBranch
        gen:I2_010_BowVerticalSideStatusBranch
        gen:I2_010_NormalFrameAngleBranch
    ) ;
    sh:message "For Polar Class PC6 or PC7, the ship must have a vertical-sided bow or qualifying normal frame angles." .

gen:I2_010_VerticalSidedBowFormBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:I2_010_BowVerticalSideStatusBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:I2_010_NormalFrameAngleBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0"^^xsd:decimal ;
                sh:maxInclusive "10"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
