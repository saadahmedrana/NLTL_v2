<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_019_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:hullAreaValue ;
                    sh:property [
                        sh:path nltl:hullAreaFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "selectedHullAreaFactor must equal the maximum hullAreaFactor among the hull areas spanned by the structural member, within a relative tolerance of 1e-6 and an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:selectedHullAreaFactor ?selectedNode .
                ?selectedNode qudt:numericValue ?selectedValue .

                {
                    SELECT ?member (MAX(?areaValue) AS ?maximumValue)
                    WHERE {
                        ?member nltl:hasSpannedHullArea ?areaNode .
                        ?areaNode nltl:hullAreaFactor ?factorNode .
                        ?factorNode qudt:numericValue ?areaValue .
                    }
                    GROUP BY ?member
                }

                FILTER (
                    ABS(?selectedValue - ?maximumValue) >
                    (0.000001 * ABS(?maximumValue) + 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
