<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .

@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_021_ShellPlateThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Thickness must equal ice-load net plate thickness plus corrosion-abrasion allowance, converted from millimetres to metres, within a relative tolerance of 1e-6 or an absolute tolerance of 1e-9 metres." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceLoadRequiredNetPlateThickness ?netNode ;
                              nltl:corrosionAbrasionAllowance ?allowanceNode ;
                              nltl:thickness ?thicknessNode .
                        ?netNode qudt:numericValue ?net .
                        ?allowanceNode qudt:numericValue ?allowance .
                        ?thicknessNode qudt:numericValue ?thickness .
                        BIND (((?net + ?allowance) * 0.001) AS ?expected)
                        BIND (IF ((ABS(?expected) * 0.000001) > 0.000000001,
                                  ABS(?expected) * 0.000001,
                                  0.000000001) AS ?tolerance)
                        FILTER (
                            (?thickness - ?expected > ?tolerance) ||
                            (?expected - ?thickness > ?tolerance)
                        )
                    }
                """
            ]
        ]
    ] .
