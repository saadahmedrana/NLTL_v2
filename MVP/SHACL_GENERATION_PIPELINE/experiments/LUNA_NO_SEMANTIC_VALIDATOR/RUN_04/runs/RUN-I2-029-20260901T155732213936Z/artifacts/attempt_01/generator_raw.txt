<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_029_CalculationCaseShape
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must represent attached shell plating as included in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The calculation case must represent attached shell plating as excluded from the shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must explicitly represent whether flange material is included in the shear area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:flangeMaterialIncludedInShearArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:flangeMaterialIncludedInShearArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:sectionCalculationCaseStructuralMember ;
                sh:minCount 1 ;
                sh:node gen:I2_029_FlangeFittedMemberShape
            ]
        ]
    ) .

gen:I2_029_FlangeFittedMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A structural member whose flange material is included in the shear area must have a fitted flange."
    ] .
<END_SHACL>
