<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-031-QuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-DemandQuantityShape>
    a sh:NodeShape ;
    sh:node <urn:nltl:generated-shape:I2-031-QuantityValueShape> ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .

<urn:nltl:generated-shape:I2-031-StrengthQuantityShape>
    a sh:NodeShape ;
    sh:node <urn:nltl:generated-shape:I2-031-QuantityValueShape> ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .

<urn:nltl:generated-shape:I2-031-LoadCaseShape>
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node <urn:nltl:generated-shape:I2-031-DemandQuantityShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:I2-031-BottomLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame
    ] .

<urn:nltl:generated-shape:I2-031-BottomMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:bottomStructureLocation
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-031-BottomLoadCaseShape>
    ] .

<urn:nltl:generated-shape:I2-031-SideMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation
    ] .

<urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:in (
            nltl:bottomStructureLocation
            nltl:sideStructureLocation
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node <urn:nltl:generated-shape:I2-031-LoadCaseShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2-031-StrengthQuantityShape> ;
        sh:minCount 1
    ] ;
    sh:or (
        <urn:nltl:generated-shape:I2-031-BottomMemberShape>
        <urn:nltl:generated-shape:I2-031-SideMemberShape>
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .
                FILTER (?demandValue > ?strengthValue)
            }
        """
    ] .

<urn:nltl:generated-shape:I2-031-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;

    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node <urn:nltl:generated-shape:I2-031-DemandQuantityShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] .
<END_SHACL>
