@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048-RequirementCaseShape>
    ] .

<urn:nltl:generated-shape:I2-048-RequirementCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify exactly one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must specify one supported steel material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must specify one supported steel strength category."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade requirement case must specify one as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
            nltl:normalStrengthHullStructuralSteel
            nltl:highStrengthHullStructuralSteel
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented steel-grade requirement case must identify exactly one actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
            nltl:normalStrengthHullStructuralSteel
            nltl:highStrengthHullStructuralSteel
        ) ;
        sh:maxCount 1 ;
        sh:message "An applicable table selection must identify one required hull structural steel grade."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required grade must be the Table 8 selection for the ship Polar Class, thickness, material class and strength category, and the actual grade must be at least that grade within the same strength category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  ?ship nltl:hasSteelGradeRequirementCase $this .
  ?ship nltl:polarClass ?pc .
  $this nltl:asBuiltPlateThickness ?thicknessNode ;
        nltl:steelMaterialClass ?material ;
        nltl:steelStrengthCategory ?category .

  ?thicknessNode qudt:numericValue ?thickness .
  ?thicknessNode qudt:unit <http://qudt.org/vocab/unit/MilliM> .

  BIND(
    IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3),
       "PC1-3",
       IF(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5),
          "PC4-5",
          IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
             "PC6-7",
             "PC1-5"
          )
       )
    ) AS ?polarGroup
  )

  BIND(
    IF(
      ?thickness <= 10,
      IF(?material = nltl:steelMaterialClassThree,
         IF(?polarGroup = "PC6-7", 1, 3),
         1
      ),
      IF(
        ?thickness <= 15,
        IF(?material = nltl:steelMaterialClassOne,
           1,
           IF(?material = nltl:steelMaterialClassTwo,
              IF(?polarGroup = "PC6-7", 1, 2),
              IF(?polarGroup = "PC6-7", 2, 3)
           )
        ),
        IF(
          ?thickness <= 25,
          IF(?material = nltl:steelMaterialClassOne,
             IF(?polarGroup = "PC6-7", 1, 2),
             IF(?material = nltl:steelMaterialClassTwo,
                IF(?polarGroup = "PC6-7", 1, 2),
                IF(?polarGroup = "PC6-7", 2, 3)
             )
          ),
          IF(
            ?thickness <= 35,
            IF(?material = nltl:steelMaterialClassOne,
               IF(?polarGroup = "PC6-7", 1, 2),
               IF(?polarGroup = "PC1-5", 3, 2)
            ),
            IF(
              ?thickness <= 45,
              IF(?material = nltl:steelMaterialClassThree,
                 IF(?polarGroup = "PC6-7", 3, 0),
                 IF(?polarGroup = "PC6-7", 2, 3)
              ),
              IF(
                ?thickness <= 50,
                IF(?material = nltl:steelMaterialClassThree,
                   IF(?polarGroup = "PC6-7", 3, 0),
                   IF(?polarGroup = "PC6-7", 2, 3)
                ),
                0
              )
            )
          )
        )
      )
    ) AS ?requiredRank
  )

  BIND(
    (?requiredRank > 0) AS ?applicable
  )

  FILTER(?applicable)

  BIND(
    IF(
      ?category = nltl:normalStrengthSteelCategory,
      IF(?requiredRank = 1, nltl:steelGradeB,
         IF(?requiredRank = 2, nltl:steelGradeD, nltl:steelGradeE)),
      IF(?requiredRank = 1, nltl:steelGradeAh,
         IF(?requiredRank = 2, nltl:steelGradeDh,
            IF(?requiredRank = 3, nltl:steelGradeEh, nltl:steelGradeFh)))
    ) AS ?expectedGrade
  )

  FILTER NOT EXISTS {
    $this nltl:requiredHullStructuralSteelGrade ?requiredGrade ;
          nltl:actualHullStructuralSteelGrade ?actualGrade .
    ?requiredGrade nltl:steelGradeOrderRank ?requiredRankValue .
    ?actualGrade nltl:steelGradeOrderRank ?actualRankValue .
    FILTER(?requiredGrade = ?expectedGrade)
    FILTER(?actualRankValue >= ?requiredRankValue)
    FILTER(
      (?category = nltl:normalStrengthSteelCategory &&
       ?actualGrade IN (
         nltl:steelGradeB,
         nltl:steelGradeD,
         nltl:steelGradeE,
         nltl:normalStrengthHullStructuralSteel
       ))
      ||
      (?category = nltl:highTensileSteelCategory &&
       ?actualGrade IN (
         nltl:steelGradeAh,
         nltl:steelGradeDh,
         nltl:steelGradeEh,
         nltl:steelGradeFh,
         nltl:highStrengthHullStructuralSteel
       ))
    )
  }
}
"""
    ] .
