<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node gen:I2_061_CalculationCaseShape ;
        sh:message "Each applicable calculation case must identify its assessed hull structure and prescribed analytical procedure."
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:node gen:I2_061_HullStructureShape
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:minCount 1 ;
        sh:message "Direct calculation must not replace the prescribed analytical procedure."
    ] .

gen:I2_061_HullStructureShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:plating
        ]
        [
            sh:class nltl:structuralMember
        ]
    ) ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
