<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-065-calculation-case> a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-065-nonlinear-readiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each nonlinear calculation case requires the declared margins, deformation and structural-dimension inputs, an externally computed satisfied assessment result, and nonlinear-analysis acceptance evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER (
                    NOT EXISTS {
                        $this nltl:fractureMargin ?fracture .
                        ?fracture qudt:numericValue ?fractureValue ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?fractureValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:majorBucklingMargin ?buckling .
                        ?buckling qudt:numericValue ?bucklingValue ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?bucklingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldingMargin ?yielding .
                        ?yielding qudt:numericValue ?yieldingValue ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?yieldingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentLateralDeformation ?lateral .
                        ?lateral qudt:numericValue ?lateralValue ;
                                 qudt:unit qudt:MilliM .
                        FILTER (datatype(?lateralValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                        ?outOfPlane qudt:numericValue ?outOfPlaneValue ;
                                    qudt:unit qudt:MilliM .
                        FILTER (datatype(?outOfPlaneValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:relevantStructuralDimension ?dimension .
                        ?dimension qudt:numericValue ?dimensionValue ;
                                   qudt:unit qudt:MilliM .
                        FILTER (datatype(?dimensionValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:minorPermanentDeformationAssessmentSatisfied true .
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                        ?evidence a nltl:evidenceArtifact .
                    }
                )
            }
        """
    ] .
<END_SHACL>
