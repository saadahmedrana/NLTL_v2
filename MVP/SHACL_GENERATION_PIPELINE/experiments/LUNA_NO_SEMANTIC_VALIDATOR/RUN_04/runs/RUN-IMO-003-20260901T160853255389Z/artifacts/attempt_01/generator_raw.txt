<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-003-ShipDesignIceConditionCategory>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship shall declare exactly one design ice condition to determine applicability of the Category C rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for open water or for ice conditions less severe than categories A and B shall have ship category C." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>

SELECT $this
WHERE {
    $this nltl:designIceCondition ?designCondition .
    FILTER (
        ?designCondition = nltl:openWaterIceCondition ||
        ?designCondition = nltl:iceConditionLessSevereThanCategoryAAndB
    )
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryC .
    }
}
"""
    ] .
<END_SHACL>
