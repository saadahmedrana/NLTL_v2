<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer ;
        sh:message "Winter growth count must be an integer when provided."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice thickness must be a metre-valued decimal quantity when provided."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship identified as having sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m must be classified as first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin .
                $this nltl:winterGrowthCount ?growth .
                $this nltl:iceThickness ?thicknessQuantity .
                ?thicknessQuantity qudt:numericValue ?thicknessValue .
                ?thicknessQuantity qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (datatype(?growth) = xsd:integer)
                FILTER (?growth <= 1)
                FILTER (datatype(?thicknessValue) = xsd:decimal)
                FILTER (
                    ?thicknessValue >= "0.3"^^xsd:decimal &&
                    ?thicknessValue <= "2.0"^^xsd:decimal
                )

                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
<END_SHACL>
