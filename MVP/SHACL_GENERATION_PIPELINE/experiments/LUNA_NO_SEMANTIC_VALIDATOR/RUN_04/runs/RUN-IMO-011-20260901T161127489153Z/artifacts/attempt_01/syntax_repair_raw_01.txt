    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDailyLowTemperatureObservation ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:dailyLowTemperatureObservation ;
            sh:property [
                sh:path nltl:dailyTemperatureObservationDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Each daily low-temperature observation must have one calendar date."
            ] ;
            sh:property [
                sh:path nltl:dailyLowTemperature ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG_C ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Each daily low-temperature observation must carry one decimal temperature quantity in degrees Celsius."
            ]
        ] ;
        sh:message "The ship must provide dated daily low-temperature observations."
    ] ;
    sh:property [
        sh:path nltl:dataPeriod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:YR ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one data period as a decimal quantity in years."
    ] ;
    sh:property [
        sh:path nltl:meanDailyLowTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one externally calculated mean daily low-temperature result in degrees Celsius."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>

                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:dataPeriod ?period .

                            FILTER (?value >= 10)
                        }
                    }
                """ ;
                sh:message "The data period must be at least 10 years, or the shorter-period approval condition must apply."
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:dataPeriod ?period .
                            ?period qudt:numericValue ?value .
                            ?period qudt:unit unit:YR .
                            FILTER (?value < 10)
                            $this nltl:administrationDatasetApprovalStatus nltl:evidenceStateApproved .
                        }
                    }
                """ ;
                sh:message "When the data period is shorter than 10 years, administration dataset approval must be Approved."
            ]
        ]
    ) .
