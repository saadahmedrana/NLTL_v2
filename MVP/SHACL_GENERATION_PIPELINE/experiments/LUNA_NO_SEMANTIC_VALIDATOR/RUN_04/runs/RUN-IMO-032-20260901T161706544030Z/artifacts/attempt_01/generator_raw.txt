    sh:property [

            nltl:polarShipCategoryA

    sh:property [

        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval must be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;

            "standard acceptable to the Organization"

    sh:property [

        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:IMO-032-CategoryCApprovalRecord> ;

    ] .

<urn:nltl:generated-shape:IMO-032-CategoryCApprovalRecord> a sh:NodeShape ;
    sh:property [

        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval must be by the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
