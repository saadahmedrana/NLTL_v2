<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:node gen:IMO_047_EssentialEngineShape
    ] .

gen:IMO_047_EssentialEngineShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each essential engine must provide combustion-air temperature control."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each essential engine must provide its combustion-air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each essential engine must provide its manufacturer minimum combustion-air temperature."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:node gen:IMO_047_TemperatureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each essential engine must provide its manufacturer maximum combustion-air temperature."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion-air temperature must be within the essential engine manufacturer's specified range." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureQuantity .
                ?temperatureQuantity qudt:numericValue ?temperature ;
                                      qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                $this nltl:manufacturerMinAirTemperature ?minimumQuantity .
                ?minimumQuantity qudt:numericValue ?minimum ;
                                 qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                $this nltl:manufacturerMaxAirTemperature ?maximumQuantity .
                ?maximumQuantity qudt:numericValue ?maximum ;
                                 qudt:unit <http://qudt.org/vocab/unit/DEG_C> .

                FILTER (
                    ?temperature < ?minimum ||
                    ?temperature > ?maximum
                )
            }
        """
    ] .

gen:IMO_047_TemperatureQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
    ] .
<END_SHACL>
