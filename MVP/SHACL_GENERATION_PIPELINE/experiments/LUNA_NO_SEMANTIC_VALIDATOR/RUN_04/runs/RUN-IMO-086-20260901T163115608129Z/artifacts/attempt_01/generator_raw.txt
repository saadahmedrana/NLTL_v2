<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-086-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node <urn:nltl:generated-shape:IMO-086-HeadingMeansShape> ;
        sh:message "The ship shall have at least two non-magnetic means to determine and display heading."
    ] .

<urn:nltl:generated-shape:IMO-086-HeadingMeansShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be connected to the ship's emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means shall be independent from another heading means on the same ship."
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means shall be independent from another distinct non-magnetic heading means linked to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                ?ship nltl:hasNonMagneticHeadingMeans $this .
                FILTER NOT EXISTS {
                    ?ship nltl:hasNonMagneticHeadingMeans ?other .
                    FILTER (?other != $this)
                    $this nltl:independentFromHeadingMeans ?other .
                }
            }
        """
    ] .
<END_SHACL>
