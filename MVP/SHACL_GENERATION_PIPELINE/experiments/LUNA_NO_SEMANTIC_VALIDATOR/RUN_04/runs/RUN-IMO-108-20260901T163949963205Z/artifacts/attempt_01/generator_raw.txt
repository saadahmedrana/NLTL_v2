<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable Category A or B ship, every represented oil fuel tank must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:constructionDate ?constructionDate ;
          nltl:aggregateOilFuelCapacity ?aggregateCapacity ;
          nltl:individualOilFuelTankCapacity ?individualCapacity .

    ?aggregateCapacity qudt:numericValue ?aggregateValue ;
                       qudt:unit unit:M3 .
    ?individualCapacity qudt:numericValue ?individualValue ;
                        qudt:unit unit:M3 .

    FILTER (
        ?category IN (
            nltl:polarShipCategoryA,
            nltl:polarShipCategoryB
        )
    )
    FILTER (datatype(?constructionDate) = xsd:date)
    FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
    FILTER (datatype(?aggregateValue) = xsd:decimal)
    FILTER (?aggregateValue < "600"^^xsd:decimal)
    FILTER (datatype(?individualValue) = xsd:decimal)
    FILTER (?individualValue > "30"^^xsd:decimal)

    FILTER NOT EXISTS {
        $this nltl:oilFuelTankOuterShellSeparation ?separation .
        ?separation qudt:numericValue ?separationValue ;
                    qudt:unit unit:M .
        FILTER (datatype(?separationValue) = xsd:decimal)
        FILTER (?separationValue >= "0.76"^^xsd:decimal)
    }
}
""" ;
    ] .
<END_SHACL>
