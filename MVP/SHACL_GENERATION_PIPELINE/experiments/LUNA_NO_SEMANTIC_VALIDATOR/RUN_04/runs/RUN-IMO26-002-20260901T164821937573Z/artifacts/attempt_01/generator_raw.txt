            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipType ?type .
                $this nltl:polarCodeChapter9Dash1Applicable ?actual .

                BIND(EXISTS {
                    $this nltl:shipType "fishing_vessel" .
                    $this nltl:lengthOverall ?lengthNode .
                    ?lengthNode q:numericValue ?lengthValue .
                    FILTER(?lengthValue >= 24)
                } AS ?fishingApplicable)

                BIND(EXISTS {
                    $this nltl:shipType "pleasure_yacht" .
                    $this nltl:grossTonnage ?tonnageNode .
                    ?tonnageNode q:numericValue ?tonnageValue .
                    $this nltl:engagedInTrade false .
                    FILTER(?tonnageValue >= 300)
                } AS ?yachtApplicable)

                BIND(EXISTS {
                    $this nltl:shipType "cargo_ship" .
                    $this nltl:grossTonnage ?cargoTonnageNode .
                    ?cargoTonnageNode q:numericValue ?cargoTonnageValue .
                    FILTER(?cargoTonnageValue >= 300 && ?cargoTonnageValue < 500)
                } AS ?cargoApplicable)

                BIND((?fishingApplicable || ?yachtApplicable || ?cargoApplicable) AS ?expected)

                FILTER(
                    (?actual = true && ?expected = false) ||
                    (?actual = false && ?expected = true)
                )
            }
        """
