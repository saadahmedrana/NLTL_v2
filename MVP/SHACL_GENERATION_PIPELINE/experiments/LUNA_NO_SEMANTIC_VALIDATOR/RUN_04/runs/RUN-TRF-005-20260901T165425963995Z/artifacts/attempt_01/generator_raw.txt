<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF005Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction stage date is required to determine applicability."
            ] ;
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction contract date is required to determine applicability."
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:iceClassRegulationEdition2002 ;
                sh:minCount 1 ;
                sh:message "Ships meeting the applicability dates must identify the 2002 Ice Class Regulations."
            ] ;
            sh:sparql [
                sh:message "The 2002 Ice Class Regulations apply when construction began on or after 1 September 2003 and the ship was contracted before 1 January 2010." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate .
                        $this nltl:constructionContractDate ?contractDate .
                        FILTER (
                            (
                                YEAR(?stageDate) > 2003 ||
                                (YEAR(?stageDate) = 2003 && MONTH(?stageDate) > 9) ||
                                (YEAR(?stageDate) = 2003 && MONTH(?stageDate) = 9 && DAY(?stageDate) >= 1)
                            )
                            &&
                            (
                                YEAR(?contractDate) < 2010
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction stage date is required to determine applicability."
            ] ;
            sh:property [
                sh:path nltl:constructionContractDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "A construction contract date is required to determine applicability."
            ] ;
            sh:sparql [
                sh:message "The 2002 Ice Class Regulations are required when construction began on or after 1 September 2003 and the ship was contracted before 1 January 2010." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate .
                        $this nltl:constructionContractDate ?contractDate .
                        FILTER (
                            !(
                                (
                                    YEAR(?stageDate) > 2003 ||
                                    (YEAR(?stageDate) = 2003 && MONTH(?stageDate) > 9) ||
                                    (YEAR(?stageDate) = 2003 && MONTH(?stageDate) = 9 && DAY(?stageDate) >= 1)
                                )
                                &&
                                YEAR(?contractDate) < 2010
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
