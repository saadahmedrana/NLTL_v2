<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

g:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For an IA or IA Super ship, the construction stage date must be on or after 1 September 2003 for this branch." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?construction .
                        FILTER (?construction < "2003-09-01"^^xsd:date)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype <http://www.w3.org/2001/XMLSchema#date> ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For an IA or IA Super ship constructed before 1 September 2003, the 2017 Ice Class Regulations method must be complied with by 1 January of the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?construction .
                        $this nltl:deliveryDate ?delivery .
                        $this nltl:assessmentDate ?assessment .
                        FILTER (?construction < "2003-09-01"^^xsd:date)
                        FILTER (
                            YEAR(?assessment) > YEAR(?delivery) + 20 ||
                            (
                                YEAR(?assessment) = YEAR(?delivery) + 20 &&
                                (
                                    MONTH(?assessment) > 1 ||
                                    (
                                        MONTH(?assessment) = 1 &&
                                        DAY(?assessment) > 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        )
    ;
    sh:message "The ship must satisfy the applicable TRF-007 ice-class and construction-date branch." .
</BEGIN_SHACL>
