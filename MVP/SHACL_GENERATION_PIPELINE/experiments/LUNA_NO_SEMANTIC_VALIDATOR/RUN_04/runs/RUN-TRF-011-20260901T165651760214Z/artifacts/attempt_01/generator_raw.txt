<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF011ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011IceWaterlineShape ;
        sh:message "The ship shall identify exactly one intended ice-operating waterline."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011IceWaterlineShape ;
        sh:message "The ship shall identify exactly one upper ice waterline."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline shall have, at each longitudinal position, the same maximum vertical coordinate as the intended ice-operating waterlines." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT ?this
            WHERE {
                ?this nltl:hasIntendedIceOperatingWaterline ?intended .
                ?this nltl:hasUpperIceWaterline ?uiwl .

                {
                    SELECT ?this ?intended ?uiwl ?x
                           (MAX(?uiValue) AS ?uiMax)
                           (MAX(?intValue) AS ?intMax)
                           (COUNT(?uiValue) AS ?uiCount)
                           (COUNT(?intValue) AS ?intCount)
                    WHERE {
                        {
                            ?uiwl nltl:hasWaterlineProfilePoint ?uiPoint .
                            ?uiPoint nltl:longitudinalPosition ?uiXQuantity ;
                                     nltl:verticalCoordinate ?uiYQuantity .
                            ?uiXQuantity qudt:numericValue ?x .
                            ?uiYQuantity qudt:numericValue ?uiValue .
                        }
                        UNION
                        {
                            ?intended nltl:hasWaterlineProfilePoint ?intPoint .
                            ?intPoint nltl:longitudinalPosition ?intXQuantity ;
                                     nltl:verticalCoordinate ?intYQuantity .
                            ?intXQuantity qudt:numericValue ?x .
                            ?intYQuantity qudt:numericValue ?intValue .
                        }
                    }
                    GROUP BY ?this ?intended ?uiwl ?x
                }

                FILTER (
                    ?uiCount = 0 ||
                    ?intCount = 0 ||
                    ?uiMax != ?intMax
                )
            }
        """
    ] .

g:TRF011IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node g:TRF011WaterlineProfilePointShape ;
        sh:message "Each waterline used by this requirement shall provide waterline profile points."
    ] .

g:TRF011WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape ;
        sh:message "Each waterline profile point shall provide one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:TRF011LengthQuantityShape ;
        sh:message "Each waterline profile point shall provide one vertical coordinate in metres."
    ] .

g:TRF011LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
