<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_016_Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:node <urn:nltl:generated-shape:TRF_016_UIWLShape> ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:displacementAtUpperIceWaterline ;
                sh:node <urn:nltl:generated-shape:TRF_016_MassQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:levelIceThickness ;
                sh:node <urn:nltl:generated-shape:TRF_016_LengthQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:draughtAmidships ;
                sh:node <urn:nltl:generated-shape:TRF_016_LengthQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:liwlDraughtAmidships ;
                sh:node <urn:nltl:generated-shape:TRF_016_LengthQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:forwardDraught ;
                sh:node <urn:nltl:generated-shape:TRF_016_LengthQuantityShape> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerHighestPointSubmerged ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasBallastTank ;
                sh:node <urn:nltl:generated-shape:TRF_016_BallastTankShape>
            ] ;
            sh:sparql [
                sh:message "Draught amidships must be at least the LIWL draught amidships, and forward draught must satisfy the level-ice-thickness and UIWL-displacement formula." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:draughtAmidships ?draughtQ ;
                              nltl:liwlDraughtAmidships ?liwlQ ;
                              nltl:forwardDraught ?forwardQ ;
                              nltl:displacementAtUpperIceWaterline ?displacementQ ;
                              nltl:levelIceThickness ?thicknessQ .
                        ?draughtQ qudt:numericValue ?draught .
                        ?liwlQ qudt:numericValue ?liwl .
                        ?forwardQ qudt:numericValue ?forward .
                        ?displacementQ qudt:numericValue ?displacement .
                        ?thicknessQ qudt:numericValue ?thickness .
                        FILTER (
                            ?draught < ?liwl ||
                            ?forward < IF(
                                ((2 + (0.00025 * ?displacement)) * ?thickness)
                                < (4 * ?thickness),
                                ((2 + (0.00025 * ?displacement)) * ?thickness),
                                (4 * ?thickness)
                            )
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The displacement at the UIWL must correspond to a UIWL waterline having the greatest displacement." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:displacementAtUpperIceWaterline ?selectedQ ;
                              nltl:hasUpperIceWaterline ?waterline .
                        ?selectedQ qudt:numericValue ?selected .
                        ?waterline nltl:waterlineDisplacement ?waterlineQ .
                        ?waterlineQ qudt:numericValue ?waterlineDisplacement .
                        FILTER NOT EXISTS {
                            $this nltl:hasUpperIceWaterline ?greaterWaterline .
                            ?greaterWaterline nltl:waterlineDisplacement ?greaterQ .
                            ?greaterQ qudt:numericValue ?greater .
                            FILTER (?greater > ?selected)
                        }
                        FILTER NOT EXISTS {
                            $this nltl:hasUpperIceWaterline ?matchingWaterline .
                            ?matchingWaterline nltl:waterlineDisplacement ?matchingQ .
                            ?matchingQ qudt:numericValue ?matching .
                            FILTER (?matching = ?selected)
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .

<urn:nltl:generated-shape:TRF_016_UIWLShape>
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:waterlineDisplacement ;
        sh:node <urn:nltl:generated-shape:TRF_016_MassQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF_016_MassQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric>
    ] .

<urn:nltl:generated-shape:TRF_016_LengthQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

<urn:nltl:generated-shape:TRF_016_BallastTankShape>
    a sh:NodeShape ;
    sh:class nltl:ballastTank ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:situatedAboveLowerIceWaterline ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:usedToReachLowerIceWaterline ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:freezingPreventionPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:situatedAboveLowerIceWaterline ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:usedToReachLowerIceWaterline ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
