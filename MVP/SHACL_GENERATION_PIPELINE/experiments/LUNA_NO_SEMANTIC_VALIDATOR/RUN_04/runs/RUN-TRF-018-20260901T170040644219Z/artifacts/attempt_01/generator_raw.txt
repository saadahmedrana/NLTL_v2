<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-018-QuantityValueShape>
    a sh:NodeShape ;
    sh:targetClass qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-018-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must specify its ice class for the applicable engine-output rule."
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:node <urn:nltl:generated-shape:TRF-018-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Calculated required power must be supplied as a single kW quantity."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node <urn:nltl:generated-shape:TRF-018-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum continuous rating power must be supplied as a single kW quantity."
    ] ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximum continuous rating power must be at least the calculated required power and 1000 kW; for IA Super, it must be at least the calculated required power and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?ratingNode .
                ?calculatedNode qudt:numericValue ?calculatedPower .
                ?ratingNode qudt:numericValue ?ratingPower .
                FILTER (
                    (
                        ?iceClass IN (nltl:iceClassIa, nltl:iceClassIb, nltl:iceClassIc)
                        && (
                            ?ratingPower < ?calculatedPower
                            || ?ratingPower < 1000
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper
                        && (
                            ?ratingPower < ?calculatedPower
                            || ?ratingPower < 2800
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
