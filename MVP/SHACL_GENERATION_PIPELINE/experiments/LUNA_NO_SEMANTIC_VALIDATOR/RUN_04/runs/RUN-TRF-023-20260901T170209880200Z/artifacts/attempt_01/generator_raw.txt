<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF023Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRuleEdition1985
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredPowerUnder1985Rules ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:hasValue nltl:iceClassRuleEdition1985
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredPowerUnder1985Rules ;
                sh:nodeKind sh:IRI ;
                sh:class qudt:QuantityValue ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:in (
                    nltl:iceClassRegulationEdition2002
                    nltl:iceClassRegulationEdition2008
                    nltl:iceClassRegulationEdition2010
                    nltl:iceClassRegulationEdition2017
                    nltl:iceClassRegulationEdition2021
                )
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC under the 1985 ice class regulations, maximum continuous rating power must be at least the required power under the 1985 rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition nltl:iceClassRuleEdition1985 ;
                      nltl:maximumContinuousRatingPower ?mcrNode ;
                      nltl:requiredPowerUnder1985Rules ?requiredNode .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )

                ?mcrNode qudt:numericValue ?mcr .
                ?requiredNode qudt:numericValue ?required .

                FILTER (?mcr < ?required)
            }
        """
    ] .
<END_SHACL>
