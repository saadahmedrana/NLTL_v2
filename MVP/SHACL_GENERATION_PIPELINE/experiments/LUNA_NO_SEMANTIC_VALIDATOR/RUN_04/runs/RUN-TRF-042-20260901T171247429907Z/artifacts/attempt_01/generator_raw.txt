<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF042PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:datatype q:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A design ice load height in metres is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:datatype q:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive 0
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A positive spacing in metres is required."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:datatype q:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path q:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path q:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The longitudinal plating factor f2 is required as a dimensionless quantity."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:datatype q:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path q:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ] ;
                    sh:property [
                        sh:path q:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:corrosionAbrasionAddition ;
                        sh:datatype q:QuantityValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path q:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:maxExclusive 2
                            ] ;
                            sh:property [
                                sh:path q:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:specialSurfaceCoatingMaintained ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:materialApprovalStatus ;
                        sh:class nltl:evidenceState ;
                        sh:hasValue nltl:evidenceStateApproved ;
                        sh:minCount 1
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable piecewise formula using designIceLoadHeight divided by spacing, within max(1e-6*ABS(expected),1e-12)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightQ ;
                      nltl:spacing ?spacingQ ;
                      nltl:longitudinalPlatingFactorF2 ?factorQ .
                ?heightQ q:numericValue ?height ;
                          q:unit <http://qudt.org/vocab/unit/M> .
                ?spacingQ q:numericValue ?spacing ;
                           q:unit <http://qudt.org/vocab/unit/M> .
                ?factorQ q:numericValue ?observed ;
                         q:unit <http://qudt.org/vocab/unit/UNITLESS> .
                BIND (?height / ?spacing AS ?ratio)
                BIND (
                    IF(
                        ?ratio <= 1,
                        0.4 / ?ratio + 0.6,
                        1.4 - 0.4 * ?ratio
                    )
                    AS ?expected
                )
                FILTER (
                    ?ratio > 1.8 ||
                    ABS(?observed - ?expected) >
                    IF(
                        1e-6 * ABS(?expected) > 1e-12,
                        1e-6 * ABS(?expected),
                        1e-12
                    )
                )
            }
        """
    ] .
<END_SHACL>
