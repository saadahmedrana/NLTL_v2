<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF043_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node nltl:TRF043_TerminationCaseShape ;
        sh:message "Each framing ice-strengthening termination case must satisfy the applicable termination constraint."
    ] .

nltl:TRF043_TerminationCaseShape
    a sh:NodeShape ;
    sh:class nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:class nltl:deckStructure
            ]
            [
                sh:class nltl:tankBoundaryPlating
            ]
            [
                sh:class nltl:tankTop
            ]
        ] ;
        sh:message "The termination adjacent boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node nltl:TRF043_ExtensionWithin250mmShape
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm."
            ]
        ]
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node nltl:TRF043_ExtensionAbove250mmShape
            ]
        ]
    ) ;
    sh:message "The extension value is required to determine whether termination at the adjacent boundary is permitted." .

nltl:TRF043_ExtensionWithin250mmShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive 250
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

nltl:TRF043_ExtensionAbove250mmShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 250
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
