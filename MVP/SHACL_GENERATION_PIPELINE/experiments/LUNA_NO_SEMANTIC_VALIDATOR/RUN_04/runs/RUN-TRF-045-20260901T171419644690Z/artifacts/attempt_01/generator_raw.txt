<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-045_FrameSpanOrdinaryScantlingsShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameSpanWithinIceStrengtheningZone ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ordinary frame scantlings are used, the frame span within the ice-strengthening zone must be less than 15% of the frame span." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true .
                FILTER NOT EXISTS {
                    $this nltl:frameSpanWithinIceStrengtheningZone ?withinNode .
                    ?withinNode qudt:numericValue ?withinValue ;
                                qudt:unit unit:M .
                    FILTER (datatype(?withinValue) = xsd:decimal)
                    $this nltl:span ?spanNode .
                    ?spanNode qudt:numericValue ?spanValue ;
                             qudt:unit unit:M .
                    FILTER (datatype(?spanValue) = xsd:decimal)
                    FILTER (?withinValue < (0.15 * ?spanValue))
                }
            }
        """
    ] .
</END_SHACL>
