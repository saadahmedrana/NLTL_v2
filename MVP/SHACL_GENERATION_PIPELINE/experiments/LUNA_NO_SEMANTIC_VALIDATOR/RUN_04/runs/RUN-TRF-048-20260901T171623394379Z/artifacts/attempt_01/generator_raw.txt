        sh:message "For a continuous beam with brackets without significantly different boundary conditions, frame moment factor m must equal 13.3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionType nltl:continuousBeamWithBracketsFrameCondition ;
                      nltl:significantlyDifferentBoundaryConditions ?different ;
                      nltl:frameMomentFactorM ?mNode .
                ?mNode qudt:numericValue ?m .
                FILTER (
                    ?different = false &&
                    (?m < 13.3 || ?m > 13.3)
                )
            }
        """
    ] ;

        sh:message "The reported f4, section modulus, and required shear area must satisfy the declared formulas within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:longitudinalFrameLoadDistributionFactorF4 ?f4Node ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?f4Node qudt:numericValue ?f4Reported .
                ?zNode qudt:numericValue ?zReported .
                ?aNode qudt:numericValue ?aReported .

                BIND ((1 - (0.2 * ?h / ?s)) AS ?f4Expected)
                BIND (((?f4Expected * ?p * ?h * ?l * ?l) / (4 * ?m * ?sigma)) * 1000000 AS ?zExpected)
                BIND ((1.7320508075688772 * ?f4Expected * 2.16 * ?p * ?h * ?l / (2 * ?sigma)) * 10000 AS ?aExpected)

                FILTER (
                    ABS(?f4Reported - ?f4Expected) >
                        (0.000001 * ABS(?f4Expected) + 0.000000001)
                    ||
                    ABS(?zReported - ?zExpected) >
                        (0.000001 * ABS(?zExpected) + 0.000000001)
                    ||
                    ABS(?aReported - ?aExpected) >
                        (0.000001 * ABS(?aExpected) + 0.000000001)
                )
            }
        """
