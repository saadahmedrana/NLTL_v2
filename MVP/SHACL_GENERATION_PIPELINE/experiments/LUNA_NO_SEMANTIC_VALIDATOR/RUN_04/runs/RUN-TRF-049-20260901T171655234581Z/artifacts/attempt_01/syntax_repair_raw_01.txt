<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF049_FrameShape a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:effectiveAttachmentConfirmed ;
                sh:datatype xsd:boolean ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:frameWebThickness ;
                sh:datatype qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node nltl:TRF049_FrameWebThicknessShape
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node nltl:TRF049_FrameAttachmentShape
            ] ;
            sh:property [
                sh:path nltl:hasSupportingWebFrame ;
                sh:node nltl:TRF049_SupportingStructureShape
            ] ;
            sh:property [
                sh:path nltl:hasSupportingBulkhead ;
                sh:node nltl:TRF049_SupportingStructureShape
            ] ;
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:webPlateConnectionSideCount ;
                        sh:datatype xsd:integer ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue false
                    ]
                ]
            ) ;
            sh:sparql [
                sh:message "Longitudinal frames must be attached by brackets to every linked supporting web frame and bulkhead; terminating transverse frames require a bracket or similar construction." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    SELECT $this
                    WHERE {
                        FILTER (
                            (
                                EXISTS {
                                    $this a nltl:longitudinalFrame .
                                    FILTER NOT EXISTS {
                                        {
                                            $this nltl:hasSupportingWebFrame ?supporting .
                                            FILTER NOT EXISTS {
                                                ?supporting nltl:hasSupportingStructureAttachment ?attachment .
                                                ?attachment nltl:hasConnectionBracket ?bracket .
                                            }
                                        }
                                        UNION
                                        {
                                            $this nltl:hasSupportingBulkhead ?supporting .
                                            FILTER NOT EXISTS {
                                                ?supporting nltl:hasSupportingStructureAttachment ?attachment .
                                                ?attachment nltl:hasConnectionBracket ?bracket .
                                            }
                                        }
                                    }
                                }
                            )
                            ||
                            (
                                EXISTS {
                                    $this a nltl:transverseFrame .
                                    $this nltl:terminatesAtDeckOrIceStringer true .
                                }
                                &&
                                NOT EXISTS {
                                    $this nltl:hasFrameAttachment ?attachment .
                                    FILTER (
                                        EXISTS {
                                            ?attachment nltl:hasConnectionBracket ?bracket .
                                        }
                                        ||
                                        EXISTS {
                                            ?attachment nltl:similarAttachmentConstructionPresent true .
                                        }
                                    )
                                }
                            )
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Each installed connection bracket must be at least as thick as the associated frame web plate." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    SELECT $this
                    WHERE {
                        $this nltl:frameWebThickness ?frameThickness .
                        ?frameThickness qudt:numericValue ?frameValue .
                        ?frameThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                        $this (
                            nltl:hasFrameAttachment |
                            nltl:hasSupportingWebFrame/nltl:hasSupportingStructureAttachment |
                            nltl:hasSupportingBulkhead/nltl:hasSupportingStructureAttachment
                        ) ?attachment .
                        ?attachment nltl:hasConnectionBracket ?bracket .
                        ?bracket nltl:bracketThickness ?bracketThickness .
                        ?bracketThickness qudt:numericValue ?bracketValue .
                        ?bracketThickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                        FILTER (?bracketValue < ?frameValue)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .

    ] .
<END_SHACL>
