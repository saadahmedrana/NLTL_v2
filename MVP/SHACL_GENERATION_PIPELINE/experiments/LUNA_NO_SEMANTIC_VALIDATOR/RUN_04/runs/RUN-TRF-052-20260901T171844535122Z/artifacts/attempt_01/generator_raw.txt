<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-052_FrameSupport> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAsymmetrical ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A frame asymmetry value is required to determine tripping-support applicability."
    ] ;
    sh:property [
        sh:path nltl:frameWebAngleToShell ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The frame web angle to the shell must be recorded as a single degree quantity."
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The frame span must be recorded as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The frame ice class must be one of the permitted canonical ice-class values."
    ] ;
    sh:property [
        sh:path nltl:equivalentSupportStatusByDirectCalculation ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:antitrippingSupportSpacing ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Antitripping support spacing must be recorded as a single millimetre quantity."
    ] ;
    sh:property [
        sh:path nltl:hasAntitrippingSupportRegion ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
            nltl:otherIceStrengthenedAreaValue
        )
    ] ;
    sh:sparql [
        sh:message "When the frame is asymmetrical or its web angle to the shell is below 90 degrees, spacing must not exceed 1300 mm unless an equivalent alternative arrangement is demonstrated." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameAsymmetrical ?asym ;
                      nltl:frameWebAngleToShell ?angleQ .
                ?angleQ qudt:numericValue ?angle ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (?asym = true || xsd:decimal(?angle) < 90)
                FILTER NOT EXISTS {
                    $this nltl:equivalentSupportStatusByDirectCalculation true
                }
                FILTER NOT EXISTS {
                    $this nltl:antitrippingSupportSpacing ?spacingQ .
                    ?spacingQ qudt:numericValue ?spacing ;
                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (xsd:decimal(?spacing) <= 1300)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For spans greater than 4 m, or for ice class IA Super with a span of 4 m or less, antitripping support must cover the bow, midbody, stern, and other ice-strengthened regions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpan ?spanQ ;
                      nltl:iceClass ?ice .
                ?spanQ qudt:numericValue ?span ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    xsd:decimal(?span) > 4 ||
                    (xsd:decimal(?span) <= 4 && ?ice = nltl:iceClassIaSuper)
                )
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                }
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                }
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:sternRegion
                }
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:otherIceStrengthenedAreaValue
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For ice class IA and a span of 4 m or less, antitripping support must cover both the bow and midbody regions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpan ?spanQ ;
                      nltl:iceClass nltl:iceClassIa .
                ?spanQ qudt:numericValue ?span ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (xsd:decimal(?span) <= 4)
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                }
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:midbodyRegion
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For ice class IB or IC and a span of 4 m or less, antitripping support must cover the bow region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpan ?spanQ ;
                      nltl:iceClass ?ice .
                ?spanQ qudt:numericValue ?span ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (xsd:decimal(?span) <= 4)
                FILTER (?ice = nltl:iceClassIb || ?ice = nltl:iceClassIc)
                FILTER NOT EXISTS {
                    $this nltl:hasAntitrippingSupportRegion nltl:bowRegion
                }
            }
        """
    ] .
<END_SHACL>
