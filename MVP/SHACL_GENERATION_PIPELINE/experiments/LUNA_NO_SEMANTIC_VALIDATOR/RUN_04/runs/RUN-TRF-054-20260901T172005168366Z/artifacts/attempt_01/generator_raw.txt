gs:TRF_054_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gs:PressureMPaShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gs:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gs:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gs:DimensionlessShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gs:PressureMPaShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gs:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gs:LengthMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gs:LineLoadKNPerMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gs:DimensionlessShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gs:SectionModulusCm3Shape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gs:AreaCm2Shape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Design line load, outside ice belt factor, section modulus, and required shear area must satisfy the declared calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?lineNode qudt:numericValue ?line .
                ?factorNode qudt:numericValue ?factor .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(?p * ?h AS ?product)
                BIND(IF(?product < 0.15, 0.15, ?product) AS ?lineMN)
                BIND(1 - (?hs / ?ls) AS ?expectedFactor)
                BIND(0.80 * 1.8 * ?lineMN * ?l * ?l /
                     (?m * ?sigma) * ?expectedFactor * 1000000 AS ?expectedZ)
                BIND(1.7320508075688772 * 0.80 * 1.8 * 1.2 * ?lineMN * ?l /
                     (2 * ?sigma) * ?expectedFactor * 10000 AS ?expectedA)

                FILTER(
                    ABS(?line - (?lineMN * 1000)) >
                        IF(ABS(?lineMN * 1000) * 0.000001 > 0.000000001,
                           ABS(?lineMN * 1000) * 0.000001,
                           0.000000001)
                    ||
                    ABS(?factor - ?expectedFactor) >
                        IF(ABS(?expectedFactor) * 0.000001 > 0.000000001,
                           ABS(?expectedFactor) * 0.000001,
                           0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                        IF(ABS(?expectedZ) * 0.000001 > 0.000000001,
                           ABS(?expectedZ) * 0.000001,
                           0.000000001)
                    ||
                    ABS(?a - ?expectedA) >
                        IF(ABS(?expectedA) * 0.000001 > 0.000000001,
                           ABS(?expectedA) * 0.000001,
                           0.000000001)
                )
            }
        """
    ] .
