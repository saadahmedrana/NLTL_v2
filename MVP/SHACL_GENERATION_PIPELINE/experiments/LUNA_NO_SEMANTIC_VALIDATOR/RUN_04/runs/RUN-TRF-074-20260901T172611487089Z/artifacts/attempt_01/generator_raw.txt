<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_074_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice class to determine applicability of the propeller-submersion rule."
    ] ;
    sh:property [
        sh:path nltl:ballastConditionPropellerTopDepth ;
        sh:node <urn:nltl:generated-shape:TRF_074_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:message "The ship must provide the ballast-condition propeller top depth."
    ] ;
    sh:property [
        sh:path nltl:requiredPropellerSubmersionDepthHi ;
        sh:node <urn:nltl:generated-shape:TRF_074_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:message "The ship must provide the required propeller submersion depth hi."
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemDesignedForIceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IB or IC, when the ballast-condition propeller top depth is less than hi, the propulsion system must be designed for ice class IA." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (
        nltl:iceClassIb,
        nltl:iceClassIc
    ))

    $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
    ?depthQuantity qudt:numericValue ?depth .
    FILTER (datatype(?depth) = xsd:decimal)

    $this nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .
    ?requiredQuantity qudt:numericValue ?requiredDepth .
    FILTER (datatype(?requiredDepth) = xsd:decimal)

    FILTER (?depth < ?requiredDepth)

    FILTER NOT EXISTS {
        $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa
    }
}
"""
    ] .

<urn:nltl:generated-shape:TRF_074_LengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
