<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF079ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design ice thickness is required in metres." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice load cycle coefficient k2 is required as a dimensionless quantity." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Propeller diameter is required in metres." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerImmersionFunction ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The propeller immersion function f is required as a dimensionless quantity." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerShaftCentrelineDepth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Propeller shaft centreline depth is required in metres." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The immersion function f and coefficient k2 do not satisfy the specified formulas; calculated decimal results use a relative tolerance of 1e-6 with an absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?hice ;
                      nltl:iceLoadCycleCoefficientK2 ?k2q ;
                      nltl:propellerDiameter ?dq ;
                      nltl:propellerImmersionFunction ?fq ;
                      nltl:propellerShaftCentrelineDepth ?h0q .
                ?hice qudt:numericValue ?hicev .
                ?k2q qudt:numericValue ?k2 .
                ?dq qudt:numericValue ?d .
                ?fq qudt:numericValue ?f .
                ?h0q qudt:numericValue ?h0 .

                BIND (((?h0 - ?hicev) / (?d / 2)) - 1 AS ?fExpected)

                FILTER (
                    ABS(?f - ?fExpected) > (0.000001 * (1 + ABS(?fExpected)))
                    ||
                    (
                        ?f < 0 &&
                        ABS(?k2 - (0.8 - ?f)) >
                            (0.000001 * (1 + ABS(0.8 - ?f)))
                    )
                    ||
                    (
                        ?f >= 0 &&
                        ?f <= 1 &&
                        ABS(?k2 - (0.8 - (0.4 * ?f))) >
                            (0.000001 * (1 + ABS(0.8 - (0.4 * ?f))))
                    )
                    ||
                    (
                        ?f > 1 &&
                        ?f <= 2.5 &&
                        ABS(?k2 - (0.6 - (0.2 * ?f))) >
                            (0.000001 * (1 + ABS(0.6 - (0.2 * ?f))))
                    )
                    ||
                    (
                        ?f > 2.5 &&
                        ABS(?k2 - 0.1) >
                            (0.000001 * (1 + ABS(0.1)))
                    )
                )
            }
        """
    ] .
