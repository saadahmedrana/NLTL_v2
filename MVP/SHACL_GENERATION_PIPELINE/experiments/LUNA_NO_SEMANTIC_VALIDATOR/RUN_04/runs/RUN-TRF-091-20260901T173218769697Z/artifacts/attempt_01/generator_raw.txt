@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF091ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:message "The ship shall identify an ice milling sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:loadCasePurposeValue ;
                sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
                sh:minCount 1 ;
                sh:message "The propulsion load case shall be used for propulsion line strength evaluation."
            ] ;
            sh:not [
                sh:property [
                    sh:path nltl:loadCasePurpose ;
                    sh:hasValue nltl:stallingAnalysisPurposeValue
                ]
            ]
        ] ;
        sh:message "The ship shall identify a propulsion load case with the required load-case purpose."
    ] .
