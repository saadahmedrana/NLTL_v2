<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-104-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:pyramidStrengthPrinciple ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall record that its propulsion line is designed according to the pyramid strength principle."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When propeller blade loss is represented, it shall not cause significant damage to any linked propeller shaft line component." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerBladeLoss true ;
                      nltl:hasPropellerShaftLineComponent ?component .
                FILTER (
                    EXISTS {
                        ?component nltl:significantDamageFromBladeLoss true .
                    }
                    ||
                    NOT EXISTS {
                        ?component nltl:significantDamageFromBladeLoss false .
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
