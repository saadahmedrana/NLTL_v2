gen:TRF130StandardChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "near_centreline_and_well_aft_if_possible" ;
        sh:message "The sea inlet must be near the centreline and well aft if possible."
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape ;
        sh:message "The recommended chest volume is required in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The chest must provide space for ice accumulation above the inlet pipe."
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A full-capacity cooling-water discharge pipe must be connected to the chest."
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape ;
        sh:message "The strainer open area is required in square metres."
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:RatioQuantityShape
    ] ;
    sh:sparql [
        sh:message "The standard chest volume must approximately equal total relevant engine power divided by 750, and strainer open area must be at least four times inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasComponent $this .
                ?ship nltl:totalRelevantEnginePower ?tp .
                ?tp qudt:numericValue ?t .
                $this nltl:recommendedChestVolume ?rv .
                ?rv qudt:numericValue ?v .
                $this nltl:inletPipeArea ?ip .
                ?ip qudt:numericValue ?i .
                $this nltl:strainerOpenArea ?sa .
                ?sa qudt:numericValue ?s .
                FILTER(
                    ABS(?v - (?t / 750)) <=
                    (ABS(?t / 750) * "0.000001"^^xsd:decimal + "0.000000001"^^xsd:decimal)
                )
                FILTER(?s >= (4 * ?i))
            }
        """
    ] ;
    sh:sparql [
        sh:message "The strainer open area ratio must correspond to strainer open area divided by inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:strainerOpenArea ?sa .
                ?sa qudt:numericValue ?s .
                $this nltl:inletPipeArea ?ip .
                ?ip qudt:numericValue ?i .
                $this nltl:strainerOpenAreaRatio ?sr .
                ?sr qudt:numericValue ?r .
                FILTER(
                    ABS(?r - (?s / ?i)) <=
                    (ABS(?s / ?i) * "0.000001"^^xsd:decimal + "0.000000001"^^xsd:decimal)
                )
            }
        """
    ] .
