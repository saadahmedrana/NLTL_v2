<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_048_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:node nltl:steelGradeRequirementCase ;
        sh:nodeKind sh:IRI
    ] .

nltl:steelGradeRequirementCase
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive 0 ;
                sh:maxInclusive 50
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade case must provide one as-built plate thickness from greater than 0 mm through 50 mm."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeAh
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeDh
            nltl:steelGradeE
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable steel-grade table selection must provide exactly one required hull structural steel grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:tableReference nltl:iacsUrI2Table8 ;
                       nltl:steelMaterialClass ?material ;
                       nltl:steelStrengthCategory ?strength ;
                       nltl:asBuiltPlateThickness ?thickness .
                ?thickness qudt:numericValue ?t .
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                       nltl:polarClass ?polar .
                FILTER (
                    ?strength = nltl:highTensileSteelCategory ||
                    ?strength = nltl:normalStrengthSteelCategory
                )
                FILTER (
                    ?material = nltl:steelMaterialClassOne ||
                    ?material = nltl:steelMaterialClassTwo ||
                    ?material = nltl:steelMaterialClassThree
                )
                FILTER (
                    ?polar IN (
                        nltl:polarClassPc1,
                        nltl:polarClassPc2,
                        nltl:polarClassPc3,
                        nltl:polarClassPc4,
                        nltl:polarClassPc5,
                        nltl:polarClassPc6,
                        nltl:polarClassPc7
                    )
                )
                FILTER (
                    ?t > 0 && ?t <= 50
                )
                FILTER NOT (
                    ?strength = nltl:normalStrengthSteelCategory &&
                    ?material = nltl:steelMaterialClassThree &&
                    (
                        (?t > 35 && ?t <= 45 &&
                         ?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
                                    nltl:polarClassPc4, nltl:polarClassPc5)) ||
                        (?t > 45 && ?t <= 50 &&
                         ?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3,
                                    nltl:polarClassPc4, nltl:polarClassPc5))
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:requiredHullStructuralSteelGrade ?required .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required hull structural steel grade must be the exact Table 8 selection for the case and the actual grade must not be below it within the same steel strength category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:tableReference nltl:iacsUrI2Table8 ;
                       nltl:steelMaterialClass ?material ;
                       nltl:steelStrengthCategory ?strength ;
                       nltl:asBuiltPlateThickness ?thickness ;
                       nltl:actualHullStructuralSteelGrade ?actual ;
                       nltl:requiredHullStructuralSteelGrade ?required .
                ?thickness qudt:numericValue ?t .
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                       nltl:polarClass ?polar .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                BIND((?strength = nltl:highTensileSteelCategory) AS ?high)
                BIND((
                    ?polar IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3)
                ) AS ?pc13)
                BIND((
                    ?polar IN (nltl:polarClassPc4, nltl:polarClassPc5)
                ) AS ?pc45)
                BIND((
                    ?polar IN (nltl:polarClassPc6, nltl:polarClassPc7)
                ) AS ?pc67)
                BIND((
                    ?polar IN (
                        nltl:polarClassPc1,
                        nltl:polarClassPc2,
                        nltl:polarClassPc3,
                        nltl:polarClassPc4,
                        nltl:polarClassPc5
                    )
                ) AS ?pc15)
                BIND((
                    IF(
                        ?t <= 10,
                        IF(
                            ?material = nltl:steelMaterialClassThree && ?pc67,
                            IF(?high, nltl:steelGradeAh, nltl:steelGradeB),
                            IF(?high, nltl:steelGradeAh, nltl:steelGradeB)
                        ),
                        IF(
                            ?t <= 20,
                            IF(
                                ?material = nltl:steelMaterialClassThree,
                                IF(?pc67,
                                   IF(?high, nltl:steelGradeDh, nltl:steelGradeD),
                                   IF(?high, nltl:steelGradeEh, nltl:steelGradeE)),
                                IF(?pc15,
                                   IF(?high, nltl:steelGradeDh, nltl:steelGradeD),
                                   IF(?high, nltl:steelGradeAh, nltl:steelGradeB))
                            ),
                            IF(
                                ?t <= 35,
                                IF(
                                    ?material = nltl:steelMaterialClassTwo,
                                    IF(?pc15,
                                       IF(?high, nltl:steelGradeEh, nltl:steelGradeE),
                                       IF(?high, nltl:steelGradeDh, nltl:steelGradeD)),
                                    IF(
                                        ?material = nltl:steelMaterialClassThree,
                                        IF(?pc67,
                                           IF(?high, nltl:steelGradeDh, nltl:steelGradeD),
                                           IF(?high, nltl:steelGradeEh, nltl:steelGradeE)),
                                        IF(?pc15,
                                           IF(?high, nltl:steelGradeDh, nltl:steelGradeD),
                                           IF(?high, nltl:steelGradeAh, nltl:steelGradeB))
                                    )
                                ),
                                IF(
                                    ?t <= 45,
                                    IF(
                                        ?material = nltl:steelMaterialClassThree,
                                        IF(?pc13,
                                           nltl:steelGradeFh,
                                           IF(?high, nltl:steelGradeEh, nltl:steelGradeE)),
                                        IF(?high, nltl:steelGradeDh, nltl:steelGradeD)
                                    ),
                                    IF(
                                        ?material = nltl:steelMaterialClassThree,
                                        IF(?pc13 || ?pc45,
                                           nltl:steelGradeFh,
                                           IF(?high, nltl:steelGradeEh, nltl:steelGradeE)),
                                        IF(?pc15,
                                           IF(?high, nltl:steelGradeEh, nltl:steelGradeE),
                                           IF(?high, nltl:steelGradeDh, nltl:steelGradeD))
                                    )
                                )
                            )
                        )
                    )
                ) AS ?expected)
                FILTER NOT (
                    ?strength = nltl:normalStrengthSteelCategory &&
                    ?material = nltl:steelMaterialClassThree &&
                    (
                        (?t > 35 && ?t <= 45 && (?pc13 || ?pc45)) ||
                        (?t > 45 && ?t <= 50 && (?pc13 || ?pc45))
                    )
                )
                FILTER (?required != ?expected || ?actualRank < ?requiredRank)
            }
        """
    ] .

nltl:steelGradeValue
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
