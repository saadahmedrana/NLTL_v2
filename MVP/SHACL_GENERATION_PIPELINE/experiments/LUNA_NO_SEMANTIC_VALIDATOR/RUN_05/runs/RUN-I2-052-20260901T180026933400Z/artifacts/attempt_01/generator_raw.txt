<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node gen:CalculationCaseShape ;
        sh:message "A calculation case for the externally determined interpolation shall be provided."
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The externally determined interpolated value shall be recorded."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The hull-girder longitudinal position from aft shall be supplied in metres."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The externally determined interpolated longitudinal distribution factor shall be recorded as a dimensionless quantity."
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case shall identify interpolation points."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case shall identify a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case shall identify an upper interpolation endpoint."
    ] .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Each interpolation point shall provide its interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Each interpolation point shall provide its point value."
    ] .

</END_SHACL>
