@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_055_StaticShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The strength criterion status must be true."
    ] ;
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single design stress quantity in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single permissible stress quantity in MPa is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The design stress must not exceed the permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designStress ?designQuantity .
                ?designQuantity qudt:numericValue ?designStress .
                $this nltl:permissibleStress ?permissibleQuantity .
                ?permissibleQuantity qudt:numericValue ?permissibleStress .
                FILTER (
                    DATATYPE(?designStress) = xsd:decimal &&
                    DATATYPE(?permissibleStress) = xsd:decimal &&
                    ?designStress > ?permissibleStress
                )
            }
        """
    ] .
