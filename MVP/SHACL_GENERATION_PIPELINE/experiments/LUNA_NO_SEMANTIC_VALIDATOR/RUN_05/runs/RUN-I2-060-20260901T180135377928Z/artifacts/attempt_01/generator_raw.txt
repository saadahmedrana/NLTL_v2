<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The ship must record that member loads do not cause instability."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where a member in way of a cutout requires stiffening, the ship must record structural stiffening." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout .
                ?member nltl:stiffeningNecessary ?necessary .
                FILTER (?necessary = true)
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (
                        isLiteral(?stiffening) &&
                        datatype(?stiffening) = xsd:string &&
                        STRLEN(STR(?stiffening)) > 0
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCutout ;
        sh:class nltl:cutout ;
        sh:minCount 1 ;
        sh:message "A structural member governed by this requirement must be linked to a cutout."
    ] ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .

<END_SHACL>
