gen:I2_064_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MegaPA
            ]
        ] ;
        sh:message "The ship shall provide one yield-strength quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:linearCalculationMethodValue ;
                sh:message "Each calculation case shall identify the linear calculation method."
            ]
        ] ;
        sh:message "The ship shall provide a calculation case for the applicable linear calculation."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:hasStructuralMemberWeb ;
                sh:node [
                    sh:class nltl:structuralMemberWeb ;
                    sh:property [
                        sh:path nltl:nominalShearStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:datatype xsd:decimal
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue qudt:MegaPA
                            ]
                        ] ;
                        sh:message "Each represented member web shall provide one nominal shear stress in MPa."
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:message "The represented member web shall document satisfied buckling criteria."
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberFlange ;
                sh:node [
                    sh:class nltl:structuralMemberFlange ;
                    sh:property [
                        sh:path nltl:nominalFlangeVonMisesStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:datatype xsd:decimal
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue qudt:MegaPA
                            ]
                        ] ;
                        sh:message "Each represented member flange shall provide one nominal von Mises stress in MPa."
                    ] ;
                    sh:property [
                        sh:path nltl:bucklingCriteriaSatisfied ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:message "The represented member flange shall document satisfied buckling criteria."
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal shear stress in each represented member web shall be less than yield strength divided by sqrt(3)." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:yieldStrength ?yieldQuantity .
    ?yieldQuantity qudt:numericValue ?yieldValue ;
                   qudt:unit qudt:MegaPA .
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasStructuralMemberWeb ?web .
    ?web nltl:nominalShearStress ?stressQuantity .
    ?stressQuantity qudt:numericValue ?stressValue ;
                    qudt:unit qudt:MegaPA .

}
"""
    ] ;

        a sh:SPARQLConstraint ;

PREFIX nltl: <https://w3id.org/nltl/vocab#>

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

}
"""
    ] .
