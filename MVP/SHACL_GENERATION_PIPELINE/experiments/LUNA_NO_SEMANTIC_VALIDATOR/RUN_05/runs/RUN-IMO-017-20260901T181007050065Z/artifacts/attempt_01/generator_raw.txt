<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_017_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "The ship shall have a Polar Ship Certificate form."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO_017_SolasScheduleShape ;
        sh:message "The ship shall have an applicable SOLAS certificate schedule."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO_017_SupplementShape ;
        sh:message "The Polar Ship Certificate shall include a supplement with required equipment records."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar certificate supplement presence indicator shall be true."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Polar Ship Certificate validity dates shall be recorded as a single date."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Polar Ship Certificate survey dates shall be recorded as a single date."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates shall be recorded as a single date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate validity, survey, and endorsement dates shall match the corresponding SOLAS certificate schedule date records." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:polarCertificateValidityDates ?validityDate ;
                       nltl:surveyDates ?surveyDate ;
                       nltl:endorsementDates ?endorsementDate .

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    FILTER NOT EXISTS {
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    FILTER NOT EXISTS {
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    FILTER NOT EXISTS {
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                nltl:certificateScheduleDate ?date .
                    }
                }

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER (?date != ?validityDate)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER (?date != ?surveyDate)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                            nltl:certificateScheduleDate ?date .
                    FILTER (?date != ?endorsementDate)
                }
            }
        """
    ] .

gen:IMO_017_SolasScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 3 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:message "The SOLAS certificate schedule shall contain the applicable validity, survey, and endorsement date records."
    ] .

gen:IMO_017_SupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar certificate supplement shall record required equipment."
    ] .

gen:IMO_017_DateRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each certificate schedule date record shall contain one xsd:date value."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record shall have a validity, survey, or endorsement category."
    ] .
<END_SHACL>
