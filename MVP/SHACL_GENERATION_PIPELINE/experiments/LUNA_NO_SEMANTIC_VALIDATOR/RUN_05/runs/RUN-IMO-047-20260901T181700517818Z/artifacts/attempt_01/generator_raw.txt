<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO047ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:IMO047NoEssentialEngine
        gs:IMO047ApplicableEngine
    ) .

gs:IMO047NoEssentialEngine
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:maxCount 0 ;
        sh:message "No essential engine is declared, so the combustion-air requirement is not applicable."
    ] .

gs:IMO047ApplicableEngine
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:minCount 1 ;
        sh:class nltl:essentialEngine ;
        sh:message "Each declared essential engine must be represented as an essential engine."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
            ]
        ] ;
        sh:message "Combustion-air temperature must be supplied as a single degree-Celsius quantity."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
            ]
        ] ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion must be supplied as a single degree-Celsius quantity."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
            ]
        ] ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion must be supplied as a single degree-Celsius quantity."
    ] ;
    sh:sparql [
        sh:message "Combustion-air temperature must be within the engine manufacturer's minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasEssentialEngine ?engine .
                $this nltl:combustionAirTemperature ?temperatureNode .
                $this nltl:manufacturerMinAirTemperature ?minimumNode .
                $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                ?temperatureNode qudt:numericValue ?temperature .
                ?minimumNode qudt:numericValue ?minimum .
                ?maximumNode qudt:numericValue ?maximum .
                FILTER (
                    ?temperature < ?minimum ||
                    ?temperature > ?maximum
                )
            }
        """
    ] .

</END_SHACL>
