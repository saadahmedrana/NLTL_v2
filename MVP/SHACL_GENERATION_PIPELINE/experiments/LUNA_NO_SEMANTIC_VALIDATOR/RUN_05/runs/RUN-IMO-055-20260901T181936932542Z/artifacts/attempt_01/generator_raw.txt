<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-055> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed isolating, pressure, or vacuum valve must have ice-accretion protection and remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:valveExposed ?exposed ;
                      nltl:valveType ?valveType .
                FILTER (?exposed = true)
                FILTER (?valveType IN ("isolating", "pressure", "vacuum"))
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceAccretionProtectionPresent true .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:continuouslyAccessible true .
                    }
                )
            }
        """
    ] .
<END_SHACL>
