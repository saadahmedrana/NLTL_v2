<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO057ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every fire pump and specified fire-protection pump must be linked to a compartment maintained above 0 deg C." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump rdf:type ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )
                FILTER NOT EXISTS {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:DEG_C .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value > 0
                    )
                }
            }
        """
    ] .

gen:IMO057PumpShape
    a sh:NodeShape ;
    sh:targetClass nltl:firePump ;
    sh:targetClass nltl:emergencyFirePump ;
    sh:targetClass nltl:waterMistPump ;
    sh:targetClass nltl:waterSprayPump ;
    sh:property [
        sh:path nltl:hasContainingCompartment ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:compartment ;
            sh:property [
                sh:path nltl:maintainedTemperature ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minExclusive "0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG_C ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] .
<END_SHACL>
