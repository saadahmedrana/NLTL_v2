<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_075_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship must indicate whether personal or group survival equipment is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:personsOnBoard ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:availableSurvivalEquipmentCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:storageAccessibilityStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When personal or group survival equipment is required, available survival equipment capacity must be at least the ceiling of 110% of the persons on board." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:personalOrGroupSurvivalEquipmentRequired true ;
                      nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (?capacity < CEIL(?persons * 1.10))
            }
        """
    ] .
<END_SHACL>
