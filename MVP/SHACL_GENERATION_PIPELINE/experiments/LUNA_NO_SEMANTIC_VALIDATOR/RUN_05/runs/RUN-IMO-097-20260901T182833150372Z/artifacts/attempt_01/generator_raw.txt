<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_097_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:hasValue true ]
            [ sh:hasValue false ]
        ] ;
        sh:message "The ship must state whether it operates in low air temperature."
    ] ;
    sh:property [
        sh:path nltl:releasedRescueBoatOrLifeboat ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:hasValue true ]
            [ sh:hasValue false ]
        ] ;
        sh:message "The ship must state whether the rescue boat or lifeboat is released for evacuation."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship operates in low air temperature and the rescue boat or lifeboat is released, every represented craft must carry the required communication devices." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipOperatesInLowAirTemperature true ;
                      nltl:releasedRescueBoatOrLifeboat true .
                FILTER (
                    EXISTS {
                        $this nltl:hasSurvivalCraft ?craft .
                        FILTER NOT EXISTS {
                            ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?distressDevice ;
                                   nltl:survivalCraftHasLocationSignalDevice ?locationDevice ;
                                   nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasRequiredRescueBoatOrLifeboat ?lifeboat .
                        FILTER NOT EXISTS {
                            ?lifeboat nltl:hasDistressAlertDevice ?distressDevice .
                        }
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
