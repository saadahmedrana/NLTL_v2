<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_105_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine whether the Arctic waters prohibition applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:hasValue "Arctic_waters"
            ] ;
            sh:property [
                sh:path nltl:oilOrOilyMixtureDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( false ) ;
                sh:message "In Arctic waters, discharge of oil or oily mixtures into the sea must be false."
            ]
        ]
        [
            sh:not [
                sh:property [
                    sh:path nltl:operatingArea ;
                    sh:hasValue "Arctic_waters"
                ]
            ]
        ]
    ) .
<END_SHACL>
