<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-106> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:class nltl:dischargedMaterialTypeValue ;
        sh:minCount 1 ;
        sh:message "A discharged material classification is required to evaluate the discharge exception."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast, the paragraph 1.1.1 oil-discharge prohibition applicability must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable "false"^^xsd:boolean .
                    }
                    ||
                    EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                        FILTER (?applicable != "false"^^xsd:boolean)
                    }
                )
            }
        """
    ] .
<END_SHACL>
