<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-108> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, every represented oil fuel tank must have at least 0.76 m separation from the outer shell." ;
        sh:select '''
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:aggregateOilFuelCapacity ?aggregateQuantity ;
                      nltl:individualOilFuelTankCapacity ?individualQuantity .

                ?aggregateQuantity qudt:numericValue ?aggregateValue ;
                                    qudt:unit unit:M3 .
                ?individualQuantity qudt:numericValue ?individualValue ;
                                     qudt:unit unit:M3 .

                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (
                    YEAR(?constructionDate) > 2017 ||
                    (
                        YEAR(?constructionDate) = 2017 &&
                        (
                            MONTH(?constructionDate) > 1 ||
                            (
                                MONTH(?constructionDate) = 1 &&
                                DAY(?constructionDate) >= 1
                            )
                        )
                    )
                )
                FILTER (?aggregateValue < 600)
                FILTER (?individualValue > 30)

                FILTER NOT EXISTS {
                    $this nltl:oilFuelTankOuterShellSeparation ?separationQuantity .
                    ?separationQuantity qudt:numericValue ?separationValue ;
                                         qudt:unit unit:M .
                    FILTER (?separationValue >= 0.76)
                }
            }
        '''
    ] .
</urn:nltl:generated-shape:IMO-108>
<END_SHACL>
