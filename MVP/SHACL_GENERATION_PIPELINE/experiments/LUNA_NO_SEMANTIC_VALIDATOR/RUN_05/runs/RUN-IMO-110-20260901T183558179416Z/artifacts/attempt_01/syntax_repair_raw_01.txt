<urn:nltl:generated-shape:IMO_110_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B oil tanker of less than 5,000 tonnes deadweight constructed on or after 1 January 2017, both required cargo-tank protection statuses must be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?constructionDate .

                ?deadweight qudt:numericValue ?deadweightValue .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && STR(?shipType) = "oil_tanker"
                    && ?deadweightValue < 5000
                    && ?constructionDate >= "2017-01-01"^^xsd:date
                )

                FILTER NOT EXISTS {
                    $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true .
                }

                UNION

                {
                    $this nltl:shipCategory ?category2 ;
                          nltl:shipType ?shipType2 ;
                          nltl:deadweight ?deadweight2 ;
                          nltl:constructionDate ?constructionDate2 .

                    ?deadweight2 qudt:numericValue ?deadweightValue2 .

                    FILTER (
                        ?category2 IN (
                            nltl:polarShipCategoryA,
                            nltl:polarShipCategoryB
                        )
                        && STR(?shipType2) = "oil_tanker"
                        && ?deadweightValue2 < 5000
                        && ?constructionDate2 >= "2017-01-01"^^xsd:date
                    )

                    FILTER NOT EXISTS {
                        $this nltl:wingTankProtectionStatus true .
                    }
                }
            }
        """
    ] .
