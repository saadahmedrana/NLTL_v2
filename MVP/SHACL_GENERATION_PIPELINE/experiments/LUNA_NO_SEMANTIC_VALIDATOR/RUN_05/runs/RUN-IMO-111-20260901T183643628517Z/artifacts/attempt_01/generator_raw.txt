<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship category must be supplied as Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single valid construction date is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, each oil residue or oily bilge water holding tank with capacity greater than 30 m3 must be separated from the outer shell by at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasComponent ?component .
                ?component a ?tankClass .
                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER (?tankClass IN (
                    nltl:oilResidueTank,
                    nltl:oilyBilgeWaterHoldingTank
                ))
                FILTER (
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )
                FILTER NOT EXISTS {
                    {
                        $this nltl:individualTankCapacity ?capacityNode .
                        ?capacityNode a qudt:QuantityValue ;
                                      qudt:numericValue ?capacity ;
                                      qudt:unit unit:M .
                        FILTER (datatype(?capacity) = xsd:decimal)
                        FILTER (?capacity <= 30)
                    }
                    UNION
                    {
                        $this nltl:individualTankCapacity ?capacityNode ;
                              nltl:tankOuterShellSeparation ?separationNode .
                        ?capacityNode a qudt:QuantityValue ;
                                      qudt:numericValue ?capacity ;
                                      qudt:unit unit:M .
                        ?separationNode a qudt:QuantityValue ;
                                        qudt:numericValue ?separation ;
                                        qudt:unit unit:M .
                        FILTER (datatype(?capacity) = xsd:decimal)
                        FILTER (datatype(?separation) = xsd:decimal)
                        FILTER (?capacity > 30)
                        FILTER (?separation >= 0.76)
                    }
                }
            }
        """
    ] .
<END_SHACL>
