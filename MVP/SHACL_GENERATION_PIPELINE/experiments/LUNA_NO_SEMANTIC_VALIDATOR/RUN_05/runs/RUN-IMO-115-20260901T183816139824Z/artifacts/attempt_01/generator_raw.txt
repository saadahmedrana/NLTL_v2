<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-115> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must record its sewage discharge type so applicability can be determined."
    ] ;
    sh:property [
        sh:path nltl:comminutedAndDisinfected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship must record whether sewage is comminuted and disinfected."
    ] ;
    sh:sparql [
        sh:message "When sewage is comminuted and disinfected, MARPOL Annex IV regulation 11.1.1 compliance, the distance threshold, and ice-condition distance records must conform." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType .
                $this nltl:comminutedAndDisinfected ?comminuted .
                FILTER(?comminuted = true)
                FILTER(str(?dischargeType) = "comminuted_and_disinfected")

                FILTER(
                    NOT EXISTS {
                        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus true
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToNearestIceShelfOrFastIce ?nearestDistance .
                        ?nearestDistance qudt:numericValue ?nearestValue ;
                            qudt:unit unit:MI_N .
                        FILTER(datatype(?nearestValue) = xsd:decimal)
                        FILTER(?nearestValue > 3)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                        ?areaDistance qudt:numericValue ?areaDistanceValue ;
                            qudt:unit unit:MI_N .
                        FILTER(datatype(?areaDistanceValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceConcentration ?iceValueNode .
                        ?iceValueNode qudt:numericValue ?iceValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER(datatype(?iceValue) = xsd:decimal)
                        FILTER(?iceValue > 0.1)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:nearbyIceConcentration ?nearbyValueNode .
                        ?nearbyValueNode qudt:numericValue ?nearbyValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER(datatype(?nearbyValue) = xsd:decimal)
                        FILTER(?nearbyValue > 0.1)
                    }
                )
            }
        """
    ] .
<END_SHACL>
