<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_118_Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:class nltl:passengerShip
        ]
    ) ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, sewage discharge to sea must be false unless compliance with Polar Code paragraph 4.2.1.3 is recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .
                FILTER (datatype(?date) = xsd:date)
                FILTER (
                    YEAR(?date) > 2017 ||
                    (
                        YEAR(?date) = 2017 &&
                        (
                            MONTH(?date) > 1 ||
                            (MONTH(?date) = 1 && DAY(?date) >= 1)
                        )
                    )
                )
                FILTER (
                    EXISTS {
                        $this nltl:shipCategory ?category .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                        )
                    } ||
                    EXISTS {
                        $this a nltl:passengerShip .
                    }
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ?status .
                    FILTER (
                        datatype(?status) = xsd:boolean &&
                        ?status = true
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:sewageDischargeToSea ?discharge .
                    FILTER (
                        datatype(?discharge) = xsd:boolean &&
                        ?discharge = false
                    )
                }
            }
        """
    ] .
<END_SHACL>
