<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node <urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When at least two non-magnetic heading means are represented, each means shall be connected to main and emergency power and be independent from another distinct means on the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means1 .
                $this nltl:hasNonMagneticHeadingMeans ?means2 .
                FILTER (?means1 != ?means2)

                {
                    FILTER NOT EXISTS {
                        ?means1 nltl:connectedToMainPower ?mainPower .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?means1 nltl:connectedToEmergencyPower ?emergencyPower .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?means1 nltl:independentFromHeadingMeans ?independentMeans .
                        $this nltl:hasNonMagneticHeadingMeans ?independentMeans .
                        FILTER (?independentMeans != ?means1)
                    }
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO26-012-HeadingMeansShape>
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans
    ] .
<END_SHACL>
