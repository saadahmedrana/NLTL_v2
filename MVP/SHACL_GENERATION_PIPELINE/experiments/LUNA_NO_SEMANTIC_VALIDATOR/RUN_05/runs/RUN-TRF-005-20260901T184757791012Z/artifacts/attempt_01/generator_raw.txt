<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF005ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate .
                        $this nltl:constructionContractDate ?contractDate .
                        FILTER (
                            !(?stageDate >= "2003-09-01"^^xsd:date &&
                              ?contractDate < "2010-01-01"^^xsd:date)
                        )
                    }
                """ ;
                sh:message "The applicability dates do not establish the applicable branch for the 2002 Ice Class Regulations."
            ]
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassRegulationEditionValue ;
                sh:minCount 1 ;
                sh:hasValue nltl:iceClassRegulationEdition2002 ;
                sh:message "Ships in the specified applicability period shall identify the 2002 Ice Class Regulations."
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?stageDate .
                        $this nltl:constructionContractDate ?contractDate .
                        FILTER (
                            ?stageDate >= "2003-09-01"^^xsd:date &&
                            ?contractDate < "2010-01-01"^^xsd:date
                        )
                    }
                """ ;
                sh:message "The applicability dates establish that the 2002 Ice Class Regulations apply."
            ]
        ]
    ) ;
    sh:message "The ship does not satisfy either the applicable or non-applicable date branch for the 2002 Ice Class Regulations." .
<END_SHACL>
