<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must have an ice class." ;
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The construction stage date is required to determine applicability." ;
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The delivery date is required to determine the compliance deadline." ;
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The assessment date is required to evaluate the compliance deadline." ;
    ] ;
    sh:sparql [
        sh:message "A ship of ice class IA Super or IA with construction before 1 September 2003 must have an allowed 2017 regulation compliance method and comply by 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:constructionStageDate ?construction ;
                      nltl:deliveryDate ?delivery ;
                      nltl:assessmentDate ?assessment .

                FILTER (
                    ?ice IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    ?construction < "2003-09-01"^^xsd:date
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (
                            ?method IN (
                                nltl:traficom2017Section3Point2Point2Method,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    }
                    ||
                    !(
                        YEAR(?assessment) < YEAR(?delivery) + 20
                        ||
                        (
                            YEAR(?assessment) = YEAR(?delivery) + 20
                            && MONTH(?assessment) = 1
                            && DAY(?assessment) = 1
                        )
                    )
                )
            }
        """ ;
    ] .
<END_SHACL>
