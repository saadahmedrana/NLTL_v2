<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-024-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of TRF-024."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA Super or IA, a valid construction stage date is required to determine whether TRF-024 applies." ;
        sh:select "PREFIX nltl: <https://w3id.org/nltl/vocab#>\nPREFIX xsd: <http://www.w3.org/2001/XMLSchema#>\nSELECT $this\nWHERE {\n    $this nltl:iceClass ?iceClass .\n    FILTER (?iceClass IN (\n        nltl:iceClassIaSuper,\n        nltl:iceClassIa\n    ))\n    FILTER NOT EXISTS {\n        $this nltl:constructionStageDate ?constructionStageDate .\n        FILTER (datatype(?constructionStageDate) = xsd:date)\n    }\n}"
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable IA Super or IA ship, delivery and assessment dates must show compliance by 1 January of the year twenty years after delivery." ;
        sh:select "PREFIX nltl: <https://w3id.org/nltl/vocab#>\nPREFIX xsd: <http://www.w3.org/2001/XMLSchema#>\nSELECT $this\nWHERE {\n    $this nltl:iceClass ?iceClass ;\n          nltl:constructionStageDate ?constructionStageDate .\n    FILTER (?iceClass IN (\n        nltl:iceClassIaSuper,\n        nltl:iceClassIa\n    ))\n    FILTER (datatype(?constructionStageDate) = xsd:date)\n    FILTER (\n        YEAR(?constructionStageDate) < 2003 ||\n        (\n            YEAR(?constructionStageDate) = 2003 &&\n            (\n                MONTH(?constructionStageDate) < 9 ||\n                (\n                    MONTH(?constructionStageDate) = 9 &&\n                    DAY(?constructionStageDate) < 1\n                )\n            )\n        )\n    )\n    FILTER NOT EXISTS {\n        $this nltl:deliveryDate ?deliveryDate ;\n              nltl:assessmentDate ?assessmentDate .\n        FILTER (\n            datatype(?deliveryDate) = xsd:date &&\n            datatype(?assessmentDate) = xsd:date\n        )\n        FILTER (\n            YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||\n            (\n                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&\n                MONTH(?assessmentDate) = 1 &&\n                DAY(?assessmentDate) <= 1\n            )\n        )\n    }\n}"
    ] .
</urn:nltl:generated-shape:TRF-024-ShipShape>
<END_SHACL>
