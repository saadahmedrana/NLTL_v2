    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF048Shape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:TRF048PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:TRF048QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:TRF048QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:TRF048QuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:TRF048DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:TRF048PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node gen:TRF048DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:TRF048SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:TRF048AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a continuous beam with brackets and no significantly different boundary conditions, m must be 13.3; f4, section modulus, and required shear area must satisfy the declared formulas within a relative tolerance of 1e-6 plus an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionType nltl:continuousBeamWithBracketsFrameCondition ;
                      nltl:significantlyDifferentBoundaryConditions ?different ;
                      nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:spacing ?spacingNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:longitudinalFrameLoadDistributionFactorF4 ?f4Node ;
                      nltl:sectionModulus ?sectionNode ;
                      nltl:requiredShearArea ?areaNode .

                ?pressureNode qudt:numericValue ?p .
                ?heightNode qudt:numericValue ?h .
                ?spacingNode qudt:numericValue ?s .
                ?spanNode qudt:numericValue ?l .
                ?momentNode qudt:numericValue ?m .
                ?yieldNode qudt:numericValue ?sigma .
                ?f4Node qudt:numericValue ?f4 .
                ?sectionNode qudt:numericValue ?z .
                ?areaNode qudt:numericValue ?a .

                BIND((1.0 - (0.2 * ?h / ?s)) AS ?expectedF4)
                BIND((?expectedF4 * ?p * ?h * ?l * ?l / (4.0 * ?m * ?sigma) * 1000000.0) AS ?expectedZ)
                BIND((1.7320508075688772 * ?expectedF4 * 2.16 * ?p * ?h * ?l / (2.0 * ?sigma) * 10000.0) AS ?expectedA)

                FILTER (
                    (
                        ?different = false &&
                        (?m < 13.3 || ?m > 13.3)
                    )
                    ||
                    (
                        ?different = true ||
                        ?different != false
                    )
                    ||
                    ABS(?f4 - ?expectedF4) > (0.000001 * ABS(?expectedF4) + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) > (0.000001 * ABS(?expectedZ) + 0.000000001)
                    ||
                    ABS(?a - ?expectedA) > (0.000001 * ABS(?expectedA) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
