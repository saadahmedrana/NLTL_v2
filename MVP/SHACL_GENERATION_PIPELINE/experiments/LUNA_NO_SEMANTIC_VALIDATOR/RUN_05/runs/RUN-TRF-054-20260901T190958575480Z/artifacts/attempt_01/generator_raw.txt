gs:TRF054_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gs:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gs:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gs:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gs:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gs:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The derived design line load, outside ice belt factor, section modulus, and required shear area must satisfy the TRF-054 formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:stringerSpan/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigmaY ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hs ;
                      nltl:stringerSpan/qudt:numericValue ?ls ;
                      nltl:designLineLoad/qudt:numericValue ?load ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?factor ;
                      nltl:sectionModulus/qudt:numericValue ?z ;
                      nltl:requiredShearArea/qudt:numericValue ?a .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?expectedLoad)
                BIND((1.0 - (?hs / ?ls)) AS ?expectedFactor)
                BIND((0.80 * 1.8 * ?expectedLoad * ?l * ?l /
                      (?m * ?sigmaY) * ?expectedFactor * 1000000.0)
                     AS ?expectedZ)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?expectedLoad * ?l /
                      (2.0 * ?sigmaY) * ?expectedFactor * 10000.0)
                     AS ?expectedA)

                FILTER (
                    ?load < (?expectedLoad -
                             (0.000001 * ABS(?expectedLoad) + 0.000000001))
                    || ?load > (?expectedLoad +
                                (0.000001 * ABS(?expectedLoad) + 0.000000001))
                    || ?factor < (?expectedFactor -
                                  (0.000001 * ABS(?expectedFactor) + 0.000000001))
                    || ?factor > (?expectedFactor +
                                  (0.000001 * ABS(?expectedFactor) + 0.000000001))
                    || ?z < (?expectedZ -
                             (0.000001 * ABS(?expectedZ) + 0.000000001))
                    || ?z > (?expectedZ +
                             (0.000001 * ABS(?expectedZ) + 0.000000001))
                    || ?a < (?expectedA -
                             (0.000001 * ABS(?expectedA) + 0.000000001))
                    || ?a > (?expectedA +
                             (0.000001 * ABS(?expectedA) + 0.000000001))
                )
            }
        """
    ] .
