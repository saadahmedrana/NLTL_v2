            SELECT ?this
            WHERE {
                ?this nltl:actualSectionModulus ?actualZ .
                ?actualZ qudt:numericValue ?actualZN .
                ?this nltl:requiredSectionModulus ?requiredZ .
                ?requiredZ qudt:numericValue ?requiredZN .
                ?this nltl:actualShearArea ?actualA .
                ?actualA qudt:numericValue ?actualAN .

        """

    a sh:NodeShape ;
    sh:property [
        sh:path nltl:veryLongHatchOpening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:permittedReducedLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN-PER-M>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:reducedLineLoadApprovedByClassificationSociety ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a very long hatch, the approved reduced line load must be at least 0.10 and less than 0.15 kN/m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:veryLongHatchOpening true .
                ?this nltl:permittedReducedLineLoad ?load .
                ?load qudt:numericValue ?loadN .
                FILTER (?loadN < 0.10 || ?loadN >= 0.15)
            }
        """
    ] .
