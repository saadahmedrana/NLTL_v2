<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A twin- or triple-screw ship shall identify a side propeller strengthening envelope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count .
                FILTER (?count = 2 || ?count = 3)
                FILTER NOT EXISTS {
                    $this nltl:hasSidePropellerStrengtheningEnvelope ?envelope .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each side propeller strengthening envelope shall have a tank-top boundary and forward and aft extents of at least 1.5 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count ;
                      nltl:hasSidePropellerStrengtheningEnvelope ?envelope .
                FILTER (?count = 2 || ?count = 3)
                FILTER NOT EXISTS {
                    ?envelope a nltl:sidePropellerStrengtheningEnvelope ;
                              nltl:strengtheningEnvelopeBoundary ?boundary ;
                              nltl:strengtheningForwardExtent ?forward ;
                              nltl:strengtheningAftExtent ?aft .

                    ?boundary a nltl:tankTop .

                    ?forward qudt:numericValue ?forwardValue ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?aft qudt:numericValue ?aftValue ;
                         qudt:unit <http://qudt.org/vocab/unit/M> .

                    FILTER (datatype(?forwardValue) = xsd:decimal)
                    FILTER (datatype(?aftValue) = xsd:decimal)
                    FILTER (?forwardValue >= 1.5)
                    FILTER (?aftValue >= 1.5)
                }
            }
        """
    ] .
<END_SHACL>
