<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF066Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its ice class."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice classes IA and IA Super, iceKnifePracticable must be a boolean and the rudder protection arrangement must provide an ice knife extending below the LIWL when practicable, or an equivalent rudder protection means otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIa ||
                    ?iceClass = nltl:iceClassIaSuper
                )
                FILTER NOT EXISTS {
                    $this nltl:iceKnifePracticable ?practicable .
                    FILTER (
                        (?practicable = true &&
                         EXISTS {
                             $this nltl:hasRudderProtectionArrangement ?arrangement .
                             ?arrangement a nltl:iceKnife ;
                                 nltl:extendsBelowLowerIceWaterline true
                         })
                        ||
                        (?practicable = false &&
                         EXISTS {
                             $this nltl:hasRudderProtectionArrangement ?arrangement .
                             ?arrangement a nltl:equivalentRudderProtectionMeans
                         })
                    )
                }
            }
        """
    ] .
</END_SHACL>
