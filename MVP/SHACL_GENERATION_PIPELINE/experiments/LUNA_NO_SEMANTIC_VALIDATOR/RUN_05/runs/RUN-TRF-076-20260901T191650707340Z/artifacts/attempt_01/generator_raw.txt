        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A propeller type is required to determine applicability of the load-case rule."
    ] ;
    sh:sparql [
        sh:message "The non-applicable branch cannot be used for CP or FP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER (?propellerType IN ("CP", "FP"))
            }
        """
    ] .
