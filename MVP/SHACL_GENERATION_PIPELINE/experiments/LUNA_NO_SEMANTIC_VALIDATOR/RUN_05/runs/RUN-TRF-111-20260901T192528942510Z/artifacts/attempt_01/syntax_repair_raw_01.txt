<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:maxCount 1 ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide one extreme ice force quantity in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-111-LoadCaseShape> ;
        sh:message "The ship must provide applicable propulsion load cases with validated load-case types and safety factors."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-111-ComponentShape> ;
        sh:message "The ship must identify the propeller shaft line components subject to the requirement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a fatigue load case applies, every linked propeller shaft line component must have a fatigue safety factor of at least 1.5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor qudt:numericValue ?value .
                    ?factor qudt:unit qudt:UNITLESS .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= 1.5
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-111-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:shaftLineLoadCaseTypeValue ;
        sh:or (
            [
                sh:hasValue nltl:extremeOperationalLoadCase ;
                sh:property [
                    sh:path nltl:componentYieldSafetyFactor ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node [
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minInclusive 1.3 ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue qudt:UNITLESS ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ]
            ]
            [
                sh:hasValue nltl:bladeFailureLoadCase ;
                sh:property [
                    sh:path nltl:componentYieldSafetyFactor ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node [
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minInclusive 1.0 ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue qudt:UNITLESS ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ]
            ]
            [
                sh:hasValue nltl:fatigueLoadCase
            ]
        ) ;
        sh:message "Each propulsion load case must identify an allowed load-case type; extreme operational and blade-failure cases require the applicable yield safety factor."
    ] .

<urn:nltl:generated-shape:TRF-111-ComponentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:fatigueSafetyFactor ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

<END_SHACL>
