<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-120-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must provide damping for thruster global natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass in kilograms."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the thruster global natural frequency in hertz."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked thruster body must have a natural-frequency calculation case." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                FILTER NOT EXISTS {
                    ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked natural-frequency calculation case must specify a vibration direction." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                FILTER NOT EXISTS {
                    ?case nltl:vibrationDirection ?direction .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every calculation case direction must be longitudinal or transverse." ;
        sh:select """
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                ?case nltl:vibrationDirection ?direction .
                FILTER (
                    ?direction NOT IN (
                        nltl:longitudinalDirection,
                        nltl:transverseDirection
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The linked thruster calculation cases must include a longitudinal direction case." ;
        sh:select """
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?bodyLong .
                    ?bodyLong a nltl:thrusterBody .
                    ?bodyLong nltl:hasNaturalFrequencyCalculationCase ?longCase .
                    ?longCase nltl:vibrationDirection nltl:longitudinalDirection .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The linked thruster calculation cases must include a transverse direction case." ;
        sh:select """
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?bodyTrans .
                    ?bodyTrans a nltl:thrusterBody .
                    ?bodyTrans nltl:hasNaturalFrequencyCalculationCase ?transCase .
                    ?transCase nltl:vibrationDirection nltl:transverseDirection .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-120-ThrusterBodyShape>
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-PER-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The thruster body must provide ship attachment stiffness in kilonewtons per metre."
    ] .
<END_SHACL>
