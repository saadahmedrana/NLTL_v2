            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;

                sh:maxCount 1 ;

PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:hasOccasionalForceLoadCase ?case .
  ?case nltl:componentYieldSafetyFactor ?factorNode ;
        nltl:occasionalForceCaseAssessedComponent ?component .
  ?factorNode qudt:numericValue ?factor .
  ?component nltl:occasionalForceComponentStress ?stressNode ;
             nltl:hasMaterialProperties ?material .
  ?stressNode qudt:numericValue ?stress .
  ?material nltl:componentMaterialYieldStrength ?yieldNode .
  ?yieldNode qudt:numericValue ?yield .
  FILTER (?stress * ?factor > ?yield)
}
