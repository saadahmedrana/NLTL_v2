<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_004_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloTON_Metric
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:KiloTON_Metric
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline displacement must be at least the displacement of every linked waterline case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiQuantity .
                ?duiQuantity qudt:numericValue ?dui .
                $this nltl:hasUpperIceWaterlineCase ?case .
                ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseDisplacement .
                FILTER (?caseDisplacement > ?dui)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline displacement must equal the greatest displacement among the linked waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiQuantity .
                ?duiQuantity qudt:numericValue ?dui .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?case .
                    ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                    ?caseQuantity qudt:numericValue ?caseDisplacement .
                    FILTER (?caseDisplacement = ?dui)
                }
            }
        """
    ] .
<END_SHACL>
