<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-026-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-026-StructuralMemberShape>
    ] .

<urn:nltl:generated-shape:I2-026-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-026-MemberSupportShape> ;
        sh:message "Each structural member must identify its support condition."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Each structural member must state whether it terminates within the ice-strengthened area."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A structural member terminating within an ice-strengthened area must have fixed support at every identified support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:terminatesWithinIceStrengthenedArea true .
                $this nltl:hasMemberSupport ?support .
                FILTER NOT EXISTS {
                    ?support nltl:frameSupportCondition nltl:fixedFrameSupport .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-026-MemberSupportShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:fixedFrameSupport
            ]
            [
                sh:hasValue nltl:simpleFrameSupport
            ]
        ) ;
        sh:message "A member support must specify fixed support or simple support."
    ] ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:continuousThroughSupport ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:connectionBracketPresent ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:demonstratedRotationalRestraint ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:significantRotationalRestraintEvidence ;
                        sh:minCount 1 ;
                        sh:nodeKind sh:IRI
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:simpleFrameSupport
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:connectionBracketPresent ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:message "Support fixity must be fixed when continuity, a connection bracket, or demonstrated significant rotational restraint justifies fixity; otherwise it must be simple."
    .
<END_SHACL>
