gen:I2_032_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "The actual net effective shear area must be at least the calculated required transverse frame shear area, and the reported required area must match the formula within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure/qudt:numericValue ?pavg ;
                      nltl:frameSpan/qudt:numericValue ?frameSpan ;
                      nltl:loadPatchHeight/qudt:numericValue ?patchHeight ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:shearArea/qudt:numericValue ?actualArea ;
                      nltl:span/qudt:numericValue ?span ;
                      nltl:yieldStrength/qudt:numericValue ?yield ;
                      nltl:hasStructuralMember ?member .

                ?member nltl:frameSpacing/qudt:numericValue ?spacing ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredTransverseFrameShearArea ?requiredQuantity .

                ?requiredQuantity qudt:numericValue ?reportedRequiredArea .

                BIND(IF(?frameSpan < ?patchHeight, ?frameSpan, ?patchHeight) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?spacing * (?af * ?ppf * ?pavg) / (0.577 * ?yield)) AS ?calculatedRequiredArea)

                FILTER(
                    ABS(?reportedRequiredArea - ?calculatedRequiredArea) >
                    (0.000000001 + 0.000001 * ABS(?calculatedRequiredArea))
                    ||
                    ?actualArea < ?calculatedRequiredArea
                )
            }
        """
    ] .
