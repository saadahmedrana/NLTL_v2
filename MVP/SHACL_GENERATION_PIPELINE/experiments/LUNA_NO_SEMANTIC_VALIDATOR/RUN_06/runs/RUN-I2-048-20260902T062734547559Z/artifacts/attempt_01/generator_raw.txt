    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual hull structural steel grade must meet or exceed the Table 8 grade rank for the same steel strength category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:asBuiltPlateThickness ?thicknessNode ;
                      nltl:steelMaterialClass ?material ;
                      nltl:steelStrengthCategory ?category ;
                      nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?thicknessNode qudt:numericValue ?thickness .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?pc .
                BIND (
                    IF(
                        ?pc IN (
                            nltl:polarClassPc1,
                            nltl:polarClassPc2,
                            nltl:polarClassPc3
                        ),
                        "PC1-3",
                        IF(
                            ?pc IN (
                                nltl:polarClassPc4,
                                nltl:polarClassPc5
                            ),
                            "PC4-5",
                            IF(
                                ?pc IN (
                                    nltl:polarClassPc1,
                                    nltl:polarClassPc2,
                                    nltl:polarClassPc3,
                                    nltl:polarClassPc4,
                                    nltl:polarClassPc5
                                ),
                                "PC1-5",
                                "PC6-7"
                            )
                        )
                    )
                    AS ?group
                )
                BIND (
                    IF(
                        ?category = nltl:normalStrengthSteelCategory,
                        IF(
                            ?material = nltl:steelMaterialClassOne,
                            IF(
                                ?thickness <= 15,
                                1,
                                IF(
                                    ?thickness <= 40,
                                    IF(?group = "PC1-5", 2, 1),
                                    IF(?group = "PC1-5", 3, 2)
                                )
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassTwo,
                                IF(
                                    ?thickness <= 10,
                                    1,
                                    IF(
                                        ?thickness <= 20,
                                        IF(?group = "PC1-5", 2, 1),
                                        IF(?group = "PC1-5", 3, 2)
                                    )
                                ),
                                IF(
                                    ?thickness <= 10,
                                    IF(?group = "PC1-5", 3, 1),
                                    IF(?group = "PC1-5", 3, 2)
                                )
                            )
                        ),
                        IF(
                            ?material = nltl:steelMaterialClassOne,
                            IF(
                                ?thickness <= 10,
                                1,
                                IF(?group = "PC1-5", 2, 1)
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassTwo,
                                IF(
                                    ?thickness <= 10,
                                    1,
                                    IF(?group = "PC1-5", 2, 1)
                                ),
                                IF(
                                    ?group = "PC1-3",
                                    IF(?thickness > 35, 4, 3),
                                    IF(
                                        ?group = "PC4-5",
                                        IF(?thickness > 45, 4, 3),
                                        IF(
                                            ?thickness <= 10,
                                            1,
                                            IF(?thickness <= 35, 2, 3)
                                        )
                                    )
                                )
                            )
                        )
                    )
                    AS ?expectedRank
                )
                FILTER (
                    ?thickness <= 50 &&
                    !(
                        ?category = nltl:normalStrengthSteelCategory &&
                        ?material = nltl:steelMaterialClassThree &&
                        (
                            (
                                ?group = "PC1-3" &&
                                ?thickness > 35
                            ) ||
                            (
                                ?group = "PC4-5" &&
                                ?thickness > 40
                            )
                        )
                    ) &&
                    (
                        ?actualRank < ?expectedRank ||
                        ?requiredRank != ?expectedRank ||
                        (
                            ?category = nltl:normalStrengthSteelCategory &&
                            ?required IN (
                                nltl:steelGradeAh,
                                nltl:steelGradeDh,
                                nltl:steelGradeEh,
                                nltl:steelGradeFh
                            )
                        ) ||
                        (
                            ?category = nltl:highTensileSteelCategory &&
                            ?required IN (
                                nltl:steelGradeB,
                                nltl:steelGradeD,
                                nltl:steelGradeE
                            )
                        )
                    )
                )
            }
        """
    ] .
