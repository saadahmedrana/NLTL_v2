<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:message "The ship must provide at least one interpolation calculation case."
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must provide the interpolated value."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case must provide interpolation points."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case must identify a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape ;
        sh:message "The calculation case must identify an upper interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "The calculation case must provide the hull girder longitudinal position from aft."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The calculation case must provide the interpolated longitudinal distribution factor."
    ] .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node gen:UnspecifiedQuantityShape ;
        sh:message "Each interpolation point must provide an interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node gen:UnspecifiedQuantityShape ;
        sh:message "Each interpolation point must provide an interpolation point value."
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The length quantity must contain a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:message "The hull girder longitudinal position must use metres."
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The dimensionless factor must contain a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:message "The interpolated longitudinal distribution factor must be unitless."
    ] .

gen:UnspecifiedQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The interpolation quantity must contain a decimal numeric value."
    ] .
<END_SHACL>
