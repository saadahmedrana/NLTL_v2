<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_060_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2_060_StructuralMemberShape>
    ] ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        sh:message "Loads carried by a structural member in way of a cut-out must not cause instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout .
                ?member nltl:memberInWayOfCutout true .
                FILTER NOT EXISTS {
                    $this nltl:memberInstability false .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Where stiffening is necessary for a structural member in way of a cut-out, structural stiffening must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout .
                ?member nltl:memberInWayOfCutout true .
                ?member nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2_060_StructuralMemberShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCutout ;
        sh:class nltl:cutout ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] .
<END_SHACL>
