<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gs:CalculationCaseShape
    ] .

gs:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:in ( nltl:prescribedAnalyticalProcedure ) ;
        sh:message "Each applicable calculation case shall use the prescribed analytical procedure rather than a direct calculation."
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gs:HullStructureShape
    ] .

gs:HullStructureShape
    a sh:NodeShape ;
    sh:class nltl:hullStructure ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node gs:ThicknessQuantityShape ;
        sh:message "The assessed hull structure shall provide its thickness."
    ] .

gs:ThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
