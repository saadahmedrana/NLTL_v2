<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_062_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record that the required direct calculation is used."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:qualifiedValueShape [
            sh:or (
                [
                    sh:class nltl:loadCarryingStringer ;
                    sh:property [
                        sh:path nltl:partOfGrillageSystem ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:grillageSystem
                        ]
                    ]
                ]
                [
                    sh:class nltl:webFrame ;
                    sh:property [
                        sh:path nltl:partOfGrillageSystem ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:grillageSystem
                        ]
                    ]
                ]
            )
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The ship must identify at least one load carrying stringer or web frame forming part of a grillage system."
    ] ;
    sh:sparql [
        sh:message "Every linked load carrying stringer or web frame must be linked to a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member rdf:type ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage rdf:type nltl:grillageSystem .
                }
            }
        """
    ] .

</END_SHACL>
