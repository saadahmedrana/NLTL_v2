@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_064_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MegaPA
            ]
        ] ;
        sh:message "The ship shall provide one yield-strength value in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:linearCalculationMethodValue ;
                sh:message "The applicable calculation case shall identify the linear calculation method."
            ]
        ] ;
        sh:message "The ship shall provide an applicable calculation case for the linear calculation method."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:bucklingCriteriaSatisfied ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "Applicable structural members shall confirm that the buckling criteria are satisfied."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberWeb ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:structuralMemberWeb ;
                    sh:property [
                        sh:path nltl:nominalShearStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:datatype xsd:decimal
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue qudt:MegaPA
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberFlange ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:structuralMemberFlange ;
                    sh:property [
                        sh:path nltl:nominalFlangeVonMisesStress ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:datatype xsd:decimal
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue qudt:MegaPA
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Nominal web shear stress shall be less than yield strength divided by sqrt(3), and nominal flange von Mises stress shall be less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:yieldStrength/qudt:numericValue ?yieldStrength .
                $this nltl:hasStructuralMember ?member .
                {
                    ?member nltl:hasStructuralMemberWeb/nltl:nominalShearStress/qudt:numericValue ?shearStress .
                    FILTER ( ! ( ?shearStress < ( ?yieldStrength / 1.7320508075688772 ) ) )
                }
                UNION
                {
                    ?member nltl:hasStructuralMemberFlange/nltl:nominalFlangeVonMisesStress/qudt:numericValue ?flangeStress .
                    FILTER ( ! ( ?flangeStress < ( 1.15 * ?yieldStrength ) ) )
                }
            }
        """
    ] .
</END_SHACL>
