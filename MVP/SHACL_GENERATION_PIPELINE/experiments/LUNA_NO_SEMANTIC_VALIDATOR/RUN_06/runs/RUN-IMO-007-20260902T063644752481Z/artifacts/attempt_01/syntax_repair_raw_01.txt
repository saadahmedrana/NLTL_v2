<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007-OpenWater>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice condition classification is required to determine whether the open-water conditions apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open-water ice conditions, sea ice concentration must be a decimal quantity in unit 1 and be less than 1/10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:seaIceConcentration ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit qudt:UNITLESS .
                    FILTER(datatype(?value) = xsd:decimal && ?value < 1)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every reported sea ice concentration for open-water ice conditions must be a decimal quantity in unit 1 and be less than 1/10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?quantity .
                FILTER NOT EXISTS {
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit qudt:UNITLESS .
                    FILTER(datatype(?value) = xsd:decimal && ?value < 1)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open-water ice conditions, ice of land origin must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:iceOfLandOriginPresent ?present .
                    FILTER(datatype(?present) = xsd:boolean && ?present = false)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every reported land-origin-ice observation for open-water ice conditions must be the boolean value false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:iceOfLandOriginPresent ?present .
                FILTER ( !(datatype(?present) = xsd:boolean && ?present = false) )
            }
        """
    ] .
<END_SHACL>
