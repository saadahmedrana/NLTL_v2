gen:IMO040ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceAccretionLikely ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "When ice accretion is likely, every relevant hatch or door must have associated ice or snow removal or prevention means." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasRelevantHatchOrDoor ?item .
                        FILTER NOT EXISTS {
                            ?item nltl:hasIceOrSnowRemovalOrPreventionMeans ?means .
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceAccretionLikely ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ] .
