<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hazardProtectionCoverage ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record hazard protection coverage."
    ] ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:machineryInstallationItem ;
        sh:node gen:IMO043MachineryInstallationShape
    ] ;
    sh:sparql [
        sh:message "Each machinery installation with an applicable hazard shall have hazard protection evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasMachineryInstallation ?installation .
                ?installation nltl:hasApplicableHazard ?hazard .
                FILTER NOT EXISTS {
                    ?installation nltl:hasHazardProtectionEvidence ?evidence .
                }
            }
        """
    ] .

gen:IMO043MachineryInstallationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:associatedEquipmentItem ;
        sh:node gen:IMO043AssociatedEquipmentShape
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceAccretionHazard
            nltl:snowAccumulationHazard
            nltl:seawaterIceIngestionHazard
            nltl:liquidFreezingOrViscosityIncreaseHazard
            nltl:seawaterIntakeTemperatureHazard
            nltl:snowIngestionHazard
        )
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact
    ] .

gen:IMO043AssociatedEquipmentShape
    a sh:NodeShape ;
    sh:class nltl:associatedEquipmentItem .

</END_SHACL>
