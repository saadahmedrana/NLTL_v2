<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO045Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:machinerySeawaterSupply ;
        sh:datatype xsd:boolean ;
        sh:message "Machinery seawater supply, when recorded, must be a boolean."
    ] ;
    sh:property [
        sh:path nltl:iceIngestionPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:message "Ice ingestion prevention presence, when recorded, must be a boolean."
    ] ;
    sh:sparql [
        sh:message "Each machinery seawater supply must either have ice-ingestion prevention or an approved alternate functionality arrangement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:machinerySeawaterSupply ?supply .
                FILTER (?supply = "true"^^xsd:boolean)
                FILTER NOT EXISTS {
                    $this nltl:iceIngestionPreventionPresent ?prevention .
                    FILTER (?prevention = "true"^^xsd:boolean)
                }
                FILTER NOT EXISTS {
                    $this nltl:alternateFunctionalityArrangementApprovalStatus
                        nltl:evidenceStateApproved .
                }
            }
        """
    ] .
<END_SHACL>
