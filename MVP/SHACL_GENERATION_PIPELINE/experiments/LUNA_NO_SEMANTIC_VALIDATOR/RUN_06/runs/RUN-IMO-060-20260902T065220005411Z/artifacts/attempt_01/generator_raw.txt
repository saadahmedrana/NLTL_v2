<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_060_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fixedWaterSystemSeparateFromMainFirePumps ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The fixed water system separation status must be provided as a boolean."
    ] ;
    sh:property [
        sh:path nltl:independentSeaSuctionUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The independent sea suction status must be provided as a boolean."
    ] ;
    sh:property [
        sh:path nltl:seaSuctionIceClearingCapability ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the fixed water system is separate from the main fire pumps and uses an independent sea suction, the sea suction must be capable of being cleared of ice accumulation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:fixedWaterSystemSeparateFromMainFirePumps true .
                $this nltl:independentSeaSuctionUsed true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaSuctionIceClearingCapability ?capabilityTrue .
                        FILTER (?capabilityTrue = true)
                    }
                    ||
                    EXISTS {
                        $this nltl:seaSuctionIceClearingCapability ?capabilityValue .
                        FILTER (?capabilityValue != true)
                    }
                )
            }
        """
    ] .
<END_SHACL>
