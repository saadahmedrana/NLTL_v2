<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization." ;
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO_104_CrewMemberShape ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each relevant PWOM procedure or equipment item assigned to a crew member shall be covered by that crew member's familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this a nltl:ship ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """ ;
    ] .

gen:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_AssignedDutyShape ;
        sh:message "Each crew member shall have an assigned duty linked to relevant PWOM items." ;
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_FamiliarizationRecordShape ;
        sh:message "Each crew member shall have a PWOM familiarization record." ;
    ] .

gen:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_PWOMItemShape ;
        sh:message "Each assigned duty shall identify the relevant PWOM procedure or equipment." ;
    ] .

gen:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_PWOMItemShape ;
        sh:message "The familiarization record shall identify the covered PWOM procedure or equipment." ;
    ] .

gen:IMO_104_PWOMItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .
<END_SHACL>
