<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-110> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:hasValue "oil_tanker"^^xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship type must be oil_tanker."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date must be on or after 1 January 2017."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must have one deadweight quantity expressed in metric tonnes."
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B oil tanker of less than 5,000 tonnes deadweight constructed on or after 1 January 2017, both required cargo-tank protection statuses must be compliant." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this rdf:type nltl:oilTanker ;
                      nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?constructionDate .

                ?deadweight qudt:numericValue ?deadweightValue ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN ( nltl:polarShipCategoryA, nltl:polarShipCategoryB )
                    && ?shipType = "oil_tanker"^^xsd:string
                    && ?deadweightValue < 5000
                    && ?constructionDate >= "2017-01-01"^^xsd:date
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?doubleBottomStatus .
                        FILTER (
                            datatype(?doubleBottomStatus) = xsd:boolean
                            && ?doubleBottomStatus = true
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus ?doubleBottomStatus .
                        FILTER (
                            !(datatype(?doubleBottomStatus) = xsd:boolean
                              && ?doubleBottomStatus = true)
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:wingTankProtectionStatus ?wingTankStatus .
                        FILTER (
                            datatype(?wingTankStatus) = xsd:boolean
                            && ?wingTankStatus = true
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:wingTankProtectionStatus ?wingTankStatus .
                        FILTER (
                            !(datatype(?wingTankStatus) = xsd:boolean
                              && ?wingTankStatus = true)
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
