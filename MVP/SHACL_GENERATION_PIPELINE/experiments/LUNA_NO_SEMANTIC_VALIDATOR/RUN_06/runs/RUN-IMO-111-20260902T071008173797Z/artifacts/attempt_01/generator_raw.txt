<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, each represented qualifying tank must have a tank outer shell separation value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:tankType ?tankType ;
                      nltl:individualTankCapacity ?capacityNode .

                ?capacityNode <http://qudt.org/schema/qudt/numericValue> ?capacity ;
                              <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M3> .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (
                    ?tankType IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )
                FILTER (?capacity > 30)
                FILTER NOT EXISTS {
                    $this nltl:tankOuterShellSeparation ?separationNode .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, a qualifying tank must be separated from the outer shell by at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:tankType ?tankType ;
                      nltl:individualTankCapacity ?capacityNode ;
                      nltl:tankOuterShellSeparation ?separationNode .

                ?capacityNode qudt:numericValue ?capacity ;
                              qudt:unit <http://qudt.org/vocab/unit/M3> .
                ?separationNode qudt:numericValue ?separation ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER (
                    ?tankType IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )
                FILTER (?capacity > 30)
                FILTER (?separation < 0.76)
            }
        """
    ] .
<END_SHACL>
