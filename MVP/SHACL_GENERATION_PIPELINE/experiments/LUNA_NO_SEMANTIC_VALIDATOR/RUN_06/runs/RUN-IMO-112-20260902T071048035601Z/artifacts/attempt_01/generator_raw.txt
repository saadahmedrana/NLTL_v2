<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_112_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must specify exactly one operating area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:hasValue "Arctic_waters"^^xsd:string
            ] ;
            sh:property [
                sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "NLS or NLS mixture discharge to sea shall be false in Arctic waters."
            ]
        ]
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:operatingArea ?area .
    FILTER (?area = "Arctic_waters"^^<http://www.w3.org/2001/XMLSchema#string>)
}
""" ;
                sh:message "The non-Arctic operating-area branch cannot apply to Arctic waters."
            ]
        ]
    ) .
<END_SHACL>
