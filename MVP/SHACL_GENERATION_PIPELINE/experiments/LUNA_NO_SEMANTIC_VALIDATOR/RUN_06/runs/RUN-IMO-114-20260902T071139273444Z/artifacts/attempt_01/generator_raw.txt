<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-114> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship category must be recorded as Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Type 3 tank NLS applicability indicator is required."
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The cargo tank type is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:administrationCarriageApprovalStatus ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Applicable carriage of the specified NLS in Type 3 cargo tanks requires Administration approval."
            ] ;
            sh:property [
                sh:path nltl:applicableCertificatePolarOperationEntryPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applicable polar operation must be reflected in the required certificate."
            ]
        ]
        [
            sh:sparql [
                sh:message "The approval and certificate-entry obligations apply only when all specified applicability conditions are met." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:constructionDate ?date ;
          nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 ;
          nltl:cargoTankType ?tankType .
    FILTER (
        !(
            ?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB)
            && ?date >= "2017-01-01"^^xsd:date
            && ?requiresType3 = true
            && ?tankType = "3"
        )
    )
}
"""
            ]
        ]
    ) .

<END_SHACL>
