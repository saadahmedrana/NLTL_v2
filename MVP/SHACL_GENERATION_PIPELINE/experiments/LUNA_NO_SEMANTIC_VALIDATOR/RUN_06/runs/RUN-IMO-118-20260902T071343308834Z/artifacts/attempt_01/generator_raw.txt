@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO118Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have exactly one construction date." 
    ] ;
    sh:or (
        gen:IMO118Before2017
        gen:IMO118CategoryAB
        gen:IMO118PassengerShip
        gen:IMO118CategoryCNonPassenger
    ) .

gen:IMO118Before2017 a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2016-12-31"^^xsd:date
    ] .

gen:IMO118CategoryAB a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:message "A ship constructed on or after 1 January 2017 in Category A or B must not discharge sewage to sea unless the discharge complies with paragraph 4.2.1.3."
    ] ;
    sh:or (
        gen:IMO118DischargeProhibited
        gen:IMO118CompliantException
    ) .

gen:IMO118PassengerShip a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:class nltl:passengerShip ;
    sh:or (
        gen:IMO118DischargeProhibited
        gen:IMO118CompliantException
    ) .

gen:IMO118CategoryCNonPassenger a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:minCount 1
    ] ;
    sh:not [
        sh:class nltl:passengerShip
    ] .

gen:IMO118DischargeProhibited a sh:NodeShape ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sewage discharge to sea must be false for the applicable ship."
    ] .

gen:IMO118CompliantException a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The exception requires compliance with paragraph 4.2.1.3."
    ] .
