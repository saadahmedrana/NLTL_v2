<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-002-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "A construction contract date, when represented, must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships contracted on or after 2019-01-01 and before 2021-07-05 must identify the 2017 Ice Class Regulations as the applicable regulation edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    ?date >= "2019-01-01"^^xsd:date &&
                    ?date < "2021-07-05"^^xsd:date
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2017 .
                }
            }
        """ ;
    ] .
</urn:nltl:generated-shape:TRF-002-ShipShape>
<END_SHACL>
