<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF005Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [ sh:node gen:TRF005ApplicableShape ]
        [ sh:node gen:TRF005NonApplicableShape ]
    ) ;
    sh:message "The construction dates must establish whether the 2002 Ice Class Regulations apply, and applicable ships must identify the 2002 regulation edition." .

gen:TRF005ApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iceClassRegulationEdition2002 ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stage ;
                      nltl:constructionContractDate ?contract .
                FILTER (
                    !(
                        ?stage >= "2003-09-01"^^xsd:date &&
                        ?contract < "2010-01-01"^^xsd:date
                    )
                )
            }
        """ ;
        sh:message "The applicable branch requires a construction stage date on or after 1 September 2003 and a construction contract date before 1 January 2010."
    ] .

gen:TRF005NonApplicableShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stage ;
                      nltl:constructionContractDate ?contract .
                FILTER (
                    ?stage >= "2003-09-01"^^xsd:date &&
                    ?contract < "2010-01-01"^^xsd:date
                )
            }
        """ ;
        sh:message "The non-applicable branch cannot be used when the construction stage date is on or after 1 September 2003 and the construction contract date is before 1 January 2010."
    ] .
<END_SHACL>
