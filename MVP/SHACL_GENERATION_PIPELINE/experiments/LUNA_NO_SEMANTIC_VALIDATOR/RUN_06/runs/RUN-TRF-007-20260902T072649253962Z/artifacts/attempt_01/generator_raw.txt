@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF007NonApplicableIbShape
        gen:TRF007NonApplicableIcShape
        gen:TRF007NonApplicableIiShape
        gen:TRF007NonApplicableIiiShape
        gen:TRF007NonApplicableIaSuperShape
        gen:TRF007NonApplicableIaShape
        gen:TRF007ApplicableIaSuperShape
        gen:TRF007ApplicableIaShape
    ) .

gen:TRF007NonApplicableIbShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIb
    ] .

gen:TRF007NonApplicableIcShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIc
    ] .

gen:TRF007NonApplicableIiShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIi
    ] .

gen:TRF007NonApplicableIiiShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIii
    ] .

gen:TRF007NonApplicableIaSuperShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper
    ] ;
    sh:node gen:TRF007PostConstructionShape .

gen:TRF007NonApplicableIaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIa
    ] ;
    sh:node gen:TRF007PostConstructionShape .

gen:TRF007PostConstructionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] .

gen:TRF007ApplicableIaSuperShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper
    ] ;
    sh:node gen:TRF007ApplicableShape .

gen:TRF007ApplicableIaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIa
    ] ;
    sh:node gen:TRF007ApplicableShape .

gen:TRF007ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:maxExclusive "2003-09-01"^^xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable IA or IA Super ship, the assessment date must be no later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:deliveryDate ?delivery ;
                      nltl:assessmentDate ?assessment .
                FILTER (
                    YEAR(?assessment) < YEAR(?delivery) + 20
                    ||
                    (
                        YEAR(?assessment) = YEAR(?delivery) + 20
                        && MONTH(?assessment) = 1
                        && DAY(?assessment) <= 1
                    )
                )
            }
        """
    ] .
