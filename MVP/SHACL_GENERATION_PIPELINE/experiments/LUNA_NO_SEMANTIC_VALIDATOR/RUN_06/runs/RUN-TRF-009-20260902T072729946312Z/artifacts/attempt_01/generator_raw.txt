<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one ice class." ;
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction stage date is required to determine applicability." ;
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessment date is required to evaluate the compliance deadline." ;
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionStageDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            ?constructionStageDate < "2003-09-01"^^xsd:date
                        )
                    }
                """ ;
                sh:message "This branch is applicable only to ships of ice class IA Super or IA whose construction stage date is before 1 September 2003." ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The delivery date is required for an applicable ship." ;
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "An applicable ship must identify compliance with section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations." ;
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionStageDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                        )
                        FILTER (
                            ?constructionStageDate < "2003-09-01"^^xsd:date
                        )
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                                &&
                                (
                                    MONTH(?assessmentDate) > 1
                                    ||
                                    (
                                        MONTH(?assessmentDate) = 1
                                        && DAY(?assessmentDate) > 1
                                    )
                                )
                            )
                        )
                    }
                """ ;
                sh:message "The applicable ship must comply by 1 January of the year twenty years after its delivery year." ;
            ]
        ]
    ) .
<END_SHACL>
