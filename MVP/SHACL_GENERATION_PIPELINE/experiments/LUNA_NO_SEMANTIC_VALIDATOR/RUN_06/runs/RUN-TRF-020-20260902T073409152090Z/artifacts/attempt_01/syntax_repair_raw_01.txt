<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF020ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "2003-09-01"^^xsd:date ;
                sh:message "An applicable ship must have a construction stage date on or after 2003-09-01."
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:message "An applicable ship must have ice class IA Super, IA, IB, or IC."
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide the externally selected engine-output coefficient."
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide the lower-ice-waterline resistance input."
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide the upper-ice-waterline resistance input."
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide propeller diameter in metres."
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applicable ship must provide the propeller count."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:controllablePitch
                    nltl:fixedPitch
                ) ;
                sh:message "The applicable ship must identify a controllable-pitch or fixed-pitch propeller."
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:conventionalPropulsionSystem
                    nltl:electricPropulsionMachinery
                    nltl:hydraulicPropulsionMachinery
                ) ;
                sh:message "The applicable ship must identify its propulsion system type."
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:iceWaterline
                ] ;
                sh:message "The applicable ship must identify its lower ice waterline."
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:iceWaterline ;
                    sh:property [
                        sh:path nltl:upperIceWaterlineLength ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:upperIceWaterlineBreadth ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ] ;
                sh:message "The applicable ship must identify the upper ice waterline with its length and breadth."
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide the externally calculated minimum power at LIWL."
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide the externally calculated minimum power at UIWL."
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The applicable ship must provide maximum continuous rating power in kW."
            ] ;
            sh:sparql [
                sh:message "Maximum continuous rating power must be at least both externally calculated minimum powers." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass ;
                              nltl:maximumContinuousRatingPower/qudt:numericValue ?mcr ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline/qudt:numericValue ?lower ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline/qudt:numericValue ?upper .
                        FILTER (
                            ?date >= "2003-09-01"^^xsd:date &&
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            )
                        )
                        FILTER (?mcr < ?lower || ?mcr < ?upper)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The ship must provide its construction stage date before applicability can be determined."
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:message "The ship must provide ice-class evidence before applicability can be determined."
            ] ;
            sh:sparql [
                sh:message "The ship is not in the applicable construction-date and ice-class branch." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass .
                        FILTER (
                            ?date >= "2003-09-01"^^xsd:date &&
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
