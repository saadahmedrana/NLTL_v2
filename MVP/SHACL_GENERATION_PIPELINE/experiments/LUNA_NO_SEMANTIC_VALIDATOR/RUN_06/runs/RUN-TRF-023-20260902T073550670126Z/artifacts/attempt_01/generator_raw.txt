<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:TRF_023_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship to which the 1985 ice class regulations apply must have ice class IB or IC and provide maximum continuous rating power in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:applicableIceClassRegulationEdition
                       nltl:iceClassRuleEdition1985 .

                FILTER (
                    NOT EXISTS {
                        $this nltl:iceClass ?iceClass .
                        FILTER (
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?powerQuantity .
                        ?powerQuantity qudt:numericValue ?powerValue ;
                                        qudt:unit qudt:KiloW .
                        FILTER (datatype(?powerValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
</BEGIN_SHACL>
