    sh:property gs:shipLengthProperty ;
    sh:property gs:iceClassDraughtProperty ;
    sh:property gs:shipBreadthProperty ;
    sh:property gs:clampedHullGeometryProperty ;
    sh:property gs:coefficientC3Property ;

    sh:property gs:coefficientG2Property ;
    sh:property gs:coefficientG3Property ;
    sh:sparql [
        sh:message "clampedHullGeometryTerm must equal the clamped value of ((shipLength * iceClassDraught) / shipBreadth squared) cubed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthValueNode ;
                      nltl:iceClassDraught ?draughtValueNode ;
                      nltl:shipBreadth ?breadthValueNode ;
                      nltl:clampedHullGeometryTerm ?resultValueNode .

                ?lengthValueNode <http://qudt.org/schema/qudt/numericValue> ?length .
                ?draughtValueNode <http://qudt.org/schema/qudt/numericValue> ?draught .
                ?breadthValueNode <http://qudt.org/schema/qudt/numericValue> ?breadth .
                ?resultValueNode <http://qudt.org/schema/qudt/numericValue> ?result .

                BIND((?length * ?draught / (?breadth * ?breadth)) *
                     (?length * ?draught / (?breadth * ?breadth)) *
                     (?length * ?draught / (?breadth * ?breadth)) AS ?raw)
                BIND(IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw)) AS ?expected)
                FILTER(ABS(?result - ?expected) >
                       (0.000001 * ABS(?expected) + 0.000000001))
            }
        """
    ] .
