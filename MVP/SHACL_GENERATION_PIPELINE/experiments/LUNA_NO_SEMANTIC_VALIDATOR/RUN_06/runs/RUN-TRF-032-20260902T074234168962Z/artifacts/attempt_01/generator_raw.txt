<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF032_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether it is ice strengthened."
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The regulation-required scantling must be supplied as a decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The classification-society-required scantling must be supplied as a decimal quantity."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceStrengtheningStatus false ;
                              nltl:regulationRequiredScantling ?regulationQuantity ;
                              nltl:classificationSocietyRequiredScantling ?classQuantity .
                        ?regulationQuantity qudt:numericValue ?regulationValue .
                        ?classQuantity qudt:numericValue ?classValue .
                        FILTER (?regulationValue >= ?classValue)
                    }
                """ ;
                sh:message "A non-ice-strengthened ship satisfies this rule when the regulation-required scantling is not less than the classification-society-required scantling."
            ]
        ]
        [
            sh:property [
                sh:path nltl:selectedDesignScantling ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceStrengtheningStatus false ;
                              nltl:regulationRequiredScantling ?regulationQuantity ;
                              nltl:classificationSocietyRequiredScantling ?classQuantity ;
                              nltl:selectedDesignScantling ?selectedQuantity .
                        ?regulationQuantity qudt:numericValue ?regulationValue .
                        ?classQuantity qudt:numericValue ?classValue .
                        ?selectedQuantity qudt:numericValue ?selectedValue .
                        FILTER (?regulationValue < ?classValue)
                        FILTER (?selectedValue = ?classValue)
                    }
                """ ;
                sh:message "When a non-ice-strengthened ship has a regulation-required scantling less than the classification-society-required scantling, the selected design scantling must equal the classification-society-required scantling."
            ]
        ]
    ) .
<END_SHACL>
