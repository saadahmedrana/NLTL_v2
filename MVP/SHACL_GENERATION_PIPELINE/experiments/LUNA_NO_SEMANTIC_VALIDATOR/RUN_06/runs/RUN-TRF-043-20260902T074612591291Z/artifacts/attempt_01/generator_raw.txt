<urn:nltl:generated-shape:TRF-043-ShipShape>

        sh:node <urn:nltl:generated-shape:TRF-043-TerminationCaseShape> ;
        sh:message "Each framing ice-strengthening termination case must conform to the termination conditions."
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-043-MillimetreQuantityShape> ;
        sh:message "The termination case must specify one extension beyond the adjacent boundary in millimetres."
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        ) ;
        sh:message "The adjacent termination boundary must be a deck structure, tank boundary plating, or tank top."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:TRF-043-ApplicableTerminationCaseShape>
        <urn:nltl:generated-shape:TRF-043-NonApplicableTerminationCaseShape>
    ) .

    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:maxInclusive 250
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm."
    ] .

<urn:nltl:generated-shape:TRF-043-NonApplicableTerminationCaseShape>

    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minExclusive 250
            ]
        ]
    ] .
