<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF_049_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame must be within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:class nltl:frameAttachment ;
        sh:node nltl:TRF_049_FrameAttachmentShape ;
        sh:minCount 1 ;
        sh:message "The frame must have an effective frame attachment."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node nltl:TRF_049_FrameWebThicknessShape
    ] ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
            [
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2 ;
                sh:message "A frame running through supporting structure must have both web-plate sides connected."
            ]
        ]
    ) ;
    sh:or (
        nltl:TRF_049_LongitudinalFrameBranchShape
        nltl:TRF_049_TransverseFrameBranchShape
    ) .

nltl:TRF_049_LongitudinalFrameBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path rdf:type ;
        sh:hasValue nltl:longitudinalFrame ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure ;
        sh:node nltl:TRF_049_SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure ;
        sh:node nltl:TRF_049_SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node nltl:TRF_049_LongitudinalAttachmentShape ;
        sh:minCount 1
    ] .

nltl:TRF_049_TransverseFrameBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path rdf:type ;
        sh:hasValue nltl:transverseFrame ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:node nltl:TRF_049_TerminatingAttachmentShape ;
                sh:minCount 1
            ]
        ]
    ) .

nltl:TRF_049_FrameAttachmentShape
    a sh:NodeShape ;
    sh:class nltl:frameAttachment ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame attachment must be confirmed effective."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node nltl:TRF_049_ConnectionBracketShape
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bracket installation requires the linked frame to provide a frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    ?frame nltl:hasFrameAttachment $this .
                    ?frame nltl:frameWebThickness ?thickness .
                    ?thickness qudt:numericValue ?value .
                    ?thickness qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                }
            }
        """
    ] .

nltl:TRF_049_LongitudinalAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node nltl:TRF_049_ConnectionBracketShape ;
        sh:minCount 1 ;
        sh:message "A longitudinal frame must be attached by brackets."
    ] .

nltl:TRF_049_TerminatingAttachmentShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasConnectionBracket ;
                sh:class nltl:connectionBracket ;
                sh:node nltl:TRF_049_ConnectionBracketShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:similarAttachmentConstructionPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
    ) .

nltl:TRF_049_SupportingStructureShape
    a sh:NodeShape ;
    sh:class nltl:supportingStructure ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:class nltl:frameAttachment ;
        sh:node nltl:TRF_049_FrameAttachmentShape ;
        sh:minCount 1 ;
        sh:message "Each linked supporting structure must have an effective attachment."
    ] .

nltl:TRF_049_ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node nltl:TRF_049_BracketThicknessShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An installed bracket must specify its thickness."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened against buckling."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The bracket thickness must be at least the thickness of the frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                ?attachment nltl:hasConnectionBracket $this .
                ?frame nltl:hasFrameAttachment ?attachment .
                ?frame nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue .
                ?this nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

nltl:TRF_049_FrameWebThicknessShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

nltl:TRF_049_BracketThicknessShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .
<END_SHACL>
