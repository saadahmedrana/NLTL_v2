<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF053_QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF053_IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltl:TRF053_IcePressureValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltl:TRF053_DesignIceLoadHeightValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node nltl:TRF053_SpanValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltl:TRF053_FrameMomentFactorValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltl:TRF053_YieldStrengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node nltl:TRF053_DesignLineLoadValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltl:TRF053_SectionModulusValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltl:TRF053_RequiredShearAreaValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "designLineLoad, sectionModulus, and requiredShearArea must satisfy the specified formulas within a relative tolerance of 1e-6 plus an absolute tolerance of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?yNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?yNode qudt:numericValue ?y .
                ?lineNode qudt:numericValue ?line .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?lineExpected)
                BIND((0.9 * 1.8 * ?lineExpected * ?l * ?l /
                      (?m * ?y) * 1000000) AS ?zExpected)
                BIND((1.7320508075688772 * 0.9 * 1.8 * 1.2 *
                      ?lineExpected * ?l / (2 * ?y) * 10000) AS ?aExpected)

                FILTER(
                    ABS(?line - ?lineExpected) >
                        (0.000001 * ABS(?lineExpected) + 0.000000001)
                    ||
                    ABS(?z - ?zExpected) >
                        (0.000001 * ABS(?zExpected) + 0.000000001)
                    ||
                    ABS(?a - ?aExpected) >
                        (0.000001 * ABS(?aExpected) + 0.000000001)
                )
            }
        """
    ] .

nltl:TRF053_IcePressureValueShape
