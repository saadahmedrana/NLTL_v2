        sh:path nltl:requiredSectionModulus ;

        sh:path nltl:requiredShearArea ;

        sh:maxCount 1 ;
        sh:node gen:TRF055ShearAreaQuantity
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Actual section modulus and actual shear area must each be at least their corresponding required value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualZNode .
                ?actualZNode qudt:numericValue ?actualZ .

                $this nltl:requiredSectionModulus ?requiredZNode .
                ?requiredZNode qudt:numericValue ?requiredZ .

                $this nltl:actualShearArea ?actualANode .
                ?actualANode qudt:numericValue ?actualA .

                $this nltl:requiredShearArea ?requiredANode .
                ?requiredANode qudt:numericValue ?requiredA .

                FILTER (?actualZ < ?requiredZ || ?actualA < ?requiredA)
            }
        """
    ] ;
    sh:or (
        gen:TRF055NotVeryLongHatch
        gen:TRF055VeryLongWithoutReduction
        gen:TRF055ApprovedVeryLongReduction
    ) .
