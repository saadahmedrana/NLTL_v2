@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-063-QuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-063-EnvelopeShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:class nltl:tankTop ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:node <urn:nltl:generated-shape:TRF-063-QuantityValueShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:node <urn:nltl:generated-shape:TRF-063-QuantityValueShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:TRF-063-TwinShipShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:node <urn:nltl:generated-shape:TRF-063-EnvelopeShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:TRF-063-TripleShipShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 3 ;
        sh:maxInclusive 3
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:node <urn:nltl:generated-shape:TRF-063-EnvelopeShape> ;
        sh:minCount 1
    ] .

<urn:nltl:generated-shape:TRF-063-OtherShipShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The twin- or triple-screw strengthening requirement must not be selected for another propeller count." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count .
                FILTER (?count IN (2, 3))
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-063-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF-063-TwinShipShape>
        <urn:nltl:generated-shape:TRF-063-TripleShipShape>
        <urn:nltl:generated-shape:TRF-063-OtherShipShape>
    ) .
