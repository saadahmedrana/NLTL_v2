<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF070ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( "azimuthing" "fixed" ) ;
        sh:message "The ship must identify an azimuthing or fixed thruster type."
    ] ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The applicable thruster must be identified as a main propulsion thruster."
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A propeller/ice interaction load in kN is required."
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A thruster-body/ice interaction load in kN is required."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The applicable local ice pressure in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:designCondition ;
            sh:property [
                sh:path nltl:designConditionLocalStrengthCapacity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "The design condition must provide the externally assessed local-strength capacity in kN."
            ]
        ] ;
        sh:message "A design condition with the local-strength assessment is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed local-strength capacity must be at least the applicable local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:localIcePressure ?pressureQuantity .
                ?pressureQuantity qudt:numericValue ?pressureValue .
                $this nltl:hasDesignCondition ?designCondition .
                ?designCondition nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacityValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .
<END_SHACL>
