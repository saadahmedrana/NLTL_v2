<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF108Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stress-life curve knowledge status must be recorded." ;
    ] ;
    sh:or (
        gen:TRF108KnownCurveBranch
        gen:TRF108UnknownCurveBranch
    ) .

gen:TRF108KnownCurveBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A known SN-curve selection must identify the propeller blade material property." ;
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A known SN-curve selection must specify the stress-life curve type." ;
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:message "A known SN-curve selection must identify the selected stress-life curve value." ;
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelectionCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Evidence must establish correspondence between the selected SN curve and the blade material properties." ;
    ] .

gen:TRF108UnknownCurveBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:stressLifeCurveKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveSelection ;
        sh:class nltl:stressLifeCurveTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:twoSlopeStressLifeCurve ;
        sh:message "When the SN curve is unknown, the selected curve must be the two-slope SN curve." ;
    ] ;
    sh:property [
        sh:path nltl:twoSlopeSnCurveUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "When the SN curve is unknown, use of the two-slope SN curve must be recorded." ;
    ] .
<END_SHACL>
