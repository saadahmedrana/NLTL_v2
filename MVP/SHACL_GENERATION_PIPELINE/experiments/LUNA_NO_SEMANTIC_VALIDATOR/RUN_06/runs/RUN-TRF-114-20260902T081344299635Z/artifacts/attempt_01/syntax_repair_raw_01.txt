<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_114_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A typical ice operating speed in knots is required."
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:node gen:TRF_114_LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "At least one applicable thruster ice impact load case is required."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A non-hemispherical contact geometry requires an equivalent impact sphere radius on the ship." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
  $this nltl:hasThrusterIceImpactLoadCase ?case .
  ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
  FILTER NOT EXISTS {
    $this nltl:equivalentImpactSphereRadius ?radius .
  }
}
"""
    ] .

gen:TRF_114_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must reference the design ice block from TRAFICOM Table 6-3."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must reference the impact load case from TRAFICOM Table 6-16."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:node [
            sh:class nltl:thrusterBody ;
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applied thruster body must provide its externally assessed resistance capacity."
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must identify the thruster body to which it applies."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must provide the externally computed thruster ice impact load."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must provide the externally computed impact demand."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Evidence for the loaded impact area is required for the impact case."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "Non-hemispherical geometry requires correspondence evidence."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed thruster resistance capacity must be at least the externally computed impact demand." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:thrusterIceImpactDemand ?demand .
  ?demand qudt:numericValue ?demandValue ;
          qudt:unit unit:KiloN .
  $this nltl:thrusterImpactCaseAppliesToBody ?body .
  ?body nltl:thrusterResistanceCapacity ?capacity .
  ?capacity qudt:numericValue ?capacityValue ;
            qudt:unit unit:KiloN .
  FILTER (?capacityValue < ?demandValue)
}
"""
    ] .
</END_SHACL>
