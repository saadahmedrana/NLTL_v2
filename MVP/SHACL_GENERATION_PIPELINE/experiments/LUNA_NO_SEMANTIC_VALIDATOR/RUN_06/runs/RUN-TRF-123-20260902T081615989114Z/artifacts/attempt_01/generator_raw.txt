PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

    ?sfQuantity qudt:numericValue ?sfValue .

    ?yieldQuantity qudt:unit <http://qudt.org/vocab/unit/MegaPA> .

        datatype(?sfValue) = xsd:decimal &&
        datatype(?stressValue) = xsd:decimal &&
