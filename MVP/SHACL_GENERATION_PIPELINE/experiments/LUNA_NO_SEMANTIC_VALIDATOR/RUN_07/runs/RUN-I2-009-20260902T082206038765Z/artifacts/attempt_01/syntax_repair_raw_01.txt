    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-009-NormalFrameAngleShape> ;
        sh:message "The normal frame angle at the centre of the foremost sub-region shall be provided in degrees."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow-form applicability shall be true exactly when the buttock angle is greater than 0 degrees and less than 80 degrees and the normal frame angle is greater than 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability ?app .
                $this nltl:buttockAngle ?buttockQuantity .
                ?buttockQuantity qudt:numericValue ?buttockValue .
                ?buttockQuantity qudt:unit <http://qudt.org/vocab/unit/DEG> .
                $this nltl:normalFrameAngle ?frameQuantity .
                ?frameQuantity qudt:numericValue ?frameValue .
                ?frameQuantity qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (
                    (?app = "true"^^xsd:boolean &&
                     !(?buttockValue > 0 &&
                       ?buttockValue < 80 &&
                       ?frameValue > 10))
                    ||
                    (?app = "false"^^xsd:boolean &&
                     ?buttockValue > 0 &&
                     ?buttockValue < 80 &&
                     ?frameValue > 10)
                )
            }
        """
    ] .
