<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_022_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:node gen:I2_022_PlatingShape
    ] .

gen:I2_022_PlatingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:platingHullAreaClassification ;
        sh:in (
            nltl:bowIntermediateBottomHullArea
            nltl:midbodyBottomHullArea
            nltl:sternBottomHullArea
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each plating must identify one permitted I2.4.2 bottom hull area classification."
    ] ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each plating must provide its framing angle omega in decimal degrees."
    ] ;
    sh:property [
        sh:path nltl:platingSupportedByStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:minCount 1 ;
        sh:message "Each plating must identify its supporting structural member."
    ] ;
    sh:sparql [
        sh:message "Applicable transversely-framed bottom plating must provide the declared calculation inputs and the externally computed required net plate thickness with the specified quantity units." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:framingAngleOmega ?omegaQuantity .
    ?omegaQuantity qudt:numericValue ?omega .
    $this nltl:platingHullAreaClassification ?area .
    FILTER (
        datatype(?omega) = xsd:decimal &&
        ?omega >= "70.0"^^xsd:decimal &&
        ?area IN (
            nltl:bowIntermediateBottomHullArea,
            nltl:midbodyBottomHullArea,
            nltl:sternBottomHullArea
        )
    )
    FILTER (NOT EXISTS {
        ?ship nltl:hasPlating $this .
        ?ship nltl:averageIcePressure ?pressureQuantity .
        ?pressureQuantity qudt:numericValue ?pressure ;
            qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
        FILTER (datatype(?pressure) = xsd:decimal)

        ?ship nltl:peakPressureFactor ?peakFactorQuantity .
        ?peakFactorQuantity qudt:numericValue ?peakFactor ;
            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
        FILTER (datatype(?peakFactor) = xsd:decimal)

        ?ship nltl:yieldStrength ?yieldQuantity .
        ?yieldQuantity qudt:numericValue ?yield ;
            qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
        FILTER (datatype(?yield) = xsd:decimal)

        ?ship nltl:loadPatchHeight ?patchHeightQuantity .
        ?patchHeightQuantity qudt:numericValue ?patchHeight ;
            qudt:unit <http://qudt.org/vocab/unit/M> .
        FILTER (datatype(?patchHeight) = xsd:decimal)

        $this nltl:platingSupportedByStructuralMember ?member .
        ?member nltl:frameSpacing ?spacingQuantity .
        ?spacingQuantity qudt:numericValue ?spacing ;
            qudt:unit <http://qudt.org/vocab/unit/M> .
        FILTER (datatype(?spacing) = xsd:decimal)

        ?member nltl:selectedHullAreaFactor ?areaFactorQuantity .
        ?areaFactorQuantity qudt:numericValue ?areaFactor ;
            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
        FILTER (datatype(?areaFactor) = xsd:decimal)

        $this nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
        ?thicknessQuantity qudt:numericValue ?thickness ;
            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
        FILTER (datatype(?thickness) = xsd:decimal)
    })
}
"""
    ] .
<END_SHACL>
