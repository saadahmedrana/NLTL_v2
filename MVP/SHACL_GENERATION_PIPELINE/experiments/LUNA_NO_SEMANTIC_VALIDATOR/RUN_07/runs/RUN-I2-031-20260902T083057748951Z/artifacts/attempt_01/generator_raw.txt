<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node nltl:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the patch load in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the plastic collapse mechanism."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node nltl:I2_031_StructuralMemberShape
    ] .

nltl:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bottomStructureLocation
            nltl:sideStructureLocation
        ) ;
        sh:message "The structural member must be located in a bottom or side structure."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node nltl:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must provide its plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node nltl:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The structural member must provide the midspan load causing plastic collapse in kN."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:loadPatchOrientation ;
                        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation
            ] ;
            sh:property [
                sh:path nltl:transverseFrameOrientation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node nltl:I2_031_LoadCaseShape ;
        sh:minCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .
                FILTER (?demandValue > ?strengthValue)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The plastic strength must equal the midspan load that produces the plastic collapse mechanism." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                $this nltl:midspanPlasticCollapseLoad ?midspanQuantity .
                ?midspanQuantity qudt:numericValue ?midspanValue .
                FILTER (?strengthValue != ?midspanValue)
            }
        """
    ] .

nltl:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node nltl:I2_031_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural-member load case must provide the combined shear and bending demand in kN."
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue ;
        sh:maxCount 1
    ] .

nltl:I2_031_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
