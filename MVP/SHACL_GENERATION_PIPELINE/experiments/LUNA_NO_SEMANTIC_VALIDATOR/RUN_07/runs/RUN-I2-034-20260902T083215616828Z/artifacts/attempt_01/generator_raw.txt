<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_StructuralMemberShape
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state whether it is longitudinally oriented."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state its structural location."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation
            ] ;
            sh:property [
                sh:path nltl:plasticStrength ;
                sh:node gen:I2_034_ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable longitudinal side frame must provide its plastic strength."
            ] ;
            sh:property [
                sh:path nltl:midspanPlasticCollapseLoad ;
                sh:node gen:I2_034_ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable longitudinal side frame must provide its midspan plastic-collapse load."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:class nltl:loadCase ;
                sh:node gen:I2_034_LoadCaseShape
            ] ;
            sh:sparql [
                sh:message "The combined shear-and-bending demand must not exceed the member plastic strength or midspan plastic-collapse load." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:plasticStrength ?strengthNode .
                        ?strengthNode qudt:numericValue ?strength .
                        $this nltl:midspanPlasticCollapseLoad ?collapseNode .
                        ?collapseNode qudt:numericValue ?collapse .
                        $this nltl:hasStructuralMemberLoadCase ?case .
                        ?case nltl:combinedShearAndBendingDemand ?demandNode .
                        ?demandNode qudt:numericValue ?demand .
                        FILTER (?demand > ?strength || ?demand > ?collapse)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue false
            ]
        ]
        [
            sh:sparql [
                sh:message "A non-side structural member is outside the side-structure applicability branch." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:structuralLocation ?location .
                        FILTER (?location != nltl:sideStructureLocation)
                    }
                """
            ]
        ]
    ) .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_034_DemandQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented load case for an applicable member must provide its combined shear-and-bending demand."
    ] .

gen:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_034_ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
