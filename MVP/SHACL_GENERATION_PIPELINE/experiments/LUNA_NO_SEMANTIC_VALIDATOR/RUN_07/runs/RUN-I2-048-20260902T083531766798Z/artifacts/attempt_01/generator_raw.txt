<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node gs:SteelGradeRequirementCaseShape
    ] .

gs:SteelGradeRequirementCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify exactly one plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The requirement case must refer exactly to IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:maxInclusive 50 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "As-built plate thickness must be a single positive millimetre quantity not exceeding 50 mm."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each case must have exactly one supported steel material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each case must have exactly one supported steel strength category."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeE
            nltl:steelGradeAh
            nltl:steelGradeDh
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each case must identify exactly one supported actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:node [
            sh:property [
                sh:path nltl:steelGradeOrderRank ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:in (
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeE
            nltl:steelGradeAh
            nltl:steelGradeDh
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:node [
            sh:property [
                sh:path nltl:steelGradeOrderRank ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required steel grade must match the Table 8 selection for the case, and the actual grade must not be lower than it within the same strength category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasSteelGradeRequirementCase $this ;
                      nltl:polarClass ?pc .
                $this nltl:asBuiltPlateThickness ?thickness ;
                      nltl:steelMaterialClass ?material ;
                      nltl:steelStrengthCategory ?category ;
                      nltl:actualHullStructuralSteelGrade ?actual .
                ?thickness qudt:numericValue ?t .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                OPTIONAL {
                    $this nltl:requiredHullStructuralSteelGrade ?required .
                    ?required nltl:steelGradeOrderRank ?requiredRank .
                }

                BIND(
                    IF(
                        ?category = nltl:normalStrengthSteelCategory,
                        IF(
                            ?material = nltl:steelMaterialClassOne,
                            IF(
                                ?t <= 35,
                                1,
                                IF(
                                    ?pc IN (
                                        nltl:polarClassPc1,
                                        nltl:polarClassPc2,
                                        nltl:polarClassPc3,
                                        nltl:polarClassPc4,
                                        nltl:polarClassPc5
                                    ),
                                    IF(?t > 40, 3, 2),
                                    2
                                )
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassTwo,
                                IF(
                                    ?t <= 10,
                                    1,
                                    IF(
                                        ?t <= 25,
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3,
                                                nltl:polarClassPc4,
                                                nltl:polarClassPc5
                                            ),
                                            2,
                                            1
                                        ),
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3,
                                                nltl:polarClassPc4,
                                                nltl:polarClassPc5
                                            ),
                                            3,
                                            2
                                        )
                                    )
                                ),
                                IF(
                                    ?t <= 10,
                                    IF(
                                        ?pc IN (
                                            nltl:polarClassPc1,
                                            nltl:polarClassPc2,
                                            nltl:polarClassPc3,
                                            nltl:polarClassPc4,
                                            nltl:polarClassPc5
                                        ),
                                        3,
                                        1
                                    ),
                                    IF(
                                        ?t <= 35,
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3,
                                                nltl:polarClassPc4,
                                                nltl:polarClassPc5
                                            ),
                                            3,
                                            2
                                        ),
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3
                                            ),
                                            -1,
                                            IF(
                                                ?pc IN (
                                                    nltl:polarClassPc4,
                                                    nltl:polarClassPc5
                                                ),
                                                IF(?t <= 45, 3, -1),
                                                3
                                            )
                                        )
                                    )
                                )
                            )
                        ),
                        IF(
                            ?material = nltl:steelMaterialClassOne,
                            IF(
                                ?t <= 10,
                                1,
                                IF(
                                    ?pc IN (
                                        nltl:polarClassPc1,
                                        nltl:polarClassPc2,
                                        nltl:polarClassPc3,
                                        nltl:polarClassPc4,
                                        nltl:polarClassPc5
                                    ),
                                    IF(?t <= 35, 2, 3),
                                    IF(?t <= 35, 1, 2)
                                )
                            ),
                            IF(
                                ?material = nltl:steelMaterialClassTwo,
                                IF(
                                    ?t <= 10,
                                    1,
                                    IF(
                                        ?pc IN (
                                            nltl:polarClassPc1,
                                            nltl:polarClassPc2,
                                            nltl:polarClassPc3,
                                            nltl:polarClassPc4,
                                            nltl:polarClassPc5
                                        ),
                                        IF(?t <= 25, 2, 3),
                                        IF(?t <= 25, 1, 2)
                                    )
                                ),
                                IF(
                                    ?t <= 10,
                                    IF(
                                        ?pc IN (
                                            nltl:polarClassPc1,
                                            nltl:polarClassPc2,
                                            nltl:polarClassPc3,
                                            nltl:polarClassPc4,
                                            nltl:polarClassPc5
                                        ),
                                        3,
                                        1
                                    ),
                                    IF(
                                        ?t <= 35,
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3,
                                                nltl:polarClassPc4,
                                                nltl:polarClassPc5
                                            ),
                                            3,
                                            2
                                        ),
                                        IF(
                                            ?pc IN (
                                                nltl:polarClassPc1,
                                                nltl:polarClassPc2,
                                                nltl:polarClassPc3
                                            ),
                                            4,
                                            IF(
                                                ?pc IN (
                                                    nltl:polarClassPc4,
                                                    nltl:polarClassPc5
                                                ),
                                                IF(?t <= 45, 3, 4),
                                                3
                                            )
                                        )
                                    )
                                )
                            )
                        )
                    ) AS ?expectedRank
                )

                FILTER(
                    (?expectedRank = -1 && BOUND(?required)) ||
                    (?expectedRank != -1 && !BOUND(?required)) ||
                    (?expectedRank != -1 && BOUND(?required) &&
                     ?requiredRank != ?expectedRank) ||
                    (BOUND(?required) && ?actualRank < ?requiredRank)
                )
            }
        """
    ] .
</END_SHACL>
