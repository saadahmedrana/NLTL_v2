<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_060_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember
    ] ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The ship's structural members in way of cut-outs must not cause instability."
    ] ;
    sh:sparql [
        sh:message "Each linked structural member in way of a cut-out must identify the cut-out relationship and whether stiffening is necessary." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    FILTER (
        NOT EXISTS {
            ?member nltl:memberInWayOfCutout true
        }
        ||
        NOT EXISTS {
            ?member nltl:stiffeningNecessary ?necessary .
            FILTER (datatype(?necessary) = xsd:boolean)
        }
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "Structural stiffening must be recorded when stiffening is necessary for a member in way of a cut-out." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:hasCutout ?cutout .
    ?member nltl:stiffeningNecessary ?necessary .
    FILTER (
        datatype(?necessary) = xsd:boolean
        && ?necessary = true
    )
    FILTER NOT EXISTS {
        $this nltl:structuralStiffening ?stiffening
    }
}
"""
    ] .
<END_SHACL>
