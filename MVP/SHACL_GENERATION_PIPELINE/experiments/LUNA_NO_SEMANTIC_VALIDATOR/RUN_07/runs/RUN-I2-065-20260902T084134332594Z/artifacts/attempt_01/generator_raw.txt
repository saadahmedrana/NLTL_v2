<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:I2_065_ComplexReadinessShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A linked non-linear calculation case must provide the required ship-level acceptance margins." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER NOT EXISTS {
                    $this nltl:fractureMargin ?fracture .
                    ?fracture qudt:numericValue ?fractureValue ;
                              qudt:unit unit:UNITLESS .
                    FILTER (datatype(?fractureValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    $this nltl:majorBucklingMargin ?buckling .
                    ?buckling qudt:numericValue ?bucklingValue ;
                              qudt:unit unit:UNITLESS .
                    FILTER (datatype(?bucklingValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    $this nltl:yieldingMargin ?yielding .
                    ?yielding qudt:numericValue ?yieldingValue ;
                              qudt:unit unit:UNITLESS .
                    FILTER (datatype(?yieldingValue) = xsd:decimal)
                }
            }
        """
    ] .

gen:I2_065_ComplexReadinessCalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked non-linear calculation case must contain the declared inputs, acceptance evidence, and satisfied permanent-deformation assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this .
                $this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER NOT EXISTS {
                    $this nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
                FILTER NOT EXISTS {
                    $this nltl:minorPermanentDeformationAssessmentSatisfied ?satisfied .
                    FILTER (datatype(?satisfied) = xsd:boolean && ?satisfied = true)
                }
                FILTER NOT EXISTS {
                    $this nltl:permanentLateralDeformation ?lateral .
                    ?lateral qudt:numericValue ?lateralValue ;
                             qudt:unit unit:MilliM .
                    FILTER (datatype(?lateralValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    $this nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                    ?outOfPlane qudt:numericValue ?outOfPlaneValue ;
                                qudt:unit unit:MilliM .
                    FILTER (datatype(?outOfPlaneValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    $this nltl:relevantStructuralDimension ?dimension .
                    ?dimension qudt:numericValue ?dimensionValue ;
                              qudt:unit unit:MilliM .
                    FILTER (datatype(?dimensionValue) = xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
