<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007-OpenWater>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide an ice condition classification to determine whether the open-water rule applies."
    ] ;
    sh:property [
        sh:path nltl:seaIceConcentration ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "Sea ice concentration must have a decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Sea ice concentration must use the unitless unit."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceOfLandOriginPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open water, sea ice concentration must be less than 1 and ice of land origin must be absent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification
                    nltl:openWaterIceCondition .
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        ?concentration qudt:numericValue ?value ;
                                       qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value < 1
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent ?present .
                        FILTER (
                            datatype(?present) = xsd:boolean &&
                            ?present = false
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
