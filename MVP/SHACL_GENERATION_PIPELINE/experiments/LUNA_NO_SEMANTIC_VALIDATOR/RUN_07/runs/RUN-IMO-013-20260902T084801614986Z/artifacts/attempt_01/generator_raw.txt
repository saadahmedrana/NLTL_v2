<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_013_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:lowestMdlt ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must declare its lowest MDLT in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must declare whether it operates in low air temperature."
    ] ;
    sh:sparql [
        sh:message "shipOperatesInLowAirTemperature must be true when lowestMdlt is below -10 degrees Celsius and false otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:lowestMdlt ?quantity ;
                      nltl:shipOperatesInLowAirTemperature ?classification .
                ?quantity qudt:numericValue ?value .
                FILTER (
                    (?value < "-10"^^xsd:decimal && ?classification != true) ||
                    (?value >= "-10"^^xsd:decimal && ?classification != false)
                )
            }
        """
    ] .
<END_SHACL>
