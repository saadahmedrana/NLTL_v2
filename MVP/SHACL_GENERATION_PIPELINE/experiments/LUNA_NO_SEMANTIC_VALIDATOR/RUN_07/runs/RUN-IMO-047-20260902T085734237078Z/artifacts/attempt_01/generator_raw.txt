<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_047_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:message "Each linked essential engine must be represented as an essential engine." ;
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present." ;
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
            ]
        ] ;
        sh:message "Combustion-air temperature must be supplied as a single degree-Celsius quantity." ;
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
            ]
        ] ;
        sh:message "The manufacturer's minimum combustion-air temperature must be supplied as a single degree-Celsius quantity." ;
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG_C> ;
            ]
        ] ;
        sh:message "The manufacturer's maximum combustion-air temperature must be supplied as a single degree-Celsius quantity." ;
    ] ;
    sh:sparql [
        sh:message "Combustion-air temperature must be within the manufacturer's minimum and maximum limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureNode .
                ?temperatureNode qudt:numericValue ?temperature .

                $this nltl:manufacturerMinAirTemperature ?minimumNode .
                ?minimumNode qudt:numericValue ?minimum .

                $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                ?maximumNode qudt:numericValue ?maximum .

                FILTER (
                    ?temperature < ?minimum ||
                    ?temperature > ?maximum
                )
            }
        """ ;
    ] .
<END_SHACL>
