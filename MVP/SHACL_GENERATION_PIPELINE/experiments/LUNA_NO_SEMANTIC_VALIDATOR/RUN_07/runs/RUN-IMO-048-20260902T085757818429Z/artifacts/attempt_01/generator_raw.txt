<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-048> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:exposedMachineryOrFoundationMaterial ;
        sh:datatype xsd:string ;
        sh:message "An exposed machinery or foundation material must be recorded as a string."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:message "The approving authority must be the Administration or a recognized organization accepted by the Administration."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:message "The approval standard must be acceptable to the Organization or offer an equivalent level of safety."
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:in ( nltl:evidenceStateApproved ) ;
        sh:message "The material approval status must be Approved."
    ] ;
    sh:property [
        sh:path nltl:polarServiceTemperature ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Polar service temperature must be a decimal quantity expressed in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each exposed machinery or foundation material must have approved evidence with an authorized approving authority and standard, based on the polar service temperature." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:exposedMachineryOrFoundationMaterial ?material .
                FILTER NOT EXISTS {
                    $this nltl:approvingAuthority ?authority .
                    FILTER (
                        ?authority IN (
                            "Administration",
                            "recognized organization accepted by the Administration"
                        )
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:approvalStandard ?standard .
                    FILTER (
                        ?standard IN (
                            "standard acceptable to the Organization",
                            "another standard offering an equivalent level of safety"
                        )
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:materialApprovalStatus nltl:evidenceStateApproved .
                }
                FILTER NOT EXISTS {
                    $this nltl:polarServiceTemperature ?temperature .
                    ?temperature qudt:numericValue ?numericValue .
                    ?temperature qudt:unit unit:DEG_C .
                }
            }
        """
    ] .
<END_SHACL>
