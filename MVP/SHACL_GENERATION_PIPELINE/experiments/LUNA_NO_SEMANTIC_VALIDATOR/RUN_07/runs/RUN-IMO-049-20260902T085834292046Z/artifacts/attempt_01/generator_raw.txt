<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO049Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship must identify the applicable machinery component type or types."
    ] ;
    sh:or (
        gen:IMO049CategoryABShape
        gen:IMO049CategoryCShape
    ) .

gen:IMO049CategoryABShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B for this approval branch."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked machinery component must have an approved scantling approval record with an authorized authority and an Organization-acceptable or equivalent-safety standard." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard ?standard .
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER (
                        STR(?standard) = "standard acceptable to the Organization" ||
                        STR(?standard) = "another standard offering an equivalent level of safety"
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked machinery component approval record must use an approved status, an authorized authority, and an Organization-acceptable or equivalent-safety standard." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component nltl:hasApprovalRecord ?record .
                FILTER NOT EXISTS {
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard ?standard .
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER (
                        STR(?standard) = "standard acceptable to the Organization" ||
                        STR(?standard) = "another standard offering an equivalent level of safety"
                    )
                }
            }
        """
    ] .

gen:IMO049CategoryCShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category C for this branch."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Category C ship must be identified as ice-strengthened."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked machinery component must have an approved scantling approval record with an authorized authority, an acceptable ice-adequate standard, and operating ice type and concentration." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard "acceptable standard adequate for operating ice type and concentration"^^xsd:string ;
                            nltl:operatingIceType ?iceType ;
                            nltl:operatingIceConcentration ?concentration .
                    ?concentration qudt:numericValue ?numericValue ;
                                   qudt:unit qudt:UNITLESS .
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER ( DATATYPE(?iceType) = xsd:string )
                    FILTER ( DATATYPE(?numericValue) = xsd:decimal )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked Category C machinery component approval record must satisfy the approved status, authority, standard, and operating-ice evidence requirements." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component nltl:hasApprovalRecord ?record .
                FILTER NOT EXISTS {
                    ?record a nltl:approvalRecord ;
                            nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                            nltl:approvingAuthority ?authority ;
                            nltl:approvalStandard "acceptable standard adequate for operating ice type and concentration"^^xsd:string ;
                            nltl:operatingIceType ?iceType ;
                            nltl:operatingIceConcentration ?concentration .
                    ?concentration qudt:numericValue ?numericValue ;
                                   qudt:unit qudt:UNITLESS .
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER ( DATATYPE(?iceType) = xsd:string )
                    FILTER ( DATATYPE(?numericValue) = xsd:decimal )
                }
            }
        """
    ] .
<END_SHACL>
