<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_055_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Valve exposure must be provided as a boolean to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Valve type must be provided as a string to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Exposed isolating, pressure, or vacuum valves must have ice-accretion protection and remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:valveExposed ?exposed .
                FILTER (datatype(?exposed) = xsd:boolean && ?exposed = true)

                $this nltl:valveType ?type .
                FILTER (
                    datatype(?type) = xsd:string &&
                    ?type IN ("isolating", "pressure", "vacuum")
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:iceAccretionProtectionPresent ?protection .
                        FILTER (
                            datatype(?protection) = xsd:boolean &&
                            ?protection = true
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:continuouslyAccessible ?accessible .
                        FILTER (
                            datatype(?accessible) = xsd:boolean &&
                            ?accessible = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
