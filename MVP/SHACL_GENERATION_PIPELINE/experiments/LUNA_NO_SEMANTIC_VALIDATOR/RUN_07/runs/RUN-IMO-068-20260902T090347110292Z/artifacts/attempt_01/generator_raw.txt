@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO068ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:or (
        gen:IMO068IceOperationApplicable
        gen:IMO068DirectIceEvacuationApplicable
        gen:IMO068NoDirectIceApplicability
    ) .

gen:IMO068RequiredEvacuationConditions
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:safeEvacuationMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Safe evacuation means must be present when the requirement applies."
    ] ;
    sh:property [
        sh:path nltl:survivalEquipmentDeploymentApprovalStatus ;
        sh:minCount 1 ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:message "Survival equipment deployment approval status must be Approved when the requirement applies."
    ] .

gen:IMO068IceOperationApplicable
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:hasValue true
    ] ;
    sh:and ( gen:IMO068RequiredEvacuationConditions ) .

gen:IMO068DirectIceEvacuationApplicable
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:hasValue true
    ] ;
    sh:and ( gen:IMO068RequiredEvacuationConditions ) .

gen:IMO068NoDirectIceApplicability
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:hasValue false
    ] ;
    sh:not [
        a sh:NodeShape ;
        sh:property [
            sh:path nltl:shipOperatesInIceCoveredWaters ;
            sh:minCount 1 ;
            sh:hasValue true
        ]
    ] ;
    sh:not [
        a sh:NodeShape ;
        sh:property [
            sh:path nltl:directEvacuationOntoIceApplicable ;
            sh:minCount 1 ;
            sh:hasValue true
        ]
    ] .
