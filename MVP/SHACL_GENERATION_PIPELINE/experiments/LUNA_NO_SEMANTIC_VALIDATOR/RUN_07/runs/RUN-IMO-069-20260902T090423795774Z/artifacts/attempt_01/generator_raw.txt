<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_069_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:addedLifeSavingDeviceRequiresPower ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:addedLifeSavingDeviceRequiresPower ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:devicePowerSourceIndependentOfMainPower ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When an added life-saving device requires power, its power source must be independent of the ship's main power source." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:addedLifeSavingDeviceRequiresPower true .
                FILTER NOT EXISTS {
                    $this nltl:devicePowerSourceIndependentOfMainPower true .
                }
            }
        """
    ] .
<END_SHACL>
