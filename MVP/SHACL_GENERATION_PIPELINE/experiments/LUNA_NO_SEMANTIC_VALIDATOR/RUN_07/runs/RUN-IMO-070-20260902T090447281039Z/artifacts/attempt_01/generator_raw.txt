<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-070-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship type is required to determine whether the passenger-ship equipment rule applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A passenger ship must have a true size compatibility status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipType "passenger_ship" .
                FILTER NOT EXISTS {
                    $this nltl:sizeCompatibilityStatus ?status .
                    FILTER(datatype(?status) = xsd:boolean && ?status = true)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each person on board a passenger ship must have one assigned immersion suit or thermal protective aid with compatible sizing." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipType "passenger_ship" .
                $this nltl:hasPersonOnBoardMember ?person .
                FILTER NOT EXISTS {
                    ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                    ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment .
                    ?equipment rdf:type nltl:personalSurvivalEquipment .
                    ?equipment rdf:type ?equipmentType .
                    FILTER(?equipmentType IN (
                        nltl:immersionSuitEquipment,
                        nltl:thermalProtectiveAidEquipment
                    ))
                    $this nltl:sizeCompatibilityStatus ?status .
                    FILTER(datatype(?status) = xsd:boolean && ?status = true)
                }
            }
        """
    ] .
</urn:nltl:generated-shape:IMO-070-ShipShape>
<END_SHACL>
