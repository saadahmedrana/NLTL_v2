@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO081Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO081NotApplicableShape
        gen:IMO081ApplicableShape
    ) ;
    sh:message "For ships constructed on or after 1 January 2017 and ice strengthened, the required echo-sounding equipment alternative shall be satisfied." .

gen:IMO081NotApplicableShape a sh:NodeShape ;
    sh:or (
        gen:IMO081Pre2017Shape
        gen:IMO081NotIceStrengthenedShape
    ) .

gen:IMO081Pre2017Shape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2016-12-31"^^xsd:date
    ] .

gen:IMO081NotIceStrengthenedShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] .

gen:IMO081ApplicableShape a sh:NodeShape ;
    sh:and (
        gen:IMO081ApplicableInputsShape
        gen:IMO081EquipmentAlternativeShape
    ) .

gen:IMO081ApplicableInputsShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] .

gen:IMO081EquipmentAlternativeShape a sh:NodeShape ;
    sh:or (
        gen:IMO081IndependentDevicesAlternativeShape
        gen:IMO081DeviceTransducerAlternativeShape
    ) .

gen:IMO081IndependentDevicesAlternativeShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:independentEchoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2
    ] .

gen:IMO081DeviceTransducerAlternativeShape a sh:NodeShape ;
    sh:and (
        gen:IMO081EchoDeviceCountShape
        gen:IMO081IndependentTransducerCountShape
    ) .

gen:IMO081EchoDeviceCountShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:echoSoundingDeviceCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 1
    ] .

gen:IMO081IndependentTransducerCountShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:independentTransducerCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2
    ] .
