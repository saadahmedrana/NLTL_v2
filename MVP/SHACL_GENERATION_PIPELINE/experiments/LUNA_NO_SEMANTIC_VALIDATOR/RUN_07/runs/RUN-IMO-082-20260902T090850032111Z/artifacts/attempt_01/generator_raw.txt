<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-082-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Bridge configuration information is required to determine whether a clear astern view is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the bridge configuration requires an astern view, clear view astern status shall be compliant." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bridgeConfigurationType
                    nltl:asternViewRequiredBridgeConfiguration .
                FILTER NOT EXISTS {
                    $this nltl:clearViewAsternStatus
                        true^^xsd:boolean .
                }
            }
        """
    ] .
<END_SHACL>
