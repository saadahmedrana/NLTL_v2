gen:IMO_087_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DEG
            ]
        ]
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:message "Latitude applicability must be established as 80 degrees or less when the over-80-degree equipment rule is not applicable." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:plannedOrActualLatitude ?latitudeQuantity .
                        ?latitudeQuantity qudt:numericValue ?latitude .
                        FILTER (?latitude <= 80)
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "Latitude applicability must be established as over 80 degrees when the additional navigation equipment requirements apply." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:plannedOrActualLatitude ?latitudeQuantity .
                        ?latitudeQuantity qudt:numericValue ?latitude .
                        FILTER (?latitude > 80)
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:gnssCompassOrEquivalentCount ;
                sh:minCount 1 ;

                sh:minInclusive 1
            ] ;
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
    ) .
