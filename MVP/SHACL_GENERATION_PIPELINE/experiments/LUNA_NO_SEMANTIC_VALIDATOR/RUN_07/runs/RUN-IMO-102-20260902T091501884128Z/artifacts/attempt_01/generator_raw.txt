<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO102ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "The ship shall have exactly one controlled ice condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:message "The ship shall have exactly one controlled polar training ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO102CrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable crew member shall have a training record whose required and recorded levels match the 12.3.1 matrix and whose STCW qualification is valid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role .

                FILTER (
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType
                        ) &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    )
                    ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        ?shipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType,
                            nltl:otherShipPolarTrainingType
                        ) &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    )
                )

                FILTER NOT EXISTS {
                    ?crew nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?requiredLevel ;
                            nltl:trainingRecordLevel ?recordLevel ;
                            nltl:stcwQualificationValid true .

                    FILTER (
                        ?requiredLevel = ?recordLevel &&
                        (
                            (
                                ?ice = nltl:openWaterIceCondition &&
                                ?shipType IN (
                                    nltl:tankerPolarTrainingShipType,
                                    nltl:passengerShipPolarTrainingShipType
                                ) &&
                                ?requiredLevel = nltl:basicPolarTrainingLevel
                            )
                            ||
                            (
                                ?ice = nltl:otherWatersIceCondition &&
                                ?role IN (
                                    nltl:masterPolarTrainingRole,
                                    nltl:chiefMatePolarTrainingRole
                                ) &&
                                ?requiredLevel = nltl:advancedPolarTrainingLevel
                            )
                            ||
                            (
                                ?ice = nltl:otherWatersIceCondition &&
                                ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                                ?requiredLevel = nltl:basicPolarTrainingLevel
                            )
                        )
                    )
                }
            }
        """
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:message "Each represented crew member shall have exactly one controlled polar training role."
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] .

gen:IMO102TrainingRecordShape
    a sh:NodeShape ;
    sh:class nltl:polarTrainingRecord ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A polar training record shall state one controlled required training level."
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:message "A polar training record shall state one controlled recorded training level."
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A represented polar training record shall state whether the STCW qualification is valid."
    ] .
<END_SHACL>
