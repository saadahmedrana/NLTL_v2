<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-105-Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must identify its operating area so Arctic-water applicability can be determined."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of oil or oily mixtures into the sea must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea "Arctic_waters"^^xsd:string .
                FILTER NOT EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea true .
                }
                FILTER NOT EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea false .
                }
            }
        """
    ] .
</urn:nltl:generated-shape:IMO-105-Static>
<END_SHACL>
