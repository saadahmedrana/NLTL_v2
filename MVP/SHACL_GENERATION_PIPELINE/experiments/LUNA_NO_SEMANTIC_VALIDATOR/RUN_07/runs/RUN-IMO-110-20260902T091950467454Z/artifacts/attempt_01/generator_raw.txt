<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_110_ShipStaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable category A or B oil tanker constructed on or after 1 January 2017 with deadweight below 5,000 tonnes, both MARPOL cargo-tank protection statuses must be compliant." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:deadweight ?deadweight ;
                      nltl:constructionDate ?constructionDate ;
                      a nltl:oilTanker .

                ?deadweight qudt:numericValue ?mass ;
                            qudt:unit unit:TON_Metric .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                    && ?shipType = "oil_tanker"^^xsd:string
                    && datatype(?mass) = xsd:decimal
                    && ?mass < "5000"^^xsd:decimal
                    && datatype(?constructionDate) = xsd:date
                    && ?constructionDate >= "2017-01-01"^^xsd:date
                )

                FILTER (
                    !EXISTS {
                        $this nltl:entireCargoTankLengthDoubleBottomProtectionStatus true
                    }
                    ||
                    !EXISTS {
                        $this nltl:wingTankProtectionStatus true
                    }
                )
            }
        """
    ] .
<END_SHACL>
