<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_114_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB nltl:polarShipCategoryC ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ship category must be one of the approved Polar Code categories."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single valid construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the carried noxious liquid substance requires a type 3 tank."
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single cargo tank type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:administrationCarriageApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:evidenceStateApproved
            nltl:evidenceStateDraft
            nltl:evidenceStateExpired
            nltl:evidenceStateRejected
            nltl:evidenceStateRevoked
            nltl:evidenceStateSubmitted
            nltl:evidenceStateUnderReview
        ) ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:applicableCertificatePolarOperationEntryPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable category A or B ship constructed on or after 1 January 2017 carrying NLS requiring a type 3 tank in cargo tanks of type 3, Administration approval and the applicable certificate polar-operation entry are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:constructionDate ?constructionDate ;
          nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 ;
          nltl:cargoTankType ?cargoTankType .
    FILTER (
        ?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB)
        && ?constructionDate >= "2017-01-01"^^xsd:date
        && ?requiresType3 = true
        && ?cargoTankType = "3"^^xsd:string
    )
    {
        FILTER NOT EXISTS {
            $this nltl:administrationCarriageApprovalStatus nltl:evidenceStateApproved .
        }
    }
    UNION
    {
        FILTER NOT EXISTS {
            $this nltl:applicableCertificatePolarOperationEntryPresent true .
        }
    }
}
""" ;
    ] .
<END_SHACL>
