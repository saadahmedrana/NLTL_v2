<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:StaticChapter9Dash1ApplicabilityShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Chapter 9-1 applicability does not match the ship type, dimensions, tonnage, and trade status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                OPTIONAL {
                    $this nltl:shipType ?type .
                }
                OPTIONAL {
                    $this nltl:lengthOverall ?lengthQuantity .
                    ?lengthQuantity qudt:numericValue ?length .
                }
                OPTIONAL {
                    $this nltl:grossTonnage ?tonnageQuantity .
                    ?tonnageQuantity qudt:numericValue ?tonnage .
                }
                OPTIONAL {
                    $this nltl:engagedInTrade ?trade .
                }
                OPTIONAL {
                    $this nltl:polarCodeChapter9Dash1Applicable ?applicable .
                }

                BIND(
                    (
                        (
                            IF(BOUND(?type), ?type = "fishing_vessel", false)
                            &&
                            IF(BOUND(?length), ?length >= 24, false)
                        )
                        ||
                        (
                            IF(BOUND(?type), ?type = "pleasure_yacht", false)
                            &&
                            IF(BOUND(?tonnage), ?tonnage >= 300, false)
                            &&
                            IF(BOUND(?trade), ?trade = false, false)
                        )
                        ||
                        (
                            IF(BOUND(?type), ?type = "cargo_ship", false)
                            &&
                            IF(BOUND(?tonnage), ?tonnage >= 300, false)
                            &&
                            IF(BOUND(?tonnage), ?tonnage < 500, false)
                        )
                    )
                    AS ?derivedApplicable
                )

                FILTER(
                    (
                        ?derivedApplicable = true
                        &&
                        (!BOUND(?applicable) || ?applicable != true)
                    )
                    ||
                    (
                        BOUND(?applicable)
                        &&
                        ?applicable = true
                        &&
                        ?derivedApplicable = false
                    )
                )
            }
        """
    ] .
<END_SHACL>
