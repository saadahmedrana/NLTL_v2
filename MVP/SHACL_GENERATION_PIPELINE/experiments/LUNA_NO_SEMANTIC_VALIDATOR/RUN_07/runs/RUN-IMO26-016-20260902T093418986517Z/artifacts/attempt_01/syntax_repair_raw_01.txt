@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_016_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's polar-waters operating status is required and must be a single boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship operating in polar waters, ship type and the applicable particulars must be present, and polar Code chapter 11-1 applicability must match the stated ship type, dimensions, gross tonnage, and trade status." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:shipOperatesInPolarWaters true .
    FILTER (
        NOT EXISTS {
            $this nltl:shipType ?type .
        }
        ||
        NOT EXISTS {
            $this nltl:polarCodeChapter11Dash1Applicable ?app .
            FILTER (datatype(?app) = xsd:boolean)
        }
        ||
        (
            EXISTS {
                $this nltl:shipType "fishing_vessel" .
            }
            &&
            NOT EXISTS {
                $this nltl:lengthOverall ?lengthValue .
                ?lengthValue qudt:numericValue ?length .
                ?lengthValue qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (datatype(?length) = xsd:decimal && ?length >= 24)
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "pleasure_yacht" .
            }
            &&
            NOT EXISTS {
                $this nltl:grossTonnage ?tonnageValue .
                ?tonnageValue qudt:numericValue ?tonnage .
                ?tonnageValue qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (datatype(?tonnage) = xsd:decimal && ?tonnage >= 300)
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "pleasure_yacht" .
            }
            &&
            NOT EXISTS {
                $this nltl:engagedInTrade false .
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "cargo_ship" .
            }
            &&
            NOT EXISTS {
                $this nltl:grossTonnage ?cargoTonnageValue .
                ?cargoTonnageValue qudt:numericValue ?cargoTonnage .
                ?cargoTonnageValue qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (
                    datatype(?cargoTonnage) = xsd:decimal &&
                    ?cargoTonnage >= 300 &&
                    ?cargoTonnage < 500
                )
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "fishing_vessel" .
                $this nltl:lengthOverall ?lengthValue2 .
                ?lengthValue2 qudt:numericValue ?length2 .
                ?lengthValue2 qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (datatype(?length2) = xsd:decimal && ?length2 >= 24)
            }
            &&
            NOT EXISTS {
                $this nltl:polarCodeChapter11Dash1Applicable true .
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "pleasure_yacht" .
                $this nltl:grossTonnage ?pleasureTonnageValue .
                ?pleasureTonnageValue qudt:numericValue ?pleasureTonnage .
                ?pleasureTonnageValue qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                $this nltl:engagedInTrade false .
                FILTER (datatype(?pleasureTonnage) = xsd:decimal && ?pleasureTonnage >= 300)
            }
            &&
            NOT EXISTS {
                $this nltl:polarCodeChapter11Dash1Applicable true .
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType "cargo_ship" .
                $this nltl:grossTonnage ?cargoTonnageValue2 .
                ?cargoTonnageValue2 qudt:numericValue ?cargoTonnage2 .
                ?cargoTonnageValue2 qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (
                    datatype(?cargoTonnage2) = xsd:decimal &&
                    ?cargoTonnage2 >= 300 &&
                    ?cargoTonnage2 < 500
                )
            }
            &&
            NOT EXISTS {
                $this nltl:polarCodeChapter11Dash1Applicable true .
            }
        )
        ||
        (
            EXISTS {
                $this nltl:shipType ?nonApplicableType .
                FILTER (
                    ?nonApplicableType != "fishing_vessel" &&
                    ?nonApplicableType != "pleasure_yacht" &&
                    ?nonApplicableType != "cargo_ship"
                )
            }
            &&
            NOT EXISTS {
                $this nltl:polarCodeChapter11Dash1Applicable false .
            }
        )
    )
}
"""
    ] .
</END_SHACL>
