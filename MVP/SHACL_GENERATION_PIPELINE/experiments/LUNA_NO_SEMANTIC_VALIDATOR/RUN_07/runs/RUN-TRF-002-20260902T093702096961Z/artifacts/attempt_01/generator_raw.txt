<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-002> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2017 Ice Class Regulations."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships contracted on or after 1 January 2019 and before 5 July 2021 shall identify the 2017 Ice Class Regulations as the applicable edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    ?date >= "2019-01-01"^^xsd:date &&
                    ?date < "2021-07-05"^^xsd:date
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2017 .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The 2017 Ice Class Regulations shall not be identified as applicable to ships contracted outside the specified date interval." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                $this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRegulationEdition2017 .
                FILTER (
                    ?date < "2019-01-01"^^xsd:date ||
                    ?date >= "2021-07-05"^^xsd:date
                )
            }
        """
    ] .
<END_SHACL>
