<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship shall have one permitted Finnish-Swedish ice-class value."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "2003-09-01"^^xsd:date ;
                sh:message "An IA Super or IA ship shall have a construction-stage date on or after 1 September 2003 when this exception applies."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:maxExclusive "2003-09-01"^^xsd:date ;
                sh:message "An applicable IA Super or IA ship shall have a construction-stage date before 1 September 2003."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable IA Super or IA ship shall have a delivery date."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable IA Super or IA ship shall have an assessment date."
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:engineOutputComplianceMethodValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "An applicable IA Super or IA ship shall comply using section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA Super or IA ship shall comply no later than 1 January in the year twenty years after its delivery year." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .

                FILTER (
                    ?iceClass = nltl:iceClassIaSuper ||
                    ?iceClass = nltl:iceClassIa
                )

                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        MONTH(?constructionStageDate) < 9
                    ) ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        MONTH(?constructionStageDate) = 9 &&
                        DAY(?constructionStageDate) < 1
                    )
                )

                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        MONTH(?assessmentDate) > 1
                    ) ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        MONTH(?assessmentDate) = 1 &&
                        DAY(?assessmentDate) > 1
                    )
                )
            }
        """
    ] .
