<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_IceWaterlineShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline shall equal the envelope of the highest intended ice-operating waterline points at every longitudinal position, with no additional positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                {
                    {
                        SELECT ?s ?x (MAX(?iy) AS ?maxY)
                        WHERE {
                            ?s nltl:hasIntendedIceOperatingWaterline ?iw .
                            ?iw nltl:hasWaterlineProfilePoint ?ip .
                            ?ip nltl:longitudinalPosition ?ixq ;
                                nltl:verticalCoordinate ?iyq .
                            ?ixq qudt:numericValue ?x ;
                                qudt:unit qudt:M .
                            ?iyq qudt:numericValue ?iy ;
                                qudt:unit qudt:M .
                        }
                        GROUP BY ?s ?x
                    }
                    FILTER NOT EXISTS {
                        ?s nltl:hasUpperIceWaterline ?uw .
                        ?uw nltl:hasWaterlineProfilePoint ?up .
                        ?up nltl:longitudinalPosition ?uxq ;
                            nltl:verticalCoordinate ?uyq .
                        ?uxq qudt:numericValue ?ux ;
                            qudt:unit qudt:M .
                        ?uyq qudt:numericValue ?uy ;
                            qudt:unit qudt:M .
                        FILTER (?ux = ?x && ?uy = ?maxY)
                    }
                }
                UNION
                {
                    SELECT ?s
                    WHERE {
                        ?s nltl:hasUpperIceWaterline ?uw .
                        ?uw nltl:hasWaterlineProfilePoint ?up .
                        ?up nltl:longitudinalPosition ?uxq ;
                            nltl:verticalCoordinate ?uyq .
                        ?uxq qudt:numericValue ?ux ;
                            qudt:unit qudt:M .
                        ?uyq qudt:numericValue ?uy ;
                            qudt:unit qudt:M .
                        FILTER NOT EXISTS {
                            ?s nltl:hasIntendedIceOperatingWaterline ?iw2 .
                            ?iw2 nltl:hasWaterlineProfilePoint ?ip2 .
                            ?ip2 nltl:longitudinalPosition ?ixq2 .
                            ?ixq2 qudt:numericValue ?ix2 ;
                                qudt:unit qudt:M .
                            FILTER (?ix2 = ?ux)
                        }
                    }
                }
            }
        """
    ] .

gen:TRF_011_IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gen:TRF_011_ProfilePointShape
    ] .

gen:TRF_011_ProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
