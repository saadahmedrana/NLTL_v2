<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix gen: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF016ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF016NavigatingInIceShape
        gen:TRF016NotNavigatingInIceShape
    ) .

gen:TRF016NotNavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .

gen:TRF016NavigatingInIceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node gen:TRF016LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node gen:TRF016LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node gen:TRF016LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node gen:TRF016LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node gen:TRF016MassQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:waterlineDisplacement ;
                sh:node gen:TRF016MassQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, draught amidships must reach the LIWL draught and forward draught must reach the applicable UIWL level-ice-thickness threshold." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true ;
                      nltl:draughtAmidships ?amidships ;
                      nltl:liwlDraughtAmidships ?liwl ;
                      nltl:forwardDraught ?forward ;
                      nltl:levelIceThickness ?thickness ;
                      nltl:hasUpperIceWaterline ?uiwl .
                ?amidships qudt:numericValue ?amidshipsValue .
                ?liwl qudt:numericValue ?liwlValue .
                ?forward qudt:numericValue ?forwardValue .
                ?thickness qudt:numericValue ?thicknessValue .
                ?uiwl nltl:waterlineDisplacement ?displacement .
                ?displacement qudt:numericValue ?displacementValue .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?otherUiwl .
                    ?otherUiwl nltl:waterlineDisplacement ?otherDisplacement .
                    ?otherDisplacement qudt:numericValue ?otherDisplacementValue .
                    FILTER (?otherDisplacementValue > ?displacementValue)
                }
                BIND (
                    IF(
                        ((2.0 + (0.00025 * ?displacementValue)) * ?thicknessValue)
                        < (4.0 * ?thicknessValue),
                        ((2.0 + (0.00025 * ?displacementValue)) * ?thicknessValue),
                        (4.0 * ?thicknessValue)
                    )
                    AS ?forwardMinimum
                )
                FILTER (
                    (?amidshipsValue < ?liwlValue) ||
                    (?forwardValue < ?forwardMinimum)
                )
            }
        """
    ] .

gen:TRF016BallastTankShape
    a sh:NodeShape ;
    sh:targetClass nltl:ballastTank ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF016BallastTankNotApplicableAbove
        gen:TRF016BallastTankNotNeeded
        gen:TRF016BallastTankProtected
    ) .

gen:TRF016BallastTankNotApplicableAbove
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:hasValue false
    ] .

gen:TRF016BallastTankNotNeeded
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:hasValue false
    ] .

gen:TRF016BallastTankProtected
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:situatedAboveLowerIceWaterline ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:freezingPreventionPresent ;
        sh:hasValue true
    ] .

gen:TRF016LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:TRF016MassQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric>
    ] .
<END_SHACL>
