gs:TRF020Shape
    a sh:NodeShape ;

    sh:property [

    ] ;
    sh:or (
        gs:TRF020ApplicableBranch
        gs:TRF020NonApplicableBranch
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be at least both externally calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrNode .
                ?mcrNode qudt:numericValue ?mcr .
                ?mcrNode qudt:unit unit:KiloW .
                $this nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerNode .
                ?lowerNode qudt:numericValue ?lowerPower .
                ?lowerNode qudt:unit unit:KiloW .
                $this nltl:minimumRequiredPowerAtUpperIceWaterline ?upperNode .
                ?upperNode qudt:numericValue ?upperPower .
                ?upperNode qudt:unit unit:KiloW .
                FILTER (?mcr < ?lowerPower || ?mcr < ?upperPower)
            }
        """
    ] .
