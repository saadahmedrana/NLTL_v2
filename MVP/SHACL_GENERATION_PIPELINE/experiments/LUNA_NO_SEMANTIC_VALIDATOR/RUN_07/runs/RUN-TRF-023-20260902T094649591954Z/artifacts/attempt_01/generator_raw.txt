<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF023ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable ice class regulation edition is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship ice class is required to determine applicability."
    ] ;
    sh:or (
        gen:TRF023ApplicableShape
        gen:TRF023NonApplicableShape
    ) .

gen:TRF023ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:hasValue nltl:iceClassRuleEdition1985
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC under the 1985 ice class regulations, maximum continuous rating power must be at least the required power under the 1985 rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrNode .
                $this nltl:requiredPowerUnder1985Rules ?requiredNode .
                ?mcrNode qudt:numericValue ?mcr .
                ?requiredNode qudt:numericValue ?required .
                FILTER (?mcr < ?required)
            }
        """
    ] .

gen:TRF023NonApplicableShape
    a sh:NodeShape ;
    sh:or (
        gen:TRF023OtherEditionShape
        gen:TRF023OtherIceClassShape
    ) .

gen:TRF023OtherEditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:in (
            nltl:iceClassRegulationEdition2002
            nltl:iceClassRegulationEdition2008
            nltl:iceClassRegulationEdition2010
            nltl:iceClassRegulationEdition2017
            nltl:iceClassRegulationEdition2021
        )
    ] .

gen:TRF023OtherIceClassShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .
<END_SHACL>
