<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF040Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must declare one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designOperationSpeedInIce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:openWaterServiceSpeed ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:sidescuttleInIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "Sidescuttles shall not be situated in the ice belt."
    ] ;
    sh:property [
        sh:path nltl:weatherDeckBelowIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The conditional ice-belt strengthening requirements are not satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                {
                    $this nltl:iceClass nltl:iceClassIaSuper .
                    FILTER (
                        NOT EXISTS {
                            $this nltl:forefootShellStrengtheningStatus ?forefoot ;
                                 nltl:bowRegionEquivalent ?bow .
                            FILTER (
                                datatype(?forefoot) = xsd:boolean &&
                                datatype(?bow) = xsd:boolean &&
                                ?forefoot = ?bow
                            )
                        }
                    )
                }
                UNION
                {
                    $this nltl:iceClass ?upperClass ;
                         nltl:openWaterServiceSpeed ?speed .
                    ?speed qudt:numericValue ?speedValue .
                    FILTER (
                        (?upperClass = nltl:iceClassIaSuper ||
                         ?upperClass = nltl:iceClassIa) &&
                        ?speedValue >= 18 &&
                        NOT EXISTS {
                            $this nltl:upperBowIceBeltStrengtheningStatus ?upper ;
                                 nltl:midbodyEquivalent ?midbody .
                            FILTER (
                                datatype(?upper) = xsd:boolean &&
                                datatype(?midbody) = xsd:boolean &&
                                ?upper = ?midbody
                            )
                        }
                    )
                }
                UNION
                {
                    $this nltl:weatherDeckBelowIceBeltUpperLimit true .
                    FILTER (
                        NOT EXISTS {
                            $this nltl:bulwarkStrengthStatus true ;
                                 nltl:freeingPortStrengthStatus true
                        }
                    )
                }
            }
        """
    ] .
<END_SHACL>
