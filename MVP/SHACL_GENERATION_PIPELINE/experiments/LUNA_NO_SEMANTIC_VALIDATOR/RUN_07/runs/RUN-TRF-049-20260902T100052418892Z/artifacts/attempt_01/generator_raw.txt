<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF049Shape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        gen:TRF049IceStrengthenedFrameShape
    ) .

gen:TRF049IceStrengthenedFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A frame governed by TRF-049 must identify that it is within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF049MillimetreLengthShape ;
        sh:message "An applicable frame must provide its web thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gen:TRF049FrameAttachmentShape ;
        sh:message "An applicable frame must be effectively attached through a frame attachment record."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket ) ;
                        sh:minCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path ( nltl:hasFrameAttachment nltl:similarAttachmentConstructionPresent ) ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue true
                    ]
                ]
            )
        )
    ) ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:message "A frame running through a supporting structure must have both web-plate sides connected."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A longitudinal frame must have its linked supporting web frames and bulkheads attached by brackets." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

            SELECT $this
            WHERE {
                $this rdf:type nltl:longitudinalFrame .
                {
                    $this nltl:hasSupportingWebFrame ?supporting .
                }
                UNION
                {
                    $this nltl:hasSupportingBulkhead ?supporting .
                }
                FILTER NOT EXISTS {
                    ?supporting nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bracket thickness must be at least the thickness of the frame web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameQuantity .
                ?frameQuantity qudt:numericValue ?frameValue .
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketQuantity .
                ?bracketQuantity qudt:numericValue ?bracketValue .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .

gen:TRF049FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame attachment must be confirmed as effective."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node gen:TRF049ConnectionBracketShape
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .

gen:TRF049ConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF049MillimetreLengthShape ;
        sh:message "An installed connection bracket must provide its thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened against buckling."
    ] .

gen:TRF049MillimetreLengthShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

<END_SHACL>
