<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceStringerTRF053Shape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:IcePressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice pressure is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:DesignIceLoadHeightQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design ice load height is required in metres."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:SpanQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Stringer span is required in metres."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:FrameMomentFactorQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Frame moment factor m is required and must be dimensionless."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:YieldStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Yield strength is required in MPa."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:DesignLineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Design line load is required in kN/m."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Section modulus is required in cm3."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:RequiredShearAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Required shear area is required in cm2."
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, and required shear area must satisfy the TRF-053 formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?areaNode .

                ?pressureNode <http://qudt.org/schema/qudt/numericValue> ?pressure .
                ?heightNode <http://qudt.org/schema/qudt/numericValue> ?height .
                ?spanNode <http://qudt.org/schema/qudt/numericValue> ?span .
                ?mNode <http://qudt.org/schema/qudt/numericValue> ?m .
                ?yieldNode <http://qudt.org/schema/qudt/numericValue> ?yield .
                ?lineNode <http://qudt.org/schema/qudt/numericValue> ?line .
                ?zNode <http://qudt.org/schema/qudt/numericValue> ?z .
                ?areaNode <http://qudt.org/schema/qudt/numericValue> ?area .

                BIND (?pressure * ?height AS ?product)
                BIND (IF(?product < 0.15, 0.15, ?product) AS ?lineMnPerM)
                BIND (?lineMnPerM * 1000 AS ?expectedLine)
                BIND (0.9 * 1.8 * ?lineMnPerM * ?span * ?span /
                      (?m * ?yield) * 1000000 AS ?expectedZ)
                BIND (1.7320508075688772 * 0.9 * 1.8 * 1.2 * ?lineMnPerM * ?span /
                      (2 * ?yield) * 10000 AS ?expectedArea)

                FILTER (
                    ABS(?line - ?expectedLine) >
                    (0.000001 * ABS(?expectedLine) + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                    (0.000001 * ABS(?expectedZ) + 0.000000001)
                    ||
                    ABS(?area - ?expectedArea) >
                    (0.000001 * ABS(?expectedArea) + 0.000000001)
                )
            }
        """
    ] .
