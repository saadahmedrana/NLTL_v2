<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice pressure in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A design ice load height in metres is required."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A span in metres is required."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A dimensionless frame moment factor m is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A yield strength in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A distance to the ice belt in metres is required."
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A stringer span in metres is required."
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:ForcePerLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A design line load in kN/m is required."
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An outside-ice-belt factor is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A section modulus in cm3 is required."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A required shear area in cm2 is required."
    ] ;
    sh:sparql [
        sh:message "The reported design line load, outside-ice-belt factor, section modulus, or required shear area does not satisfy the specified calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?lineNode qudt:numericValue ?line .
                ?factorNode qudt:numericValue ?factor .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF((?p * ?h) > 0.15, (?p * ?h), 0.15) AS ?loadMNPerM)
                BIND(?loadMNPerM * 1000 AS ?expectedLine)
                BIND(1 - (?hs / ?ls) AS ?expectedFactor)
                BIND(0.80 * 1.8 * ?loadMNPerM * (?ls * ?ls) /
                     (?m * ?sigma) * (1 - (?hs / ?ls)) * 1000000
                     AS ?expectedZ)
                BIND(1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                     ?loadMNPerM * ?ls / (2 * ?sigma) *
                     (1 - (?hs / ?ls)) * 10000
                     AS ?expectedA)

                FILTER(
                    ABS(?line - ?expectedLine) >
                        (ABS(?expectedLine) * 0.000001 + 0.000000001)
                    ||
                    ABS(?factor - ?expectedFactor) >
                        (ABS(?expectedFactor) * 0.000001 + 0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                        (ABS(?expectedZ) * 0.000001 + 0.000000001)
                    ||
                    ABS(?a - ?expectedA) >
                        (ABS(?expectedA) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
