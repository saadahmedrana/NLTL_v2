@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF070Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:in ( "azimuthing" "fixed" ) ;
        sh:minCount 1 ;
        sh:message "A main-propulsion thruster type of azimuthing or fixed is required."
    ] ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The applicable thruster shall be identified as main propulsion."
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:maxCount 1 ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The propeller/ice interaction load shall be provided in kN."
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:maxCount 1 ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The thruster-body/ice interaction load shall be provided in kN."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:maxCount 1 ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The local ice pressure shall be provided in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:class nltl:designCondition ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:designConditionLocalStrengthCapacity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:maxCount 1 ;
                        sh:minCount 1
                    ]
                ] ;
                sh:message "The externally assessed local-strength capacity shall be provided in kN."
            ]
        ] ;
        sh:message "At least one design condition with local-strength capacity is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For every design condition, the local-strength capacity shall be at least the local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:localIcePressure ?pressure .
                ?pressure qudt:numericValue ?pressureValue .
                $this nltl:hasDesignCondition ?condition .
                ?condition nltl:designConditionLocalStrengthCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .
