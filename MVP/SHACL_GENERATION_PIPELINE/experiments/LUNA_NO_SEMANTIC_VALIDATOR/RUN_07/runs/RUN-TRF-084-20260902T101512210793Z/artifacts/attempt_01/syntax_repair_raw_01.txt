<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthInMetresShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF_084_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:node gen:LengthInMetresShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller with a known pitch value, the propeller pitch at MCR in bollard condition must be provided."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:node gen:LengthInMetresShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:node gen:LengthInMetresShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "For a CP propeller whose pitch value is not known, the bollard-condition pitch must equal 0.7 times the free-running pitch within a relative tolerance of 1e-6 and an absolute tolerance of 1e-6 m." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerType ?type ;
                              nltl:pitchValueKnown ?known ;
                              nltl:propellerPitchAtMcrFreeRunning ?freeQuantity ;
                              nltl:propellerPitchAtMcrBollard ?bollardQuantity .

                        FILTER (?type = "CP" && ?known = false)

                        ?freeQuantity qudt:numericValue ?freeValue ;
                                      qudt:unit unit:M .
                        ?bollardQuantity qudt:numericValue ?bollardValue ;
                                         qudt:unit unit:M .

                        BIND ("0.7"^^xsd:decimal * ?freeValue AS ?expected)
                        BIND (
                            IF(
                                ABS(?expected) * "0.000001"^^xsd:decimal > "0.000001"^^xsd:decimal,
                                ABS(?expected) * "0.000001"^^xsd:decimal,
                                "0.000001"^^xsd:decimal
                            )
                            AS ?tolerance
                        )

                        FILTER (ABS(?bollardValue - ?expected) > ?tolerance)
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "This requirement is not applicable because the propeller is not a CP propeller." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerType ?type .
                        FILTER (?type != "CP")
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
