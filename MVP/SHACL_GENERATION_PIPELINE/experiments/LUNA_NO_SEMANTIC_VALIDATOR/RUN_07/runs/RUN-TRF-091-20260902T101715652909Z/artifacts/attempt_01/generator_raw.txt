<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<nltl:generated-shape:TRF-091-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:message "The ship shall identify an ice milling load sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
                sh:message "The propulsion load case shall be used for propulsion line strength evaluation."
            ]
        ] ;
        sh:message "The ship shall identify a propulsion load case for the required strength evaluation."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice milling load sequence shall not be used for propulsion system stalling analyses." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:loadCasePurpose nltl:stallingAnalysisPurposeValue .
            }
        """
    ] .
<END_SHACL>
