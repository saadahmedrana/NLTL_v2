<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF123ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:node <urn:nltl:generated-shape:TRF123OccasionalCaseShape> ;
        sh:message "Each represented occasional-force load case must provide the required applicability evidence and conformance inputs."
    ] .

<urn:nltl:generated-shape:TRF123OccasionalCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The occasional-force load case must provide one dimensionless component yield safety factor."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:class nltl:shipComponent ;
                sh:node <urn:nltl:generated-shape:TRF123ComponentShape> ;
                sh:minCount 1 ;
                sh:message "A non-excluded occasional-force case must identify the assessed component."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a non-excluded occasional-force case, the externally calculated component stress multiplied by the safety factor must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:propellerBladeExcludedFromOccasionalForceScope ?excluded .
                FILTER (?excluded = false)

                $this nltl:componentYieldSafetyFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factor .

                $this nltl:occasionalForceCaseAssessedComponent ?component .
                ?component nltl:occasionalForceComponentStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress .

                ?component nltl:hasMaterialProperties ?material .
                ?material nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .

                FILTER (!(?stress * ?factor <= ?yield))
            }
        """
    ] .

<urn:nltl:generated-shape:TRF123ComponentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The assessed component must provide one occasional-force stress quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:class nltl:materialProperties ;
        sh:node <urn:nltl:generated-shape:TRF123MaterialShape> ;
        sh:minCount 1 ;
        sh:message "The assessed component must identify its material properties."
    ] .

<urn:nltl:generated-shape:TRF123MaterialShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The component material properties must provide one material yield strength quantity in MPa."
    ] .
<END_SHACL>
