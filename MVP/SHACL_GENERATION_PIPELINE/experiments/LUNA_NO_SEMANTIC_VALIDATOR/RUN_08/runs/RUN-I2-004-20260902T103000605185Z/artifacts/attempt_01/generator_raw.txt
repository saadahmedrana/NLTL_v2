<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-004-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Upper ice waterline displacement must equal the greatest linked waterline case displacement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?resultQuantity .
                ?resultQuantity qudt:numericValue ?resultValue .
                {
                    SELECT $this (MAX(?caseValue) AS ?maximumCaseValue)
                    WHERE {
                        $this nltl:hasUpperIceWaterlineCase ?case .
                        ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                        ?caseQuantity qudt:numericValue ?caseValue ;
                                      qudt:unit unit:KiloTON_Metric .
                    }
                    GROUP BY $this
                }
                FILTER (?resultValue != ?maximumCaseValue)
            }
        """
    ] .
<END_SHACL>
