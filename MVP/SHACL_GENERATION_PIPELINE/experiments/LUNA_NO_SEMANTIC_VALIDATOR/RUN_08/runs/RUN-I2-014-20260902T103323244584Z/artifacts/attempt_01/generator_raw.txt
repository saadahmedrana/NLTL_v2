@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_014_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "bowSubregionCount must be exactly four."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:node gen:I2_014_BowSubregionCaseShape ;
        sh:message "The ship must have exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_SelectedBowForceShape
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_SelectedBowLineLoadShape
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_014_SelectedBowPressureShape
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowForce must be at least every reported bow subregion calculated force and must equal one of those reported maxima." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                FILTER (
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?quantity .
                        ?quantity qudt:numericValue ?value .
                        FILTER (?value > ?selectedValue)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedForce ?quantity2 .
                        ?quantity2 qudt:numericValue ?value2 .
                        FILTER (?value2 = ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowLineLoad must be at least every reported bow subregion calculated line load and must equal one of those reported maxima." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowLineLoad ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                FILTER (
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedLineLoad ?quantity .
                        ?quantity qudt:numericValue ?value .
                        FILTER (?value > ?selectedValue)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedLineLoad ?quantity2 .
                        ?quantity2 qudt:numericValue ?value2 .
                        FILTER (?value2 = ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowPressure must be at least every reported bow subregion calculated pressure and must equal one of those reported maxima." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowPressure ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                FILTER (
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedPressure ?quantity .
                        ?quantity qudt:numericValue ?value .
                        FILTER (?value > ?selectedValue)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedPressure ?quantity2 .
                        ?quantity2 qudt:numericValue ?value2 .
                        FILTER (?value2 = ?selectedValue)
                    }
                )
            }
        """
    ] .

    ] ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:bowSubregionCalculatedLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:bowSubregionCalculatedPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [

    ] .
