        sh:message "Average ice pressure must equal ice force divided by load patch height and length, with kN converted to MN, within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?averageQuantity ;
                      nltl:iceForce ?forceQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:loadPatchLength ?lengthQuantity .
                ?averageQuantity qudt:numericValue ?average .
                ?forceQuantity qudt:numericValue ?force .
                ?heightQuantity qudt:numericValue ?height .
                ?lengthQuantity qudt:numericValue ?length .

                FILTER (?length != 0)
                BIND ((?force / 1000) / (?height * ?length) AS ?expected)
                FILTER (ABS(?average - ?expected) > (0.000000001 + (0.000001 * ABS(?expected))))
            }
        """

            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor ?shipFactorQuantity ;
                      nltl:hasPeakPressureFactorLookupCase ?lookupCase .
                ?lookupCase nltl:selectedPeakPressureFactor ?selectedFactorQuantity .
                ?shipFactorQuantity qudt:numericValue ?shipFactor .
                ?selectedFactorQuantity qudt:numericValue ?selectedFactor .
                FILTER (ABS(?shipFactor - ?selectedFactor) >
                    (0.000000001 + (0.000001 * ABS(?selectedFactor))))
            }
        """
    ] .
