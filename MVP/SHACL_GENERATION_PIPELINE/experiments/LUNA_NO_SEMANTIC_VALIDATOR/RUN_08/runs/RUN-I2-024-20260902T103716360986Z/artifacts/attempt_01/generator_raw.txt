<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:node gen:ObliquePlatingInterpolationCaseShape ;
        sh:message "Each oblique plating interpolation case must provide the applicable angle and interpolation structure."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:message "Each linked calculation case must provide the declared framing thickness inputs."
    ] .

gen:ObliquePlatingInterpolationCaseShape a sh:NodeShape ;
    sh:class nltl:obliquePlatingInterpolationCase ;
    sh:or (
        gen:ApplicableObliquePlatingCaseShape
        gen:NonApplicableObliquePlatingCaseShape
    ) .

gen:ApplicableObliquePlatingCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:node gen:FramingAngleValueShape ;
        sh:minCount 1 ;
        sh:message "An applicable oblique-plating case must provide its framing angle."
    ] ;
    sh:property [
        sh:path nltl:interpolatedNetPlateThickness ;
        sh:node gen:InterpolatedNetPlateThicknessValueShape ;
        sh:minCount 1 ;
        sh:message "An applicable oblique-plating case must provide the externally calculated interpolated net plate thickness."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:message "An applicable oblique-plating case must provide a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:node gen:InterpolationPointShape ;
        sh:minCount 1 ;
        sh:message "An applicable oblique-plating case must provide an upper interpolation endpoint."
    ] ;
    sh:sparql [
        sh:message "The framing angle for an applicable oblique-plating case must be greater than 20 degrees and less than 70 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:framingAngle ?angle .
                ?angle qudt:numericValue ?value .
                FILTER ( ?value <= 20 || ?value >= 70 )
            }
        """
    ] .

gen:NonApplicableObliquePlatingCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:node gen:FramingAngleValueShape ;
        sh:minCount 1 ;
        sh:message "A non-applicable case must provide its framing angle to establish non-applicability."
    ] ;
    sh:sparql [
        sh:message "A non-applicable case must have a framing angle at or below 20 degrees or at or above 70 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:framingAngle ?angle .
                ?angle qudt:numericValue ?value .
                FILTER ( ?value > 20 && ?value < 70 )
            }
        """
    ] .

gen:CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:longitudinalFramingNetPlateThickness ;
        sh:node gen:MillimetreQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "A calculation case must provide the longitudinal framing net plate thickness input."
    ] ;
    sh:property [
        sh:path nltl:transverseFramingNetPlateThickness ;
        sh:node gen:MillimetreQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "A calculation case must provide the transverse framing net plate thickness input."
    ] .

gen:InterpolationPointShape a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:node gen:DimensionlessQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "Each interpolation point must provide its coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:node gen:UnspecifiedQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "Each interpolation point must provide its externally calculated result."
    ] .

gen:FramingAngleValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:InterpolatedNetPlateThicknessValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnspecifiedQuantityValueShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

<END_SHACL>
