<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape ;
        sh:message "The ship must provide one patch load in kN."
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:message "The ship must identify the frame profile type."
    ] .

gs:StructuralMemberShape
    a sh:NodeShape ;
    sh:or (
        gs:BottomStructuralMemberBranch
        gs:SideStructuralMemberBranch
    ) ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gs:LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "Each represented structural member must have a structural-member load case."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape ;
        sh:message "Each represented structural member must provide one plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape ;
        sh:message "Each represented structural member must provide one midspan plastic-collapse load in kN."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed plastic strength, and plastic strength must equal the midspan load producing the plastic collapse mechanism." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                $this nltl:plasticStrength ?strength .
                ?strength qudt:numericValue ?strengthValue .
                $this nltl:midspanPlasticCollapseLoad ?midspanLoad .
                ?midspanLoad qudt:numericValue ?midspanValue .
                FILTER (
                    ?demandValue > ?strengthValue ||
                    ?strengthValue != ?midspanValue
                )
            }
        """
    ] .

gs:BottomStructuralMemberBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:bottomStructureLocation ;
        sh:message "A bottom structural member must be identified as being in the bottom structure."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gs:BottomLoadCaseShape
    ] .

gs:SideStructuralMemberBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:sideStructureLocation ;
        sh:message "A side structural member must be identified as being in the side structure."
    ] .

gs:LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape ;
        sh:message "Each load case must provide one combined shear and bending demand in kN."
    ] .

gs:BottomLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
        sh:message "For a bottom structural member, the patch-load dimension b must be parallel with the frame direction."
    ] .

gs:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
