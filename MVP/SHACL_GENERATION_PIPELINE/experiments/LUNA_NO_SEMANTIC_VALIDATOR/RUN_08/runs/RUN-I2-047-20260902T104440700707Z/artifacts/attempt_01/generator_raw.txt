<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_047_SteelRenewalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be required when gauged thickness is less than net thickness plus 0.5 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:gaugedThickness ?gaugedQuantity ;
                      nltl:netThickness ?netQuantity ;
                      nltl:steelRenewalRequired ?renewalRequired .

                ?gaugedQuantity qudt:numericValue ?gaugedThicknessValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                ?netQuantity qudt:numericValue ?netThicknessValue ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    ?gaugedThicknessValue <
                    (?netThicknessValue + 0.0005)
                )
                FILTER (?renewalRequired != true)
            }
        """
    ] .
<END_SHACL>
