<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix gen: <urn:nltl:generated-shape:> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A dimensionless quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A dimensionless quantity must use the unitless unit."
    ] .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A force quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A design vertical ice force must use the meganewton unit."
    ] .

    sh:property [

        sh:maxCount 1 ;
        sh:message "A stem angle must use the degree unit."

        sh:path q:unit ;
        sh:hasValue unit:M ;

gen:TableLookupCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each table lookup case must provide the crushing failure class factor."
    ] ;
    sh:property [
        sh:path nltl:flexuralFailureClassFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each table lookup case must provide the flexural failure class factor."
    ] ;
    sh:property [
        sh:path nltl:longitudinalStrengthClassFactor ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each table lookup case must provide the longitudinal strength class factor."
    ] .

    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TableLookupCaseShape ;

    sh:property [
        sh:path nltl:hullFormCoefficientKh ;

        sh:minCount 1 ;
        sh:message "The ship must provide the hull form coefficient kh."

    sh:property [
