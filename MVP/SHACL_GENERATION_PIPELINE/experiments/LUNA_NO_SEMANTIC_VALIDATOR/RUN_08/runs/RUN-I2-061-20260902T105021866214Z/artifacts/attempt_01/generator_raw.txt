<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-061-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node <urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    ] .

<urn:nltl:generated-shape:I2-061-CalculationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "Each applicable calculation case must use the prescribed analytical procedure."
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-HullStructureShape>
    ] ;
    sh:sparql [
        sh:message "A direct calculation method must not be used as an alternative to the prescribed analytical procedure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:calculationMethod nltl:directCalculationMethodValue .
            }
        """
    ] .

<urn:nltl:generated-shape:I2-061-HullStructureShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-061-ThicknessQuantityShape> ;
        sh:message "The assessed hull structure must provide its thickness."
    ] .

<urn:nltl:generated-shape:I2-061-ThicknessQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
