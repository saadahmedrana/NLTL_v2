<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record that direct calculation is used."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked load carrying stringer or web frame shall be identified as part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:partOfGrillageSystem ?grillage .
                    ?grillage a nltl:grillageSystem .
                }
            }
        """
    ] .
<END_SHACL>
