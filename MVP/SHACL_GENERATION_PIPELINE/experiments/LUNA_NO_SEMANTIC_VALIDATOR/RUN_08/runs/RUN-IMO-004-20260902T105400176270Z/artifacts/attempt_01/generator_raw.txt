<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_004_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "When the origin is sea ice, winter growth is no more than one, and thickness is between 0.3 m and 2.0 m inclusive, the ice type classification must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin .
                $this nltl:winterGrowthCount ?growth .
                $this nltl:iceThickness ?thicknessNode .
                ?thicknessNode qudt:numericValue ?thickness ;
                    qudt:unit unit:M .
                FILTER ( ?growth &lt;= 1 )
                FILTER ( ?thickness &gt;= "0.3"^^xsd:decimal &&
                         ?thickness &lt;= "2.0"^^xsd:decimal )
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
</BEGIN_SHACL>
