@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO005IceFreeWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare its ice condition." ;
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare whether any ice is present." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:totalIcePresent ;
                sh:hasValue false ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:totalIcePresent ;
                sh:hasValue true ;
            ] ;
            sh:sparql [
                sh:message "The ice condition 'ice free waters' must not be used when ice is present." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceCondition ?condition .
                        FILTER (
                            ?condition = "ice free waters"^^xsd:string
                        )
                    }
                """ ;
            ]
        ]
    ) .

<END_SHACL>
