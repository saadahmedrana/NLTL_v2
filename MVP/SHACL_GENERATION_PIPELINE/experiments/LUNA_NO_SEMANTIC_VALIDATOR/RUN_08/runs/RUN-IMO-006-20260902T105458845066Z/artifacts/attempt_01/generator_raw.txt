<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO-006_MediumFirstYearIceThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:iceTypeValue
        ] ;
        sh:message "A ship must provide its ice type classification."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For medium first-year ice, ice thickness must be a QuantityValue in metres with a decimal value from 0.7 m through 1.2 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?quantity .
                    ?quantity a qudt:QuantityValue ;
                              qudt:numericValue ?value ;
                              qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.7"^^xsd:decimal &&
                        ?value <= "1.2"^^xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every reported medium first-year ice thickness must be a decimal QuantityValue in metres within the range 0.7 m through 1.2 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce ;
                      nltl:iceThickness ?quantity .
                FILTER NOT EXISTS {
                    ?quantity a qudt:QuantityValue ;
                              qudt:numericValue ?value ;
                              qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.7"^^xsd:decimal &&
                        ?value <= "1.2"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
