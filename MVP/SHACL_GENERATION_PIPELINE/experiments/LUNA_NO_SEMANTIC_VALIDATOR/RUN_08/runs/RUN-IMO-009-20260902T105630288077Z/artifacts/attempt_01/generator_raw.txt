<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO009_BergyWatersShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one ice condition to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, land-origin ice concentration must be less than 1/10 and total ice concentration must not exceed 1/10, using dimensionless QUDT quantities." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "bergy waters")

                FILTER (
                    NOT EXISTS {
                        $this nltl:landOriginIceConcentration ?landQuantity .
                        ?landQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?landValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?landValue) = xsd:decimal &&
                            ?landValue < "1"^^xsd:decimal
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:landOriginIceConcentration ?landQuantity .
                        FILTER NOT EXISTS {
                            ?landQuantity a qudt:QuantityValue ;
                                qudt:numericValue ?landValue ;
                                qudt:unit unit:UNITLESS .
                            FILTER (
                                datatype(?landValue) = xsd:decimal &&
                                ?landValue < "1"^^xsd:decimal
                            )
                        }
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:totalIceConcentration ?totalQuantity .
                        ?totalQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?totalValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (
                            datatype(?totalValue) = xsd:decimal &&
                            ?totalValue <= "1"^^xsd:decimal
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:totalIceConcentration ?totalQuantity .
                        FILTER NOT EXISTS {
                            ?totalQuantity a qudt:QuantityValue ;
                                qudt:numericValue ?totalValue ;
                                qudt:unit unit:UNITLESS .
                            FILTER (
                                datatype(?totalValue) = xsd:decimal &&
                                ?totalValue <= "1"^^xsd:decimal
                            )
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
