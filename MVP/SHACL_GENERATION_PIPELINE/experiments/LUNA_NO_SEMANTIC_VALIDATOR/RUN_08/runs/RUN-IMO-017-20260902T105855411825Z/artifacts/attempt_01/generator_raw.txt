<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship shall record Polar Ship Certificate validity dates."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship shall record survey dates."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The ship shall record endorsement dates."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate shall include a supplement."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:minCount 1 ;
        sh:message "The ship shall link its Polar Certificate supplement."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "The ship shall link its Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:minCount 1 ;
        sh:message "The ship shall link the applicable SOLAS certificate schedule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each Polar Ship Certificate validity date shall match the corresponding SOLAS certificate schedule date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:polarCertificateValidityDates ?polarDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDate ?scheduleDate ;
                            nltl:certificateScheduleDateType nltl:validityDateCategory .
                    FILTER (
                        YEAR(?polarDate) = YEAR(?scheduleDate) &&
                        MONTH(?polarDate) = MONTH(?scheduleDate) &&
                        DAY(?polarDate) = DAY(?scheduleDate)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each survey date shall match the corresponding SOLAS certificate schedule date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:surveyDates ?polarDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDate ?scheduleDate ;
                            nltl:certificateScheduleDateType nltl:surveyDateCategory .
                    FILTER (
                        YEAR(?polarDate) = YEAR(?scheduleDate) &&
                        MONTH(?polarDate) = MONTH(?scheduleDate) &&
                        DAY(?polarDate) = DAY(?scheduleDate)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each endorsement date shall match the corresponding SOLAS certificate schedule date." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:endorsementDates ?polarDate .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDate ?scheduleDate ;
                            nltl:certificateScheduleDateType nltl:endorsementDateCategory .
                    FILTER (
                        YEAR(?polarDate) = YEAR(?scheduleDate) &&
                        MONTH(?polarDate) = MONTH(?scheduleDate) &&
                        DAY(?polarDate) = DAY(?scheduleDate)
                    )
                }
            }
        """
    ] .

        sh:class nltl:requiredEquipmentRecords ;

gen:IMO017ScheduleShape a sh:NodeShape ;
    sh:targetClass nltl:solasCertificateSchedule ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:message "The SOLAS certificate schedule shall contain schedule-date records."
    ] .

gen:IMO017DateRecordShape a sh:NodeShape ;
    sh:targetClass nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule-date record shall contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule-date record shall identify a validity, survey, or endorsement date category."
    ] .
<END_SHACL>
