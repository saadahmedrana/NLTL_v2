<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_033_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO_033_ApplicableShape
        gen:IMO_033_NotApplicableShape
    ) .

gen:IMO_033_NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceAccretionLikely must be explicitly false when the icing allowance rule is not applicable."
    ] .

gen:IMO_033_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceAccretionLikely must be explicitly true to apply the icing allowance rule."
    ] ;
    sh:property [
        sh:path nltl:deckIcingAllowance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "30"^^xsd:decimal ;
                sh:maxInclusive "30"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM-PER-M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:sideIcingAllowance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "7.5"^^xsd:decimal ;
                sh:maxInclusive "7.5"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM-PER-M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:continuousSurfaceProjectedArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:continuousSurfaceStaticMoment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:discontinuousSurfaceProjectedArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:discontinuousSurfaceStaticMoment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "When ice accretion is likely, discontinuous surface projected area must equal continuous surface projected area multiplied by 1.05, and discontinuous surface static moment must equal continuous surface static moment multiplied by 1.10, within a relative tolerance of 1e-6 with an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:continuousSurfaceProjectedArea ?continuousAreaNode ;
                      nltl:discontinuousSurfaceProjectedArea ?discontinuousAreaNode ;
                      nltl:continuousSurfaceStaticMoment ?continuousMomentNode ;
                      nltl:discontinuousSurfaceStaticMoment ?discontinuousMomentNode .

                ?continuousAreaNode qudt:numericValue ?continuousArea .
                ?discontinuousAreaNode qudt:numericValue ?discontinuousArea .
                ?continuousMomentNode qudt:numericValue ?continuousMoment .
                ?discontinuousMomentNode qudt:numericValue ?discontinuousMoment .

                BIND((?continuousArea * 1.05) AS ?expectedArea)
                BIND(IF(?expectedArea >= 0,
                        ?expectedArea * 0.000001,
                        -?expectedArea * 0.000001) AS ?relativeAreaTolerance)
                BIND(IF(?relativeAreaTolerance > 0,
                        ?relativeAreaTolerance,
                        0.000001) AS ?areaTolerance)

                BIND((?continuousMoment * 1.10) AS ?expectedMoment)
                BIND(IF(?expectedMoment >= 0,
                        ?expectedMoment * 0.000001,
                        -?expectedMoment * 0.000001) AS ?relativeMomentTolerance)
                BIND(IF(?relativeMomentTolerance > 0,
                        ?relativeMomentTolerance,
                        0.000001) AS ?momentTolerance)

                FILTER (
                    ?discontinuousArea < (?expectedArea - ?areaTolerance) ||
                    ?discontinuousArea > (?expectedArea + ?areaTolerance) ||
                    ?discontinuousMoment < (?expectedMoment - ?momentTolerance) ||
                    ?discontinuousMoment > (?expectedMoment + ?momentTolerance)
                )
            }
        """
    ] .
<END_SHACL>
