<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO038Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Damage-center position relative to maximum breadth must be supplied as exactly one boolean value."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Upper ice waterline length must be supplied as one length quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Upper ice waterline draught must be supplied as one length quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Longitudinal damage extent must be supplied as one length quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:node gen:LengthMillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Transverse penetration must be supplied as one length quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:node gen:LengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Vertical damage extent must be supplied as one length quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:node gen:NonNegativeLengthMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Selected vertical damage position must be supplied as one non-negative length quantity in metres."
    ] ;
    sh:sparql [
        sh:message "Longitudinal damage extent must equal 4.5% of upper ice waterline length when the damage centre is forward of maximum breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth true ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:longitudinalDamageExtent ?extentNode .
                ?lengthNode qudt:numericValue ?lengthValue .
                ?extentNode qudt:numericValue ?extentValue .
                BIND (0.045 * ?lengthValue AS ?expected)
                BIND (IF (ABS(?expected) * 0.000001 > 0.000000001,
                          ABS(?expected) * 0.000001,
                          0.000000001) AS ?tolerance)
                FILTER (?extentValue < (?expected - ?tolerance) ||
                        ?extentValue > (?expected + ?tolerance))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Longitudinal damage extent must equal 1.5% of upper ice waterline length when the damage centre is not forward of maximum breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth false ;
                      nltl:upperIceWaterlineLength ?lengthNode ;
                      nltl:longitudinalDamageExtent ?extentNode .
                ?lengthNode qudt:numericValue ?lengthValue .
                ?extentNode qudt:numericValue ?extentValue .
                BIND (0.015 * ?lengthValue AS ?expected)
                BIND (IF (ABS(?expected) * 0.000001 > 0.000000001,
                          ABS(?expected) * 0.000001,
                          0.000000001) AS ?tolerance)
                FILTER (?extentValue < (?expected - ?tolerance) ||
                        ?extentValue > (?expected + ?tolerance))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Transverse penetration must be exactly 760 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:transversePenetration ?penetrationNode .
                ?penetrationNode qudt:numericValue ?penetrationValue .
                FILTER (?penetrationValue < 760 || ?penetrationValue > 760)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Vertical damage extent must equal the lesser of 20% of upper ice waterline draught and longitudinal damage extent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:longitudinalDamageExtent ?longitudinalNode ;
                      nltl:verticalDamageExtent ?verticalNode .
                ?draughtNode qudt:numericValue ?draughtValue .
                ?longitudinalNode qudt:numericValue ?longitudinalValue .
                ?verticalNode qudt:numericValue ?verticalValue .
                BIND (0.20 * ?draughtValue AS ?draughtLimit)
                BIND (IF (?draughtLimit < ?longitudinalValue,
                          ?draughtLimit,
                          ?longitudinalValue) AS ?expected)
                BIND (IF (ABS(?expected) * 0.000001 > 0.000000001,
                          ABS(?expected) * 0.000001,
                          0.000000001) AS ?tolerance)
                FILTER (?verticalValue < (?expected - ?tolerance) ||
                        ?verticalValue > (?expected + ?tolerance))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected vertical damage position must not exceed 120% of upper ice waterline draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraught ?draughtNode ;
                      nltl:selectedVerticalDamagePosition ?positionNode .
                ?draughtNode qudt:numericValue ?draughtValue .
                ?positionNode qudt:numericValue ?positionValue .
                FILTER (?positionValue > (1.20 * ?draughtValue))
            }
        """
    ] .

gen:LengthMetreQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NonNegativeLengthMetreQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthMillimetreQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 0 ;
        sh:maxInclusive 760
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
