<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltl:IMO059ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFirefighterOutfit ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:firefighterOutfitItem ;
        sh:node nltl:IMO059FirefighterOutfitShape ;
        sh:message "Each linked firefighter outfit shall be stored at a warm-class location."
    ] .

nltl:IMO059FirefighterOutfitShape
    a sh:NodeShape ;
    sh:class nltl:equipmentItem ;
    sh:property [
        sh:path nltl:hasStorageLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:physicalStorageLocation ;
        sh:node nltl:IMO059StorageLocationShape ;
        sh:minCount 1 ;
        sh:message "A firefighter outfit shall have a storage location."
    ] .

nltl:IMO059StorageLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:storageTemperatureClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:storageTemperatureClassValue ;
        sh:hasValue nltl:warmStorageLocation ;
        sh:minCount 1 ;
        sh:message "The storage location shall be classified as warm."
    ] .
<END_SHACL>
