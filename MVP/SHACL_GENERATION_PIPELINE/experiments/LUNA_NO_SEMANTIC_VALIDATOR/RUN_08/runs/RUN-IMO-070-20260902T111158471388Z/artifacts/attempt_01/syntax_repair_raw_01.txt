<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_070_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship type is required to determine applicability of the passenger-ship survival-equipment rule."
    ] ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Size compatibility must be recorded as true for the assigned survival equipment."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a passenger ship, every person on board must have an assignment to a properly sized immersion suit or thermal protective aid." ;
        sh:select """ 
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:shipType ?shipType .
                    FILTER (?shipType = "passenger_ship"^^xsd:string)
                }
                FILTER EXISTS {
                    $this nltl:sizeCompatibilityStatus true
                }
                $this nltl:hasPersonOnBoardMember ?person .
                FILTER NOT EXISTS {
                    ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                    ?assignment nltl:assignedPersonalSurvivalEquipment ?equipment .
                    ?equipment rdf:type ?equipmentType .
                    FILTER (
                        ?equipmentType IN (
                            nltl:immersionSuitEquipment,
                            nltl:thermalProtectiveAidEquipment
                        )
                    )
                }
            }
        """
    ] .
</BEGIN_SHACL>
