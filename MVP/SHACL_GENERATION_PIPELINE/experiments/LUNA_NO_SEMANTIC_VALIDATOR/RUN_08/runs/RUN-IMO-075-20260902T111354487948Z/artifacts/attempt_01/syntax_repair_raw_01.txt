<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-075> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The survival equipment applicability status must be provided." ;
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue false ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue true ;
            ] ;
            sh:property [
                sh:path nltl:personsOnBoard ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:availableSurvivalEquipmentCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:message "Available survival equipment capacity must be at least 110% of the persons on board, rounded up." ;
            ] ;
            sh:property [
                sh:path nltl:storageAccessibilityStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "Required personal or group survival equipment must be stored in an accessible location." ;
            ] ;
            sh:sparql [
                sh:message "Available survival equipment capacity must be at least ceil(persons on board multiplied by 1.10)." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:personsOnBoard ?persons ;
                              nltl:availableSurvivalEquipmentCapacity ?capacity .
                        FILTER (?capacity < CEIL(?persons * 1.10))
                    }
                """ ;
            ]
        ]
    ) .
<END_SHACL>
