<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_079_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide exactly one persons-on-board count."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide exactly one maximum expected rescue time in days."
    ] ;
    sh:property [
        sh:path nltl:availableEmergencyRationPerson ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DAY ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each ship must provide exactly one emergency ration capacity in person-days."
    ] ;
    sh:sparql [
        sh:message "Available emergency ration person-days must be at least persons on board multiplied by maximum expected rescue time in days." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:personsOnBoard ?persons ;
                      nltl:maximumExpectedRescueTime ?rescueNode ;
                      nltl:availableEmergencyRationPerson ?rationNode .
                ?rescueNode qudt:numericValue ?rescueTime .
                ?rationNode qudt:numericValue ?rationCapacity .
                FILTER (
                    ?rationCapacity <
                    (xsd:decimal(?persons) * ?rescueTime)
                )
            }
        """
    ] .
<END_SHACL>
