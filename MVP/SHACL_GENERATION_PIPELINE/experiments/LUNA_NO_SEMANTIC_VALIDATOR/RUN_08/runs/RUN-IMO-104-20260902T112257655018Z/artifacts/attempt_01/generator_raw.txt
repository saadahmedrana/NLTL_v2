<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_104_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization status."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gen:IMO_104_CrewMemberShape
    ] .

gen:IMO_104_CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:class nltl:assignedDuty ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_AssignedDutyShape ;
        sh:message "Each crew member must have an assigned duty linked to relevant PWOM items."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:class nltl:familiarizationRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO_104_FamiliarizationRecordShape ;
        sh:message "Each crew member must have a PWOM familiarization record."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each PWOM procedure or equipment item relevant to a crew member's assigned duties must be included in that crew member's familiarization record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:crewMember .
                $this nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    $this nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gen:IMO_104_AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:message "Each assigned duty must identify its relevant PWOM procedure or equipment."
    ] .

gen:IMO_104_FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:class nltl:polarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:message "Each familiarization record must identify the relevant PWOM procedure or equipment covered."
    ] .
<END_SHACL>
