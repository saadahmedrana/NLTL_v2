<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string ;
        sh:in ( "oil_residue_tank" "oily_bilge_water_holding_tank" )
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 2017-01-01, each applicable oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have at least 0.76 m separation from the outer shell." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:tankType ?tankType ;
                      nltl:individualTankCapacity ?capacity .

                ?capacity qudt:numericValue ?capacityValue .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER (?date >= "2017-01-01"^^xsd:date)
                FILTER (
                    ?tankType IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )
                FILTER (?capacityValue > 30)

                FILTER NOT EXISTS {
                    $this nltl:tankOuterShellSeparation ?separation .
                    ?separation qudt:numericValue ?separationValue .
                    ?separation qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (?separationValue >= 0.76)
                }
            }
        """
    ] .
<END_SHACL>
