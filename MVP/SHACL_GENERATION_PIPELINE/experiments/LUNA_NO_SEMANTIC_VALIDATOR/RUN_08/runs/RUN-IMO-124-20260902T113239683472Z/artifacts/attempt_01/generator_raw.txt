<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-124> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an Antarctic-area MARPOL Annex V regulation 6.1 discharge, the required ice-distance records and prohibition must conform." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this a nltl:ship ;
        nltl:operatingArea ?operatingArea ;
        nltl:antarcticArea ?antarcticArea ;
        nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .

    FILTER (
        str(?operatingArea) = "Antarctic_area" &&
        str(?antarcticArea) = "Antarctic_area" &&
        ?discharge = true
    )

    FILTER (
        NOT EXISTS {
            $this nltl:distanceToNearestFastIce ?fastIceDistance .
            ?fastIceDistance a qudt:QuantityValue ;
                qudt:numericValue ?fastIceValue ;
                qudt:unit unit:MI_N .
            FILTER (
                datatype(?fastIceValue) = xsd:decimal &&
                ?fastIceValue >= "12"^^xsd:decimal
            )
        }
        ||
        NOT EXISTS {
            $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?concentrationAreaDistance .
            ?concentrationAreaDistance a qudt:QuantityValue ;
                qudt:numericValue ?concentrationAreaDistanceValue ;
                qudt:unit unit:MI_N .
            FILTER (datatype(?concentrationAreaDistanceValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            $this nltl:nearbyIceConcentration ?nearbyConcentration .
            ?nearbyConcentration a qudt:QuantityValue ;
                qudt:numericValue ?nearbyConcentrationValue ;
                qudt:unit unit:UNITLESS .
            FILTER (datatype(?nearbyConcentrationValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            $this nltl:iceConcentration ?concentration .
            ?concentration a qudt:QuantityValue ;
                qudt:numericValue ?concentrationValue ;
                qudt:unit unit:UNITLESS .
            FILTER (datatype(?concentrationValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            $this nltl:foodWasteDischargeOntoIce ?foodWasteDischarge .
            FILTER (?foodWasteDischarge = false)
        }
    )
}
""" ;
    ] .
<END_SHACL>
