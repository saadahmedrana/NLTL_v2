<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-002_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "A construction contract date, when supplied, must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ships contracted on or after 1 January 2019 and before 5 July 2021, the applicable ice class regulation edition must be 2017." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    YEAR(?date) > 2019 ||
                    (YEAR(?date) = 2019 && MONTH(?date) >= 1)
                )
                FILTER (
                    YEAR(?date) < 2021 ||
                    (YEAR(?date) = 2021 &&
                     (MONTH(?date) < 7 ||
                      (MONTH(?date) = 7 && DAY(?date) < 5)))
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2017 .
                }
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition ?otherEdition .
                    FILTER (?otherEdition != nltl:iceClassRegulationEdition2017)
                }
            }
        """
    ] .
<END_SHACL>
