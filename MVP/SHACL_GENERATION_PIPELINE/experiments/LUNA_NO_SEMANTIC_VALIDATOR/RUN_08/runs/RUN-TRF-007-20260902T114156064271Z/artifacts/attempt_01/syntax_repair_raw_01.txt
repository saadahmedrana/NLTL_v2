<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must have an ice class to determine applicability."
    ] ;
    sh:or (
        [
            a sh:NodeShape ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The conditional requirements apply only to ice class IA Super or IA ships meeting the construction-stage condition." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER (?iceClass IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa
                        ))
                    }
                """
            ]
        ]
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "An applicable ship must have its construction-stage date."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "An applicable ship must have its delivery date."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:message "An applicable ship must have an assessment date."
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:engineOutputComplianceMethodValue ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:minCount 1 ;
                sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4 of the 2017 Ice Class Regulations."
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "For an IA Super or IA ship, the construction-stage date must be before 1 September 2003, and compliance must be established no later than 1 January of the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass ;
                              nltl:constructionStageDate ?constructionStageDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            ?iceClass NOT IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            )
                            ||
                            YEAR(?constructionStageDate) > 2003
                            ||
                            (
                                YEAR(?constructionStageDate) = 2003 &&
                                (
                                    MONTH(?constructionStageDate) > 9
                                    ||
                                    (
                                        MONTH(?constructionStageDate) = 9 &&
                                        DAY(?constructionStageDate) >= 1
                                    )
                                )
                            )
                            ||
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                MONTH(?assessmentDate) > 1
                            )
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
