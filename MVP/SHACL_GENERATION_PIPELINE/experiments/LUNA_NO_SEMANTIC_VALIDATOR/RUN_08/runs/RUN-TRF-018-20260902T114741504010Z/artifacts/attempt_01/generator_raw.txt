<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-018> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one supported ice class." ;
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:class qudt:QuantityValue ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Each ship must provide one calculated required power quantity in kW." ;
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:class qudt:QuantityValue ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "Each ship must provide one maximum continuous rating power quantity in kW." ;
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximum continuous rating power must be at least the calculated required power and 1000 kW; for IA Super, it must be at least the calculated required power and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?maximumNode .
                ?calculatedNode qudt:numericValue ?calculated .
                ?maximumNode qudt:numericValue ?maximum .
                FILTER (
                    (
                        ?ice IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        )
                        &&
                        (
                            ?maximum < ?calculated ||
                            ?maximum < 1000
                        )
                    )
                    ||
                    (
                        ?ice = nltl:iceClassIaSuper
                        &&
                        (
                            ?maximum < ?calculated ||
                            ?maximum < 2800
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
