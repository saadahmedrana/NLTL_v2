<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF_024_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide an ice class to determine applicability of the IA/IA Super rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA or IA Super with construction begun before 1 September 2003, the ship must have the section 3.2.2 compliance method by 1 January of the year twenty years after delivery." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .

    OPTIONAL {
        $this nltl:constructionStageDate ?constructionStageDate .
    }
    OPTIONAL {
        $this nltl:deliveryDate ?deliveryDate .
    }
    OPTIONAL {
        $this nltl:assessmentDate ?assessmentDate .
    }

    FILTER (
        ?iceClass = nltl:iceClassIaSuper ||
        ?iceClass = nltl:iceClassIa
    )

    FILTER (
        !BOUND(?constructionStageDate) ||
        DATATYPE(?constructionStageDate) != xsd:date ||
        (
            ?constructionStageDate < "2003-09-01"^^xsd:date &&
            (
                !BOUND(?deliveryDate) ||
                DATATYPE(?deliveryDate) != xsd:date ||
                !BOUND(?assessmentDate) ||
                DATATYPE(?assessmentDate) != xsd:date ||
                (
                    YEAR(?assessmentDate) >= YEAR(?deliveryDate) + 20 &&
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod
                            nltl:traficom2017Section3Point2Point2Method .
                    }
                )
            )
        )
    )
}
""" ;
    ] .
<END_SHACL>
