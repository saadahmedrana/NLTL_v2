gen:TRF042PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042SpacingQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042FactorQuantityShape
    ] ;
    sh:or (
        gen:TRF042CorrosionNormalBranch
        gen:TRF042CorrosionCoatingBranch
    ) ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable piecewise formula of designIceLoadHeight divided by spacing within a tolerance of max(1e-6*ABS(expected),1e-12), and the ratio must not exceed 1.8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .

                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?actual .

                BIND((?h / ?s) AS ?ratio)
                BIND(
                    IF(
                        ?ratio <= 1.0,
                        (0.4 / ?ratio) + 0.6,
                        IF(
                            ?ratio <= 1.8,
                            1.4 - (0.4 * ?ratio),
                            ?actual + 1.0
                        )
                    ) AS ?expected
                )
                BIND(
                    IF(
                        (ABS(?expected) * 0.000001) > 0.000000000001,
                        ABS(?expected) * 0.000001,
                        0.000000000001
                    ) AS ?tolerance
                )

                FILTER(
                    ?ratio > 1.8 ||
                    ABS(?actual - ?expected) > ?tolerance
                )
            }
        """
    ] .

</END_SHACL>
