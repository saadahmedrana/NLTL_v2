g:TRF050FrameShape a sh:NodeShape ;

    sh:property [
        sh:path nltl:frameAttachment ;
        sh:minCount 1 ;

        sh:message "Each frame must have a frame attachment record."
    ] .

    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:datatype nltl:weldTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "The frame attachment must specify one double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Shell plate butt crossing must be recorded to select the scalloping rule."
    ] ;
    sh:property [
        sh:path nltl:scallopingPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:minCount 1 ;
                sh:hasValue false ;
                sh:message "Scalloping is not allowed unless the frame crosses a shell plate butt."
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue true
            ]
        ]
    ) .
