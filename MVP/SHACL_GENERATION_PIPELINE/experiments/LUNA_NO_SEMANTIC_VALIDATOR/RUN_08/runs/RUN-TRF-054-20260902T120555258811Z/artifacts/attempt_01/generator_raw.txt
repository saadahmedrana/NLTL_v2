        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LineLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "Design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy the TRF-054 formulas; calculated-result tolerance is relative 1e-6 with an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lineNode qudt:numericValue ?line .
                ?factorNode qudt:numericValue ?factor .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?designLoadMNPerM)
                BIND(?designLoadMNPerM * 1000 AS ?expectedLine)
                BIND(1 - (?hs / ?l) AS ?expectedFactor)
                BIND(0.80 * 1.8 * ?designLoadMNPerM * (?l * ?l) /
                     (?m * ?sigma) * ?expectedFactor * 1000000 AS ?expectedZ)
                BIND(1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                     ?designLoadMNPerM * ?l /
                     (2 * ?sigma) * ?expectedFactor * 10000 AS ?expectedA)

                FILTER(
                    (ABS(?line - ?expectedLine) >
                     (ABS(?expectedLine) * 0.000001 + 0.000000001))
                    ||
                    (ABS(?factor - ?expectedFactor) >
                     (ABS(?expectedFactor) * 0.000001 + 0.000000001))
                    ||
                    (ABS(?z - ?expectedZ) >
                     (ABS(?expectedZ) * 0.000001 + 0.000000001))
                    ||
                    (ABS(?a - ?expectedA) >
                     (ABS(?expectedA) * 0.000001 + 0.000000001))
                )
            }
        """
    ] .
