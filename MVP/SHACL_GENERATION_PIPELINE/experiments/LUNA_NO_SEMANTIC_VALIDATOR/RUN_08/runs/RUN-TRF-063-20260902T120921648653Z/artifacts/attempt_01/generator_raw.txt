<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF063ApplicableShipShape
        gen:TRF063NonApplicableLowCountShape
        gen:TRF063NonApplicableHighCountShape
    ) .

gen:TRF063ApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 3 ;
        sh:message "A twin- or triple-screw ship must declare a propeller count of 2 or 3."
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:node gen:TRF063EnvelopeShape ;
        sh:message "A twin- or triple-screw ship must provide a side-propeller strengthening envelope."
    ] .

gen:TRF063NonApplicableLowCountShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxInclusive 1
    ] .

gen:TRF063NonApplicableHighCountShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 4
    ] .

gen:TRF063EnvelopeShape
    a sh:NodeShape ;
    sh:class nltl:sidePropellerStrengtheningEnvelope ;
    sh:property [
        sh:path nltl:strengtheningEnvelopeBoundary ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tankTop ;
        sh:minCount 1 ;
        sh:message "The strengthening envelope boundary must be a tank top."
    ] ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063ForwardExtentQuantityShape ;
        sh:message "The strengthening must extend at least 1.5 metres forward of the side propeller."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node gen:TRF063AftExtentQuantityShape ;
        sh:message "The strengthening must extend at least 1.5 metres aft of the side propeller."
    ] .

gen:TRF063ForwardExtentQuantityShape
    a sh:NodeShape ;
    sh:nodeKind sh:IRI ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF063AftExtentQuantityShape
    a sh:NodeShape ;
    sh:nodeKind sh:IRI ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
