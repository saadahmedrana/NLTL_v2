<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF064Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF064EnclosedBranch
        gen:TRF064DetachedStrutBranch
    ) .

gen:TRF064EnclosedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:detachedStrut ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The detached-strut applicability value must be false for the plated-bossing alternative."
    ] ;
    sh:property [
        sh:path (
            nltl:hasSidePropellerShafting
            nltl:enclosedByPlatedBossing
        ) ;
        sh:minCount 1 ;
        sh:message "Side propeller shafting shall be enclosed within plated bossing when detached struts are not used."
    ] ;
    sh:property [
        sh:path (
            nltl:hasSternTube
            nltl:enclosedByPlatedBossing
        ) ;
        sh:minCount 1 ;
        sh:message "Stern tubes of side propellers shall be enclosed within plated bossing when detached struts are not used."
    ] .

gen:TRF064DetachedStrutBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:detachedStrut ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The detached-strut applicability value must be true for the detached-strut alternative."
    ] ;
    sh:property [
        sh:path nltl:hasDetachedStrut ;
        sh:minCount 1 ;
        sh:message "A detached-strut structure shall be identified when detached struts are used."
    ] ;
    sh:property [
        sh:path (
            nltl:hasDetachedStrut
            nltl:hasStrutDesignEvidence
        ) ;
        sh:minCount 1 ;
        sh:message "Design evidence shall be provided for each detached strut."
    ] ;
    sh:property [
        sh:path (
            nltl:hasDetachedStrut
            nltl:hasStrutStrengthEvidence
        ) ;
        sh:minCount 1 ;
        sh:message "Strength evidence shall be provided for each detached strut."
    ] ;
    sh:property [
        sh:path (
            nltl:hasDetachedStrut
            nltl:hasStrutHullAttachmentEvidence
        ) ;
        sh:minCount 1 ;
        sh:message "Hull-attachment evidence shall be provided for each detached strut."
    ] .
<END_SHACL>
