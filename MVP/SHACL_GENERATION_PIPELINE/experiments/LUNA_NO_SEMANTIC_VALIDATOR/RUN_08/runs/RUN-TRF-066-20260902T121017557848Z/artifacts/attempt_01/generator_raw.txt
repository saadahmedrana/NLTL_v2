<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_066_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must specify exactly one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "The ship must state whether an ice knife is practicable."
    ] ;
    sh:or (
        gen:TRF_066_ApplicableShape
        gen:TRF_066_NonApplicableShape
    ) .

gen:TRF_066_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIa
    ] ;
    sh:or (
        gen:TRF_066_PracticableShape
        gen:TRF_066_NotPracticableShape
    ) .

gen:TRF_066_PracticableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasRudderProtectionArrangement ;
        sh:minCount 1 ;
        sh:class nltl:rudderProtectionArrangement ;
        sh:node gen:TRF_066_IceKnifeArrangementShape ;
        sh:message "For an applicable ship with a practicable ice knife, a rudder protection arrangement with an ice knife extending below the LIWL is required."
    ] .

gen:TRF_066_NotPracticableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceKnifePracticable ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:hasRudderProtectionArrangement ;
        sh:minCount 1 ;
        sh:class nltl:rudderProtectionArrangement ;
        sh:node gen:TRF_066_EquivalentArrangementShape ;
        sh:message "For an applicable ship where an ice knife is not practicable, an equivalent rudder protection means is required."
    ] .

gen:TRF_066_NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF_066_IceKnifeArrangementShape
    a sh:NodeShape ;
    sh:class nltl:iceKnife ;
    sh:property [
        sh:path nltl:extendsBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ice knife must extend below the lower ice waterline."
    ] .

gen:TRF_066_EquivalentArrangementShape
    a sh:NodeShape ;
    sh:class nltl:equivalentRudderProtectionMeans .

<END_SHACL>
