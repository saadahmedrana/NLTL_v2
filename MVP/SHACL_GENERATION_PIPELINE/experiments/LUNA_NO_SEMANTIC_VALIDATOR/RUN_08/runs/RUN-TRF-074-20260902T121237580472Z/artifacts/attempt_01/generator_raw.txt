<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_074_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice class to determine applicability of the propeller submersion rule."
    ] ;
    sh:sparql [
        sh:message "For ice classes IB and IC, the ballast-condition propeller top depth and required depth hi must be valid metre quantities; if the propeller top is not at least hi below the water surface, the propulsion system must be designed for ice class IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                FILTER (
                    EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        FILTER NOT EXISTS {
                            ?depthNode qudt:numericValue ?depthValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?depthValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        FILTER NOT EXISTS {
                            ?hiNode qudt:numericValue ?hiValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?hiValue) = xsd:decimal)
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:ballastConditionPropellerTopDepth ?validDepthNode .
                        ?validDepthNode qudt:numericValue ?validDepth ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?validDepth) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPropellerSubmersionDepthHi ?validHiNode .
                        ?validHiNode qudt:numericValue ?validHi ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?validHi) = xsd:decimal)
                    }
                    ||
                    (
                        NOT EXISTS {
                            $this nltl:ballastConditionPropellerTopDepth ?depthNode2 .
                            ?depthNode2 qudt:numericValue ?depthValue2 ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?depthValue2) = xsd:decimal)

                            $this nltl:requiredPropellerSubmersionDepthHi ?hiNode2 .
                            ?hiNode2 qudt:numericValue ?hiValue2 ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            FILTER (datatype(?hiValue2) = xsd:decimal)

                            FILTER (?depthValue2 >= ?hiValue2)
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
