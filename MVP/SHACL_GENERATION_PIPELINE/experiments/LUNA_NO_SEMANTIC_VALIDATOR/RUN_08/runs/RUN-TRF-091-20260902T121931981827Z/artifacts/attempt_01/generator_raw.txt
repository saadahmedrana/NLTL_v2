<urn:nltl:generated-shape:TRF-091_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceMillingSequenceCase ;
        sh:minCount 1 ;
        sh:message "The ship shall identify an ice-milling sequence case."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-091_PropulsionLoadCaseShape> ;
        sh:message "The ship shall identify a propulsion load case for the prescribed load sequence."
    ] .

<urn:nltl:generated-shape:TRF-091_PropulsionLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadCasePurpose ;

        sh:in ( nltl:propulsionLineStrengthEvaluationPurpose ) ;
        sh:minCount 1 ;
        sh:message "The load case shall be designated for propulsion-line strength evaluation and not for propulsion-system stalling analysis."
    ] ;
    sh:property [
        sh:path nltl:loadCasePurpose ;
        sh:not [
            sh:hasValue nltl:stallingAnalysisPurposeValue
        ]
    ] .
