nltlg:FormulaAlternativeShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeExpandedAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:leadingEdgeChordPortionAtZeroPointEightRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:trailingEdgeChordPortionAtZeroPointEightRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureSpindleTorqueFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureSpindleTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:TorqueQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumSpindleTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlg:TorqueQuantityShape
    ] ;
    sh:sparql [
        sh:message "The reported spindle torque factor and spindle torques must satisfy the declared formula within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bladeExpandedAreaRatio ?earNode ;
                      nltl:propellerBladeCount ?z ;
                      nltl:leadingEdgeChordPortionAtZeroPointEightRadius ?leNode ;
                      nltl:trailingEdgeChordPortionAtZeroPointEightRadius ?teNode ;
                      nltl:extremeIceForce ?forceNode ;
                      nltl:bladeFailureSpindleTorqueFactor ?factorNode ;
                      nltl:bladeFailureSpindleTorque ?failureTorqueNode ;
                      nltl:maximumSpindleTorque ?maximumTorqueNode .

                ?earNode qudt:numericValue ?ear .
                ?leNode qudt:numericValue ?le .
                ?teNode qudt:numericValue ?te .
                ?forceNode qudt:numericValue ?force .
                ?factorNode qudt:numericValue ?factor .
                ?failureTorqueNode qudt:numericValue ?failureTorque .
                ?maximumTorqueNode qudt:numericValue ?maximumTorque .

                BIND((4.0 * ?ear / ?z) AS ?ratio)
                BIND((?ratio * ?ratio * ?ratio) AS ?ratioCube)
                BIND((0.7 * (1.0 - ?ratioCube)) AS ?rawFactor)
                BIND(IF(?rawFactor < 0.3, 0.3, ?rawFactor) AS ?expectedFactor)
                BIND(IF(?le > (0.8 * ?te), ?le, (0.8 * ?te)) AS ?chordTerm)
                BIND((?chordTerm * ?expectedFactor * ?force) AS ?expectedTorque)

                FILTER (
                    ABS(?factor - ?expectedFactor) >
                        (0.000001 * ABS(?expectedFactor) + 0.000001)
                    ||
                    ABS(?failureTorque - ?expectedTorque) >
                        (0.000001 * ABS(?expectedTorque) + 0.000001)
                    ||
                    ABS(?maximumTorque - ?expectedTorque) >
                        (0.000001 * ABS(?expectedTorque) + 0.000001)
                )
            }
        """
    ] .

nltlg:TRF102Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltlg:StressAnalysisAlternativeShape
        nltlg:FormulaAlternativeShape
    ) ;
    sh:message "The maximum spindle torque must be supported either by an approved appropriate stress analysis or by the declared blade-failure torque calculation." .
