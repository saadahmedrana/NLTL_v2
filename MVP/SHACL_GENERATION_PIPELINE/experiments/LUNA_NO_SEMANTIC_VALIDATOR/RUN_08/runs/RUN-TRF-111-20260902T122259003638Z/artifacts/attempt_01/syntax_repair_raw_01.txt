    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the extreme ice force input in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ] ;
        sh:message "The ship must identify the applicable propeller shaft line components."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [
                    sh:class nltl:loadCase ;
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.3 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:class nltl:loadCase ;
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:class nltl:loadCase ;
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive 1.0 ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            )
        ] ;
        sh:message "Each propulsion load case must identify an applicable shaft-line load-case type and satisfy its declared readiness check."
    ] ;

        a sh:SPARQLConstraint ;
        sh:message "When a fatigue load case applies, every linked propeller shaft line component must provide a fatigue safety factor of at least 1.5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?fatigueCase .
                ?fatigueCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor <http://qudt.org/schema/qudt/numericValue> ?numericValue .
                    ?numericValue <http://www.w3.org/2001/XMLSchema#decimal> ?lexicalValue .
                    FILTER ( ?numericValue >= 1.5 )
                    ?factor <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/UNITLESS> .
                }
            }
        """
    ] .
