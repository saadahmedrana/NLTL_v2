<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:TRF127AirReceiverCapacityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:airReceiverServesAdditionalPurpose ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applicability value must indicate that the air receiver serves an additional purpose."
            ] ;
            sh:property [
                sh:path nltl:airReceiverCapacity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M3 ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Air receiver capacity is required in cubic metres when additional-purpose service applies."
            ] ;
            sh:property [
                sh:path nltl:additionalPurposeRequiredAirCapacity ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Additional-purpose required air capacity is required when additional-purpose service applies."
            ] ;
            sh:sparql [
                sh:message "Air receiver capacity must be at least the additional-purpose required air capacity." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:airReceiverServesAdditionalPurpose true ;
                              nltl:airReceiverCapacity ?receiverCapacity ;
                              nltl:additionalPurposeRequiredAirCapacity ?requiredCapacity .
                        ?receiverCapacity qudt:numericValue ?receiverValue .
                        ?requiredCapacity qudt:numericValue ?requiredValue .
                        FILTER (
                            DATATYPE(?receiverValue) = xsd:decimal &&
                            DATATYPE(?requiredValue) = xsd:decimal &&
                            ?receiverValue < ?requiredValue
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:airReceiverServesAdditionalPurpose ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applicability value must indicate that the air receiver serves no additional purpose."
            ]
        ]
    ) .
</END_SHACL>
