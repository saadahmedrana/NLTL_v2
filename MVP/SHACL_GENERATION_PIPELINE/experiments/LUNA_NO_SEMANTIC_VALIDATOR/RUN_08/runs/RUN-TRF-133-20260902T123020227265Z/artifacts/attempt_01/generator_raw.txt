gen:TRF133ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:nodeKind sh:IRI ;
        sh:node gen:IceClassDraughtMarkingShape ;
        sh:minCount 1 ;
        sh:message "An ice-class draught marking conforming to the requirement is required."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:nodeKind sh:IRI ;
        sh:node gen:WarningTriangleMarkingShape ;
        sh:minCount 1 ;
        sh:message "A warning-triangle marking conforming to the requirement is required."
    ] ;
    sh:or (
        gen:TRF133TimberApplicableShape
        gen:TRF133TimberNotApplicableShape
    ) .
