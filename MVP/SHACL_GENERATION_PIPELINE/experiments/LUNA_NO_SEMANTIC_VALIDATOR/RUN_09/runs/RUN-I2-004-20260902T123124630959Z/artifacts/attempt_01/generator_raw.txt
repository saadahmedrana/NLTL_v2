<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_004_StaticShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must specify one upper ice waterline displacement in kilotonnes."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Each upper ice waterline case must specify one displacement in kilotonnes."
            ]
        ] ;
        sh:message "The ship must link at least one upper ice waterline case."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline displacement must equal the greatest displacement among the linked upper ice waterline cases." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>

SELECT $this
WHERE {
    $this nltl:upperIceWaterlineDisplacement ?duiNode .
    ?duiNode qudt:numericValue ?duiValue .

    FILTER (
        EXISTS {
            $this nltl:hasUpperIceWaterlineCase ?greaterCase .
            ?greaterCase nltl:waterlineCaseDisplacement ?greaterNode .
            ?greaterNode qudt:numericValue ?greaterValue .
            FILTER (?greaterValue > ?duiValue)
        }
        ||
        NOT EXISTS {
            $this nltl:hasUpperIceWaterlineCase ?matchingCase .
            ?matchingCase nltl:waterlineCaseDisplacement ?matchingNode .
            ?matchingNode qudt:numericValue ?matchingValue .
            FILTER (?matchingValue = ?duiValue)
        }
    )
}
"""
    ] .
<END_SHACL>
