@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_021_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Thickness must equal the ice-load net plate thickness plus the corrosion and abrasion allowance, with millimetre operands converted to metres; comparison tolerance is relative 1e-6 with a 1e-6 metre fallback near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceLoadRequiredNetPlateThickness ?netNode ;
                              nltl:corrosionAbrasionAllowance ?allowanceNode ;
                              nltl:thickness ?thicknessNode .
                        ?netNode qudt:numericValue ?net ;
                                 qudt:unit unit:MilliM .
                        ?allowanceNode qudt:numericValue ?allowance ;
                                         qudt:unit unit:MilliM .
                        ?thicknessNode qudt:numericValue ?actual ;
                                       qudt:unit unit:M .
                        BIND ((?net + ?allowance) / 1000 AS ?expected)
                        FILTER (
                            ABS(?actual - ?expected) >
                            IF(
                                ABS(?expected) < 0.000001,
                                0.000001,
                                0.000001 * ABS(?expected)
                            )
                        )
                    }
                """
            ]
        ]
    ] .
