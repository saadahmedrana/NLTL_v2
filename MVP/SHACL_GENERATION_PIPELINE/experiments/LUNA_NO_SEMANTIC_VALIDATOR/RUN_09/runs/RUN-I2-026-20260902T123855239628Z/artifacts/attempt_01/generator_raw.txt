<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:FramingSupportShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node nltl:FramingMemberShape ;
        sh:message "Each represented structural member must satisfy the framing support conditions."
    ] .

nltl:FramingMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:node nltl:MemberSupportCaseShape ;
        sh:minCount 1 ;
        sh:message "A represented framing member must have a support with a permitted support condition."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member's termination within the ice-strengthened area must be stated."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node nltl:FixedMemberSupportShape
            ]
        ]
    ) ;
    sh:message "A member terminating within an ice-strengthened area must have fixed support."

nltl:MemberSupportCaseShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:connectionBracketPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:connectionBracketPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:demonstratedRotationalRestraint ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:simpleFrameSupport
            ]
        ]
    ) ;
    sh:message "Support must be fixed for continuous framing, a connection bracket, or demonstrated significant rotational restraint; otherwise it must be simple support."

nltl:FixedMemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:SimpleMemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:simpleFrameSupport ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
