                      nltl:peakPressureFactor ?peakFactorQ ;
                      nltl:shearArea ?actualAreaQ ;
                      nltl:span ?spanQ ;

                ?heightQ qudt:numericValue ?height .
                ?peakFactorQ qudt:numericValue ?peakFactor .
                ?actualAreaQ qudt:numericValue ?actualArea .
                ?spanQ qudt:numericValue ?span .
                ?yieldQ qudt:numericValue ?yield .

                ?member nltl:frameSpacing ?spacingQ ;
                        nltl:selectedHullAreaFactor ?areaFactorQ ;
                        nltl:requiredTransverseFrameShearArea ?requiredAreaQ .

                ?spacingQ qudt:numericValue ?spacing .
                ?areaFactorQ qudt:numericValue ?areaFactor .
                ?requiredAreaQ qudt:numericValue ?reportedRequiredArea .

                BIND(IF(?span < ?height, ?span, ?height) AS ?ll)
                BIND((100 * 100 * 0.5 * ?ll * ?spacing *
                      (?areaFactor * ?peakFactor * ?pressure) /
                      (0.577 * ?yield)) AS ?calculatedRequiredAreaCm2)

                FILTER (
                    ?actualArea * 10000 < ?calculatedRequiredAreaCm2
                    ||
                    ABS(?reportedRequiredArea - ?calculatedRequiredAreaCm2)
                    > (
                        0.000001 * ABS(?calculatedRequiredAreaCm2)
                        + 0.000001
                    )
                )
            }
        """
