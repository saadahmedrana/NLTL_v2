<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A represented structural member must state whether it is longitudinal."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A represented structural member must identify its structural location."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:LoadCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each longitudinal frame in a side structure, the combined shear-and-bending demand must not exceed either the plastic strength or the midspan plastic collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:hasStructuralMemberLoadCase ?case .

                ?case nltl:combinedShearAndBendingDemand ?demand ;
                      nltl:plasticStrength ?plasticStrength ;
                      nltl:midspanPlasticCollapseLoad ?midspanLoad .

                ?demand qudt:numericValue ?demandValue .
                ?plasticStrength qudt:numericValue ?plasticStrengthValue .
                ?midspanLoad qudt:numericValue ?midspanValue .

                FILTER (
                    ?demandValue > ?plasticStrengthValue ||
                    ?demandValue > ?midspanValue
                )
            }
        """
    ] .

gen:LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DemandQuantityShape ;
        sh:message "Each represented load case must provide one combined shear-and-bending demand."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:message "Each represented load case must provide one plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:message "Each represented load case must provide one midspan plastic collapse load."
    ] .

gen:DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
