                      nltl:loadPatchHeight/qudt:numericValue ?a ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;

                      nltl:yieldStrength/qudt:numericValue ?sigmaY ;

                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reportedAL .

                BIND (?b / ?s AS ?bprime)
                BIND (IF (?bprime < 2, ?b * (1 - 0.25 * ?bprime), ?s) AS ?b2)

                    ABS(?reportedAL - ?expectedAL) >
                    (0.000001 * ABS(?expectedAL) + 0.000000001)
                    ||
                    ?actualAw < (?expectedAL * 0.0001)
                )
            }
        """
