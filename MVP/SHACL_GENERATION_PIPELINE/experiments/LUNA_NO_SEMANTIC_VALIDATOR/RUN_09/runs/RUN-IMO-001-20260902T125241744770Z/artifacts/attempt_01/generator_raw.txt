<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship design ice condition is required to determine applicability of the category rule."
    ] ;
    sh:or (
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:mediumFirstYearIceWithPossibleOldIceInclusions
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:polarShipCategoryValue ;
                sh:minCount 1 ;
                sh:hasValue nltl:polarShipCategoryA ;
                sh:message "A ship designed for at least medium first-year ice shall have Polar Code category A."
            ]
        ]
        [
            a sh:NodeShape ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:designIceCondition ?condition .
                        FILTER NOT EXISTS {
                            $this nltl:designIceCondition
                                nltl:mediumFirstYearIceWithPossibleOldIceInclusions .
                        }
                    }
                """ ;
                sh:message "The non-applicable branch requires a design ice condition other than at least medium first-year ice."
            ]
        ]
    ] .
<END_SHACL>
