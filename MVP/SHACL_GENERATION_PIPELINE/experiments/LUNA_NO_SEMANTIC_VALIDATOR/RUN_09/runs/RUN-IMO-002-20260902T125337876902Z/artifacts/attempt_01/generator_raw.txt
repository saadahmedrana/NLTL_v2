<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-002-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare exactly one design ice condition." ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare exactly one Polar Code ship category." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship that is not category A and is designed for at least thin first-year ice with possible old-ice inclusions must have category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition nltl:thinFirstYearIceWithPossibleOldIceInclusions ;
                      nltl:shipCategory ?category .
                FILTER (
                    ?category != nltl:polarShipCategoryA &&
                    ?category != nltl:polarShipCategoryB
                )
            }
        """ ;
    ] .
<END_SHACL>
