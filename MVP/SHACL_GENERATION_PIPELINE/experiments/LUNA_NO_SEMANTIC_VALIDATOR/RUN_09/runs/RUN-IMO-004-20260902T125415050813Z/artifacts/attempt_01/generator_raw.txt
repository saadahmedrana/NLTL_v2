<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_004_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:node [
            sh:class nltl:iceOriginValue
        ]
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:node [
            sh:class nltl:iceTypeValue
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceOriginType ;
                sh:minCount 1 ;
                sh:hasValue nltl:seaIceOrigin
            ] ;
            sh:property [
                sh:path nltl:winterGrowthCount ;
                sh:minCount 1 ;
                sh:datatype xsd:integer ;
                sh:maxInclusive 1
            ] ;
            sh:property [
                sh:path nltl:iceThickness ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:minInclusive "0.3"^^xsd:decimal ;
                        sh:maxInclusive "2.0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:iceTypeClassification ;
                sh:minCount 1 ;
                sh:hasValue nltl:firstYearIce
            ]
        ]
        [
            sh:sparql [
                sh:message "When the ship has sea ice of no more than one winter growth and thickness from 0.3 m through 2.0 m, it must be classified as first-year ice." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:iceOriginType nltl:seaIceOrigin ;
                               nltl:winterGrowthCount ?growth ;
                               nltl:iceThickness ?thickness .
                        ?thickness qudt:numericValue ?value ;
                                   qudt:unit unit:M .
                        FILTER (
                            datatype(?growth) = xsd:integer &&
                            datatype(?value) = xsd:decimal &&
                            ?growth <= 1 &&
                            ?value >= 0.3 &&
                            ?value <= 2.0
                        )
                        FILTER NOT EXISTS {
                            $this nltl:iceTypeClassification nltl:firstYearIce .
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
