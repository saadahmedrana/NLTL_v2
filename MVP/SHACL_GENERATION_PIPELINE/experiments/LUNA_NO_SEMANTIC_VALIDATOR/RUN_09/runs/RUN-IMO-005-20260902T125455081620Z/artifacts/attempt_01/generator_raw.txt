<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "The ice condition 'ice free waters' is permitted only when total ice present is false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (STR(?condition) = "ice free waters")
                FILTER NOT EXISTS {
                    $this nltl:totalIcePresent ?present .
                    FILTER (?present = false)
                }
            }
        """
    ] .
<END_SHACL>
