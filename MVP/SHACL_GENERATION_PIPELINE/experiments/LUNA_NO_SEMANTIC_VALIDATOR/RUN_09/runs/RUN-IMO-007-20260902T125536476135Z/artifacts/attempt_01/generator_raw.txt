<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO007OpenWaterShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "The ship must record an ice condition classification."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an open-water classification, sea ice concentration must be a unitless decimal less than 1 and ice of land origin must be recorded as false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceConditionClassification nltl:openWaterIceCondition .
    FILTER (
        NOT EXISTS {
            $this nltl:iceOfLandOriginPresent ?landOrigin .
            FILTER (
                datatype(?landOrigin) = xsd:boolean &&
                ?landOrigin = false
            )
        }
        ||
        NOT EXISTS {
            $this nltl:seaIceConcentration ?concentration .
            ?concentration qudt:numericValue ?value .
            ?concentration qudt:unit qudt:UNITLESS .
            FILTER (
                datatype(?value) = xsd:decimal &&
                ?value < 1
            )
        }
        ||
        EXISTS {
            $this nltl:seaIceConcentration ?concentration .
            ?concentration qudt:numericValue ?value .
            FILTER (
                datatype(?value) != xsd:decimal ||
                ?value >= 1
            )
        }
        ||
        EXISTS {
            $this nltl:seaIceConcentration ?concentration .
            ?concentration qudt:unit ?unit .
            FILTER (?unit != qudt:UNITLESS)
        }
    )
}
"""
    ] .
<END_SHACL>
