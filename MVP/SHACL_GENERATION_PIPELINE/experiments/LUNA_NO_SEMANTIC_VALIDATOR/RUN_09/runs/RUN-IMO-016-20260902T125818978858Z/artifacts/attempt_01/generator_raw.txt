<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO016Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:certificateFormModel ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:appendixIPolarShipCertificateModel ;
                sh:message "The Polar Ship Certificate shall use the Appendix I model."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Polar Ship Certificate language must be recorded."
    ] ;
    sh:sparql [
        sh:message "If the certificate language is not English, French, or Spanish, an approved translation must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (?language NOT IN ("English", "French", "Spanish"))
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent true
                }
            }
        """
    ] .
<END_SHACL>
