<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_055_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Exposed isolating, pressure, or vacuum valves shall have ice accretion protection and remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:valveExposed ?exposed ;
                      nltl:valveType ?type .
                FILTER (
                    datatype(?exposed) = xsd:boolean &&
                    ?exposed = true &&
                    (
                        ?type = "isolating" ||
                        ?type = "pressure" ||
                        ?type = "vacuum"
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceAccretionProtectionPresent ?protected .
                        FILTER (
                            datatype(?protected) = xsd:boolean &&
                            ?protected = true
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:continuouslyAccessible ?accessible .
                        FILTER (
                            datatype(?accessible) = xsd:boolean &&
                            ?accessible = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
