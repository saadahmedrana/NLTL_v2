<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_068_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipOperatesInIceCoveredWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The ship's ice-covered-water operating condition is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:directEvacuationOntoIceApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Whether direct evacuation onto ice is applicable is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:safeEvacuationMeansPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:survivalEquipmentDeploymentApprovalStatus ;
        sh:class nltl:evidenceState
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When operating in ice-covered waters or evacuating directly onto the ice, safe evacuation means must be present and survival equipment deployment approval status must be approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipOperatesInIceCoveredWaters ?operates ;
                      nltl:directEvacuationOntoIceApplicable ?direct .
                FILTER (?operates = true || ?direct = true)
                {
                    FILTER NOT EXISTS {
                        $this nltl:safeEvacuationMeansPresent ?safe .
                        FILTER (?safe = true)
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:survivalEquipmentDeploymentApprovalStatus
                            nltl:evidenceStateApproved .
                    }
                }
            }
        """
    ] .
<END_SHACL>
