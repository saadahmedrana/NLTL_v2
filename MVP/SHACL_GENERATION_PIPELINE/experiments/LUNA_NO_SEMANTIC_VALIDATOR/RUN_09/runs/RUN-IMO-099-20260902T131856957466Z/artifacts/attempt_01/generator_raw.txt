<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO099Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:batteryLifeManagementProcedurePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery life management procedure must be present."
    ] ;
    sh:property [
        sh:path nltl:procedureImplementationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The battery life management procedure must be implemented."
    ] ;
    sh:property [
        sh:path nltl:communicationDeviceAvailability ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DayQuantityShape ;
        sh:message "Communication device availability must be supplied in days."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DayQuantityShape ;
        sh:message "The maximum expected rescue time must be supplied in days."
    ] ;
    sh:sparql [
        sh:message "Communication device availability must be at least the maximum expected rescue time." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:communicationDeviceAvailability ?availabilityNode ;
                      nltl:maximumExpectedRescueTime ?rescueNode .
                ?availabilityNode qudt:numericValue ?availabilityValue .
                ?rescueNode qudt:numericValue ?rescueValue .
                FILTER (
                    datatype(?availabilityValue) = xsd:decimal &&
                    datatype(?rescueValue) = xsd:decimal &&
                    ?availabilityValue < ?rescueValue
                )
            }
        """
    ] .

gen:DayQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:DAY
    ] .
<END_SHACL>
