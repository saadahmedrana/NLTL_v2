<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO104ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship shall record completed PWOM familiarization."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gs:IMO104CrewMemberShape
    ] .

gs:IMO104CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:minCount 1 ;
        sh:node gs:IMO104AssignedDutyShape ;
        sh:message "Each crew member shall have an assigned duty linked to relevant PWOM items."
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:minCount 1 ;
        sh:node gs:IMO104FamiliarizationRecordShape ;
        sh:message "Each crew member shall have a PWOM familiarization record."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each PWOM procedure or equipment item relevant to a crew member's assigned duties shall be represented in that crew member's familiarization records." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    $this nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gs:IMO104AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:class nltl:polarWaterOperationalManualItem
    ] .

gs:IMO104FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:minCount 1 ;
        sh:class nltl:polarWaterOperationalManualItem
    ] .
<END_SHACL>
