<urn:nltl:generated-shape:IMO-120-Shape> a sh:NodeShape ;

            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For a non-Arctic operating area, operatingArea must not be Arctic_waters." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:operatingArea ?area .
                        FILTER (DATATYPE(?area) = xsd:string)
                        FILTER (?area != "Arctic_waters")
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:foodWasteDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:hasValue "Arctic_waters" ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:foodWasteDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;

                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minInclusive "12"^^xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MI_N ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:nearbyIceConcentration ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;

                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;

                        sh:maxCount 1

                ]
            ] ;
            sh:property [
                sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
                sh:minCount 1 ;

                    sh:class qudt:QuantityValue ;

                        sh:path qudt:numericValue ;

                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;

                ]
            ]
        ]
