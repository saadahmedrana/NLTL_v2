<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:generated-shape:TRF-005> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:sparql [
        sh:message "Ships whose construction stage was on or after 2003-09-01 and whose construction contract was before 2010-01-01 must identify the 2002 Ice Class Regulations as the applicable regulation edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate ;
                      nltl:constructionContractDate ?contractDate .
                FILTER (DATATYPE(?stageDate) = xsd:date)
                FILTER (DATATYPE(?contractDate) = xsd:date)
                FILTER (
                    YEAR(?stageDate) > 2003 ||
                    (YEAR(?stageDate) = 2003 &&
                     (MONTH(?stageDate) > 9 ||
                      (MONTH(?stageDate) = 9 && DAY(?stageDate) >= 1)))
                )
                FILTER (
                    YEAR(?contractDate) < 2010 ||
                    (YEAR(?contractDate) = 2010 &&
                     (MONTH(?contractDate) = 1 && DAY(?contractDate) < 1))
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRegulationEdition2002 .
                }
            }
        """
    ] .
</END_SHACL>
