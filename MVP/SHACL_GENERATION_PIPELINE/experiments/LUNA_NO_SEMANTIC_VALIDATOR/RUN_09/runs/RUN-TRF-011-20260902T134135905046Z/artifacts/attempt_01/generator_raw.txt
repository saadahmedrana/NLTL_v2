<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
        ;
        sh:message "The ship must identify the intended ice-operating waterline data needed to establish the UIWL envelope."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
        ;
        sh:message "The ship must identify its upper ice waterline."
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal, at every longitudinal position, the maximum vertical coordinate among the intended ice-operating waterline points, in both directions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT ?this
            WHERE {
                {
                    {
                        SELECT ?this ?x (MAX(?iv) AS ?maxv)
                        WHERE {
                            ?this nltl:hasIntendedIceOperatingWaterline ?iw .
                            ?iw nltl:hasWaterlineProfilePoint ?ip .
                            ?ip nltl:longitudinalPosition ?ixq ;
                                nltl:verticalCoordinate ?ivq .
                            ?ixq qudt:numericValue ?x ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?ivq qudt:numericValue ?iv ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                        GROUP BY ?this ?x
                    }
                    FILTER NOT EXISTS {
                        ?this nltl:hasUpperIceWaterline ?uw .
                        ?uw nltl:hasWaterlineProfilePoint ?up .
                        ?up nltl:longitudinalPosition ?uxq ;
                            nltl:verticalCoordinate ?uvq .
                        ?uxq qudt:numericValue ?x ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?uvq qudt:numericValue ?uv ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (
                            ABS(?uv - ?maxv) <=
                            IF(
                                ABS(?maxv) * 0.000001 > 0.000000001,
                                ABS(?maxv) * 0.000001,
                                0.000000001
                            )
                        )
                    }
                }
                UNION
                {
                    ?this nltl:hasUpperIceWaterline ?uw .
                    ?uw nltl:hasWaterlineProfilePoint ?up .
                    ?up nltl:longitudinalPosition ?uxq ;
                        nltl:verticalCoordinate ?uvq .
                    ?uxq qudt:numericValue ?x ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?uvq qudt:numericValue ?uv ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER NOT EXISTS {
                        ?this nltl:hasIntendedIceOperatingWaterline ?iw2 .
                        ?iw2 nltl:hasWaterlineProfilePoint ?ip2 .
                        ?ip2 nltl:longitudinalPosition ?ixq2 .
                        ?ixq2 qudt:numericValue ?x ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                }
                UNION
                {
                    ?this nltl:hasUpperIceWaterline ?uw .
                    ?uw nltl:hasWaterlineProfilePoint ?up .
                    ?up nltl:longitudinalPosition ?uxq ;
                        nltl:verticalCoordinate ?uvq .
                    ?uxq qudt:numericValue ?x ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?uvq qudt:numericValue ?uv ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    {
                        SELECT ?this ?x (MAX(?iv2) AS ?maxv2)
                        WHERE {
                            ?this nltl:hasIntendedIceOperatingWaterline ?iw3 .
                            ?iw3 nltl:hasWaterlineProfilePoint ?ip3 .
                            ?ip3 nltl:longitudinalPosition ?ixq3 ;
                                nltl:verticalCoordinate ?ivq3 .
                            ?ixq3 qudt:numericValue ?x ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?ivq3 qudt:numericValue ?iv2 ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                        GROUP BY ?this ?x
                    }
                    FILTER (
                        ABS(?uv - ?maxv2) >
                        IF(
                            ABS(?maxv2) * 0.000001 > 0.000000001,
                            ABS(?maxv2) * 0.000001,
                            0.000000001
                        )
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-ProfilePointShape>
    ] .

<urn:nltl:generated-shape:TRF-011-ProfilePointShape>
    a sh:NodeShape ;
    sh:targetClass nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    ] .

<urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
