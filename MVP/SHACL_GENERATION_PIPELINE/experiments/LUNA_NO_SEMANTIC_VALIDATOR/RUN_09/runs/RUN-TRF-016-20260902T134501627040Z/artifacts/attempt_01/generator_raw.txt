<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF016_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue true
            ] ;
            sh:node <urn:nltl:generated-shape:TRF016_ApplicableShape>
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue false
            ]
        ]
    ) .

<urn:nltl:generated-shape:TRF016_ApplicableShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:displacementAtUpperIceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF016_MassQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF016_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:forwardDraught ;
        sh:node <urn:nltl:generated-shape:TRF016_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:liwlDraughtAmidships ;
        sh:node <urn:nltl:generated-shape:TRF016_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:node <urn:nltl:generated-shape:TRF016_LengthQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerHighestPointSubmerged ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:class nltl:iceWaterline ;
        sh:node <urn:nltl:generated-shape:TRF016_UpperIceWaterlineShape> ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasBallastTank ;
        sh:class nltl:ballastTank ;
        sh:node <urn:nltl:generated-shape:TRF016_BallastTankShape>
    ] ;
    sh:sparql [
        sh:message "When navigating in ice, amidships draught must reach the LIWL draught and forward draught must meet the applicable level-ice-thickness threshold; displacement must be taken from a greatest-displacement UIWL." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:displacementAtUpperIceWaterline ?displacementQuantity ;
                      nltl:draughtAmidships ?draughtQuantity ;
                      nltl:forwardDraught ?forwardQuantity ;
                      nltl:liwlDraughtAmidships ?liwlQuantity ;
                      nltl:levelIceThickness ?thicknessQuantity ;
                      nltl:hasUpperIceWaterline ?upperWaterline .

                ?displacementQuantity <http://qudt.org/schema/qudt/numericValue> ?displacement ;
                                      <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/TON_Metric> .
                ?draughtQuantity <http://qudt.org/schema/qudt/numericValue> ?draught ;
                               <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?forwardQuantity <http://qudt.org/schema/qudt/numericValue> ?forward ;
                                <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?liwlQuantity <http://qudt.org/schema/qudt/numericValue> ?liwl ;
                             <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?thicknessQuantity <http://qudt.org/schema/qudt/numericValue> ?thickness ;
                                   <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .

                ?upperWaterline nltl:waterlineDisplacement ?waterlineDisplacement .
                ?waterlineDisplacement <http://qudt.org/schema/qudt/numericValue> ?waterlineMass ;
                                        <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/TON_Metric> .

                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterline ?greaterWaterline .
                    ?greaterWaterline nltl:waterlineDisplacement ?greaterDisplacement .
                    ?greaterDisplacement <http://qudt.org/schema/qudt/numericValue> ?greaterMass ;
                                         <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/TON_Metric> .
                    FILTER (?greaterMass > ?waterlineMass)
                }

                BIND(
                    IF(
                        ((2 + (0.00025 * ?displacement)) * ?thickness) < (4 * ?thickness),
                        ((2 + (0.00025 * ?displacement)) * ?thickness),
                        (4 * ?thickness)
                    )
                    AS ?minimumForwardDraught
                )

                FILTER (
                    ?displacement != ?waterlineMass ||
                    ?draught < ?liwl ||
                    ?forward < ?minimumForwardDraught
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF016_UpperIceWaterlineShape>

        sh:node <urn:nltl:generated-shape:TRF016_MassQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
