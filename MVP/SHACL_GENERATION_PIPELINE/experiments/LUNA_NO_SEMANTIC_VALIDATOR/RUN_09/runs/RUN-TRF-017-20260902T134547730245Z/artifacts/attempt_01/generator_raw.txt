<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF017Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propulsionOutputRestrictionApplies ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:propulsionOutputRestrictionApplies ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:restrictedPropulsionOutput ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node g:PowerQuantityShape
            ] ;
            sh:sparql [
                sh:message "When propulsion output is restricted, maximum continuous rating power shall equal the restricted propulsion output." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    SELECT $this
                    WHERE {
                        $this nltl:maximumContinuousRatingPower ?maximumQuantity .
                        ?maximumQuantity qudt:numericValue ?maximumValue ;
                                          qudt:unit unit:KiloW .
                        $this nltl:restrictedPropulsionOutput ?restrictedQuantity .
                        ?restrictedQuantity qudt:numericValue ?restrictedValue ;
                                            qudt:unit unit:KiloW .
                        FILTER (?maximumValue != ?restrictedValue)
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Maximum continuous rating power shall equal the maximum continuous propulsion machinery output." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?maximumQuantity .
                ?maximumQuantity qudt:numericValue ?maximumValue ;
                                  qudt:unit unit:KiloW .
                $this nltl:propulsionMachineryContinuousOutput ?outputQuantity .
                ?outputQuantity qudt:numericValue ?outputValue ;
                                qudt:unit unit:KiloW .
                FILTER (?maximumValue != ?outputValue)
            }
        """
    ] .

g:PowerQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
