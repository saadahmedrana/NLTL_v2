<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF018Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF018ApplicableShape
        gen:TRF018NonApplicableShape
    ) ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximumContinuousRatingPower must be at least the greater of calculatedRequiredPower and 1000 kW; for IA Super, it must be at least the greater of calculatedRequiredPower and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedQuantity ;
                      nltl:maximumContinuousRatingPower ?ratingQuantity .

                ?calculatedQuantity qudt:numericValue ?calculatedValue .
                ?ratingQuantity qudt:numericValue ?ratingValue .

                FILTER (
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        )
                        && (
                            ?ratingValue < ?calculatedValue
                            || ?ratingValue < 1000
                        )
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper
                        && (
                            ?ratingValue < ?calculatedValue
                            || ?ratingValue < 2800
                        )
                    )
                )
            }
        """
    ] .

gen:TRF018ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property gen:TRF018CalculatedPowerProperty ;
    sh:property gen:TRF018MaximumRatingPowerProperty .

gen:TRF018NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF018CalculatedPowerProperty
    sh:path nltl:calculatedRequiredPower ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:class qudt:QuantityValue ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        a sh:NodeShape ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:nodeKind sh:IRI ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW>
        ]
    ] .

gen:TRF018MaximumRatingPowerProperty
    sh:path nltl:maximumContinuousRatingPower ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:class qudt:QuantityValue ;
    sh:minCount 1 ;
    sh:maxCount 1 ;
    sh:node [
        a sh:NodeShape ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:nodeKind sh:IRI ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:hasValue <http://qudt.org/vocab/unit/KiloW>
        ]
    ] .
<END_SHACL>
