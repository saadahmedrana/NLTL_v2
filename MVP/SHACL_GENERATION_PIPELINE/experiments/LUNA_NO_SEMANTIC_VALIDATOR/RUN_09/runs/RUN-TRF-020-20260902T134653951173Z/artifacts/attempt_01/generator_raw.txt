<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF020ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single construction stage date is required to select the applicable branch."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single controlled ice class value is required to select the applicable branch."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:node gen:TRF020DimensionlessQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:node gen:TRF020UnspecifiedQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:node gen:TRF020UnspecifiedQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:node gen:TRF020LengthQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:class nltl:propellerPitchControlTypeValue ;
                sh:in (
                    nltl:controllablePitch
                    nltl:fixedPitch
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:class nltl:propulsionSystemTypeValue ;
                sh:in (
                    nltl:conventionalPropulsionSystem
                    nltl:electricPropulsionMachinery
                    nltl:hydraulicPropulsionMachinery
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:node gen:TRF020LowerWaterlineShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:node gen:TRF020UpperWaterlineShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node gen:TRF020PowerQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:node gen:TRF020PowerQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:node gen:TRF020PowerQuantityShape ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For the applicable ice class and construction date, the externally calculated minimum-power results and required input data must be present." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass .
                        FILTER (
                            ?date < "2003-09-01"^^xsd:date ||
                            ?iceClass NOT IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            )
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Maximum continuous rating power must be at least both externally calculated minimum powers." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    SELECT $this
                    WHERE {
                        $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerQuantity ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline ?upperQuantity .
                        ?mcrQuantity qudt:numericValue ?mcr .
                        ?lowerQuantity qudt:numericValue ?lower .
                        ?upperQuantity qudt:numericValue ?upper .
                        FILTER (?mcr < ?lower || ?mcr < ?upper)
                    }
                """
            ]
        ]
        [
            sh:sparql [
                sh:message "The non-applicable branch requires a construction date before 1 September 2003 or an ice class outside IA Super, IA, IB, and IC." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass .
                        FILTER (
                            ?date >= "2003-09-01"^^xsd:date &&
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            )
                        )
                    }
                """
            ]
        ]
    ) .

gen:TRF020PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF020LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF020DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF020UnspecifiedQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF020LowerWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline .

gen:TRF020UpperWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:node gen:TRF020LengthQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineBreadth ;
        sh:node gen:TRF020LengthQuantityShape ;
        sh:minCount 1
    ] .
<END_SHACL>
