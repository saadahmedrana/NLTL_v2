<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship of ice class IB or IC subject to the 1985 ice class regulations, maximum continuous rating power must be at least the required power under the 1985 rules, with both quantities expressed in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:applicableIceClassRegulationEdition
                    nltl:iceClassRuleEdition1985 ;
                    nltl:iceClass ?iceClass .

                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
                        nltl:requiredPowerUnder1985Rules ?requiredQuantity .

                    ?mcrQuantity qudt:numericValue ?mcrValue ;
                        qudt:unit qudt:KiloW .
                    ?requiredQuantity qudt:numericValue ?requiredValue ;
                        qudt:unit qudt:KiloW .

                    FILTER (
                        datatype(?mcrValue) = xsd:decimal &&
                        datatype(?requiredValue) = xsd:decimal &&
                        ?mcrValue >= ?requiredValue
                    )
                }
            }
        """
    ] .
<END_SHACL>
