@prefix gen: <urn:nltl:generated-shape:> .

@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF027Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF027ApplicableShape
        gen:TRF027NonApplicableShape
    ) ;
    sh:message "The ship must identify exactly one permitted ice class and construction stage date; applicable IA and IA Super ships must satisfy the Table 3-3 and clamped-geometry constraints." .

gen:TRF027ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
        )
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:maxInclusive "2003-08-31"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:shipLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:C3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:C4QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC5 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:C5QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F1QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:F4QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G1QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:G3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:clampedHullGeometryTerm ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:sparql [
        sh:message "clampedHullGeometryTerm must equal ((shipLength times iceClassDraught) divided by shipBreadth squared) cubed, clamped to the inclusive range 5 to 20, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:clampedHullGeometryTerm ?resultNode .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .
                ?breadthNode qudt:numericValue ?breadth .
                ?resultNode qudt:numericValue ?reported .
                BIND (((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth)) *
                      ((?length * ?draught) / (?breadth * ?breadth))) AS ?raw)
                BIND (IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw)) AS ?expected)
                FILTER (ABS(?reported - ?expected) >
                        ((1e-6 * ABS(?expected)) + 1e-9))
            }
        """
    ] .

gen:TRF027NonApplicableShape

    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:minInclusive "2003-09-01"^^xsd:date
            ]
        ]
    ) .
