<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must contain one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A quantity value must contain one unit."
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The pressure must be expressed in MPa."
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length must be expressed in metres."
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:node gen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The area factor must be dimensionless."
    ] .

gen:DirectAnalysisCaseBaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must state its load patch length."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must state its vertical load position."
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must state its horizontal load position."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineReferencePosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must identify the UIWL reference position."
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterlineReferencePosition ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must identify the LIWL reference position."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:class nltl:loadPositionTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must identify its vertical load position type."
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:class nltl:loadPositionTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must identify its horizontal load position type."
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The case must confirm a capacity-minimizing load position."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The case must confirm evaluation of combined bending and shear."
    ] .

gen:DirectAnalysisCaseWithAreaFactorShape
    a sh:NodeShape ;
    sh:node gen:DirectAnalysisCaseBaseShape ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each alternative load-length case must state its corresponding ca factor."
    ] .

gen:DirectAnalysisCaseWithoutAreaFactorShape
    a sh:NodeShape ;
    sh:node gen:DirectAnalysisCaseBaseShape .

gen:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the externally determined ice pressure."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the load patch height."
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether load length was determined from the arrangement."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the externally computed applied ice pressure."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:node gen:DirectAnalysisCaseWithoutAreaFactorShape ;
                sh:minCount 2 ;
                sh:message "Several direct analysis cases are required."
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:node gen:DirectAnalysisCaseWithAreaFactorShape ;
                sh:minCount 2 ;
                sh:message "When arrangement does not determine load length, several paired load-length and ca cases are required."
            ]
        ]
    ) .
<END_SHACL>
