<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_032_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Ice strengthening status is required to determine whether the classification society scantling rule applies."
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The scantling derived from the regulations is required."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The classification society required scantling is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue false
            ] ;
            sh:sparql [
                sh:message "For a ship that is not ice strengthened, the regulation-derived scantling must not be less than the classification society required scantling unless the classification society scantling is selected." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:regulationRequiredScantling ?regulationQuantity .
                        ?regulationQuantity qudt:numericValue ?regulationValue .
                        $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                        ?classificationQuantity qudt:numericValue ?classificationValue .
                        FILTER (!(?regulationValue >= ?classificationValue))
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:selectedDesignScantling ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "When the regulation-derived scantling is less than the classification society required scantling for a non-ice-strengthened ship, the selected design scantling must equal the classification society required scantling." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:regulationRequiredScantling ?regulationQuantity .
                        ?regulationQuantity qudt:numericValue ?regulationValue .
                        $this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                        ?classificationQuantity qudt:numericValue ?classificationValue .
                        $this nltl:selectedDesignScantling ?selectedQuantity .
                        ?selectedQuantity qudt:numericValue ?selectedValue .
                        FILTER (?regulationValue < ?classificationValue)
                        FILTER (?selectedValue != ?classificationValue)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
