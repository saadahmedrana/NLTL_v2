<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-047> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A main frame or intermediate ice frame must have a strengthened part whose lower end is attached to an eligible deck, tank plating, tank top, or ice stringer, or satisfies the permitted horizontal-member connection alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this a ?frameType .
                FILTER (?frameType IN (nltl:mainFrame, nltl:intermediateIceFrame))

                FILTER NOT EXISTS {
                    {
                        $this nltl:hasStrengthenedPart ?part .
                        ?part nltl:hasLowerEnd ?lowerEnd .
                        ?lowerEnd nltl:hasAttachedSupportingStructure ?support .
                        ?support a ?supportType .
                        FILTER (?supportType IN (
                            nltl:deckStructure,
                            nltl:tankBoundaryPlating,
                            nltl:tankTop,
                            nltl:iceStringer
                        ))
                    }
                    UNION
                    {
                        $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
                        $this nltl:hasStrengthenedPart ?part2 .
                        ?part2 nltl:hasLowerEnd ?lowerEnd2 .
                        ?lowerEnd2 nltl:hasTerminationBelowSupportingStructure ?belowSupport .
                        ?belowSupport a ?belowSupportType .
                        FILTER (?belowSupportType IN (
                            nltl:deckStructure,
                            nltl:tankBoundaryPlating,
                            nltl:tankTop,
                            nltl:iceStringer
                        ))
                        ?lowerEnd2 nltl:hasHorizontalConnectionMember ?member .
                        ?member nltl:connectsToAdjacentMainFrame ?adjacentMain .
                        ?adjacentMain a nltl:mainFrame .
                        $this nltl:sameScantlingsAsMainFrame true .
                        ?adjacentMain nltl:mainFrameBelowIceBeltStrengthened true
                    }
                }
            }
        """
    ] .
<END_SHACL>
