<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:TRF048_LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:continuousBeamWithBracketsFrameCondition
            nltl:bulkCarrierTopWingTankFrameCondition
            nltl:multiDeckOrStringerContinuousFrameCondition
            nltl:singleDeckTankTopToMainDeckFrameCondition
            nltl:twoDeckFrameCondition
        ) ;
        sh:message "A permitted frame boundary condition type is required."
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Significantly different boundary conditions must be specified."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:message "Ice pressure in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "Design ice load height in metres is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "Frame spacing in metres is required."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "Frame span in metres is required."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:DimensionlessQuantityShape ;
        sh:message "The frame moment factor m is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:message "Yield strength in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:DimensionlessQuantityShape ;
        sh:message "The calculated load distribution factor f4 is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:SectionModulusQuantityShape ;
        sh:message "The calculated section modulus is required."
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltlgen:AreaQuantityShape ;
        sh:message "The calculated required shear area is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:continuousBeamWithBracketsFrameCondition
            ] ;
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node nltlgen:Exact13Point3QuantityShape
            ]
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:significantlyDifferentBoundaryConditions ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:frameBoundaryConditionType ;
                        sh:in (
                            nltl:bulkCarrierTopWingTankFrameCondition
                            nltl:multiDeckOrStringerContinuousFrameCondition
                            nltl:singleDeckTankTopToMainDeckFrameCondition
                            nltl:twoDeckFrameCondition
                        )
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        sh:message "f4, section modulus, and required shear area must satisfy the declared formulas within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?syNode ;
                      nltl:longitudinalFrameLoadDistributionFactorF4 ?f4Node ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?syNode qudt:numericValue ?sy .
                ?f4Node qudt:numericValue ?f4 .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND (1 - (0.2 * ?h / ?s) AS ?expectedF4)
                BIND ((?expectedF4 * ?p * ?h * ?l * ?l /
                       (4 * ?m * ?sy)) * 1000000 AS ?expectedZ)
                BIND ((1.7320508075688772 * ?expectedF4 * 2.16 * ?p * ?h * ?l /
                       (2 * ?sy)) * 10000 AS ?expectedA)

                FILTER (
                    ABS(?f4 - ?expectedF4) >
                        (ABS(?expectedF4) * 0.000001 + 0.000001)
                    ||
                    ABS(?z - ?expectedZ) >
                        (ABS(?expectedZ) * 0.000001 + 0.000001)
                    ||
                    ABS(?a - ?expectedA) >
                        (ABS(?expectedA) * 0.000001 + 0.000001)
                )
            }
        """
    ] .

nltlgen:QuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:PressureQuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:LengthQuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:AreaQuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM2> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:SectionModulusQuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:QuantityValueShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM3> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:Exact13Point3QuantityShape
    a sh:NodeShape ;
    sh:node nltlgen:DimensionlessQuantityShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 13.3 ;
        sh:maxInclusive 13.3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
