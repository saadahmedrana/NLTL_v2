<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-049-FrameShape> a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node <urn:nltl:generated-shape:TRF-049-FrameAttachmentShape> ;
        sh:minCount 1 ;
        sh:message "A frame must have an effective frame attachment."
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:node <urn:nltl:generated-shape:TRF-049-SupportingStructureShape>
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:node <urn:nltl:generated-shape:TRF-049-SupportingStructureShape>
    ] ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node <urn:nltl:generated-shape:TRF-049-FrameWebThicknessShape> ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "A frame running through a supporting structure must have both web-plate connection sides recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount ?count .
                    FILTER (datatype(?count) = xsd:integer && ?count = 2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each longitudinal frame must be attached by a bracket to every supporting web frame and bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:longitudinalFrame .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar attachment construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:transverseFrame .
                $this nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every bracket must be at least as thick as the attached frame web." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?frameThicknessNode .
                    ?frameThicknessNode qudt:numericValue ?frameThickness .
                    ?frameThicknessNode qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?bracket nltl:bracketThickness ?bracketThicknessNode .
                    ?bracketThicknessNode qudt:numericValue ?bracketThickness .
                    ?bracketThicknessNode qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (?bracketThickness >= ?frameThickness)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-049-FrameAttachmentShape> a sh:NodeShape ;

        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;

        sh:path nltl:hasAttachedSupportingStructure ;

    ] .

    sh:property [

        sh:minCount 1 ;

        sh:message "An installed bracket must have a single thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
    ] .

        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

<urn:nltl:generated-shape:TRF-049-BracketThicknessShape> a sh:NodeShape ;

    ] ;
    sh:property [

    ] .

<urn:nltl:generated-shape:TRF-049-FrameEndShape> a sh:NodeShape ;

    ] .
<END_SHACL>
