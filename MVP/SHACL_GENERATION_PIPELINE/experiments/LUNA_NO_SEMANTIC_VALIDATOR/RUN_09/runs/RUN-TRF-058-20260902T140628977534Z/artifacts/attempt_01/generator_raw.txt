<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF058WebFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:iceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:ForceQuantityShape
            ] ;
            sh:property [
                sh:path nltl:stringerOutsideIceBeltHeight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:stringerSpan ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:PositiveLengthQuantityShape
            ] ;
            sh:property [
                sh:path nltl:adjustedIceLoadForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:ForceQuantityShape
            ] ;
            sh:sparql [
                sh:message "When the supported stringer is outside the ice belt, adjustedIceLoadForce must equal iceLoadForce multiplied by (1 - stringerOutsideIceBeltHeight / stringerSpan), within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9 near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:supportedStringerOutsideIceBelt true ;
                              nltl:iceLoadForce ?forceQuantity ;
                              nltl:stringerOutsideIceBeltHeight ?heightQuantity ;
                              nltl:stringerSpan ?spanQuantity ;
                              nltl:adjustedIceLoadForce ?adjustedQuantity .

                        ?forceQuantity qudt:numericValue ?force .
                        ?heightQuantity qudt:numericValue ?height .
                        ?spanQuantity qudt:numericValue ?span .
                        ?adjustedQuantity qudt:numericValue ?adjusted .

                        FILTER (
                            ABS(
                                ?adjusted -
                                (?force * (1 - (?height / ?span)))
                            )
                            >
                            (
                                0.000001 *
                                ABS(?force * (1 - (?height / ?span)))
                                + 0.000000001
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportedStringerOutsideIceBelt ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:PositiveLengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
