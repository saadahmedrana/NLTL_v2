<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_080_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceLoadCycleCount is required as a single integer value."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "propellerBladeCount is required as a single integer value."
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "effectiveIceLoadCycleCount must provide one unitless decimal quantity value."
    ] ;
    sh:sparql [
        sh:message "effectiveIceLoadCycleCount must equal iceLoadCycleCount multiplied by propellerBladeCount within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?iceCycles ;
                      nltl:propellerBladeCount ?bladeCount ;
                      nltl:effectiveIceLoadCycleCount ?effectiveQuantity .
                ?effectiveQuantity <http://qudt.org/schema/qudt/numericValue> ?actual .
                BIND((?iceCycles * ?bladeCount) AS ?expected)
                FILTER (
                    ABS(?actual - ?expected) >
                    IF(
                        ABS(?expected) > 0,
                        ABS(?expected) * "0.000001"^^xsd:decimal,
                        "0.000001"^^xsd:decimal
                    )
                )
            }
        """
    ] .
<END_SHACL>
