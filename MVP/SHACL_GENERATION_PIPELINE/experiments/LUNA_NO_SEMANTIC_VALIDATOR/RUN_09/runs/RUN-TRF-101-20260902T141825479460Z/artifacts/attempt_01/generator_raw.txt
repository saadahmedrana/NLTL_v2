<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BladeFailureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Propeller diameter is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade tip offset is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade failure ultimate load is required in kN."
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:node gen:BladeFailureLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "A blade-failure load case is required."
    ] ;
    sh:or (
        gen:FormulaBranchShape
        gen:NonlinearAnalysisBranchShape
    ) ;
    sh:sparql [
        sh:message "Blade failure status must indicate failure when tip offset exceeds 10% of propeller diameter and non-failure otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameterNode ;
                      nltl:bladeTipOffset ?offsetNode ;
                      nltl:bladeFailureStatus ?status .
                ?diameterNode qudt:numericValue ?diameter .
                ?offsetNode qudt:numericValue ?offset .
                FILTER (
                    (?offset > (0.1 * ?diameter) && ?status != true) ||
                    (?offset <= (0.1 * ?diameter) && ?status != false)
                )
            }
        """
    ] .

gen:FormulaBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade chord length at the weakest section is required for the formula calculation."
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade cylindrical root radius is required for the formula calculation."
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade maximum thickness at the weakest section is required for the formula calculation."
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade minimum yield strength is required for the formula calculation."
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Blade ultimate tensile strength is required for the formula calculation."
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The reference blade strength is required for the formula calculation."
    ] ;
    sh:sparql [
        sh:message "Blade failure ultimate load and reference strength must satisfy the prescribed calculation, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?dNode ;
                      nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:bladeMinimumYieldStrength ?yieldNode ;
                      nltl:bladeUltimateTensileStrength ?ultimateNode ;
                      nltl:bladeReferenceStrength1 ?referenceNode ;
                      nltl:bladeFailureUltimateLoad ?loadNode .
                ?dNode qudt:numericValue ?d .
                ?cNode qudt:numericValue ?c .
                ?tNode qudt:numericValue ?t .
                ?rNode qudt:numericValue ?r .
                ?yieldNode qudt:numericValue ?yield .
                ?ultimateNode qudt:numericValue ?ultimate .
                ?referenceNode qudt:numericValue ?reference .
                ?loadNode qudt:numericValue ?load .
                BIND((0.6 * ?yield + 0.4 * ?ultimate) AS ?expectedReference)
                BIND((300 * ?c * ?t * ?t * ?expectedReference) / (0.8 * ?d - 2 * ?r) AS ?expectedLoad)
                FILTER (
                    (ABS(?reference - ?expectedReference) > (ABS(?expectedReference) * 0.000001)
                     && ABS(?reference - ?expectedReference) > 0.000000001)
                    ||
                    (ABS(?load - ?expectedLoad) > (ABS(?expectedLoad) * 0.000001)
                     && ABS(?load - ?expectedLoad) > 0.000000001)
                )
            }
        """
    ] .

gen:NonlinearAnalysisBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:approvedNonlinearBladeStressAnalysis ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An approved nonlinear blade stress analysis is required when the direct formula is not used."
    ] .

gen:BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The load case must apply the load at a radius ratio of 0.8."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The blade failure load must act in the weakest direction."
    ] ;
    sh:sparql [
        sh:message "The blade failure load application radius ratio must equal 0.8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeFailureLoadApplicationRadiusRatio ?ratioNode .
                ?ratioNode qudt:numericValue ?ratio .
                FILTER (?ratio < 0.8 || ?ratio > 0.8)
            }
        """
    ] .
