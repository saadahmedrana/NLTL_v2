<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF111Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the externally determined extreme ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:extremeOperationalLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:minInclusive 1.3
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue qudt:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:bladeFailureLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:componentYieldSafetyFactor ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:minInclusive 1.0
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue qudt:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shaftLineLoadCaseType ;
                        sh:hasValue nltl:fatigueLoadCase ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ] ;
        sh:message "Each propulsion load case must identify an allowed shaft-line load-case type and satisfy its applicable readiness checks."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:node [
            sh:class nltl:propellerShaftLineComponent
        ]
    ] ;
    sh:sparql [
        sh:message "When a fatigue load case is applicable, each linked shaft-line component must provide a fatigue safety factor in the required unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor qudt:numericValue ?value ;
                            qudt:unit qudt:UNITLESS .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A fatigue safety factor must use the unitless unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                ?component nltl:fatigueSafetyFactor ?factor .
                ?factor qudt:unit ?unit .
                FILTER (?unit != qudt:UNITLESS)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A fatigue safety factor numeric value must be an xsd:decimal." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                $this nltl:hasPropellerShaftLineComponent ?component .
                ?component nltl:fatigueSafetyFactor ?factor .
                ?factor qudt:numericValue ?value .
                FILTER (datatype(?value) != xsd:decimal)
            }
        """
    ] .
<END_SHACL>
