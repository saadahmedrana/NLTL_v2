    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;

                sh:hasValue nltl:traficomTable6Dash3Reference ;

            ] ;
            sh:property [
                sh:path nltl:impactLoadCaseTableReference ;
                sh:hasValue nltl:traficomTable6Dash16Reference ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must reference the TRAFICOM Table 6-16 load case."
            ] ;
            sh:property [
                sh:path nltl:thrusterImpactCaseAppliesToBody ;
                sh:node [
                    sh:class nltl:thrusterBody ;
                    sh:property [
                        sh:path nltl:thrusterResistanceCapacity ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The impact case must identify the applicable thruster body and its externally assessed resistance capacity."

            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "The impact case must provide evidence for the loaded area assessment."
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactLoad ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;

                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:thrusterIceImpactDemand ;

                    ] ;
                    sh:property [
                        sh:path qudt:unit ;

                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;

                    }
