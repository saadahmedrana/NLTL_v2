<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-127-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:generated-shape:TRF-127-NonApplicableBranch>
        <urn:nltl:generated-shape:TRF-127-ApplicableBranch>
    ) .

<urn:nltl:generated-shape:TRF-127-NonApplicableBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:airReceiverServesAdditionalPurpose ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:message "The air receiver additional-purpose applicability value must be explicitly recorded as false when the rule is non-applicable."
    ] .

<urn:nltl:generated-shape:TRF-127-ApplicableBranch>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:airReceiverServesAdditionalPurpose ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The air receiver additional-purpose applicability value must be explicitly recorded as true when additional capacity is required."
    ] ;
    sh:property [
        sh:path nltl:airReceiverCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "An air receiver capacity in cubic metres is required when the receiver serves an additional purpose."
    ] ;
    sh:property [
        sh:path nltl:additionalPurposeRequiredAirCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "The additional-purpose required air capacity is required when the receiver serves an additional purpose."
    ] ;
    sh:sparql [
        sh:message "The air receiver capacity must be at least the capacity required for additional purposes." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:airReceiverServesAdditionalPurpose true .
                $this nltl:airReceiverCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                $this nltl:additionalPurposeRequiredAirCapacity ?required .
                ?required qudt:numericValue ?requiredValue .
                FILTER (
                    datatype(?capacityValue) = xsd:decimal &&
                    datatype(?requiredValue) = xsd:decimal &&
                    ?capacityValue < ?requiredValue
                )
            }
        """
    ] .
<END_SHACL>
