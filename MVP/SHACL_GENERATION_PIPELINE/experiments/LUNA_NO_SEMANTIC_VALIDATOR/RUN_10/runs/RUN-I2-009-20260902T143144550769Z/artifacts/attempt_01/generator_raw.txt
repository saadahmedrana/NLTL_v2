            ]

        sh:message "Average ice pressure shall be provided as a single MPa quantity." ;
    ] ;
    sh:property [

        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;

                sh:minInclusive 3.6 ;
                sh:maxInclusive 3.6
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;

            ]
        ] ;
        sh:message "The load patch aspect ratio shall be exactly 3.6 and dimensionless." ;
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:minCount 1 ;

        sh:message "The buttock angle shall be provided as a single degree quantity." ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When bow-form design ice forces are applicable, the buttock angle shall be greater than 0 and less than 80 degrees, and the normal frame angle shall be greater than 10 degrees." ;

PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:bowFormApplicability true ;
        nltl:buttockAngle ?gamma ;
        nltl:normalFrameAngle ?beta .
  ?gamma qudt:numericValue ?gammaValue .
  ?beta qudt:numericValue ?betaValue .
  FILTER (
    !(
      ?gammaValue > 0 &&
      ?gammaValue < 80 &&
      ?betaValue > 10
    )
  )
}
"""
    ] .
