gen:I2_014_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:node gen:BowSubregionCalculationCaseShape ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:message "A ship must provide exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The selected maximum bow force must be reported."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:node gen:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The selected maximum bow line load must be reported."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The selected maximum bow pressure must be reported."
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowForce must equal the maximum calculated force across the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForce .
                ?selectedForce qudt:numericValue ?selectedValue .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?force .
                        ?force qudt:numericValue ?value .
                        FILTER (?value = ?selectedValue)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedForce ?force2 .
                        ?force2 qudt:numericValue ?value2 .
                        FILTER (?value2 > ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowLineLoad must equal the maximum calculated line load across the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowLineLoad ?selectedLineLoad .
                ?selectedLineLoad qudt:numericValue ?selectedValue .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedLineLoad ?lineLoad .
                        ?lineLoad qudt:numericValue ?value .
                        FILTER (?value = ?selectedValue)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedLineLoad ?lineLoad2 .
                        ?lineLoad2 qudt:numericValue ?value2 .
                        FILTER (?value2 > ?selectedValue)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "selectedMaximumBowPressure must equal the maximum calculated pressure across the four bow subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowPressure ?selectedPressure .
                ?selectedPressure qudt:numericValue ?selectedValue .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedPressure ?pressure .
                        ?pressure qudt:numericValue ?value .
                        FILTER (?value = ?selectedValue)
                    }
                    ||
                    EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionCalculatedPressure ?pressure2 .
                        ?pressure2 qudt:numericValue ?value2 .
                        FILTER (?value2 > ?selectedValue)
                    }
                )
            }
        """
    ] .
