<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node g:I2_026_MemberShape
    ] .

g:I2_026_MemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:class nltl:memberSupport ;
        sh:node g:I2_026_SupportShape ;
        sh:message "Each structural member must identify a member support and its support condition."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node g:I2_026_FixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node g:I2_026_BracketFixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node g:I2_026_RestraintFixedSupportShape
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact ;
                sh:message "Demonstrated significant rotational restraint must have supporting evidence."
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node g:I2_026_FixedSupportShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node g:I2_026_SimpleSupportShape
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Simple support cannot be selected when continuous support, a connection bracket, demonstrated rotational restraint, or termination within the ice-strengthened area is present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:frameSupportCondition nltl:simpleFrameSupport .
                FILTER (
                    EXISTS {
                        $this nltl:continuousThroughSupport true .
                    }
                    ||
                    EXISTS {
                        ?support nltl:connectionBracketPresent true .
                    }
                    ||
                    EXISTS {
                        ?support nltl:demonstratedRotationalRestraint true .
                    }
                    ||
                    EXISTS {
                        $this nltl:terminatesWithinIceStrengthenedArea true .
                    }
                )
            }
        """
    ] .

g:I2_026_SupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:frameSupportConditionValue ;
        sh:or (
            [
                sh:hasValue nltl:fixedFrameSupport
            ]
            [
                sh:hasValue nltl:simpleFrameSupport
            ]
        ) ;
        sh:message "A member support must specify either fixed or simple frame support."
    ] .

g:I2_026_FixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] .

g:I2_026_BracketFixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

g:I2_026_RestraintFixedSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:fixedFrameSupport
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

g:I2_026_SimpleSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:simpleFrameSupport
    ] .
<END_SHACL>
