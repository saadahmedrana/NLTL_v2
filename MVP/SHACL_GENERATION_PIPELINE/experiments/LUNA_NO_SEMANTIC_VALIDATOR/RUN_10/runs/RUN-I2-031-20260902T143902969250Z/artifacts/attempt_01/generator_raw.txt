<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-031-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node <urn:nltl:generated-shape:I2-031-PatchLoadShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the patch load."
    ] .

<urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bottomStructureLocation
            nltl:sideStructureLocation
        ) ;
        sh:message "Each structural member must identify a bottom or side structure location."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2-031-PlasticStrengthShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural member must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node <urn:nltl:generated-shape:I2-031-MidspanPlasticCollapseLoadShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each structural member must provide its midspan plastic collapse load."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:class nltl:loadCase ;
        sh:node <urn:nltl:generated-shape:I2-031-LoadCaseShape> ;
        sh:minCount 1 ;
        sh:message "Each structural member must have a load case."
    ] ;
    sh:sparql [
        sh:message "For a bottom structural member, the patch-load dimension b must be parallel with the frame direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:structuralLocation nltl:bottomStructureLocation .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    ?loadCase nltl:loadPatchOrientation
                              nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The combined shear and bending demand must not exceed the plastic strength of the member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                ?strengthQuantity qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .

                FILTER (
                    xsd:decimal(?demandValue) >
                    xsd:decimal(?strengthValue)
                )
            }
        """
    ] .

<urn:nltl:generated-shape:I2-031-LoadCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:loadCase ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue ;
        sh:maxCount 1 ;
        sh:message "A load case must not contain conflicting patch-load orientations."
    ] ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node <urn:nltl:generated-shape:I2-031-CombinedDemandShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each load case must provide the combined shear and bending demand."
    ] .

<urn:nltl:generated-shape:I2-031-PatchLoadShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-PlasticStrengthShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-MidspanPlasticCollapseLoadShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-CombinedDemandShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
