gen:I2_032_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AverageIcePressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FrameSpanQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LoadPatchHeightQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PeakPressureFactorQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ShearAreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SpanQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:YieldStrengthQuantityShape
    ] ;
    sh:sparql [
        sh:message "Actual net effective shear area must be at least the calculated required transverse frame shear area; the reported required area must match the calculation within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pavg ;
                      nltl:frameSpan ?frameSpan ;
                      nltl:loadPatchHeight ?patchHeight ;
                      nltl:peakPressureFactor ?ppf ;
                      nltl:shearArea ?shearArea ;
                      nltl:yieldStrength ?yieldStrength ;
                      nltl:hasStructuralMember ?member .

                ?pavg qudt:numericValue ?pavgValue .
                ?frameSpan qudt:numericValue ?frameSpanValue .
                ?patchHeight qudt:numericValue ?patchHeightValue .
                ?ppf qudt:numericValue ?ppfValue .
                ?shearArea qudt:numericValue ?shearAreaValue .
                ?yieldStrength qudt:numericValue ?yieldStrengthValue .

                ?member nltl:frameSpacing ?spacing ;
                        nltl:selectedHullAreaFactor ?af ;
                        nltl:requiredTransverseFrameShearArea ?requiredArea .

                ?spacing qudt:numericValue ?spacingValue .
                ?af qudt:numericValue ?afValue .
                ?requiredArea qudt:numericValue ?requiredAreaValue .

                ?hullAreaValue a nltl:hullAreaValue ;
                               nltl:hullAreaFactor ?hullAreaFactor .
                ?hullAreaFactor qudt:numericValue ?hullAreaFactorValue .

                BIND(IF(?frameSpanValue < ?patchHeightValue,
                        ?frameSpanValue,
                        ?patchHeightValue) AS ?llValue)
                BIND((10000 * 0.5 * ?llValue * ?spacingValue *
                      (?afValue * ?ppfValue * ?pavgValue) /
                      (0.577 * ?yieldStrengthValue)) AS ?expectedAreaValue)

                FILTER(
                    ABS(?requiredAreaValue - ?expectedAreaValue) >
                    (ABS(?expectedAreaValue) * 0.000001 + 0.000000001)
                    ||
                    (?shearAreaValue * 10000 < ?expectedAreaValue)
                )
            }
        """
    ] .
