<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_048_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one supported Polar Class."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node gen:I2_048_CaseShape
    ] .

gen:I2_048_CaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must identify exactly one assessed plating."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each steel-grade requirement case must have one as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must specify one supported steel material class."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must specify one supported steel strength category."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeE
            nltl:steelGradeAh
            nltl:steelGradeDh
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each steel-grade requirement case must specify exactly one supported actual hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:steelGradeB
            nltl:steelGradeD
            nltl:steelGradeE
            nltl:steelGradeAh
            nltl:steelGradeDh
            nltl:steelGradeEh
            nltl:steelGradeFh
        ) ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required grade must be selected from Table 8 using Polar Class, thickness, material class, and strength category; non-applicable combinations must not have a required grade, and the actual grade must be at least the required grade within its category." ;
        sh:select """PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  ?ship nltl:hasSteelGradeRequirementCase $this ;
        nltl:polarClass ?pc .
  $this nltl:asBuiltPlateThickness ?thicknessNode ;
        nltl:steelMaterialClass ?material ;
        nltl:steelStrengthCategory ?category ;
        nltl:actualHullStructuralSteelGrade ?actual .
  ?thicknessNode qudt:numericValue ?thickness ;
                 qudt:unit <http://qudt.org/vocab/unit/MilliM> .
  OPTIONAL { $this nltl:requiredHullStructuralSteelGrade ?required . }

  BIND(
    IF(?pc IN (nltl:polarClassPc1, nltl:polarClassPc2, nltl:polarClassPc3),
       "PC1-3",
       IF(?pc IN (nltl:polarClassPc4, nltl:polarClassPc5),
          "PC4-5",
          IF(?pc IN (nltl:polarClassPc6, nltl:polarClassPc7),
             "PC6-7", "UNKNOWN")))
    AS ?polarGroup
  )

  BIND(
    IF(
      ?thickness <= 50,
      IF(
        ?category = nltl:normalStrengthSteelCategory,
        IF(
          ?material = nltl:steelMaterialClassOne,
          IF(?thickness <= 15, nltl:steelGradeB,
             IF(?thickness <= 35, nltl:steelGradeD, nltl:steelGradeE)),
          IF(
            ?material = nltl:steelMaterialClassTwo,
            IF(?thickness <= 10, nltl:steelGradeB,
               IF(?thickness <= 25, nltl:steelGradeD, nltl:steelGradeE)),
            IF(
              ?polarGroup = "PC1-3",
              IF(?thickness <= 35, nltl:steelGradeE, ?actual),
              IF(
                ?polarGroup = "PC4-5",
                IF(?thickness <= 45, nltl:steelGradeE, ?actual),
                IF(?thickness <= 10, nltl:steelGradeB,
                   IF(?thickness <= 35, nltl:steelGradeD, nltl:steelGradeE))
              )
            )
          )
        ),
        IF(
          ?material IN (nltl:steelMaterialClassOne, nltl:steelMaterialClassTwo),
          IF(?thickness <= 10, nltl:steelGradeAh, nltl:steelGradeDh),
          IF(
            ?polarGroup = "PC1-3",
            IF(?thickness <= 10, nltl:steelGradeAh,
               IF(?thickness <= 35, nltl:steelGradeEh, nltl:steelGradeFh)),
            IF(
              ?polarGroup = "PC4-5",
              IF(?thickness <= 10, nltl:steelGradeAh,
                 IF(?thickness <= 45, nltl:steelGradeEh, nltl:steelGradeFh)),
              IF(?thickness <= 10, nltl:steelGradeAh, nltl:steelGradeEh)
            )
          )
        )
      ),
      ?actual
    )
    AS ?expected
  )

  BIND(
    ?category = nltl:normalStrengthSteelCategory &&
    ?material = nltl:steelMaterialClassThree &&
    (
      (?polarGroup = "PC1-3" && ?thickness > 35) ||
      (?polarGroup = "PC4-5" && ?thickness > 45)
    )
    AS ?notApplicableNormal
  )

  BIND(
    ?category = nltl:normalStrengthSteelCategory &&
    ?material = nltl:steelMaterialClassThree &&
    ?polarGroup IN ("PC1-3", "PC4-5") &&
    (
      (?polarGroup = "PC1-3" && ?thickness > 35) ||
      (?polarGroup = "PC4-5" && ?thickness > 45)
    )
    AS ?notApplicable
  )

  BIND(
    IF(?actual IN (nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE),
       IF(?actual = nltl:steelGradeB, 1,
          IF(?actual = nltl:steelGradeD, 2, 3)),
       IF(?actual = nltl:steelGradeAh, 1,
          IF(?actual = nltl:steelGradeDh, 2,
             IF(?actual = nltl:steelGradeEh, 3, 4))))
    AS ?actualRank
  )

  BIND(
    IF(?required = nltl:steelGradeB, 1,
       IF(?required = nltl:steelGradeD, 2,
          IF(?required = nltl:steelGradeE, 3,
             IF(?required = nltl:steelGradeAh, 1,
                IF(?required = nltl:steelGradeDh, 2,
                   IF(?required = nltl:steelGradeEh, 3,
                      IF(?required = nltl:steelGradeFh, 4, 0)))))))
    AS ?requiredRank
  )

  FILTER(
    !BOUND(?expected) ||
    (
      ?notApplicable &&
      BOUND(?required)
    ) ||
    (
      ! ?notApplicable &&
      (
        !BOUND(?required) ||
        ?required != ?expected ||
        (
          ?category = nltl:normalStrengthSteelCategory &&
          !(?actual IN (nltl:steelGradeB, nltl:steelGradeD, nltl:steelGradeE))
        ) ||
        (
          ?category = nltl:highTensileSteelCategory &&
          !(?actual IN (nltl:steelGradeAh, nltl:steelGradeDh, nltl:steelGradeEh, nltl:steelGradeFh))
        ) ||
        ?actualRank < ?requiredRank
      )
    )
  )
}
""" 
    ] .
</END_SHACL>
