<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Each interpolation point shall provide a numeric interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Each interpolation point shall provide a numeric interpolation value."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each interpolation calculation case shall provide the hull-girder longitudinal position from aft in metres."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each interpolation calculation case shall provide the longitudinal distribution factor."
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "Each interpolation calculation case shall identify its interpolation points."
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "Each interpolation calculation case shall identify a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node gen:InterpolationPointShape ;
        sh:message "Each interpolation calculation case shall identify an upper interpolation endpoint."
    ] .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node gen:CalculationCaseShape ;
        sh:message "The ship shall provide a calculation case for the externally determined interpolation."
    ] ;
    sh:property [
        sh:path nltl:interpolatedValue ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The ship shall provide the externally determined interpolated value."
    ] .
<END_SHACL>
