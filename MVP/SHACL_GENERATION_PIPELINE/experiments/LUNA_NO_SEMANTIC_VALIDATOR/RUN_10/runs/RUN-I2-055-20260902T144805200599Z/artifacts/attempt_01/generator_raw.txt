nltl:GeneratedShape_I2_055 a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:GeneratedShape_I2_055_DesignStressQuantity ;
        sh:message "A single design stress in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node nltl:GeneratedShape_I2_055_PermissibleStressQuantity ;
        sh:message "A single permissible stress in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The strength criteria must be satisfied."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Design stress must not exceed permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designStress ?designQuantity .
                ?designQuantity qudt:numericValue ?designValue .
                $this nltl:permissibleStress ?permissibleQuantity .
                ?permissibleQuantity qudt:numericValue ?permissibleValue .
                FILTER (xsd:decimal(?designValue) > xsd:decimal(?permissibleValue))
            }
        """
    ] .
