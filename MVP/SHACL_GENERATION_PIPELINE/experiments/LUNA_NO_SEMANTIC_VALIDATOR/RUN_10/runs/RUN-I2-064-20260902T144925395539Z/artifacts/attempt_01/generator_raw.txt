<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-064-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:class nltl:calculationMethodValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:or (
                    [
                        sh:hasValue nltl:linearCalculationMethodValue
                    ]
                    [
                        sh:hasValue nltl:nonlinearCalculationMethodValue
                    ]
                    [
                        sh:hasValue nltl:prescribedAnalyticalProcedure
                    ]
                ]
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node [
            sh:property [
                sh:path nltl:hasStructuralMemberFlange ;
                sh:class nltl:structuralMemberFlange
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberWeb ;
                sh:class nltl:structuralMemberWeb
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A linear calculation case requires yield strength, member web and flange stress results, and buckling-criteria evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                $this nltl:hasStructuralMember ?member .
                FILTER NOT EXISTS {
                    $this nltl:yieldStrength ?yield .
                    ?yield qudt:numericValue ?yieldValue ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    FILTER (datatype(?yieldValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    ?member nltl:bucklingCriteriaSatisfied true .
                }
                FILTER NOT EXISTS {
                    ?member nltl:hasStructuralMemberWeb ?web .
                    ?web nltl:nominalShearStress ?shear .
                    ?shear qudt:numericValue ?shearValue ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    FILTER (datatype(?shearValue) = xsd:decimal)
                }
                FILTER NOT EXISTS {
                    ?member nltl:hasStructuralMemberFlange ?flange .
                    ?flange nltl:nominalFlangeVonMisesStress ?stress .
                    ?stress qudt:numericValue ?stressValue ;
                            qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    FILTER (datatype(?stressValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Nominal shear stress in each member web must be less than yield strength divided by sqrt(3) for a linear calculation case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:yieldStrength ?yield ;
                      nltl:hasStructuralMember ?member .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                ?yield qudt:numericValue ?yieldValue ;
                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?shear .
                ?shear qudt:numericValue ?shearValue ;
                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (datatype(?yieldValue) = xsd:decimal)
                FILTER (datatype(?shearValue) = xsd:decimal)
                FILTER (?shearValue >= (?yieldValue / 1.7320508075688772))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Nominal flange von Mises stress must be less than 1.15 times yield strength for a linear calculation case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:yieldStrength ?yield ;
                      nltl:hasStructuralMember ?member .
                ?case nltl:calculationMethod nltl:linearCalculationMethodValue .
                ?yield qudt:numericValue ?yieldValue ;
                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stress .
                ?stress qudt:numericValue ?stressValue ;
                        qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (datatype(?yieldValue) = xsd:decimal)
                FILTER (datatype(?stressValue) = xsd:decimal)
                FILTER (?stressValue >= (1.15 * ?yieldValue))
            }
        """
    ] .
<END_SHACL>
