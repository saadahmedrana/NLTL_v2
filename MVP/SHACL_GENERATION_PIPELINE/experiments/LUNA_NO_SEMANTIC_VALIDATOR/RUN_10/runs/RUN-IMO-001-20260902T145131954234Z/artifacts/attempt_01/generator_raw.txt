<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_001_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship must declare a design ice condition to determine applicability of the category rule."
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:mediumFirstYearIceWithPossibleOldIceInclusions
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:polarShipCategoryA ;
                sh:message "A ship designed for at least medium first-year ice with possible old-ice inclusions must have Polar Ship Category A."
            ]
        ]
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceConditionLessSevereThanCategoryAAndB
                    nltl:iceFreeIceCondition
                    nltl:openWaterIceCondition
                    nltl:otherWatersIceCondition
                    nltl:thinFirstYearIceWithPossibleOldIceInclusions
                )
            ]
        ]
    ) .
</END_SHACL>
