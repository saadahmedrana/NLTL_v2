<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:sparql [
        sh:message "When the ship has sea ice with no more than one winter growth and ice thickness from 0.3 m through 2.0 m, its ice type classification must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                      nltl:winterGrowthCount ?growth ;
                      nltl:iceThickness ?thicknessNode .
                ?thicknessNode qudt:numericValue ?thickness ;
                                qudt:unit unit:M .
                FILTER (datatype(?growth) = xsd:integer)
                FILTER (?growth <= 1)
                FILTER (datatype(?thickness) = xsd:decimal)
                FILTER (?thickness >= 0.3 && ?thickness <= 2.0)
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
<END_SHACL>
