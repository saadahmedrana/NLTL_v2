<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-005-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must provide exactly one total ice presence value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice condition 'ice free waters' may be used only when total ice present is false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceCondition ?condition ;
                      nltl:totalIcePresent ?icePresent .
                FILTER (
                    STR(?condition) = "ice free waters" &&
                    (
                        ?icePresent != false ||
                        DATATYPE(?icePresent) != xsd:boolean
                    )
                )
            }
        """ ;
    ] .
<END_SHACL>
