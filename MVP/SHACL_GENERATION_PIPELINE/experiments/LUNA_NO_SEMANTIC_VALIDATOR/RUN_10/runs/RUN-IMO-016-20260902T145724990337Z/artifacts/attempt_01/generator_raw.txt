        sh:minCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;

    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:datatype xsd:string ;

            ]
        ]
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:minCount 1 ;
                sh:datatype xsd:string
            ] ;

                sh:datatype xsd:boolean ;
                sh:minCount 1 ;

            ]
        ]
    ) .
