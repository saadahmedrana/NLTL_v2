<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-017_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate form is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:minCount 1 ;
        sh:message "The applicable SOLAS certificate schedule is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate supplement is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement presence indicator must be true."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each Polar Ship Certificate validity, survey, or endorsement date must match a SOLAS certificate schedule date of the corresponding category." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this ?dateProperty ?dateValue .
    FILTER (
        ?dateProperty = nltl:polarCertificateValidityDates ||
        ?dateProperty = nltl:surveyDates ||
        ?dateProperty = nltl:endorsementDates
    )
    FILTER NOT EXISTS {
        $this nltl:hasSolasCertificateSchedule /
              nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDate ?dateValue ;
                nltl:certificateScheduleDateType ?dateType .
        FILTER (
            (?dateProperty = nltl:polarCertificateValidityDates &&
             ?dateType = nltl:validityDateCategory) ||
            (?dateProperty = nltl:surveyDates &&
             ?dateType = nltl:surveyDateCategory) ||
            (?dateProperty = nltl:endorsementDates &&
             ?dateType = nltl:endorsementDateCategory)
        )
    }
}
"""
    ] .

<urn:nltl:generated-shape:IMO-017_SupplementShape>
    a sh:NodeShape ;
    sh:targetClass nltl:polarCertificateSupplement ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Certificate supplement must contain required equipment records."
    ] .

<urn:nltl:generated-shape:IMO-017_ScheduleShape>
    a sh:NodeShape ;
    sh:targetClass nltl:solasCertificateSchedule ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:message "The SOLAS certificate schedule must contain certificate schedule date records."
    ] .

<urn:nltl:generated-shape:IMO-017_DateRecordShape>
    a sh:NodeShape ;
    sh:targetClass nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A certificate schedule date record must specify a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:or (
            [ sh:hasValue nltl:validityDateCategory ]
            [ sh:hasValue nltl:surveyDateCategory ]
            [ sh:hasValue nltl:endorsementDateCategory ]
        ) ;
        sh:message "A certificate schedule date record must have a validity, survey, or endorsement category."
    ] .
<END_SHACL>
