<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_041_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hydraulicFluidFreezingOrViscosityControlPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Hydraulically operated hatches or doors require hydraulic-fluid freezing or excessive-viscosity control." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuationType .
                FILTER (STR(?actuationType) = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent ?controlPresent .
                    FILTER (?controlPresent = true)
                }
            }
        """
    ] .
<END_SHACL>
