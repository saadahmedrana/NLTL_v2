<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:IMO-047:> .

gen:Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:message "Each linked essential engine must be represented as an essential engine."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TemperatureQuantityShape ;
        sh:message "A single combustion-air temperature quantity in degrees Celsius is required."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TemperatureQuantityShape ;
        sh:message "A single manufacturer minimum air-temperature limit in degrees Celsius is required."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TemperatureQuantityShape ;
        sh:message "A single manufacturer maximum air-temperature limit in degrees Celsius is required."
    ] ;
    sh:sparql [
        sh:message "Combustion-air temperature must be within the manufacturer minimum and maximum limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperatureNode .
                ?temperatureNode qudt:numericValue ?temperatureValue .
                $this nltl:manufacturerMinAirTemperature ?minimumNode .
                ?minimumNode qudt:numericValue ?minimumValue .
                $this nltl:manufacturerMaxAirTemperature ?maximumNode .
                ?maximumNode qudt:numericValue ?maximumValue .
                FILTER (
                    ?temperatureValue < ?minimumValue ||
                    ?temperatureValue > ?maximumValue
                )
            }
        """
    ] .

gen:TemperatureQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG_C>
    ] .
<END_SHACL>
