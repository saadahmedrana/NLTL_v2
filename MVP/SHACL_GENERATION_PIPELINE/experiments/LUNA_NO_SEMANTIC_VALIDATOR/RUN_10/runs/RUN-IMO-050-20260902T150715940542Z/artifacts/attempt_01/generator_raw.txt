<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO_050_Static> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fireSafetyComponentExposed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceAndSnowProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A fire-safety component recorded as exposed shall have ice and snow protection recorded as present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:fireSafetyComponentExposed ?exposed .
                FILTER(datatype(?exposed) = xsd:boolean)
                FILTER(?exposed = true)
                FILTER NOT EXISTS {
                    $this nltl:iceAndSnowProtectionPresent ?protected .
                    FILTER(datatype(?protected) = xsd:boolean)
                    FILTER(?protected = true)
                }
            }
        """
    ] .
<END_SHACL>
