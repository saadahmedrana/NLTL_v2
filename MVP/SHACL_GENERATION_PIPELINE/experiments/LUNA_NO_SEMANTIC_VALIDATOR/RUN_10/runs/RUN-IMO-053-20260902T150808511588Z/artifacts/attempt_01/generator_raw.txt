<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_053_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasExtinguishingMedium ;
        sh:node gen:IMO_053_ExtinguishingMediumItemShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an extinguishing medium must have intended operation compatibility status approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasExtinguishingMedium ?medium .
                FILTER NOT EXISTS {
                    $this nltl:intendedOperationCompatibilityStatus ?status .
                    FILTER (
                        datatype(?status) = xsd:boolean &&
                        ?status = true
                    )
                }
            }
        """
    ] .

gen:IMO_053_ExtinguishingMediumItemShape
    a sh:NodeShape ;
    sh:class nltl:extinguishingMediumItem .
<END_SHACL>
