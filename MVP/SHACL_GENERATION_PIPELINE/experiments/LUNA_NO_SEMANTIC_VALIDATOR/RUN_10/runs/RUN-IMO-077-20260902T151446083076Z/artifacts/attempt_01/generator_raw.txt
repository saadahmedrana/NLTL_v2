<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO077ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO077ApplicableShape
        gen:IMO077NotApplicableShape
    ) .

gen:IMO077ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:personsCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:additionalEquipmentCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadedSurvivalCraftRequirement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:survivalCraftAvailableCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:launchingApplianceCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Survival-craft or launching-appliance capacity is below the required loaded capacity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:personsCapacityRequirement ?personsValue .
                $this nltl:additionalEquipmentCapacityRequirement ?additionalValue .
                $this nltl:loadedSurvivalCraftRequirement ?loadedNode .
                ?loadedNode qudt:numericValue ?loadedValue .
                $this nltl:survivalCraftAvailableCapacity ?availableNode .
                ?availableNode qudt:numericValue ?availableValue .
                $this nltl:launchingApplianceCapacity ?launchNode .
                ?launchNode qudt:numericValue ?launchValue .
                FILTER (
                    ?availableValue < (?personsValue + ?additionalValue)
                    || ?launchValue < ?loadedValue
                )
            }
        """
    ] .

gen:IMO077NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
