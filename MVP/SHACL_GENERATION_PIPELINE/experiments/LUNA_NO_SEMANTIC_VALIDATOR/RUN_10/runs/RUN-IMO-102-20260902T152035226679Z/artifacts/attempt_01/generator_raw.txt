<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO102ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled ice condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one controlled polar training ship type."
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasCrewMemberInventory ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented crew member must have exactly one controlled polar training role."
    ] ;
    sh:sparql [
        sh:message "Each applicable crew member must have a matching required training level, training record level, and valid STCW qualification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                      nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role .

                FILTER (
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType
                        ) &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole,
                            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                        )
                    ) ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole
                        )
                    ) ||
                    (
                        ?ice = nltl:otherWatersIceCondition &&
                        ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
                    )
                )

                FILTER NOT EXISTS {
                    ?this nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?requiredLevel ;
                            nltl:trainingRecordLevel ?recordLevel ;
                            nltl:stcwQualificationValid true .

                    FILTER (
                        ?requiredLevel = ?recordLevel &&
                        (
                            (
                                ?ice = nltl:openWaterIceCondition &&
                                ?shipType IN (
                                    nltl:tankerPolarTrainingShipType,
                                    nltl:passengerShipPolarTrainingShipType
                                ) &&
                                ?requiredLevel = nltl:basicPolarTrainingLevel
                            ) ||
                            (
                                ?ice = nltl:otherWatersIceCondition &&
                                ?role IN (
                                    nltl:masterPolarTrainingRole,
                                    nltl:chiefMatePolarTrainingRole
                                ) &&
                                ?requiredLevel = nltl:advancedPolarTrainingLevel
                            ) ||
                            (
                                ?ice = nltl:otherWatersIceCondition &&
                                ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole &&
                                ?requiredLevel = nltl:basicPolarTrainingLevel
                            )
                        )
                    )
                }
            }
        """ ;
    ] .

gen:IMO102TrainingRecordShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasCrewTrainingRecord ;
    sh:targetObjectsOf nltl:hasPolarTrainingRecord ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:minCount 1 ;
        sh:message "A polar training record must state one controlled required training level."
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        ) ;
        sh:minCount 1 ;
        sh:message "A polar training record must state one controlled recorded training level."
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A polar training record must state whether the STCW qualification is valid."
    ] .
<END_SHACL>
