<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO112ArcticNlsDischargeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An operating area is required to determine whether the Arctic discharge prohibition applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "Arctic_waters"
            ] ;
            sh:property [
                sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false ;
                sh:message "Discharge of noxious liquid substances or mixtures containing them shall be recorded as prohibited in Arctic waters."
            ]
        ]
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:pattern "^(?!Arctic_waters$).*$"
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When the operating area includes Arctic_waters, the noxious liquid substance discharge record shall not contain a value other than false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                FILTER ( ?area = "Arctic_waters"^^xsd:string )
                FILTER ( ?discharge != false )
            }
        """
    ] ;
    sh:sparql [
        sh:message "When the operating area is Arctic_waters, a noxious liquid substance discharge record is required and shall be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER ( ?area = "Arctic_waters"^^xsd:string )
                FILTER NOT EXISTS {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                }
            }
        """
    ] .
<END_SHACL>
