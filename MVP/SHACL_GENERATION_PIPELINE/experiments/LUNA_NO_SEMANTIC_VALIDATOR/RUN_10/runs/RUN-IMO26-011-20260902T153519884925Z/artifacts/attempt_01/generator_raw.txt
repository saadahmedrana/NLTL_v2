<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_011_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO26_011_CategoryC
        gen:IMO26_011_CategoryAB_Before2026
        gen:IMO26_011_Applicable
    ) ;
    sh:message "For Category A or B ships constructed on or after 2026-01-01, bridge wings must be enclosed, have the required protection design, or have an Administration-approved equivalent arrangement." .

gen:IMO26_011_CategoryC
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryC ) ;
        sh:minCount 1
    ] .

gen:IMO26_011_CategoryAB_Before2026
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2025-12-31"^^xsd:date ;
        sh:minCount 1
    ] .

gen:IMO26_011_Applicable
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2026-01-01"^^xsd:date ;
        sh:minCount 1
    ] ;
    sh:or (
        gen:IMO26_011_Enclosed
        gen:IMO26_011_DesignProtected
        gen:IMO26_011_EquivalentProtected
    ) .

gen:IMO26_011_Enclosed
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bridgeWingsEnclosed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] .

gen:IMO26_011_DesignProtected
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bridgeWingProtectionDesignStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] .

gen:IMO26_011_EquivalentProtected
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:administrationApprovedEquivalentBridgeWingProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] .
<END_SHACL>
