<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO26_012_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to main power."
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to emergency power."
            ] ;
            sh:property [
                sh:path nltl:independentFromHeadingMeans ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:nonMagneticHeadingMeans ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall be independent from another heading means."
            ]
        ]
    ] ;
    sh:sparql [
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER (
                    NOT EXISTS {
                        ?means nltl:independentFromHeadingMeans ?other .
                        FILTER (?other != ?means)
                        $this nltl:hasNonMagneticHeadingMeans ?other .
                    }
                )
            }
        """ ;
        sh:message "Each represented non-magnetic heading means shall be independent from another distinct means owned by the same ship."
    ] .
<END_SHACL>
