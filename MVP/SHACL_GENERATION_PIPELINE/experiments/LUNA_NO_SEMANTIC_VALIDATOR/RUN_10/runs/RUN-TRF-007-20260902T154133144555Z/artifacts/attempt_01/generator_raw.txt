<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship must have an ice class to determine applicability of TRF-007."
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?ice .
                        FILTER (?ice NOT IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa
                        ))
                        FILTER NOT EXISTS {
                            ?this nltl:iceClass ?targetIce .
                            FILTER (?targetIce IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa
                            ))
                        }
                    }
                """ ;
                sh:message "The ship's ice class makes TRF-007 inapplicable."
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A construction stage date is required to determine TRF-007 applicability for ice class IA Super or IA."
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?ice ;
                              nltl:constructionStageDate ?stage .
                        FILTER (?ice IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa
                        ))
                        FILTER (
                            YEAR(?stage) > 2003 ||
                            (
                                YEAR(?stage) = 2003 &&
                                (
                                    MONTH(?stage) > 9 ||
                                    (MONTH(?stage) = 9 && DAY(?stage) >= 1)
                                )
                            )
                        )
                    }
                """ ;
                sh:message "The construction stage was on or after 1 September 2003, so the conditional TRF-007 obligation does not apply."
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A construction stage date is required to determine TRF-007 applicability."
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A delivery date is required to evaluate the TRF-007 deadline."
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An assessment date is required to evaluate the TRF-007 deadline."
            ] ;
            sh:property [
                sh:path nltl:engineOutputComplianceMethod ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:traficom2017Section3Point2Point2Method
                    nltl:traficom2017Section3Point2Point4Method
                ) ;
                sh:message "An applicable ship must identify compliance with section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations."
            ] ;
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT ?this
                    WHERE {
                        ?this nltl:iceClass ?ice ;
                              nltl:constructionStageDate ?stage ;
                              nltl:deliveryDate ?delivery ;
                              nltl:assessmentDate ?assessment .
                        FILTER (?ice IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa
                        ))
                        FILTER (
                            YEAR(?stage) < 2003 ||
                            (
                                YEAR(?stage) = 2003 &&
                                (
                                    MONTH(?stage) < 9 ||
                                    (MONTH(?stage) = 9 && DAY(?stage) < 1)
                                )
                            )
                        )
                        FILTER (
                            YEAR(?assessment) < YEAR(?delivery) + 20 ||
                            (
                                YEAR(?assessment) = YEAR(?delivery) + 20 &&
                                MONTH(?assessment) = 1 &&
                                DAY(?assessment) <= 1
                            )
                        )
                    }
                """ ;
                sh:message "The applicable TRF-007 compliance deadline has not been exceeded."
            ]
        ]
    ] .
<END_SHACL>
