<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "For an applicable ship, the assessment date must not be later than 1 January of the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?constructionStageDate ;
                              nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?constructionStageDate) = xsd:date)
                        FILTER (datatype(?deliveryDate) = xsd:date)
                        FILTER (datatype(?assessmentDate) = xsd:date)
                        FILTER (?constructionStageDate < "2003-09-01"^^xsd:date)
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                (
                                    MONTH(?assessmentDate) > 1 ||
                                    (
                                        MONTH(?assessmentDate) = 1 &&
                                        DAY(?assessmentDate) > 1
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        ]
        [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
    ) .
<END_SHACL>
