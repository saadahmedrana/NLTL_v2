<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF011_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF011_IceWaterlineShape> ;
        sh:message "The ship must identify the intended ice-operating waterline and its profile." ;
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF011_IceWaterlineShape> ;
        sh:message "The ship must identify the upper ice waterline and its profile." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must equal, pointwise, the maximum vertical coordinate of the intended ice-operating waterlines at each longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            {
                SELECT ?iw ?x (MAX(?yi) AS ?maxY)
                WHERE {
                    $this nltl:hasIntendedIceOperatingWaterline ?iw .
                    ?iw nltl:hasWaterlineProfilePoint ?ip .
                    ?ip nltl:longitudinalPosition ?xv .
                    ?xv qudt:numericValue ?x .
                    ?ip nltl:verticalCoordinate ?yiv .
                    ?yiv qudt:numericValue ?yi .
                }
                GROUP BY ?iw ?x
            }
            $this nltl:hasUpperIceWaterline ?uw .
            FILTER (
                NOT EXISTS {
                    ?uw nltl:hasWaterlineProfilePoint ?up .
                    ?up nltl:longitudinalPosition ?xuv .
                    ?xuv qudt:numericValue ?xu .
                    ?up nltl:verticalCoordinate ?yuv .
                    ?yuv qudt:numericValue ?yu .
                    FILTER (?xu = ?x && ?yu = ?maxY)
                }
                ||
                EXISTS {
                    ?uw nltl:hasWaterlineProfilePoint ?up2 .
                    ?up2 nltl:longitudinalPosition ?xuv2 .
                    ?xuv2 qudt:numericValue ?xu2 .
                    ?up2 nltl:verticalCoordinate ?yuv2 .
                    ?yuv2 qudt:numericValue ?yu2 .
                    FILTER (?xu2 = ?x && ?yu2 != ?maxY)
                }
            }

            UNION

            {
                $this nltl:hasUpperIceWaterline ?uwExtra .
                ?uwExtra nltl:hasWaterlineProfilePoint ?upExtra .
                ?upExtra nltl:longitudinalPosition ?xuvExtra .
                ?xuvExtra qudt:numericValue ?xExtra .
                FILTER NOT EXISTS {
                    $this nltl:hasIntendedIceOperatingWaterline ?iwExtra .
                    ?iwExtra nltl:hasWaterlineProfilePoint ?ipExtra .
                    ?ipExtra nltl:longitudinalPosition ?xivExtra .
                    ?xivExtra qudt:numericValue ?xExtra2 .
                    FILTER (?xExtra2 = ?xExtra)
                }
            }
        """ ;
    ] .

<urn:nltl:generated-shape:TRF011_IceWaterlineShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF011_ProfilePointShape> ;
        sh:message "Each ice waterline must provide its profile points." ;
    ] .

<urn:nltl:generated-shape:TRF011_ProfilePointShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF011_LengthQuantityShape> ;
        sh:message "Each waterline profile point must have one longitudinal position in metres." ;
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF011_LengthQuantityShape> ;
        sh:message "Each waterline profile point must have one vertical coordinate in metres." ;
    ] .

<urn:nltl:generated-shape:TRF011_LengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
