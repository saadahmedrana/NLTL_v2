@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF016Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether it is navigating in ice."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:node [
                    sh:property [
                        sh:path nltl:waterlineDisplacement ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:datatype xsd:decimal
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue qudt:TON_Metric
                            ]
                        ]
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:liwlDraughtAmidships ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:M
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:draughtAmidships ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:M
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:forwardDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:M
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:levelIceThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue qudt:M
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:propellerHighestPointSubmerged ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasBallastTank ;
                sh:node [
                    sh:property [
                        sh:path nltl:situatedAboveLowerIceWaterline ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:usedToReachLowerIceWaterline ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:sparql [
                        sh:message "A ballast tank situated above the LIWL and needed to reach the LIWL must have freezing prevention."
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                $this nltl:situatedAboveLowerIceWaterline true ;
                                      nltl:usedToReachLowerIceWaterline true .
                                FILTER NOT EXISTS {
                                    $this nltl:freezingPreventionPresent true .
                                }
                            }
                        """
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Draught amidships must be at least the LIWL draught amidships."
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:draughtAmidships ?draughtNode ;
                              nltl:liwlDraughtAmidships ?liwlNode .
                        ?draughtNode qudt:numericValue ?draught .
                        ?liwlNode qudt:numericValue ?liwl .
                        FILTER (?draught < ?liwl)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Forward draught must satisfy the UIWL displacement and level-ice-thickness formula."
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasUpperIceWaterline ?uiwl ;
                              nltl:levelIceThickness ?iceNode ;
                              nltl:forwardDraught ?forwardNode .
                        ?uiwl nltl:waterlineDisplacement ?dispNode .
                        ?dispNode qudt:numericValue ?delta .
                        ?iceNode qudt:numericValue ?hi .
                        ?forwardNode qudt:numericValue ?reported .
                        FILTER NOT EXISTS {
                            $this nltl:hasUpperIceWaterline ?otherUiwl .
                            ?otherUiwl nltl:waterlineDisplacement ?otherDispNode .
                            ?otherDispNode qudt:numericValue ?otherDelta .
                            FILTER (?otherDelta > ?delta)
                        }
                        BIND ((2.0 + (0.00025 * ?delta)) * ?hi AS ?uncapped)
                        BIND (4.0 * ?hi AS ?cap)
                        BIND (IF(?uncapped < ?cap, ?uncapped, ?cap) AS ?expected)
                        FILTER (
                            ABS(?reported - ?expected) >
                            ((0.000001 * ABS(?expected)) + 0.000000001)
                        )
                    }
                """
            ]
        ]
    ) .
