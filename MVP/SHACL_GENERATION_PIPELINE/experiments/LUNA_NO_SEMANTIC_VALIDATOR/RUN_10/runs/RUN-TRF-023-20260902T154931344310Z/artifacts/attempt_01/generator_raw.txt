<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship retaining ice class IB or IC must identify that the 1985 ice class regulations apply." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                         nltl:iceClassRegulationEdition2002 .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC ships subject to the 1985 ice class regulations, maximum continuous rating power must be at least the required power under the 1985 rules, with both quantities expressed in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRegulationEdition2002 .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?maximumPower .
                    ?maximumPower qudt:numericValue ?maximumValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    $this nltl:requiredPowerUnder1985Rules ?requiredPower .
                    ?requiredPower qudt:numericValue ?requiredValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    FILTER (
                        datatype(?maximumValue) = xsd:decimal &&
                        datatype(?requiredValue) = xsd:decimal &&
                        ?maximumValue >= ?requiredValue
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-023-MaximumPower> a sh:NodeShape ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:message "Maximum continuous rating power must be a quantity value in kW when supplied."
    ] .

<urn:nltl:generated-shape:TRF-023-RequiredPower> a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:message "Required power under the 1985 rules must be a quantity value in kW when supplied."
    ] .
<END_SHACL>
