<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-036-ice-parameters>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasIceClassDesignParameterSet
        ] ;
        sh:class nltl:ship ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path nltl:iceClass ;
                sh:class nltl:iceClassValue ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The design parameter set must be linked to exactly one ship with a permitted ice class."
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Level ice thickness must be provided as a single metre quantity."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design ice load height must be provided as a single metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Level ice thickness or design ice load height does not match the Table 4-1 value for the ship's ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                       nltl:iceClass ?iceClass .

                $this nltl:levelIceThickness ?levelQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity .

                ?levelQuantity qudt:numericValue ?levelValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?heightQuantity qudt:numericValue ?heightValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        (?levelValue > "1.0"^^xsd:decimal ||
                         ?heightValue < "0.35"^^xsd:decimal ||
                         ?heightValue > "0.35"^^xsd:decimal)
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIa &&
                        (?levelValue > "0.8"^^xsd:decimal ||
                         ?heightValue < "0.30"^^xsd:decimal ||
                         ?heightValue > "0.30"^^xsd:decimal)
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIb &&
                        (?levelValue > "0.6"^^xsd:decimal ||
                         ?heightValue < "0.25"^^xsd:decimal ||
                         ?heightValue > "0.25"^^xsd:decimal)
                    ) ||
                    (
                        ?iceClass = nltl:iceClassIc &&
                        (?levelValue > "0.4"^^xsd:decimal ||
                         ?heightValue < "0.22"^^xsd:decimal ||
                         ?heightValue > "0.22"^^xsd:decimal)
                    )
                )
            }
        """
    ] .
<END_SHACL>
