<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF037Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientK ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientB ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassFactorCp ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:nominalIcePressureP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:directAnalysisCase ;
            sh:property [
                sh:path nltl:iceLoadAreaFactorCa ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:DimensionlessQuantityShape
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputFactorCd ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:sparql [
        sh:message "The capped ship-size engine output factor or design ice pressure does not match its declared formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipSizeEngineOutputCoefficientA ?aNode ;
                      nltl:shipSizeEngineOutputCoefficientK ?kNode ;
                      nltl:shipSizeEngineOutputCoefficientB ?bNode ;
                      nltl:iceClassFactorCp ?cpNode ;
                      nltl:nominalIcePressureP0 ?p0Node ;
                      nltl:shipSizeEngineOutputFactorCd ?cdNode ;
                      nltl:icePressure ?pNode ;
                      nltl:hasDirectAnalysisCase ?case .

                ?case nltl:iceLoadAreaFactorCa ?caNode .
                ?aNode qudt:numericValue ?a .
                ?kNode qudt:numericValue ?k .
                ?bNode qudt:numericValue ?b .
                ?cpNode qudt:numericValue ?cp .
                ?p0Node qudt:numericValue ?p0 .
                ?caNode qudt:numericValue ?ca .
                ?cdNode qudt:numericValue ?cd .
                ?pNode qudt:numericValue ?p .

                BIND((?a * ?k + ?b) / 1000 AS ?rawCd)
                BIND(IF(?rawCd > 1, 1, ?rawCd) AS ?expectedCd)
                BIND(?expectedCd * ?cp * ?ca * ?p0 AS ?expectedPressure)

                FILTER(
                    ABS(?cd - ?expectedCd) >
                    ((0.000001 * ABS(?expectedCd)) + 0.000000001)
                    ||
                    ABS(?p - ?expectedPressure) >
                    ((0.000001 * ABS(?expectedPressure)) + 0.000000001)
                )
            }
        """
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
