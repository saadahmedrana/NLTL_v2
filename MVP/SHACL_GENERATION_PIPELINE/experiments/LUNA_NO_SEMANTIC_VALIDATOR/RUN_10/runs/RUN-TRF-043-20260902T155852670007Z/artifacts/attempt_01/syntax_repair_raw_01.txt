<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-043-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:node <urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    ] .

<urn:nltl:generated-shape:TRF-043-TerminationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:framingIceStrengtheningTerminationCase ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-043-MillimetreQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive 250
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minExclusive 250
                    ]
                ]
            ]
        ]
    ) .

<urn:nltl:generated-shape:TRF-043-MillimetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .

<urn:nltl:generated-shape:TRF-043-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-043-MetreQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-043-MetreQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing ice-strengthening extents must match the applicable Table 4-6 values and required reach conditions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below ;
                      nltl:upperBowIceBeltRequired ?upperRequired .
                ?above qudt:numericValue ?aboveValue ;
                       qudt:unit <http://qudt.org/vocab/unit/M> .
                ?below qudt:numericValue ?belowValue ;
                       qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper &&
                     ?aboveValue != 1.2 &&
                     ?belowValue != 2.0)
                    ||
                    (?iceClass = nltl:iceClassIaSuper &&
                     ?region = nltl:sternRegion &&
                     ?aboveValue != 1.2 &&
                     ?belowValue != 1.6)
                    ||
                    (?iceClass IN (nltl:iceClassIa, nltl:iceClassIb, nltl:iceClassIc) &&
                     ?region = nltl:bowRegion &&
                     (?aboveValue != 1.0 || ?belowValue != 1.6))
                    ||
                    (?iceClass IN (nltl:iceClassIa, nltl:iceClassIb, nltl:iceClassIc) &&
                     ?region = nltl:midbodyRegion &&
                     (?aboveValue != 1.0 || ?belowValue != 1.3))
                    ||
                    (?iceClass IN (nltl:iceClassIa, nltl:iceClassIb, nltl:iceClassIc) &&
                     ?region = nltl:sternRegion &&
                     (?aboveValue != 1.0 || ?belowValue != 1.0))
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bow frame must reach the tank top or below the top of floors, and a required upper bow ice belt must be reached." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hullRegion ?region .
                FILTER (
                    (?region = nltl:bowRegion &&
                     NOT EXISTS {
                         $this nltl:reachesTankTopOrBelowFloorTop true
                     })
                    ||
                    (EXISTS {
                         $this nltl:upperBowIceBeltRequired true
                     } &&
                     NOT EXISTS {
                         $this nltl:reachesUpperBowIceBeltTop true
                     })
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-043-MetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
