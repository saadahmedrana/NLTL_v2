<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_049_FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each represented frame attachment must be effectively confirmed."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node gen:TRF_049_ConnectionBracketShape
    ] .

gen:TRF_049_ConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket edge must be appropriately stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An installed bracket must have buckling stiffening."
    ] ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:MilliM
            ]
        ] ;
        sh:message "An installed bracket must have a single thickness quantity in millimetres."
    ] .

gen:TRF_049_FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1 ;
                sh:node gen:TRF_049_FrameAttachmentShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Within the ice-strengthened area, every linked supporting structure must be connected through a frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A longitudinal frame must be attached by a bracket to each linked supporting web frame or bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      a nltl:longitudinalFrame ;
                      (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A transversal frame terminating at a deck or ice stringer must have a bracket or similar construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      a nltl:transverseFrame ;
                      nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A frame running through a supporting structure must connect both sides of its web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount ?count .
                    FILTER (
                        datatype(?count) = xsd:integer &&
                        ?count >= 2
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each installed bracket must be at least as thick as the frame web plate, with both quantities expressed in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:hasFrameAttachment ?attachment ;
                      nltl:frameWebThickness ?webQuantity .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?webQuantity qudt:numericValue ?webValue ;
                              qudt:unit qudt:MilliM .
                ?bracket nltl:bracketThickness ?bracketQuantity .
                ?bracketQuantity qudt:numericValue ?bracketValue ;
                                  qudt:unit qudt:MilliM .
                FILTER (?bracketValue < ?webValue)
            }
        """
    ] .

</END_SHACL>
