        sh:message "The reported design line load, outside ice belt factor, section modulus, and required shear area must satisfy the specified formulas within a relative tolerance of 1e-6, with an absolute fallback of 1e-9."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:outsideIceBeltFactor ?factorNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?lineNode qudt:numericValue ?line .
                ?factorNode qudt:numericValue ?factor .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(?p * ?h AS ?rawLine)
                BIND(IF(?rawLine < 0.15, 0.15, ?rawLine) AS ?expectedLine)
                BIND(1 - (?hs / ?ls) AS ?expectedFactor)
                BIND(0.80 * 1.8 * ?expectedLine * ?l * ?l /
                     (?m * ?sigma) * ?expectedFactor * 1000000
                     AS ?expectedZ)
                BIND(1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                     ?expectedLine * ?l / (2 * ?sigma) *
                     ?expectedFactor * 10000
                     AS ?expectedA)

                FILTER(
                    ABS(?line - ?expectedLine) >
                    IF(ABS(?expectedLine) * 0.000001 > 0.000000001,
                       ABS(?expectedLine) * 0.000001,
                       0.000000001)
                    ||
                    ABS(?factor - ?expectedFactor) >
                    IF(ABS(?expectedFactor) * 0.000001 > 0.000000001,
                       ABS(?expectedFactor) * 0.000001,
                       0.000000001)
                    ||
                    ABS(?z - ?expectedZ) >
                    IF(ABS(?expectedZ) * 0.000001 > 0.000000001,
                       ABS(?expectedZ) * 0.000001,
                       0.000000001)
                    ||
                    ABS(?a - ?expectedA) >
                    IF(ABS(?expectedA) * 0.000001 > 0.000000001,
                       ABS(?expectedA) * 0.000001,
                       0.000000001)
                )
            }
        """
    ] .
