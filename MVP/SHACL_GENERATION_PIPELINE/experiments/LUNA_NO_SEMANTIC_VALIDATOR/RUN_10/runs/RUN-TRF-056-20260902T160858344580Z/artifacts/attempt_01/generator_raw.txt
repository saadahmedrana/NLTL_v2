<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-056-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node <urn:nltl:generated-shape:TRF-056-WeatherdeckHatchShape>
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A ship with a weatherdeck hatch must provide its breadth to determine hatch-opening applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                FILTER NOT EXISTS {
                    $this nltl:shipBreadth ?breadth .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-056-WeatherdeckHatchShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:message "This hatch is not applicable to the very-long-opening rule because its opening length is not greater than half the ship breadth." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        ?ship nltl:hasWeatherdeckHatch $this ;
                              nltl:shipBreadth ?breadthQuantity .
                        $this nltl:hatchOpeningLength ?lengthQuantity .
                        ?breadthQuantity qudt:numericValue ?breadth .
                        ?lengthQuantity qudt:numericValue ?length .
                        FILTER (
                            datatype(?breadth) = xsd:decimal &&
                            datatype(?length) = xsd:decimal &&
                            ?length <= (?breadth / 2)
                        )
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:shipSideDeflection ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasHatchCoverDesignEvidence ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:hatchCoverDesignEvidence
                ]
            ] ;
            sh:property [
                sh:path nltl:hasHatchFittingDesignEvidence ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:hatchFittingDesignEvidence
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "This hatch has an opening longer than half the ship breadth and therefore is subject to the hatch-cover deflection consideration." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        ?ship nltl:hasWeatherdeckHatch $this ;
                              nltl:shipBreadth ?breadthQuantity .
                        $this nltl:hatchOpeningLength ?lengthQuantity .
                        ?breadthQuantity qudt:numericValue ?breadth .
                        ?lengthQuantity qudt:numericValue ?length .
                        FILTER (
                            datatype(?breadth) = xsd:decimal &&
                            datatype(?length) = xsd:decimal &&
                            ?length > (?breadth / 2)
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
