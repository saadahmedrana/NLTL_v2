<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF070_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:in ( "azimuthing" "fixed" ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify an azimuthing or fixed thruster type."
    ] ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the thruster is for main propulsion."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:TRF070_ApplicableThrusterShape>
        <urn:nltl:generated-shape:TRF070_NonApplicableThrusterShape>
    ) .

<urn:nltl:generated-shape:TRF070_ApplicableThrusterShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:node <urn:nltl:generated-shape:TRF070_KiloNewtonQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable main-propulsion thruster must provide the propeller/ice interaction load."
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:node <urn:nltl:generated-shape:TRF070_KiloNewtonQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable main-propulsion thruster must provide the thruster-body/ice interaction load."
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:node <urn:nltl:generated-shape:TRF070_MegaPascalQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable main-propulsion thruster must provide the local ice pressure."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:node <urn:nltl:generated-shape:TRF070_DesignConditionShape> ;
        sh:minCount 1 ;
        sh:message "An applicable main-propulsion thruster must provide a design condition with externally assessed local-strength capacity."
    ] .

<urn:nltl:generated-shape:TRF070_NonApplicableThrusterShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF070_DesignConditionShape>
    a sh:NodeShape ;
    sh:class nltl:designCondition ;
    sh:property [
        sh:path nltl:designConditionLocalStrengthCapacity ;
        sh:node <urn:nltl:generated-shape:TRF070_KiloNewtonQuantityShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design condition must provide the externally assessed local-strength capacity."
    ] ;
    sh:sparql [
        sh:message "The design-condition local-strength capacity must be at least the ship's local ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacityValue .
                ?ship nltl:hasDesignCondition $this .
                ?ship nltl:localIcePressure ?pressureQuantity .
                ?pressureQuantity qudt:numericValue ?pressureValue .
                FILTER (?capacityValue < ?pressureValue)
            }
        """
    ] .

<urn:nltl:generated-shape:TRF070_KiloNewtonQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF070_MegaPascalQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
