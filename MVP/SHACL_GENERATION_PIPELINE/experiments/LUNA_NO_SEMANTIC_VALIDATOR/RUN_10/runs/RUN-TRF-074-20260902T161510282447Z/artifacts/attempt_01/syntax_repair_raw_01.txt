<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:DepthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:RequiredDepthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:node gen:DepthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:node gen:RequiredDepthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "For ice class IB, the propulsion system shall be designed for ice class IA when the ballast-condition propeller depth is below hi." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        ?depthNode qudt:numericValue ?depth .
                        ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        ?hiNode qudt:numericValue ?hi .
                        ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (?depth < ?hi)
                        FILTER NOT EXISTS {
                            $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:node gen:DepthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:node gen:RequiredDepthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "For ice class IC, the propulsion system shall be designed for ice class IA when the ballast-condition propeller depth is below hi." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:ballastConditionPropellerTopDepth ?depthNode .
                        ?depthNode qudt:numericValue ?depth .
                        ?depthNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        $this nltl:requiredPropellerSubmersionDepthHi ?hiNode .
                        ?hiNode qudt:numericValue ?hi .
                        ?hiNode qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (?depth < ?hi)
                        FILTER NOT EXISTS {
                            $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
                        }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:iceClassValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .
<END_SHACL>
