<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF076LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF076CPBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "CP"
    ] .

gen:TRF076FPBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "FP"
    ] ;
    sh:property [
        sh:path nltl:propellerReversingCapability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF076Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( "CP" "FP" )
    ] ;
    sh:or (
        gen:TRF076CPBranch
        gen:TRF076FPBranch
    ) ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:node gen:TRF076LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "CP and FP propellers must cover load cases 1-4; reversing FP propellers must also cover load case 5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case1 .
                        ?case1 nltl:propellerBladeLoadCaseNumber ?number1 .
                        FILTER (?number1 = 1)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case2 .
                        ?case2 nltl:propellerBladeLoadCaseNumber ?number2 .
                        FILTER (?number2 = 2)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case3 .
                        ?case3 nltl:propellerBladeLoadCaseNumber ?number3 .
                        FILTER (?number3 = 3)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasPropellerBladeLoadCase ?case4 .
                        ?case4 nltl:propellerBladeLoadCaseNumber ?number4 .
                        FILTER (?number4 = 4)
                    }
                    ||
                    EXISTS {
                        $this nltl:propellerType "FP" ;
                              nltl:propellerReversingCapability true .
                        FILTER NOT EXISTS {
                            $this nltl:hasPropellerBladeLoadCase ?case5 .
                            ?case5 nltl:propellerBladeLoadCaseNumber ?number5 .
                            FILTER (?number5 = 5)
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
