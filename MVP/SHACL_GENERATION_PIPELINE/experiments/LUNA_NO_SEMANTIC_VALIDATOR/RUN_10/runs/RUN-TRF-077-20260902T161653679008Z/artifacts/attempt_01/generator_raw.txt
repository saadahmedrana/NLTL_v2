<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BladeIceLoadQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

gen:MaximumBladeIceLoadQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

gen:LoadSpectrumShapeParameterQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .

gen:Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:bladeIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:BladeIceLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumBladeIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MaximumBladeIceLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadSpectrumShapeParameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LoadSpectrumShapeParameterQuantityShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:string ;
                sh:hasValue "open"
            ] ;
            sh:property [
                sh:path nltl:loadSpectrumShapeParameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal ;
                        sh:minInclusive "0.75"^^xsd:decimal ;
                        sh:maxInclusive "0.75"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:UNITLESS
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:string ;
                sh:hasValue "ducted"
            ] ;
            sh:property [
                sh:path nltl:loadSpectrumShapeParameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal ;
                        sh:minInclusive "1.0"^^xsd:decimal ;
                        sh:maxInclusive "1.0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:UNITLESS
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The blade ice load must satisfy 0 <= Fice <= maximum blade ice load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bladeIceLoad ?ficeNode .
                $this nltl:maximumBladeIceLoad ?maxNode .
                ?ficeNode qudt:numericValue ?ficeValue .
                ?maxNode qudt:numericValue ?maxValue .
                FILTER (
                    xsd:decimal(?ficeValue) < 0 ||
                    xsd:decimal(?ficeValue) > xsd:decimal(?maxValue)
                )
            }
        """
    ] .
