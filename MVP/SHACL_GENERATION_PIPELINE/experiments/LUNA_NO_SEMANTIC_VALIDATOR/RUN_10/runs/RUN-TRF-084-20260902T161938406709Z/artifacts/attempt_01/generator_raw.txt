<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_084_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller type is required to determine whether the conditional propeller-pitch rule applies."
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a CP propeller, pitchValueKnown must select a conforming case: when true, propellerPitchAtMcrBollard must be present in metres; when false, propellerPitchAtMcrBollard must equal 0.7 times propellerPitchAtMcrFreeRunning within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER (STR(?propellerType) = "CP")

                FILTER NOT EXISTS {
                    $this nltl:pitchValueKnown true .
                    $this nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                    ?bollardQuantity qudt:numericValue ?bollardValue ;
                        qudt:unit unit:M .
                    FILTER (datatype(?bollardValue) = xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:pitchValueKnown false .
                    $this nltl:propellerPitchAtMcrFreeRunning ?freeRunningQuantity .
                    ?freeRunningQuantity qudt:numericValue ?freeRunningValue ;
                        qudt:unit unit:M .
                    $this nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                    ?bollardQuantity qudt:numericValue ?bollardValue ;
                        qudt:unit unit:M .
                    FILTER (datatype(?freeRunningValue) = xsd:decimal)
                    FILTER (datatype(?bollardValue) = xsd:decimal)
                    BIND ((0.7 * ?freeRunningValue) AS ?expectedValue)
                    FILTER (
                        ABS(?bollardValue - ?expectedValue) <=
                        (ABS(?expectedValue) * 0.000001 + 0.000000001)
                    )
                }
            }
        """
    ] .
<END_SHACL>
