<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF114ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:node gs:DecimalKnotQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide a typical ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node gs:DecimalMetreQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:node gs:TRF114ImpactLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide at least one applicable thruster ice impact load case."
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical contact geometry requires a valid equivalent impact sphere radius on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                    ?radius qudt:numericValue ?value ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] .

gs:TRF114ImpactLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must reference TRAFICOM Table 6-16 for its impact load case."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:node [
            sh:class nltl:thrusterBody ;
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:node gs:DecimalKiloNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The applicable thruster body must provide its externally assessed resistance capacity."
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must identify the thruster body to which it applies."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:node gs:DecimalKiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must provide the externally determined thruster ice impact load."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:node gs:DecimalKiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The impact case must provide the externally determined thruster ice impact demand."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case must include evidence for the loaded area assessment."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "A non-hemispherical contact geometry must include correspondence evidence."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The externally assessed thruster resistance capacity must be at least the impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demand ;
                      nltl:thrusterImpactCaseAppliesToBody ?body .
                ?demand qudt:numericValue ?demandValue .
                ?body nltl:thrusterResistanceCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER (
                    datatype(?demandValue) = xsd:decimal &&
                    datatype(?capacityValue) = xsd:decimal &&
                    ?capacityValue >= ?demandValue
                )
            }
        """
    ] .
<END_SHACL>
