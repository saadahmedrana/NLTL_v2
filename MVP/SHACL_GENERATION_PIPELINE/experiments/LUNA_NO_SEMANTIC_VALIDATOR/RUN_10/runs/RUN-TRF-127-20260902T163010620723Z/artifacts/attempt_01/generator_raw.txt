<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-127> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:airReceiverServesAdditionalPurpose ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether its air receivers serve an additional purpose."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:airReceiverServesAdditionalPurpose ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:airReceiverCapacity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M3 ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "When air receivers serve an additional purpose, air receiver capacity with cubic-metre units is required."
            ] ;
            sh:property [
                sh:path nltl:additionalPurposeRequiredAirCapacity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "When air receivers serve an additional purpose, the required additional air capacity is required."
            ]
        ]
        [
            sh:property [
                sh:path nltl:airReceiverServesAdditionalPurpose ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Air receiver capacity must be sufficient for the additional purpose." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:airReceiverServesAdditionalPurpose true ;
                      nltl:airReceiverCapacity ?capacityNode ;
                      nltl:additionalPurposeRequiredAirCapacity ?requiredNode .
                ?capacityNode qudt:numericValue ?capacityValue .
                ?requiredNode qudt:numericValue ?requiredValue .
                FILTER (?capacityValue < ?requiredValue)
            }
        """
    ] .
<END_SHACL>
