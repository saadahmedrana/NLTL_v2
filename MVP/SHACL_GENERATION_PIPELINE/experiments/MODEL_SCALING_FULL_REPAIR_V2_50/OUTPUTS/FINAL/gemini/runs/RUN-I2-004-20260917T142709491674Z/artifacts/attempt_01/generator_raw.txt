<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's upper ice waterline displacement (DUI) must be equal to the maximum displacement among its upper ice waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            
            SELECT $this (nltl:upperIceWaterlineDisplacement AS ?path) (?duiQty AS ?value)
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiQty .
                ?duiQty qudt:numericValue ?dui .
                {
                    SELECT $this (MAX(?caseVal) AS ?maxCaseVal)
                    WHERE {
                        $this nltl:hasUpperIceWaterlineCase ?case .
                        ?case nltl:waterlineCaseDisplacement ?caseQty .
                        ?caseQty qudt:numericValue ?caseVal .
                    }
                    GROUP BY $this
                }
                FILTER (abs(?dui - ?maxCaseVal) > 1e-6 && (?maxCaseVal = 0.0 || (abs(?dui - ?maxCaseVal) / ?maxCaseVal) > 1e-6))
            }
        """
    ] .
<END_SHACL>
