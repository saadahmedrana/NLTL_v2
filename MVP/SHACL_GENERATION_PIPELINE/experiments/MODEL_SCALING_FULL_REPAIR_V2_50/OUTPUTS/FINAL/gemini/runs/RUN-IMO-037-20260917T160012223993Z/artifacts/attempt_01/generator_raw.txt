<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1 ;
        sh:message "Construction date must be a single xsd:date." ;
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:message "Ship category must be a polar ship category value IRI." ;
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionCase ;
        sh:class nltl:loadingConditionCase ;
        sh:message "Each loading condition case must be a valid loadingConditionCase." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For Category A or B ships constructed on/after 2017-01-01, each loading condition must have a residual stability factor s_i of 1, or comply with the alternative instrument criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?cat .
                FILTER (?cat IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                
                $this nltl:constructionDate ?date .
                FILTER (?date >= "2017-01-01"^^xsd:date)
                
                $this nltl:hasLoadingConditionCase ?case .
                
                # Check that neither the exact factor of 1 is met...
                FILTER NOT EXISTS {
                    ?case nltl:residualStabilityFactorSI ?siNode .
                    ?siNode qudt:numericValue ?siVal .
                    FILTER (?siVal = 1)
                }
                # ...nor is the alternative instrument applicable and status compliant (true)
                FILTER NOT EXISTS {
                    ?case nltl:alternativeInstrumentApplicable true .
                    ?case nltl:alternativeInstrumentResidualStabilityStatus true .
                }
            }
        """ ;
    ] .

<urn:nltl:generated-shape:LoadingConditionCaseShape> a sh:NodeShape ;
    sh:targetClass nltl:loadingConditionCase ;
    sh:property [
        sh:path nltl:loadingConditionIdentifier ;
        sh:datatype xsd:string ;
        sh:maxCount 1 ;
        sh:message "Loading condition identifier must be a single string." ;
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentApplicable ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "Alternative instrument applicability status must be a boolean." ;
    ] ;
    sh:property [
        sh:path nltl:alternativeInstrumentResidualStabilityStatus ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "Alternative instrument residual stability status must be a boolean." ;
    ] ;
    sh:property [
        sh:path nltl:hasLoadingConditionResultEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:message "Loading condition result evidence must reference an evidenceArtifact." ;
    ] ;
    sh:property [
        sh:path nltl:residualStabilityFactorSI ;
        sh:class qudt:QuantityValue ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:message "Residual stability factor unit must be UNITLESS." ;
        ] ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:minCount 1 ;
            sh:maxCount 1 ;
            sh:message "Residual stability factor numeric value is required." ;
        ] ;
        sh:message "Residual stability factor s_i must be a valid QuantityValue with UNITLESS unit." ;
    ] .
<END_SHACL>
