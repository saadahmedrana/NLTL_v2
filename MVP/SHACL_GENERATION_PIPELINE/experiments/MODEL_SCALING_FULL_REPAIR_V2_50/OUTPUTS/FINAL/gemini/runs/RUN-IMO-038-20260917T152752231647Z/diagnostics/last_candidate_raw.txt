<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix urn: <urn:nltl:generated-shape:> .

urn:IceDamageExtentShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damageCenterForwardOfMaxBreadth ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:transversePenetration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MilliMeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalDamageExtent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedVerticalDamagePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node urn:MeterQuantityShape
    ] ;
    sh:sparql [
        sh:message "Assumed ice damage extents or vertical positions do not comply with the piecewise calculation formulas or limits specified in Polar Code 4.3.2.2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:damageCenterForwardOfMaxBreadth ?isForward ;
                      nltl:upperIceWaterlineLength [ qudt:numericValue ?L_val ] ;
                      nltl:upperIceWaterlineDraught [ qudt:numericValue ?D_val ] ;
                      nltl:longitudinalDamageExtent [ qudt:numericValue ?longExtent ] ;
                      nltl:transversePenetration [ qudt:numericValue ?transPen ] ;
                      nltl:verticalDamageExtent [ qudt:numericValue ?vertExtent ] ;
                      nltl:selectedVerticalDamagePosition [ qudt:numericValue ?vertPos ] .

                BIND(IF(?isForward = true, 0.045 * ?L_val, 0.015 * ?L_val) AS ?expectedLong)
                BIND(IF(0.20 * ?D_val < ?longExtent, 0.20 * ?D_val, ?longExtent) AS ?expectedVert)

                BIND(abs(?longExtent - ?expectedLong) AS ?diffLong)
                BIND((1e-6 * ?expectedLong) + 1e-9 AS ?tolLong)

                BIND(abs(?transPen - 760.0) AS ?diffTrans)
                BIND(1e-5 AS ?tolTrans)

                BIND(abs(?vertExtent - ?expectedVert) AS ?diffVert)
                BIND((1e-6 * ?expectedVert) + 1e-9 AS ?tolVert)

                BIND(1.20 * ?D_val AS ?maxVertPos)

                FILTER (
                    ?diffLong > ?tolLong ||
                    ?diffTrans > ?tolTrans ||
                    ?diffVert > ?tolVert ||
                    ?vertPos < -1e-9 ||
                    ?vertPos > ?maxVertPos + 1e-9
                )
            }
        """
    ] .

urn:MeterQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

urn:MilliMeterQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
