<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix genshape: <urn:nltl:generated-shape:> .

genshape:ShipIceAccretionShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "When ice accretion is likely, every relevant hatch or door must have an associated ice or snow removal or prevention means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceAccretionLikely true .
                $this nltl:hasRelevantHatchOrDoor ?hatchOrDoor .
                FILTER NOT EXISTS {
                    ?hatchOrDoor nltl:hasIceOrSnowRemovalOrPreventionMeans ?means .
                }
            }
        """
    ] .
<END_SHACL>
