<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:IMOPolarCodeChapter2NlsDischargeProhibitionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ships operating in Arctic waters, discharge of noxious liquid substances (NLS) or NLS mixtures into the sea is prohibited." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?dischargeVal
            WHERE {
                {
                    $this nltl:operatingArea ?area .
                    FILTER(str(?area) = "Arctic_waters" || str(?area) = "Arctic waters")
                } UNION {
                    $this nltl:arcticWaters ?arctic .
                    FILTER(str(?arctic) = "true" || str(?arctic) = "yes" || str(?arctic) = "Arctic_waters" || str(?arctic) = "Arctic waters")
                }
                OPTIONAL {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?val .
                }
                FILTER(!BOUND(?val) || ?val != false)
                BIND(IF(BOUND(?val), ?val, "Missing Discharge Record") AS ?dischargeVal)
            }
        """
    ] .
<END_SHACL>
