<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:IMO-114-Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Category A or B ships constructed on or after 2017-01-01 carrying NLS requiring Type 3 tank in cargo tank type 3 must be approved by the Administration and have a polar operation entry present on the certificate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
  $this nltl:shipCategory ?cat .
  FILTER (?cat IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
  
  $this nltl:constructionDate ?date .
  FILTER (?date >= "2017-01-01"^^xsd:date)
  
  $this nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank true .
  
  $this nltl:cargoTankType "3" .
  
  OPTIONAL {
    $this nltl:administrationCarriageApprovalStatus ?approval .
  }
  OPTIONAL {
    $this nltl:applicableCertificatePolarOperationEntryPresent ?cert .
  }
  
  FILTER (!BOUND(?approval) || ?approval != nltl:evidenceStateApproved || !BOUND(?cert) || ?cert != true)
}
""" ;
    ] .
<END_SHACL>
