@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO26-014-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatesOnlyInContinuousDaylight ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:visualIceDetectionIlluminationMeansCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ship does not operate only in continuous daylight, it must have at least 2 visual ice detection illumination means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?value
            WHERE {
                $this nltl:operatesOnlyInContinuousDaylight false .
                OPTIONAL { $this nltl:visualIceDetectionIlluminationMeansCount ?value }
                FILTER (!BOUND(?value) || ?value < 2)
            }
        """
    ] .
