@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:PolarCodeChapter11ApplicabilityShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeChapter11Applicable ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Code chapter 11 applies only when the ship is certified in accordance with SOLAS Chapter I." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarCodeChapter11Applicable true .
                FILTER NOT EXISTS {
                    $this nltl:solasChapterICertified true .
                }
            }
        """
    ] .
