<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF002Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an ice class contracted for construction on or after 1 January 2019 but before 5 July 2021 must have the applicable ice class regulation edition set to Ice Class Regulations of 2017." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                $this nltl:constructionContractDate ?date .
                FILTER (?date >= "2019-01-01"^^xsd:date && ?date < "2021-07-05"^^xsd:date)
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition nltl:iceClassRegulationEdition2017 .
                }
            }
        """
    ] .
<END_SHACL>
