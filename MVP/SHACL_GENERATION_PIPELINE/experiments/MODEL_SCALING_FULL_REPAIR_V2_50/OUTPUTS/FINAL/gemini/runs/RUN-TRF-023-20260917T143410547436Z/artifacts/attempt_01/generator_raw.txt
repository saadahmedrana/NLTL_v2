<BEGIN_SHACL>
@prefix gshape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gshape:Ship1985RulesPowerShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For ships subject to the 1985 ice class regulations and having ice class IB or IC, the maximum continuous rating power must be at least the required minimum engine output under the 1985 rules." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?this ?value ?message
WHERE {
  ?this nltl:applicableIceClassRegulationEdition nltl:iceClassRuleEdition1985 .
  ?this nltl:iceClass ?iceClass .
  FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
  
  OPTIONAL {
    ?this nltl:maximumContinuousRatingPower/qudt:numericValue ?mcrVal .
  }
  OPTIONAL {
    ?this nltl:requiredPowerUnder1985Rules/qudt:numericValue ?reqVal .
  }
  
  BIND(
    IF(!BOUND(?mcrVal), "Missing maximum continuous rating power.",
    IF(!BOUND(?reqVal), "Missing required power under 1985 rules.",
    IF(?mcrVal < ?reqVal, "Maximum continuous rating power is less than the required power under 1985 rules.",
    ""))) AS ?message
  )
  FILTER (?message != "")
  BIND (?this AS ?value)
}
        """ ;
    ] .
<END_SHACL>
