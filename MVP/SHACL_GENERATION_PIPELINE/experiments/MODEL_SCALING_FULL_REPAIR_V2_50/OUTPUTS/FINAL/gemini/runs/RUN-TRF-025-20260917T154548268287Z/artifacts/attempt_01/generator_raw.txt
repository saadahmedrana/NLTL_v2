<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix s: <urn:nltl:generated-shape:> .

s:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one ice class." ;
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Each ship must specify whether a bulbous bow is present." ;
    ] ;
    sh:or (
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:not [ sh:hasValue nltl:iceClassIaSuper ] ;
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bulbousBowPresent ;
                        sh:hasValue true ;
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper ;
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false ;
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:PositiveLengthInMeters ;
                sh:message "For IA Super ships without a bulb, ship length (L) must be a positive quantity in meters." ;
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:PositiveLengthInMeters ;
                sh:message "For IA Super ships without a bulb, ship breadth (B) must be a positive quantity in meters." ;
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:PositiveLengthInMeters ;
                sh:message "For IA Super ships without a bulb, ice class draught (T) must be a positive quantity in meters." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNPerM2 ;
                sh:message "Coefficient f1 must be specified in N/m2." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNPerM ;
                sh:message "Coefficient f2 must be specified in N/m." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNPerM ;
                sh:message "Coefficient f3 must be specified in N/m." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNPerM2 ;
                sh:message "Coefficient f4 must be specified in N/m2." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInN ;
                sh:message "Coefficient g1 must be specified in N." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNPerM ;
                sh:message "Coefficient g2 must be specified in N/m." ;
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInNewtonPerMetreToPowerOnePointFiveUnit ;
                sh:message "Coefficient g3 must be specified in N/m^1.5." ;
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInN ;
                sh:message "Brash ice resistance coefficient C1 must be specified in N." ;
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node s:QuantityInN ;
                sh:message "Brash ice resistance coefficient C2 must be specified in N." ;
            ] ;
        ]
    ) .

s:PositiveLengthInMeters a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
    ] .

s:QuantityInNPerM2 a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N-PER-M2 ;
    ] .

s:QuantityInNPerM a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N-PER-M ;
    ] .

s:QuantityInN a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N ;
    ] .

s:QuantityInNewtonPerMetreToPowerOnePointFiveUnit a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:newtonPerMetreToPowerOnePointFiveUnit ;
    ] .
<END_SHACL>
