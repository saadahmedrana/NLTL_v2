        sh:message "Web frame ice load F does not match the calculated value F = 1.8 * max(p * h, 0.15) * S [MN] converted to kN" ;

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                $this nltl:webFrameSpacing/qudt:numericValue ?S .

                $this nltl:iceLoadHeight/qudt:numericValue ?h .

                BIND(IF(?ph < 0.15, 0.15, ?ph) AS ?ph_eff)
                BIND(1800.0 * ?ph_eff * ?S AS ?F_expected)

                BIND(ABS(?F - ?F_expected) AS ?diff)
                BIND(ABS(?F - ?F_expected) / ?F_expected AS ?rel_diff)

                FILTER (?diff > 1e-5 && ?rel_diff > 1e-5)
            }
        """

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                $this nltl:webFrameSpacing/qudt:numericValue ?S .
                $this nltl:loadLengthParameter/qudt:numericValue ?la .

                BIND(ABS(?la - ?la_expected) AS ?diff)
                BIND(ABS(?la - ?la_expected) / ?la_expected AS ?rel_diff)

                FILTER (?diff > 1e-5 && ?rel_diff > 1e-5)
            }
