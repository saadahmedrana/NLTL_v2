<BEGIN_SHACL>
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterExtremeLoad ;
        sh:minCount 1 ;
        sh:node shape:QuantityValueForceShape ;
    ] ;
    sh:property [
        sh:path nltl:yieldSafetyFactor ;
        sh:minCount 1 ;
        sh:node shape:QuantityValueDimensionlessShape ;
    ] ;
    sh:property [
        sh:path nltl:nominalVonMisesStress ;
        sh:minCount 1 ;
        sh:node shape:QuantityValuePressureShape ;
    ] ;
    sh:property [
        sh:path nltl:localStressConcentration ;
        sh:minCount 1 ;
        sh:node shape:QuantityValueDimensionlessShape ;
    ] ;
    sh:property [
        sh:path nltl:hasFactoredExtremeLoadOperabilityAssessment ;
        sh:minCount 1 ;
        sh:node shape:FactoredExtremeLoadOperabilityAssessmentShape ;
    ] .

shape:FactoredExtremeLoadOperabilityAssessmentShape a sh:NodeShape ;
    sh:class nltl:factoredExtremeLoadOperabilityAssessment ;
    sh:property [
        sh:path nltl:assessedFactoredExtremeLoad ;
        sh:minCount 1 ;
        sh:node shape:QuantityValueForceShape ;
    ] ;
    sh:property [
        sh:path nltl:assessmentOperabilityMaintained ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Assessed operability must be maintained." ;
    ] ;
    sh:property [
        sh:path nltl:assessmentRepairRequired ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Assessed repair must not be required." ;
    ] ;
    sh:property [
        sh:path nltl:assessedThrusterComponent ;
        sh:minCount 1 ;
        sh:node shape:ShipComponentShape ;
    ] .

shape:ShipComponentShape a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node shape:MaterialPropertiesShape ;
    ] .

shape:MaterialPropertiesShape a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:node shape:QuantityValuePressureShape ;
    ] .

shape:QuantityValueForceShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

shape:QuantityValuePressureShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

shape:QuantityValueDimensionlessShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
