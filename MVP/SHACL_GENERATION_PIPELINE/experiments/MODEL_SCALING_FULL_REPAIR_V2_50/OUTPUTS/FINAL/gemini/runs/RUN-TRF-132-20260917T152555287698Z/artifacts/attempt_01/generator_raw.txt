<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:ShipEngineOutputRequirementShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( nltl:iceClassIa nltl:iceClassIaSuper nltl:iceClassIb nltl:iceClassIc nltl:iceClassIi nltl:iceClassIii ) ;
        sh:message "Each ship must have a valid ice class value."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Maximum continuous rating power must be specified in kW."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula base power P0 must be specified in kW."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula displacement must have a numeric value."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula factor f1 must be unitless."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula factor f2 must be unitless."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula factor f3 must be unitless."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Engine output formula factor f4 must have a numeric value."
    ] ;
    sh:sparql [
        sh:message "Engine output (maximumContinuousRatingPower) must be no less than calculated P = f1*f2*f3*(f4*Delta + P0) kW, and in no case less than 740 kW for ice classes IB and IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower/qudt:numericValue ?mcr .
                $this nltl:engineOutputFormulaFactorF1/qudt:numericValue ?f1 .
                $this nltl:engineOutputFormulaFactorF2/qudt:numericValue ?f2 .
                $this nltl:engineOutputFormulaFactorF3/qudt:numericValue ?f3 .
                $this nltl:engineOutputFormulaFactorF4/qudt:numericValue ?f4 .
                $this nltl:engineOutputFormulaDisplacement/qudt:numericValue ?delta .
                $this nltl:engineOutputFormulaBasePowerP0/qudt:numericValue ?p0 .
                $this nltl:iceClass ?iceClass .

                BIND(?f1 * ?f2 * ?f3 * (?f4 * ?delta + ?p0) AS ?calculatedP)

                FILTER (
                    (?mcr < (?calculatedP - 1e-6)) ||
                    ((?iceClass = nltl:iceClassIb || ?iceClass = nltl:iceClassIc) && ?mcr < 740)
                )
            }
        """
    ] .
<END_SHACL>
