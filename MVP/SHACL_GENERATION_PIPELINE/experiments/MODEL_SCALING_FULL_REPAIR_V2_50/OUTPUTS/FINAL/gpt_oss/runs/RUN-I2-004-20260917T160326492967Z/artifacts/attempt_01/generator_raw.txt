<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix : <urn:nltl:generated-shape:> .

:UpperIceWaterlineDisplacementShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    # Presence, datatype and unit of the ship displacement (DUI)
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    # At least one upper‑ice‑waterline case linked to the ship
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            # Each case must provide its waterline case displacement
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;

    # Consistency: ship DUI must equal the greatest waterline case displacement
    sh:sparql [
        sh:message "The ship displacement (upperIceWaterlineDisplacement) must be equal to the greatest waterline case displacement." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?shipValue ?maxCaseVal
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiNode .
                ?duiNode qudt:numericValue ?shipValue .
                FILTER(datatype(?shipValue) = xsd:decimal)

                $this nltl:hasUpperIceWaterlineCase ?case .
                ?case nltl:waterlineCaseDisplacement ?caseNode .
                ?caseNode qudt:numericValue ?caseValue .
                FILTER(datatype(?caseValue) = xsd:decimal)

                BIND (MAX(?caseValue) AS ?maxCaseVal)
            }
            FILTER (?shipValue != ?maxCaseVal)
        """
