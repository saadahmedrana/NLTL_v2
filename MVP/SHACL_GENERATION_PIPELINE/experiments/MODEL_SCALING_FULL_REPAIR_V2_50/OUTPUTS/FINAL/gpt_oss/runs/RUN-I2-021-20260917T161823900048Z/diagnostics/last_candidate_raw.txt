<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:ThicknessCalcShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;

            # required thickness
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ; sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ; sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;   # corrected unit to mm
                        sh:minCount 1 ; sh:maxCount 1
                    ]
                ]
            ] ;

            # net plate thickness required for ice load
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ; sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ; sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ; sh:maxCount 1
                    ]
                ]
            ] ;

            # corrosion‑abrasion allowance
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ; sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ; sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ; sh:maxCount 1
                    ]
                ]
            ] ;

            # calculation: thickness = net + allowance (with tolerance)
            sh:sparql [
                sh:message "Shell plate thickness must equal net plate thickness plus corrosion‑abrasion allowance (within tolerance)." ;
                sh:select """
                    PREFIX sh: <http://www.w3.org/ns/shacl#>
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasPlating ?plating .
                        ?plating nltl:thickness ?thickQV .
                        ?plating nltl:iceLoadRequiredNetPlateThickness ?netQV .
                        ?plating nltl:corrosionAbrasionAllowance ?caQV .
                        ?thickQV qudt:numericValue ?tVal .
                        ?netQV qudt:numericValue ?netVal .
                        ?caQV qudt:numericValue ?caVal .
                        FILTER (
                            abs(?tVal - (?netVal + ?caVal)) >
                            ( "0.000001"^^xsd:decimal * (abs(?netVal + ?caVal) + "0.000000001"^^xsd:decimal) )
                        )
                    }
                """ ;
            ]
        ]
    ] .
<END_SHACL>
