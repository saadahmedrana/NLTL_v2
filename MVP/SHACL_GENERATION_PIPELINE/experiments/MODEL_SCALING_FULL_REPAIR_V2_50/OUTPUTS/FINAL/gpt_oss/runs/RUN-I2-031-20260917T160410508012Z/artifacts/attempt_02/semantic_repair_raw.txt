<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix ex: <urn:nltl:generated-shape:> .

ex:MemberStrengthCheck
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear and bending demand exceeds plastic strength for this structural member." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
              $this nltl:structuralLocation ?locValue .
              FILTER(?locValue IN (nltl:bottomStructureLocation, nltl:sideStructureLocation))
              $this nltl:hasStructuralMemberLoadCase ?lc .
              ?lc nltl:combinedShearAndBendingDemand ?demandQty .
              ?demandQty qudt:numericValue ?demand .
              $this nltl:plasticStrength ?strengthQty .
              ?strengthQty qudt:numericValue ?strength .
              FILTER(?demand > ?strength)
            }
        """ ;
    ] .

ex:BottomPatchOrientationCheck
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bottom‑structure members the patch load must be applied with the dimension (b) parallel to the frame direction." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
              $this nltl:structuralLocation nltl:bottomStructureLocation .
              $this nltl:hasStructuralMemberLoadCase ?lc .
              FILTER NOT EXISTS {
                ?lc nltl:loadPatchOrientation nltl:loadPatchHeightParallelToFrame .
              }
            }
        """ ;
    ] .

# NEW: Ensure ship instances have at least one structural member
ex:ShipHasStructuralMemberCheck
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:message "A ship must have at least one structural member linked via nltl:hasStructuralMember." ;
    ] .

# NEW: Ensure each structural member specifies its frame profile type
ex:StructuralMemberFrameProfileTypeCheck
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:message "Every structural member must have a frame profile type defined via nltl:frameProfileType." ;
    ] .

</END_SHACL>
