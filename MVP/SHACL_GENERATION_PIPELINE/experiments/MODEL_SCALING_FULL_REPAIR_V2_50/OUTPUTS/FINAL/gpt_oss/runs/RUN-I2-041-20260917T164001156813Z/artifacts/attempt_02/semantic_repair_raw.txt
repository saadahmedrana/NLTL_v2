<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

# Ship level shape – requires at least one structural member
gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape ;
        sh:message "A ship must be linked to at least one structural member via nltl:hasStructuralMember." ;
    ] .

# Structural member shape – required quantity values (numeric literals)
gen:StructuralMemberShape a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;

    # net attached shell plate thickness (t_pn)
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "netAttachedShellPlateThickness (t_pn) must be provided as a decimal numeric value." ;
    ] ;

    # yield strength (sigma_y)
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "yieldStrength must be provided as a decimal numeric value." ;
    ] ;

    # net web thickness (t_wn) – result term
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "netWebThickness (t_wn) must be provided as a decimal numeric value." ;
    ] .
<END_SHACL>
