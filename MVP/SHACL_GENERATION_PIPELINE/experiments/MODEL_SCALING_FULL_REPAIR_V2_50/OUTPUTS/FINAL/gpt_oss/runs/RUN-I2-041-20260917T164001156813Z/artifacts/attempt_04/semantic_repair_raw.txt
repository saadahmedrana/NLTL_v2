<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt#> .

# Declare external QUDT namespace to satisfy deterministic validation
sh:declare [
    sh:prefix "qudt" ;
    sh:namespace "http://qudt.org/schema/qudt#" ;
] .

# Ship level shape – requires at least one structural member
gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape ;
        sh:message "A ship must be linked to at least one structural member via nltl:hasStructuralMember." ;
    ] .

# Structural member shape – required quantity values (QUDT QuantityValue nodes)
gen:StructuralMemberShape a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;

    # net attached shell plate thickness (t_pn)
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:message "netAttachedShellPlateThickness numeric value must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:message "netAttachedShellPlateThickness unit must be milli‑metre (mm)." ;
            ] ;
        ] ;
        sh:message "netAttachedShellPlateThickness (t_pn) must be provided as a QUDT QuantityValue node." ;
    ] ;

    # yield strength (sigma_y)
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:message "yieldStrength numeric value must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:message "yieldStrength unit must be megapascals (MPa)." ;
            ] ;
        ] ;
        sh:message "yieldStrength must be provided as a QUDT QuantityValue node." ;
    ] ;

    # net web thickness (t_wn) – result term
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:message "netWebThickness numeric value must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:message "netWebThickness unit must be milli‑metre (mm)." ;
            ] ;
        ] ;
        sh:message "netWebThickness (t_wn) must be provided as a QUDT QuantityValue node." ;
    ] .
<END_SHACL>
