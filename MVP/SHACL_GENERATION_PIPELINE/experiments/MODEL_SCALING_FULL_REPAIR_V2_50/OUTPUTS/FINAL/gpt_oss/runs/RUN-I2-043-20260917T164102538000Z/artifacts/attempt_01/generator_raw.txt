<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .
@prefix u: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

# Shape for each structural member required by the requirement
gen:StructuralMemberShape a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    # net flange thickness quantity node
    sh:property [
        sh:path nltl:netFlangeThickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "netFlangeThickness must have a decimal numeric value." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue u:MilliM ;
                sh:minCount 1 ;
                sh:message "netFlangeThickness must use unit milli‑meter (mm)." ;
            ] ;
        ] ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide netFlangeThickness quantity." ;
    ] ;
    # yield strength quantity node
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "yieldStrength must have a decimal numeric value." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue u:MegaPA ;
                sh:minCount 1 ;
                sh:message "yieldStrength must use unit megapascal (MPa)." ;
            ] ;
        ] ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide yieldStrength quantity." ;
    ] ;
    # flange outstand quantity node (result of external calculation)
    sh:property [
        sh:path nltl:flangeOutstand ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "flangeOutstand must have a decimal numeric value." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue u:MilliM ;
                sh:minCount 1 ;
                sh:message "flangeOutstand must use unit milli‑meter (mm)." ;
            ] ;
        ] ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide flangeOutstand quantity." ;
    ] .

# Ship shape linking to structural members
gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "Ship must have at least one structuralMember with required quantities." ;
    ] .
<END_SHACL>
