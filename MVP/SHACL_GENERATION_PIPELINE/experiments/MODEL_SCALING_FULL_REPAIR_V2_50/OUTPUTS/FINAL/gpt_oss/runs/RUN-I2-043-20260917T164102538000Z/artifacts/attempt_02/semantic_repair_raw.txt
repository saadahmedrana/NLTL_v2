# Shape for each structural member required by the requirement

    sh:targetClass nltl:structuralMember ;
    # net flange thickness quantity (decimal, mm)
    sh:property [
        sh:path nltl:netFlangeThickness ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide netFlangeThickness as a decimal value (mm)." ;
    ] ;
    # yield strength quantity (decimal, MPa)
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide yieldStrength as a decimal value (MPa)." ;
    ] ;
    # flange outstand quantity (decimal, mm)
    sh:property [
        sh:path nltl:flangeOutstand ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Each structural member must provide flangeOutstand as a decimal value (mm)." ;
    ] .

# Ship shape linking to structural members
gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "Ship must have at least one structuralMember with required quantities." ;
