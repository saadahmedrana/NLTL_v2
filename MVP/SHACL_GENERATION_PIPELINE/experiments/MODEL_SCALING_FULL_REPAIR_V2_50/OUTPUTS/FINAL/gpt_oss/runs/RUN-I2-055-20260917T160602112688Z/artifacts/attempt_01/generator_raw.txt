<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

urn:nltl:generated-shape:StrengthCriteriaShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    # designStress must be present
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:message "Design stress quantity must be present." ;
    ] ;

    # permissibleStress must be present
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:message "Permissible stress quantity must be present." ;
    ] ;

    # strengthCriterionStatus must be present and true
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Strength criterion status must be true." ;
    ] ;

    # numeric comparison: designStress must not exceed permissibleStress
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Design stress exceeds permissible stress." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:designStress ?dQ .
                ?dQ qudt:numericValue ?dVal .
                $this nltl:permissibleStress ?pQ .
                ?pQ qudt:numericValue ?pVal .
                FILTER( ?dVal > ?pVal )
            }
        """ ;
    ] .

<END_SHACL>
