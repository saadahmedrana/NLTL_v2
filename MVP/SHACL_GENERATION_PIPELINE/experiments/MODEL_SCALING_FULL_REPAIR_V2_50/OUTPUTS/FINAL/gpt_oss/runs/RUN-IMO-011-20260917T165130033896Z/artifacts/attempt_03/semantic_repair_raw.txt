<BEGIN_SHACL>
@prefix sh:   <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix qudt: <http://qudt.org/schema/qudt#> .
@prefix gen:  <urn:nltl:generated-shape:> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # dataPeriod must be a QUDT QuantityValue (years)
    sh:property [
        sh:path nltl:dataPeriod ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:YR
            ]
        ] ;
        sh:minCount 1
    ] ;
    # meanDailyLowTemperature must be a QUDT QuantityValue (deg C)
    sh:property [
        sh:path nltl:meanDailyLowTemperature ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C
            ]
        ] ;
        sh:minCount 1
    ] ;
    # administrationDatasetApprovalStatus limited to Approved
    sh:property [
        sh:path nltl:administrationDatasetApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:in ( nltl:evidenceStateApproved )
    ] ;
    # ship must have at least one daily low temperature observation
    sh:property [
        sh:path nltl:hasDailyLowTemperatureObservation ;
        sh:node gen:DailyLowTempObsShape ;
        sh:minCount 1
    ] ;
    # Conditional rule: if dataPeriod < 10 years then approval must be Approved
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When dataPeriod is less than 10 years, administrationDatasetApprovalStatus must be Approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:dataPeriod ?dpNode .
                ?dpNode qudt:numericValue ?dp .
                FILTER ( ?dp < 10 ) .
                FILTER NOT EXISTS { $this nltl:administrationDatasetApprovalStatus nltl:evidenceStateApproved . }
            }
        """ ;
    ] .

gen:DailyLowTempObsShape
    a sh:NodeShape ;
    sh:targetClass nltl:dailyLowTemperatureObservation ;
    # observation date
    sh:property [
        sh:path nltl:dailyTemperatureObservationDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    # daily low temperature value as QUDT QuantityValue (deg C)
    sh:property [
        sh:path nltl:dailyLowTemperature ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG_C
            ]
        ] ;
        sh:minCount 1
    ] .
<END_SHACL>
