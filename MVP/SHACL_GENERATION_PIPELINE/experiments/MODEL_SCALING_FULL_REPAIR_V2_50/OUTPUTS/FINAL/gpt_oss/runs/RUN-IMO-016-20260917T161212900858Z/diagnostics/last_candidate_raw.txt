<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix : <urn:nltl:generated-shape:> .

:IMO-016_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    # Certificate form must be the Appendix I model
    sh:property [
        sh:path (nltl:hasPolarShipCertificate nltl:certificateFormModel) ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "Polar ship certificate must use the Appendix I model." ;
    ] ;

    # Certificate language must be recorded
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Certificate language must be provided." ;
    ] ;

    # Approved translation presence (boolean) – optional unless needed by condition
    sh:property [
        sh:path nltl:approvedTranslationPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "Approved translation presence must be a boolean value." ;
    ] ;

    # Conditional rule: non‑English/French/Spanish language → approved translation must be true
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the certificate language is not English, French, or Spanish, an approved translation must be present and true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this (STR(?lang) AS ?nonCompliantLanguage)
            WHERE {
                $this nltl:certificateLanguage ?lang .
                FILTER ( !( ?lang IN ("English", "French", "Spanish") ) )
                OPTIONAL { $this nltl:approvedTranslationPresent ?ap }
                FILTER ( (!bound(?ap)) || (?ap != true) )
            }
        """ ;
    ] .
<END_SHACL>
