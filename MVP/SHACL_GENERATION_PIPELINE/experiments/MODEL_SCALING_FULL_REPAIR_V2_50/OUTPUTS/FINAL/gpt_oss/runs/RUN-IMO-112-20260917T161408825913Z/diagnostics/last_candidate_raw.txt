<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix : <urn:nltl:generated-shape:> .

:IMO-112 a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When operating area is Arctic waters, discharge of noxious liquid substances or mixtures to sea must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea ?operatingArea .
                FILTER ( ?operatingArea = nltl:arcticWaters )
                $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?nls .
                FILTER ( bound(?nls) && ?nls = true )
            }
        """ ;
    ] .
<END_SHACL>
