<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

urn:nltl:generated-shape:IMO114Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ships of Polar Category A or B built on or after 2017‑01‑01 that carry NLS requiring a type‑3 tank and have cargo tank type “3”, the administration carriage approval status must be Approved and the polar‑operation certificate entry must be present (true)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>
            SELECT $this WHERE {
                $this nltl:shipCategory ?cat .
                FILTER(?cat IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))

                $this nltl:constructionDate ?cdate .
                FILTER(?cdate >= "2017-01-01"^^xsd:date)

                $this nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?cNLS .
                FILTER(?cNLS = true)

                $this nltl:cargoTankType ?cTank .
                FILTER(?cTank = "3"^^xsd:string)

                OPTIONAL { $this nltl:administrationCarriageApprovalStatus ?admStatus . }
                OPTIONAL { $this nltl:applicableCertificatePolarOperationEntryPresent ?certPresent . }

                FILTER( !bound(?admStatus) || ?admStatus != nltl:evidenceStateApproved ||
                        !bound(?certPresent) || ?certPresent != true )
            }
        """ ;
    ] .
<END_SHACL>
