<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix g: <urn:nltl:generated-shape:> .

g:TRF-002_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # construction contract date must be present and be a date
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must have a construction contract date (xsd:date)." ;
    ] ;
    # the applicable ice‑class regulation edition may be present; when present it must be a regulated edition value
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:maxCount 1 ;
    ] ;
    # Conditional applicability: if the contract date is on/after 2019‑01‑01 and before 2021‑07‑05,
    # the ship must reference the 2017 ice‑class regulation edition.
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the construction contract date is on or after 2019‑01‑01 and before 2021‑07‑05, the applicable ice‑class regulation edition must be the 2017 edition." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER ( ?date >= "2019-01-01"^^xsd:date && ?date < "2021-07-05"^^xsd:date )
                OPTIONAL { $this nltl:applicableIceClassRegulationEdition ?edition . }
                FILTER ( !bound(?edition) || ?edition != nltl:iceClassRegulationEdition2017 )
            }
        """ ;
    ] .
<END_SHACL>
