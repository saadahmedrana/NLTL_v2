<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix : <urn:nltl:generated-shape:> .

:TRF-006_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    # Construction stage must be >= 1986-11-01 and < 2003-09-01
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "1986-11-01"^^xsd:date ;
        sh:maxExclusive "2003-09-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    # Owner request flag must be present and boolean
    sh:property [
        sh:path nltl:ownerRequested2008EngineOutputRequirements ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    # Applicable Ice‑Class Regulation Edition is fixed to 1985 rule edition
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:in ( nltl:iceClassRuleEdition1985 ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    # Conditional: when owner requests 2008 engine‑output requirements, the
    # engineOutputRegulationEdition must be the 2008 edition.
    sh:property [
        sh:path nltl:engineOutputRegulationEdition ;
        sh:class nltl:iceClassRegulationEditionValue ;
        sh:sparql [
            a sh:SPARQLConstraint ;
            sh:message "When ownerRequested2008EngineOutputRequirements is true, engineOutputRegulationEdition must be iceClassRegulationEdition2008." ;
            sh:select """
                PREFIX nltl: <https://w3id.org/nltl/vocab#>
                PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                SELECT $this (COUNT(*) AS ?violations) WHERE {
                  FILTER EXISTS {
                    $this nltl:ownerRequested2008EngineOutputRequirements true^^xsd:boolean .
                  }
                  FILTER NOT EXISTS {
                    $this nltl:engineOutputRegulationEdition nltl:iceClassRegulationEdition2008 .
                  }
                }
                HAVING (COUNT(*) > 0)
            """ ;
        ]
    ] .
<END_SHACL>
