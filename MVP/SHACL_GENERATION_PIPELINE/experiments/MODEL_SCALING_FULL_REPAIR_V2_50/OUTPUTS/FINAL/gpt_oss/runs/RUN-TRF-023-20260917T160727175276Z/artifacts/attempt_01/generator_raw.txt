@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF023Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:iceClassIb nltl:iceClassIc ) ;
        sh:message "Ship must have ice class IB or IC." ;
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:hasValue nltl:iceClassRuleEdition1985 ;
        sh:message "Applicable ice class regulation edition must be 1985 edition." ;
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:nodeKind sh:BlankNode ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum continuous rating power must be provided as a QuantityValue." ;
    ] ;
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:nodeKind sh:BlankNode ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Required power under 1985 rules must be provided as a QuantityValue." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be greater than or equal to required power under 1985 rules." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ( ?mcrVal AS ?value1 ) ( ?reqVal AS ?value2 )
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrQty .
                ?mcrQty qudt:numericValue ?mcrVal .
                $this nltl:requiredPowerUnder1985Rules ?reqQty .
                ?reqQty qudt:numericValue ?reqVal .
                FILTER ( ?mcrVal < ?reqVal )
            }
        """ ;
    ] .

@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .

    a sh:NodeShape ;

    # Applicability: ice class must be IB or IC
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:iceClassIb nltl:iceClassIc ) ;
        sh:message "Ship must have ice class IB or IC." ;

            }
