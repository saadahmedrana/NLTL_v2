<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF023Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # Applicability: ice class must be IB or IC
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:iceClassIb nltl:iceClassIc ) ;
        sh:message "Ship must have ice class IB or IC." ;
    ] ;
    # Applicability: regulation edition must be the 1985 edition
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:hasValue nltl:iceClassRuleEdition1985 ;
        sh:message "Applicable ice class regulation edition must be the 1985 edition." ;
    ] ;
    # Required input: maximum continuous rating power (numeric value in kW)
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum continuous rating power must be provided as a decimal value (kW)." ;
    ] ;
    # Required input: required power under 1985 rules (numeric value in kW)
    sh:property [
        sh:path nltl:requiredPowerUnder1985Rules ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Required power under 1985 rules must be provided as a decimal value (kW)." ;
    ] ;
    # Comparison: MCR power >= required power
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be greater than or equal to required power under 1985 rules." ;
        sh:select """
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ( ?mcrVal AS ?value1 ) ( ?reqVal AS ?value2 )
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrVal .
                $this nltl:requiredPowerUnder1985Rules ?reqVal .
                FILTER ( ?mcrVal < ?reqVal )
            }
        """ ;
    ] .
<END_SHACL>
