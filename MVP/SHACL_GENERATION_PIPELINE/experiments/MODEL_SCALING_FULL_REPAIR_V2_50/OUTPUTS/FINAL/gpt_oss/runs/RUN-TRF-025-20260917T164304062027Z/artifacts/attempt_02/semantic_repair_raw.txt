<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/2001/sw/DataAccess/rf1/result-set#> .

@prefix qudt: <http://qudt.org/schema/qudt#> .

<urn:nltl:generated-shape:TRF-025-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    # --- Required ice class (must be IA Super) ---
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:iceClassIaSuper
    ] ;

    # --- Required bulbous bow presence (must be false) ---
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;

    # --- Operand quantities ----------------------------------------------------
    # Ship breadth (B)
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "0"^^xsd:decimal
    ] ;

    # Ship length (L)
    sh:property [
        sh:path nltl:shipLength ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "0"^^xsd:decimal
    ] ;

    # Ice class draught (T)
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "0"^^xsd:decimal
    ] ;

    # Coefficient f1

        sh:path nltl:coefficientF1 ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
