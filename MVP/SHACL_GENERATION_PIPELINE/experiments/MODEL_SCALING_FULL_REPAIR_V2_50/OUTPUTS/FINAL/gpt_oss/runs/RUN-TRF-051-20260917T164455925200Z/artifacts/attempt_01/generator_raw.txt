<BEGIN_SHACL>
@prefix sh:   <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

urn:nltl:generated-shape:TRF-051_FrameWebThickness_Min9mm
    a sh:NodeShape ;
    sh:targetClass nltl:hullStructure ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "9"^^xsd:decimal ;
                sh:message "Frame web thickness must be at least 9 mm." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:message "Frame web thickness must be expressed in millimetres (mm)." ;
            ] ;
        ] ;
    ] .
<END_SHACL>
