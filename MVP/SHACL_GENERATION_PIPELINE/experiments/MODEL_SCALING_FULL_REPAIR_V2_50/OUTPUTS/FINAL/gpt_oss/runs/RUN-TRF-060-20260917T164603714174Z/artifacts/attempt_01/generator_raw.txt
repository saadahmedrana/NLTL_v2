<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix gen: <urn:nltl:generated-shape:> .

# Reusable shape for a QUDT QuantityValue with correct unit
gen:QuantityValueShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Numeric value (xsd:decimal) is required." ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Unit IRI is required." ;
    ] .

# Shape for the table lookup case that supplies gamma selection evidence
gen:TableLookupCaseShape a sh:NodeShape ;
    sh:targetClass nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "Evidence of table lookup selection is required." ;
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable4Dash8 ;
        sh:message "Table reference must be the controlled individual traficomTable4Dash8." ;
    ] .

# Main shape targeting the ship
