<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix ex: <urn:nltl:generated-shape:> .

# Shape targeting each Ship
ex:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node ex:PropBladeLoadCaseShape
    ] .

# Shape for a propeller blade load case
ex:PropBladeLoadCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseBlade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:propellerBlade
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseIceBlock ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:iceBlock
    ] ;
    sh:xone ( ex:BackwardForceChoice ex:ForwardForceChoice ) .

# Exclusive choice: exactly one of backwardBladeForce or forwardBladeForce must be present

ex:BackwardForceChoice
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node ex:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:maxCount 0
    ] .

ex:ForwardForceChoice
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node ex:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:maxCount 0
    ] .

# Quantity shape for backwardBladeForce / forwardBladeForce
ex:ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
