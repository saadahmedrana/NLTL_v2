    a sh:NodeShape ;

        sh:minCount 1 ;

    # ---- alternative compliance paths (approved analysis OR calculation) ----------
    sh:or (

              sh:minCount 1 ;
              sh:maxCount 1

        # Path B: formula‑based calculation must be consistent
        [ sh:sparql [

              sh:select """\
                  PREFIX nltl: <https://w3id.org/nltl/vocab#>

                  PREFIX qudt: <http://qudt.org/schema/qudt#>
