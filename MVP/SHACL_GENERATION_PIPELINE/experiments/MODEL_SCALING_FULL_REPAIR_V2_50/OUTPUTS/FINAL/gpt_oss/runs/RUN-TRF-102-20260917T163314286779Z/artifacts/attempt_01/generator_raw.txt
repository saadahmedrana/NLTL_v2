  FILTER NOT EXISTS { $this nltl:approvedSpindleTorqueStressAnalysis ?analysis . }

  $this nltl:bladeExpandedAreaRatio ?earQty .

  FILTER (datatype(?ear) = xsd:decimal || datatype(?ear) = xsd:float || datatype(?ear) = xsd:double)

  FILTER (datatype(?bladeCount) = xsd:integer)

  $this nltl:leadingEdgeChordPortionAtZeroPointEightRadius ?cLeQty .
  ?cLeQty qudt:numericValue ?cLe .
  $this nltl:trailingEdgeChordPortionAtZeroPointEightRadius ?cTeQty .
  ?cTeQty qudt:numericValue ?cTe .

  $this nltl:extremeIceForce ?iceForceQty .
  ?iceForceQty qudt:numericValue ?Fex .

  # compute C_spex
  BIND ((4.0 * ?ear) / ?bladeCount AS ?x)
  BIND (?x * ?x * ?x AS ?x3)
  BIND (0.7 * (1.0 - ?x3) AS ?cTemp)

  $this nltl:maximumSpindleTorque ?maxTorqueQty .
  ?maxTorqueQty qudt:numericValue ?maxTorqueReported .

  # tolerance

  BIND (ABS(?factorReported - ?cSpexCalc) AS ?diffFactor)
  BIND (ABS(?torqueReported - ?qSexCalc) AS ?diffTorque)
