@prefix sh:   <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .

@prefix qudt: <http://qudt.org/schema/qudt#> .

# -----------------------------------------------------------------
# Shape targeting the ship entity for the TRF‑116 requirement

urn:nltl:generated-shape:TRF-116_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # -------------------------------------------------------------
    # Boolean flag indicating the exception case
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:propellerHubOrThrusterEndCapImpact ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Flag propellerHubOrThrusterEndCapImpact (true if the impact is on a propeller hub or thruster end‑cap) must be provided." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Non‑hemispherical impact contact area (A)
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "numericValue for nonHemisphericalImpactContactArea must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:message "Unit for nonHemisphericalImpactContactArea must be square metres (unit:M2)." ;
            ] ;
        ] ;
        sh:message "Non‑hemispherical impact contact area (A) must be supplied with a numeric value and unit m²." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Design ice thickness (H_iced)
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "numericValue for designIceThickness must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:message "Unit for designIceThickness must be metres (unit:M)." ;
            ] ;
        ] ;
        sh:message "Design ice thickness (H_iced) must be supplied with a numeric value and unit m." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Equivalent impact sphere radius (R_ceq) – result of the calculation
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:message "numericValue for equivalentImpactSphereRadius must be a decimal." ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:message "Unit for equivalentImpactSphereRadius must be metres (unit:M)." ;
