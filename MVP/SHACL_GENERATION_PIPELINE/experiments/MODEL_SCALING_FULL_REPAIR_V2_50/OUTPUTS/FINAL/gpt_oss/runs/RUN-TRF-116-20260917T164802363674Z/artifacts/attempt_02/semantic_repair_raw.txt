<BEGIN_SHACL>
@prefix sh:   <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix g:    <urn:nltl:generated-shape:> .

# -----------------------------------------------------------------
# Shape targeting the ship entity for the TRF‑116 requirement
# -----------------------------------------------------------------
g:TRF-116_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # -------------------------------------------------------------
    # Boolean flag indicating the exception case
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:propellerHubOrThrusterEndCapImpact ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Flag propellerHubOrThrusterEndCapImpact (true if the impact is on a propeller hub or thruster end‑cap) must be provided." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Non‑hemispherical impact contact area (A)
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Non‑hemispherical impact contact area (A) must be supplied as a decimal value (metres²)." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Design ice thickness (H_iced)
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Design ice thickness (H_iced) must be supplied as a decimal value (metres)." ;
    ] ;
    # -------------------------------------------------------------
    # Quantity: Equivalent impact sphere radius (R_ceq) – result of the calculation
    # -------------------------------------------------------------
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Equivalent impact sphere radius (R_ceq) must be provided as a decimal value (metres) when the rule applies." ;
    ] ;
    # -------------------------------------------------------------
    # Conditional SPARQL constraint:
    #   If the exception flag is false, the three quantities above must all be present.
    # -------------------------------------------------------------
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When propellerHubOrThrusterEndCapImpact = false, the ship must provide nonHemisphericalImpactContactArea, designIceThickness and equivalentImpactSphereRadius." ;
        sh:select """ 
            PREFIX sh:   <http://www.w3.org/ns/shacl#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:propellerHubOrThrusterEndCapImpact ?exc .
                FILTER ( ?exc = false ) .
                FILTER NOT EXISTS { $this nltl:nonHemisphericalImpactContactArea ?a . }
                FILTER NOT EXISTS { $this nltl:designIceThickness ?d . }
                FILTER NOT EXISTS { $this nltl:equivalentImpactSphereRadius ?r . }
            }
        """ ;
    ] .
<END_SHACL>
