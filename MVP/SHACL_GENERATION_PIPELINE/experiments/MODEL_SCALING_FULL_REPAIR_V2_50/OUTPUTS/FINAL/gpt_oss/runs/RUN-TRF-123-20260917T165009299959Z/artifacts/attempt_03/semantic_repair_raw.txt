<BEGIN_SHACL>
@prefix sh:   <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .

<urn:nltl:generated-shape:TRF123Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Propeller blades must be excluded from occasional‑force scope (true)." 
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "Component stress exceeds the allowed limit (material yield strength × safety factor)." ;
                sh:select """ 
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt#>
                    PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this WHERE {
                        $this nltl:occasionalForceCaseAssessedComponent ?comp .

                        # Retrieve numeric stress value from QuantityValue
                        ?comp nltl:occasionalForceComponentStress ?stressNode .
                        ?stressNode qudt:numericValue ?stressVal .
                        FILTER(datatype(?stressVal) = xsd:decimal) .

                        # Retrieve numeric yield strength from QuantityValue
                        ?comp nltl:hasMaterialProperties ?mat .
                        ?mat nltl:componentMaterialYieldStrength ?yieldNode .
                        ?yieldNode qudt:numericValue ?yieldVal .
                        FILTER(datatype(?yieldVal) = xsd:decimal) .

                        # Retrieve numeric safety factor from QuantityValue
                        $this nltl:componentYieldSafetyFactor ?sfNode .
                        ?sfNode qudt:numericValue ?sfVal .
                        FILTER(datatype(?sfVal) = xsd:decimal) .

                        BIND(?yieldVal * ?sfVal AS ?allowable) .
                        FILTER( ?stressVal > ?allowable )
                    }
                """ ;
            ]
        ]
    ] .
<END_SHACL>
