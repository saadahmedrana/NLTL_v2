@prefix gen: <urn:nltl:generated-shape:> .

@prefix shacl: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF-132Shape a shacl:NodeShape ;

    # --- Engine output formula base power P0 (kW) ---

        shacl:path nltl:engineOutputFormulaBasePowerP0 ;

        shacl:maxCount 1 ;

            shacl:class qudt:QuantityValue ;
            shacl:property [
                shacl:path qudt:numericValue ;
                shacl:datatype xsd:decimal ;

                shacl:maxCount 1
            ] ;
            shacl:property [

                shacl:maxCount 1

                shacl:maxCount 1
            ] ;

                shacl:path qudt:unit ;

                shacl:minCount 1 ;
                shacl:maxCount 1
