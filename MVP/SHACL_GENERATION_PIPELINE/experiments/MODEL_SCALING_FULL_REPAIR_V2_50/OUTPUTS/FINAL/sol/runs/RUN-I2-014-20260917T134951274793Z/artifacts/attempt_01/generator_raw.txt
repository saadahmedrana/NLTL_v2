gs:BowSubregionCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:BowLengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The bow region must be represented as four subregions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:node gs:BowSubregionCalculationCaseShape ;
        sh:message "Exactly four bow subregion calculation cases are required."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:LineLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:PressureQuantityShape
    ] ;
    sh:sparql [
        sh:message "The four calculation cases must represent the mid-length positions of four equal bow-length subregions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?lengthQuantity qudt:numericValue ?length .
                ?case nltl:bowSubregionMidLengthPosition ?positionQuantity .
                ?positionQuantity qudt:numericValue ?position .

                BIND((?length / "8"^^xsd:decimal) AS ?expected1)
                BIND((?length * "3"^^xsd:decimal / "8"^^xsd:decimal) AS ?expected2)
                BIND((?length * "5"^^xsd:decimal / "8"^^xsd:decimal) AS ?expected3)
                BIND((?length * "7"^^xsd:decimal / "8"^^xsd:decimal) AS ?expected4)

                BIND(IF(
                    ABS(?expected1) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                    ABS(?expected1) * "0.000001"^^xsd:decimal,
                    "0.000000001"^^xsd:decimal
                ) AS ?tolerance1)
                BIND(IF(
                    ABS(?expected2) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                    ABS(?expected2) * "0.000001"^^xsd:decimal,
                    "0.000000001"^^xsd:decimal
                ) AS ?tolerance2)
                BIND(IF(
                    ABS(?expected3) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                    ABS(?expected3) * "0.000001"^^xsd:decimal,
                    "0.000000001"^^xsd:decimal
                ) AS ?tolerance3)
                BIND(IF(
                    ABS(?expected4) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                    ABS(?expected4) * "0.000001"^^xsd:decimal,
                    "0.000000001"^^xsd:decimal
                ) AS ?tolerance4)

                BIND(IF(ABS(?position - ?expected1) <= ?tolerance1, 1, 0) AS ?match1)
                BIND(IF(ABS(?position - ?expected2) <= ?tolerance2, 1, 0) AS ?match2)
                BIND(IF(ABS(?position - ?expected3) <= ?tolerance3, 1, 0) AS ?match3)
                BIND(IF(ABS(?position - ?expected4) <= ?tolerance4, 1, 0) AS ?match4)
            }
            GROUP BY $this
            HAVING (
                SUM(?match1) != 1 ||
                SUM(?match2) != 1 ||
                SUM(?match3) != 1 ||
                SUM(?match4) != 1
            )
        """
    ] ;
    sh:sparql [
        sh:message "The selected bow force, line load and pressure must equal the respective maxima reported by the four subregion calculation cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?selectedForce ?selectedLineLoad ?selectedPressure
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForceQuantity ;
                      nltl:selectedMaximumBowLineLoad ?selectedLineLoadQuantity ;
                      nltl:selectedMaximumBowPressure ?selectedPressureQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .

                ?selectedForceQuantity qudt:numericValue ?selectedForce .
                ?selectedLineLoadQuantity qudt:numericValue ?selectedLineLoad .
                ?selectedPressureQuantity qudt:numericValue ?selectedPressure .

                ?case nltl:bowSubregionCalculatedForce ?forceQuantity ;
                      nltl:bowSubregionCalculatedLineLoad ?lineLoadQuantity ;
                      nltl:bowSubregionCalculatedPressure ?pressureQuantity .

                ?forceQuantity qudt:numericValue ?force .
                ?lineLoadQuantity qudt:numericValue ?lineLoad .
                ?pressureQuantity qudt:numericValue ?pressure .
            }
            GROUP BY $this ?selectedForce ?selectedLineLoad ?selectedPressure
            HAVING (
                ABS(?selectedForce - MAX(?force)) >
                    IF(
                        ABS(MAX(?force)) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                        ABS(MAX(?force)) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                ||
                ABS(?selectedLineLoad - MAX(?lineLoad)) >
                    IF(
                        ABS(MAX(?lineLoad)) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                        ABS(MAX(?lineLoad)) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                ||
                ABS(?selectedPressure - MAX(?pressure)) >
                    IF(
                        ABS(MAX(?pressure)) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                        ABS(MAX(?pressure)) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
            )
        """
    ] .
