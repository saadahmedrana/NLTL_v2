    sh:property [
        sh:path nltl:platingHullAreaClassification ;

                        nltl:bowIntermediateBottomHullArea

                    sh:class nltl:hullAreaValue

        ] ;
        sh:message "Plating must have a hull-area classification."
    ] ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:I2-022AngleQuantityShape ;
        sh:message "Plating must have one framing angle in degrees."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:framingAngleOmega ;
                sh:node gs:I2-022AngleBelowApplicabilityShape

            sh:property [

                sh:node gs:I2-022ApplicableAngleShape

            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;

                sh:minCount 1 ;

                sh:message "Applicable plating must identify its supporting structural member and member-owned calculation inputs."

                sh:message "Applicable plating must be linked to a ship providing the ship-owned calculation inputs."
