<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO016ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:node gen:IMO016PolarShipCertificateFormShape ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate corresponding to the Appendix I model is required."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The certificate language is required."
    ] ;
    sh:property [
        sh:path nltl:approvedTranslationPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:in (
                    "English"^^xsd:string
                    "French"^^xsd:string
                    "Spanish"^^xsd:string
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:approvedTranslationPresent ;
                sh:hasValue true ;
                sh:message "An approved translation is required when the certificate language is not English, French, or Spanish."
            ]
        ]
    ) .

gen:IMO016PolarShipCertificateFormShape
    a sh:NodeShape ;
    sh:class nltl:polarShipCertificateForm ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:in (
            nltl:appendixIPolarShipCertificateModel
        ) ;
        sh:message "The Polar Ship Certificate form must correspond to the Appendix I model."
    ] .
<END_SHACL>
