<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF070ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF070NonApplicableShipState
        gen:TRF070ApplicableShipState
    ) .

gen:TRF070NonApplicableShipState
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF070ApplicableShipState
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:in ( "azimuthing" "fixed" ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gen:TRF070ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:node gen:TRF070ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:minCount 1 ;
        sh:node gen:TRF070PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyLocalStrength ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The externally assessed thruster-body local strength must be sufficient to withstand the applicable local ice pressure."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:class nltl:designCondition ;
        sh:minCount 1 ;
        sh:node gen:TRF070DesignConditionShape
    ] .

gen:TRF070DesignConditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designConditionLocalStrengthCapacity ;
        sh:minCount 1 ;
        sh:node gen:TRF070ForceQuantityShape
    ] .

gen:TRF070ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:maxCount 1
    ] .

gen:TRF070PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:maxCount 1
    ] .
<END_SHACL>
