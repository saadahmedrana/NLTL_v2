<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-069-Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:devicePowerSourceIndependentOfMainPower ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If an added life-saving device requires power, its power source must be independent of the ship's main power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this nltl:addedLifeSavingDeviceRequiresPower true .
                FILTER NOT EXISTS {
                    ?this nltl:devicePowerSourceIndependentOfMainPower true .
                }
            }
        """
    ] .
</END_SHACL>
