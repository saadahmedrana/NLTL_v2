<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:ShipReadinessShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape shape:BenchmarkEntityReadinessShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "Ship must have a benchmark entity component with required properties." ;
    ] ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Design ice thickness must be provided with unit M." ;
    ] ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Non hemispherical impact contact area must be provided with unit M2." ;
    ] ;
    sh:property [
        sh:path nltl:propellerHubOrThrusterEndCapImpact ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Propeller hub or thruster end cap impact indicator must be a boolean." ;
    ] .

shape:BenchmarkEntityReadinessShape a sh:NodeShape ;
    sh:class nltl:benchmarkEntity ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "Equivalent impact sphere radius must be estimated and provided with unit M." ;
    ] ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Non hemispherical impact area identifier must be a string." ;
    ] .
<END_SHACL>
