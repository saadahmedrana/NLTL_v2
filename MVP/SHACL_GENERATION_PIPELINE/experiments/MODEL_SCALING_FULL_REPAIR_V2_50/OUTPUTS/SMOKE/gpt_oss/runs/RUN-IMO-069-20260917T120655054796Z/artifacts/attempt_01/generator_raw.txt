<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix rs: <urn:nltl:generated-shape:> .

rs:AddedDevicePowerShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    
    # datatype checks (optional presence)
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean ;

    sh:property [
        sh:path nltl:devicePowerSourceIndependentOfMainPower ;
        sh:datatype xsd:boolean ;
    ] ;
    
    # conditional implication: if power‑requiring device => independent source must be true
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If a life‑saving device requires power, its power source must be independent of the ship's main power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:addedLifeSavingDeviceRequiresPower ?req .
                FILTER(?req = true) .
                $this nltl:devicePowerSourceIndependentOfMainPower ?ind .
                FILTER(?ind != true)
            }
        """ ;
    ] .
<END_SHACL>
