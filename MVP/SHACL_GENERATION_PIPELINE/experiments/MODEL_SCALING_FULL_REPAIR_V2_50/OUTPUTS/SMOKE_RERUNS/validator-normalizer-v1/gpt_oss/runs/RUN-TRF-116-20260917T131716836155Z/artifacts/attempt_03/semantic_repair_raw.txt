<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt#> .

<urn:nltl:generated-shape:TRF-116Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    
    ## Operand: non‑hemispherical impact contact area (A)
    sh:property [
        a sh:PropertyShape ;
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Non‑hemispherical impact contact area (A) must be provided as decimal value."
    ] ;
    
    ## Operand: design ice thickness (H_iced)
    sh:property [
        a sh:PropertyShape ;
        sh:path nltl:designIceThickness ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "Design ice thickness (H_iced) must be provided as decimal value."
    ] ;
    
    ## Operand: propeller‑hub or thruster‑end‑cap impact flag
    sh:property [
        a sh:PropertyShape ;
        sh:path nltl:propellerHubOrThrusterEndCapImpact ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Flag indicating propeller‑hub or thruster‑end‑cap impact must be provided."
    ] ;
    
    ## Result: equivalent impact sphere radius (R_ceq)
    sh:property [
        a sh:PropertyShape ;
        sh:path nltl:equivalentImpactSphereRadius ;
