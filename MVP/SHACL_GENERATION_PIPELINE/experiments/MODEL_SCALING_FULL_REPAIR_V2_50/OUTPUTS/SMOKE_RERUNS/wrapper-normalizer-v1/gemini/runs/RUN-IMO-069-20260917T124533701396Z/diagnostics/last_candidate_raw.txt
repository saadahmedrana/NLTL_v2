<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:ShipPowerIndependenceShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:devicePowerSourceIndependentOfMainPower ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If an added life-saving device requires power, its power source must operate independently of the ship's main power source." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:addedLifeSavingDeviceRequiresPower true .
                FILTER NOT EXISTS {
                    $this nltl:devicePowerSourceIndependentOfMainPower true .
                }
            }
        """ ;
    ] .
<END_SHACL>
