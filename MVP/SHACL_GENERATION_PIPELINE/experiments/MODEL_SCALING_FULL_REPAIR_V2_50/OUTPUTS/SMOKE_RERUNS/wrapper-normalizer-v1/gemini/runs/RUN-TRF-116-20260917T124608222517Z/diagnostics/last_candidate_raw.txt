<BEGIN_SHACL>
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:QuantityValueM a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

shape:QuantityValueM2 a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

shape:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:node shape:QuantityValueM ;
    ] ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:node shape:QuantityValueM2 ;
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node shape:QuantityValueM ;
    ] ;
    sh:property [
        sh:path nltl:propellerHubOrThrusterEndCapImpact ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactArea ;
        sh:datatype xsd:string ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a non-hemispherical impact area is specified, the design ice thickness, non-hemispherical impact contact area, propeller hub/thruster end cap impact flag, and estimated equivalent impact sphere radius must all be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:nonHemisphericalImpactArea ?area .
                FILTER (
                    NOT EXISTS { $this nltl:designIceThickness ?t } ||
                    NOT EXISTS { $this nltl:nonHemisphericalImpactContactArea ?a } ||
                    NOT EXISTS { $this nltl:propellerHubOrThrusterEndCapImpact ?p } ||
                    NOT EXISTS { $this nltl:equivalentImpactSphereRadius ?r }
                )
            }
        """
    ] .
<END_SHACL>
