<BEGIN_SHACL>
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:ShipPowerDependencyShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    # the property that triggers the conditional rule – must be present and boolean
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    # the dependent property – datatype checked when present
    sh:property [
        sh:path nltl:devicePowerSourceIndependentOfMainPower ;
        sh:datatype xsd:boolean
    ] ;
    # conditional check: when the trigger is true, the dependent must also be true
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If a ship has an added life‑saving device that requires power, its power source must be independent of the ship's main power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this WHERE {
                $this nltl:addedLifeSavingDeviceRequiresPower ?trigger .
                FILTER ( ?trigger = true )
                $this nltl:devicePowerSourceIndependentOfMainPower ?independent .
                FILTER ( ?independent != true )
            }
        """ ;
    ] .
<END_SHACL>
