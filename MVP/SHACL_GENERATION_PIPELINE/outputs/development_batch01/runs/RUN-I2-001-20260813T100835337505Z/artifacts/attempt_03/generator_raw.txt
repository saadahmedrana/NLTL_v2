<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ConstructionContractDateShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must have exactly one construction contract date as an xsd:date."
    ] .

gs:UR_I2_Rev4ApplicableShipShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT ?this
WHERE {
    ?this a nltl:ship ;
          nltl:constructionContractDate ?contractDate .
    FILTER(
        datatype(?contractDate) = xsd:date &&
        ?contractDate >= "2021-01-01"^^xsd:date
    )
}
"""
    ] ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2021-01-01"^^xsd:date
    ] .
<END_SHACL>
