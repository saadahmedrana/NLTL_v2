<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:PolarClassUrI2ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A Polar Class ship according to UR I1 must have UR I1 compliance and UR I2 applicability set to true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:polarClass ?polarClass .
    FILTER (
        ?polarClass IN (
            nltl:polarClassPc1,
            nltl:polarClassPc2,
            nltl:polarClassPc3,
            nltl:polarClassPc4,
            nltl:polarClassPc5,
            nltl:polarClassPc6,
            nltl:polarClassPc7
        )
    )
    FILTER NOT EXISTS {
        $this nltl:urI1ComplianceStatus ?urI1Status ;
              nltl:urI2Applicability ?urI2Status .
        FILTER (
            datatype(?urI1Status) = xsd:boolean &&
            datatype(?urI2Status) = xsd:boolean &&
            ?urI1Status = "true"^^xsd:boolean &&
            ?urI2Status = "true"^^xsd:boolean
        )
    }
}
"""
    ] .
<END_SHACL>
