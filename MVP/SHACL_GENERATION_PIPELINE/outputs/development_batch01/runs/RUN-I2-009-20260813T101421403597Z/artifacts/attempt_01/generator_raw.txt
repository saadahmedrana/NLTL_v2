<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceLoadParametersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gen:PressureValueShape
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node gen:LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node gen:LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node gen:DimensionlessValueShape
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:node gen:AngleValueShape ;
        sh:message "A buttock angle in degrees is required to determine bow-form applicability."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:node gen:AngleValueShape ;
        sh:message "A normal frame angle in degrees is required to determine bow-form applicability."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Bow-form applicability must be recorded as a boolean."
    ] ;
    sh:or (
        gen:ApplicableBowFormShape
        gen:NonApplicableBowFormShape
    ) ;
    sh:sparql [
        sh:message "Other ice-strengthened areas require average ice pressure, non-bow load-patch dimensions, and a load-patch aspect ratio of 3.6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hullArea ?area .
                FILTER (LCASE(STR(?area)) = "other ice-strengthened areas")

                OPTIONAL {
                    $this nltl:averageIcePressure ?pressure .
                    ?pressure qudt:numericValue ?pressureValue ;
                              qudt:unit unit:MegaPA .
                }
                OPTIONAL {
                    $this nltl:nonBowLoadPatchHeight ?height .
                    ?height qudt:numericValue ?heightValue ;
                            qudt:unit unit:M .
                }
                OPTIONAL {
                    $this nltl:nonBowLoadPatchLength ?length .
                    ?length qudt:numericValue ?lengthValue ;
                            qudt:unit unit:M .
                }
                OPTIONAL {
                    $this nltl:loadPatchAspectRatio ?ratio .
                    ?ratio qudt:numericValue ?ratioValue ;
                           qudt:unit unit:UNITLESS .
                    FILTER (?ratioValue = "3.6"^^xsd:decimal)
                }

                FILTER (
                    !BOUND(?pressureValue) ||
                    !BOUND(?heightValue) ||
                    !BOUND(?lengthValue) ||
                    !BOUND(?ratioValue)
                )
            }
        """
    ] .

gen:PressureValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:MegaPA )
    ] .

gen:LengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:M )
    ] .

gen:DimensionlessValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:UNITLESS )
    ] .

gen:AngleValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:DEG )
    ] .

gen:ApplicableBowFormShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node gen:PositiveButLessThan80AngleShape
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node gen:GreaterThan10AngleShape
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:in ( true )
    ] .

gen:NonApplicableBowFormShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:in ( false )
    ] ;
    sh:or (
        gen:NonPositiveButtockAngleShape
        gen:ButtockAngleAtLeast80Shape
        gen:NormalFrameAngleAtMost10Shape
    ) .

gen:PositiveButLessThan80AngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minExclusive "0"^^xsd:decimal ;
        sh:maxExclusive "80"^^xsd:decimal
    ] .

gen:GreaterThan10AngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minExclusive "10"^^xsd:decimal
    ] .

gen:NonPositiveButtockAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node gen:AtMostZeroAngleValueShape
    ] .

gen:ButtockAngleAtLeast80Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node gen:AtLeast80AngleValueShape
    ] .

gen:NormalFrameAngleAtMost10Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node gen:AtMost10AngleValueShape
    ] .

gen:AtMostZeroAngleValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:maxInclusive "0"^^xsd:decimal
    ] .

gen:AtLeast80AngleValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive "80"^^xsd:decimal
    ] .

gen:AtMost10AngleValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:maxInclusive "10"^^xsd:decimal
    ] .
<END_SHACL>
