gs:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:ButtockAngleValueShape ;
        sh:message "The buttock angle must be greater than 0 degrees and less than 80 degrees."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:NormalFrameAngleValueShape ;
        sh:message "The normal frame angle must be greater than 10 degrees."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hullArea ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullArea ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:averageIcePressure ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:AverageIcePressureValueShape ;
                sh:message "Other ice-strengthened areas require one average ice pressure in MPa."
            ] ;
            sh:property [
                sh:path nltl:nonBowLoadPatchLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:NonBowLengthValueShape ;
                sh:message "Other ice-strengthened areas require one non-bow load-patch length in metres."
            ] ;
            sh:property [
                sh:path nltl:nonBowLoadPatchHeight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:NonBowLengthValueShape ;
                sh:message "Other ice-strengthened areas require one non-bow load-patch height in metres."
            ] ;
            sh:property [
                sh:path nltl:loadPatchAspectRatio ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:LoadPatchAspectRatioValueShape ;
                sh:message "The fixed load-patch aspect ratio must be 3.6."
            ] ;
            sh:sparql [
                sh:message "The non-bow load-patch length must equal 3.6 times its height within numerical tolerance." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hullArea ?area ;
                              nltl:nonBowLoadPatchLength ?lengthQuantity ;
                              nltl:nonBowLoadPatchHeight ?heightQuantity .
                        ?lengthQuantity qudt:numericValue ?length ;
                                        qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?heightQuantity qudt:numericValue ?height ;
                                        qudt:unit <http://qudt.org/vocab/unit/M> .
                        BIND((3.6 * ?height) AS ?expectedLength)
                        BIND((ABS(?expectedLength) * 0.000001) AS ?relativeTolerance)
                        BIND(
                            IF(
                                ?relativeTolerance > 0.000000000001,
                                ?relativeTolerance,
                                0.000000000001
                            )
                            AS ?tolerance
                        )
                        FILTER(ABS(?length - ?expectedLength) > ?tolerance)
                    }
                """
            ]
        ]
    ) .
