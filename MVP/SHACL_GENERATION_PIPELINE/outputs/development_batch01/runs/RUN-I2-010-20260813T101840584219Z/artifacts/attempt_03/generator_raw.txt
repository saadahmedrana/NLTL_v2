<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010RequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:NonPc6Pc7ShipShape
        gen:ApplicablePc6Pc7ShipShape
        gen:NonApplicablePc6Pc7ShipShape
    ) .

gen:Pc6Pc7ShipShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc6
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc7
            ]
        ]
    ) .

gen:NonPc6Pc7ShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:qualifiedValueShape [
            sh:in (
                nltl:polarClassPc1
                nltl:polarClassPc2
                nltl:polarClassPc3
                nltl:polarClassPc4
                nltl:polarClassPc5
            )
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:not gen:Pc6Pc7ShipShape .

gen:BowVerticalTrueShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] .

gen:BowVerticalFalseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] .

gen:NormalFrameAngle0To10DegreeValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 0.0 ;
        sh:maxInclusive 10.0 ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1
    ] .

gen:NormalFrameAngleInRangeShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:qualifiedValueShape gen:NormalFrameAngle0To10DegreeValueShape ;
        sh:qualifiedMinCount 1
    ] .

gen:ApplicablePc6Pc7ShipShape
    a sh:NodeShape ;
    sh:node gen:Pc6Pc7ShipShape ;
    sh:or (
        [
            sh:node gen:BowVerticalTrueShape
        ]
        [
            sh:node gen:NormalFrameAngleInRangeShipShape
        ]
    ) ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string ;
        sh:hasValue "I2.3.2.1 (iv)"^^xsd:string ;
        sh:message "Applicable PC6 or PC7 ships must use the design ice force method in I2.3.2.1 (iv)."
    ] .

gen:NonApplicablePc6Pc7ShipShape
    a sh:NodeShape ;
    sh:node gen:Pc6Pc7ShipShape ;
    sh:node gen:BowVerticalFalseShape ;
    sh:not gen:BowVerticalTrueShape ;
    sh:not gen:NormalFrameAngleInRangeShipShape .
<END_SHACL>
