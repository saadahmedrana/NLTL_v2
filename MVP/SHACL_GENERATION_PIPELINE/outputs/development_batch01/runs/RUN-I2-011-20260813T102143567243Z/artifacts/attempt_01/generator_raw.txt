<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BowDesignForceValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1
    ] .

gen:ShapeCoefficientValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 0.6 ;
        sh:maxInclusive 0.6 ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1
    ] .

gen:LoadPatchAspectRatioValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.3 ;
        sh:maxInclusive 1.3 ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1
    ] .

gen:PolarClassBulbousBowDesignForceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                )
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                )
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:bowDesignForce ;
                sh:class qudt:QuantityValue ;
                sh:node gen:BowDesignForceValueShape ;
                sh:minCount 1 ;
                sh:message "A bow design force in kilonewtons is required for a PC6 or PC7 ship equipped with a bulbous bow."
            ] ;
            sh:property [
                sh:path nltl:shapeCoefficient ;
                sh:class qudt:QuantityValue ;
                sh:node gen:ShapeCoefficientValueShape ;
                sh:minCount 1 ;
                sh:message "The applicable shape coefficient must be 0.6 unitless."
            ] ;
            sh:property [
                sh:path nltl:loadPatchAspectRatio ;
                sh:class qudt:QuantityValue ;
                sh:node gen:LoadPatchAspectRatioValueShape ;
                sh:minCount 1 ;
                sh:message "The applicable load patch aspect ratio must be 1.3 unitless."
            ]
        ]
    ) .
<END_SHACL>
