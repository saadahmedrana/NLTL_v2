<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:AverageIcePressureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The selected ice force must be provided in kN."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive "0"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The selected positive load-patch height must be provided in metres."
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive "0"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The selected positive load-patch length must be provided in metres."
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Average ice pressure must be provided in MPa."
    ] ;
    sh:sparql [
        sh:message "Average ice pressure must equal ice force divided by load-patch height and length, with kN converted to MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceForce ?forceQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:loadPatchLength ?lengthQuantity ;
                      nltl:averageIcePressure ?pressureQuantity .
                ?forceQuantity qudt:numericValue ?force ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?lengthQuantity qudt:numericValue ?length ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?pressureQuantity qudt:numericValue ?pressure ;
                                  qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                BIND(
                    xsd:decimal(?force) /
                    (xsd:decimal(?height) * xsd:decimal(?length) * "1000"^^xsd:decimal)
                    AS ?expected
                )
                BIND(ABS(?expected) * "0.000001"^^xsd:decimal AS ?relativeTolerance)
                BIND(
                    IF(
                        ?relativeTolerance > "0.000000000001"^^xsd:decimal,
                        ?relativeTolerance,
                        "0.000000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(xsd:decimal(?pressure) - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
