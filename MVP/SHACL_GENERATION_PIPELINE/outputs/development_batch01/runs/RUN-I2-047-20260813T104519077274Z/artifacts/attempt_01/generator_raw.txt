<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SteelRenewalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one gauged thickness in metres is required."
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one net thickness in metres is required."
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be marked as required when gauged thickness is less than net thickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:gaugedThickness ?gaugedQuantity ;
                      nltl:netThickness ?netQuantity .
                ?gaugedQuantity qudt:numericValue ?gauged .
                ?netQuantity qudt:numericValue ?net .
                FILTER (?gauged < (?net + "0.0005"^^xsd:decimal))
                FILTER NOT EXISTS {
                    $this nltl:steelRenewalRequired ?renewal .
                    FILTER (?renewal = true)
                }
            }
        """
    ] .
<END_SHACL>
