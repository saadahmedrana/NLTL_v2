<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:StressQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "A stress quantity requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:in ( <http://qudt.org/vocab/unit/MegaPA> ) ;
        sh:message "A stress quantity must use megapascal."
    ] .

gen:StrengthCriteriaShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:node gen:StressQuantityValueShape ;
        sh:message "Design stress is required and must be expressed in megapascal."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:node gen:StressQuantityValueShape ;
        sh:message "Permissible stress is required and must be expressed in megapascal."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:message "The strength criterion status must be true."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Design stress must not exceed permissible stress." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:designStress ?designQuantity ;
          nltl:permissibleStress ?permissibleQuantity .
    ?designQuantity qudt:numericValue ?designValue ;
                    qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
    ?permissibleQuantity qudt:numericValue ?permissibleValue ;
                         qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
    FILTER(?designValue > ?permissibleValue)
}
"""
    ] .
<END_SHACL>
