<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_065_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:nonlinearCalculationMethod ;
                sh:maxCount 0
            ]
        ]
        [
            sh:node gen:I2_065_NonlinearEvaluationShape
        ]
    ) ;
    sh:message "A ship evaluated using a non-linear calculation method requires evidence addressing buckling, plastic and permanent deformation, and the specified structural margins." .

gen:I2_065_NonlinearEvaluationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:nonlinearCalculationMethod ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:bucklingResponse ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Evidence of the buckling response is required for a non-linear evaluation."
    ] ;
    sh:property [
        sh:path nltl:plasticDeformation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Evidence of plastic deformation behavior is required for a non-linear evaluation."
    ] ;
    sh:property [
        sh:path nltl:permanentDeformation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Evidence concerning permanent lateral and out-of-plane deformation is required."
    ] ;
    sh:property [
        sh:path nltl:fractureMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_DimensionlessMarginShape ;
        sh:message "A dimensionless fracture margin is required."
    ] ;
    sh:property [
        sh:path nltl:majorBucklingMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_DimensionlessMarginShape ;
        sh:message "A dimensionless major buckling margin is required."
    ] ;
    sh:property [
        sh:path nltl:yieldingMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_DimensionlessMarginShape ;
        sh:message "A dimensionless yielding margin is required."
    ] .

gen:I2_065_DimensionlessMarginShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS
    ] .
<END_SHACL>
