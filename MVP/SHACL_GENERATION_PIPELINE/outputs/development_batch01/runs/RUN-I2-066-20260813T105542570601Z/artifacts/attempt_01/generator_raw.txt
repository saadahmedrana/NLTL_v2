<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix shape: <urn:nltl:generated-shape:> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:IceStrengthenedAreaWeldingShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengthenedArea ;
        sh:datatype xsd:string ;
        sh:message "Ice-strengthened area values must be strings."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where an ice-strengthened area is identified, every supplied weld type must be the xsd:string value 'double continuous', and at least one weld type must be supplied." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceStrengthenedArea ?area .
    FILTER (
        NOT EXISTS {
            $this nltl:weldType ?weldType .
        }
        ||
        EXISTS {
            $this nltl:weldType ?reportedWeldType .
            FILTER (
                !isLiteral(?reportedWeldType)
                ||
                DATATYPE(?reportedWeldType) != xsd:string
                ||
                STR(?reportedWeldType) != "double continuous"
            )
        }
    )
}
"""
    ] .
<END_SHACL>
