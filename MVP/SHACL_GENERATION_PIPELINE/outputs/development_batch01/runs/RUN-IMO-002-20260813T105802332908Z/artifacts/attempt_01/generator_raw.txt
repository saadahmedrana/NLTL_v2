    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship's design ice condition is required to determine category B applicability."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:message "The ship must declare a Polar Code category."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship that is not category A and is designed for operation in at least thin first-year ice, including possible old-ice inclusions, must be category B." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:designIceCondition ?iceCondition .
    FILTER(
        ?iceCondition =
        "at least thin first-year ice, including possible old-ice inclusions"^^xsd:string
    )
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryA .
    }
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryB .
    }
}
"""
    ] .
