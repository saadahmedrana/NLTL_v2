<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_002_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A design ice condition is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must declare exactly one permitted Polar Code category."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship that is not category A and is designed for at least thin first-year ice must be category B." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:designIceCondition ?iceCondition ;
          nltl:shipCategory ?category .
    FILTER(
        ?iceCondition =
        "at least thin first-year ice, including possible old-ice inclusions"^^xsd:string
    )
    FILTER(?category != nltl:polarShipCategoryA)
    FILTER(?category != nltl:polarShipCategoryB)
}
"""
    ] .
<END_SHACL>
