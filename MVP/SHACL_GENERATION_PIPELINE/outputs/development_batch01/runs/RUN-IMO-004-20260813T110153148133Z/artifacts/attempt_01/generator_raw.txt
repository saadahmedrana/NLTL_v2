<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO004Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Sea ice of not more than one winter growth and thickness from 0.3 m to 2.0 m must be classified as first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:iceOrigin ?origin ;
          nltl:winterGrowthCount ?growth ;
          nltl:iceThickness ?thicknessValue .

    ?thicknessValue rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?thickness ;
                    qudt:unit unit:M .

    FILTER(
        DATATYPE(?origin) = xsd:string &&
        STR(?origin) = "sea ice" &&
        DATATYPE(?growth) = xsd:integer &&
        ?growth <= 1 &&
        DATATYPE(?thickness) = xsd:decimal &&
        ?thickness >= 0.3 &&
        ?thickness <= 2.0
    )

    FILTER NOT EXISTS {
        $this nltl:iceType ?type .
        FILTER(
            DATATYPE(?type) = xsd:string &&
            STR(?type) = "first-year ice"
        )
    }
}
"""
    ] .
<END_SHACL>
