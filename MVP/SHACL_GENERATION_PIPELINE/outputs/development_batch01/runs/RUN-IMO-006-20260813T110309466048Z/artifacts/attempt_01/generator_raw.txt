<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Medium first-year ice must have a decimal thickness from 0.7 m to 1.2 m." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:iceType ?iceType .
    FILTER(STR(?iceType) = "medium first-year ice")

    OPTIONAL {
        $this nltl:iceThickness ?thickness .
        OPTIONAL {
            ?thickness a qudt:QuantityValue .
            BIND(true AS ?hasQuantityValueClass)
        }
        OPTIONAL {
            ?thickness qudt:numericValue ?value .
        }
        OPTIONAL {
            ?thickness qudt:unit unit:M .
            BIND(true AS ?hasMetreUnit)
        }
    }

    FILTER(
        DATATYPE(?iceType) != xsd:string ||
        !BOUND(?thickness) ||
        !BOUND(?hasQuantityValueClass) ||
        !BOUND(?value) ||
        !BOUND(?hasMetreUnit) ||
        (BOUND(?value) && DATATYPE(?value) != xsd:decimal) ||
        (
            BOUND(?value) &&
            DATATYPE(?value) = xsd:decimal &&
            (
                ?value < "0.7"^^xsd:decimal ||
                ?value > "1.2"^^xsd:decimal
            )
        )
    )
}
""" ;
    ] .
<END_SHACL>
