<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:OpenWaterDefinitionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Ice condition is required to determine whether the open-water constraints apply."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open water, sea ice concentration must be less than 1 in unitless measure and ice of land origin must be false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER(?condition = "open water")
    FILTER NOT EXISTS {
        $this nltl:seaIceConcentration ?concentration ;
              nltl:iceOfLandOriginPresent ?landOriginPresent .
        ?concentration qudt:numericValue ?concentrationValue ;
                       qudt:unit unit:UNITLESS .
        FILTER(datatype(?concentrationValue) = xsd:decimal)
        FILTER(?concentrationValue < 1)
        FILTER(datatype(?landOriginPresent) = xsd:boolean)
        FILTER(?landOriginPresent = false)
    }
}
"""
    ] .
<END_SHACL>
