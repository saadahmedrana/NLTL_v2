<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:OpenWaterShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one string-valued ice condition is required."
    ] ;
    sh:property [
        sh:path nltl:seaIceConcentration ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Sea ice concentration and its numeric value and unit must be single-valued."
    ] ;
    sh:property [
        sh:path nltl:iceOfLandOriginPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
        sh:message "Ice-of-land-origin presence must be a single Boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Open water requires a unitless decimal sea ice concentration below 1 and no ice of land origin." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER(?condition = "open water"^^xsd:string)
    FILTER NOT EXISTS {
        $this nltl:seaIceConcentration ?concentration ;
              nltl:iceOfLandOriginPresent ?landOriginPresent .
        ?concentration qudt:numericValue ?concentrationValue ;
                       qudt:unit unit:UNITLESS .
        FILTER(
            datatype(?concentrationValue) = xsd:decimal &&
            ?concentrationValue < 1.0 &&
            datatype(?landOriginPresent) = xsd:boolean &&
            ?landOriginPresent = false
        )
    }
}
"""
    ] .
<END_SHACL>
