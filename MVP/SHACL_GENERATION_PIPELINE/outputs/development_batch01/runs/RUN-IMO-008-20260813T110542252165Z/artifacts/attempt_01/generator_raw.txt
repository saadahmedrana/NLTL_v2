<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ThinFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceType ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Ice type is required to determine whether the thin first-year ice thickness constraint applies."
    ] ;
    sh:sparql [
        sh:message "Thin first-year ice must have a thickness from 0.3 m to 0.7 m inclusive, recorded as a decimal QUDT quantity in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceType ?iceType .
                FILTER(?iceType = "thin first-year ice"^^xsd:string)
                OPTIONAL {
                    $this nltl:iceThickness ?thickness .
                }
                FILTER(
                    !BOUND(?thickness) ||
                    NOT EXISTS {
                        ?thickness a qudt:QuantityValue ;
                            qudt:numericValue ?value ;
                            qudt:unit unit:M .
                        FILTER(
                            DATATYPE(?value) = xsd:decimal &&
                            ?value >= "0.3"^^xsd:decimal &&
                            ?value <= "0.7"^^xsd:decimal
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
