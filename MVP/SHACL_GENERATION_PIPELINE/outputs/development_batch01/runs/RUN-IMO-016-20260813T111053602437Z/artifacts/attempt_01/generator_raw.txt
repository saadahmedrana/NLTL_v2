<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:PolarShipCertificateLanguageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The certificate language must be recorded."
    ] ;
    sh:property [
        sh:path nltl:approvedTranslationPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A certificate in a language other than English, French, or Spanish must include an approved translation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER(?language NOT IN ("English", "French", "Spanish"))
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent ?translation .
                    FILTER(?translation = "true"^^xsd:boolean)
                }
            }
        """
    ] .
<END_SHACL>
