<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:RequiredAccessIceSnowMeansShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:requiredAccess ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceOrSnowRemovalOrPreventionMeans ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship recording a required access must also record an ice or snow removal or prevention means." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:requiredAccess ?access .
    FILTER NOT EXISTS {
        $this nltl:iceOrSnowRemovalOrPreventionMeans ?means .
        FILTER(datatype(?means) = xsd:string)
    }
}
"""
    ] .
<END_SHACL>
