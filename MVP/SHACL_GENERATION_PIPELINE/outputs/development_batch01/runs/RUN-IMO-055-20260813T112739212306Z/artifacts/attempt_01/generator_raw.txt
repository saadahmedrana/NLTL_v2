<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ValveIceProtectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Valve exposure status is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Valve type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed isolating, pressure, or vacuum valve must be protected from ice accretion and continuously accessible." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:valveExposed ?exposed ;
          nltl:valveType ?type .
    FILTER(?exposed = true && ?type IN ("isolating", "pressure", "vacuum"))
    FILTER NOT EXISTS {
        $this nltl:iceAccretionProtectionPresent ?protected ;
              nltl:continuouslyAccessible ?accessible .
        FILTER(?protected = true && ?accessible = true)
    }
}
"""
    ] .
<END_SHACL>
