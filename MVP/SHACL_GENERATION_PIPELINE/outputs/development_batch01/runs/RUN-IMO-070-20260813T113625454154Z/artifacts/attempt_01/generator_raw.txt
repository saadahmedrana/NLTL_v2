<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PassengerSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipType ;
                sh:minCount 1 ;
                sh:datatype xsd:string ;
                sh:pattern "^(?!passenger_ship$).+"
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipType ;
                sh:minCount 1 ;
                sh:datatype xsd:string ;
                sh:hasValue "passenger_ship"
            ] ;
            sh:property [
                sh:path nltl:personOnBoard ;
                sh:minCount 1 ;
                sh:datatype xsd:integer ;
                sh:minInclusive 0
            ] ;
            sh:property [
                sh:path nltl:assignedImmersionSuit ;
                sh:datatype xsd:string
            ] ;
            sh:property [
                sh:path nltl:assignedThermalProtectiveAid ;
                sh:datatype xsd:string
            ] ;
            sh:property [
                sh:path nltl:sizeCompatibilityStatus ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "A passenger ship must record at least one distinct immersion-suit or thermal-protective-aid assignment for each person on board." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipType ?shipType ;
                              nltl:personOnBoard ?persons .
                        FILTER(
                            ?shipType = "passenger_ship"^^xsd:string &&
                            datatype(?persons) = xsd:integer &&
                            ?persons >= 0
                        )
                        OPTIONAL {
                            $this (nltl:assignedImmersionSuit|nltl:assignedThermalProtectiveAid) ?assignment .
                        }
                    }
                    GROUP BY $this ?persons
                    HAVING(COUNT(DISTINCT ?assignment) < ?persons)
                """
            ]
        ]
    ) .
<END_SHACL>
