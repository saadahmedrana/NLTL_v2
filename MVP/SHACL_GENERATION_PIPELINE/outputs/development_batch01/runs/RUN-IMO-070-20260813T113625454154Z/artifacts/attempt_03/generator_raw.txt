<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PassengerSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:passengerShip ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:maxCount 1 ;
        sh:message "A passenger ship must not have more than one ship type value."
    ] ;
    sh:property [
        sh:path nltl:personOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A passenger ship must have exactly one persons-on-board count."
    ] ;
    sh:property [
        sh:path nltl:assignedImmersionSuit ;
        sh:datatype xsd:string ;
        sh:disjoint nltl:assignedThermalProtectiveAid ;
        sh:message "An assignment identifier must not identify both an immersion suit assignment and a thermal protective aid assignment."
    ] ;
    sh:property [
        sh:path nltl:assignedThermalProtectiveAid ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( true ) ;
        sh:message "A passenger ship must have exactly one true size compatibility status."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The distinct union of immersion suit and thermal protective aid assignment identifiers must equal the persons-on-board count." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:personOnBoard ?persons .
    {
        SELECT ?ship (COUNT(DISTINCT ?assignment) AS ?assignedCount)
        WHERE {
            ?ship nltl:personOnBoard ?recordedPersons .
            OPTIONAL {
                {
                    ?ship nltl:assignedImmersionSuit ?assignment .
                }
                UNION
                {
                    ?ship nltl:assignedThermalProtectiveAid ?assignment .
                }
            }
        }
        GROUP BY ?ship
    }
    FILTER (?ship = $this)
    FILTER (?assignedCount != ?persons)
}
"""
    ] .
<END_SHACL>
