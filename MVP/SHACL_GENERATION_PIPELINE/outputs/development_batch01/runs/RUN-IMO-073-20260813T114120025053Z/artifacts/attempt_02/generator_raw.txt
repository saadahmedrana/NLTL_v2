    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:lifeboatType ;
        sh:datatype xsd:string ;
        sh:in ( "partially enclosed" "totally enclosed" ) ;
        sh:minCount 1 ;
        sh:message "Lifeboat type must be partially enclosed or totally enclosed."
    ] .
