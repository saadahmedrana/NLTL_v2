<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SurvivalCraftCapacityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:AdditionalEquipmentNotCarriedShape
        gen:AdditionalEquipmentCarriedShape
    ) .

gen:AdditionalEquipmentNotCarriedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( false )
    ] .

gen:AdditionalEquipmentCarriedShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:survivalCraftAvailableCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Survival-craft available capacity is required when additional equipment is carried."
    ] ;
    sh:property [
        sh:path nltl:personsCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Persons capacity requirement is required for the survival-craft capacity calculation."
    ] ;
    sh:property [
        sh:path nltl:additionalEquipmentCapacityRequirement ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Additional equipment capacity requirement is required for the survival-craft capacity calculation."
    ] ;
    sh:property [
        sh:path nltl:launchingApplianceCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Launching-appliance capacity in kilograms is required when additional equipment is carried."
    ] ;
    sh:property [
        sh:path nltl:loadedSurvivalCraftRequirement ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1
            ]
        ] ;
        sh:message "Loaded survival-craft requirement in kilograms is required when additional equipment is carried."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Survival-craft capacity must cover persons plus additional equipment, and launching-appliance capacity must cover the loaded survival-craft requirement." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:survivalCraftAvailableCapacity ?available ;
          nltl:personsCapacityRequirement ?persons ;
          nltl:additionalEquipmentCapacityRequirement ?additional ;
          nltl:launchingApplianceCapacity ?launchingQuantity ;
          nltl:loadedSurvivalCraftRequirement ?loadedQuantity .
    ?launchingQuantity qudt:numericValue ?launchingCapacity .
    ?loadedQuantity qudt:numericValue ?loadedRequirement .
    FILTER (
        xsd:decimal(?available) <
            (xsd:decimal(?persons) + xsd:decimal(?additional))
        ||
        xsd:decimal(?launchingCapacity) <
            xsd:decimal(?loadedRequirement)
    )
}
"""
    ] .
<END_SHACL>
