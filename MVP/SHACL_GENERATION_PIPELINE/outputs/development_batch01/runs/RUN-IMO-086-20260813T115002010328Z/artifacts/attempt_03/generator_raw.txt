<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO086HeadingMeansShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:nonMagneticHeadingMeansCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 2 ;
        sh:message "The non-magnetic heading means count must be at least two."
    ] ;
    sh:property [
        sh:path nltl:headingMeansIndependent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The non-magnetic heading means must be independent."
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:equals nltl:connectedToEmergencyPower ;
        sh:message "Every heading means connected to either power source must be connected to both main and emergency power."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The declared heading means count must equal the number of distinct means connected to both main and emergency power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:nonMagneticHeadingMeansCount ?declaredCount .
                {
                    SELECT $this (COUNT(DISTINCT ?means) AS ?actualCount)
                    WHERE {
                        $this nltl:connectedToMainPower ?means .
                        $this nltl:connectedToEmergencyPower ?means .
                    }
                    GROUP BY $this
                }
                FILTER (?actualCount != xsd:integer(?declaredCount))
            }
        """
    ] .
<END_SHACL>
