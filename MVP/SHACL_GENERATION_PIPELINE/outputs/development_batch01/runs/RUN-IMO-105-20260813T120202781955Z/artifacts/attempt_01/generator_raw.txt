<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ArcticOilDischargeProhibitionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Operating area is required to determine whether the Arctic waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, every recorded oil or oily mixture discharge-to-sea value must be false, and a false discharge record is required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:operatingArea ?area .
    FILTER(?area = "Arctic_waters"^^xsd:string)
    OPTIONAL {
        $this nltl:oilOrOilyMixtureDischargeToSea ?discharge .
    }
    FILTER(
        !BOUND(?discharge) ||
        !isLiteral(?discharge) ||
        DATATYPE(?discharge) != xsd:boolean ||
        ?discharge != "false"^^xsd:boolean
    )
}
"""
    ] .
<END_SHACL>
