<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO106Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialType ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Discharged material classification is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean or segregated ballast, paragraph 1.1.1 applicability must be false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:dischargedMaterialType ?material .
    FILTER(?material IN ("clean_ballast", "segregated_ballast"))
    FILTER(
        !EXISTS {
            $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?acceptable .
            FILTER(
                isLiteral(?acceptable) &&
                datatype(?acceptable) = xsd:boolean &&
                ?acceptable = false
            )
        }
        ||
        EXISTS {
            $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?reported .
            FILTER(
                !isLiteral(?reported) ||
                datatype(?reported) != xsd:boolean ||
                ?reported != false
            )
        }
    )
}
"""
    ] .
<END_SHACL>
