<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO112Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, the NLS or NLS-mixture discharge-to-sea value must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER(?area = "Arctic_waters"^^xsd:string)
                FILTER(
                    NOT EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?allowed .
                        FILTER(
                            DATATYPE(?allowed) = xsd:boolean &&
                            ?allowed = false
                        )
                    } ||
                    EXISTS {
                        $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?reported .
                        FILTER(
                            !(
                                DATATYPE(?reported) = xsd:boolean &&
                                ?reported = false
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
