<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO116Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Sewage discharge type is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For sewage that is not comminuted or disinfected, MARPOL Annex IV regulation 11.1.1 compliance, a distance greater than 12 nautical miles, and nearby ice-concentration data are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:sewageDischargeType ?dischargeType .
    FILTER(STR(?dischargeType) = "not_comminuted_or_disinfected")
    FILTER NOT EXISTS {
        $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus ?status ;
              nltl:distanceToNearestIceShelfOrFastIce ?distanceNode ;
              nltl:nearbyIceConcentration ?nearbyIceNode .
        ?distanceNode a qudt:QuantityValue ;
                      qudt:numericValue ?distance ;
                      qudt:unit unit:MI_N .
        ?nearbyIceNode a qudt:QuantityValue ;
                       qudt:numericValue ?nearbyIce ;
                       qudt:unit unit:UNITLESS .
        FILTER(
            datatype(?status) = xsd:boolean &&
            ?status = "true"^^xsd:boolean &&
            datatype(?distance) = xsd:decimal &&
            ?distance > "12"^^xsd:decimal &&
            datatype(?nearbyIce) = xsd:decimal
        )
    }
}
"""
    ] .
<END_SHACL>
