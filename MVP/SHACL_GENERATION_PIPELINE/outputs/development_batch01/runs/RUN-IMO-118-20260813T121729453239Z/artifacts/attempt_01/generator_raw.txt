<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO-118-Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionDate ;
                sh:maxExclusive "2017-01-01"^^xsd:date
            ]
        ]
        [
            sh:class nltl:passengerShip ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable ship, sewage discharge to sea must be false unless paragraph 4.2.1.3 compliance is true." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionDate ?date .
    FILTER(
        YEAR(?date) > 2017 ||
        (YEAR(?date) = 2017 && MONTH(?date) > 1) ||
        (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
    )
    FILTER(
        EXISTS {
            $this a nltl:passengerShip .
        } ||
        EXISTS {
            $this nltl:shipCategory ?category .
            FILTER(?category IN (
                nltl:polarShipCategoryA,
                nltl:polarShipCategoryB
            ))
        }
    )
    FILTER NOT EXISTS {
        $this nltl:sewageDischargeToSea "false"^^xsd:boolean .
    }
    FILTER NOT EXISTS {
        $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus "true"^^xsd:boolean .
    }
}
"""
    ] .
<END_SHACL>
