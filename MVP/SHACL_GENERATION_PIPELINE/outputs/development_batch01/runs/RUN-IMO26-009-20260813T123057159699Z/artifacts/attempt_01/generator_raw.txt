<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceAccumulationPreventionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice-accretion likelihood must be recorded to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:requiredNavigationOrCommunicationAntenna ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationPreventionMeansPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice accretion is likely and a required navigation or communication antenna is recorded, ice-accumulation prevention means must be present." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceAccretionLikely true ;
          nltl:requiredNavigationOrCommunicationAntenna ?antenna .
    FILTER NOT EXISTS {
        $this nltl:iceAccumulationPreventionMeansPresent true .
    }
}
"""
    ] .
<END_SHACL>
