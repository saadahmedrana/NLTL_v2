<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PolarCodeChapter11ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeChapter11Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:solasChapterICertified ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Code chapter 11 may apply only when the ship is certified under SOLAS chapter I." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:polarCodeChapter11Applicable ?applicable .
    FILTER(?applicable = true)
    FILTER NOT EXISTS {
        $this nltl:solasChapterICertified ?certified .
        FILTER(?certified = true)
    }
}
"""
    ] .
<END_SHACL>
