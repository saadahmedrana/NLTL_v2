<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:Chapter11Dash1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipOperatesInPolarWaters ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        gen:PolarOperationShape
    ) .

gen:PolarOperationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInPolarWaters ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A polar-operating ship must provide exactly one ship type."
    ] ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A polar-operating ship must provide its Chapter 11-1 applicability state."
    ] ;
    sh:or (
        gen:FishingVesselApplicabilityShape
        gen:PleasureYachtApplicabilityShape
        gen:CargoShipApplicabilityShape
        gen:OtherShipTypeApplicabilityShape
    ) .

gen:FishingVesselApplicabilityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipType ;
        sh:hasValue "fishing_vessel"
    ] ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A polar fishing vessel must provide length overall in metres."
    ] ;
    sh:sparql [
        sh:message "Chapter 11-1 applies to a polar fishing vessel exactly when its length overall is at least 24 metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipType ?type ;
          nltl:lengthOverall ?quantity ;
          nltl:polarCodeChapter11Dash1Applicable ?actual .
    ?quantity qudt:numericValue ?length ;
              qudt:unit unit:M .
    FILTER(
        ?type = "fishing_vessel" &&
        (
            (?length >= "24"^^xsd:decimal && ?actual != true) ||
            (?length < "24"^^xsd:decimal && ?actual != false)
        )
    )
}
"""
    ] .

gen:PleasureYachtApplicabilityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipType ;
        sh:hasValue "pleasure_yacht"
    ] ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A polar pleasure yacht must provide gross tonnage as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A polar pleasure yacht must state whether it is engaged in trade."
    ] ;
    sh:sparql [
        sh:message "Chapter 11-1 applies to a polar pleasure yacht exactly when gross tonnage is at least 300 and it is not engaged in trade." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipType ?type ;
          nltl:grossTonnage ?quantity ;
          nltl:engagedInTrade ?trade ;
          nltl:polarCodeChapter11Dash1Applicable ?actual .
    ?quantity qudt:numericValue ?tonnage ;
              qudt:unit unit:UNITLESS .
    FILTER(
        ?type = "pleasure_yacht" &&
        (
            (?tonnage >= "300"^^xsd:decimal && ?trade = false && ?actual != true) ||
            ((?tonnage < "300"^^xsd:decimal || ?trade = true) && ?actual != false)
        )
    )
}
"""
    ] .

gen:CargoShipApplicabilityShape

        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A polar cargo ship must provide gross tonnage as a unitless quantity."
    ] ;
    sh:sparql [
        sh:message "Chapter 11-1 applies to a polar cargo ship exactly when gross tonnage is at least 300 and below 500." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>

PREFIX unit: <http://qudt.org/vocab/unit/>

    ] .

gen:OtherShipTypeApplicabilityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeChapter11Dash1Applicable ;
        sh:hasValue false
    ] ;
    sh:sparql [
        sh:message "The other-ship-type branch cannot be used for a fishing vessel, pleasure yacht, or cargo ship." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipType ?type .
    FILTER(?type IN ("fishing_vessel", "pleasure_yacht", "cargo_ship"))
}
"""
    ] .
<END_SHACL>
