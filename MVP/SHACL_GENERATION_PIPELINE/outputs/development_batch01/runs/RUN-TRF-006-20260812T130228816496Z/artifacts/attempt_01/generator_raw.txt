<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF006ApplicationOf1985RulesShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:message "A stated construction stage date must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a construction stage from 1 November 1986 through 31 August 2003, the applicable ice class regulation edition must be the 1985 Ice Class Rules." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionStageDate ?date .
    FILTER(
        ?date >= "1986-11-01"^^xsd:date &&
        ?date < "2003-09-01"^^xsd:date
    )
    FILTER(
        NOT EXISTS {
            $this nltl:applicableIceClassRegulationEdition
                nltl:iceClassRuleEdition1985 .
        } ||
        EXISTS {
            $this nltl:applicableIceClassRegulationEdition ?edition .
            FILTER(?edition != nltl:iceClassRuleEdition1985)
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship in the 1985-rules construction period, a stated engine-output regulation edition must be either the 1985 Ice Class Rules or the 2008 Ice Class Regulations." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionStageDate ?date ;
          nltl:engineOutputRegulationEdition ?engineEdition .
    FILTER(
        ?date >= "1986-11-01"^^xsd:date &&
        ?date < "2003-09-01"^^xsd:date
    )
    FILTER(
        ?engineEdition NOT IN (
            nltl:iceClassRuleEdition1985,
            nltl:iceClassRegulationEdition2008
        )
    )
}
"""
    ] .
<END_SHACL>
