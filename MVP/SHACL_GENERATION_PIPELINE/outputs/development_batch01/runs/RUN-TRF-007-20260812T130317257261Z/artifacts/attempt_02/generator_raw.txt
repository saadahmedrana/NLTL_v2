<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF007ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship requires a valid delivery date to determine its compliance deadline." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate .
    VALUES ?iceClass {
        nltl:iceClassIaSuper
        nltl:iceClassIa
    }
    FILTER(
        DATATYPE(?constructionDate) = xsd:date &&
        ?constructionDate < "2003-09-01"^^xsd:date
    )
    FILTER NOT EXISTS {
        $this nltl:deliveryDate ?deliveryDate .
        FILTER(DATATYPE(?deliveryDate) = xsd:date)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship with a valid delivery date requires a valid assessment date to determine whether its compliance deadline has been reached." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate .
    VALUES ?iceClass {
        nltl:iceClassIaSuper
        nltl:iceClassIa
    }
    FILTER(
        DATATYPE(?constructionDate) = xsd:date &&
        ?constructionDate < "2003-09-01"^^xsd:date &&
        DATATYPE(?deliveryDate) = xsd:date
    )
    FILTER NOT EXISTS {
        $this nltl:assessmentDate ?assessmentDate .
        FILTER(DATATYPE(?assessmentDate) = xsd:date)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The applicable ship has reached its compliance deadline without a permitted 2017 Ice Class Regulations compliance method." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate ;
          nltl:assessmentDate ?assessmentDate .
    VALUES ?iceClass {
        nltl:iceClassIaSuper
        nltl:iceClassIa
    }
    FILTER(
        DATATYPE(?constructionDate) = xsd:date &&
        ?constructionDate < "2003-09-01"^^xsd:date &&
        DATATYPE(?deliveryDate) = xsd:date &&
        DATATYPE(?assessmentDate) = xsd:date
    )
    BIND(
        xsd:date(
            CONCAT(
                STR(YEAR(?deliveryDate) + 20),
                "-01-01"
            )
        )
        AS ?complianceDeadline
    )
    FILTER(!(?assessmentDate < ?complianceDeadline))
    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod ?method .
        VALUES ?method {
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        }
    }
}
"""
    ] .
<END_SHACL>
