<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF007Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IA Super or IA require a valid construction-stage date to determine applicability." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa
        )
    )
    FILTER NOT EXISTS {
        $this nltl:constructionStageDate ?constructionDate .
        FILTER (DATATYPE(?constructionDate) = xsd:date)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A valid delivery date and assessment date are required for an applicable ship to determine whether the compliance deadline has elapsed." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa
        )
    )
    FILTER EXISTS {
        $this nltl:constructionStageDate ?constructionDate .
        FILTER (DATATYPE(?constructionDate) = xsd:date)
        FILTER (
            YEAR(?constructionDate) < 2003 ||
            (
                YEAR(?constructionDate) = 2003 &&
                MONTH(?constructionDate) < 9
            )
        )
    }
    FILTER (
        NOT EXISTS {
            $this nltl:deliveryDate ?deliveryDate .
            FILTER (DATATYPE(?deliveryDate) = xsd:date)
        }
        ||
        NOT EXISTS {
            $this nltl:assessmentDate ?assessmentDate .
            FILTER (DATATYPE(?assessmentDate) = xsd:date)
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "After the applicable deadline, the ship must use the 2017 section 3.2.2 or section 3.2.4 engine-output compliance method." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa
        )
    )
    FILTER EXISTS {
        $this nltl:constructionStageDate ?constructionDate .
        FILTER (DATATYPE(?constructionDate) = xsd:date)
        FILTER (
            YEAR(?constructionDate) < 2003 ||
            (
                YEAR(?constructionDate) = 2003 &&
                MONTH(?constructionDate) < 9
            )
        )
    }
    FILTER EXISTS {
        $this nltl:deliveryDate ?deliveryDate ;
              nltl:assessmentDate ?assessmentDate .
        FILTER (DATATYPE(?deliveryDate) = xsd:date)
        FILTER (DATATYPE(?assessmentDate) = xsd:date)
        FILTER (
            YEAR(?assessmentDate) >= YEAR(?deliveryDate) + 20
        )
    }
    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod ?method .
        FILTER (
            ?method IN (
                nltl:traficom2017Section3Point2Point2Method,
                nltl:traficom2017Section3Point2Point4Method
            )
        )
    }
}
"""
    ] .
<END_SHACL>
