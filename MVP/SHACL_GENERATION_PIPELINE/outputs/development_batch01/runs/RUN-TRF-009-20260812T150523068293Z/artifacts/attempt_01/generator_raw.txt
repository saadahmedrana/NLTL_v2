<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF009Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship must provide a valid delivery date so that the compliance deadline can be calculated." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate .
    FILTER(
        ?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa)
        && datatype(?constructionDate) = xsd:date
        && ?constructionDate < "2003-09-01"^^xsd:date
    )
    FILTER NOT EXISTS {
        $this nltl:deliveryDate ?deliveryDate .
        FILTER(datatype(?deliveryDate) = xsd:date)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship must use the compliance method from section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate .
    FILTER(
        ?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa)
        && datatype(?constructionDate) = xsd:date
        && ?constructionDate < "2003-09-01"^^xsd:date
    )
    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod ?method .
        FILTER(
            ?method IN (
                nltl:traficom2017Section3Point2Point2Method,
                nltl:traficom2017Section3Point2Point4Method
            )
        )
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Compliance must be assessed no later than 1 January of the year twenty years after the delivery year." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate .
    FILTER(
        ?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa)
        && datatype(?constructionDate) = xsd:date
        && ?constructionDate < "2003-09-01"^^xsd:date
        && datatype(?deliveryDate) = xsd:date
    )
    FILTER NOT EXISTS {
        $this nltl:assessmentDate ?assessmentDate .
        FILTER(
            datatype(?assessmentDate) = xsd:date
            && ?assessmentDate <= xsd:date(
                CONCAT(STR(YEAR(?deliveryDate) + 20), "-01-01")
            )
        )
    }
}
"""
    ] .
<END_SHACL>
