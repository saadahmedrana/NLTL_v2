<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UpperIceWaterlineEnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "An upper ice waterline with profile-point data is required."
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "Intended ice operating waterline data is required to verify the upper ice waterline envelope."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented longitudinal position of an intended ice operating waterline must have an upper ice waterline point at the highest represented vertical coordinate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
    ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedXValue .
    ?intendedXValue qudt:numericValue ?x ;
        qudt:unit unit:M .

    FILTER NOT EXISTS {
        $this nltl:hasUpperIceWaterline ?upperWaterline .
        ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
        ?upperPoint nltl:longitudinalPosition ?upperXValue ;
            nltl:verticalCoordinate ?upperZValue .
        ?upperXValue qudt:numericValue ?upperX ;
            qudt:unit unit:M .
        ?upperZValue qudt:numericValue ?upperZ ;
            qudt:unit unit:M .
        FILTER(?upperX = ?x)

        FILTER EXISTS {
            $this nltl:hasIntendedIceOperatingWaterline ?matchingWaterline .
            ?matchingWaterline nltl:hasWaterlineProfilePoint ?matchingPoint .
            ?matchingPoint nltl:longitudinalPosition ?matchingXValue ;
                nltl:verticalCoordinate ?matchingZValue .
            ?matchingXValue qudt:numericValue ?matchingX ;
                qudt:unit unit:M .
            ?matchingZValue qudt:numericValue ?matchingZ ;
                qudt:unit unit:M .
            FILTER(?matchingX = ?x && ?matchingZ = ?upperZ)
        }

        FILTER NOT EXISTS {
            $this nltl:hasIntendedIceOperatingWaterline ?higherWaterline .
            ?higherWaterline nltl:hasWaterlineProfilePoint ?higherPoint .
            ?higherPoint nltl:longitudinalPosition ?higherXValue ;
                nltl:verticalCoordinate ?higherZValue .
            ?higherXValue qudt:numericValue ?higherX ;
                qudt:unit unit:M .
            ?higherZValue qudt:numericValue ?higherZ ;
                qudt:unit unit:M .
            FILTER(?higherX = ?x && ?higherZ > ?upperZ)
        }
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each upper ice waterline profile point must coincide with the highest represented intended operating waterline point at that longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasUpperIceWaterline ?upperWaterline .
    ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
    ?upperPoint nltl:longitudinalPosition ?upperXValue ;
        nltl:verticalCoordinate ?upperZValue .
    ?upperXValue qudt:numericValue ?upperX ;
        qudt:unit unit:M .
    ?upperZValue qudt:numericValue ?upperZ ;
        qudt:unit unit:M .

    FILTER NOT EXISTS {
        $this nltl:hasIntendedIceOperatingWaterline ?matchingWaterline .
        ?matchingWaterline nltl:hasWaterlineProfilePoint ?matchingPoint .
        ?matchingPoint nltl:longitudinalPosition ?matchingXValue ;
            nltl:verticalCoordinate ?matchingZValue .
        ?matchingXValue qudt:numericValue ?matchingX ;
            qudt:unit unit:M .
        ?matchingZValue qudt:numericValue ?matchingZ ;
            qudt:unit unit:M .
        FILTER(?matchingX = ?upperX && ?matchingZ = ?upperZ)

        FILTER NOT EXISTS {
            $this nltl:hasIntendedIceOperatingWaterline ?higherWaterline .
            ?higherWaterline nltl:hasWaterlineProfilePoint ?higherPoint .
            ?higherPoint nltl:longitudinalPosition ?higherXValue ;
                nltl:verticalCoordinate ?higherZValue .
            ?higherXValue qudt:numericValue ?higherX ;
                qudt:unit unit:M .
            ?higherZValue qudt:numericValue ?higherZ ;
                qudt:unit unit:M .
            FILTER(?higherX = ?upperX && ?higherZ > ?upperZ)
        }
    }
}
"""
    ] .

gen:IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:WaterlineProfilePointShape ;
        sh:minCount 1 ;
        sh:message "A waterline must have represented profile points."
    ] .

gen:WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:node gen:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "A waterline profile point requires a longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:node gen:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:message "A waterline profile point requires a vertical coordinate in metres."
    ] .

gen:LengthQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:in (unit:M) ;
        sh:minCount 1
    ] .
<END_SHACL>
