<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UpperIceWaterlineEnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one upper ice waterline with a usable profile."
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "At least one intended ice operating waterline with a usable profile is required to check the upper envelope."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each intended ice operating waterline point must have an upper-waterline point at the same longitudinal position and at least the same vertical coordinate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasUpperIceWaterline ?upper ;
          nltl:hasIntendedIceOperatingWaterline ?intended .
    ?intended nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedLongitudinal ;
                   nltl:verticalCoordinate ?intendedVertical .
    ?intendedLongitudinal qudt:numericValue ?intendedX .
    ?intendedVertical qudt:numericValue ?intendedZ .
    FILTER NOT EXISTS {
        ?upper nltl:hasWaterlineProfilePoint ?upperPoint .
        ?upperPoint nltl:longitudinalPosition ?upperLongitudinal ;
                    nltl:verticalCoordinate ?upperVertical .
        ?upperLongitudinal qudt:numericValue ?upperX .
        ?upperVertical qudt:numericValue ?upperZ .
        FILTER(
            xsd:decimal(?upperX) = xsd:decimal(?intendedX) &&
            xsd:decimal(?upperZ) >= xsd:decimal(?intendedZ)
        )
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each upper ice waterline point must coincide with a highest intended-operating-waterline point." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasUpperIceWaterline ?upper .
    ?upper nltl:hasWaterlineProfilePoint ?upperPoint .
    ?upperPoint nltl:longitudinalPosition ?upperLongitudinal ;
                nltl:verticalCoordinate ?upperVertical .
    ?upperLongitudinal qudt:numericValue ?upperX .
    ?upperVertical qudt:numericValue ?upperZ .
    FILTER NOT EXISTS {
        $this nltl:hasIntendedIceOperatingWaterline ?intended .
        ?intended nltl:hasWaterlineProfilePoint ?intendedPoint .
        ?intendedPoint nltl:longitudinalPosition ?intendedLongitudinal ;
                       nltl:verticalCoordinate ?intendedVertical .
        ?intendedLongitudinal qudt:numericValue ?intendedX .
        ?intendedVertical qudt:numericValue ?intendedZ .
        FILTER(
            xsd:decimal(?intendedX) = xsd:decimal(?upperX) &&
            xsd:decimal(?intendedZ) = xsd:decimal(?upperZ)
        )
    }
}
"""
    ] .

gen:IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:WaterlineProfilePointShape ;
        sh:minCount 1 ;
        sh:message "A waterline must have at least one profile point with usable coordinates."
    ] .

gen:WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:node gen:MetreQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A waterline profile point must have exactly one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:node gen:MetreQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A waterline profile point must have exactly one vertical coordinate in metres."
    ] .

gen:MetreQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The coordinate quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:M ;
        sh:maxCount 1 ;
        sh:message "The coordinate quantity must use metre as its single unit."
    ] .
<END_SHACL>
