<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:UpperIceWaterlineEnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gs:WaterlineProfileShape ;
        sh:message "At least one intended ice operating waterline with a valid profile is required."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gs:WaterlineProfileShape ;
        sh:message "Exactly one upper ice waterline with a valid profile is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each upper ice waterline point must coincide, within tolerance, with an intended operating waterline point at the same longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasUpperIceWaterline ?upperWaterline .
    ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
    ?upperPoint nltl:longitudinalPosition ?upperXQuantity ;
                nltl:verticalCoordinate ?upperZQuantity .
    ?upperXQuantity qudt:numericValue ?upperX .
    ?upperZQuantity qudt:numericValue ?upperZ .

    FILTER NOT EXISTS {
        $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
        ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
        ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                       nltl:verticalCoordinate ?intendedZQuantity .
        ?intendedXQuantity qudt:numericValue ?intendedX .
        ?intendedZQuantity qudt:numericValue ?intendedZ .

        BIND(IF(ABS(?upperX) > ABS(?intendedX), ABS(?upperX), ABS(?intendedX)) AS ?xScale)
        BIND(IF((0.000001 * ?xScale) > 0.000000001,
                0.000001 * ?xScale,
                0.000000001) AS ?xTolerance)
        BIND(IF(ABS(?upperZ) > ABS(?intendedZ), ABS(?upperZ), ABS(?intendedZ)) AS ?zScale)
        BIND(IF((0.000001 * ?zScale) > 0.000000001,
                0.000001 * ?zScale,
                0.000000001) AS ?zTolerance)

        FILTER(ABS(?upperX - ?intendedX) <= ?xTolerance)
        FILTER(ABS(?upperZ - ?intendedZ) <= ?zTolerance)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An intended operating waterline point is higher than the upper ice waterline at the corresponding longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasUpperIceWaterline ?upperWaterline ;
          nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .

    ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
    ?upperPoint nltl:longitudinalPosition ?upperXQuantity ;
                nltl:verticalCoordinate ?upperZQuantity .
    ?upperXQuantity qudt:numericValue ?upperX .
    ?upperZQuantity qudt:numericValue ?upperZ .

    ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                   nltl:verticalCoordinate ?intendedZQuantity .
    ?intendedXQuantity qudt:numericValue ?intendedX .
    ?intendedZQuantity qudt:numericValue ?intendedZ .

    BIND(IF(ABS(?upperX) > ABS(?intendedX), ABS(?upperX), ABS(?intendedX)) AS ?xScale)
    BIND(IF((0.000001 * ?xScale) > 0.000000001,
            0.000001 * ?xScale,
            0.000000001) AS ?xTolerance)
    BIND(IF(ABS(?upperZ) > ABS(?intendedZ), ABS(?upperZ), ABS(?intendedZ)) AS ?zScale)
    BIND(IF((0.000001 * ?zScale) > 0.000000001,
            0.000001 * ?zScale,
            0.000000001) AS ?zTolerance)

    FILTER(ABS(?upperX - ?intendedX) <= ?xTolerance)
    FILTER(?intendedZ > (?upperZ + ?zTolerance))
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every intended operating waterline longitudinal position must be represented by an upper ice waterline point." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
    ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity .
    ?intendedXQuantity qudt:numericValue ?intendedX .

    FILTER NOT EXISTS {
        $this nltl:hasUpperIceWaterline ?upperWaterline .
        ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
        ?upperPoint nltl:longitudinalPosition ?upperXQuantity .
        ?upperXQuantity qudt:numericValue ?upperX .

        BIND(IF(ABS(?upperX) > ABS(?intendedX), ABS(?upperX), ABS(?intendedX)) AS ?xScale)
        BIND(IF((0.000001 * ?xScale) > 0.000000001,
                0.000001 * ?xScale,
                0.000000001) AS ?xTolerance)

        FILTER(ABS(?upperX - ?intendedX) <= ?xTolerance)
    }
}
"""
    ] .

gs:WaterlineProfileShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gs:WaterlineProfilePointShape ;
        sh:message "A waterline must have at least one valid profile point."
    ] .

gs:WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gs:MetreDecimalQuantityShape ;
        sh:message "Each profile point must have exactly one decimal longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gs:MetreDecimalQuantityShape ;
        sh:message "Each profile point must have exactly one decimal vertical coordinate in metres."
    ] .

gs:MetreDecimalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
