<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UpperIceWaterlineEnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify exactly one upper ice waterline."
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:node gen:IceWaterlineShape ;
        sh:minCount 1 ;
        sh:message "At least one intended ice operating waterline is required to establish the upper-waterline envelope."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline must contain, at every intended longitudinal position, exactly the highest intended vertical coordinate, and every upper point must correspond to such a highest intended point." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    FILTER (
        EXISTS {
            $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
            ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
            ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity .
            ?intendedXQuantity qudt:numericValue ?intendedX .

            FILTER NOT EXISTS {
                $this nltl:hasUpperIceWaterline ?upperWaterline .
                ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                ?upperPoint nltl:longitudinalPosition ?upperXQuantity ;
                            nltl:verticalCoordinate ?upperYQuantity .
                ?upperXQuantity qudt:numericValue ?upperX .
                ?upperYQuantity qudt:numericValue ?upperY .

                $this nltl:hasIntendedIceOperatingWaterline ?matchingWaterline .
                ?matchingWaterline nltl:hasWaterlineProfilePoint ?matchingPoint .
                ?matchingPoint nltl:longitudinalPosition ?matchingXQuantity ;
                               nltl:verticalCoordinate ?matchingYQuantity .
                ?matchingXQuantity qudt:numericValue ?matchingX .
                ?matchingYQuantity qudt:numericValue ?matchingY .

                FILTER (
                    ?upperX = ?intendedX &&
                    ?matchingX = ?upperX &&
                    ?matchingY = ?upperY
                )

                FILTER NOT EXISTS {
                    $this nltl:hasIntendedIceOperatingWaterline ?higherWaterline .
                    ?higherWaterline nltl:hasWaterlineProfilePoint ?higherPoint .
                    ?higherPoint nltl:longitudinalPosition ?higherXQuantity ;
                                 nltl:verticalCoordinate ?higherYQuantity .
                    ?higherXQuantity qudt:numericValue ?higherX .
                    ?higherYQuantity qudt:numericValue ?higherY .
                    FILTER (?higherX = ?upperX && ?higherY > ?upperY)
                }
            }
        }
        ||
        EXISTS {
            $this nltl:hasUpperIceWaterline ?checkedUpperWaterline .
            ?checkedUpperWaterline nltl:hasWaterlineProfilePoint ?checkedUpperPoint .
            ?checkedUpperPoint nltl:longitudinalPosition ?checkedUpperXQuantity ;
                               nltl:verticalCoordinate ?checkedUpperYQuantity .
            ?checkedUpperXQuantity qudt:numericValue ?checkedUpperX .
            ?checkedUpperYQuantity qudt:numericValue ?checkedUpperY .

            FILTER NOT EXISTS {
                $this nltl:hasIntendedIceOperatingWaterline ?correspondingWaterline .
                ?correspondingWaterline nltl:hasWaterlineProfilePoint ?correspondingPoint .
                ?correspondingPoint nltl:longitudinalPosition ?correspondingXQuantity ;
                                    nltl:verticalCoordinate ?correspondingYQuantity .
                ?correspondingXQuantity qudt:numericValue ?correspondingX .
                ?correspondingYQuantity qudt:numericValue ?correspondingY .

                FILTER (
                    ?correspondingX = ?checkedUpperX &&
                    ?correspondingY = ?checkedUpperY
                )

                FILTER NOT EXISTS {
                    $this nltl:hasIntendedIceOperatingWaterline ?greaterWaterline .
                    ?greaterWaterline nltl:hasWaterlineProfilePoint ?greaterPoint .
                    ?greaterPoint nltl:longitudinalPosition ?greaterXQuantity ;
                                  nltl:verticalCoordinate ?greaterYQuantity .
                    ?greaterXQuantity qudt:numericValue ?greaterX .
                    ?greaterYQuantity qudt:numericValue ?greaterY .
                    FILTER (
                        ?greaterX = ?checkedUpperX &&
                        ?greaterY > ?checkedUpperY
                    )
                }
            }
        }
    )
}
"""
    ] .

gen:IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:node gen:WaterlineProfilePointShape ;
        sh:minCount 1 ;
        sh:message "An ice waterline must have at least one profile point."
    ] .

gen:WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MetreDecimalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must have exactly one decimal longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MetreDecimalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each waterline profile point must have exactly one decimal vertical coordinate in metres."
    ] .

gen:MetreDecimalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must have exactly one xsd:decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must use metre as its sole unit."
    ] .
<END_SHACL>
