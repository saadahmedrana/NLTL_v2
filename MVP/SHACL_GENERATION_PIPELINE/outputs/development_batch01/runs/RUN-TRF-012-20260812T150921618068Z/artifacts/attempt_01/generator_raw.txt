<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LowerIceWaterlineShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node gen:IceWaterlineShape ;
        sh:message "A lower ice waterline with profile data is required."
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node gen:IceWaterlineShape ;
        sh:message "Intended ice operating waterline data is required to establish the lower envelope."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The lower ice waterline profile shall coincide with the lowest intended ice operating waterline profile at each represented longitudinal position." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
  {
    $this nltl:hasLowerIceWaterline ?lowerWaterline .
    ?lowerWaterline nltl:hasWaterlineProfilePoint ?lowerPoint .
    ?lowerPoint nltl:longitudinalPosition ?lowerXQuantity ;
                nltl:verticalCoordinate ?lowerZQuantity .
    ?lowerXQuantity qudt:numericValue ?lowerX ;
                    qudt:unit unit:M .
    ?lowerZQuantity qudt:numericValue ?lowerZ ;
                    qudt:unit unit:M .
    FILTER(datatype(?lowerX) = xsd:decimal)
    FILTER(datatype(?lowerZ) = xsd:decimal)

    FILTER NOT EXISTS {
      $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
      ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
      ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                     nltl:verticalCoordinate ?intendedZQuantity .
      ?intendedXQuantity qudt:numericValue ?intendedX ;
                         qudt:unit unit:M .
      ?intendedZQuantity qudt:numericValue ?intendedZ ;
                         qudt:unit unit:M .
      FILTER(datatype(?intendedX) = xsd:decimal)
      FILTER(datatype(?intendedZ) = xsd:decimal)
      FILTER(?intendedX = ?lowerX)
      FILTER(?intendedZ = ?lowerZ)

      FILTER NOT EXISTS {
        $this nltl:hasIntendedIceOperatingWaterline ?otherWaterline .
        ?otherWaterline nltl:hasWaterlineProfilePoint ?otherPoint .
        ?otherPoint nltl:longitudinalPosition ?otherXQuantity ;
                    nltl:verticalCoordinate ?otherZQuantity .
        ?otherXQuantity qudt:numericValue ?otherX ;
                        qudt:unit unit:M .
        ?otherZQuantity qudt:numericValue ?otherZ ;
                        qudt:unit unit:M .
        FILTER(datatype(?otherX) = xsd:decimal)
        FILTER(datatype(?otherZ) = xsd:decimal)
        FILTER(?otherX = ?intendedX)
        FILTER(?otherZ < ?intendedZ)
      }
    }
  }
  UNION
  {
    $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
    ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
    ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                   nltl:verticalCoordinate ?intendedZQuantity .
    ?intendedXQuantity qudt:numericValue ?intendedX ;
                       qudt:unit unit:M .
    ?intendedZQuantity qudt:numericValue ?intendedZ ;
                       qudt:unit unit:M .
    FILTER(datatype(?intendedX) = xsd:decimal)
    FILTER(datatype(?intendedZ) = xsd:decimal)

    FILTER NOT EXISTS {
      $this nltl:hasIntendedIceOperatingWaterline ?lowerIntendedWaterline .
      ?lowerIntendedWaterline nltl:hasWaterlineProfilePoint ?lowerIntendedPoint .
      ?lowerIntendedPoint nltl:longitudinalPosition ?otherXQuantity ;
                          nltl:verticalCoordinate ?otherZQuantity .
      ?otherXQuantity qudt:numericValue ?otherX ;
                      qudt:unit unit:M .
      ?otherZQuantity qudt:numericValue ?otherZ ;
                      qudt:unit unit:M .
      FILTER(datatype(?otherX) = xsd:decimal)
      FILTER(datatype(?otherZ) = xsd:decimal)
      FILTER(?otherX = ?intendedX)
      FILTER(?otherZ < ?intendedZ)
    }

    FILTER NOT EXISTS {
      $this nltl:hasLowerIceWaterline ?lowerWaterline .
      ?lowerWaterline nltl:hasWaterlineProfilePoint ?lowerPoint .
      ?lowerPoint nltl:longitudinalPosition ?lowerXQuantity ;
                  nltl:verticalCoordinate ?lowerZQuantity .
      ?lowerXQuantity qudt:numericValue ?lowerX ;
                      qudt:unit unit:M .
      ?lowerZQuantity qudt:numericValue ?lowerZ ;
                      qudt:unit unit:M .
      FILTER(datatype(?lowerX) = xsd:decimal)
      FILTER(datatype(?lowerZ) = xsd:decimal)
      FILTER(?lowerX = ?intendedX)
      FILTER(?lowerZ = ?intendedZ)
    }
  }
}
"""
    ] .

gen:IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:node gen:WaterlineProfilePointShape ;
        sh:message "A waterline requires at least one profile point."
    ] .

gen:WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:node gen:LengthQuantityValueShape ;
        sh:message "A waterline profile point requires a longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:node gen:LengthQuantityValueShape ;
        sh:message "A waterline profile point requires a vertical coordinate in metres."
    ] .

gen:LengthQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
