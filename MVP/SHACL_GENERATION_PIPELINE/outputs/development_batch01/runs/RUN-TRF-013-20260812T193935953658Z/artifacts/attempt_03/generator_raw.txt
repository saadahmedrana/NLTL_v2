<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceWaterlineDraughtShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:draughtAtForePerpendicular ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:draughtAtAftPerpendicular ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityValueShape
    ] .

gen:MaximumMinimumDraughtShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must have exactly one recognized ice class."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:IceWaterlineDraughtShape
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:IceWaterlineDraughtShape
            ] ;
            sh:property [
                sh:path nltl:operatingForeDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:operatingAftDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:LengthQuantityValueShape
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "The operating fore and aft draughts must be between the corresponding lower and upper ice-waterline draughts." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (
        nltl:iceClassIaSuper,
        nltl:iceClassIa,
        nltl:iceClassIb,
        nltl:iceClassIc
    ))
    {
        $this nltl:hasLowerIceWaterline ?lower ;
              nltl:operatingForeDraught ?operatingFore .
        ?lower nltl:draughtAtForePerpendicular ?lowerFore .
        ?lowerFore qudt:numericValue ?lowerForeValue .
        ?operatingFore qudt:numericValue ?operatingForeValue .
        FILTER (
            xsd:decimal(?lowerForeValue) >
            xsd:decimal(?operatingForeValue)
        )
    }
    UNION
    {
        $this nltl:hasUpperIceWaterline ?upper ;
              nltl:operatingForeDraught ?operatingFore .
        ?upper nltl:draughtAtForePerpendicular ?upperFore .
        ?upperFore qudt:numericValue ?upperForeValue .
        ?operatingFore qudt:numericValue ?operatingForeValue .
        FILTER (
            xsd:decimal(?operatingForeValue) >
            xsd:decimal(?upperForeValue)
        )
    }
    UNION
    {
        $this nltl:hasLowerIceWaterline ?lower ;
              nltl:operatingAftDraught ?operatingAft .
        ?lower nltl:draughtAtAftPerpendicular ?lowerAft .
        ?lowerAft qudt:numericValue ?lowerAftValue .
        ?operatingAft qudt:numericValue ?operatingAftValue .
        FILTER (
            xsd:decimal(?lowerAftValue) >
            xsd:decimal(?operatingAftValue)
        )
    }
    UNION
    {
        $this nltl:hasUpperIceWaterline ?upper ;
              nltl:operatingAftDraught ?operatingAft .
        ?upper nltl:draughtAtAftPerpendicular ?upperAft .
        ?upperAft qudt:numericValue ?upperAftValue .
        ?operatingAft qudt:numericValue ?operatingAftValue .
        FILTER (
            xsd:decimal(?operatingAftValue) >
            xsd:decimal(?upperAftValue)
        )
    }
}
"""
            ]
        ]
    ) .
<END_SHACL>
