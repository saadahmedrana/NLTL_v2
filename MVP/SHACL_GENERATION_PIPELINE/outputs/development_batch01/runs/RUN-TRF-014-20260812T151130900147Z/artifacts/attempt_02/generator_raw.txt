@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:class nltl:iceDraughtRestrictionDocument ;
    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:hasValue "true"^^xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF014ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:firstScheduledDryDockingDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:node gen:IceDraughtRestrictionDocumentShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasClassCertificate ;
        sh:node gen:ClassCertificateShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node gen:MetreQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship built before 1 July 2007 must record exactly one first scheduled dry-docking date after 1 July 2007." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:constructionDate ?constructionDate .
    FILTER (?constructionDate < "2007-07-01"^^xsd:date)
    FILTER NOT EXISTS {
        $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
        FILTER (
            datatype(?dryDockingDate) = xsd:date &&
            ?dryDockingDate > "2007-07-01"^^xsd:date
        )
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where the summer load line draught exceeds the UIWL draught, the warning triangle and ice-class draught mark must be present and the mark draught must be recorded." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:summerLoadLineFreshWaterDraught ?summerNode ;
          nltl:upperIceWaterlineDraught ?uiwlNode .
    ?summerNode qudt:numericValue ?summerValue ;
                qudt:unit unit:M .
    ?uiwlNode qudt:numericValue ?uiwlValue ;
              qudt:unit unit:M .
    FILTER (xsd:decimal(?summerValue) > xsd:decimal(?uiwlValue))
    FILTER NOT EXISTS {
        $this nltl:warningTrianglePresent "true"^^xsd:boolean ;
              nltl:iceClassDraughtMarkPresent "true"^^xsd:boolean ;
              nltl:iceClassDraughtMarkDraughtAmidships ?markNode .
        ?markNode qudt:numericValue ?markValue ;
                  qudt:unit unit:M .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice-class draught mark amidships must equal the maximum amidships draught on each conforming linked class certificate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:summerLoadLineFreshWaterDraught ?summerNode ;
          nltl:upperIceWaterlineDraught ?uiwlNode ;
          nltl:iceClassDraughtMarkDraughtAmidships ?markNode ;
          nltl:hasClassCertificate ?certificate .
    ?summerNode qudt:numericValue ?summerValue ;
                qudt:unit unit:M .
    ?uiwlNode qudt:numericValue ?uiwlValue ;
              qudt:unit unit:M .
    ?markNode qudt:numericValue ?markValue ;
              qudt:unit unit:M .
    ?certificate nltl:maximumIceClassDraughtAmidships ?maximumNode .
    ?maximumNode qudt:numericValue ?maximumValue ;
                 qudt:unit unit:M .
    FILTER (xsd:decimal(?summerValue) > xsd:decimal(?uiwlValue))
    FILTER (xsd:decimal(?markValue) != xsd:decimal(?maximumValue))
}
"""
    ] .
