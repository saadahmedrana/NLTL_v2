gen:TRF014MarkingApplicabilityShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
                    nltl:upperIceWaterlineDraught ?uiwlQuantity .
                ?summerQuantity qudt:numericValue ?summerValue ;
                    qudt:unit unit:M .
                ?uiwlQuantity qudt:numericValue ?uiwlValue ;
                    qudt:unit unit:M .
                FILTER(datatype(?summerValue) = xsd:decimal)
                FILTER(datatype(?uiwlValue) = xsd:decimal)
                FILTER(?summerValue > ?uiwlValue)
            }
        """
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2007-07-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:node gen:IceClassDraughtMarkingShape
        ]
        [
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2007-07-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:firstScheduledDryDockingDate ;
                sh:datatype xsd:date ;
                sh:minExclusive "2007-07-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:node gen:IceClassDraughtMarkingShape
        ]
    ) ;
    sh:message "An applicable ship must satisfy the post-2007 or pre-2007 ice-class draught-marking conditions." .

gen:IceClassDraughtMarkingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A warning triangle must be present."
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice-class draught mark must be present."
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node gen:MetreQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The amidships ice-class draught-mark value must be recorded in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The amidships mark must equal the certificate maximum amidships draught within relative tolerance 1e-6 or absolute tolerance 1e-9 metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity ;
                    nltl:hasClassCertificate ?certificate .
                ?markQuantity qudt:numericValue ?markValue ;
                    qudt:unit unit:M .
                ?certificate nltl:maximumIceClassDraughtAmidships ?maximumQuantity .
                ?maximumQuantity qudt:numericValue ?maximumValue ;
                    qudt:unit unit:M .
                FILTER(datatype(?markValue) = xsd:decimal)
                FILTER(datatype(?maximumValue) = xsd:decimal)
                FILTER(
                    ABS(xsd:decimal(?markValue) - xsd:decimal(?maximumValue))
                        > (0.000001 * ABS(xsd:decimal(?maximumValue)))
                    &&
                    ABS(xsd:decimal(?markValue) - xsd:decimal(?maximumValue))
                        > 0.000000001
                )
            }
        """
    ] .
