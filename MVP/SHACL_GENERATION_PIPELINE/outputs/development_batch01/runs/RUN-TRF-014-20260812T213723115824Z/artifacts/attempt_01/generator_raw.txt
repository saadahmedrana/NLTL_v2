<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthInMetresShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:TRF014ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship's ice class is required to establish conformance."
    ] ;

    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "The construction date is required to determine the applicable marking provision."
    ] ;

    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceDraughtRestrictionDocument ;
        sh:minCount 1 ;
        sh:message "Restrictions on draughts when operating in ice shall be documented."
    ] ;

    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ice draught restriction documentation shall be retained on board."
    ] ;

    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ice draught restriction documentation shall be readily available to the master."
    ] ;

    sh:property [
        sh:path nltl:hasClassCertificate ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:classCertificate ;
        sh:minCount 1 ;
        sh:message "A class certificate indicating the ice class draughts is required."
    ] ;

    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:maximumIceClassDraughtAmidships ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:minimumIceClassDraughtAmidships ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1 ;
        sh:message "The summer load line fresh-water draught is required to determine marking applicability."
    ] ;

    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthInMetresShape ;
        sh:minCount 1 ;
        sh:message "The upper ice waterline draught is required to determine marking applicability."
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where the summer load line in fresh water is above the upper ice waterline, a warning triangle and an ice class draught mark at the maximum permissible ice class draught amidships are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionDate ?constructionDate ;
          nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
          nltl:upperIceWaterlineDraught ?uiwlQuantity ;
          nltl:maximumIceClassDraughtAmidships ?maximumQuantity .

    ?summerQuantity qudt:numericValue ?summerDraught ;
                    qudt:unit unit:M .
    ?uiwlQuantity qudt:numericValue ?uiwlDraught ;
                  qudt:unit unit:M .
    ?maximumQuantity qudt:numericValue ?maximumDraught ;
                     qudt:unit unit:M .

    FILTER(?summerDraught > ?uiwlDraught)

    FILTER(
        NOT EXISTS {
            $this nltl:warningTrianglePresent true .
        }
        ||
        NOT EXISTS {
            $this nltl:iceClassDraughtMarkPresent true .
        }
        ||
        NOT EXISTS {
            $this nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity .
            ?markQuantity qudt:numericValue ?markDraught ;
                          qudt:unit unit:M .
            FILTER(?markDraught = ?maximumDraught)
        }
    )
}
"""
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship built before 1 July 2007 to which the marking provision applies, the first scheduled dry docking after 1 July 2007 is required to establish the marking deadline." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionDate ?constructionDate ;
          nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
          nltl:upperIceWaterlineDraught ?uiwlQuantity .

    ?summerQuantity qudt:numericValue ?summerDraught ;
                    qudt:unit unit:M .
    ?uiwlQuantity qudt:numericValue ?uiwlDraught ;
                  qudt:unit unit:M .

    FILTER(
        YEAR(?constructionDate) < 2007
        ||
        (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) < 7)
        ||
        (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) = 7 && DAY(?constructionDate) < 1)
    )
    FILTER(?summerDraught > ?uiwlDraught)

    FILTER NOT EXISTS {
        $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
        FILTER(
            datatype(?dryDockingDate) = xsd:date
            &&
            (
                YEAR(?dryDockingDate) > 2007
                ||
                (YEAR(?dryDockingDate) = 2007 && MONTH(?dryDockingDate) > 7)
                ||
                (YEAR(?dryDockingDate) = 2007 && MONTH(?dryDockingDate) = 7 && DAY(?dryDockingDate) > 1)
            )
        )
    }
}
"""
    ] .
<END_SHACL>
