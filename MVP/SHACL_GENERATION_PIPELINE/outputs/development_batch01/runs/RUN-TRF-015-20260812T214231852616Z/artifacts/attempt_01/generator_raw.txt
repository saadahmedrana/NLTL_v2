gs:TRF015Shape

                sh:path nltl:navigatingInIce ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:in ( false ) ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:operatingDraught ;
                sh:minCount 1 ;

                sh:message "Operating draught in metres is required when navigating in ice."

            sh:property [

                sh:message "Upper ice waterline draught in metres is required when navigating in ice."

                sh:path nltl:operatingTrim ;
                sh:minCount 1 ;
                sh:node gs:MetreQuantityShape ;
                sh:message "Operating trim in metres is required when navigating in ice."
            ] ;
            sh:property [
                sh:path nltl:maximumPermittedIceTrim ;
                sh:minCount 1 ;
                sh:node gs:MetreQuantityShape ;

                sh:message "Loading-calculation sea-water salinity in grams per kilogram is required when navigating in ice."

            sh:sparql [

                sh:message "Operating draught must not exceed the upper ice waterline draught." ;
                sh:select """

PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
