<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF017EngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:message "Engine output P must be recorded in kilowatts." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propulsionMachineryContinuousOutput ;
        sh:minCount 1 ;
        sh:message "The propulsion machinery continuous output must be recorded in kilowatts." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propulsionOutputRestrictionApplies ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:restrictedPropulsionOutput ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an output restriction applies, engine output P must equal a restricted propulsion output in kilowatts." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propulsionOutputRestrictionApplies ?applies ;
          nltl:maximumContinuousRatingPower ?p .
    FILTER(?applies = true)
    ?p qudt:numericValue ?pValue ;
       qudt:unit unit:KiloW .
    FILTER NOT EXISTS {
        $this nltl:restrictedPropulsionOutput ?restricted .
        ?restricted qudt:numericValue ?restrictedValue ;
                    qudt:unit unit:KiloW .
        FILTER(xsd:decimal(?pValue) = xsd:decimal(?restrictedValue))
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an output restriction does not apply, engine output P must equal the propulsion machinery continuous output in kilowatts." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propulsionOutputRestrictionApplies ?applies ;
          nltl:maximumContinuousRatingPower ?p .
    FILTER(?applies = false)
    ?p qudt:numericValue ?pValue ;
       qudt:unit unit:KiloW .
    FILTER NOT EXISTS {
        $this nltl:propulsionMachineryContinuousOutput ?continuous .
        ?continuous qudt:numericValue ?continuousValue ;
                    qudt:unit unit:KiloW .
        FILTER(xsd:decimal(?pValue) = xsd:decimal(?continuousValue))
    }
}
"""
    ] .
<END_SHACL>
