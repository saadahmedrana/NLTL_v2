<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF018RequiredEngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximum continuous rating power must be at least the calculated required power and 1000 kW; for IA Super, it must be at least the calculated required power and 2800 kW. Both power inputs must be decimal quantities in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceClass ?iceClass .

                VALUES (?iceClass ?minimumPower) {
                    (nltl:iceClassIaSuper "2800"^^xsd:decimal)
                    (nltl:iceClassIa "1000"^^xsd:decimal)
                    (nltl:iceClassIb "1000"^^xsd:decimal)
                    (nltl:iceClassIc "1000"^^xsd:decimal)
                }

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?mcrQuantity .
                        ?mcrQuantity
                            qudt:numericValue ?mcrValue ;
                            qudt:unit unit:KiloW .
                        FILTER(datatype(?mcrValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:calculatedRequiredPower ?requiredQuantity .
                        ?requiredQuantity
                            qudt:numericValue ?requiredValue ;
                            qudt:unit unit:KiloW .
                        FILTER(datatype(?requiredValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?mcrQuantity .
                        FILTER NOT EXISTS {
                            ?mcrQuantity
                                qudt:numericValue ?mcrValue ;
                                qudt:unit unit:KiloW .
                            FILTER(datatype(?mcrValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?requiredQuantity .
                        FILTER NOT EXISTS {
                            ?requiredQuantity
                                qudt:numericValue ?requiredValue ;
                                qudt:unit unit:KiloW .
                            FILTER(datatype(?requiredValue) = xsd:decimal)
                        }
                    }
                    ||
                    EXISTS {
                        $this
                            nltl:maximumContinuousRatingPower ?mcrQuantity ;
                            nltl:calculatedRequiredPower ?requiredQuantity .
                        ?mcrQuantity
                            qudt:numericValue ?mcrValue ;
                            qudt:unit unit:KiloW .
                        ?requiredQuantity
                            qudt:numericValue ?requiredValue ;
                            qudt:unit unit:KiloW .
                        FILTER(
                            datatype(?mcrValue) = xsd:decimal &&
                            datatype(?requiredValue) = xsd:decimal &&
                            (?mcrValue < ?requiredValue || ?mcrValue < ?minimumPower)
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
