<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF020EngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For an applicable ship, engine-output inputs must use the required units, Ke must match Table 3-1, both draught outputs must satisfy formula 3.1 within a relative tolerance of 1e-6 (with a 1e-9 absolute fallback near zero), and MCR power must be at least both outputs." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:constructionStageDate ?construction ;
                      nltl:iceClass ?iceClass .

                FILTER (
                    DATATYPE(?construction) = xsd:date &&
                    (
                        YEAR(?construction) > 2003 ||
                        (
                            YEAR(?construction) = 2003 &&
                            (
                                MONTH(?construction) > 9 ||
                                (
                                    MONTH(?construction) = 9 &&
                                    DAY(?construction) >= 1
                                )
                            )
                        )
                    )
                )
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?mcrNode ;
                          nltl:minimumRequiredPowerAtUpperIceWaterline ?pUiNode ;
                          nltl:minimumRequiredPowerAtLowerIceWaterline ?pLiNode ;
                          nltl:brashIceChannelResistanceAtUpperIceWaterline ?rUiNode ;
                          nltl:brashIceChannelResistanceAtLowerIceWaterline ?rLiNode ;
                          nltl:propellerDiameter ?diameterNode ;
                          nltl:engineOutputCoefficientKe ?keNode ;
                          nltl:propellerCount ?propellerCount ;
                          nltl:propulsionSystemType ?systemType .

                    OPTIONAL {
                        $this nltl:propellerPitchControlType ?pitchType .
                    }

                    ?mcrNode qudt:numericValue ?mcr ;
                             qudt:unit unit:KiloW .
                    ?pUiNode qudt:numericValue ?pUi ;
                             qudt:unit unit:KiloW .
                    ?pLiNode qudt:numericValue ?pLi ;
                             qudt:unit unit:KiloW .
                    ?rUiNode qudt:numericValue ?rUi ;
                             qudt:unit unit:N .
                    ?rLiNode qudt:numericValue ?rLi ;
                             qudt:unit unit:N .
                    ?diameterNode qudt:numericValue ?diameter ;
                                  qudt:unit unit:M .
                    ?keNode qudt:numericValue ?ke ;
                            qudt:unit unit:UNITLESS .

                    FILTER (
                        DATATYPE(?mcr) = xsd:decimal &&
                        DATATYPE(?pUi) = xsd:decimal &&
                        DATATYPE(?pLi) = xsd:decimal &&
                        DATATYPE(?rUi) = xsd:decimal &&
                        DATATYPE(?rLi) = xsd:decimal &&
                        DATATYPE(?diameter) = xsd:decimal &&
                        DATATYPE(?ke) = xsd:decimal &&
                        DATATYPE(?propellerCount) = xsd:integer
                    )

                    FILTER (
                        ?diameter > 0 &&
                        ?rUi >= 0 &&
                        ?rLi >= 0 &&
                        ?pUi >= 0 &&
                        ?pLi >= 0
                    )

                    FILTER (
                        (
                            ?propellerCount = 1 &&
                            (
                                (
                                    (
                                        (
                                            ?systemType = nltl:conventionalPropulsionSystem &&
                                            ?pitchType = nltl:controllablePitch
                                        ) ||
                                        ?systemType IN (
                                            nltl:electricPropulsionMachinery,
                                            nltl:hydraulicPropulsionMachinery
                                        )
                                    ) &&
                                    ?ke = 2.03
                                ) ||
                                (
                                    ?systemType = nltl:conventionalPropulsionSystem &&
                                    ?pitchType = nltl:fixedPitch &&
                                    ?ke = 2.26
                                )
                            )
                        ) ||
                        (
                            ?propellerCount = 2 &&
                            (
                                (
                                    (
                                        (
                                            ?systemType = nltl:conventionalPropulsionSystem &&
                                            ?pitchType = nltl:controllablePitch
                                        ) ||
                                        ?systemType IN (
                                            nltl:electricPropulsionMachinery,
                                            nltl:hydraulicPropulsionMachinery
                                        )
                                    ) &&
                                    ?ke = 1.44
                                ) ||
                                (
                                    ?systemType = nltl:conventionalPropulsionSystem &&
                                    ?pitchType = nltl:fixedPitch &&
                                    ?ke = 1.60
                                )
                            )
                        ) ||
                        (
                            ?propellerCount = 3 &&
                            (
                                (
                                    (
                                        (
                                            ?systemType = nltl:conventionalPropulsionSystem &&
                                            ?pitchType = nltl:controllablePitch
                                        ) ||
                                        ?systemType IN (
                                            nltl:electricPropulsionMachinery,
                                            nltl:hydraulicPropulsionMachinery
                                        )
                                    ) &&
                                    ?ke = 1.18
                                ) ||
                                (
                                    ?systemType = nltl:conventionalPropulsionSystem &&
                                    ?pitchType = nltl:fixedPitch &&
                                    ?ke = 1.31
                                )
                            )
                        )
                    )

                    BIND (?rUi / 1000 AS ?rUiKilonewton)
                    BIND (?rLi / 1000 AS ?rLiKilonewton)
                    BIND (
                        (?ke * ?ke *
                         ?rUiKilonewton * ?rUiKilonewton * ?rUiKilonewton) /
                        (?diameter * ?diameter)
                        AS ?expectedUiSquared
                    )
                    BIND (
                        (?ke * ?ke *
                         ?rLiKilonewton * ?rLiKilonewton * ?rLiKilonewton) /
                        (?diameter * ?diameter)
                        AS ?expectedLiSquared
                    )

                    FILTER (
                        (
                            ?expectedUiSquared = 0 &&
                            ABS(?pUi) <= 0.000000001
                        ) ||
                        (
                            ?expectedUiSquared > 0 &&
                            (?pUi * ?pUi) >=
                                (?expectedUiSquared * 0.999999 * 0.999999) &&
                            (?pUi * ?pUi) <=
                                (?expectedUiSquared * 1.000001 * 1.000001)
                        )
                    )
                    FILTER (
                        (
                            ?expectedLiSquared = 0 &&
                            ABS(?pLi) <= 0.000000001
                        ) ||
                        (
                            ?expectedLiSquared > 0 &&
                            (?pLi * ?pLi) >=
                                (?expectedLiSquared * 0.999999 * 0.999999) &&
                            (?pLi * ?pLi) <=
                                (?expectedLiSquared * 1.000001 * 1.000001)
                        )
                    )
                    FILTER (?mcr >= ?pUi && ?mcr >= ?pLi)
                }
            }
        """
    ] .
<END_SHACL>
