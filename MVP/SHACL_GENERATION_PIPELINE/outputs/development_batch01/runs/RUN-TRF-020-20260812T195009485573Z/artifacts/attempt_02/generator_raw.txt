<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:N ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DiameterQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:CoefficientQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:EngineOutputRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] ;
    sh:or (
        [
            sh:not [
                sh:or (
                    [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIaSuper ] ]
                    [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIa ] ]
                    [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIb ] ]
                    [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIc ] ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:maxExclusive "2003-09-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2003-09-01"^^xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:in ( 1 2 3 ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Exactly one propeller count of 1, 2, or 3 is required."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:controllablePitch
                    nltl:fixedPitch
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:conventionalPropulsionSystem
                    nltl:electricPropulsionMachinery
                    nltl:hydraulicPropulsionMachinery
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:node gen:DiameterQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:node gen:CoefficientQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:node gen:ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:node gen:ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceRch ;
                sh:node gen:ForceQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node gen:PowerQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 1 ] ;
                    sh:or (
                        [
                            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                            sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:controllablePitch ]
                        ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ]
                    ) ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 2.03 ]
                        ]
                    ]
                ]
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 1 ] ;
                    sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                    sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:fixedPitch ] ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 2.26 ]
                        ]
                    ]
                ]
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 2 ] ;
                    sh:or (
                        [
                            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                            sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:controllablePitch ]
                        ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ]
                    ) ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 1.44 ]
                        ]
                    ]
                ]
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 2 ] ;
                    sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                    sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:fixedPitch ] ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 1.60 ]
                        ]
                    ]
                ]
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 3 ] ;
                    sh:or (
                        [
                            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                            sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:controllablePitch ]
                        ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ]
                        [ sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ]
                    ) ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 1.18 ]
                        ]
                    ]
                ]
                [
                    sh:property [ sh:path nltl:propellerCount ; sh:hasValue 3 ] ;
                    sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
                    sh:property [ sh:path nltl:propellerPitchControlType ; sh:hasValue nltl:fixedPitch ] ;
                    sh:property [
                        sh:path nltl:engineOutputCoefficientKe ;
                        sh:node [
                            sh:property [ sh:path qudt:numericValue ; sh:hasValue 1.31 ]
                        ]
                    ]
                ]
            ) ;
            sh:sparql [
                sh:message "The UIWL and LIWL minimum powers must satisfy formula 3.1 within a relative power tolerance of 1e-6, and MCR power must be no less than both outputs." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:engineOutputCoefficientKe/qudt:numericValue ?ke ;
                              nltl:propellerDiameter/qudt:numericValue ?diameter ;
                              nltl:brashIceChannelResistanceAtUpperIceWaterline/qudt:numericValue ?rchUpper ;
                              nltl:brashIceChannelResistanceAtLowerIceWaterline/qudt:numericValue ?rchLower ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline/qudt:numericValue ?powerUpper ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline/qudt:numericValue ?powerLower ;
                              nltl:maximumContinuousRatingPower/qudt:numericValue ?mcr .

                        BIND ((?powerUpper * ?diameter) / ?ke AS ?scaledUpper)
                        BIND (?scaledUpper * ?scaledUpper AS ?leftUpper)
                        BIND (?rchUpper / xsd:decimal("1000") AS ?resistanceUpper)
                        BIND (?resistanceUpper * ?resistanceUpper * ?resistanceUpper AS ?rightUpper)

                        BIND ((?powerLower * ?diameter) / ?ke AS ?scaledLower)
                        BIND (?scaledLower * ?scaledLower AS ?leftLower)
                        BIND (?rchLower / xsd:decimal("1000") AS ?resistanceLower)
                        BIND (?resistanceLower * ?resistanceLower * ?resistanceLower AS ?rightLower)

                        FILTER (
                            (
                                ABS(?leftUpper - ?rightUpper) > xsd:decimal("0.000000000001") &&
                                (
                                    ?leftUpper < (?rightUpper * xsd:decimal("0.999998000001")) ||
                                    ?leftUpper > (?rightUpper * xsd:decimal("1.000002000001"))
                                )
                            ) ||
                            (
                                ABS(?leftLower - ?rightLower) > xsd:decimal("0.000000000001") &&
                                (
                                    ?leftLower < (?rightLower * xsd:decimal("0.999998000001")) ||
                                    ?leftLower > (?rightLower * xsd:decimal("1.000002000001"))
                                )
                            ) ||
                            ?mcr < ?powerUpper ||
                            ?mcr < ?powerLower
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
