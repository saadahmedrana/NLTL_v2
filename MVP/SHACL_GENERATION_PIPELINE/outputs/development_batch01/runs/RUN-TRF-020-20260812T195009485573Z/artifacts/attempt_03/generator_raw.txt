gen:TRF020Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF020NonApplicableIceClassShape
        gen:TRF020PreCutoffShape
        gen:TRF020ApplicableShape
    ) .

gen:TRF020NonApplicableIceClassShape
    a sh:NodeShape ;
    sh:not [
        sh:property [
            sh:path nltl:iceClass ;
            sh:hasValue nltl:iceClassIaSuper
        ]
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:iceClass ;
            sh:hasValue nltl:iceClassIa
        ]
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:iceClass ;
            sh:hasValue nltl:iceClassIb
        ]
    ] ;
    sh:not [
        sh:property [
            sh:path nltl:iceClass ;
            sh:hasValue nltl:iceClassIc
        ]
    ] .

gen:TRF020PreCutoffShape
    a sh:NodeShape ;
    sh:or (
        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIaSuper ] ]
        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIa ] ]
        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIb ] ]
        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIc ] ]
    ) ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxExclusive "2003-09-01"^^xsd:date ;
        sh:message "Applicable-class ships treated as pre-cutoff must provide one construction-stage date before 2003-09-01."
    ] .

        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIb ] ]
        [ sh:property [ sh:path nltl:iceClass ; sh:hasValue nltl:iceClassIc ] ]
    ) ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( 1 2 3 )
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        )
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveUnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveKilowattQuantityShape
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveKilowattQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveKilowattQuantityShape
    ] ;
    sh:or (
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 1 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke203Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 2 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke144Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:electricPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 3 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke118Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 1 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke203Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 2 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke144Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:hydraulicPropulsionMachinery ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 3 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke118Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;

                sh:path nltl:propellerPitchControlType ;

                sh:maxCount 1 ;

            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 1 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke203Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:nodeKind sh:IRI ;

                sh:maxCount 1 ;

            ] ;
            sh:property [ sh:path nltl:propellerCount ; sh:hasValue 2 ] ;
            sh:property [ sh:path nltl:engineOutputCoefficientKe ; sh:node gen:Ke144Shape ]
        ]
        [
            sh:property [ sh:path nltl:propulsionSystemType ; sh:hasValue nltl:conventionalPropulsionSystem ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;

    sh:sparql [
        sh:message "UIWL and LIWL minimum powers must satisfy formula 3.1 within relative tolerance 1e-6, and MCR must be no less than both powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter/qudt:numericValue ?diameter ;
                      nltl:engineOutputCoefficientKe/qudt:numericValue ?ke ;
                      nltl:brashIceChannelResistanceAtUpperIceWaterline/qudt:numericValue ?resistanceUI ;
                      nltl:brashIceChannelResistanceAtLowerIceWaterline/qudt:numericValue ?resistanceLI ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline/qudt:numericValue ?powerUI ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline/qudt:numericValue ?powerLI ;
                      nltl:maximumContinuousRatingPower/qudt:numericValue ?mcr .

                BIND(xsd:decimal(?diameter) AS ?d)
                BIND(xsd:decimal(?ke) AS ?k)
                BIND(xsd:decimal(?resistanceUI) / 1000.0 AS ?rui)
                BIND(xsd:decimal(?resistanceLI) / 1000.0 AS ?rli)
                BIND((xsd:decimal(?powerUI) * ?d) / ?k AS ?scaledUI)
                BIND((xsd:decimal(?powerLI) * ?d) / ?k AS ?scaledLI)
                BIND(
                    (?scaledUI * ?scaledUI) / (?rui * ?rui * ?rui)
                    AS ?formulaRatioUI
                )
                BIND(
                    (?scaledLI * ?scaledLI) / (?rli * ?rli * ?rli)
                    AS ?formulaRatioLI
                )

                FILTER(
                    ?formulaRatioUI < 0.999998000001 ||
                    ?formulaRatioUI > 1.000002000001 ||
                    ?formulaRatioLI < 0.999998000001 ||
                    ?formulaRatioLI > 1.000002000001 ||
                    xsd:decimal(?mcr) < xsd:decimal(?powerUI) ||
                    xsd:decimal(?mcr) < xsd:decimal(?powerLI)
                )
            }
        """
    ] .

gen:PositiveMetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive 0.0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
