<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PositiveUnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:in ( unit:UNITLESS ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PositiveMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:in ( unit:M ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PositiveNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [

        sh:in ( unit:N ) ;

gen:TRF020ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass .
                        FILTER (?iceClass IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ))
                        FILTER (
                            YEAR(?date) > 2003 ||
                            (YEAR(?date) = 2003 && MONTH(?date) > 9) ||
                            (YEAR(?date) = 2003 && MONTH(?date) = 9 && DAY(?date) >= 1)
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterline ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The upper ice waterline draught operand is required."
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:in ( 1 2 3 ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:conventionalPropulsionSystem
                    nltl:electricPropulsionMachinery
                    nltl:hydraulicPropulsionMachinery
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:nodeKind sh:IRI ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:node gen:PositiveUnitlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:node gen:PositiveNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:node gen:PositiveNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:node gen:PositiveKilowattQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:node gen:PositiveKilowattQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node gen:KilowattQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "A conventional propulsion system requires fixed-pitch or controllable-pitch classification." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propulsionSystemType nltl:conventionalPropulsionSystem .
                        FILTER NOT EXISTS {
                            $this nltl:propellerPitchControlType ?pitch .
                            FILTER (?pitch IN (
                                nltl:controllablePitch,
                                nltl:fixedPitch
                            ))
                        }
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Ke must equal the applicable Table 3-1 value for electric or hydraulic propulsion machinery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propulsionSystemType ?system ;
                              nltl:propellerCount ?count ;
                              nltl:engineOutputCoefficientKe ?keQuantity .
                        ?keQuantity qudt:numericValue ?ke .
                        FILTER (?system IN (
                            nltl:electricPropulsionMachinery,
                            nltl:hydraulicPropulsionMachinery
                        ))
                        FILTER (
                            (?count = 1 && ?ke != 2.03) ||
                            (?count = 2 && ?ke != 1.44) ||
                            (?count = 3 && ?ke != 1.18)
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Ke must equal the applicable Table 3-1 value for the conventional propulsion pitch type and propeller count." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propulsionSystemType nltl:conventionalPropulsionSystem ;
                              nltl:propellerPitchControlType ?pitch ;
                              nltl:propellerCount ?count ;
                              nltl:engineOutputCoefficientKe ?keQuantity .
                        ?keQuantity qudt:numericValue ?ke .
                        FILTER (
                            (?pitch = nltl:controllablePitch && (
                                (?count = 1 && ?ke != 2.03) ||
                                (?count = 2 && ?ke != 1.44) ||
                                (?count = 3 && ?ke != 1.18)
                            )) ||
                            (?pitch = nltl:fixedPitch && (
                                (?count = 1 && ?ke != 2.26) ||
                                (?count = 2 && ?ke != 1.60) ||
                                (?count = 3 && ?ke != 1.31)
                            ))
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The UIWL and LIWL minimum powers must satisfy P_min = Ke*(R_CH/1000)^(3/2)/D within relative tolerance 1e-6 and absolute fallback 1e-12." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:engineOutputCoefficientKe ?keQuantity ;
                              nltl:propellerDiameter ?diameterQuantity ;
                              nltl:brashIceChannelResistanceAtUpperIceWaterline ?upperResistanceQuantity ;
                              nltl:brashIceChannelResistanceAtLowerIceWaterline ?lowerResistanceQuantity ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline ?upperPowerQuantity ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerPowerQuantity .
                        ?keQuantity qudt:numericValue ?ke .
                        ?diameterQuantity qudt:numericValue ?diameter .
                        ?upperResistanceQuantity qudt:numericValue ?upperResistance .
                        ?lowerResistanceQuantity qudt:numericValue ?lowerResistance .
                        ?upperPowerQuantity qudt:numericValue ?upperPower .
                        ?lowerPowerQuantity qudt:numericValue ?lowerPower .
                        BIND (
                            (?upperPower * ?upperPower * ?diameter * ?diameter)
                            AS ?upperActualSquared
                        )
                        BIND (
                            (?ke * ?ke *
                             (?upperResistance / 1000.0) *
                             (?upperResistance / 1000.0) *
                             (?upperResistance / 1000.0))
                            AS ?upperExpectedSquared
                        )
                        BIND (
                            (?lowerPower * ?lowerPower * ?diameter * ?diameter)
                            AS ?lowerActualSquared
                        )
                        BIND (
                            (?ke * ?ke *
                             (?lowerResistance / 1000.0) *
                             (?lowerResistance / 1000.0) *
                             (?lowerResistance / 1000.0))
                            AS ?lowerExpectedSquared
                        )
                        FILTER (
                            ABS(?upperActualSquared - ?upperExpectedSquared) >
                                (0.000001 * ABS(?upperExpectedSquared) + 0.000000000001)
                            ||
                            ABS(?lowerActualSquared - ?lowerExpectedSquared) >
                                (0.000001 * ABS(?lowerExpectedSquared) + 0.000000000001)
                        )
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Maximum continuous rating power must be no less than both the UIWL and LIWL minimum required powers." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline ?upperPowerQuantity ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerPowerQuantity .
                        ?mcrQuantity qudt:numericValue ?mcr .
                        ?upperPowerQuantity qudt:numericValue ?upperPower .
                        ?lowerPowerQuantity qudt:numericValue ?lowerPower .
                        FILTER (?mcr < ?upperPower || ?mcr < ?lowerPower)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
