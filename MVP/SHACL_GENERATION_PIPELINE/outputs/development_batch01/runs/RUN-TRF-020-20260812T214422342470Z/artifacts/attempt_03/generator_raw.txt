gen:TRF020Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:sparql [
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?iceClass .
                        FILTER (
                            DATATYPE(?date) = xsd:date &&
                            ?iceClass IN (
                                nltl:iceClassIaSuper,
                                nltl:iceClassIa,
                                nltl:iceClassIb,
                                nltl:iceClassIc
                            ) &&
                            (
                                YEAR(?date) > 2003 ||
                                (
                                    YEAR(?date) = 2003 &&
                                    (
                                        MONTH(?date) > 9 ||
                                        (MONTH(?date) = 9 && DAY(?date) >= 1)
                                    )
                                )
                            )
                        )
                    }
                """
            ]
        ]
        gen:TRF020ApplicableShape
    ) .

gen:TRF020ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterline ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable calculation requires exactly one UIWL operand."
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterline ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The applicable calculation requires exactly one LIWL operand."
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:in ( 1 2 3 ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerPitchControlType ;
        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:N ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:N ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minExclusive 0 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Ke must equal the applicable Table 3-1 coefficient." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerCount ?count ;
                      nltl:propellerPitchControlType ?pitch ;
                      nltl:propulsionSystemType ?system ;
                      nltl:engineOutputCoefficientKe ?keQuantity .
                ?keQuantity qudt:numericValue ?ke .
                BIND(
                    IF(
                        ?count = 1,
                        IF(
                            ?pitch = nltl:controllablePitch ||
                            ?system IN (
                                nltl:electricPropulsionMachinery,
                                nltl:hydraulicPropulsionMachinery
                            ),
                            2.03,
                            2.26
                        ),
                        IF(
                            ?count = 2,
                            IF(
                                ?pitch = nltl:controllablePitch ||
                                ?system IN (
                                    nltl:electricPropulsionMachinery,
                                    nltl:hydraulicPropulsionMachinery
                                ),
                                1.44,
                                1.60
                            ),
                            IF(
                                ?pitch = nltl:controllablePitch ||
                                ?system IN (
                                    nltl:electricPropulsionMachinery,
                                    nltl:hydraulicPropulsionMachinery
                                ),
                                1.18,
                                1.31
                            )
                        )
                    )
                    AS ?expectedKe
                )
                FILTER (?ke != ?expectedKe)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The UIWL and LIWL minimum powers must satisfy formula 3.1; the squared positive-value comparison uses relative tolerance 1e-6 with absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterline ?uiwl ;
                      nltl:lowerIceWaterline ?liwl ;
                      nltl:engineOutputCoefficientKe ?keQuantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:brashIceChannelResistanceAtUpperIceWaterline ?uiResistanceQuantity ;
                      nltl:brashIceChannelResistanceAtLowerIceWaterline ?liResistanceQuantity ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline ?uiPowerQuantity ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline ?liPowerQuantity .
                ?keQuantity qudt:numericValue ?ke .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?uiResistanceQuantity qudt:numericValue ?uiResistance .
                ?liResistanceQuantity qudt:numericValue ?liResistance .
                ?uiPowerQuantity qudt:numericValue ?uiPower .
                ?liPowerQuantity qudt:numericValue ?liPower .

                BIND(
                    ?uiPower * ?uiPower * ?diameter * ?diameter
                    AS ?uiActualSquared
                )
                BIND(
                    ?ke * ?ke *
                    (?uiResistance / 1000.0) *
                    (?uiResistance / 1000.0) *
                    (?uiResistance / 1000.0)
                    AS ?uiExpectedSquared
                )
                BIND(
                    IF(
                        ABS(?uiExpectedSquared) * 0.000001 > 0.000000000001,
                        ABS(?uiExpectedSquared) * 0.000001,
                        0.000000000001
                    )
                    AS ?uiTolerance
                )

                BIND(
                    ?liPower * ?liPower * ?diameter * ?diameter
                    AS ?liActualSquared
                )
                BIND(
                    ?ke * ?ke *
                    (?liResistance / 1000.0) *
                    (?liResistance / 1000.0) *
                    (?liResistance / 1000.0)
                    AS ?liExpectedSquared
                )
                BIND(
                    IF(
                        ABS(?liExpectedSquared) * 0.000001 > 0.000000000001,
                        ABS(?liExpectedSquared) * 0.000001,
                        0.000000000001
                    )
                    AS ?liTolerance
                )

                FILTER (
                    ABS(?uiActualSquared - ?uiExpectedSquared) > ?uiTolerance ||
                    ABS(?liActualSquared - ?liExpectedSquared) > ?liTolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be no less than both calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline ?uiPowerQuantity ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline ?liPowerQuantity .
                ?mcrQuantity qudt:numericValue ?mcr .
                ?uiPowerQuantity qudt:numericValue ?uiPower .
                ?liPowerQuantity qudt:numericValue ?liPower .
                FILTER (?mcr < ?uiPower || ?mcr < ?liPower)
            }
        """
    ] .
