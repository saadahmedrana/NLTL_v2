    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:node gs:NonNegativeNewtonValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:node gs:NonNegativeKilowattValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:node gs:NonNegativeKilowattValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gs:NonNegativeKilowattValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gs:OnePropellerLowerCoefficientCase
        gs:OnePropellerHigherCoefficientCase
        gs:TwoPropellerLowerCoefficientCase
        gs:TwoPropellerHigherCoefficientCase
        gs:ThreePropellerLowerCoefficientCase
        gs:ThreePropellerHigherCoefficientCase
    ) ;
    sh:sparql [
        sh:message "Each draught-specific minimum power must satisfy formula 3.1 within a squared-equation relative tolerance of 2.000001e-6 (absolute fallback 1e-12), and MCR power must be no less than both minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:engineOutputCoefficientKe/qudt:numericValue ?ke ;
                      nltl:propellerDiameter/qudt:numericValue ?diameter ;
                      nltl:brashIceChannelResistanceAtUpperIceWaterline/qudt:numericValue ?upperResistance ;
                      nltl:brashIceChannelResistanceAtLowerIceWaterline/qudt:numericValue ?lowerResistance ;
                      nltl:minimumRequiredPowerAtUpperIceWaterline/qudt:numericValue ?upperPower ;
                      nltl:minimumRequiredPowerAtLowerIceWaterline/qudt:numericValue ?lowerPower ;
                      nltl:maximumContinuousRatingPower/qudt:numericValue ?mcrPower .

                BIND (?upperResistance / 1000.0 AS ?upperResistanceKilonewtons)
                BIND (?lowerResistance / 1000.0 AS ?lowerResistanceKilonewtons)

                BIND (
                    (?ke * ?ke *
                     ?upperResistanceKilonewtons *
                     ?upperResistanceKilonewtons *
                     ?upperResistanceKilonewtons) /
                    (?diameter * ?diameter)
                    AS ?expectedUpperSquared
                )
                BIND (
                    (?ke * ?ke *
                     ?lowerResistanceKilonewtons *
                     ?lowerResistanceKilonewtons *
                     ?lowerResistanceKilonewtons) /
                    (?diameter * ?diameter)
                    AS ?expectedLowerSquared
                )

                BIND (?upperPower * ?upperPower AS ?actualUpperSquared)
                BIND (?lowerPower * ?lowerPower AS ?actualLowerSquared)

                BIND (
                    IF(
                        ABS(?expectedUpperSquared) > 0.000000000001,
                        ABS(?expectedUpperSquared) * 0.000002000001,
                        0.000000000001
                    )
                    AS ?upperTolerance
                )
                BIND (
                    IF(
                        ABS(?expectedLowerSquared) > 0.000000000001,
                        ABS(?expectedLowerSquared) * 0.000002000001,
                        0.000000000001
                    )
                    AS ?lowerTolerance
                )

                FILTER (
                    ABS(?actualUpperSquared - ?expectedUpperSquared) > ?upperTolerance ||
                    ABS(?actualLowerSquared - ?expectedLowerSquared) > ?lowerTolerance ||
                    ?mcrPower < ?upperPower ||
                    ?mcrPower < ?lowerPower
                )
            }
        """
    ] .
