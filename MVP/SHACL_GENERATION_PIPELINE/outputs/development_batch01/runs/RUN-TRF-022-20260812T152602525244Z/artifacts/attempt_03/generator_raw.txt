    sh:property [
        sh:path nltl:shipLength ;

    sh:property [
        sh:path nltl:shipBreadth ;

        sh:path nltl:clampedHullGeometryTerm ;

        sh:path nltl:bowRakeAtQuarterBreadth ;

    ] ;
    sh:property [
        sh:path nltl:waterlineAngleAtQuarterBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DegreeQuantityShape

    sh:property [
        sh:path nltl:flareAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DegreeQuantityShape
    ] ;
    sh:sparql [
        sh:message "Ship breadth must be non-zero for the clamped hull geometry calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadth .
                FILTER (?breadth = "0"^^xsd:decimal)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Clamped hull geometry term must equal max(5, min(20, ((L*T)/(B*B))^3))." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:clampedHullGeometryTerm ?clampedQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                ?draughtQuantity qudt:numericValue ?draught .
                ?breadthQuantity qudt:numericValue ?breadth .
                ?clampedQuantity qudt:numericValue ?clamped .
                FILTER (?breadth != "0"^^xsd:decimal)
                BIND ((?length * ?draught) / (?breadth * ?breadth) AS ?ratio)
                BIND (?ratio * ?ratio * ?ratio AS ?raw)
                BIND (
                    IF(
                        ?raw < "5"^^xsd:decimal,
                        "5"^^xsd:decimal,
                        IF(
                            ?raw > "20"^^xsd:decimal,
                            "20"^^xsd:decimal,
                            ?raw
                        )
                    ) AS ?expected
                )
                FILTER (?clamped != ?expected)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Waterline angle makes the denominator of the flare-angle calculation zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:waterlineAngleAtQuarterBreadth ?angleQuantity .
                ?angleQuantity qudt:numericValue ?angle .
                FILTER (
                    ?angle =
                    ("180"^^xsd:decimal *
                     xsd:integer(?angle / "180"^^xsd:decimal))
                )
            }
        """
    ] .
