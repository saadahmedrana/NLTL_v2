<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:NewShipHullCoefficientShape
    a sh:NodeShape ;
    sh:targetClass nltl:benchmarkEntity ;
    sh:sparql [
        sh:message "For new ships, the prescribed coefficients, clamped hull geometry term, and flare-angle relation must have the required values and units." ;
        sh:select """
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:constructionStageDate ?constructionDate .

    FILTER(
        DATATYPE(?constructionDate) = xsd:date &&
        (
            YEAR(?constructionDate) > 2017 ||
            (YEAR(?constructionDate) = 2017 && MONTH(?constructionDate) > 9) ||
            (YEAR(?constructionDate) = 2017 && MONTH(?constructionDate) = 9 &&
             DAY(?constructionDate) >= 1)
        )
    )

    OPTIONAL {
        $this nltl:coefficientC3 ?c3Node .
        ?c3Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?c3 ;
                qudt:unit nltl:kilogramPerSquareMetreSquareSecondUnit .
    }
    OPTIONAL {
        $this nltl:coefficientC4 ?c4Node .
        ?c4Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?c4 ;
                qudt:unit nltl:kilogramPerSquareMetreSquareSecondUnit .
    }
    OPTIONAL {
        $this nltl:coefficientC5 ?c5Node .
        ?c5Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?c5 ;
                qudt:unit nltl:kilogramPerSquareSecondUnit .
    }
    OPTIONAL {
        $this nltl:coefficientF1 ?f1Node .
        ?f1Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?f1 ;
                qudt:unit unit:N-PER-M2 .
    }
    OPTIONAL {
        $this nltl:coefficientF2 ?f2Node .
        ?f2Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?f2 ;
                qudt:unit unit:N-PER-M .
    }
    OPTIONAL {
        $this nltl:coefficientF3 ?f3Node .
        ?f3Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?f3 ;
                qudt:unit unit:N-PER-M .
    }
    OPTIONAL {
        $this nltl:coefficientF4 ?f4Node .
        ?f4Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?f4 ;
                qudt:unit unit:N-PER-M2 .
    }
    OPTIONAL {
        $this nltl:coefficientG1 ?g1Node .
        ?g1Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?g1 ;
                qudt:unit unit:N .
    }
    OPTIONAL {
        $this nltl:coefficientG2 ?g2Node .
        ?g2Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?g2 ;
                qudt:unit unit:N-PER-M .
    }
    OPTIONAL {
        $this nltl:coefficientG3 ?g3Node .
        ?g3Node rdf:type qudt:QuantityValue ;
                qudt:numericValue ?g3 ;
                qudt:unit nltl:newtonPerMetreToPowerOnePointFiveUnit .
    }
    OPTIONAL {
        $this nltl:shipLength ?lengthNode .
        ?lengthNode rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?length ;
                    qudt:unit unit:M .
    }
    OPTIONAL {
        $this nltl:iceClassDraught ?draughtNode .
        ?draughtNode rdf:type qudt:QuantityValue ;
                     qudt:numericValue ?draught ;
                     qudt:unit unit:M .
    }
    OPTIONAL {
        $this nltl:shipBreadth ?breadthNode .
        ?breadthNode rdf:type qudt:QuantityValue ;
                     qudt:numericValue ?breadth ;
                     qudt:unit unit:M .
    }
    OPTIONAL {
        $this nltl:clampedHullGeometryTerm ?clampedNode .
        ?clampedNode rdf:type qudt:QuantityValue ;
                     qudt:numericValue ?clamped ;
                     qudt:unit unit:UNITLESS .
    }
    OPTIONAL {
        $this nltl:flareAngle ?psiNode .
        ?psiNode rdf:type qudt:QuantityValue ;
                 qudt:numericValue ?psi ;
                 qudt:unit unit:DEG .
    }
    OPTIONAL {
        $this nltl:bowRakeAtQuarterBreadth ?phi2Node .
        ?phi2Node rdf:type qudt:QuantityValue ;
                  qudt:numericValue ?phi2 ;
                  qudt:unit unit:DEG .
    }
    OPTIONAL {
        $this nltl:waterlineAngleAtQuarterBreadth ?alphaNode .
        ?alphaNode rdf:type qudt:QuantityValue ;
                   qudt:numericValue ?alpha ;
                   qudt:unit unit:DEG .
    }

    BIND(3.1415926535897932384626433832795 AS ?pi)
    BIND((?length * ?draught) / (?breadth * ?breadth) AS ?geometryRatio)
    BIND(?geometryRatio * ?geometryRatio * ?geometryRatio AS ?rawGeometryTerm)
    BIND(
        IF(?rawGeometryTerm < 5.0, 5.0,
            IF(?rawGeometryTerm > 20.0, 20.0, ?rawGeometryTerm))
        AS ?expectedClamped
    )
    BIND(
        IF(
            ABS(?expectedClamped) * 0.000001 > 0.000000001,
            ABS(?expectedClamped) * 0.000001,
            0.000000001
        )
        AS ?clampedTolerance
    )

    BIND((?phi2 * ?pi) / 180.0 AS ?phi2Radians)
    BIND((?alpha * ?pi) / 180.0 AS ?alphaRadians)
    BIND(
        math:atan(math:tan(?phi2Radians) / math:sin(?alphaRadians))
        AS ?expectedPsiRadians
    )
    BIND((?expectedPsiRadians * 180.0) / ?pi AS ?expectedPsi)
    BIND(
        IF(
            ABS(?expectedPsi) * 0.000001 > 0.000000001,
            ABS(?expectedPsi) * 0.000001,
            0.000000001
        )
        AS ?psiTolerance
    )

    FILTER(
        !BOUND(?c3) || DATATYPE(?c3) != xsd:decimal || ?c3 != 845.0 ||
        !BOUND(?c4) || DATATYPE(?c4) != xsd:decimal || ?c4 != 42.0 ||
        !BOUND(?c5) || DATATYPE(?c5) != xsd:decimal || ?c5 != 825.0 ||
        !BOUND(?f1) || DATATYPE(?f1) != xsd:decimal || ?f1 != 23.0 ||
        !BOUND(?f2) || DATATYPE(?f2) != xsd:decimal || ?f2 != 45.8 ||
        !BOUND(?f3) || DATATYPE(?f3) != xsd:decimal || ?f3 != 14.7 ||
        !BOUND(?f4) || DATATYPE(?f4) != xsd:decimal || ?f4 != 29.0 ||
        !BOUND(?g1) || DATATYPE(?g1) != xsd:decimal || ?g1 != 1530.0 ||
        !BOUND(?g2) || DATATYPE(?g2) != xsd:decimal || ?g2 != 170.0 ||
        !BOUND(?g3) || DATATYPE(?g3) != xsd:decimal || ?g3 != 400.0 ||
        !BOUND(?length) || DATATYPE(?length) != xsd:decimal ||
        !BOUND(?draught) || DATATYPE(?draught) != xsd:decimal ||
        !BOUND(?breadth) || DATATYPE(?breadth) != xsd:decimal ||
        !BOUND(?clamped) || DATATYPE(?clamped) != xsd:decimal ||
        !BOUND(?expectedClamped) || !BOUND(?clampedTolerance) ||
        ABS(?clamped - ?expectedClamped) > ?clampedTolerance ||
        !BOUND(?psi) || DATATYPE(?psi) != xsd:decimal ||
        !BOUND(?phi2) || DATATYPE(?phi2) != xsd:decimal ||
        !BOUND(?alpha) || DATATYPE(?alpha) != xsd:decimal ||
        !BOUND(?expectedPsi) || !BOUND(?psiTolerance) ||
        ABS(?psi - ?expectedPsi) > ?psiTolerance
    )
}
        """
    ] .
<END_SHACL>
