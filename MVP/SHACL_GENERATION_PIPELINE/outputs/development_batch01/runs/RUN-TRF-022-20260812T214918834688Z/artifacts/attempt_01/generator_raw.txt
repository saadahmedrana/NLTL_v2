    sh:sparql [
        sh:message "The clamped hull geometry term must equal ((L*T)/B^2)^3 clamped to [5,20], within relative tolerance 1e-6, and breadth must be non-zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:clampedHullGeometryTerm ?clampedQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                ?draughtQuantity qudt:numericValue ?draught .
                ?breadthQuantity qudt:numericValue ?breadth .
                ?clampedQuantity qudt:numericValue ?actual .
                BIND(IF(?breadth = 0, 0,
                        (?length * ?draught) / (?breadth * ?breadth))
                     AS ?ratio)
                BIND(?ratio * ?ratio * ?ratio AS ?raw)
                BIND(IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw))
                     AS ?expected)
                BIND(IF(ABS(?expected) > 0.000000000001,
                        0.000001 * ABS(?expected),
                        0.000000000001)
                     AS ?tolerance)
                FILTER(
                    ?breadth = 0 ||
                    ABS(?actual - ?expected) > ?tolerance
                )
            }
        """
    ] .
