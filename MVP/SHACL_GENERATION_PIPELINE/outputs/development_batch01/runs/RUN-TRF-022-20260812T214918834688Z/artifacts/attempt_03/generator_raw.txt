        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:bowRakeAtQuarterBreadth ?phiNode ;
                      nltl:waterlineAngleAtQuarterBreadth ?alphaNode ;
                      nltl:flareAngle ?psiNode .
                ?phiNode qudt:numericValue ?phiDegrees .
                ?alphaNode qudt:numericValue ?alphaDegrees .
                ?psiNode qudt:numericValue ?psiDegrees .

                BIND(xsd:double(?phiDegrees) * 0.017453292519943295 AS ?phiRadians)
                BIND(xsd:double(?alphaDegrees) * 0.017453292519943295 AS ?alphaRadians)
                BIND(math:sin(?alphaRadians) AS ?sinAlpha)
                BIND(math:cos(?phiRadians) AS ?cosPhi)
                BIND(math:tan(?phiRadians) AS ?tanPhi)
                BIND(math:atan(?tanPhi / ?sinAlpha) * 57.29577951308232 AS ?expectedPsi)
                BIND(ABS(?expectedPsi) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000001,
                        ?relativeTolerance,
                        0.000000001) AS ?tolerance)

                FILTER (
                    ABS(?sinAlpha) <= 0.000000000001 ||
                    ABS(?cosPhi) <= 0.000000000001 ||
                    !BOUND(?expectedPsi) ||
                    ABS(xsd:double(?psiDegrees) - ?expectedPsi) > ?tolerance
                )
            }
        """
    ] ;

            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:clampedHullGeometryTerm ?resultNode .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .
                ?breadthNode qudt:numericValue ?breadth .
                ?resultNode qudt:numericValue ?actual .

                BIND(
                    ((xsd:double(?length) * xsd:double(?draught)) /
                     (xsd:double(?breadth) * xsd:double(?breadth))) AS ?ratio
                )
                BIND(?ratio * ?ratio * ?ratio AS ?rawValue)
                BIND(
                    IF(?rawValue < 5.0,
                       5.0,
                       IF(?rawValue > 20.0, 20.0, ?rawValue))
                    AS ?expected
                )
                BIND(ABS(?expected) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000001,
                        ?relativeTolerance,
                        0.000000001) AS ?tolerance)

                FILTER (
                    xsd:double(?breadth) = 0.0 ||
                    !BOUND(?expected) ||
                    ABS(xsd:double(?actual) - ?expected) > ?tolerance
                )
            }
        """
