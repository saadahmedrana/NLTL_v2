        sh:message "Flare angle must equal atan(tan(bow rake at quarter breadth) / sin(waterline angle at quarter breadth)); the sine denominator must be non-zero." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
SELECT $this
WHERE {
    $this nltl:bowRakeAtQuarterBreadth ?phiQuantity ;
          nltl:waterlineAngleAtQuarterBreadth ?alphaQuantity ;
          nltl:flareAngle ?psiQuantity .
    ?phiQuantity qudt:numericValue ?phi .
    ?alphaQuantity qudt:numericValue ?alpha .
    ?psiQuantity qudt:numericValue ?psi .
    BIND((xsd:double(?phi) * 3.141592653589793 / 180.0) AS ?phiRadians)
    BIND((xsd:double(?alpha) * 3.141592653589793 / 180.0) AS ?alphaRadians)
    BIND(math:sin(?alphaRadians) AS ?sinAlpha)
    BIND(
        math:atan(math:tan(?phiRadians) / ?sinAlpha)
        * 180.0 / 3.141592653589793
        AS ?expectedPsi
    )
    BIND(ABS(?expectedPsi) * 0.000001 AS ?relativeTolerance)
    BIND(
        IF(?relativeTolerance > 0.000000001,
           ?relativeTolerance,
           0.000000001)
        AS ?tolerance
    )
    FILTER(
        ABS(?sinAlpha) <= 0.000000000001 ||
        ABS(xsd:double(?psi) - ?expectedPsi) > ?tolerance
    )
}
"""
    ] ;

        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipLength ?lengthQuantity ;
          nltl:iceClassDraught ?draughtQuantity ;
          nltl:shipBreadth ?breadthQuantity ;
          nltl:clampedHullGeometryTerm ?reportedQuantity .
    ?lengthQuantity qudt:numericValue ?length .
    ?draughtQuantity qudt:numericValue ?draught .
    ?breadthQuantity qudt:numericValue ?breadth .
    ?reportedQuantity qudt:numericValue ?reported .
    BIND(xsd:double(?breadth) AS ?breadthNumber)
    BIND(
        (
            (xsd:double(?length) * xsd:double(?draught))
            / (?breadthNumber * ?breadthNumber)
        ) AS ?ratio
    )
    BIND((?ratio * ?ratio * ?ratio) AS ?rawTerm)
    BIND(
        IF(?rawTerm < 5.0, 5.0,
           IF(?rawTerm > 20.0, 20.0, ?rawTerm))
        AS ?expectedTerm
    )
    BIND(ABS(?expectedTerm) * 0.000001 AS ?relativeTolerance)
    BIND(
        IF(?relativeTolerance > 0.000000001,
           ?relativeTolerance,
           0.000000001)
        AS ?tolerance
    )
    FILTER(
        ABS(?breadthNumber) <= 0.000000000001 ||
        ABS(xsd:double(?reported) - ?expectedTerm) > ?tolerance
    )
}
"""
    ] .
