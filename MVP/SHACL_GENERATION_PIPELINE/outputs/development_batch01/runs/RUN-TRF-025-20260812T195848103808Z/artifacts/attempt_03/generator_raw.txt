            sh:sparql [
                sh:message "For an IA Super ship without a bulb, B must be nonzero, L must be positive, the C1 denominator must be nonzero, and C1 and C2 must satisfy the prescribed formulas within relative tolerance 1e-6 or absolute tolerance 1e-9." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipBreadth/qudt:numericValue ?B ;
          nltl:shipLength/qudt:numericValue ?L ;
          nltl:iceClassDraught/qudt:numericValue ?T ;
          nltl:coefficientF1/qudt:numericValue ?f1 ;
          nltl:coefficientF2/qudt:numericValue ?f2 ;
          nltl:coefficientF3/qudt:numericValue ?f3 ;
          nltl:coefficientF4/qudt:numericValue ?f4 ;
          nltl:coefficientG1/qudt:numericValue ?g1 ;
          nltl:coefficientG2/qudt:numericValue ?g2 ;
          nltl:coefficientG3/qudt:numericValue ?g3 ;
          nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?C1 ;
          nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?C2 .
    BIND ((2.0 * ?T / ?B + 1.0) AS ?c1Denominator)
    BIND ((?f1 * (?B * ?L) / ?c1Denominator) + 1.84 * (?f2 * ?B + ?f3 * ?L + ?f4 * ?B * ?L) AS ?expectedC1)
    BIND (3.52 * (?g1 + ?g2 * ?B) + ?g3 * (1.0 + 1.2 * ?T / ?B) * ?B * ?B / SQRT(?L) AS ?expectedC2)
    FILTER (
        ?B = 0.0 ||
        ?L <= 0.0 ||
        ?c1Denominator = 0.0 ||
        (
            ABS(?C1 - ?expectedC1) > 0.000001 * ABS(?expectedC1) &&
            ABS(?C1 - ?expectedC1) > 0.000000001
        ) ||
        (
            ABS(?C2 - ?expectedC2) > 0.000001 * ABS(?expectedC2) &&
            ABS(?C2 - ?expectedC2) > 0.000000001
        )
    )
}
