<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF025Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "Exactly one recognized ice class is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "IA Super ships must have exactly one boolean bulbous-bow value."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:message "IA Super ships without a bulb must record one false bulbous-bow value."
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:PositiveMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:PositiveMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:PositiveMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantity
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonQuantity
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:NewtonQuantity
            ] ;
            sh:sparql [
                sh:message "C1 and C2 must satisfy the IA Super no-bulb formulas within relative tolerance 1e-6 and absolute fallback 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipBreadth/qudt:numericValue ?B ;
                              nltl:shipLength/qudt:numericValue ?L ;
                              nltl:iceClassDraught/qudt:numericValue ?T ;
                              nltl:coefficientF1/qudt:numericValue ?f1 ;
                              nltl:coefficientF2/qudt:numericValue ?f2 ;
                              nltl:coefficientF3/qudt:numericValue ?f3 ;
                              nltl:coefficientF4/qudt:numericValue ?f4 ;
                              nltl:coefficientG1/qudt:numericValue ?g1 ;
                              nltl:coefficientG2/qudt:numericValue ?g2 ;
                              nltl:coefficientG3/qudt:numericValue ?g3 ;
                              nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?c1 ;
                              nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?c2 .

                        BIND(
                            ?f1 * (?B * ?L) / (2 * ?T / ?B + 1)
                            + 1.84 * (?f2 * ?B + ?f3 * ?L + ?f4 * ?B * ?L)
                            AS ?expectedC1
                        )

                        BIND(IF(?L >= 1, ?L, 1) AS ?s0)
                        BIND((?s0 + ?L / ?s0) / 2 AS ?s1)
                        BIND((?s1 + ?L / ?s1) / 2 AS ?s2)
                        BIND((?s2 + ?L / ?s2) / 2 AS ?s3)
                        BIND((?s3 + ?L / ?s3) / 2 AS ?s4)
                        BIND((?s4 + ?L / ?s4) / 2 AS ?s5)
                        BIND((?s5 + ?L / ?s5) / 2 AS ?s6)
                        BIND((?s6 + ?L / ?s6) / 2 AS ?s7)
                        BIND((?s7 + ?L / ?s7) / 2 AS ?s8)
                        BIND((?s8 + ?L / ?s8) / 2 AS ?s9)
                        BIND((?s9 + ?L / ?s9) / 2 AS ?s10)
                        BIND((?s10 + ?L / ?s10) / 2 AS ?s11)
                        BIND((?s11 + ?L / ?s11) / 2 AS ?s12)
                        BIND((?s12 + ?L / ?s12) / 2 AS ?s13)
                        BIND((?s13 + ?L / ?s13) / 2 AS ?s14)
                        BIND((?s14 + ?L / ?s14) / 2 AS ?sqrtL)

                        BIND(
                            3.52 * (?g1 + ?g2 * ?B)
                            + ?g3 * (1 + 1.2 * ?T / ?B) * ?B * ?B / ?sqrtL
                            AS ?expectedC2
                        )

                        BIND(
                            IF(
                                0.000001 * ABS(?expectedC1) > 0.000000001,
                                0.000001 * ABS(?expectedC1),
                                0.000000001
                            )
                            AS ?toleranceC1
                        )
                        BIND(
                            IF(
                                0.000001 * ABS(?expectedC2) > 0.000000001,
                                0.000001 * ABS(?expectedC2),
                                0.000000001
                            )
                            AS ?toleranceC2
                        )

                        FILTER(
                            ABS(?c1 - ?expectedC1) > ?toleranceC1
                            ||
                            ABS(?c2 - ?expectedC2) > ?toleranceC2
                        )
                    }
                """
            ]
        ]
    ) .

gen:PositiveMetreQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
