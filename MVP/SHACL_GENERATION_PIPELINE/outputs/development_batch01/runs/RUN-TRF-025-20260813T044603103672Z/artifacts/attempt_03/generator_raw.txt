        ) ;
        sh:message "Exactly one recognized ice class is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [

            sh:property [

        [

            sh:property [
                sh:path nltl:coefficientF1 ;

                sh:minCount 1 ;

            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;

            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:node gen:NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:node gen:NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "For an IA Super ship without a bulb, C1 and C2 must satisfy the prescribed formulas within a relative tolerance of 0.000001 and an absolute fallback of 0.000000001." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:shipBreadth/qudt:numericValue ?b ;
          nltl:shipLength/qudt:numericValue ?l ;
          nltl:iceClassDraught/qudt:numericValue ?t ;
          nltl:coefficientF1/qudt:numericValue ?f1 ;
          nltl:coefficientF2/qudt:numericValue ?f2 ;
          nltl:coefficientF3/qudt:numericValue ?f3 ;
          nltl:coefficientF4/qudt:numericValue ?f4 ;
          nltl:coefficientG1/qudt:numericValue ?g1 ;
          nltl:coefficientG2/qudt:numericValue ?g2 ;
          nltl:coefficientG3/qudt:numericValue ?g3 ;
          nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?c1 ;
          nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?c2 .

    BIND(
        ?f1 * (?b * ?l) / (2 * ?t / ?b + 1)
        + 1.84 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
        AS ?expectedC1
    )
    BIND(
        IF(
            ABS(?expectedC1) > 0.001,
            0.000001 * ABS(?expectedC1),
            0.000000001
        )
        AS ?c1Tolerance
    )

    BIND(3.52 * (?g1 + ?g2 * ?b) AS ?c2Base)
    BIND(?g3 * (1 + 1.2 * ?t / ?b) * ?b * ?b AS ?rootTerm)
    BIND(?c2 - ?c2Base AS ?c2Residual)
    BIND(
        IF(
            ABS(?c2) > 0.001,
            0.000001 * ABS(?c2),
            0.000000001
        )
        AS ?c2Tolerance
    )
    BIND(
        IF(?rootTerm > 0, ?c2Residual, -?c2Residual)
        AS ?signedResidual
    )
    BIND(ABS(?rootTerm) AS ?absoluteRootTerm)
    BIND(?signedResidual + ?c2Tolerance AS ?upperBound)
    BIND(?signedResidual - ?c2Tolerance AS ?lowerBound)

    FILTER(
        ABS(?c1 - ?expectedC1) > ?c1Tolerance
        ||
        (
            ?rootTerm = 0
            &&
            ABS(?c2Residual) > ?c2Tolerance
        )
        ||
        (
            ?rootTerm != 0
            &&
            (
                ?upperBound < 0
                ||
                ?absoluteRootTerm * ?absoluteRootTerm
                    > ?upperBound * ?upperBound * ?l
                ||
                (
                    ?lowerBound > 0
                    &&
                    ?absoluteRootTerm * ?absoluteRootTerm
                        < ?lowerBound * ?lowerBound * ?l
                )
            )
        )
    )
}
"""
            ]
        ]
    ) .
