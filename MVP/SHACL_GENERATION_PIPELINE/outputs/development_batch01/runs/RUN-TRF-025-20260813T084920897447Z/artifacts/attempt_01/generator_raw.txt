    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF025ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one ice class to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:node gen:NewtonQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:node gen:NewtonQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:node gen:NewtonPerSquareMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:node gen:NewtonPerMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:node gen:NewtonPerMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:node gen:NewtonPerSquareMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:node gen:NewtonQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:node gen:NewtonPerMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:node gen:PositiveMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:node gen:PositiveMetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:node gen:MetreQuantityValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For an IA Super ship without a bulb, C1 and C2 must satisfy the prescribed formulas within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent false ;
                      nltl:brashIceResistanceCoefficientC1 ?c1Node ;
                      nltl:brashIceResistanceCoefficientC2 ?c2Node ;
                      nltl:coefficientF1 ?f1Node ;
                      nltl:coefficientF2 ?f2Node ;
                      nltl:coefficientF3 ?f3Node ;
                      nltl:coefficientF4 ?f4Node ;
                      nltl:coefficientG1 ?g1Node ;
                      nltl:coefficientG2 ?g2Node ;
                      nltl:coefficientG3 ?g3Node ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode .

                ?c1Node qudt:numericValue ?c1 .
                ?c2Node qudt:numericValue ?c2 .
                ?f1Node qudt:numericValue ?f1 .
                ?f2Node qudt:numericValue ?f2 .
                ?f3Node qudt:numericValue ?f3 .
                ?f4Node qudt:numericValue ?f4 .
                ?g1Node qudt:numericValue ?g1 .
                ?g2Node qudt:numericValue ?g2 .
                ?g3Node qudt:numericValue ?g3 .
                ?breadthNode qudt:numericValue ?breadth .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .

                BIND((2.0 * ?draught / ?breadth) + 1.0 AS ?c1Denominator)
                BIND(
                    (?f1 * ?breadth * ?length / ?c1Denominator)
                    + 1.84 * (
                        ?f2 * ?breadth
                        + ?f3 * ?length
                        + ?f4 * ?breadth * ?length
                    )
                    AS ?expectedC1
                )
                BIND(ABS(?expectedC1) * 0.000001 AS ?relativeC1Tolerance)
                BIND(
                    IF(
                        ?relativeC1Tolerance > 0.000000001,
                        ?relativeC1Tolerance,
                        0.000000001
                    )
                    AS ?c1Tolerance
                )

                BIND(3.52 * (?g1 + ?g2 * ?breadth) AS ?c2Base)
                BIND(
                    ?g3
                    * (1.0 + 1.2 * ?draught / ?breadth)
                    * ?breadth * ?breadth
                    AS ?c2RadicalNumerator
                )
                BIND(?c2 - ?c2Base AS ?c2Difference)
                BIND(
                    ?c2Difference * ?c2Difference * ?length
                    AS ?observedSquaredC2Term
                )
                BIND(
                    ?c2RadicalNumerator * ?c2RadicalNumerator
                    AS ?expectedSquaredC2Term
                )
                BIND(
                    ABS(?expectedSquaredC2Term) * 0.000002000001
                    AS ?relativeSquaredC2Tolerance
                )
                BIND(
                    IF(
                        ?relativeSquaredC2Tolerance > 0.000000000001,
                        ?relativeSquaredC2Tolerance,
                        0.000000000001
                    )
                    AS ?squaredC2Tolerance
                )

                FILTER(
                    ?c1Denominator = 0.0
                    || ABS(?c1 - ?expectedC1) > ?c1Tolerance
                    || (?c2Difference * ?c2RadicalNumerator) < 0.0
                    || ABS(
                        ?observedSquaredC2Term - ?expectedSquaredC2Term
                    ) > ?squaredC2Tolerance
                )
            }
        """
    ] .
<END_SHACL>
