<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "An applicable IA Super ship with a bulb requires valid formula inputs, prescribed units, B != 0, L > 0, and 2*T/B+1 != 0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true .
                FILTER NOT EXISTS {
                    $this nltl:shipBreadth ?bq ;
                          nltl:shipLength ?lq ;
                          nltl:iceClassDraught ?tq ;
                          nltl:coefficientF1 ?f1q ;
                          nltl:coefficientF2 ?f2q ;
                          nltl:coefficientF3 ?f3q ;
                          nltl:coefficientF4 ?f4q ;
                          nltl:coefficientG1 ?g1q ;
                          nltl:coefficientG2 ?g2q ;
                          nltl:coefficientG3 ?g3q ;
                          nltl:brashIceResistanceCoefficientC1 ?c1q ;
                          nltl:brashIceResistanceCoefficientC2 ?c2q .

                    ?bq qudt:numericValue ?b ;
                        qudt:unit unit:M .
                    ?lq qudt:numericValue ?l ;
                        qudt:unit unit:M .
                    ?tq qudt:numericValue ?t ;
                        qudt:unit unit:M .
                    ?f1q qudt:numericValue ?f1 ;
                         qudt:unit unit:N-PER-M2 .
                    ?f2q qudt:numericValue ?f2 ;
                         qudt:unit unit:N-PER-M .
                    ?f3q qudt:numericValue ?f3 ;
                         qudt:unit unit:N-PER-M .
                    ?f4q qudt:numericValue ?f4 ;
                         qudt:unit unit:N-PER-M2 .
                    ?g1q qudt:numericValue ?g1 ;
                         qudt:unit unit:N .
                    ?g2q qudt:numericValue ?g2 ;
                         qudt:unit unit:N-PER-M .
                    ?g3q qudt:numericValue ?g3 ;
                         qudt:unit unit:N-PER-M1pt5 .
                    ?c1q qudt:numericValue ?c1 ;
                         qudt:unit unit:N .
                    ?c2q qudt:numericValue ?c2 ;
                         qudt:unit unit:N .

                    FILTER (
                        datatype(?b) = xsd:decimal &&
                        datatype(?l) = xsd:decimal &&
                        datatype(?t) = xsd:decimal &&
                        datatype(?f1) = xsd:decimal &&
                        datatype(?f2) = xsd:decimal &&
                        datatype(?f3) = xsd:decimal &&
                        datatype(?f4) = xsd:decimal &&
                        datatype(?g1) = xsd:decimal &&
                        datatype(?g2) = xsd:decimal &&
                        datatype(?g3) = xsd:decimal &&
                        datatype(?c1) = xsd:decimal &&
                        datatype(?c2) = xsd:decimal &&
                        ?b != 0.0 &&
                        ?l > 0.0 &&
                        ((2.0 * ?t / ?b) + 1.0) != 0.0
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "C1 and C2 must equal the prescribed IA Super bulbous-bow formula results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true ;
                      nltl:shipBreadth ?bq ;
                      nltl:shipLength ?lq ;
                      nltl:iceClassDraught ?tq ;
                      nltl:coefficientF1 ?f1q ;
                      nltl:coefficientF2 ?f2q ;
                      nltl:coefficientF3 ?f3q ;
                      nltl:coefficientF4 ?f4q ;
                      nltl:coefficientG1 ?g1q ;
                      nltl:coefficientG2 ?g2q ;
                      nltl:coefficientG3 ?g3q ;
                      nltl:brashIceResistanceCoefficientC1 ?c1q ;
                      nltl:brashIceResistanceCoefficientC2 ?c2q .

                ?bq qudt:numericValue ?b ;
                    qudt:unit unit:M .
                ?lq qudt:numericValue ?l ;
                    qudt:unit unit:M .
                ?tq qudt:numericValue ?t ;
                    qudt:unit unit:M .
                ?f1q qudt:numericValue ?f1 ;
                     qudt:unit unit:N-PER-M2 .
                ?f2q qudt:numericValue ?f2 ;
                     qudt:unit unit:N-PER-M .
                ?f3q qudt:numericValue ?f3 ;
                     qudt:unit unit:N-PER-M .
                ?f4q qudt:numericValue ?f4 ;
                     qudt:unit unit:N-PER-M2 .
                ?g1q qudt:numericValue ?g1 ;
                     qudt:unit unit:N .
                ?g2q qudt:numericValue ?g2 ;
                     qudt:unit unit:N-PER-M .
                ?g3q qudt:numericValue ?g3 ;
                     qudt:unit unit:N-PER-M1pt5 .
                ?c1q qudt:numericValue ?c1 ;
                     qudt:unit unit:N .
                ?c2q qudt:numericValue ?c2 ;
                     qudt:unit unit:N .

                FILTER (
                    datatype(?b) = xsd:decimal &&
                    datatype(?l) = xsd:decimal &&
                    datatype(?t) = xsd:decimal &&
                    datatype(?f1) = xsd:decimal &&
                    datatype(?f2) = xsd:decimal &&
                    datatype(?f3) = xsd:decimal &&
                    datatype(?f4) = xsd:decimal &&
                    datatype(?g1) = xsd:decimal &&
                    datatype(?g2) = xsd:decimal &&
                    datatype(?g3) = xsd:decimal &&
                    datatype(?c1) = xsd:decimal &&
                    datatype(?c2) = xsd:decimal &&
                    ?b != 0.0 &&
                    ?l > 0.0 &&
                    ((2.0 * ?t / ?b) + 1.0) != 0.0
                )

                BIND(
                    ?f1 * (?b * ?l) / ((2.0 * ?t / ?b) + 1.0) +
                    2.89 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                    AS ?expectedC1
                )
                BIND(
                    6.67 * (?g1 + ?g2 * ?b) +
                    ?g3 * (1.0 + 1.2 * ?t / ?b) * (?b * ?b) / SQRT(?l)
                    AS ?expectedC2
                )
                FILTER (?c1 != ?expectedC1 || ?c2 != ?expectedC2)
            }
        """
    ] .
<END_SHACL>
