<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "An IA Super ship must provide a boolean bulbous-bow applicability value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER NOT EXISTS {
                    $this nltl:bulbousBowPresent ?bulb .
                    FILTER (
                        datatype(?bulb) = xsd:boolean &&
                        (?bulb = true || ?bulb = false)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ship recorded as having a bulbous bow must provide its ice class to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bulbousBowPresent true .
                FILTER NOT EXISTS {
                    $this nltl:iceClass ?iceClass .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An IA Super ship with a bulbous bow must provide decimal quantity values in the required units for C1, C2, f1-f4, g1-g3, breadth, length, and ice-class draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipBreadth ?bq .
                        ?bq qudt:numericValue ?b ;
                            qudt:unit unit:M .
                        FILTER (datatype(?b) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:shipLength ?lq .
                        ?lq qudt:numericValue ?l ;
                            qudt:unit unit:M .
                        FILTER (datatype(?l) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:iceClassDraught ?tq .
                        ?tq qudt:numericValue ?t ;
                            qudt:unit unit:M .
                        FILTER (datatype(?t) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientF1 ?f1q .
                        ?f1q qudt:numericValue ?f1 ;
                             qudt:unit unit:N-PER-M2 .
                        FILTER (datatype(?f1) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientF2 ?f2q .
                        ?f2q qudt:numericValue ?f2 ;
                             qudt:unit unit:N-PER-M .
                        FILTER (datatype(?f2) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientF3 ?f3q .
                        ?f3q qudt:numericValue ?f3 ;
                             qudt:unit unit:N-PER-M .
                        FILTER (datatype(?f3) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientF4 ?f4q .
                        ?f4q qudt:numericValue ?f4 ;
                             qudt:unit unit:N-PER-M2 .
                        FILTER (datatype(?f4) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientG1 ?g1q .
                        ?g1q qudt:numericValue ?g1 ;
                             qudt:unit unit:N .
                        FILTER (datatype(?g1) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientG2 ?g2q .
                        ?g2q qudt:numericValue ?g2 ;
                             qudt:unit unit:N-PER-M .
                        FILTER (datatype(?g2) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:coefficientG3 ?g3q .
                        ?g3q qudt:numericValue ?g3 ;
                             qudt:unit nltl:newtonPerMetreToPowerOnePointFiveUnit .
                        FILTER (datatype(?g3) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:brashIceResistanceCoefficientC1 ?c1q .
                        ?c1q qudt:numericValue ?c1 ;
                             qudt:unit unit:N .
                        FILTER (datatype(?c1) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:brashIceResistanceCoefficientC2 ?c2q .
                        ?c2q qudt:numericValue ?c2 ;
                             qudt:unit unit:N .
                        FILTER (datatype(?c2) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Breadth and length must be positive and the C1 denominator must be non-zero for the required calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true ;
                      nltl:shipBreadth ?bq ;
                      nltl:shipLength ?lq ;
                      nltl:iceClassDraught ?tq .
                ?bq qudt:numericValue ?b ;
                    qudt:unit unit:M .
                ?lq qudt:numericValue ?l ;
                    qudt:unit unit:M .
                ?tq qudt:numericValue ?t ;
                    qudt:unit unit:M .
                FILTER (
                    datatype(?b) = xsd:decimal &&
                    datatype(?l) = xsd:decimal &&
                    datatype(?t) = xsd:decimal
                )
                FILTER (
                    ?b <= 0 ||
                    ?l <= 0 ||
                    ((2 * ?t / ?b) + 1) = 0
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "C1 must equal f1*(B*L)/(2*T/B+1) + 2.89*(f2*B+f3*L+f4*B*L)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true ;
                      nltl:shipBreadth ?bq ;
                      nltl:shipLength ?lq ;
                      nltl:iceClassDraught ?tq ;
                      nltl:coefficientF1 ?f1q ;
                      nltl:coefficientF2 ?f2q ;
                      nltl:coefficientF3 ?f3q ;
                      nltl:coefficientF4 ?f4q ;
                      nltl:brashIceResistanceCoefficientC1 ?c1q .
                ?bq qudt:numericValue ?b ;
                    qudt:unit unit:M .
                ?lq qudt:numericValue ?l ;
                    qudt:unit unit:M .
                ?tq qudt:numericValue ?t ;
                    qudt:unit unit:M .
                ?f1q qudt:numericValue ?f1 ;
                     qudt:unit unit:N-PER-M2 .
                ?f2q qudt:numericValue ?f2 ;
                     qudt:unit unit:N-PER-M .
                ?f3q qudt:numericValue ?f3 ;
                     qudt:unit unit:N-PER-M .
                ?f4q qudt:numericValue ?f4 ;
                     qudt:unit unit:N-PER-M2 .
                ?c1q qudt:numericValue ?c1 ;
                     qudt:unit unit:N .
                FILTER (
                    datatype(?b) = xsd:decimal &&
                    datatype(?l) = xsd:decimal &&
                    datatype(?t) = xsd:decimal &&
                    datatype(?f1) = xsd:decimal &&
                    datatype(?f2) = xsd:decimal &&
                    datatype(?f3) = xsd:decimal &&
                    datatype(?f4) = xsd:decimal &&
                    datatype(?c1) = xsd:decimal &&
                    ?b > 0 &&
                    ((2 * ?t / ?b) + 1) != 0
                )
                FILTER (
                    ?c1 != (
                        (?f1 * ?b * ?l) / ((2 * ?t / ?b) + 1) +
                        2.89 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "C2 must equal 6.67*(g1+g2*B) + g3*(1+1.2*T/B)*B^2/sqrt(L)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true ;
                      nltl:shipBreadth ?bq ;
                      nltl:shipLength ?lq ;
                      nltl:iceClassDraught ?tq ;
                      nltl:coefficientG1 ?g1q ;
                      nltl:coefficientG2 ?g2q ;
                      nltl:coefficientG3 ?g3q ;
                      nltl:brashIceResistanceCoefficientC2 ?c2q .
                ?bq qudt:numericValue ?b ;
                    qudt:unit unit:M .
                ?lq qudt:numericValue ?l ;
                    qudt:unit unit:M .
                ?tq qudt:numericValue ?t ;
                    qudt:unit unit:M .
                ?g1q qudt:numericValue ?g1 ;
                     qudt:unit unit:N .
                ?g2q qudt:numericValue ?g2 ;
                     qudt:unit unit:N-PER-M .
                ?g3q qudt:numericValue ?g3 ;
                     qudt:unit nltl:newtonPerMetreToPowerOnePointFiveUnit .
                ?c2q qudt:numericValue ?c2 ;
                     qudt:unit unit:N .
                FILTER (
                    datatype(?b) = xsd:decimal &&
                    datatype(?l) = xsd:decimal &&
                    datatype(?t) = xsd:decimal &&
                    datatype(?g1) = xsd:decimal &&
                    datatype(?g2) = xsd:decimal &&
                    datatype(?g3) = xsd:decimal &&
                    datatype(?c2) = xsd:decimal &&
                    ?b > 0 &&
                    ?l > 0
                )
                BIND (6.67 * (?g1 + ?g2 * ?b) AS ?base)
                BIND (?g3 * (1 + 1.2 * ?t / ?b) * ?b * ?b AS ?scaled)
                BIND (?c2 - ?base AS ?difference)
                FILTER (
                    (?difference * ?difference * ?l) != (?scaled * ?scaled) ||
                    (?difference * ?scaled) < 0
                )
            }
        """
    ] .
<END_SHACL>
