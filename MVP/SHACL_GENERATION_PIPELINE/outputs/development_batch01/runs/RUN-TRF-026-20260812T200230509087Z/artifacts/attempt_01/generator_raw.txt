<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:M
    ] .

gen:NewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:N
    ] .

gen:NewtonPerSquareMetreQuantityShape

    sh:property [

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:benchmarkEntity ;
    sh:or (
        [
            sh:not [
                sh:and (
                    [
                        sh:property [
                            sh:path nltl:iceClass ;
                            sh:hasValue nltl:iceClassIaSuper
                        ]
                    ]
                    [
                        sh:property [
                            sh:path nltl:bulbousBowPresent ;
                            sh:hasValue true
                        ]
                    ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape ;
                sh:message "Ship breadth B is required in metres for the C1 and C2 calculations."
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape ;
                sh:message "Ship length L is required in metres for the C1 and C2 calculations."
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape ;
                sh:message "Ice-class draught T is required in metres for the C1 and C2 calculations."
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ] ;
            sh:sparql [
                sh:message "C1 must satisfy its formula within relative tolerance 1e-6 or absolute tolerance 1e-9; C2 uses the equivalent signed squared identity with relative tolerance 1e-6 and an absolute fallback when its square-root term is zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipBreadth/qudt:numericValue ?b ;
                              nltl:shipLength/qudt:numericValue ?l ;
                              nltl:iceClassDraught/qudt:numericValue ?t ;
                              nltl:coefficientF1/qudt:numericValue ?f1 ;
                              nltl:coefficientF2/qudt:numericValue ?f2 ;
                              nltl:coefficientF3/qudt:numericValue ?f3 ;
                              nltl:coefficientF4/qudt:numericValue ?f4 ;
                              nltl:coefficientG1/qudt:numericValue ?g1 ;
                              nltl:coefficientG2/qudt:numericValue ?g2 ;
                              nltl:coefficientG3/qudt:numericValue ?g3 ;
                              nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?c1 ;
                              nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?c2 .

                        BIND(IF(?b = 0, 1, ?b) AS ?safeB)
                        BIND((2 * ?t / ?safeB + 1) AS ?denominator)
                        BIND(IF(?denominator = 0, 1, ?denominator) AS ?safeDenominator)

                        BIND(
                            ?f1 * (?b * ?l) / ?safeDenominator
                            + 2.89 * (?f2 * ?b + ?f3 * ?l + ?f4 * ?b * ?l)
                            AS ?expectedC1
                        )

                        BIND((6.67 * (?g1 + ?g2 * ?b)) AS ?c2Base)
                        BIND((?g3 * (1 + 1.2 * ?t / ?safeB) * ?b * ?b) AS ?c2Numerator)
                        BIND((?c2 - ?c2Base) AS ?c2Difference)
                        BIND((?c2Difference * ?c2Difference * ?l) AS ?c2LeftSquare)
                        BIND((?c2Numerator * ?c2Numerator) AS ?c2RightSquare)

                        FILTER(
                            ?b = 0
                            || ?l <= 0
                            || ?denominator = 0
                            || (
                                ABS(?c1 - ?expectedC1) > 0.000001 * ABS(?expectedC1)
                                && ABS(?c1 - ?expectedC1) > 0.000000001
                            )
                            || (
                                ?c2Numerator = 0
                                && ABS(?c2Difference) > 0.000000001
                            )
                            || (
                                ?c2Numerator != 0
                                && (
                                    ?c2Difference * ?c2Numerator < 0
                                    || ?c2LeftSquare < ?c2RightSquare * 0.999998000001
                                    || ?c2LeftSquare > ?c2RightSquare * 1.000002000001
                                )
                            )
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
