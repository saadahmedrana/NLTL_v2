<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PositiveMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:property [

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one ice class is required."
    ] ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:iceClass ;
                    sh:hasValue nltl:iceClassIaSuper
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:node gen:NewtonPerMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:node gen:NewtonPerMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:node gen:NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:node gen:NewtonPerMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:node gen:NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:node gen:NewtonQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "C1 and C2 must match the IA Super bulb formulas within relative tolerance 1e-6 and absolute fallback 1e-12." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:shipBreadth/qudt:numericValue ?b ;
                              nltl:shipLength/qudt:numericValue ?l ;
                              nltl:iceClassDraught/qudt:numericValue ?t ;
                              nltl:coefficientF1/qudt:numericValue ?f1 ;
                              nltl:coefficientF2/qudt:numericValue ?f2 ;
                              nltl:coefficientF3/qudt:numericValue ?f3 ;
                              nltl:coefficientF4/qudt:numericValue ?f4 ;
                              nltl:coefficientG1/qudt:numericValue ?g1 ;
                              nltl:coefficientG2/qudt:numericValue ?g2 ;
                              nltl:coefficientG3/qudt:numericValue ?g3 ;
                              nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?c1 ;
                              nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?c2 .
                        BIND(
                            (?f1 * (?b * ?l) / ((2 * ?t / ?b) + 1))
                            + (2.89 * ((?f2 * ?b) + (?f3 * ?l) + (?f4 * ?b * ?l)))
                            AS ?expectedC1
                        )
                        BIND(
                            (6.67 * (?g1 + (?g2 * ?b)))
                            + (?g3 * (1 + (1.2 * ?t / ?b)) * (?b * ?b) / SQRT(?l))
                            AS ?expectedC2
                        )
                        BIND(ABS(?expectedC1) * 0.000001 AS ?relativeC1)
                        BIND(ABS(?expectedC2) * 0.000001 AS ?relativeC2)
                        BIND(
                            IF(?relativeC1 > 0.000000000001,
                               ?relativeC1,
                               0.000000000001)
                            AS ?toleranceC1
                        )
                        BIND(
                            IF(?relativeC2 > 0.000000000001,
                               ?relativeC2,
                               0.000000000001)
                            AS ?toleranceC2
                        )
                        FILTER(
                            ABS(?c1 - ?expectedC1) > ?toleranceC1
                            || ABS(?c2 - ?expectedC2) > ?toleranceC2
                        )
                    }
                """
            ]
        ]
    ) ;
    sh:message "For ice class IA Super, bulb presence must be recorded exactly once; ships with a bulb must provide conforming C1 and C2 calculations." .
<END_SHACL>
