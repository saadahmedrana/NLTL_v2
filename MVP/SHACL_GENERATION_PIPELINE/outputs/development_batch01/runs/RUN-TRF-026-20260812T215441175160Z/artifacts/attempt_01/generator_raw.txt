<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .

gen:NewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N
    ] .

gen:NewtonPerMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N-PER-M
    ] .

gen:NewtonPerSquareMetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:N-PER-M2
    ] .

gen:NewtonPerMetrePowerOnePointFiveQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:newtonPerMetreToPowerOnePointFiveUnit
    ] .

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:not [
                sh:and (
                    [
                        sh:property [
                            sh:path nltl:iceClass ;
                            sh:hasValue nltl:iceClassIaSuper
                        ]
                    ]
                    [
                        sh:property [
                            sh:path nltl:bulbousBowPresent ;
                            sh:hasValue true
                        ]
                    ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipBreadth ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:shipLength ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:iceClassDraught ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF3 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientF4 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerSquareMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:coefficientG3 ;
                sh:minCount 1 ;
                sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC1 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ] ;
            sh:property [
                sh:path nltl:brashIceResistanceCoefficientC2 ;
                sh:minCount 1 ;
                sh:node gen:NewtonQuantityShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For an IA Super ship with a bulb, C1 and C2 must match the prescribed formulas within the greater of 1e-6 relative tolerance and 1e-9 N absolute tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:bulbousBowPresent true ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:coefficientF1 ?f1Quantity ;
                      nltl:coefficientF2 ?f2Quantity ;
                      nltl:coefficientF3 ?f3Quantity ;
                      nltl:coefficientF4 ?f4Quantity ;
                      nltl:coefficientG1 ?g1Quantity ;
                      nltl:coefficientG2 ?g2Quantity ;
                      nltl:coefficientG3 ?g3Quantity ;
                      nltl:brashIceResistanceCoefficientC1 ?c1Quantity ;
                      nltl:brashIceResistanceCoefficientC2 ?c2Quantity .

                ?breadthQuantity qudt:numericValue ?breadth ;
                                 qudt:unit unit:M .
                ?lengthQuantity qudt:numericValue ?length ;
                                qudt:unit unit:M .
                ?draughtQuantity qudt:numericValue ?draught ;
                                 qudt:unit unit:M .
                ?f1Quantity qudt:numericValue ?f1 ;
                            qudt:unit unit:N-PER-M2 .
                ?f2Quantity qudt:numericValue ?f2 ;
                            qudt:unit unit:N-PER-M .
                ?f3Quantity qudt:numericValue ?f3 ;
                            qudt:unit unit:N-PER-M .
                ?f4Quantity qudt:numericValue ?f4 ;
                            qudt:unit unit:N-PER-M2 .
                ?g1Quantity qudt:numericValue ?g1 ;
                            qudt:unit unit:N .
                ?g2Quantity qudt:numericValue ?g2 ;
                            qudt:unit unit:N-PER-M .
                ?g3Quantity qudt:numericValue ?g3 ;
                            qudt:unit nltl:newtonPerMetreToPowerOnePointFiveUnit .
                ?c1Quantity qudt:numericValue ?c1 ;
                            qudt:unit unit:N .
                ?c2Quantity qudt:numericValue ?c2 ;
                            qudt:unit unit:N .

                BIND(
                    (?f1 * (?breadth * ?length) /
                        ((2.0 * ?draught / ?breadth) + 1.0))
                    + 2.89 * (
                        (?f2 * ?breadth)
                        + (?f3 * ?length)
                        + (?f4 * ?breadth * ?length)
                    )
                    AS ?expectedC1
                )
                BIND(
                    6.67 * (?g1 + (?g2 * ?breadth))
                    + ?g3
                      * (1.0 + (1.2 * ?draught / ?breadth))
                      * ?breadth * ?breadth
                      / SQRT(?length)
                    AS ?expectedC2
                )
                FILTER(
                    (
                        ABS(?c1 - ?expectedC1) > 0.000000001
                        && ABS(?c1 - ?expectedC1)
                           > 0.000001 * ABS(?expectedC1)
                    )
                    ||
                    (
                        ABS(?c2 - ?expectedC2) > 0.000000001
                        && ABS(?c2 - ?expectedC2)
                           > 0.000001 * ABS(?expectedC2)
                    )
                )
            }
        """
    ] .
<END_SHACL>
