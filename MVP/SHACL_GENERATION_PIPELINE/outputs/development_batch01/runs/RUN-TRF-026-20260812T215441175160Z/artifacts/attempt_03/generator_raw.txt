<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF026Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:iceClass ;
                    sh:hasValue nltl:iceClassIaSuper
                ] ;
                sh:property [
                    sh:path nltl:bulbousBowPresent ;
                    sh:hasValue true
                ]
            ]
        ]
        [
            sh:node gen:TRF026CalculationShape
        ]
    ) .

gen:TRF026CalculationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:coefficientF1 ;
        sh:node gen:NewtonPerSquareMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientF2 ;
        sh:node gen:NewtonPerMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientF3 ;
        sh:node gen:NewtonPerMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientF4 ;
        sh:node gen:NewtonPerSquareMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientG1 ;
        sh:node gen:NewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:node gen:NewtonPerMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:node gen:PositiveMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipLength ;
        sh:node gen:PositiveMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:node gen:PositiveMetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceResistanceCoefficientC1 ;
        sh:node gen:NewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceResistanceCoefficientC2 ;
        sh:node gen:NewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "C1 and C2 must satisfy the IA Super bulb formulas within relative tolerance 1e-6 or absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:coefficientF1 ?f1q ;
                      nltl:coefficientF2 ?f2q ;
                      nltl:coefficientF3 ?f3q ;
                      nltl:coefficientF4 ?f4q ;
                      nltl:coefficientG1 ?g1q ;
                      nltl:coefficientG2 ?g2q ;
                      nltl:coefficientG3 ?g3q ;
                      nltl:shipBreadth ?bq ;
                      nltl:shipLength ?lq ;
                      nltl:iceClassDraught ?tq ;
                      nltl:brashIceResistanceCoefficientC1 ?c1q ;
                      nltl:brashIceResistanceCoefficientC2 ?c2q .
                ?f1q qudt:numericValue ?f1 .
                ?f2q qudt:numericValue ?f2 .
                ?f3q qudt:numericValue ?f3 .
                ?f4q qudt:numericValue ?f4 .
                ?g1q qudt:numericValue ?g1 .
                ?g2q qudt:numericValue ?g2 .
                ?g3q qudt:numericValue ?g3 .
                ?bq qudt:numericValue ?b .
                ?lq qudt:numericValue ?l .
                ?tq qudt:numericValue ?t .
                ?c1q qudt:numericValue ?c1 .
                ?c2q qudt:numericValue ?c2 .
                FILTER (
                    (
                        ABS(
                            ?c1 -
                            (
                                (?f1 * (?b * ?l) / ((2 * ?t / ?b) + 1)) +
                                (2.89 * ((?f2 * ?b) + (?f3 * ?l) + (?f4 * ?b * ?l)))
                            )
                        ) > 0.000000001
                        &&
                        ABS(
                            ?c1 -
                            (
                                (?f1 * (?b * ?l) / ((2 * ?t / ?b) + 1)) +
                                (2.89 * ((?f2 * ?b) + (?f3 * ?l) + (?f4 * ?b * ?l)))
                            )
                        ) >
                        (
                            0.000001 *
                            ABS(
                                (?f1 * (?b * ?l) / ((2 * ?t / ?b) + 1)) +
                                (2.89 * ((?f2 * ?b) + (?f3 * ?l) + (?f4 * ?b * ?l)))
                            )
                        )
                    )
                    ||
                    (
                        (
                            ?c2 - (6.67 * (?g1 + (?g2 * ?b)))
                        ) *
                        (
                            ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                        ) < 0
                    )
                    ||
                    (
                        ABS(
                            (
                                (
                                    ?c2 - (6.67 * (?g1 + (?g2 * ?b)))
                                ) *
                                (
                                    ?c2 - (6.67 * (?g1 + (?g2 * ?b)))
                                ) *
                                ?l
                            ) -
                            (
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                ) *
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                )
                            )
                        ) > 0.000000001
                        &&
                        ABS(
                            (
                                (
                                    ?c2 - (6.67 * (?g1 + (?g2 * ?b)))
                                ) *
                                (
                                    ?c2 - (6.67 * (?g1 + (?g2 * ?b)))
                                ) *
                                ?l
                            ) -
                            (
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                ) *
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                )
                            )
                        ) >
                        (
                            0.000001 *
                            ABS(
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                ) *
                                (
                                    ?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b
                                )
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:PositiveMetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive 0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
