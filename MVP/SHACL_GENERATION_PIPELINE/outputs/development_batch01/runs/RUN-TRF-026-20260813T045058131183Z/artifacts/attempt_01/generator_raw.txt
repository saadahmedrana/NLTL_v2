@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:node gen:TRF026FormulaShape
        ]
    ) .

gen:TRF026FormulaShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerSquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerSquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonPerMetrePowerOnePointFiveQuantityShape
    ] ;
    sh:property [
        sh:path nltl:brashIceResistanceCoefficientC1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:brashIceResistanceCoefficientC2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:NewtonQuantityShape
    ] ;
    sh:sparql [
        sh:message "C1 must follow the prescribed formula within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipBreadth/qudt:numericValue ?B ;
                      nltl:shipLength/qudt:numericValue ?L ;
                      nltl:iceClassDraught/qudt:numericValue ?T ;
                      nltl:coefficientF1/qudt:numericValue ?f1 ;
                      nltl:coefficientF2/qudt:numericValue ?f2 ;
                      nltl:coefficientF3/qudt:numericValue ?f3 ;
                      nltl:coefficientF4/qudt:numericValue ?f4 ;
                      nltl:brashIceResistanceCoefficientC1/qudt:numericValue ?c1 .
                BIND(IF(?B = 0, 0, ?T / ?B) AS ?draughtBreadthRatio)
                BIND(2 * ?draughtBreadthRatio + 1 AS ?denominator)
                BIND(
                    IF(
                        ?denominator = 0,
                        0,
                        ?f1 * (?B * ?L) / ?denominator
                        + 2.89 * (?f2 * ?B + ?f3 * ?L + ?f4 * ?B * ?L)
                    )
                    AS ?expectedC1
                )
                BIND(
                    IF(
                        ABS(?expectedC1) > 0.000000001,
                        ABS(?expectedC1) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(
                    ?B = 0
                    || ?denominator = 0
                    || ABS(?c1 - ?expectedC1) > ?tolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "C2 must follow the prescribed formula within a relative tolerance of 1e-6, with a 1e-9 absolute fallback when the square-root term is zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipBreadth/qudt:numericValue ?B ;
                      nltl:shipLength/qudt:numericValue ?L ;
                      nltl:iceClassDraught/qudt:numericValue ?T ;
                      nltl:coefficientG1/qudt:numericValue ?g1 ;
                      nltl:coefficientG2/qudt:numericValue ?g2 ;
                      nltl:coefficientG3/qudt:numericValue ?g3 ;
                      nltl:brashIceResistanceCoefficientC2/qudt:numericValue ?c2 .
                BIND(IF(?B = 0, 0, ?T / ?B) AS ?draughtBreadthRatio)
                BIND(6.67 * (?g1 + ?g2 * ?B) AS ?baseTerm)
                BIND(
                    ?g3 * (1 + 1.2 * ?draughtBreadthRatio) * ?B * ?B
                    AS ?squareRootNumerator
                )
                BIND(?c2 - ?baseTerm AS ?reportedSquareRootTerm)
                BIND(
                    ?reportedSquareRootTerm * ?reportedSquareRootTerm * ?L
                    AS ?reportedSquared
                )
                BIND(
                    ?squareRootNumerator * ?squareRootNumerator
                    AS ?expectedSquared
                )
                FILTER(
                    ?B = 0
                    || ?L <= 0
                    || (?reportedSquareRootTerm * ?squareRootNumerator < 0)
                    || (
                        ?expectedSquared = 0
                        && ABS(?reportedSquareRootTerm) > 0.000000001
                    )
                    || (
                        ?expectedSquared != 0
                        && (
                            ?reportedSquared < 0.999998000001 * ?expectedSquared
                            || ?reportedSquared > 1.000002000001 * ?expectedSquared
                        )
                    )
                )
            }
        """
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
