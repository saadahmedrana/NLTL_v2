<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MeterQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1
    ] .

gen:ClampedHullGeometryTermShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipLength ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:node gen:MeterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:clampedHullGeometryTerm ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:sparql [
        sh:message "Clamped hull geometry term must equal ((L*T)/B^2)^3, limited to the range 5 through 20." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipLength/qudt:numericValue ?length ;
                      nltl:iceClassDraught/qudt:numericValue ?draught ;
                      nltl:shipBreadth/qudt:numericValue ?breadth ;
                      nltl:clampedHullGeometryTerm/qudt:numericValue ?actual .
                FILTER (?breadth != "0"^^xsd:decimal)
                BIND((?length * ?draught) / (?breadth * ?breadth) AS ?ratio)
                BIND(?ratio * ?ratio * ?ratio AS ?rawTerm)
                BIND(
                    IF(
                        ?rawTerm < "5"^^xsd:decimal,
                        "5"^^xsd:decimal,
                        IF(
                            ?rawTerm > "20"^^xsd:decimal,
                            "20"^^xsd:decimal,
                            ?rawTerm
                        )
                    )
                    AS ?requiredTerm
                )
                FILTER (?actual != ?requiredTerm)
            }
        """
    ] .
<END_SHACL>
