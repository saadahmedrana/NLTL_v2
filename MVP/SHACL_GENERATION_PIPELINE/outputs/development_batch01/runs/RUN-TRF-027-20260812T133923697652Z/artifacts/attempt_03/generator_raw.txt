@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ClampedHullGeometryShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "An applicable IA Super or IA ship must have exactly one decimal ship length in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                ?iceClass skos:prefLabel ?iceClassLabel .
                FILTER (STR(?iceClassLabel) IN ("IA Super", "IA"))
                FILTER (
                    NOT EXISTS {
                        $this nltl:shipLength ?quantity .
                    }
                    ||
                    EXISTS {
                        $this nltl:shipLength ?quantity1, ?quantity2 .
                        FILTER (?quantity1 != ?quantity2)
                    }
                    ||
                    EXISTS {
                        $this nltl:shipLength ?quantity .
                        FILTER (
                            NOT EXISTS { ?quantity a qudt:QuantityValue }
                            ||
                            NOT EXISTS { ?quantity qudt:numericValue ?value }
                            ||
                            EXISTS {
                                ?quantity qudt:numericValue ?value1, ?value2 .
                                FILTER (?value1 != ?value2)
                            }
                            ||
                            EXISTS {
                                ?quantity qudt:numericValue ?value .
                                FILTER (!isLiteral(?value) || DATATYPE(?value) != xsd:decimal)
                            }
                            ||
                            NOT EXISTS { ?quantity qudt:unit unit:M }
                            ||
                            EXISTS {
                                ?quantity qudt:unit ?otherUnit .
                                FILTER (?otherUnit != unit:M)
                            }
                        )
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable IA Super or IA ship must have exactly one decimal ice-class draught in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

    ] ;
    sh:sparql [
        sh:message "An applicable IA Super or IA ship must have exactly one non-zero decimal ship breadth in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                ?iceClass skos:prefLabel ?iceClassLabel .

            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

    sh:sparql [
        sh:message "The clamped hull geometry term must equal min(max(((L*T)/(B^2))^3, 5), 20)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:clampedHullGeometryTerm ?termQuantity .
                ?iceClass skos:prefLabel ?iceClassLabel .
                FILTER (STR(?iceClassLabel) IN ("IA Super", "IA"))
                ?lengthQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?length ;
                    qudt:unit unit:M .
                ?draughtQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?draught ;
                    qudt:unit unit:M .
                ?breadthQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?breadth ;
                    qudt:unit unit:M .
                ?termQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?actual ;
                    qudt:unit unit:UNITLESS .
                FILTER (
                    DATATYPE(?length) = xsd:decimal
                    &&
                    DATATYPE(?draught) = xsd:decimal
                    &&
                    DATATYPE(?breadth) = xsd:decimal
                    &&
                    DATATYPE(?actual) = xsd:decimal
                    &&
                    ?breadth != "0"^^xsd:decimal
                )
                BIND ((?length * ?draught) / (?breadth * ?breadth) AS ?ratio)
                BIND (?ratio * ?ratio * ?ratio AS ?rawTerm)
                BIND ((?rawTerm + "5"^^xsd:decimal + ABS(?rawTerm - "5"^^xsd:decimal)) / "2"^^xsd:decimal AS ?lowerClamped)
                BIND ((?lowerClamped + "20"^^xsd:decimal - ABS(?lowerClamped - "20"^^xsd:decimal)) / "2"^^xsd:decimal AS ?expected)
                FILTER (?actual != ?expected)
            }
        """
    ] .
