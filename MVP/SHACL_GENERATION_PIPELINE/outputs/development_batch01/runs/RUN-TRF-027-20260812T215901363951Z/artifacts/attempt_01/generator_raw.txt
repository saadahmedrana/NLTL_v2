            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this

                    ?year < 2003 ||

                        ?month < 9 ||
                        (?month = 9 && ?day < 1)

                        $this nltl:coefficientC3 ?c3 .

    sh:sparql [
        sh:message "For an applicable IA Super or IA ship, the unitless hull-geometry term must equal ((L*T)/B^2)^3 clamped to 5 through 20, within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?date .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (DATATYPE(?date) = xsd:date)
                BIND (YEAR(?date) AS ?year)
                BIND (MONTH(?date) AS ?month)
                BIND (DAY(?date) AS ?day)
                FILTER (
                    ?year < 2003 ||
                    (?year = 2003 && (
                        ?month < 9 ||
                        (?month = 9 && ?day < 1)
                    ))
                )
                FILTER NOT EXISTS {
                    $this nltl:shipLength ?lengthQuantity ;
                          nltl:iceClassDraught ?draughtQuantity ;
                          nltl:shipBreadth ?breadthQuantity ;
                          nltl:clampedHullGeometryTerm ?termQuantity .
                    ?lengthQuantity a qudt:QuantityValue ;
                                    qudt:numericValue ?length ;
                                    qudt:unit unit:M .
                    ?draughtQuantity a qudt:QuantityValue ;
                                     qudt:numericValue ?draught ;
                                     qudt:unit unit:M .
                    ?breadthQuantity a qudt:QuantityValue ;
                                     qudt:numericValue ?breadth ;
                                     qudt:unit unit:M .
                    ?termQuantity a qudt:QuantityValue ;
                                  qudt:numericValue ?actualTerm ;
                                  qudt:unit unit:UNITLESS .
                    FILTER (
                        DATATYPE(?length) = xsd:decimal &&
                        DATATYPE(?draught) = xsd:decimal &&
                        DATATYPE(?breadth) = xsd:decimal &&
                        DATATYPE(?actualTerm) = xsd:decimal &&
                        ?breadth != "0"^^xsd:decimal
                    )
                    BIND (
                        (?length * ?draught) /
                        (?breadth * ?breadth)
                        AS ?ratio
                    )
                    BIND (?ratio * ?ratio * ?ratio AS ?rawTerm)
                    BIND (
                        IF(
                            ?rawTerm < "5"^^xsd:decimal,
                            "5"^^xsd:decimal,
                            IF(
                                ?rawTerm > "20"^^xsd:decimal,
                                "20"^^xsd:decimal,
                                ?rawTerm
                            )
                        )
                        AS ?expectedTerm
                    )
                    BIND (
                        ABS(?expectedTerm) * "0.000001"^^xsd:decimal
                        AS ?relativeTolerance
                    )
                    BIND (
                        IF(
                            ?relativeTolerance > "0.000000001"^^xsd:decimal,
                            ?relativeTolerance,
                            "0.000000001"^^xsd:decimal
                        )
                        AS ?tolerance
                    )
                    FILTER (ABS(?actualTerm - ?expectedTerm) <= ?tolerance)
                }
            }
        """
    ] .
