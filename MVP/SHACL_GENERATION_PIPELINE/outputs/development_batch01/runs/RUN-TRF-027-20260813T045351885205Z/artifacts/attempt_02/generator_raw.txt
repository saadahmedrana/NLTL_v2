    sh:property [
        sh:path nltl:coefficientG1 ;

    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:minCount 1 ;

        sh:node gen:CoefficientG2ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientG3ValueShape
    ] ;
    sh:sparql [
        sh:message "The clamped hull geometry term must equal ((L*T)/(B*B))^3 clamped to the range 5 through 20, and breadth must be non-zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:clampedHullGeometryTerm ?actualQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                ?draughtQuantity qudt:numericValue ?draught .
                ?breadthQuantity qudt:numericValue ?breadth .
                ?actualQuantity qudt:numericValue ?actual .
                BIND((?length * ?draught) / (?breadth * ?breadth) AS ?r)
                BIND(?r * ?r * ?r AS ?raw)
                BIND(IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw)) AS ?expected)
                BIND(ABS(?actual - ?expected) AS ?difference)
                FILTER(
                    ?breadth = 0 ||
                    (
                        ?difference > 0.000000001 &&
                        ?difference > (0.000001 * ABS(?expected))
                    )
                )
            }
        """
    ] .
