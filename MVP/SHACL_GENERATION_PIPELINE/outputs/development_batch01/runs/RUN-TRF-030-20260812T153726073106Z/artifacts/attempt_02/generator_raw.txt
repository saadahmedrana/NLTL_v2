<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:message "Direct-analysis evidence must be linked through typed ship components."
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:hasDirectAnalysisCase ) ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:TRF030DirectAnalysisCaseShape ;
        sh:minCount 1 ;
        sh:message "At least one conforming typed direct-analysis case is required."
    ] ;
    sh:sparql [
        sh:message "A typed direct-analysis case centred at the upper ice waterline is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component .
                    ?component a nltl:shipComponent ;
                               nltl:hasDirectAnalysisCase ?case .
                    ?case a nltl:directAnalysisCase ;
                          nltl:verticalLoadPositionType nltl:upperIceWaterlineCenteredPosition .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A typed direct-analysis case positioned half a load height below the lower ice waterline is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component .
                    ?component a nltl:shipComponent ;
                               nltl:hasDirectAnalysisCase ?case .
                    ?case a nltl:directAnalysisCase ;
                          nltl:verticalLoadPositionType nltl:halfLoadHeightBelowLowerIceWaterlinePosition .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two typed intermediate direct-analysis cases at distinct vertical positions are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component1 .
                    ?component1 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case1 .
                    ?case1 a nltl:directAnalysisCase ;
                           nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionNode1 .
                    ?positionNode1 qudt:numericValue ?position1 ;
                                   qudt:unit unit:M .

                    $this nltl:hasComponent ?component2 .
                    ?component2 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case2 .
                    ?case2 a nltl:directAnalysisCase ;
                           nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionNode2 .
                    ?positionNode2 qudt:numericValue ?position2 ;
                                   qudt:unit unit:M .

                    FILTER (?case1 != ?case2 && ?position1 != ?position2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A typed direct-analysis case centred at the mid-span or spacing is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component .
                    ?component a nltl:shipComponent ;
                               nltl:hasDirectAnalysisCase ?case .
                    ?case a nltl:directAnalysisCase ;
                          nltl:horizontalLoadPositionType nltl:midSpanOrSpacingCenteredPosition .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two typed direct-analysis cases at distinct horizontal positions are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component1 .
                    ?component1 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case1 .
                    ?case1 a nltl:directAnalysisCase ;
                           nltl:horizontalLoadPosition ?positionNode1 .
                    ?positionNode1 qudt:numericValue ?position1 ;
                                   qudt:unit unit:M .

                    $this nltl:hasComponent ?component2 .
                    ?component2 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case2 .
                    ?case2 a nltl:directAnalysisCase ;
                           nltl:horizontalLoadPosition ?positionNode2 .
                    ?positionNode2 qudt:numericValue ?position2 ;
                                   qudt:unit unit:M .

                    FILTER (?case1 != ?case2 && ?position1 != ?position2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "When load length cannot be determined from the arrangement, at least two distinct load lengths with corresponding area factors are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:hasComponent ?triggerComponent .
                    ?triggerComponent a nltl:shipComponent ;
                                      nltl:hasDirectAnalysisCase ?triggerCase .
                    ?triggerCase a nltl:directAnalysisCase ;
                                 nltl:loadLengthDeterminedFromArrangement false .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?component1 .
                    ?component1 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case1 .
                    ?case1 a nltl:directAnalysisCase ;
                           nltl:loadLengthDeterminedFromArrangement false ;
                           nltl:loadPatchLength ?lengthNode1 ;
                           nltl:iceLoadAreaFactorCa ?factorNode1 .
                    ?lengthNode1 qudt:numericValue ?length1 ;
                                 qudt:unit unit:M .
                    ?factorNode1 qudt:numericValue ?factor1 ;
                                 qudt:unit unit:UNITLESS .

                    $this nltl:hasComponent ?component2 .
                    ?component2 a nltl:shipComponent ;
                                nltl:hasDirectAnalysisCase ?case2 .
                    ?case2 a nltl:directAnalysisCase ;
                           nltl:loadLengthDeterminedFromArrangement false ;
                           nltl:loadPatchLength ?lengthNode2 ;
                           nltl:iceLoadAreaFactorCa ?factorNode2 .
                    ?lengthNode2 qudt:numericValue ?length2 ;
                                 qudt:unit unit:M .
                    ?factorNode2 qudt:numericValue ?factor2 ;
                                 qudt:unit unit:UNITLESS .

                    FILTER (?case1 != ?case2 && ?length1 != ?length2)
                }
            }
        """
    ] .

gen:TRF030DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030PressureValueShape
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030PressureValueShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030UnitlessValueShape
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF030LengthValueShape
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times the section 4.2.2 ice pressure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:appliedIcePressure ?appliedNode .
                ?pressureNode qudt:numericValue ?pressure ;
                              qudt:unit unit:MegaPA .
                ?appliedNode qudt:numericValue ?applied ;
                             qudt:unit unit:MegaPA .
                FILTER (?applied != (xsd:decimal("1.8") * ?pressure))
            }
        """
    ] .

gen:TRF030PressureValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF030LengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF030UnitlessValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
