<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseShape a sh:NodeShape ;
    sh:targetClass nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:spacing ;
                sh:node gen:MetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:span ;
                sh:node gen:MetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:iceLoadAreaFactorCa ;
                sh:node gen:UnitlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times the ice pressure within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:appliedIcePressure ?appliedQuantity .
                ?pressureQuantity qudt:numericValue ?pressure .
                ?appliedQuantity qudt:numericValue ?applied .
                FILTER (
                    ABS(?applied - (1.8 * ?pressure)) >
                    IF(
                        ABS(1.8 * ?pressure) * 0.000001 > 0.000000001,
                        ABS(1.8 * ?pressure) * 0.000001,
                        0.000000001
                    )
                )
            }
        """
    ] .

gen:ShipDirectAnalysisShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:DirectAnalysisCaseShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ]
        ] ;
        sh:qualifiedMinCount 2
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:midSpanOrSpacingCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:otherHorizontalPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:sparql [
        sh:message "When load length cannot be determined from the arrangement, at least two distinct checked load-patch lengths with corresponding area factors are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasDirectAnalysisCase ?triggerCase .
                ?triggerCase nltl:loadLengthDeterminedFromArrangement false .
                OPTIONAL {
                    $this nltl:hasDirectAnalysisCase ?lengthCase .
                    ?lengthCase nltl:loadLengthDeterminedFromArrangement false ;
                                nltl:loadPatchLength ?lengthQuantity ;
                                nltl:iceLoadAreaFactorCa ?areaFactorQuantity .
                    ?lengthQuantity qudt:numericValue ?lengthValue ;
                                    qudt:unit unit:M .
                    ?areaFactorQuantity qudt:numericValue ?areaFactorValue ;
                                        qudt:unit unit:UNITLESS .
                }
            }
            GROUP BY $this
            HAVING (COUNT(DISTINCT ?lengthValue) < 2)
        """
    ] .
<END_SHACL>
