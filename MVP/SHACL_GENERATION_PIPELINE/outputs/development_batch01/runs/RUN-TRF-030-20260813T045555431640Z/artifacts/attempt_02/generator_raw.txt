<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MeterQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must have one load-patch length la in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineReferencePosition ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterlineReferencePosition ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The case must confirm that the load position minimizes structural capacity."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The case must evaluate the combined effects of bending and shear."
    ] .

gen:DirectAnalysisCaseWithCaShape
    a sh:NodeShape ;
    sh:node gen:DirectAnalysisCaseShape ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each alternative load-patch length must be paired with one corresponding ca value."
    ] .

gen:TRF030Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "One ice pressure p in megapascals is required."
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "One applied ice pressure in megapascals is required."
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "One load-patch height h in metres is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "It must be stated whether la is determined directly from the structural arrangement."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:DirectAnalysisCaseShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one case must be centred at the upper ice waterline."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one case must be positioned half a load height below the lower ice waterline."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ]
        ] ;
        sh:qualifiedMinCount 2 ;
        sh:message "At least two intermediate vertical analysis cases are required."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:midSpanOrSpacingCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one case must be centred at the mid-span or mid-spacing."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:otherHorizontalPosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one additional horizontal load position is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:node gen:DirectAnalysisCaseWithCaShape ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times p within relative tolerance 1e-6 and absolute fallback 1e-12 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pQuantity ;
                      nltl:appliedIcePressure ?appliedQuantity .
                ?pQuantity qudt:numericValue ?p .
                ?appliedQuantity qudt:numericValue ?applied .
                BIND((xsd:decimal("1.8") * xsd:decimal(?p)) AS ?expected)
                BIND(ABS(xsd:decimal(?applied) - ?expected) AS ?difference)
                FILTER(
                    ?difference >
                    ((xsd:decimal("0.000001") * ABS(?expected)) +
                     xsd:decimal("0.000000000001"))
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Vertical load positions must correspond to UIWL, half a load height below LIWL, or lie strictly between those limits." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:loadPatchHeight ?heightQuantity ;
                      nltl:hasDirectAnalysisCase ?case .
                ?heightQuantity qudt:numericValue ?height .
                ?case nltl:verticalLoadPositionType ?positionType ;
                      nltl:verticalLoadPosition ?positionQuantity ;
                      nltl:upperIceWaterlineReferencePosition ?upperQuantity ;
                      nltl:lowerIceWaterlineReferencePosition ?lowerQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?upperQuantity qudt:numericValue ?upper .
                ?lowerQuantity qudt:numericValue ?lower .
                BIND(
                    xsd:decimal(?lower) -
                    (xsd:decimal("0.5") * xsd:decimal(?height))
                    AS ?lowerLoadPosition
                )
                BIND(
                    (xsd:decimal("0.000001") * ABS(xsd:decimal(?upper))) +
                    xsd:decimal("0.000000000001")
                    AS ?upperTolerance
                )
                BIND(
                    (xsd:decimal("0.000001") * ABS(?lowerLoadPosition)) +
                    xsd:decimal("0.000000000001")
                    AS ?lowerTolerance
                )
                FILTER(
                    (?positionType = nltl:upperIceWaterlineCenteredPosition &&
                     ABS(xsd:decimal(?position) - xsd:decimal(?upper)) > ?upperTolerance)
                    ||
                    (?positionType = nltl:halfLoadHeightBelowLowerIceWaterlinePosition &&
                     ABS(xsd:decimal(?position) - ?lowerLoadPosition) > ?lowerTolerance)
                    ||
                    (?positionType = nltl:intermediateVerticalPosition &&
                     !(xsd:decimal(?position) > ?lowerLoadPosition &&
                       xsd:decimal(?position) < xsd:decimal(?upper)))
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Intermediate vertical checks must use at least two distinct vertical positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1 ;
                          nltl:hasDirectAnalysisCase ?case2 .
                    ?case1 nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionQuantity1 .
                    ?case2 nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionQuantity2 .
                    ?positionQuantity1 qudt:numericValue ?position1 .
                    ?positionQuantity2 qudt:numericValue ?position2 .
                    FILTER(xsd:decimal(?position1) != xsd:decimal(?position2))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Direct analyses must use at least two distinct horizontal load positions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1 ;
                          nltl:hasDirectAnalysisCase ?case2 .
                    ?case1 nltl:horizontalLoadPosition ?positionQuantity1 .
                    ?case2 nltl:horizontalLoadPosition ?positionQuantity2 .
                    ?positionQuantity1 qudt:numericValue ?position1 .
                    ?positionQuantity2 qudt:numericValue ?position2 .
                    FILTER(xsd:decimal(?position1) != xsd:decimal(?position2))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "When la is not determined from the arrangement, at least two distinct case-level la values with corresponding ca values must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:loadLengthDeterminedFromArrangement false .
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1 ;
                          nltl:hasDirectAnalysisCase ?case2 .
                    ?case1 nltl:loadPatchLength ?lengthQuantity1 ;
                           nltl:iceLoadAreaFactorCa ?caQuantity1 .
                    ?case2 nltl:loadPatchLength ?lengthQuantity2 ;
                           nltl:iceLoadAreaFactorCa ?caQuantity2 .
                    ?lengthQuantity1 qudt:numericValue ?length1 .
                    ?lengthQuantity2 qudt:numericValue ?length2 .
                    ?caQuantity1 qudt:numericValue ?ca1 .
                    ?caQuantity2 qudt:numericValue ?ca2 .
                    FILTER(xsd:decimal(?length1) != xsd:decimal(?length2))
                }
            }
        """
    ] .
<END_SHACL>
