<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The analysis case must confirm a capacity-minimizing load position."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The analysis case must evaluate combined bending and shear."
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineReferencePosition ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterlineReferencePosition ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseWithAreaFactorShape
    a sh:NodeShape ;
    sh:node gen:DirectAnalysisCaseShape ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gen:UnitlessValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each checked load length must have a corresponding ice load area factor ca."
    ] .

gen:TRF030Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gen:PressureValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:DirectAnalysisCaseShape ;
        sh:minCount 1 ;
        sh:message "At least one conforming direct analysis case is required."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "A load case centered at the upper ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "A load case positioned half a load height below the lower ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ]
        ] ;
        sh:qualifiedMinCount 2 ;
        sh:message "Several intermediate vertical load cases are required."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:midSpanOrSpacingCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "A load case centered at mid-span or mid-spacing is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:node gen:DirectAnalysisCaseWithAreaFactorShape ;
                sh:minCount 2
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times the section 4.2.2 ice pressure within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:appliedIcePressure ?appliedNode .
                ?pressureNode qudt:numericValue ?pressureRaw .
                ?appliedNode qudt:numericValue ?appliedRaw .
                BIND(xsd:decimal(?pressureRaw) AS ?pressure)
                BIND(xsd:decimal(?appliedRaw) AS ?applied)
                BIND((1.8 * ?pressure) AS ?expected)
                BIND(IF((ABS(?expected) * 0.000001) > 0.000000001,
                        (ABS(?expected) * 0.000001),
                        0.000000001) AS ?tolerance)
                FILTER(ABS(?applied - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Vertical load positions must match the specified UIWL, half-height-below-LIWL, or intermediate-location rule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:loadPatchHeight ?heightNode ;
                      nltl:hasDirectAnalysisCase ?case .
                ?heightNode qudt:numericValue ?heightRaw .
                ?case nltl:verticalLoadPositionType ?type ;
                      nltl:verticalLoadPosition ?positionNode ;
                      nltl:upperIceWaterlineReferencePosition ?upperNode ;
                      nltl:lowerIceWaterlineReferencePosition ?lowerNode .
                ?positionNode qudt:numericValue ?positionRaw .
                ?upperNode qudt:numericValue ?upperRaw .
                ?lowerNode qudt:numericValue ?lowerRaw .
                BIND(xsd:decimal(?heightRaw) AS ?height)
                BIND(xsd:decimal(?positionRaw) AS ?position)
                BIND(xsd:decimal(?upperRaw) AS ?upper)
                BIND(xsd:decimal(?lowerRaw) AS ?lower)
                BIND((?lower - (0.5 * ?height)) AS ?lowerRequired)
                BIND(IF(
                    ?type = nltl:upperIceWaterlineCenteredPosition,
                    ?upper,
                    ?lowerRequired
                ) AS ?expected)
                BIND(IF((ABS(?expected) * 0.000001) > 0.000000001,
                        (ABS(?expected) * 0.000001),
                        0.000000001) AS ?tolerance)
                FILTER(
                    (
                        ?type IN (
                            nltl:upperIceWaterlineCenteredPosition,
                            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
                        )
                        && ABS(?position - ?expected) > ?tolerance
                    )
                    ||
                    (
                        ?type = nltl:intermediateVerticalPosition
                        && !(?position > ?lowerRequired && ?position < ?upper)
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two distinct intermediate vertical load positions must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    SELECT $this (COUNT(DISTINCT ?position) AS ?positionCount)
                    WHERE {
                        $this nltl:hasDirectAnalysisCase ?case .
                        ?case nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                              nltl:verticalLoadPosition ?positionNode .
                        ?positionNode qudt:numericValue ?positionRaw .
                        BIND(xsd:decimal(?positionRaw) AS ?position)
                    }
                    GROUP BY $this
                }
                FILTER(?positionCount < 2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two distinct horizontal load positions must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    SELECT $this (COUNT(DISTINCT ?position) AS ?positionCount)
                    WHERE {
                        $this nltl:hasDirectAnalysisCase ?case .
                        ?case nltl:horizontalLoadPosition ?positionNode .
                        ?positionNode qudt:numericValue ?positionRaw .
                        BIND(xsd:decimal(?positionRaw) AS ?position)
                    }
                    GROUP BY $this
                }
                FILTER(?positionCount < 2)
            }
        """
    ] ;
    sh:sparql [
        sh:message "When load length cannot be determined from the arrangement, at least two distinct load lengths with corresponding ca values must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:loadLengthDeterminedFromArrangement ?determined .
                FILTER(?determined = false)
                {
                    SELECT $this (COUNT(DISTINCT ?length) AS ?lengthCount)
                    WHERE {
                        $this nltl:hasDirectAnalysisCase ?case .
                        ?case nltl:loadPatchLength ?lengthNode ;
                              nltl:iceLoadAreaFactorCa ?areaFactorNode .
                        ?lengthNode qudt:numericValue ?lengthRaw .
                        ?areaFactorNode qudt:numericValue ?areaFactorRaw .
                        BIND(xsd:decimal(?lengthRaw) AS ?length)
                    }
                    GROUP BY $this
                }
                FILTER(?lengthCount < 2)
            }
        """
    ] .
<END_SHACL>
