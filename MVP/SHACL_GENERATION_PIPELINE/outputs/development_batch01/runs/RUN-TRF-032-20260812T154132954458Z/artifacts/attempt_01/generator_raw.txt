<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:ScantlingSelectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Regulation and classification-society scantlings must provide comparable decimal quantity values in the same unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?component
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:hasComponent ?component .
                {
                    ?component nltl:regulationRequiredScantling ?candidate .
                }
                UNION
                {
                    ?component nltl:classificationSocietyRequiredScantling ?candidate .
                }
                UNION
                {
                    ?component nltl:selectedDesignScantling ?candidate .
                }
                FILTER NOT EXISTS {
                    ?component
                        nltl:regulationRequiredScantling ?regulationScantling ;
                        nltl:classificationSocietyRequiredScantling ?classificationScantling .
                    ?regulationScantling
                        a qudt:QuantityValue ;
                        qudt:numericValue ?regulationValue ;
                        qudt:unit ?unit .
                    ?classificationScantling
                        a qudt:QuantityValue ;
                        qudt:numericValue ?classificationValue ;
                        qudt:unit ?unit .
                    FILTER(
                        datatype(?regulationValue) = xsd:decimal &&
                        datatype(?classificationValue) = xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the regulation-required scantling is less than the classification-society-required scantling, the selected design scantling must equal the classification-society value in the same unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?component
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:hasComponent ?component .
                ?component
                    nltl:regulationRequiredScantling ?regulationScantling ;
                    nltl:classificationSocietyRequiredScantling ?classificationScantling .
                ?regulationScantling
                    a qudt:QuantityValue ;
                    qudt:numericValue ?regulationValue ;
                    qudt:unit ?unit .
                ?classificationScantling
                    a qudt:QuantityValue ;
                    qudt:numericValue ?classificationValue ;
                    qudt:unit ?unit .
                FILTER(
                    datatype(?regulationValue) = xsd:decimal &&
                    datatype(?classificationValue) = xsd:decimal &&
                    ?regulationValue < ?classificationValue
                )
                FILTER NOT EXISTS {
                    ?component nltl:selectedDesignScantling ?selectedScantling .
                    ?selectedScantling
                        a qudt:QuantityValue ;
                        qudt:numericValue ?selectedValue ;
                        qudt:unit ?unit .
                    FILTER(
                        datatype(?selectedValue) = xsd:decimal &&
                        ?selectedValue = ?classificationValue
                    )
                }
            }
        """
    ] .
<END_SHACL>
