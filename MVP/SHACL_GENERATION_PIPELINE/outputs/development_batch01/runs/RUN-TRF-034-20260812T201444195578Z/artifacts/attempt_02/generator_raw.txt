<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:EffectiveMemberCrossSectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:hasEffectiveMemberCrossSection ;
        sh:minCount 1 ;
        sh:class nltl:effectiveMemberCrossSection ;
        sh:message "An effective member cross section is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:message "Section modulus is required." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal ;
                sh:message "Section modulus requires a decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:CentiM3 ;
                sh:message "Section modulus must use cubic centimetres."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:message "Shear area is required." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal ;
                sh:message "Shear area requires a decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:message "Shear area must use square metres."
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:memberNormalToPlating ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Member orientation relative to the plating is required."
    ] ;
    sh:sparql [
        sh:message "A member not normal to the plating requires classification-society section-property calculation evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:memberNormalToPlating ?orientation .
                FILTER (?orientation = "false"^^xsd:boolean)
                FILTER NOT EXISTS {
                    $this nltl:hasClassificationSocietySectionPropertyCalculationEvidence ?evidence .
                    ?evidence a nltl:evidenceArtifact .
                }
            }
        """
    ] .
<END_SHACL>
