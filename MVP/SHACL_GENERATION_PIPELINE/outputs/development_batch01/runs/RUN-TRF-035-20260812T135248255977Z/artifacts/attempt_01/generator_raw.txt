<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:LengthInMetresShape
    a sh:NodeShape ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The length requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:message "The length requires metre as its unit."
    ] .

gs:ClassificationSocietyRuleLengthShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:minCount 1 ;
        sh:node gs:LengthInMetresShape ;
        sh:message "The ship requires a rule length expressed in metres."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:minCount 1 ;
        sh:node gs:LengthInMetresShape ;
        sh:message "The classification society rule length is required in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length must equal the classification society rule length." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER EXISTS {
        $this nltl:ruleLength ?ruleLength .
        ?ruleLength qudt:numericValue ?ruleValue ;
                    qudt:unit unit:M .
        FILTER(datatype(?ruleValue) = xsd:decimal)
    }
    FILTER EXISTS {
        $this nltl:classificationSocietyRuleLength ?societyLength .
        ?societyLength qudt:numericValue ?societyValue ;
                       qudt:unit unit:M .
        FILTER(datatype(?societyValue) = xsd:decimal)
    }
    FILTER NOT EXISTS {
        $this nltl:ruleLength ?matchingRuleLength ;
              nltl:classificationSocietyRuleLength ?matchingSocietyLength .
        ?matchingRuleLength qudt:numericValue ?matchingRuleValue ;
                            qudt:unit unit:M .
        ?matchingSocietyLength qudt:numericValue ?matchingSocietyValue ;
                               qudt:unit unit:M .
        FILTER(
            datatype(?matchingRuleValue) = xsd:decimal &&
            datatype(?matchingSocietyValue) = xsd:decimal &&
            xsd:decimal(?matchingRuleValue) = xsd:decimal(?matchingSocietyValue)
        )
    }
}
"""
    ] .
<END_SHACL>
