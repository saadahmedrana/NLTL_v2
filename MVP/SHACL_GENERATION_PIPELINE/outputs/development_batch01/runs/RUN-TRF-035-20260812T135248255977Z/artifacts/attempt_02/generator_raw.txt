<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:RuleLengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The length quantity must have exactly one unit, metre."
    ] .

gen:ClassificationSocietyRuleLengthShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:node gen:RuleLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one rule length."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:node gen:RuleLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one classification society rule length."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length must equal the classification society rule length." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:ruleLength ?ruleLength ;
          nltl:classificationSocietyRuleLength ?societyLength .
    ?ruleLength qudt:numericValue ?ruleValue .
    ?societyLength qudt:numericValue ?societyValue .
    FILTER (xsd:decimal(?ruleValue) != xsd:decimal(?societyValue))
}
"""
    ] .
<END_SHACL>
