<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IceLoadHeightShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "An applicable ice-class ship must provide a design parameter set containing decimal level-ice-thickness and design-ice-load-height quantities in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                VALUES ?applicableClass {
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                }
                $this nltl:iceClass ?applicableClass .
                FILTER NOT EXISTS {
                    $this nltl:hasIceClassDesignParameterSet ?parameterSet .
                    ?parameterSet a nltl:iceClassDesignParameterSet ;
                        nltl:levelIceThickness ?iceThickness ;
                        nltl:designIceLoadHeight ?loadHeight .
                    ?iceThickness
                        qudt:numericValue ?iceThicknessValue ;
                        qudt:unit unit:M .
                    ?loadHeight
                        qudt:numericValue ?loadHeightValue ;
                        qudt:unit unit:M .
                    FILTER (
                        DATATYPE(?iceThicknessValue) = xsd:decimal &&
                        DATATYPE(?loadHeightValue) = xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The design ice load height must be less than the corresponding level ice thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                VALUES ?applicableClass {
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                }
                $this nltl:iceClass ?applicableClass ;
                    nltl:hasIceClassDesignParameterSet ?parameterSet .
                ?parameterSet
                    nltl:levelIceThickness ?iceThickness ;
                    nltl:designIceLoadHeight ?loadHeight .
                ?iceThickness
                    qudt:numericValue ?iceThicknessValue ;
                    qudt:unit unit:M .
                ?loadHeight
                    qudt:numericValue ?loadHeightValue ;
                    qudt:unit unit:M .
                FILTER (
                    DATATYPE(?iceThicknessValue) = xsd:decimal &&
                    DATATYPE(?loadHeightValue) = xsd:decimal &&
                    ?loadHeightValue >= ?iceThicknessValue
                )
            }
        """
    ] .
<END_SHACL>
