<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IceLoadHeightTableShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "The ice-class design parameter set must contain the Table 4-1 level ice thickness and design ice load height in metres for the ship's ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?iceClass
            WHERE {
                $this nltl:iceClass ?iceClass .

                VALUES (?iceClass ?expectedHi ?expectedH) {
                    (nltl:iceClassIaSuper "1.0"^^xsd:decimal "0.35"^^xsd:decimal)
                    (nltl:iceClassIa      "0.8"^^xsd:decimal "0.30"^^xsd:decimal)
                    (nltl:iceClassIb      "0.6"^^xsd:decimal "0.25"^^xsd:decimal)
                    (nltl:iceClassIc      "0.4"^^xsd:decimal "0.22"^^xsd:decimal)
                }

                FILTER NOT EXISTS {
                    $this nltl:hasIceClassDesignParameterSet ?parameterSet .

                    ?parameterSet
                        nltl:levelIceThickness ?hiQuantity ;
                        nltl:designIceLoadHeight ?hQuantity .

                    ?hiQuantity
                        qudt:numericValue ?hiValue ;
                        qudt:unit unit:M .

                    ?hQuantity
                        qudt:numericValue ?hValue ;
                        qudt:unit unit:M .

                    FILTER (
                        datatype(?hiValue) = xsd:decimal &&
                        datatype(?hValue) = xsd:decimal &&
                        ?hiValue = ?expectedHi &&
                        ?hValue = ?expectedH
                    )
                }
            }
        """
    ] .
<END_SHACL>
