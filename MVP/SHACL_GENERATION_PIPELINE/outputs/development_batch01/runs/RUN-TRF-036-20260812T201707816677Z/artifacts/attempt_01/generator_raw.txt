<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF036HeightOfIceLoadAreaShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "The ice-class design parameter set must provide the prescribed level ice thickness and design ice load height in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )

                BIND(
                    IF(?iceClass = nltl:iceClassIaSuper, xsd:decimal("1.0"),
                    IF(?iceClass = nltl:iceClassIa, xsd:decimal("0.8"),
                    IF(?iceClass = nltl:iceClassIb, xsd:decimal("0.6"),
                                                        xsd:decimal("0.4"))))
                    AS ?expectedHi
                )

                BIND(
                    IF(?iceClass = nltl:iceClassIaSuper, xsd:decimal("0.35"),
                    IF(?iceClass = nltl:iceClassIa, xsd:decimal("0.30"),
                    IF(?iceClass = nltl:iceClassIb, xsd:decimal("0.25"),
                                                        xsd:decimal("0.22"))))
                    AS ?expectedH
                )

                FILTER NOT EXISTS {
                    $this nltl:hasIceClassDesignParameterSet ?parameterSet .

                    ?parameterSet
                        a nltl:iceClassDesignParameterSet ;
                        nltl:levelIceThickness ?hiQuantity ;
                        nltl:designIceLoadHeight ?hQuantity .

                    ?hiQuantity
                        qudt:numericValue ?hi ;
                        qudt:unit unit:M .

                    ?hQuantity
                        qudt:numericValue ?h ;
                        qudt:unit unit:M .

                    FILTER (
                        DATATYPE(?hi) = xsd:decimal &&
                        DATATYPE(?h) = xsd:decimal &&
                        ?hi = ?expectedHi &&
                        ?h = ?expectedH
                    )
                }
            }
        """
    ] .
<END_SHACL>
