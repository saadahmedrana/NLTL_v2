<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IcePressureCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:nominalIcePressureP0 ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassFactorCp ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputFactorCd ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path (
            nltl:shipSizeEngineOutputFactorCd
            qudt:numericValue
        ) ;
        sh:maxInclusive "1"^^xsd:decimal ;
        sh:message "The ship-size and engine-output factor c_d must not exceed 1."
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientA ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientB ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientK ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:continuousPropulsionPowerAvailableInIce ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Ice pressure and c_d must satisfy p = c_d*c_p*c_a*p_0 and c_d = (a*k+b)/1000 within relative tolerance 1e-6 and absolute tolerance 1e-12 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?p ;
                      nltl:nominalIcePressureP0/qudt:numericValue ?p0 ;
                      nltl:iceClassFactorCp/qudt:numericValue ?cp ;
                      nltl:iceLoadAreaFactorCa/qudt:numericValue ?ca ;
                      nltl:shipSizeEngineOutputFactorCd/qudt:numericValue ?cd ;
                      nltl:shipSizeEngineOutputCoefficientA/qudt:numericValue ?a ;
                      nltl:shipSizeEngineOutputCoefficientB/qudt:numericValue ?b ;
                      nltl:shipSizeEngineOutputCoefficientK/qudt:numericValue ?k .
                BIND((?a * ?k + ?b) / 1000.0 AS ?expectedCd)
                BIND(?cd * ?cp * ?ca * ?p0 AS ?expectedP)
                BIND(IF(ABS(?expectedCd) * 0.000001 > 0.000000000001,
                        ABS(?expectedCd) * 0.000001,
                        0.000000000001) AS ?cdTolerance)
                BIND(IF(ABS(?expectedP) * 0.000001 > 0.000000000001,
                        ABS(?expectedP) * 0.000001,
                        0.000000000001) AS ?pTolerance)
                FILTER(
                    ABS(?cd - ?expectedCd) > ?cdTolerance ||
                    ABS(?p - ?expectedP) > ?pTolerance
                )
            }
        """
    ] .
<END_SHACL>
