<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:YieldStrengthShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each graded hull component must have a valid yield strength in MPa satisfying the grade default or supported by approved material evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                {
                    $this nltl:hasComponent ?component .
                    ?component nltl:hullStructuralSteelGrade ?grade .
                    FILTER NOT EXISTS {
                        ?component nltl:yieldStrength ?yieldQuantity .
                        ?yieldQuantity qudt:numericValue ?yieldValue ;
                                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?yieldValue) = xsd:decimal)
                    }
                }
                UNION
                {
                    $this nltl:hasComponent ?component .
                    ?component nltl:hullStructuralSteelGrade ?grade ;
                               nltl:yieldStrength ?yieldQuantity .
                    FILTER NOT EXISTS {
                        ?yieldQuantity qudt:numericValue ?yieldValue ;
                                       qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?yieldValue) = xsd:decimal)
                    }
                }
                UNION
                {
                    $this nltl:hasComponent ?component .
                    ?component nltl:hullStructuralSteelGrade ?grade ;
                               nltl:yieldStrength ?yieldQuantity .
                    ?yieldQuantity qudt:numericValue ?yieldValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    FILTER (datatype(?yieldValue) = xsd:decimal)
                    FILTER (
                        !(
                            (?grade = nltl:normalStrengthHullStructuralSteel &&
                             ?yieldValue = "235"^^xsd:decimal)
                            ||
                            (?grade = nltl:highStrengthHullStructuralSteel &&
                             ?yieldValue >= "315"^^xsd:decimal)
                            ||
                            EXISTS {
                                ?component nltl:materialApprovalStatus
                                    nltl:evidenceStateApproved .
                            }
                        )
                    )
                }
            }
        """
    ] .
<END_SHACL>
