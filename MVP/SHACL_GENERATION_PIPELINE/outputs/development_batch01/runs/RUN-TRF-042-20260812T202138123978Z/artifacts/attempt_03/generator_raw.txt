<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:YieldStrengthBySteelGradeShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:hullStructuralSteelGrade ;
    sh:property [
        sh:path nltl:hullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Hull structural steel grade must be identified."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:message "Yield strength must be provided as a quantity value." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal ;
                sh:message "Yield strength must have a decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:message "Yield strength must be expressed in MPa."
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Normal-strength steel requires 235 MPa, high-strength steel requires at least 315 MPa, and another grade requires an approved actual yield value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade ?grade ;
                      nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit unit:MegaPA .
                FILTER (
                    (
                        ?grade = nltl:normalStrengthHullStructuralSteel
                        && ?yield != xsd:decimal("235")
                    )
                    ||
                    (
                        ?grade = nltl:highStrengthHullStructuralSteel
                        && ?yield < xsd:decimal("315")
                    )
                    ||
                    (
                        ?grade NOT IN (
                            nltl:normalStrengthHullStructuralSteel,
                            nltl:highStrengthHullStructuralSteel
                        )
                        && NOT EXISTS {
                            $this nltl:materialApprovalStatus
                                  nltl:evidenceStateApproved .
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
