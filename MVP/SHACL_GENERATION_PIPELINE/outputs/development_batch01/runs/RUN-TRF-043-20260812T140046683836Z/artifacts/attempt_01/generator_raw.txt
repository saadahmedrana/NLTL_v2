<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:upperBowIceBeltRequired ) ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:reachesUpperBowIceBeltTop ) ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ) ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:extensionBeyondAdjacentDeckOrTankBoundary ) ;
        sh:node gen:ExtensionBeyondBoundaryQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing for a required upper bow ice belt must reach at least the top of that belt." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?component
WHERE {
    $this nltl:hasComponent ?component .
    ?component nltl:upperBowIceBeltRequired "true"^^xsd:boolean .
    FILTER NOT EXISTS {
        ?component nltl:reachesUpperBowIceBeltTop "true"^^xsd:boolean .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary must be recorded as permitted when the extension is no more than 250 mm." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?component ?quantity
WHERE {
    $this nltl:hasComponent ?component .
    ?component nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
    ?quantity qudt:numericValue ?amount ;
              qudt:unit unit:MilliM .
    FILTER(?amount <= xsd:decimal("250"))
    FILTER NOT EXISTS {
        ?component nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted "true"^^xsd:boolean .
    }
}
"""
    ] .

gen:ExtensionBeyondBoundaryQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The extension quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:message "The extension quantity must use millimetres."
    ] .
<END_SHACL>
