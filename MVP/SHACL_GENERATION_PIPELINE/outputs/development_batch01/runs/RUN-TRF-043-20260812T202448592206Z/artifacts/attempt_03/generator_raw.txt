<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:FramingIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;

    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:not [
                    sh:hasValue true
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "Framing must reach the upper bow ice-belt top when that belt is required."
            ]
        ]
    ) ;

    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:not [
                    sh:in (
                        nltl:iceClassIaSuper
                        nltl:iceClassIa
                        nltl:iceClassIb
                        nltl:iceClassIc
                    )
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullRegion ;
                sh:minCount 1 ;
                sh:in (
                    nltl:bowRegion
                    nltl:midbodyRegion
                    nltl:sternRegion
                ) ;
                sh:message "A bow, midbody, or stern region is required for applicable framing."
            ]
        ]
    ) ;

    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue unit:M
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue unit:MilliM
            ]
        ]
    ] ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Framing extents do not meet the applicable Table 4-6 minima where the adjacent-boundary extension exceeds 250 mm." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:hullRegion ?region ;
          nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .

    ?extension qudt:numericValue ?extensionValue ;
               qudt:unit unit:MilliM .

    FILTER(datatype(?extensionValue) = xsd:decimal)
    FILTER(?extensionValue > "250"^^xsd:decimal)

    FILTER(
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        )
    )
    FILTER(
        ?region IN (
            nltl:bowRegion,
            nltl:midbodyRegion,
            nltl:sternRegion
        )
    )

    FILTER(
        !(
            ?iceClass = nltl:iceClassIaSuper &&
            ?region = nltl:bowRegion
        )
    )

    BIND(
        IF(
            ?iceClass = nltl:iceClassIaSuper,
            "1.2"^^xsd:decimal,
            "1.0"^^xsd:decimal
        ) AS ?minimumAbove
    )

    BIND(
        IF(
            ?iceClass = nltl:iceClassIaSuper &&
            ?region = nltl:midbodyRegion,
            "2.0"^^xsd:decimal,
            IF(
                ?iceClass = nltl:iceClassIaSuper &&
                ?region = nltl:sternRegion,
                "1.6"^^xsd:decimal,
                IF(
                    ?region = nltl:bowRegion,
                    "1.6"^^xsd:decimal,
                    IF(
                        ?region = nltl:midbodyRegion,
                        "1.3"^^xsd:decimal,
                        "1.0"^^xsd:decimal
                    )
                )
            )
        ) AS ?minimumBelow
    )

    FILTER(
        NOT EXISTS {
            $this nltl:framingIceStrengtheningAboveUpperIceWaterline ?aboveQuantity .
            ?aboveQuantity qudt:numericValue ?aboveValue ;
                           qudt:unit unit:M .
            FILTER(
                datatype(?aboveValue) = xsd:decimal &&
                ?aboveValue >= ?minimumAbove
            )
        }
        ||
        NOT EXISTS {
            $this nltl:framingIceStrengtheningBelowLowerIceWaterline ?belowQuantity .
            ?belowQuantity qudt:numericValue ?belowValue ;
                           qudt:unit unit:M .
            FILTER(
                datatype(?belowValue) = xsd:decimal &&
                ?belowValue >= ?minimumBelow
            )
        }
    )
}
"""
    ] .
<END_SHACL>
