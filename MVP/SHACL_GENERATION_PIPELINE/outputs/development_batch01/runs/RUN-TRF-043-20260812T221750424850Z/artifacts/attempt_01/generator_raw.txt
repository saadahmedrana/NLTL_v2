<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameVerticalIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;

    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node gen:MetreQuantity
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node gen:MetreQuantity
    ] ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node gen:MillimetreQuantity
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean
    ] ;

    sh:or (
        [ sh:not gen:IceClassIaSuperCondition ]
        gen:AboveAtLeast1_2MetreRequirement
    ) ;

    sh:or (
        [ sh:not gen:IceClassIaIbIcCondition ]
        gen:AboveAtLeast1_0MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaSuperCondition
                    gen:BowRegionCondition
                )
            ]
        ]
        gen:ReachesTankTopRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaSuperCondition
                    gen:MidbodyRegionCondition
                )
            ]
        ]
        gen:BelowAtLeast2_0MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaSuperCondition
                    gen:SternRegionCondition
                )
            ]
        ]
        gen:BelowAtLeast1_6MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaIbIcCondition
                    gen:BowRegionCondition
                )
            ]
        ]
        gen:BelowAtLeast1_6MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaIbIcCondition
                    gen:MidbodyRegionCondition
                )
            ]
        ]
        gen:BelowAtLeast1_3MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:and (
                    gen:IceClassIaIbIcCondition
                    gen:SternRegionCondition
                )
            ]
        ]
        gen:BelowAtLeast1_0MetreRequirement
    ) ;

    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:upperBowIceBeltRequired ;
                    sh:hasValue true
                ]
            ]
        ]
        gen:ReachesUpperBowIceBeltTopRequirement
    ) ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension beyond it is no more than 250 mm." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .
    ?extension qudt:numericValue ?value ;
               qudt:unit unit:MilliM .
    FILTER(?value <= "250"^^xsd:decimal)
    FILTER NOT EXISTS {
        $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
    }
}
"""
    ] .

gen:MetreQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:MillimetreQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1
    ] .

gen:MetreAtLeast1_0
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 1.0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:MetreAtLeast1_2
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 1.2
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:MetreAtLeast1_3
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 1.3
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;

    ] .

gen:MetreAtLeast1_6

    sh:property [
        sh:path qudt:numericValue ;

        sh:minCount 1 ;
        sh:minInclusive 1.6
    ] ;
    sh:property [
        sh:path qudt:unit ;

gen:MetreAtLeast2_0
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 2.0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:IceClassIaSuperCondition
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:hasValue nltl:iceClassIaSuper
    ] .

gen:IceClassIaIbIcCondition
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ]
        ]
    ) .

gen:BowRegionCondition
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:hasValue nltl:bowRegion
    ] .

gen:MidbodyRegionCondition
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:hasValue nltl:midbodyRegion
    ] .

gen:SternRegionCondition
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:hasValue nltl:sternRegion
    ] .

gen:AboveAtLeast1_0MetreRequirement
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:node gen:MetreAtLeast1_0 ;
        sh:message "Framing must extend at least 1.0 m above the upper ice waterline."
    ] .

    a sh:NodeShape ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "IA Super bow framing must reach the tank top or below the top of the floors."
    ] .

gen:ReachesUpperBowIceBeltTopRequirement
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
