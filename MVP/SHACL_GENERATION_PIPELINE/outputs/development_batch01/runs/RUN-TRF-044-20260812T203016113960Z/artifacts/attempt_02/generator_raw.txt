        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TransverseFrameSectionModulusAndShearAreaShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bulkCarrierTopWingTankFrameCondition
            nltl:singleDeckTankTopToMainDeckFrameCondition
            nltl:multiDeckOrStringerContinuousFrameCondition
            nltl:twoDeckFrameCondition
        )
    ] ;
    sh:property [
        sh:path nltl:frameBoundaryConditionFactorM0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:UnitlessDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorMt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:UnitlessDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameShearDistributionFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:UnitlessDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MegaPascalDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:effectiveFrameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MegaPascalDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:CubicCentimetreDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:SquareCentimetreDecimalQuantityShape
    ] ;
    sh:sparql [
        sh:message "The table-selected m0 and fixed f3 must be exact; mt, section modulus, and shear area must match their formulas within relative tolerance 1e-6 with an absolute fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionType ?boundaryCondition ;
                      nltl:frameBoundaryConditionFactorM0/qudt:numericValue ?m0 ;
                      nltl:frameMomentFactorMt/qudt:numericValue ?mt ;
                      nltl:frameShearDistributionFactorF3/qudt:numericValue ?f3 ;
                      nltl:icePressure/qudt:numericValue ?p ;
                      nltl:spacing/qudt:numericValue ?s ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:effectiveFrameSpan/qudt:numericValue ?l ;
                      nltl:yieldStrength/qudt:numericValue ?sigmaY ;
                      nltl:sectionModulus/qudt:numericValue ?sectionModulus ;
                      nltl:requiredShearArea/qudt:numericValue ?shearArea .

                BIND(
                    IF(
                        ?boundaryCondition = nltl:bulkCarrierTopWingTankFrameCondition,
                        "7"^^xsd:decimal,
                        IF(
                            ?boundaryCondition = nltl:singleDeckTankTopToMainDeckFrameCondition,
                            "6"^^xsd:decimal,
                            IF(
                                ?boundaryCondition = nltl:multiDeckOrStringerContinuousFrameCondition,
                                "5.7"^^xsd:decimal,
                                "5"^^xsd:decimal
                            )
                        )
                    )
                    AS ?expectedM0
                )

                BIND(IF(?l = "0"^^xsd:decimal, "1"^^xsd:decimal, ?l) AS ?safeL)
                BIND(
                    "7"^^xsd:decimal -
                    ("5"^^xsd:decimal * ?h / ?safeL)
                    AS ?mtDenominator
                )
                BIND(
                    IF(
                        ?mtDenominator = "0"^^xsd:decimal,
                        "1"^^xsd:decimal,
                        ?mtDenominator
                    )
                    AS ?safeMtDenominator
                )
                BIND(
                    "7"^^xsd:decimal * ?m0 / ?safeMtDenominator
                    AS ?expectedMt
                )

                BIND(IF(?mt = "0"^^xsd:decimal, "1"^^xsd:decimal, ?mt) AS ?safeMt)
                BIND(
                    IF(
                        ?sigmaY = "0"^^xsd:decimal,
                        "1"^^xsd:decimal,
                        ?sigmaY
                    )
                    AS ?safeSigmaY
                )

                BIND(
                    ?p * ?s * ?h * ?l /
                    (?safeMt * ?safeSigmaY) *
                    "1000000"^^xsd:decimal
                    AS ?expectedSectionModulus
                )
                BIND(
                    "1.7320508075688772"^^xsd:decimal *
                    ?f3 * ?p * ?h * ?s /
                    ("2"^^xsd:decimal * ?safeSigmaY) *
                    "10000"^^xsd:decimal
                    AS ?expectedShearArea
                )

                BIND(
                    IF(
                        ABS(?expectedMt) > "0.000000000001"^^xsd:decimal,
                        ABS(?expectedMt) * "0.000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal
                    )
                    AS ?mtTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedSectionModulus) > "0.000000000001"^^xsd:decimal,
                        ABS(?expectedSectionModulus) * "0.000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal
                    )
                    AS ?sectionModulusTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedShearArea) > "0.000000000001"^^xsd:decimal,
                        ABS(?expectedShearArea) * "0.000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal
                    )
                    AS ?shearAreaTolerance
                )

                FILTER(
                    ?l = "0"^^xsd:decimal ||
                    ?sigmaY = "0"^^xsd:decimal ||
                    ?mt = "0"^^xsd:decimal ||
                    ?mtDenominator = "0"^^xsd:decimal ||
                    ?m0 != ?expectedM0 ||
                    ?f3 != "1.2"^^xsd:decimal ||
                    ABS(?mt - ?expectedMt) > ?mtTolerance ||
                    ABS(?sectionModulus - ?expectedSectionModulus) > ?sectionModulusTolerance ||
                    ABS(?shearArea - ?expectedShearArea) > ?shearAreaTolerance
                )
            }
        """
    ] .
<END_SHACL>
