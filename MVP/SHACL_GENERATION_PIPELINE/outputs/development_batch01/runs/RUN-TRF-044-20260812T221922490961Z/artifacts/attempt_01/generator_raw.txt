            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                BIND(xsd:decimal(?pRaw) AS ?p)
                BIND(xsd:decimal(?sRaw) AS ?s)
                BIND(xsd:decimal(?hRaw) AS ?h)
                BIND(xsd:decimal(?lRaw) AS ?l)

                BIND(xsd:decimal(?zRaw) AS ?z)
                BIND(xsd:decimal(?aRaw) AS ?a)

                BIND(

                        IF(

                BIND(7.0 * ?m0 / (7.0 - 5.0 * ?h / ?l) AS ?expectedMt)
                BIND(?p * ?s * ?h * ?l / (?mt * ?sigma) * 1000000.0 AS ?expectedZ)

                    ?m0 != ?tableM0 ||
                    ?f3 != 1.2 ||
                    !BOUND(?expectedMt) ||
                    !BOUND(?expectedZ) ||
                    !BOUND(?expectedA) ||
                    (
                        ABS(?mt - ?expectedMt) >
                            0.000001 * ABS(?expectedMt) &&
                        ABS(?mt - ?expectedMt) > 0.000000001
                    ) ||
                    (
                        ABS(?z - ?expectedZ) >
                            0.000001 * ABS(?expectedZ) &&
                        ABS(?z - ?expectedZ) > 0.000000001
                    ) ||
                    (
                        ABS(?a - ?expectedA) >
                            0.000001 * ABS(?expectedA) &&
                        ABS(?a - ?expectedA) > 0.000000001
                    )
                )
            }
