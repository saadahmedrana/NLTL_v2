<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:OrdinaryFrameScantlingsShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:ordinaryFrameScantlingsUsed ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame span is required in metres when ordinary frame scantlings are used." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:ordinaryFrameScantlingsUsed true .
    FILTER NOT EXISTS {
        $this nltl:span ?spanQuantity .
        ?spanQuantity qudt:numericValue ?spanValue ;
                      qudt:unit unit:M .
        FILTER(datatype(?spanValue) = xsd:decimal)
    }
}
""" ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame span within the ice-strengthening zone is required in metres when ordinary frame scantlings are used." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:ordinaryFrameScantlingsUsed true .
    FILTER NOT EXISTS {
        $this nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
        ?zoneQuantity qudt:numericValue ?zoneValue ;
                      qudt:unit unit:M .
        FILTER(datatype(?zoneValue) = xsd:decimal)
    }
}
""" ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary frame scantlings may be used only when less than 15% of the frame span is within the ice-strengthening zone." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:ordinaryFrameScantlingsUsed true ;
          nltl:span ?spanQuantity ;
          nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
    ?spanQuantity qudt:numericValue ?spanValue ;
                  qudt:unit unit:M .
    ?zoneQuantity qudt:numericValue ?zoneValue ;
                  qudt:unit unit:M .
    FILTER(datatype(?spanValue) = xsd:decimal)
    FILTER(datatype(?zoneValue) = xsd:decimal)
    FILTER(?zoneValue >= (xsd:decimal("0.15") * ?spanValue))
}
""" ;
    ] .
<END_SHACL>
