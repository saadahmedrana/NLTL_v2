<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:AllowedAttachmentStructure
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:tankBoundaryPlating ]
        [ sh:class nltl:iceStringer ]
    ) .

gen:AllowedTerminationStructure
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:iceStringer ]
    ) .

gen:DirectAttachmentEnd
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:qualifiedValueShape gen:AllowedAttachmentStructure ;
        sh:qualifiedMinCount 1 ;
        sh:message "The upper end must be attached to a deck structure, tank boundary plating, or ice stringer."
    ] .

gen:TerminationAboveAllowedStructureEnd
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasTerminationAboveSupportingStructure ;
        sh:qualifiedValueShape gen:AllowedTerminationStructure ;
        sh:qualifiedMinCount 1 ;
        sh:message "Termination above a supporting structure must identify a deck structure or ice stringer."
    ] .

gen:ConnectingHorizontalMember
    a sh:NodeShape ;
    sh:class nltl:horizontalConnectionMember ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:qualifiedValueShape [
            sh:class nltl:mainFrame
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "The horizontal member must connect to an adjacent main frame."
    ] .

gen:HorizontalConnectionEnd
    a sh:NodeShape ;
    sh:node gen:TerminationAboveAllowedStructureEnd ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:qualifiedValueShape gen:ConnectingHorizontalMember ;
        sh:qualifiedMinCount 1 ;
        sh:message "The intermediate-frame upper end must have a horizontal member connected to an adjacent main frame."
    ] .

gen:IntermediateApplicableEnd
    a sh:NodeShape ;
    sh:or (
        [ sh:node gen:DirectAttachmentEnd ]
        [ sh:node gen:HorizontalConnectionEnd ]
    ) .

gen:MainFrameUpperEndShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DirectAttachmentEnd ;
        sh:message "A main frame must have one upper end satisfying the attachment requirement."
    ] ;
    sh:property [
        sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasUpperEnd ;
                    sh:node gen:TerminationAboveAllowedStructureEnd
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "A deck or stringer below a terminating frame must be at or above the ice-belt upper limit."
            ]
        ]
    ) ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:ordinaryFrameScantlingsUsed ;
                    sh:hasValue true
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasUpperEnd ;
                sh:node gen:TerminationAboveAllowedStructureEnd
            ] ;
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "Ordinary frame scantlings may be used only above a qualifying deck or stringer at or above the ice-belt upper limit."
            ]
        ]
    ) .

gen:IntermediateIceFrameUpperEndShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:IntermediateApplicableEnd ;
        sh:message "An intermediate ice frame must have one upper end satisfying an allowed attachment or horizontal-connection alternative."
    ] ;
    sh:property [
        sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasUpperEnd ;
                sh:node gen:DirectAttachmentEnd
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasUpperEnd ;
                sh:node gen:HorizontalConnectionEnd
            ] ;
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "The horizontal-member alternative requires a deck or stringer at or above the ice-belt upper limit."
            ] ;
            sh:property [
                sh:path nltl:sameScantlingsAsMainFrame ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "The horizontal member must have the same scantlings as the main frame."
            ]
        ]
    ) ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:hasUpperEnd ;
                    sh:node gen:TerminationAboveAllowedStructureEnd
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "A deck or stringer below a terminating frame must be at or above the ice-belt upper limit."
            ]
        ]
    ) ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:ordinaryFrameScantlingsUsed ;
                    sh:hasValue true
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasUpperEnd ;
                sh:node gen:TerminationAboveAllowedStructureEnd
            ] ;
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:minCount 1 ;
                sh:hasValue true ;
                sh:message "Ordinary frame scantlings may be used only above a qualifying deck or stringer at or above the ice-belt upper limit."
            ]
        ]
    ) .
<END_SHACL>
