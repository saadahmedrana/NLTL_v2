<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:class nltl:frame ;
    sh:class nltl:transverseFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:class nltl:frameStrengthenedPart ;
        sh:node gs:StrengthenedPartShape ;
        sh:message "A main frame shall identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:class nltl:frameEnd ;
        sh:node gs:PermittedUpperEndShape ;
        sh:message "The upper end shall be attached to a permitted supporting structure or satisfy the termination-above exception."
    ] ;
    sh:sparql [
        sh:message "Ordinary frame scantlings may be used only where the frame terminates above a deck or ice stringer at or above the ice-belt upper limit." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStrengthenedPart ?part .
    ?part nltl:ordinaryFrameScantlingsUsed true .
    FILTER NOT EXISTS {
        $this nltl:hasUpperEnd ?end .
        ?end nltl:hasTerminationAboveSupportingStructure ?support .
        ?support nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
        ?support a ?supportType .
        FILTER(?supportType IN (nltl:deckStructure, nltl:iceStringer))
    }
}
"""
    ] .

gs:IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:class nltl:frame ;
    sh:class nltl:transverseFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:class nltl:frameStrengthenedPart ;
        sh:node gs:StrengthenedPartShape ;
        sh:message "An intermediate ice frame shall identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:class nltl:frameEnd ;
        sh:node gs:IntermediateUpperEndShape ;
        sh:message "The upper end shall be attached to a permitted supporting structure or satisfy the termination-above exception."
    ] ;
    sh:sparql [
        sh:message "Ordinary frame scantlings may be used only where the frame terminates above a deck or ice stringer at or above the ice-belt upper limit." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStrengthenedPart ?part .
    ?part nltl:ordinaryFrameScantlingsUsed true .
    FILTER NOT EXISTS {
        $this nltl:hasUpperEnd ?end .
        ?end nltl:hasTerminationAboveSupportingStructure ?support .
        ?support nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
        ?support a ?supportType .
        FILTER(?supportType IN (nltl:deckStructure, nltl:iceStringer))
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A horizontal upper-end connection is permitted only for an end terminating above a deck or ice stringer at or above the ice-belt upper limit." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasUpperEnd ?end .
    ?end nltl:hasHorizontalConnectionMember ?member .
    FILTER NOT EXISTS {
        ?end nltl:hasTerminationAboveSupportingStructure ?support .
        ?support nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
        ?support a ?supportType .
        FILTER(?supportType IN (nltl:deckStructure, nltl:iceStringer))
    }
}
"""
    ] .

gs:StrengthenedPartShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean
    ] .

gs:PermittedUpperEndShape
    a sh:NodeShape ;
    sh:or (
        gs:AttachedUpperEndShape
        gs:TerminationAboveUpperLimitShape
    ) .

gs:IntermediateUpperEndShape
    a sh:NodeShape ;
    sh:or (
        gs:AttachedUpperEndShape
        gs:TerminationAboveUpperLimitShape
    ) ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:class nltl:horizontalConnectionMember ;
        sh:node gs:HorizontalConnectionMemberShape
    ] .

gs:AttachedUpperEndShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure ;
        sh:node gs:PermittedAttachedStructureShape ;
        sh:message "The upper end shall be attached to a deck, tank-boundary plating, or ice stringer."
    ] .

gs:PermittedAttachedStructureShape
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:tankBoundaryPlating ]
        [ sh:class nltl:iceStringer ]
    ) .

gs:TerminationAboveUpperLimitShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasTerminationAboveSupportingStructure ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure ;
        sh:node gs:UpperLimitSupportingStructureShape ;
        sh:message "The exception requires termination above a deck or ice stringer at or above the ice-belt upper limit."
    ] .

gs:UpperLimitSupportingStructureShape
    a sh:NodeShape ;
    sh:or (
        [ sh:class nltl:deckStructure ]
        [ sh:class nltl:iceStringer ]
    ) ;
    sh:property [
        sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] .

gs:HorizontalConnectionMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The horizontal connection member shall have the same scantlings as the main frame."
    ] ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:minCount 1 ;
        sh:class nltl:mainFrame ;
        sh:message "The horizontal member shall connect to an adjacent main frame."
    ] .
<END_SHACL>
