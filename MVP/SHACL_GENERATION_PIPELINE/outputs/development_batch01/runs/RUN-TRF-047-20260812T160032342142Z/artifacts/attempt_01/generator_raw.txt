<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TransverseFrameLowerEndShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The lower end must be attached to an accepted supporting structure or satisfy the permitted intermediate-frame connection alternative." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER NOT EXISTS {
        $this nltl:hasLowerEnd ?lowerEnd .
        ?lowerEnd nltl:hasAttachedSupportingStructure ?support .
        ?support a ?supportType .
        FILTER (
            ?supportType IN (
                nltl:deckStructure,
                nltl:tankBoundaryPlating,
                nltl:tankTop,
                nltl:iceStringer
            )
        )
    }
    FILTER NOT EXISTS {
        $this a nltl:intermediateIceFrame ;
            nltl:hasLowerEnd ?lowerEnd ;
            nltl:hasTerminationBelowSupportingStructure ?support .
        ?support a ?supportType ;
            nltl:supportingStructureAtOrBelowIceBeltLowerLimit true .
        FILTER (
            ?supportType IN (
                nltl:deckStructure,
                nltl:tankBoundaryPlating,
                nltl:tankTop,
                nltl:iceStringer
            )
        )
        ?lowerEnd nltl:hasHorizontalConnectionMember ?member .
        ?member nltl:sameScantlingsAsMainFrame true ;
            nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
        ?adjacentMainFrame a nltl:mainFrame .
    }
}
"""
    ] .

gen:MainFrameBelowIceBeltStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The main frame below the lower edge of the ice belt must be recorded as ice strengthened."
    ] .
<END_SHACL>
