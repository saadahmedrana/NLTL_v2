<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TransverseFrameLowerEndShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The strengthened lower end must use an approved direct attachment or, for an eligible intermediate ice frame, a compliant horizontal connection to two distinct strengthened adjacent main frames." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this a nltl:transverseFrame ;
          a ?frameKind .
    FILTER (?frameKind IN (nltl:mainFrame, nltl:intermediateIceFrame))

    FILTER (
        (
            EXISTS {
                $this a nltl:mainFrame .
            }
            &&
            NOT EXISTS {
                $this nltl:mainFrameBelowIceBeltStrengthened "true"^^xsd:boolean .
            }
        )
        ||
        (
            NOT EXISTS {
                $this nltl:hasStrengthenedPart ?strengthenedPart ;
                      nltl:hasLowerEnd ?lowerEnd .
                ?lowerEnd nltl:hasAttachedSupportingStructure ?attachedStructure .
                ?attachedStructure a ?attachedStructureType .
                FILTER (
                    ?attachedStructureType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    )
                )
            }
            &&
            NOT EXISTS {
                $this a nltl:intermediateIceFrame ;
                      nltl:hasStrengthenedPart ?strengthenedPart ;
                      nltl:hasLowerEnd ?lowerEnd ;
                      nltl:supportingStructureAtOrBelowIceBeltLowerLimit "true"^^xsd:boolean ;
                      nltl:sameScantlingsAsMainFrame "true"^^xsd:boolean .
                ?lowerEnd nltl:hasTerminationBelowSupportingStructure ?supportingStructure ;
                          nltl:hasHorizontalConnectionMember ?horizontalMember .
                ?supportingStructure a ?supportingStructureType .
                FILTER (
                    ?supportingStructureType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    )
                )
                ?horizontalMember
                    nltl:connectsToAdjacentMainFrame ?adjacentMainFrame1,
                                                           ?adjacentMainFrame2 .
                ?adjacentMainFrame1
                    a nltl:mainFrame ;
                    nltl:mainFrameBelowIceBeltStrengthened "true"^^xsd:boolean .
                ?adjacentMainFrame2
                    a nltl:mainFrame ;
                    nltl:mainFrameBelowIceBeltStrengthened "true"^^xsd:boolean .
                FILTER (?adjacentMainFrame1 != ?adjacentMainFrame2)
            }
        )
    )
}
""" 
    ] .
<END_SHACL>
