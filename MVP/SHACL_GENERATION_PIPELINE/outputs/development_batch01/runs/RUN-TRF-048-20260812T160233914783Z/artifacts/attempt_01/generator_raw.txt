<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:AreaCm2QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2
    ] .

gen:AreaCm3QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM3
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS
    ] .

gen:LengthMQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .

gen:PressureMPaQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA
    ] .

gen:LongitudinalFrameCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:node gen:LengthMQuantityShape
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:node gen:LengthMQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:node gen:LengthMQuantityShape
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:node gen:PressureMPaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gen:PressureMPaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameShearFactorF5 ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:node gen:AreaCm3QuantityShape
    ] ;
    sh:property [
        sh:path nltl:grossFrameShearArea ;
        sh:minCount 1 ;
        sh:node gen:AreaCm2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:actualFrameShearArea ;
        sh:minCount 1 ;
        sh:node gen:AreaCm2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:bracketArea ;
        sh:minCount 1 ;
        sh:node gen:AreaCm2QuantityShape
    ] ;
    sh:sparql [
        sh:message "Frame factors, section modulus, shear area, and bracket-area exclusion must satisfy the verified longitudinal-frame formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:designIceLoadHeight/qudt:numericValue ?h ;
                          nltl:spacing/qudt:numericValue ?s ;
                          nltl:span/qudt:numericValue ?l ;
                          nltl:icePressure/qudt:numericValue ?p ;
                          nltl:yieldStrength/qudt:numericValue ?yield ;
                          nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?f4 ;
                          nltl:longitudinalFrameShearFactorF5/qudt:numericValue ?f5 ;
                          nltl:frameMomentFactorM/qudt:numericValue ?m ;
                          nltl:sectionModulus/qudt:numericValue ?z ;
                          nltl:grossFrameShearArea/qudt:numericValue ?grossArea ;
                          nltl:actualFrameShearArea/qudt:numericValue ?actualArea ;
                          nltl:bracketArea/qudt:numericValue ?bracketArea .

                    FILTER (
                        ?s != "0"^^xsd:decimal &&
                        ?yield != "0"^^xsd:decimal &&
                        ?m != "0"^^xsd:decimal &&
                        ?f4 = (
                            "1"^^xsd:decimal -
                            ("0.2"^^xsd:decimal * ?h / ?s)
                        ) &&
                        ?f5 = "2.16"^^xsd:decimal &&
                        ?z = (
                            ?f4 * ?p * ?h * ?l * ?l *
                            "1000000"^^xsd:decimal /
                            ("4"^^xsd:decimal * ?m * ?yield)
                        ) &&
                        ?grossArea >= "0"^^xsd:decimal &&
                        (
                            ?f4 * ?f5 * ?p * ?h * ?l /
                            ("2"^^xsd:decimal * ?yield)
                        ) >= "0"^^xsd:decimal &&
                        (?grossArea * ?grossArea) = (
                            "3"^^xsd:decimal *
                            (
                                ?f4 * ?f5 * ?p * ?h * ?l *
                                "10000"^^xsd:decimal /
                                ("2"^^xsd:decimal * ?yield)
                            ) *
                            (
                                ?f4 * ?f5 * ?p * ?h * ?l *
                                "10000"^^xsd:decimal /
                                ("2"^^xsd:decimal * ?yield)
                            )
                        ) &&
                        ?actualArea = (?grossArea - ?bracketArea)
                    )
                }
            }
        """
    ] .
<END_SHACL>
