<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:AreaQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;

        sh:maxCount 1

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;

gen:LongitudinalFrameCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;

    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameShearFactorF5 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:grossFrameShearArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bracketArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualFrameShearArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasFrameBoundaryConditionEvidence ;
        sh:nodeKind sh:IRI
    ] ;

    sh:sparql [
        sh:message "The longitudinal frame load distribution factor f4 must equal 1 - 0.2*h/s." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:spacing/qudt:numericValue ?s ;
                      nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?f4 .
                FILTER (?s = 0 || ?f4 != (1 - (0.2 * ?h / ?s)))
            }
        """
    ] ;

    sh:sparql [
        sh:message "The longitudinal frame shear factor f5 must equal 2.16." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameShearFactorF5/qudt:numericValue ?f5 .
                FILTER (?f5 != 2.16)
            }
        """
    ] ;

    sh:sparql [
        sh:message "The frame moment factor m must be 13.3 unless documented significantly different boundary conditions require a positive smaller value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameMomentFactorM/qudt:numericValue ?m .
                FILTER (
                    ?m != 13.3 &&
                    (
                        ?m <= 0 ||
                        ?m >= 13.3 ||
                        NOT EXISTS {
                            $this nltl:significantlyDifferentBoundaryConditions true .
                        } ||
                        NOT EXISTS {
                            $this nltl:hasFrameBoundaryConditionEvidence ?evidence .
                        }
                    )
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Section modulus must equal f4*p*h*l^2/(4*m*yieldStrength)*10^6 cm3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?f4 ;
                      nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:span/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
                      nltl:sectionModulus/qudt:numericValue ?sectionModulus .
                FILTER (
                    ?m = 0 ||
                    ?yieldStrength = 0 ||
                    ?sectionModulus !=
                        ((?f4 * ?p * ?h * ?l * ?l) /
                         (4 * ?m * ?yieldStrength) * 1000000)
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Actual frame shear area must equal gross frame shear area minus bracket area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:grossFrameShearArea/qudt:numericValue ?grossArea ;
                      nltl:bracketArea/qudt:numericValue ?bracketArea ;
                      nltl:actualFrameShearArea/qudt:numericValue ?actualArea .
                FILTER (?actualArea != (?grossArea - ?bracketArea))
            }
        """
    ] ;

    sh:sparql [
        sh:message "Actual frame shear area must be at least sqrt(3)*f4*2.16*p*h*l/(2*yieldStrength)*10^4 cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?f4 ;
                      nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:span/qudt:numericValue ?l ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
                      nltl:actualFrameShearArea/qudt:numericValue ?actualArea .
                FILTER (
                    ?yieldStrength = 0 ||
                    ?actualArea <
                        ((1.7320508075688772 * ?f4 * 2.16 * ?p * ?h * ?l) /
                         (2 * ?yieldStrength) * 10000)
                )
            }
        """
    ] .
<END_SHACL>
