<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SectionModulusValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;

gen:LongitudinalFrameScantlingShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;

    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:PressureValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one ice pressure in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one design ice-load height in metres is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one frame spacing in metres is required."
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:LengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one frame span in metres is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:PressureValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one yield strength in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node gen:DimensionlessValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one dimensionless load-distribution factor f4 is required."
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameShearFactorF5 ;
        sh:node gen:DimensionlessValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one dimensionless shear factor f5 is required."
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DimensionlessValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one dimensionless frame moment factor m is required."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:SectionModulusValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one section modulus in cubic centimetres is required."
    ] ;
    sh:property [
        sh:path nltl:actualFrameShearArea ;
        sh:node gen:AreaValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one actual frame shear area in square centimetres is required."
    ] ;
    sh:property [
        sh:path nltl:grossFrameShearArea ;
        sh:node gen:AreaValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one gross frame shear area in square centimetres is required."
    ] ;
    sh:property [
        sh:path nltl:bracketArea ;
        sh:node gen:AreaValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one bracket area in square centimetres is required."
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The boundary-condition state must be recorded as one boolean value."
    ] ;
    sh:property [
        sh:path nltl:hasFrameBoundaryConditionEvidence ;
        sh:class nltl:frameBoundaryConditionEvidence
    ] ;

    sh:sparql [
        sh:message "The frame inputs, factors, minimum section modulus, effective shear area, boundary-condition evidence, or bracket-exclusion calculation do not satisfy clause 4.4.3." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:icePressure/qudt:numericValue ?pressure ;
          nltl:designIceLoadHeight/qudt:numericValue ?height ;
          nltl:spacing/qudt:numericValue ?spacing ;
          nltl:span/qudt:numericValue ?span ;
          nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
          nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?f4 ;
          nltl:longitudinalFrameShearFactorF5/qudt:numericValue ?f5 ;
          nltl:frameMomentFactorM/qudt:numericValue ?m ;
          nltl:sectionModulus/qudt:numericValue ?sectionModulus ;
          nltl:actualFrameShearArea/qudt:numericValue ?actualArea ;
          nltl:grossFrameShearArea/qudt:numericValue ?grossArea ;
          nltl:bracketArea/qudt:numericValue ?bracketArea ;
          nltl:significantlyDifferentBoundaryConditions ?different .

    BIND(
        "1.0"^^xsd:decimal -
        ("0.2"^^xsd:decimal * ?height / ?spacing)
        AS ?calculatedF4
    )
    BIND(
        IF(
            ABS(?calculatedF4) > "0.000000001"^^xsd:decimal,
            ABS(?calculatedF4) * "0.000001"^^xsd:decimal,
            "0.000000001"^^xsd:decimal
        )
        AS ?f4Tolerance
    )
    BIND(
        ?f4 * ?pressure * ?height * ?span * ?span /
        ("4.0"^^xsd:decimal * ?m * ?yieldStrength) *
        "1000000.0"^^xsd:decimal
        AS ?requiredSectionModulus
    )
    BIND(
        "1.7320508075688772935"^^xsd:decimal *
        ?f4 * ?f5 * ?pressure * ?height * ?span /
        ("2.0"^^xsd:decimal * ?yieldStrength) *
        "10000.0"^^xsd:decimal
        AS ?requiredShearArea
    )
    BIND(?grossArea - ?bracketArea AS ?areaWithoutBracket)
    BIND(
        IF(
            ABS(?areaWithoutBracket) > "0.000000001"^^xsd:decimal,
            ABS(?areaWithoutBracket) * "0.000001"^^xsd:decimal,
            "0.000000001"^^xsd:decimal
        )
        AS ?areaTolerance
    )

    FILTER(
        ?pressure <= "0.0"^^xsd:decimal ||
        ?height <= "0.0"^^xsd:decimal ||
        ?spacing <= "0.0"^^xsd:decimal ||
        ?span <= "0.0"^^xsd:decimal ||
        ?yieldStrength <= "0.0"^^xsd:decimal ||
        ?m <= "0.0"^^xsd:decimal ||
        ?f4 <= "0.0"^^xsd:decimal ||
        ?calculatedF4 <= "0.0"^^xsd:decimal ||
        ABS(?f4 - ?calculatedF4) > ?f4Tolerance ||
        ?f5 != "2.16"^^xsd:decimal ||
        (?different = false && ?m != "13.3"^^xsd:decimal) ||
        (
            ?different = true &&
            (
                ?m >= "13.3"^^xsd:decimal ||
                NOT EXISTS {
                    $this nltl:hasFrameBoundaryConditionEvidence ?boundaryEvidence .
                }
            )
        ) ||
        ?sectionModulus < ?requiredSectionModulus ||
        ?actualArea < ?requiredShearArea ||
        ABS(?actualArea - ?areaWithoutBracket) > ?areaTolerance
    )
}
        """
    ] .
<END_SHACL>
