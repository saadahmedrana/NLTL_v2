<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityInMillimetresShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        sh:message "Each identified supporting web frame must be linked through a frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?support
            WHERE {
                $this nltl:hasSupportingWebFrame ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:frameAttachment ?support .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each identified supporting bulkhead must be linked through a frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?support
            WHERE {
                $this nltl:hasSupportingBulkhead ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:frameAttachment ?support .
                }
            }
        """
    ] .

gen:WebPlateConnectionShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:webPlateConnectionSideCount ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:message "A recorded web plate connection must connect both sides."
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:targetClass nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:node gen:QuantityInMillimetresShape ;
        sh:message "Bracket thickness in millimetres is required for comparison with the frame web thickness."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The bracket stiffening must protect against buckling."
    ] ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasFrameAttachment
        ] ;
        sh:minCount 1 ;
        sh:class nltl:frame ;
        sh:node [
            sh:property [
                sh:path nltl:frameWebThickness ;
                sh:minCount 1 ;
                sh:node gen:QuantityInMillimetresShape
            ]
        ] ;
        sh:message "The bracket must identify an attached frame with a frame web thickness in millimetres."
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?bracketThickness ?frameThickness
            WHERE {
                ?frame nltl:hasFrameAttachment $this ;
                       nltl:frameWebThickness ?frameThickness .
                $this nltl:bracketThickness ?bracketThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                                qudt:unit unit:MilliM .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                  qudt:unit unit:MilliM .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .
<END_SHACL>
