<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:InstalledConnectionBracketShape
    a sh:NodeShape ;
    sh:targetClass nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM
            ]
        ] ;
        sh:message "An installed connection bracket must provide its thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM
            ]
        ] ;
        sh:message "An installed connection bracket must provide the frame web thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The installed bracket edge must be stiffened against buckling."
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:bracketThickness ?bracketQuantity ;
                      nltl:frameWebThickness ?frameQuantity .
                ?bracketQuantity qudt:numericValue ?bracketThickness ;
                                 qudt:unit unit:MilliM .
                ?frameQuantity qudt:numericValue ?frameWebThickness ;
                               qudt:unit unit:MilliM .
                FILTER (
                    xsd:decimal(?bracketThickness) <
                    xsd:decimal(?frameWebThickness)
                )
            }
        """
    ] .
<END_SHACL>
