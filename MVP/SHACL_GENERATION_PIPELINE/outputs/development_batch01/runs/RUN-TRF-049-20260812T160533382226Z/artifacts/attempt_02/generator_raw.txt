<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:targetClass nltl:transverseFrame ;
    sh:sparql [
        sh:message "A frame within the ice-strengthened area must have effective attachment confirmed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:effectiveAttachmentConfirmed ?confirmed .
                    FILTER (
                        datatype(?confirmed) = xsd:boolean &&
                        ?confirmed = true
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame passing through a supporting structure must have at least two connected web-plate sides." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                {
                    FILTER NOT EXISTS {
                        $this nltl:webPlateConnectionSideCount ?count .
                        FILTER (
                            datatype(?count) = xsd:integer &&
                            ?count >= 2
                        )
                    }
                }
                UNION
                {
                    $this nltl:webPlateConnectionSideCount ?count .
                    FILTER (
                        datatype(?count) != xsd:integer ||
                        ?count < 2
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Bracket and frame-web thickness values in millimetres are required to verify bracket thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                {
                    $this nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingWebFrame ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingBulkhead ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                FILTER (
                    NOT EXISTS {
                        $this nltl:frameWebThickness ?frameThickness .
                        ?frameThickness qudt:numericValue ?frameValue ;
                                        qudt:unit unit:MilliM .
                        FILTER (datatype(?frameValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?bracket nltl:bracketThickness ?bracketThickness .
                        ?bracketThickness qudt:numericValue ?bracketValue ;
                                          qudt:unit unit:MilliM .
                        FILTER (datatype(?bracketValue) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A bracket must be at least as thick as the web plate of its owning frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:frameWebThickness ?frameThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                                qudt:unit unit:MilliM .
                {
                    $this nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingWebFrame ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingBulkhead ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                  qudt:unit unit:MilliM .
                FILTER (
                    datatype(?frameValue) = xsd:decimal &&
                    datatype(?bracketValue) = xsd:decimal &&
                    ?bracketValue < ?frameValue
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each installed bracket must have its edge stiffened against buckling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                {
                    $this nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingWebFrame ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                UNION
                {
                    $this nltl:hasSupportingBulkhead ?support .
                    ?support nltl:hasConnectionBracket ?bracket .
                }
                FILTER NOT EXISTS {
                    ?bracket nltl:bracketEdgeStiffened ?stiffened .
                    FILTER (
                        datatype(?stiffened) = xsd:boolean &&
                        ?stiffened = true
                    )
                }
            }
        """
    ] .

gen:LongitudinalFrameBracketShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:sparql [
        sh:message "Each supporting web frame and bulkhead of a longitudinal frame must have a connection bracket." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                {
                    $this nltl:hasSupportingWebFrame ?support .
                }
                UNION
                {
                    $this nltl:hasSupportingBulkhead ?support .
                }
                FILTER NOT EXISTS {
                    ?support nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] .

gen:TerminatingTransverseFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have a connection bracket or a similar frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasConnectionBracket ?bracket .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                }
            }
        """
    ] .
<END_SHACL>
