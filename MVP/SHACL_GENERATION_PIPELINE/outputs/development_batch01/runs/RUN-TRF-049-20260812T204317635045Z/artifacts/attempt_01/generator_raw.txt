<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:targetClass nltl:transverseFrame ;
    sh:sparql [
        sh:message "Within the ice-strengthened area, every identified supporting web frame or bulkhead must have an effective frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:frameAttachment ?support ;
                                nltl:effectiveAttachmentConfirmed true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Within the ice-strengthened area, each identified supporting web frame or bulkhead of a longitudinal frame must have a bracketed attachment." ;
        sh:select """
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this rdf:type nltl:longitudinalFrame ;
                      nltl:withinIceStrengthenedArea true ;
                      (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:frameAttachment ?support ;
                                nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame passing through a supporting structure must record connection of at least two web-plate sides." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount ?sideCount .
                    FILTER (
                        datatype(?sideCount) = xsd:integer &&
                        ?sideCount >= 2
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An installed bracket must have a decimal thickness in millimetres, the frame must have a decimal web thickness in millimetres, and the bracket edge must be stiffened." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER (
                    NOT EXISTS {
                        $this nltl:frameWebThickness ?frameThickness .
                        ?frameThickness qudt:numericValue ?frameValue ;
                                        qudt:unit unit:MilliM .
                        FILTER (datatype(?frameValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?bracket nltl:bracketThickness ?bracketThickness .
                        ?bracketThickness qudt:numericValue ?bracketValue ;
                                          qudt:unit unit:MilliM .
                        FILTER (datatype(?bracketValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?bracket nltl:bracketEdgeStiffened true .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "An installed bracket must be at least as thick as the web plate of its frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:frameWebThickness ?frameThickness ;
                      nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?frameThickness qudt:numericValue ?frameValue ;
                                qudt:unit unit:MilliM .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                  qudt:unit unit:MilliM .
                FILTER (
                    datatype(?frameValue) = xsd:decimal &&
                    datatype(?bracketValue) = xsd:decimal &&
                    ?bracketValue < ?frameValue
                )
            }
        """
    ] .
<END_SHACL>
