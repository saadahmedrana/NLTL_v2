<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameAttachmentRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:targetClass nltl:transverseFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:node gen:ApplicableFrameShape
        ]
    ) .

gen:ApplicableFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:message "A frame passing through supporting structure must have both web-plate sides connected."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each identified supporting structure must share an effectively confirmed attachment with the frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:effectiveAttachmentConfirmed true .
                }
            }
        """
    ] .

gen:LongitudinalFrameRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "A longitudinal frame must use a shared bracketed attachment for each identified supporting web frame or bulkhead." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:withinIceStrengthenedArea true ;
                              (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
                        FILTER NOT EXISTS {
                            $this nltl:hasFrameAttachment ?attachment .
                            ?support nltl:hasSupportingStructureAttachment ?attachment .
                            ?attachment nltl:hasConnectionBracket ?bracket .
                        }
                    }
                """
            ]
        ]
    ) .

gen:TransverseFrameRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:hasFrameAttachment ;
                        sh:qualifiedValueShape [
                            sh:or (
                                [
                                    sh:property [
                                        sh:path nltl:hasConnectionBracket ;
                                        sh:minCount 1
                                    ]
                                ]
                                [
                                    sh:property [
                                        sh:path nltl:similarAttachmentConstructionPresent ;
                                        sh:datatype xsd:boolean ;
                                        sh:minCount 1 ;
                                        sh:hasValue true
                                    ]
                                ]
                            )
                        ] ;
                        sh:qualifiedMinCount 1 ;
                        sh:message "A terminating transverse frame requires a bracket or similar attachment construction."
                    ]
                ]
            )
        ]
    ) .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "Bracket thickness must be recorded in millimetres."
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasConnectionBracket ]
            [ sh:inversePath nltl:hasFrameAttachment ]
            nltl:frameWebThickness
        ) ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "The attached frame web thickness must be recorded in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket must be stiffened against buckling."
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the attached frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bracketThickness ?bracketQuantity .
                ?bracketQuantity qudt:numericValue ?bracketThickness ;
                                 qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?attachment nltl:hasConnectionBracket $this .
                ?frame nltl:hasFrameAttachment ?attachment ;
                       nltl:frameWebThickness ?webQuantity .
                ?webQuantity qudt:numericValue ?webThickness ;
                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (?bracketThickness < ?webThickness)
            }
        """
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .
<END_SHACL>
