<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameShellAttachmentShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:frameShellAttachment ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "The frame-shell attachment must use a double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:scallopingPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:hasValue false ;
                sh:in ( false )
            ]
        ]
    ) ;
    sh:message "Scalloping must be absent unless the attachment crosses a shell plate butt." .
<END_SHACL>
