<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1
    ] .

gen:MegapascalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1
    ] .

gen:FrameWebThicknessShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:frameProfileType ;
    sh:targetSubjectsOf nltl:frameWebThickness ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:in (
            nltl:profileSection
            nltl:flatBarSection
        ) ;
        sh:message "Frame profile type must identify a profile section or flat bar section."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "Frame web thickness must be provided in millimetres."
    ] ;
    sh:property [
        sh:path nltl:adjacentFrameHeight ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "Adjacent frame height must be provided in millimetres."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gen:MegapascalQuantityShape ;
        sh:message "Yield strength must be provided in megapascals."
    ] ;
    sh:property [
        sh:path nltl:netShellPlatingThickness ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "Net shell-plating thickness must be provided in millimetres."
    ] ;
    sh:sparql [
        sh:message "Frame web thickness must be at least the profile-dependent calculated value, half the net shell-plating thickness, and 9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:frameProfileType ?profile ;
                      nltl:frameWebThickness/qudt:numericValue ?webThickness ;
                      nltl:adjacentFrameHeight/qudt:numericValue ?frameHeight ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
                      nltl:netShellPlatingThickness/qudt:numericValue ?shellThickness .
                VALUES (?profile ?constant) {
                    (nltl:profileSection "805"^^xsd:decimal)
                    (nltl:flatBarSection "282"^^xsd:decimal)
                }
                FILTER (
                    ?webThickness < "9"^^xsd:decimal ||
                    ?webThickness < ("0.5"^^xsd:decimal * ?shellThickness) ||
                    ((?webThickness * ?constant) * (?webThickness * ?constant)
                        < (?frameHeight * ?frameHeight * ?yieldStrength))
                )
            }
        """
    ] .
<END_SHACL>
