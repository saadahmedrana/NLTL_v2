<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:FrameWebThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:shipComponent ;
    sh:sparql [
        sh:message "A frame requires nonnegative decimal values for web thickness, adjacent frame height, yield strength, and net shell-plating thickness in their prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                VALUES ?profileType {
                    nltl:profileSection
                    nltl:flatBarSection
                }
                $this nltl:frameProfileType ?profileType .
                FILTER (
                    NOT EXISTS {
                        $this nltl:frameWebThickness ?actualQuantity .
                        ?actualQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?actual ;
                            qudt:unit unit:MilliM .
                        FILTER(datatype(?actual) = xsd:decimal && ?actual >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:adjacentFrameHeight ?heightQuantity .
                        ?heightQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?height ;
                            qudt:unit unit:MilliM .
                        FILTER(datatype(?height) = xsd:decimal && ?height >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yieldQuantity .
                        ?yieldQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?yieldStrength ;
                            qudt:unit unit:MegaPA .
                        FILTER(datatype(?yieldStrength) = xsd:decimal && ?yieldStrength >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:netShellPlatingThickness ?shellQuantity .
                        ?shellQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?netShellThickness ;
                            qudt:unit unit:MilliM .
                        FILTER(datatype(?netShellThickness) = xsd:decimal && ?netShellThickness >= 0)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Frame web thickness must be at least the calculated value, half the net shell-plating thickness, and 9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                VALUES (?profileType ?coefficient) {
                    (nltl:profileSection "805"^^xsd:decimal)
                    (nltl:flatBarSection "282"^^xsd:decimal)
                }
                $this nltl:frameProfileType ?profileType ;
                    nltl:frameWebThickness ?actualQuantity ;
                    nltl:adjacentFrameHeight ?heightQuantity ;
                    nltl:yieldStrength ?yieldQuantity ;
                    nltl:netShellPlatingThickness ?shellQuantity .

                ?actualQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?actual ;
                    qudt:unit unit:MilliM .
                ?heightQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?height ;
                    qudt:unit unit:MilliM .
                ?yieldQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?yieldStrength ;
                    qudt:unit unit:MegaPA .
                ?shellQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?netShellThickness ;
                    qudt:unit unit:MilliM .

                FILTER(
                    datatype(?actual) = xsd:decimal &&
                    datatype(?height) = xsd:decimal &&
                    datatype(?yieldStrength) = xsd:decimal &&
                    datatype(?netShellThickness) = xsd:decimal &&
                    ?actual >= 0 &&
                    ?height >= 0 &&
                    ?yieldStrength >= 0 &&
                    ?netShellThickness >= 0
                )
                FILTER(
                    ?actual < 9 ||
                    (?actual * 2) < ?netShellThickness ||
                    (?actual * ?actual * ?coefficient * ?coefficient)
                        < (?height * ?height * ?yieldStrength)
                )
            }
        """
    ] .

gen:SubstitutePlateThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:shipComponent ;
    sh:sparql [
        sh:message "A component with substitute plate thickness requires nonnegative decimal thickness, adjacent frame height, yield strength, and net shell-plating thickness values in their prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:thickness ?declaredThickness .
                }
                FILTER (
                    NOT EXISTS {
                        $this nltl:thickness ?actualQuantity .
                        ?actualQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?actualMetres ;
                            qudt:unit unit:M .
                        FILTER(datatype(?actualMetres) = xsd:decimal && ?actualMetres >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:adjacentFrameHeight ?heightQuantity .
                        ?heightQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?height ;
                            qudt:unit unit:MilliM .
                        FILTER(datatype(?height) = xsd:decimal && ?height >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yieldQuantity .
                        ?yieldQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?yieldStrength ;
                            qudt:unit unit:MegaPA .
                        FILTER(datatype(?yieldStrength) = xsd:decimal && ?yieldStrength >= 0)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:netShellPlatingThickness ?shellQuantity .
                        ?shellQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?netShellThickness ;
                            qudt:unit unit:MilliM .
                        FILTER(datatype(?netShellThickness) = xsd:decimal && ?netShellThickness >= 0)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Substitute plate thickness must be at least the C = 805 calculated value, half the net shell-plating thickness, and 9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thickness ?actualQuantity ;
                    nltl:adjacentFrameHeight ?heightQuantity ;
                    nltl:yieldStrength ?yieldQuantity ;
                    nltl:netShellPlatingThickness ?shellQuantity .

                ?actualQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?actualMetres ;
                    qudt:unit unit:M .
                ?heightQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?height ;
                    qudt:unit unit:MilliM .
                ?yieldQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?yieldStrength ;
                    qudt:unit unit:MegaPA .
                ?shellQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?netShellThickness ;
                    qudt:unit unit:MilliM .

                FILTER(
                    datatype(?actualMetres) = xsd:decimal &&
                    datatype(?height) = xsd:decimal &&
                    datatype(?yieldStrength) = xsd:decimal &&
                    datatype(?netShellThickness) = xsd:decimal &&
                    ?actualMetres >= 0 &&
                    ?height >= 0 &&
                    ?yieldStrength >= 0 &&
                    ?netShellThickness >= 0
                )
                BIND(?actualMetres * 1000 AS ?actualMillimetres)
                FILTER(
                    ?actualMillimetres < 9 ||
                    (?actualMillimetres * 2) < ?netShellThickness ||
                    (?actualMillimetres * ?actualMillimetres * 805 * 805)
                        < (?height * ?height * ?yieldStrength)
                )
            }
        """
    ] .
<END_SHACL>
