gen:TRF051FrameWebThicknessShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:profileSection
            nltl:flatBarSection
        ) ;
        sh:message "An applicable frame must have exactly one supported profile classification."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:adjacentFrameHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:netShellPlatingThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Frame web thickness must be at least the formula value, half the net shell-plating thickness, and 9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameProfileType ?profile ;
                      nltl:frameWebThickness ?webQuantity ;
                      nltl:adjacentFrameHeight ?heightQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:netShellPlatingThickness ?shellQuantity .

                ?webQuantity qudt:numericValue ?web ;
                             qudt:unit unit:MilliM .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit unit:MilliM .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit unit:MegaPA .
                ?shellQuantity qudt:numericValue ?shell ;
                               qudt:unit unit:MilliM .

                VALUES (?profile ?constant) {
                    (nltl:profileSection "805"^^xsd:decimal)
                    (nltl:flatBarSection "282"^^xsd:decimal)
                }

                FILTER (
                    ?web >= "0"^^xsd:decimal &&
                    ?height >= "0"^^xsd:decimal &&
                    ?yield >= "0"^^xsd:decimal &&
                    ?shell >= "0"^^xsd:decimal &&
                    (
                        (?web * ?web * ?constant * ?constant) < (?height * ?height * ?yield) ||
                        ?web < (?shell / "2"^^xsd:decimal) ||
                        ?web < "9"^^xsd:decimal
                    )
                )
            }
        """
    ] .
