g:AntitrippingSupportShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable frame must have antitripping support spacing not exceeding 1300 mm unless equivalent support is demonstrated by direct calculation." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    {
        ?frame nltl:frameAsymmetrical "true"^^xsd:boolean .
    }
    UNION
    {
        ?frame nltl:frameWebAngleToShell ?angle .
        ?angle qudt:numericValue ?angleValue ;
               qudt:unit unit:DEG .
        FILTER(datatype(?angleValue) = xsd:decimal)
        FILTER(?angleValue < "90"^^xsd:decimal)
    }
    FILTER NOT EXISTS {
        ?frame nltl:equivalentSupportStatusByDirectCalculation "true"^^xsd:boolean .
    }
    FILTER NOT EXISTS {
        ?frame nltl:antitrippingSupportSpacing ?spacing .
        ?spacing qudt:numericValue ?spacingValue ;
                 qudt:unit unit:MilliM .
        FILTER(datatype(?spacingValue) = xsd:decimal)
        FILTER(?spacingValue <= "1300"^^xsd:decimal)
    }
}
""" ;
    ] ;
