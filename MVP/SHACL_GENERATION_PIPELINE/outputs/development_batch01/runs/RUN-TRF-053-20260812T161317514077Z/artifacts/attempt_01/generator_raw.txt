<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF053Shape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:sparql [
        sh:message "An ice stringer within the ice belt must provide decimal calculation quantities in the required units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:locatedWithinIceBelt true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:icePressure ?pNode .
                        ?pNode qudt:numericValue ?p ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?p) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:designIceLoadHeight ?hNode .
                        ?hNode qudt:numericValue ?h ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?h) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:designLineLoad ?qNode .
                        ?qNode qudt:numericValue ?q ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                        FILTER (datatype(?q) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:span ?lNode .
                        ?lNode qudt:numericValue ?l ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (datatype(?l) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:frameMomentFactorM ?mNode .
                        ?mNode qudt:numericValue ?m ;
                               qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?m) = xsd:decimal && ?m != 0)
                    } ||
                    NOT EXISTS {
                        $this nltl:iceStringerDistributionFactorF6 ?f6Node .
                        ?f6Node qudt:numericValue ?f6 ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?f6) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:iceStringerSafetyFactorF7 ?f7Node .
                        ?f7Node qudt:numericValue ?f7 ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?f7) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:iceStringerShearFactorF8 ?f8Node .
                        ?f8Node qudt:numericValue ?f8 ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (datatype(?f8) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yieldNode .
                        ?yieldNode qudt:numericValue ?yield ;
                                   qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                        FILTER (datatype(?yield) = xsd:decimal && ?yield != 0)
                    } ||
                    NOT EXISTS {
                        $this nltl:sectionModulus ?zNode .
                        ?zNode qudt:numericValue ?z ;
                               qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                        FILTER (datatype(?z) = xsd:decimal)
                    } ||
                    NOT EXISTS {
                        $this nltl:requiredShearArea ?aNode .
                        ?aNode qudt:numericValue ?a ;
                               qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                        FILTER (datatype(?a) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ice-stringer factors must be f6 = 0.9, f7 = 1.8, and f8 = 1.2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:locatedWithinIceBelt true .
                FILTER NOT EXISTS {
                    $this nltl:iceStringerDistributionFactorF6 ?f6Node ;
                          nltl:iceStringerSafetyFactorF7 ?f7Node ;
                          nltl:iceStringerShearFactorF8 ?f8Node .
                    ?f6Node qudt:numericValue ?f6 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?f7Node qudt:numericValue ?f7 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?f8Node qudt:numericValue ?f8 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (
                        datatype(?f6) = xsd:decimal &&
                        datatype(?f7) = xsd:decimal &&
                        datatype(?f8) = xsd:decimal &&
                        ?f6 = 0.9 &&
                        ?f7 = 1.8 &&
                        ?f8 = 1.2
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Design line load must equal the greater of ice pressure times design ice load height and 0.15 MN/m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:locatedWithinIceBelt true .
                FILTER NOT EXISTS {
                    $this nltl:icePressure ?pNode ;
                          nltl:designIceLoadHeight ?hNode ;
                          nltl:designLineLoad ?qNode .
                    ?pNode qudt:numericValue ?p ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?hNode qudt:numericValue ?h ;
                           qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?qNode qudt:numericValue ?q ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                    FILTER (
                        datatype(?p) = xsd:decimal &&
                        datatype(?h) = xsd:decimal &&
                        datatype(?q) = xsd:decimal
                    )
                    BIND(IF((?p * ?h) < 0.15, 150.0, 1000.0 * ?p * ?h) AS ?requiredQ)
                    FILTER (?q = ?requiredQ)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Section modulus must satisfy the prescribed ice-stringer formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:locatedWithinIceBelt true .
                FILTER NOT EXISTS {
                    $this nltl:designLineLoad ?qNode ;
                          nltl:span ?lNode ;
                          nltl:frameMomentFactorM ?mNode ;
                          nltl:yieldStrength ?yieldNode ;
                          nltl:iceStringerDistributionFactorF6 ?f6Node ;
                          nltl:iceStringerSafetyFactorF7 ?f7Node ;
                          nltl:sectionModulus ?zNode .
                    ?qNode qudt:numericValue ?q ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                    ?lNode qudt:numericValue ?l ;
                           qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?mNode qudt:numericValue ?m ;
                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?yieldNode qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?f6Node qudt:numericValue ?f6 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?f7Node qudt:numericValue ?f7 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?zNode qudt:numericValue ?z ;
                           qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                    FILTER (
                        datatype(?q) = xsd:decimal &&
                        datatype(?l) = xsd:decimal &&
                        datatype(?m) = xsd:decimal &&
                        datatype(?yield) = xsd:decimal &&
                        datatype(?f6) = xsd:decimal &&
                        datatype(?f7) = xsd:decimal &&
                        datatype(?z) = xsd:decimal &&
                        ?m != 0 &&
                        ?yield != 0 &&
                        ?z = ((?f6 * ?f7 * ?q * ?l * ?l * 1000.0) / (?m * ?yield))
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Required shear area must satisfy the prescribed ice-stringer formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:locatedWithinIceBelt true .
                FILTER NOT EXISTS {
                    $this nltl:designLineLoad ?qNode ;
                          nltl:span ?lNode ;
                          nltl:yieldStrength ?yieldNode ;
                          nltl:iceStringerDistributionFactorF6 ?f6Node ;
                          nltl:iceStringerSafetyFactorF7 ?f7Node ;
                          nltl:iceStringerShearFactorF8 ?f8Node ;
                          nltl:requiredShearArea ?aNode .
                    ?qNode qudt:numericValue ?q ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN-PER-M> .
                    ?lNode qudt:numericValue ?l ;
                           qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?yieldNode qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?f6Node qudt:numericValue ?f6 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?f7Node qudt:numericValue ?f7 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?f8Node qudt:numericValue ?f8 ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?aNode qudt:numericValue ?a ;
                           qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                    FILTER (
                        datatype(?q) = xsd:decimal &&
                        datatype(?l) = xsd:decimal &&
                        datatype(?yield) = xsd:decimal &&
                        datatype(?f6) = xsd:decimal &&
                        datatype(?f7) = xsd:decimal &&
                        datatype(?f8) = xsd:decimal &&
                        datatype(?a) = xsd:decimal &&
                        ?yield != 0 &&
                        ?a >= 0 &&
                        (4.0 * ?yield * ?yield * ?a * ?a) =
                        (3.0 * (?f6 * ?f7 * ?f8 * ?q * ?l * 10.0) *
                               (?f6 * ?f7 * ?f8 * ?q * ?l * 10.0))
                    )
                }
            }
        """
    ] .
<END_SHACL>
