<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceStringerWithinIceBeltShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:locatedWithinIceBelt ;
                    sh:hasValue true
                ]
            ]
        ]
        gen:IceStringerCalculationShape
    ) .

gen:IceStringerCalculationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPAQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveUnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveMegaPAQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStringerDistributionFactorF6 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FactorF6QuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStringerSafetyFactorF7 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FactorF7QuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStringerShearFactorF8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FactorF8QuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CubicCentimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareCentimetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "Design line load, section modulus, or required shear area differs from the prescribed calculation beyond relative tolerance 1e-6 with absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:designLineLoad/qudt:numericValue ?q ;
                      nltl:span/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:iceStringerDistributionFactorF6/qudt:numericValue ?f6 ;
                      nltl:iceStringerSafetyFactorF7/qudt:numericValue ?f7 ;
                      nltl:iceStringerShearFactorF8/qudt:numericValue ?f8 ;
                      nltl:sectionModulus/qudt:numericValue ?z ;
                      nltl:requiredShearArea/qudt:numericValue ?a .

                BIND(
                    IF(
                        (?p * ?h) < xsd:decimal("0.15"),
                        xsd:decimal("150"),
                        (?p * ?h * xsd:decimal("1000"))
                    )
                    AS ?expectedQ
                )
                BIND(
                    (
                        ?f6 * ?f7 * (?q / xsd:decimal("1000")) *
                        ?l * ?l /
                        (?m * ?sigma) *
                        xsd:decimal("1000000")
                    )
                    AS ?expectedZ
                )
                BIND(
                    (
                        xsd:decimal("1.7320508075688772") *
                        ?f6 * ?f7 * ?f8 *
                        (?q / xsd:decimal("1000")) * ?l /
                        (xsd:decimal("2") * ?sigma) *
                        xsd:decimal("10000")
                    )
                    AS ?expectedA
                )

                FILTER(
                    ABS(?q - ?expectedQ) >
                        (xsd:decimal("0.000000001") +
                         xsd:decimal("0.000001") * ABS(?expectedQ))
                    ||
                    ABS(?z - ?expectedZ) >
                        (xsd:decimal("0.000000001") +
                         xsd:decimal("0.000001") * ABS(?expectedZ))
                    ||
                    ABS(?a - ?expectedA) >
                        (xsd:decimal("0.000000001") +
                         xsd:decimal("0.000001") * ABS(?expectedA))
                )
            }
        """
    ] .

gen:MegaPAQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PositiveMegaPAQuantityShape
    a sh:NodeShape ;
    sh:node gen:MegaPAQuantityShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minExclusive 0
    ] .
