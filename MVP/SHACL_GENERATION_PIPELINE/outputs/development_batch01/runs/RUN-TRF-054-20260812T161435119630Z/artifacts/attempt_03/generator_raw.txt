<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:StringersOutsideIceBeltShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:sparql [
        sh:message "An applicable ice stringer requires decimal, correctly unit-bearing formula inputs and results, with nonzero divisors." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true .

                FILTER (
                    NOT EXISTS {
                        $this nltl:icePressure ?p .
                        ?p qudt:numericValue ?pValue ;
                           qudt:unit unit:MegaPA .
                        FILTER (DATATYPE(?pValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:designIceLoadHeight ?h .
                        ?h qudt:numericValue ?hValue ;
                           qudt:unit unit:M .
                        FILTER (DATATYPE(?hValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:stringerSpan ?l .
                        ?l qudt:numericValue ?lValue ;
                           qudt:unit unit:M .
                        FILTER (DATATYPE(?lValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToAdjacentIceStringer ?ls .
                        ?ls qudt:numericValue ?lsValue ;
                            qudt:unit unit:M .
                        FILTER (
                            DATATYPE(?lsValue) = xsd:decimal &&
                            ?lsValue != 0
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:distanceToIceBelt ?hs .
                        ?hs qudt:numericValue ?hsValue ;
                            qudt:unit unit:M .
                        FILTER (DATATYPE(?hsValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:frameMomentFactorM ?m .
                        ?m qudt:numericValue ?mValue ;
                           qudt:unit unit:UNITLESS .
                        FILTER (
                            DATATYPE(?mValue) = xsd:decimal &&
                            ?mValue != 0
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yield .
                        ?yield qudt:numericValue ?yieldValue ;
                               qudt:unit unit:MegaPA .
                        FILTER (
                            DATATYPE(?yieldValue) = xsd:decimal &&
                            ?yieldValue != 0
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:outsideIceBeltDistributionFactorF9 ?f9 .
                        ?f9 qudt:numericValue ?f9Value ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?f9Value) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:outsideIceBeltSafetyFactorF10 ?f10 .
                        ?f10 qudt:numericValue ?f10Value ;
                             qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?f10Value) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:outsideIceBeltShearFactorF11 ?f11 .
                        ?f11 qudt:numericValue ?f11Value ;
                             qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?f11Value) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:designLineLoad ?lineLoad .
                        ?lineLoad qudt:numericValue ?lineLoadValue ;
                                  qudt:unit unit:KiloN-PER-M .
                        FILTER (DATATYPE(?lineLoadValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:outsideIceBeltFactor ?outsideFactor .
                        ?outsideFactor qudt:numericValue ?outsideFactorValue ;
                                       qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?outsideFactorValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:sectionModulus ?sectionModulus .
                        ?sectionModulus qudt:numericValue ?sectionModulusValue ;
                                        qudt:unit unit:CentiM3 .
                        FILTER (DATATYPE(?sectionModulusValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredShearArea ?shearArea .
                        ?shearArea qudt:numericValue ?shearAreaValue ;
                                   qudt:unit unit:CentiM2 .
                        FILTER (DATATYPE(?shearAreaValue) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The design line load, outside-ice-belt factor, section modulus, or effective shear area does not satisfy the required formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true .

                FILTER NOT EXISTS {
                    $this nltl:icePressure ?p ;
                          nltl:designIceLoadHeight ?h ;
                          nltl:stringerSpan ?l ;
                          nltl:distanceToAdjacentIceStringer ?ls ;
                          nltl:distanceToIceBelt ?hs ;
                          nltl:frameMomentFactorM ?m ;
                          nltl:yieldStrength ?yield ;
                          nltl:outsideIceBeltDistributionFactorF9 ?f9 ;
                          nltl:outsideIceBeltSafetyFactorF10 ?f10 ;
                          nltl:outsideIceBeltShearFactorF11 ?f11 ;
                          nltl:designLineLoad ?lineLoad ;
                          nltl:outsideIceBeltFactor ?outsideFactor ;
                          nltl:sectionModulus ?sectionModulus ;
                          nltl:requiredShearArea ?shearArea .

                    ?p qudt:numericValue ?pValue ;
                       qudt:unit unit:MegaPA .
                    ?h qudt:numericValue ?hValue ;
                       qudt:unit unit:M .
                    ?l qudt:numericValue ?lValue ;
                       qudt:unit unit:M .
                    ?ls qudt:numericValue ?lsValue ;
                        qudt:unit unit:M .
                    ?hs qudt:numericValue ?hsValue ;
                        qudt:unit unit:M .
                    ?m qudt:numericValue ?mValue ;
                       qudt:unit unit:UNITLESS .
                    ?yield qudt:numericValue ?yieldValue ;
                           qudt:unit unit:MegaPA .
                    ?f9 qudt:numericValue ?f9Value ;
                        qudt:unit unit:UNITLESS .
                    ?f10 qudt:numericValue ?f10Value ;
                         qudt:unit unit:UNITLESS .
                    ?f11 qudt:numericValue ?f11Value ;
                         qudt:unit unit:UNITLESS .
                    ?lineLoad qudt:numericValue ?lineLoadValue ;
                              qudt:unit unit:KiloN-PER-M .
                    ?outsideFactor qudt:numericValue ?outsideFactorValue ;
                                   qudt:unit unit:UNITLESS .
                    ?sectionModulus qudt:numericValue ?sectionModulusValue ;
                                    qudt:unit unit:CentiM3 .
                    ?shearArea qudt:numericValue ?shearAreaValue ;
                               qudt:unit unit:CentiM2 .

                    FILTER (
                        DATATYPE(?pValue) = xsd:decimal &&
                        DATATYPE(?hValue) = xsd:decimal &&
                        DATATYPE(?lValue) = xsd:decimal &&
                        DATATYPE(?lsValue) = xsd:decimal &&
                        DATATYPE(?hsValue) = xsd:decimal &&
                        DATATYPE(?mValue) = xsd:decimal &&
                        DATATYPE(?yieldValue) = xsd:decimal &&
                        DATATYPE(?f9Value) = xsd:decimal &&
                        DATATYPE(?f10Value) = xsd:decimal &&
                        DATATYPE(?f11Value) = xsd:decimal &&
                        DATATYPE(?lineLoadValue) = xsd:decimal &&
                        DATATYPE(?outsideFactorValue) = xsd:decimal &&
                        DATATYPE(?sectionModulusValue) = xsd:decimal &&
                        DATATYPE(?shearAreaValue) = xsd:decimal &&
                        ?lsValue != 0 &&
                        ?mValue != 0 &&
                        ?yieldValue != 0 &&
                        ?f9Value = 0.80 &&
                        ?f10Value = 1.8 &&
                        ?f11Value = 1.2
                    )

                    BIND(
                        IF(
                            (?pValue * ?hValue) < 0.15,
                            0.15,
                            (?pValue * ?hValue)
                        )
                        AS ?designLineLoadMNPerM
                    )
                    BIND(
                        ?designLineLoadMNPerM * 1000
                        AS ?expectedLineLoadKiloNPerM
                    )
                    BIND(
                        1 - (?hsValue / ?lsValue)
                        AS ?expectedOutsideFactor
                    )
                    BIND(
                        (
                            ?f9Value *
                            ?f10Value *
                            ?designLineLoadMNPerM *
                            ?lValue *
                            ?lValue
                        )
                        /
                        (?mValue * ?yieldValue)
                        *
                        ?expectedOutsideFactor
                        *
                        1000000
                        AS ?expectedSectionModulus
                    )
                    BIND(
                        1.73205080756887729352744634150587236694280525381038
                        AS ?sqrtThree
                    )
                    BIND(
                        (
                            ?sqrtThree *
                            ?f9Value *
                            ?f10Value *
                            ?f11Value *
                            ?designLineLoadMNPerM *
                            ?lValue
                        )
                        /
                        (2 * ?yieldValue)
                        *
                        ?expectedOutsideFactor
                        *
                        10000
                        AS ?expectedShearArea
                    )

                    FILTER (
                        ?lineLoadValue = ?expectedLineLoadKiloNPerM &&
                        ?outsideFactorValue = ?expectedOutsideFactor &&
                        ?sectionModulusValue = ?expectedSectionModulus &&
                        ?shearAreaValue = ?expectedShearArea
                    )
                }
            }
        """
    ] .
<END_SHACL>
