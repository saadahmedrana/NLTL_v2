<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:StringersOutsideIceBeltShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:sparql [
        sh:message "An applicable ice stringer must have correctly unit-tagged decimal inputs and results satisfying the design line load, outside-ice-belt factor, section modulus, and shear-area formulas within relative tolerance 1e-6 or absolute tolerance 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:locatedOutsideIceBelt true ;
          nltl:supportsIceStrengthenedFrames true .

    FILTER NOT EXISTS {
        $this nltl:icePressure ?pNode ;
              nltl:designIceLoadHeight ?hNode ;
              nltl:designLineLoad ?qNode ;
              nltl:stringerSpan ?lNode ;
              nltl:frameMomentFactorM ?mNode ;
              nltl:yieldStrength ?yieldNode ;
              nltl:distanceToAdjacentIceStringer ?lsNode ;
              nltl:distanceToIceBelt ?hsNode ;
              nltl:outsideIceBeltDistributionFactorF9 ?f9Node ;
              nltl:outsideIceBeltSafetyFactorF10 ?f10Node ;
              nltl:outsideIceBeltShearFactorF11 ?f11Node ;
              nltl:outsideIceBeltFactor ?outsideFactorNode ;
              nltl:sectionModulus ?zNode ;
              nltl:requiredShearArea ?areaNode .

        ?pNode qudt:numericValue ?p ;
               qudt:unit unit:MegaPA .
        ?hNode qudt:numericValue ?h ;
               qudt:unit unit:M .
        ?qNode qudt:numericValue ?q ;
               qudt:unit unit:KiloN-PER-M .
        ?lNode qudt:numericValue ?l ;
               qudt:unit unit:M .
        ?mNode qudt:numericValue ?m ;
               qudt:unit unit:UNITLESS .
        ?yieldNode qudt:numericValue ?yieldStrength ;
                   qudt:unit unit:MegaPA .
        ?lsNode qudt:numericValue ?ls ;
                qudt:unit unit:M .
        ?hsNode qudt:numericValue ?hs ;
                qudt:unit unit:M .
        ?f9Node qudt:numericValue ?f9 ;
                qudt:unit unit:UNITLESS .
        ?f10Node qudt:numericValue ?f10 ;
                 qudt:unit unit:UNITLESS .
        ?f11Node qudt:numericValue ?f11 ;
                 qudt:unit unit:UNITLESS .
        ?outsideFactorNode qudt:numericValue ?outsideFactor ;
                           qudt:unit unit:UNITLESS .
        ?zNode qudt:numericValue ?z ;
               qudt:unit unit:CentiM3 .
        ?areaNode qudt:numericValue ?area ;
                  qudt:unit unit:CentiM2 .

        FILTER(
            datatype(?p) = xsd:decimal &&
            datatype(?h) = xsd:decimal &&
            datatype(?q) = xsd:decimal &&
            datatype(?l) = xsd:decimal &&
            datatype(?m) = xsd:decimal &&
            datatype(?yieldStrength) = xsd:decimal &&
            datatype(?ls) = xsd:decimal &&
            datatype(?hs) = xsd:decimal &&
            datatype(?f9) = xsd:decimal &&
            datatype(?f10) = xsd:decimal &&
            datatype(?f11) = xsd:decimal &&
            datatype(?outsideFactor) = xsd:decimal &&
            datatype(?z) = xsd:decimal &&
            datatype(?area) = xsd:decimal
        )

        FILTER(
            ?m != 0 &&
            ?yieldStrength != 0 &&
            ?ls != 0 &&
            ?f9 = 0.80 &&
            ?f10 = 1.8 &&
            ?f11 = 1.2
        )

        BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?expectedLineLoadMNPerM)
        BIND((?q / 1000) AS ?lineLoadMNPerM)
        BIND((1 - (?hs / ?ls)) AS ?expectedOutsideFactor)

        BIND(
            (?f9 * ?f10 * ?lineLoadMNPerM * ?l * ?l *
             ?outsideFactor * 1000000) /
            (?m * ?yieldStrength)
            AS ?expectedZ
        )

        BIND(
            (1.7320508075688772 * ?f9 * ?f10 * ?f11 *
             ?lineLoadMNPerM * ?l * ?outsideFactor * 10000) /
            (2 * ?yieldStrength)
            AS ?expectedArea
        )

        FILTER(
            (
                ABS(?lineLoadMNPerM - ?expectedLineLoadMNPerM)
                    <= 0.000001 * ABS(?expectedLineLoadMNPerM)
                ||
                ABS(?lineLoadMNPerM - ?expectedLineLoadMNPerM)
                    <= 0.000000001
            ) &&
            (
                ABS(?outsideFactor - ?expectedOutsideFactor)
                    <= 0.000001 * ABS(?expectedOutsideFactor)
                ||
                ABS(?outsideFactor - ?expectedOutsideFactor)
                    <= 0.000000001
            ) &&
            (
                ABS(?z - ?expectedZ)
                    <= 0.000001 * ABS(?expectedZ)
                ||
                ABS(?z - ?expectedZ)
                    <= 0.000000001
            ) &&
            (
                ABS(?area - ?expectedArea)
                    <= 0.000001 * ABS(?expectedArea)
                ||
                ABS(?area - ?expectedArea)
                    <= 0.000000001
            )
        )
    }
}
""" 
    ] .
<END_SHACL>
