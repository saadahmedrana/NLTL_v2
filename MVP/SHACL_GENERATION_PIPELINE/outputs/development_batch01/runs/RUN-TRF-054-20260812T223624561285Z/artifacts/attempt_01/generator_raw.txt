<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IceStringerOutsideIceBeltShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:locatedOutsideIceBelt ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:supportsIceStrengthenedFrames ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:not [
                sh:property [
                    sh:path nltl:locatedOutsideIceBelt ;
                    sh:hasValue true
                ] ;
                sh:property [
                    sh:path nltl:supportsIceStrengthenedFrames ;
                    sh:hasValue true
                ]
            ]
        ]
        gen:OutsideIceBeltCalculationInputsShape
    ) ;
    sh:sparql [
        sh:message "For an outside-belt stringer supporting ice-strengthened frames, design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy clauses 4.13 and 4.14 within relative tolerance 1e-6 or absolute tolerance 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:locatedOutsideIceBelt true ;
          nltl:supportsIceStrengthenedFrames true ;
          nltl:icePressure/qudt:numericValue ?p ;
          nltl:designIceLoadHeight/qudt:numericValue ?h ;
          nltl:designLineLoad/qudt:numericValue ?lineLoad ;
          nltl:stringerSpan/qudt:numericValue ?l ;
          nltl:frameMomentFactorM/qudt:numericValue ?m ;
          nltl:yieldStrength/qudt:numericValue ?sigmaY ;
          nltl:distanceToAdjacentIceStringer/qudt:numericValue ?ls ;
          nltl:distanceToIceBelt/qudt:numericValue ?hs ;
          nltl:outsideIceBeltDistributionFactorF9/qudt:numericValue ?f9 ;
          nltl:outsideIceBeltSafetyFactorF10/qudt:numericValue ?f10 ;
          nltl:outsideIceBeltShearFactorF11/qudt:numericValue ?f11 ;
          nltl:outsideIceBeltFactor/qudt:numericValue ?outsideFactor ;
          nltl:sectionModulus/qudt:numericValue ?sectionModulus ;
          nltl:requiredShearArea/qudt:numericValue ?shearArea .

    BIND (?p * ?h AS ?calculatedLineLoadMNPerM)
    BIND (
        IF(
            ?calculatedLineLoadMNPerM < 0.15,
            150.0,
            1000.0 * ?calculatedLineLoadMNPerM
        ) AS ?expectedLineLoadKNPerM
    )
    BIND (1.0 - (?hs / ?ls) AS ?expectedOutsideFactor)
    BIND (
        ?f9 * ?f10 * (?expectedLineLoadKNPerM / 1000.0) *
        ?l * ?l / (?m * ?sigmaY) *
        ?expectedOutsideFactor * 1000000.0
        AS ?expectedSectionModulus
    )
    BIND (
        1.7320508075688772 * ?f9 * ?f10 * ?f11 *
        (?expectedLineLoadKNPerM / 1000.0) * ?l /
        (2.0 * ?sigmaY) *
        ?expectedOutsideFactor * 10000.0
        AS ?expectedShearArea
    )

    FILTER (
        (
            ABS(?lineLoad - ?expectedLineLoadKNPerM) > 0.000000001 &&
            ABS(?lineLoad - ?expectedLineLoadKNPerM) >
                0.000001 * ABS(?expectedLineLoadKNPerM)
        ) ||
        (
            ABS(?outsideFactor - ?expectedOutsideFactor) > 0.000000001 &&
            ABS(?outsideFactor - ?expectedOutsideFactor) >
                0.000001 * ABS(?expectedOutsideFactor)
        ) ||
        (
            ABS(?sectionModulus - ?expectedSectionModulus) > 0.000000001 &&
            ABS(?sectionModulus - ?expectedSectionModulus) >
                0.000001 * ABS(?expectedSectionModulus)
        ) ||
        (
            ABS(?shearArea - ?expectedShearArea) > 0.000000001 &&
            ABS(?shearArea - ?expectedShearArea) >
                0.000001 * ABS(?expectedShearArea)
        )
    )
}
"""
    ] .
