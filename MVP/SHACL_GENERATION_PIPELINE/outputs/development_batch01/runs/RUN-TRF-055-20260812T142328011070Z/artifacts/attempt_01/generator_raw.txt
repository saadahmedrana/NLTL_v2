<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF055NarrowDeckStripShape
    a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:sparql [
        sh:message "A narrow deck strip serving as an ice stringer must provide section modulus data in cubic centimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:servesAsIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:sectionModulus ?quantity .
                    ?quantity qudt:numericValue ?numericValue ;
                              qudt:unit unit:CentiM3 .
                    FILTER(datatype(?numericValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A permitted reduced line load must be at least 100 kN/m and less than 150 kN/m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?load
            WHERE {
                $this nltl:servesAsIceStringer true ;
                      nltl:permittedReducedLineLoad ?load .
                FILTER NOT EXISTS {
                    ?load qudt:numericValue ?numericValue ;
                          qudt:unit unit:KiloN-PER-M .
                    FILTER(
                        datatype(?numericValue) = xsd:decimal &&
                        ?numericValue >= "100"^^xsd:decimal &&
                        ?numericValue < "150"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
