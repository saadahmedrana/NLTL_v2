<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .

gen:NarrowDeckStripShearAreaShape
    a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:sparql [
        sh:message "For a narrow deck strip associated with a hatch and serving as an ice stringer, shear area must be at least the required shear area, using m2 and cm2 respectively." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:servesAsIceStringer ?iceStringer ;
                      nltl:hatchOpeningLength ?hatchOpening .
                FILTER (?iceStringer = true)
                FILTER NOT EXISTS {
                    $this nltl:shearArea ?actualQuantity ;
                          nltl:requiredShearArea ?requiredQuantity .
                    ?actualQuantity
                        qudt:numericValue ?actualArea ;
                        qudt:unit unit:M2 .
                    ?requiredQuantity
                        qudt:numericValue ?requiredArea ;
                        qudt:unit unit:CentiM2 .
                    FILTER (
                        datatype(?actualArea) = xsd:decimal &&
                        datatype(?requiredArea) = xsd:decimal &&
                        ?actualArea >= (?requiredArea * 0.0001)
                    )
                }
            }
        """
    ] .
<END_SHACL>
