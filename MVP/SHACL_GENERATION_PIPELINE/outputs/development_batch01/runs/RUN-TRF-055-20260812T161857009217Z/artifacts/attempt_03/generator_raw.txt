<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:NarrowDeckStripShape
    a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:sparql [
        sh:message "An applicable ice-stringer deck strip requires a decimal section modulus in cubic centimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:servesAsIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:sectionModulus ?sectionModulusQuantity .
                    ?sectionModulusQuantity
                        qudt:numericValue ?sectionModulus ;
                        qudt:unit unit:CentiM3 .
                    FILTER (datatype(?sectionModulus) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ice-stringer deck strip requires decimal actual and required shear areas in their prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:servesAsIceStringer true .
                FILTER (
                    NOT EXISTS {
                        $this nltl:shearArea ?actualQuantity .
                        ?actualQuantity
                            qudt:numericValue ?actualArea ;
                            qudt:unit unit:M2 .
                        FILTER (datatype(?actualArea) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredShearArea ?requiredQuantity .
                        ?requiredQuantity
                            qudt:numericValue ?requiredArea ;
                            qudt:unit unit:CentiM2 .
                        FILTER (datatype(?requiredArea) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual shear area must be at least the required shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this
                    nltl:servesAsIceStringer true ;
                    nltl:shearArea ?actualQuantity ;
                    nltl:requiredShearArea ?requiredQuantity .
                ?actualQuantity
                    qudt:numericValue ?actualArea ;
                    qudt:unit unit:M2 .
                ?requiredQuantity
                    qudt:numericValue ?requiredArea ;
                    qudt:unit unit:CentiM2 .
                FILTER (
                    datatype(?actualArea) = xsd:decimal &&
                    datatype(?requiredArea) = xsd:decimal &&
                    (?actualArea * 10000) < ?requiredArea
                )
            }
        """
    ] .
<END_SHACL>
