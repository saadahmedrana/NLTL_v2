<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF056Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Ship side deflection in metres is required when a hatch opening is longer than half the ship breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipBreadth ?breadthNode ;
                      nltl:hasComponent ?component .
                ?breadthNode qudt:numericValue ?breadthValue ;
                             qudt:unit unit:M .
                ?component nltl:hatchOpeningLength ?openingNode .
                ?openingNode qudt:numericValue ?openingValue ;
                             qudt:unit unit:M .
                FILTER (
                    datatype(?breadthValue) = xsd:decimal &&
                    datatype(?openingValue) = xsd:decimal &&
                    ?openingValue > (?breadthValue / 2)
                )
                FILTER NOT EXISTS {
                    $this nltl:shipSideDeflection ?deflectionNode .
                    ?deflectionNode qudt:numericValue ?deflectionValue ;
                                    qudt:unit unit:M .
                    FILTER (datatype(?deflectionValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch cover design evidence is required when the hatch opening is longer than half the ship breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipBreadth ?breadthNode ;
                      nltl:hasComponent ?component .
                ?breadthNode qudt:numericValue ?breadthValue ;
                             qudt:unit unit:M .
                ?component nltl:hatchOpeningLength ?openingNode .
                ?openingNode qudt:numericValue ?openingValue ;
                             qudt:unit unit:M .
                FILTER (
                    datatype(?breadthValue) = xsd:decimal &&
                    datatype(?openingValue) = xsd:decimal &&
                    ?openingValue > (?breadthValue / 2)
                )
                FILTER NOT EXISTS {
                    ?component nltl:hasHatchCoverDesignEvidence ?evidence .
                    ?evidence rdf:type nltl:hatchCoverDesignEvidence .
                    FILTER (isIRI(?evidence))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch fitting design evidence is required when the hatch opening is longer than half the ship breadth." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipBreadth ?breadthNode ;
                      nltl:hasComponent ?component .
                ?breadthNode qudt:numericValue ?breadthValue ;
                             qudt:unit unit:M .
                ?component nltl:hatchOpeningLength ?openingNode .
                ?openingNode qudt:numericValue ?openingValue ;
                             qudt:unit unit:M .
                FILTER (
                    datatype(?breadthValue) = xsd:decimal &&
                    datatype(?openingValue) = xsd:decimal &&
                    ?openingValue > (?breadthValue / 2)
                )
                FILTER NOT EXISTS {
                    ?component nltl:hasHatchFittingDesignEvidence ?evidence .
                    ?evidence rdf:type nltl:hatchFittingDesignEvidence .
                    FILTER (isIRI(?evidence))
                }
            }
        """
    ] .
<END_SHACL>
