<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:WeatherdeckHatchDeflectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each linked weatherdeck hatch must be an IRI typed as weatherdeckHatch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                FILTER (
                    !isIRI(?hatch) ||
                    NOT EXISTS {
                        ?hatch rdf:type nltl:weatherdeckHatch .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ship breadth and hatch opening length must be decimal quantities in metres to determine whether the hatch opening exceeds B/2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                {
                    FILTER NOT EXISTS {
                        $this nltl:shipBreadth ?breadthQuantity .
                        ?breadthQuantity qudt:numericValue ?breadthValue ;
                                         qudt:unit unit:M .
                        FILTER(DATATYPE(?breadthValue) = xsd:decimal)
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?hatch nltl:hatchOpeningLength ?openingQuantity .
                        ?openingQuantity qudt:numericValue ?openingValue ;
                                         qudt:unit unit:M .
                        FILTER(DATATYPE(?openingValue) = xsd:decimal)
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ship-side deflection must be provided as a decimal quantity in metres for a hatch opening greater than B/2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadthValue ;
                                 qudt:unit unit:M .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingValue ;
                                 qudt:unit unit:M .
                FILTER(DATATYPE(?breadthValue) = xsd:decimal)
                FILTER(DATATYPE(?openingValue) = xsd:decimal)
                FILTER(?openingValue > (?breadthValue / 2))
                FILTER NOT EXISTS {
                    $this nltl:shipSideDeflection ?deflectionQuantity .
                    ?deflectionQuantity qudt:numericValue ?deflectionValue ;
                                        qudt:unit unit:M .
                    FILTER(DATATYPE(?deflectionValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch-cover design evidence is required for a weatherdeck hatch opening greater than B/2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadthValue ;
                                 qudt:unit unit:M .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingValue ;
                                 qudt:unit unit:M .
                FILTER(DATATYPE(?breadthValue) = xsd:decimal)
                FILTER(DATATYPE(?openingValue) = xsd:decimal)
                FILTER(?openingValue > (?breadthValue / 2))
                FILTER NOT EXISTS {
                    ?hatch nltl:hasHatchCoverDesignEvidence ?evidence .
                    ?evidence rdf:type nltl:hatchCoverDesignEvidence .
                    FILTER(isIRI(?evidence))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Hatch-fitting design evidence is required for a weatherdeck hatch opening greater than B/2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch ;
                      nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadthValue ;
                                 qudt:unit unit:M .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingValue ;
                                 qudt:unit unit:M .
                FILTER(DATATYPE(?breadthValue) = xsd:decimal)
                FILTER(DATATYPE(?openingValue) = xsd:decimal)
                FILTER(?openingValue > (?breadthValue / 2))
                FILTER NOT EXISTS {
                    ?hatch nltl:hasHatchFittingDesignEvidence ?evidence .
                    ?evidence rdf:type nltl:hatchFittingDesignEvidence .
                    FILTER(isIRI(?evidence))
                }
            }
        """
    ] .
<END_SHACL>
