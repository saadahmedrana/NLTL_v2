<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF057WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasComponent ;
    sh:sparql [
        sh:message "For a ship component receiving load from an ice stringer or longitudinal framing, the required quantities, units, constants, load-length relation, and web-frame ice-load formula must conform to clause 4.6.1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasComponent $this .
                $this nltl:iceLoadSourceType ?source .
                FILTER (
                    ?source IN (
                        nltl:iceStringerLoadSource,
                        nltl:longitudinalFramingLoadSource
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:icePressure ?pressureQuantity ;
                          nltl:iceLoadHeight ?heightQuantity ;
                          nltl:webFrameSpacing ?spacingQuantity ;
                          nltl:loadLengthParameter ?loadLengthQuantity ;
                          nltl:minimumLineLoad ?minimumLineLoadQuantity ;
                          nltl:webFrameSafetyFactor ?factorQuantity ;
                          nltl:webFrameIceLoad ?loadQuantity .

                    ?pressureQuantity
                        qudt:numericValue ?pressure ;
                        qudt:unit unit:MegaPA .
                    ?heightQuantity
                        qudt:numericValue ?height ;
                        qudt:unit unit:M .
                    ?spacingQuantity
                        qudt:numericValue ?spacing ;
                        qudt:unit unit:M .
                    ?loadLengthQuantity
                        qudt:numericValue ?loadLength ;
                        qudt:unit unit:M .
                    ?minimumLineLoadQuantity
                        qudt:numericValue ?minimumLineLoad ;
                        qudt:unit unit:KiloN-PER-M .
                    ?factorQuantity
                        qudt:numericValue ?factor ;
                        qudt:unit unit:UNITLESS .
                    ?loadQuantity
                        qudt:numericValue ?load ;
                        qudt:unit unit:KiloN .

                    FILTER (
                        datatype(?pressure) = xsd:decimal &&
                        datatype(?height) = xsd:decimal &&
                        datatype(?spacing) = xsd:decimal &&
                        datatype(?loadLength) = xsd:decimal &&
                        datatype(?minimumLineLoad) = xsd:decimal &&
                        datatype(?factor) = xsd:decimal &&
                        datatype(?load) = xsd:decimal
                    )
                    FILTER (?factor = 1.8)
                    FILTER (?minimumLineLoad = 150)
                    FILTER (?loadLength = 2 * ?spacing)
                    FILTER (
                        ?load =
                            ?factor *
                            IF(
                                (1000 * ?pressure * ?height) < ?minimumLineLoad,
                                ?minimumLineLoad,
                                (1000 * ?pressure * ?height)
                            ) *
                            ?spacing
                    )
                }
            }
        """
    ] .
<END_SHACL>
