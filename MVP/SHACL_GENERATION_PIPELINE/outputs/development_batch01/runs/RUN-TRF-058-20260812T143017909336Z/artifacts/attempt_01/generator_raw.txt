<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF058ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node gen:TRF058StringerAdjustmentShape
    ] .

gen:TRF058StringerAdjustmentShape
    a sh:NodeShape ;
    sh:sparql [
        sh:message "A supported stringer outside the ice belt requires valid ice-load force, height, span, and adjusted-force quantities in the prescribed units; the span must be nonzero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true .
                FILTER NOT EXISTS {
                    $this nltl:iceLoadForce ?force ;
                          nltl:stringerOutsideIceBeltHeight ?height ;
                          nltl:stringerSpan ?span ;
                          nltl:adjustedIceLoadForce ?adjusted .
                    ?force qudt:numericValue ?forceValue ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    ?height qudt:numericValue ?heightValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?span qudt:numericValue ?spanValue ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?adjusted qudt:numericValue ?adjustedValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER (
                        datatype(?forceValue) = xsd:decimal &&
                        datatype(?heightValue) = xsd:decimal &&
                        datatype(?spanValue) = xsd:decimal &&
                        datatype(?adjustedValue) = xsd:decimal &&
                        ?spanValue != 0
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Adjusted ice load force must equal ice load force multiplied by (1 - stringerOutsideIceBeltHeight/stringerSpan)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:supportedStringerOutsideIceBelt true .
                FILTER EXISTS {
                    $this nltl:iceLoadForce ?availableForce ;
                          nltl:stringerOutsideIceBeltHeight ?availableHeight ;
                          nltl:stringerSpan ?availableSpan ;
                          nltl:adjustedIceLoadForce ?availableAdjusted .
                    ?availableForce qudt:numericValue ?availableForceValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    ?availableHeight qudt:numericValue ?availableHeightValue ;
                                     qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?availableSpan qudt:numericValue ?availableSpanValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?availableAdjusted qudt:numericValue ?availableAdjustedValue ;
                                       qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER (
                        datatype(?availableForceValue) = xsd:decimal &&
                        datatype(?availableHeightValue) = xsd:decimal &&
                        datatype(?availableSpanValue) = xsd:decimal &&
                        datatype(?availableAdjustedValue) = xsd:decimal &&
                        ?availableSpanValue != 0
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:iceLoadForce ?force ;
                          nltl:stringerOutsideIceBeltHeight ?height ;
                          nltl:stringerSpan ?span ;
                          nltl:adjustedIceLoadForce ?adjusted .
                    ?force qudt:numericValue ?forceValue ;
                           qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    ?height qudt:numericValue ?heightValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?span qudt:numericValue ?spanValue ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?adjusted qudt:numericValue ?adjustedValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER (
                        datatype(?forceValue) = xsd:decimal &&
                        datatype(?heightValue) = xsd:decimal &&
                        datatype(?spanValue) = xsd:decimal &&
                        datatype(?adjustedValue) = xsd:decimal &&
                        ?spanValue != 0 &&
                        ?adjustedValue = (?forceValue * (1 - (?heightValue / ?spanValue)))
                    )
                }
            }
        """
    ] .
<END_SHACL>
