gs:WebFrameEffectiveShearAreaShape
    a sh:NodeShape ;
    sh:targetClass nltl:shipComponent ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gs:NonnegativeAreaValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumCalculatedShearForce ;
        sh:node gs:NonnegativeForceValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gs:PositiveYieldStrengthValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webFrameShearFactorAlpha ;
        sh:node gs:NonnegativeUnitlessValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webFrameShearDistributionFactorF13 ;
        sh:node gs:F13ValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freeFlangeToWebAreaRatio ;
        sh:node gs:NonnegativeUnitlessValueShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:freeFlangeArea ;
        sh:node gs:NonnegativeAreaValueShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webPlateArea ;
        sh:node gs:PositiveAreaValueShape ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Alpha selection requires either the flange-to-web area ratio or both flange and web areas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER (
                    NOT EXISTS {
                        $this nltl:freeFlangeToWebAreaRatio ?ratioValue .
                        ?ratioValue qudt:numericValue ?ratio .
                    }
                    &&
                    (
                        NOT EXISTS {
                            $this nltl:freeFlangeArea ?flangeValue .
                            ?flangeValue qudt:numericValue ?flangeArea .
                        }
                        ||
                        NOT EXISTS {
                            $this nltl:webPlateArea ?webValue .
                            ?webValue qudt:numericValue ?webArea .
                        }
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The direct flange-to-web area ratio must equal freeFlangeArea divided by webPlateArea." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:freeFlangeToWebAreaRatio ?ratioValue ;
                      nltl:freeFlangeArea ?flangeValue ;
                      nltl:webPlateArea ?webValue .
                ?ratioValue qudt:numericValue ?ratio .
                ?flangeValue qudt:numericValue ?flangeArea .
                ?webValue qudt:numericValue ?webArea .
                FILTER (?webArea > "0"^^xsd:decimal)
                FILTER (?ratio != (?flangeArea / ?webArea))
            }
        """
    ] ;
    sh:sparql [
        sh:message "webFrameShearFactorAlpha does not match the Table 4-8 flange-to-web area-ratio selection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:webFrameShearFactorAlpha ?alphaValue .
                ?alphaValue qudt:numericValue ?alpha .
                OPTIONAL {
                    $this nltl:freeFlangeToWebAreaRatio ?ratioValue .
                    ?ratioValue qudt:numericValue ?directRatio .
                }
                OPTIONAL {
                    $this nltl:freeFlangeArea ?flangeValue ;
                          nltl:webPlateArea ?webValue .
                    ?flangeValue qudt:numericValue ?flangeArea .
                    ?webValue qudt:numericValue ?webArea .
                    FILTER (?webArea > "0"^^xsd:decimal)
                }
                BIND(
                    IF(
                        BOUND(?directRatio),
                        ?directRatio,
                        ?flangeArea / ?webArea
                    )
                    AS ?ratio
                )
                BIND(
                    IF(
                        ?ratio < "0.1"^^xsd:decimal,
                        "1.5"^^xsd:decimal,
                        "1.0"^^xsd:decimal
                    )
                    AS ?requiredAlpha
                )
                FILTER (?alpha != ?requiredAlpha)
            }
        """
    ] ;
    sh:sparql [
        sh:message "requiredShearArea must satisfy the effective web-frame shear-area formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:requiredShearArea ?areaValue ;
                      nltl:maximumCalculatedShearForce ?forceValue ;
                      nltl:yieldStrength ?yieldValue ;
                      nltl:webFrameShearFactorAlpha ?alphaValue ;
                      nltl:webFrameShearDistributionFactorF13 ?f13Value .
                ?areaValue qudt:numericValue ?area .
                ?forceValue qudt:numericValue ?force .
                ?yieldValue qudt:numericValue ?yieldStrength .
                ?alphaValue qudt:numericValue ?alpha .
                ?f13Value qudt:numericValue ?f13 .
                FILTER (
                    (?area * ?area * ?yieldStrength * ?yieldStrength)
                    !=
                    (
                        "300"^^xsd:decimal
                        * ?alpha * ?alpha
                        * ?f13 * ?f13
                        * ?force * ?force
                    )
                )
            }
        """
    ] .
