<BEGIN_SHACL>
@prefix shape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:TRF063ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Propeller count is required to determine whether the twin- or triple-screw requirement applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A twin- or triple-screw ship must state both shell and framing ice-strengthening extents as strings." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propellerCount ?propellerCount .
    FILTER (?propellerCount IN (2, 3))
    FILTER NOT EXISTS {
        $this nltl:shellIceStrengtheningExtent ?shellExtent ;
              nltl:framingIceStrengtheningExtent ?framingExtent .
        FILTER (
            datatype(?shellExtent) = xsd:string &&
            datatype(?framingExtent) = xsd:string
        )
    }
}
"""
    ] .
<END_SHACL>
