<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF066Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA or IA Super, lowerIceWaterline must have an xsd:string value." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
    FILTER NOT EXISTS {
        $this nltl:lowerIceWaterline ?liwl .
        FILTER (DATATYPE(?liwl) = xsd:string)
    }
}
"""
    ] .
<END_SHACL>
