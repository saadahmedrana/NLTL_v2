<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF069Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable propulsion-machinery case requires maximum ice torque as a decimal QUDT quantity in kilonewton-metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    )
                )
                FILTER (
                    ?propellerType IN (
                        "open"^^xsd:string,
                        "ducted"^^xsd:string
                    )
                )
                FILTER (
                    ?pitchType IN (
                        "controllable"^^xsd:string,
                        "fixed"^^xsd:string
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:maximumIceTorque ?torque .
                    ?torque rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?numericValue ;
                            qudt:unit unit:KiloN-M .
                    FILTER (DATATYPE(?numericValue) = xsd:decimal)
                }
            }
        """
    ] .
<END_SHACL>
