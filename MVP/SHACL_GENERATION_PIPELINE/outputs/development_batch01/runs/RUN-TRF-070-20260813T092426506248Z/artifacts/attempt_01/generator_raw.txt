<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ThrusterIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:node gen:KiloNewtonQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:node gen:KiloNewtonQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:extremeLoad ;
        sh:node gen:KiloNewtonQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:node gen:MegaPascalQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:serviceLife ;
        sh:node gen:YearQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyLocalStrength ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "An applicable thruster ice-load assessment that records either interaction load must record both the propeller and thruster-body interaction loads." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterType ?type .
                FILTER (LCASE(STR(?type)) IN ("azimuthing", "fixed"))
                FILTER (
                    EXISTS { $this nltl:propellerIceInteractionLoad ?propellerLoad }
                    ||
                    EXISTS { $this nltl:thrusterBodyIceInteractionLoad ?bodyLoad }
                )
                FILTER (
                    NOT EXISTS { $this nltl:propellerIceInteractionLoad ?requiredPropellerLoad }
                    ||
                    NOT EXISTS { $this nltl:thrusterBodyIceInteractionLoad ?requiredBodyLoad }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A recorded azimuthing-thruster body ice load requires the ship service life used for that load assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterType ?type ;
                      nltl:thrusterBodyIceInteractionLoad ?bodyLoad .
                FILTER (LCASE(STR(?type)) = "azimuthing")
                FILTER NOT EXISTS {
                    $this nltl:serviceLife ?serviceLife .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A thruster-body extreme-load assessment requires local ice pressure and confirmation of sufficient local strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterType ?type ;
                      nltl:thrusterBodyIceInteractionLoad ?bodyLoad ;
                      nltl:extremeLoad ?extremeLoad .
                FILTER (LCASE(STR(?type)) IN ("azimuthing", "fixed"))
                FILTER (
                    NOT EXISTS {
                        $this nltl:localIcePressure ?localPressure .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:thrusterBodyLocalStrength ?strength .
                        FILTER (?strength = true)
                    }
                )
            }
        """
    ] .

gen:KiloNewtonQuantityValueShape
    a sh:NodeShape ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1
    ] .

gen:MegaPascalQuantityValueShape
    a sh:NodeShape ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1
    ] .

gen:YearQuantityValueShape
    a sh:NodeShape ;
    sh:nodeKind sh:BlankNodeOrIRI ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:YR ;
        sh:minCount 1
    ] .
<END_SHACL>
