<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF084PropellerPitchShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Propeller type is required to determine whether the CP-propeller rule applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A CP propeller requires a boolean indication of whether its bollard-condition pitch is known." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:propellerType ?type .
    FILTER(?type = "CP"^^xsd:string)
    FILTER NOT EXISTS {
        $this nltl:pitchValueKnown ?known .
        FILTER(datatype(?known) = xsd:boolean)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the CP-propeller pitch is known, the bollard-condition MCR pitch must be recorded as a decimal quantity in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:propellerType ?type ;
          nltl:pitchValueKnown ?known .
    FILTER(?type = "CP"^^xsd:string)
    FILTER(?known = true)
    FILTER NOT EXISTS {
        $this nltl:propellerPitchAtMcrBollard ?bollardQuantity .
        ?bollardQuantity qudt:numericValue ?bollardPitch ;
                         qudt:unit unit:M .
        FILTER(datatype(?bollardPitch) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the CP-propeller pitch is not known, the bollard-condition pitch must equal 0.7 times the free-running MCR pitch in metres, within a relative tolerance of 1e-6 and an absolute fallback of 1e-12 metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:propellerType ?type ;
          nltl:pitchValueKnown ?known .
    FILTER(?type = "CP"^^xsd:string)
    FILTER(?known = false)
    FILTER NOT EXISTS {
        $this nltl:propellerPitchAtMcrBollard ?bollardQuantity ;
              nltl:propellerPitchAtMcrFreeRunning ?freeRunningQuantity .
        ?bollardQuantity qudt:numericValue ?bollardPitch ;
                         qudt:unit unit:M .
        ?freeRunningQuantity qudt:numericValue ?freeRunningPitch ;
                             qudt:unit unit:M .
        FILTER(datatype(?bollardPitch) = xsd:decimal)
        FILTER(datatype(?freeRunningPitch) = xsd:decimal)
        BIND("0.7"^^xsd:decimal * ?freeRunningPitch AS ?expectedPitch)
        BIND(
            IF(
                ABS(?expectedPitch) * "0.000001"^^xsd:decimal >
                    "0.000000000001"^^xsd:decimal,
                ABS(?expectedPitch) * "0.000001"^^xsd:decimal,
                "0.000000000001"^^xsd:decimal
            )
            AS ?tolerance
        )
        FILTER(ABS(?bollardPitch - ?expectedPitch) <= ?tolerance)
    }
}
"""
    ] .
<END_SHACL>
