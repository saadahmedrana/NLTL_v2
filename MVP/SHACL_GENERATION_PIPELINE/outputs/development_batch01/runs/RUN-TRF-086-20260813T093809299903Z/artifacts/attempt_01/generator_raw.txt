            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:propellerType ?propellerType .
                FILTER(
                    DATATYPE(?propellerType) = xsd:string &&
                    STR(?propellerType) = "CP"

            }
        """
    ] ;

                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;

                        sh:path qudt:unit ;

                sh:minCount 1 ;
                sh:maxCount 1 ;

                sh:path nltl:propellerPitchAtMcrFreeRunning ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [

                        sh:path qudt:numericValue ;

                sh:message "The bollard-condition pitch must equal 0.7 times the free-running pitch within relative tolerance 1e-6 or absolute tolerance 1e-12 near zero." ;

                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerPitchAtMcrBollard ?bollardQuantity ;

                        ?freeRunningQuantity qudt:numericValue ?freeRunningPitch .

                            >
                            IF(
                                ABS("0.7"^^xsd:decimal * ?freeRunningPitch) * "0.000001"^^xsd:decimal

                            )
                        )
                    }
                """
            ]
        ]
