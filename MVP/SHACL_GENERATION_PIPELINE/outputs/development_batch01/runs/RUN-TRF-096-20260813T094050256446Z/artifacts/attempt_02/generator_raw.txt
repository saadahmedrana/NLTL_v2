<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:MaximumTorqueQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum torque must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KN ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum torque must use the canonical KN unit."
    ] .

gen:EngineStandstillShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:engineStandstillReached ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Engine standstill status is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:engineStandstillReached ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:engineStandstillReached ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:intendedServicePowerAdequacy ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false ;
                sh:message "Intended-service power adequacy must be false when engine standstill is reached."
            ] ;
            sh:property [
                sh:path nltl:maximumTorqueDuringSpeedDrop ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:MaximumTorqueQuantityShape ;
                sh:message "Maximum torque during the speed-drop process is required when engine standstill is reached."
            ]
        ]
    ) .
<END_SHACL>
