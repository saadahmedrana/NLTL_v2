<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FailureLengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal numeric value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:message "The length unit must be metre."
    ] .

gen:BladeFailureStatusShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Blade failure status is required."
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node gen:FailureLengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "Blade tip offset in metres is required."
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:FailureLengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "Propeller diameter in metres is required."
    ] ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:bladeFailureStatus ?status ;
                      nltl:bladeTipOffset ?offsetQuantity ;
                      nltl:propellerDiameter ?diameterQuantity .
                ?offsetQuantity qudt:numericValue ?offset ;
                                qudt:unit unit:M .
                ?diameterQuantity qudt:numericValue ?diameter ;
                                  qudt:unit unit:M .
                FILTER (
                    ?offset > ("0.10"^^xsd:decimal * ?diameter)
                    && ?status != true
                )
            }
        """
    ] .
<END_SHACL>
