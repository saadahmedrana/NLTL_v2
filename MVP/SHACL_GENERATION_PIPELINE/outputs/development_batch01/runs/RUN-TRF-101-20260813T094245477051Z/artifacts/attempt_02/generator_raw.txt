<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthInMetresShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:in ( unit:M )
    ] .

gen:ForceInKilonewtonsShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:in ( unit:KiloN )
    ] .

gen:StressInMegapascalsShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:in ( unit:MegaPA )
    ] .

gen:BladeFailureRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:node gen:LengthInMetresShape ;
        sh:message "Propeller diameter is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:node gen:LengthInMetresShape ;
        sh:message "Blade tip offset is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:node gen:ForceInKilonewtonsShape ;
        sh:message "Blade-failure ultimate load is required in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:node gen:StressInMegapascalsShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:bladeTipOffset ?offsetQuantity ;
                      nltl:propellerDiameter ?diameterQuantity .
                ?offsetQuantity qudt:numericValue ?offset .
                ?diameterQuantity qudt:numericValue ?diameter .
                FILTER (
                    ?offset > (xsd:decimal("0.10") * ?diameter) &&
                    NOT EXISTS {
                        $this nltl:bladeFailureStatus true .
                    }
                )
            }
        """
    ] .
<END_SHACL>
