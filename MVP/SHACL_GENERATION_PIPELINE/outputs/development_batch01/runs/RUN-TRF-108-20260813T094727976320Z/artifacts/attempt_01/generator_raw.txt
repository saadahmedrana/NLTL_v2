<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PropellerBladeSnCurveShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerBladeMaterialProperty ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Propeller blade material properties are required for SN-curve selection."
    ] ;
    sh:property [
        sh:path nltl:stressLifeCurveType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:twoSlopeSnCurveUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:stressLifeCurveType ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:twoSlopeSnCurveUsed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The two-slope SN curve must be used when the SN-curve type is unknown." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:stressLifeCurveType ?curveType .
    FILTER(LCASE(STR(?curveType)) = "unknown")
    FILTER NOT EXISTS {
        $this nltl:twoSlopeSnCurveUsed ?used .
        FILTER(?used = true)
    }
}
"""
    ] .
<END_SHACL>
