<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF132Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Ships of ice class IB or IC must have maximum continuous rating power of at least 740 kW, represented as a decimal QUDT QuantityValue in kilowatts." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?power .
                    ?power a qudt:QuantityValue ;
                           qudt:numericValue ?value ;
                           qudt:unit unit:KiloW .
                    FILTER (
                        isLiteral(?value) &&
                        datatype(?value) = xsd:decimal &&
                        ?value >= "740"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
