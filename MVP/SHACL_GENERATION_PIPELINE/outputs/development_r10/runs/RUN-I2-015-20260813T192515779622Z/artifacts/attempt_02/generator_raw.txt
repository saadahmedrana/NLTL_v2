<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PositiveMetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive "0"^^xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:class qudt:QuantityValue ;

gen:BowSubregionCalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:bowSubregionCalculationCase ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:bowSubregionMidLengthPosition ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked bow subregion calculation case requires one mid-length position in metres."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionWaterlineAngle ;
        sh:node gen:DegreeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked bow subregion calculation case requires one waterline angle in degrees."
    ] ;
    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:node gen:PositiveUnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked bow subregion calculation case requires one positive crushing failure class factor."
    ] ;
    sh:property [
        sh:path nltl:flexuralFailureClassFactor ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked bow subregion calculation case requires one flexural failure class factor."
    ] ;
    sh:property [
        sh:path nltl:bowShapeCoefficient ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked bow subregion calculation case requires one reported bow shape coefficient."
    ] .

gen:I2_015ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasBowSubregionCalculationCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasBowSubregionCalculationCase ;
                sh:minCount 1 ;
                sh:node gen:BowSubregionCalculationCaseShape
            ] ;
            sh:property [
                sh:path nltl:betaIPrime ;
                sh:node gen:PositiveUnitlessQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A ship with a bow subregion calculation case requires one positive beta i prime value."
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineLengthLUI ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A ship with a bow subregion calculation case requires one positive upper ice waterline length."
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineDraughtDUI ;
                sh:node gen:PositiveMetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A ship with a bow subregion calculation case requires one positive upper ice waterline draught."
            ] ;
            sh:sparql [
                sh:message "The bow shape coefficient must equal min(f_ai1, f_ai2, 0.60) within relative tolerance 1e-6 or absolute tolerance 1e-12 near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case ;
                              nltl:betaIPrime ?betaQuantity ;
                              nltl:upperIceWaterlineLengthLUI ?lengthQuantity ;
                              nltl:upperIceWaterlineDraughtDUI ?draughtQuantity .

                        ?case a nltl:bowSubregionCalculationCase ;
                              a nltl:tableLookupCase ;
                              nltl:bowSubregionMidLengthPosition ?positionQuantity ;
                              nltl:bowSubregionWaterlineAngle ?angleQuantity ;
                              nltl:crushingFailureClassFactor ?crushingQuantity ;
                              nltl:flexuralFailureClassFactor ?flexuralQuantity ;
                              nltl:bowShapeCoefficient ?reportedQuantity .

                        ?betaQuantity qudt:numericValue ?beta ;
                                      qudt:unit unit:UNITLESS .
                        ?lengthQuantity qudt:numericValue ?length ;
                                        qudt:unit unit:M .
                        ?draughtQuantity qudt:numericValue ?draught ;
                                         qudt:unit unit:M .
                        ?positionQuantity qudt:numericValue ?position ;
                                          qudt:unit unit:M .
                        ?angleQuantity qudt:numericValue ?angle ;
                                       qudt:unit unit:DEG .
                        ?crushingQuantity qudt:numericValue ?crushing ;
                                           qudt:unit unit:UNITLESS .
                        ?flexuralQuantity qudt:numericValue ?flexural ;
                                           qudt:unit unit:UNITLESS .
                        ?reportedQuantity qudt:numericValue ?reported ;
                                          qudt:unit unit:UNITLESS .

                        BIND(
                            ("0.097"^^xsd:decimal -
                             "0.68"^^xsd:decimal *
                             ((?position / ?length) - "0.15"^^xsd:decimal) *
                             ((?position / ?length) - "0.15"^^xsd:decimal)) *
                            ?angle /
                            math:pow(?beta, "0.5"^^xsd:decimal)
                            AS ?fai1
                        )
                        BIND(
                            "1.2"^^xsd:decimal * ?flexural /
                            (math:sin(?beta) * ?crushing *
                             math:pow(?draught, "0.64"^^xsd:decimal))
                            AS ?fai2
                        )
                        BIND(
                            IF(
                                ?fai1 <= ?fai2 && ?fai1 <= "0.60"^^xsd:decimal,
                                ?fai1,
                                IF(
                                    ?fai2 <= "0.60"^^xsd:decimal,
                                    ?fai2,
                                    "0.60"^^xsd:decimal
                                )
                            )
                            AS ?expected
                        )
                        BIND(
                            IF(
                                ABS(?expected) * "0.000001"^^xsd:decimal >
                                "0.000000000001"^^xsd:decimal,
                                ABS(?expected) * "0.000001"^^xsd:decimal,
                                "0.000000000001"^^xsd:decimal
                            )
                            AS ?tolerance
                        )
                        FILTER(ABS(?reported - ?expected) > ?tolerance)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "The beta i prime value must give a non-zero sine for the f_ai2 calculation." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case ;
                              nltl:betaIPrime ?betaQuantity .
                        ?case a nltl:bowSubregionCalculationCase .
                        ?betaQuantity qudt:numericValue ?beta ;
                                      qudt:unit unit:UNITLESS .
                        FILTER(math:sin(?beta) = "0"^^xsd:decimal)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
