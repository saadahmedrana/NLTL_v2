@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:UnitlessDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:class nltl:hullAreaValue
    ] ;
    sh:property [
        sh:path nltl:spansHullArea ;
        sh:class nltl:hullRegionValue
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:node gs:UnitlessDecimalQuantityShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:spansHullArea ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 2
            ] ;
            sh:property [
                sh:path nltl:spansHullArea ;
                sh:minCount 2
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .

gs:I2_019_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:node gs:UnitlessDecimalQuantityShape
    ] ;
    sh:sparql [
        sh:message "A member spanning multiple hull areas and regions must use the largest available hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasSpannedHullArea ?area1 ;
                        nltl:hasSpannedHullArea ?area2 ;
                        nltl:spansHullArea ?region1 ;
                        nltl:spansHullArea ?region2 .
                FILTER (?area1 != ?area2)
                FILTER (?region1 != ?region2)
                FILTER NOT EXISTS {
                    $this nltl:hullAreaFactor ?largestNode .
                    ?largestNode qudt:numericValue ?largest ;
                                 qudt:unit unit:UNITLESS .
                    ?member nltl:selectedHullAreaFactor ?selectedNode .
                    ?selectedNode qudt:numericValue ?selected ;
                                  qudt:unit unit:UNITLESS .
                    FILTER (?selected = ?largest)
                    FILTER NOT EXISTS {
                        $this nltl:hullAreaFactor ?otherNode .
                        ?otherNode qudt:numericValue ?other ;
                                   qudt:unit unit:UNITLESS .
                        FILTER (?other > ?largest)
                    }
                }
            }
        """
    ] .
<END_SHACL>
