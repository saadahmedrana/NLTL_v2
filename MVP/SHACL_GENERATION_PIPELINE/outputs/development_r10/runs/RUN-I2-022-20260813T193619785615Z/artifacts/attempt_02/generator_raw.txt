<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipDataShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:MegaPascalQuantityShape
    ] .

gen:PlatingDataShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:node gen:DegreeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:node gen:MillimetreQuantityShape
    ] .

gen:StructuralMemberDataShape
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:node gen:UnitlessQuantityShape
    ] .

gen:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1
    ] .
<END_SHACL>
