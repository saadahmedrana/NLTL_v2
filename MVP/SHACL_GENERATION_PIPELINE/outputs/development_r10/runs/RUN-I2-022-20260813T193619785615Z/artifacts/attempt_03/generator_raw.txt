@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2-022-ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:framingAngleOmega ;
                sh:minCount 1 ;
                sh:node gs:AngleAtLeast70Quantity
            ] ;
            sh:property [
                sh:path nltl:hullArea ;
                sh:minCount 1 ;
                sh:datatype xsd:string ;
                sh:in ( "BI_b" "M_b" "S_b" )
            ] ;
            sh:property [
                sh:path nltl:frameSpacing ;
                sh:minCount 1 ;
                sh:node gs:NonNegativeMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:loadPatchHeight ;
                sh:minCount 1 ;
                sh:node gs:PositiveMetreQuantity
            ] ;
            sh:property [
                sh:path nltl:peakPressureFactor ;
                sh:minCount 1 ;
                sh:node gs:NonNegativeUnitlessQuantity
            ] ;
            sh:property [
                sh:path nltl:averageIcePressure ;
                sh:minCount 1 ;
                sh:node gs:NonNegativeMegaPascalQuantity
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:node gs:PositiveMegaPascalQuantity
            ] ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:node gs:NonNegativeMillimetreQuantity
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:qualifiedValueShape [
                    sh:class nltl:structuralMember ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:minCount 1 ;
                        sh:node gs:NonNegativeUnitlessQuantity
                    ]
                ] ;
                sh:qualifiedMinCount 1
            ] ;
            sh:sparql [
                sh:message "For Omega >= 70 degrees in hull areas BI_b, M_b, or S_b, t_net must satisfy the stated formula within a relative tolerance of 1e-6, with a 1e-9 mm fallback near zero." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:hullArea ?area ;
                              nltl:framingAngleOmega ?omegaQuantity ;
                              nltl:frameSpacing ?spacingQuantity ;
                              nltl:loadPatchHeight ?heightQuantity ;
                              nltl:peakPressureFactor ?peakFactorQuantity ;
                              nltl:averageIcePressure ?pressureQuantity ;
                              nltl:yieldStrength ?yieldQuantity ;
                              nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity ;
                              nltl:hasStructuralMember ?member .
                        ?omegaQuantity qudt:numericValue ?omega .
                        ?spacingQuantity qudt:numericValue ?spacing .
                        ?heightQuantity qudt:numericValue ?height .
                        ?peakFactorQuantity qudt:numericValue ?peakFactor .
                        ?pressureQuantity qudt:numericValue ?pressure .
                        ?yieldQuantity qudt:numericValue ?yieldStrength .
                        ?thicknessQuantity qudt:numericValue ?reportedThickness .
                        ?member nltl:selectedHullAreaFactor ?areaFactorQuantity .
                        ?areaFactorQuantity qudt:numericValue ?areaFactor .
                        FILTER (
                            ?omega >= "70"^^xsd:decimal &&
                            ?area IN ("BI_b", "M_b", "S_b")
                        )
                        BIND (
                            ("500"^^xsd:decimal * ?spacing) /
                            ("1"^^xsd:decimal + ?spacing / ("2"^^xsd:decimal * ?height))
                            AS ?coefficient
                        )
                        BIND (
                            ?coefficient * ?coefficient *
                            ((?areaFactor * ?peakFactor * ?pressure) / ?yieldStrength)
                            AS ?expectedSquared
                        )
                        BIND (?reportedThickness * ?reportedThickness AS ?reportedSquared)
                        FILTER (
                            ?reportedThickness < "0"^^xsd:decimal ||
                            (
                                ?expectedSquared > "0.000000000000000001"^^xsd:decimal &&
                                (
                                    ?reportedSquared <
                                        ?expectedSquared *
                                        "0.999998000001"^^xsd:decimal ||
                                    ?reportedSquared >
                                        ?expectedSquared *
                                        "1.000002000001"^^xsd:decimal
                                )
                            ) ||
                            (
                                ?expectedSquared <= "0.000000000000000001"^^xsd:decimal &&
                                ABS(?reportedThickness) > "0.000000001"^^xsd:decimal
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:framingAngleOmega ;
                sh:minCount 1 ;
                sh:node gs:AngleBelow70Quantity
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullArea ;
                sh:minCount 1 ;
                sh:datatype xsd:string ;
                sh:pattern "^(?!(BI_b|M_b|S_b)$).+"
            ]
        ]
    ) .
