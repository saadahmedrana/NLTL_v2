    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .

gen:I2_023_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gen:MegapascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:MegapascalQuantityShape
    ] ;
    sh:property [
        sh:path ( nltl:hasStructuralMember nltl:frameSpacing ) ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path ( nltl:hasStructuralMember nltl:selectedHullAreaFactor ) ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:framingAngleOmega ) ;
        sh:node gen:DegreeQuantityShape
    ] ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:iceLoadRequiredNetPlateThickness ) ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "Each linked plating needs a decimal framing angle in degrees to determine whether the longitudinally-framed formula applies." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?plate .
                ?plate a nltl:plating .
                FILTER NOT EXISTS {
                    ?plate nltl:framingAngleOmega ?omegaQuantity .
                    ?omegaQuantity qudt:numericValue ?omega ;
                                   qudt:unit unit:DEG .
                    FILTER(datatype(?omega) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Applicable plating lacks a complete, unit-correct, and numerically calculable I2.4.2 input or net-thickness result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?plate .
                ?plate a nltl:plating ;
                       nltl:framingAngleOmega ?omegaQuantity .
                ?omegaQuantity qudt:numericValue ?omega ;
                               qudt:unit unit:DEG .
                FILTER(datatype(?omega) = xsd:decimal && ?omega <= 20)

                OPTIONAL {
                    $this nltl:averageIcePressure ?pressureQuantity ;
                          nltl:frameSpan ?spanQuantity ;
                          nltl:hullAreaFactor ?hullAreaFactorQuantity ;
                          nltl:loadPatchHeight ?heightQuantity ;
                          nltl:peakPressureFactor ?peakFactorQuantity ;
                          nltl:yieldStrength ?yieldQuantity ;
                          nltl:hasStructuralMember ?member .

                    ?pressureQuantity qudt:numericValue ?pressure ;
                                      qudt:unit unit:MegaPA .
                    ?spanQuantity qudt:numericValue ?span ;
                                  qudt:unit unit:M .
                    ?hullAreaFactorQuantity qudt:numericValue ?hullAreaFactor ;
                                            qudt:unit unit:UNITLESS .
                    ?heightQuantity qudt:numericValue ?height ;
                                    qudt:unit unit:M .
                    ?peakFactorQuantity qudt:numericValue ?peakFactor ;
                                        qudt:unit unit:UNITLESS .
                    ?yieldQuantity qudt:numericValue ?yieldStrength ;
                                   qudt:unit unit:MegaPA .

                    ?member a nltl:structuralMember ;
                            nltl:frameSpacing ?spacingQuantity ;
                            nltl:selectedHullAreaFactor ?selectedAreaFactorQuantity .
                    ?spacingQuantity qudt:numericValue ?spacing ;
                                     qudt:unit unit:M .
                    ?selectedAreaFactorQuantity qudt:numericValue ?selectedAreaFactor ;
                                                qudt:unit unit:UNITLESS .

                    ?plate nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
                    ?thicknessQuantity qudt:numericValue ?thickness ;
                                       qudt:unit unit:MilliM .

                    FILTER(
                        datatype(?pressure) = xsd:decimal &&
                        datatype(?span) = xsd:decimal &&
                        datatype(?hullAreaFactor) = xsd:decimal &&
                        datatype(?height) = xsd:decimal &&
                        datatype(?peakFactor) = xsd:decimal &&
                        datatype(?yieldStrength) = xsd:decimal &&
                        datatype(?spacing) = xsd:decimal &&
                        datatype(?selectedAreaFactor) = xsd:decimal &&
                        datatype(?thickness) = xsd:decimal &&
                        ?yieldStrength > 0 &&
                        ?spacing > 0 &&
                        ?span != 0 &&
                        (1 + ?spacing / (2 * ?span)) != 0 &&
                        (?selectedAreaFactor * ?peakFactor * ?pressure) >= 0 &&
                        (
                            ?height >= ?spacing ||
                            (
                                ?height < ?spacing &&
                                (2 * ?height / ?spacing -
                                 (?height / ?spacing) * (?height / ?spacing)) >= 0
                            )
                        )
                    )
                }
                FILTER(!BOUND(?thickness))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Net plate thickness does not satisfy the applicable I2.4.2 formula within an approximately 1e-6 relative thickness tolerance, evaluated through the equivalent squared formula with a 1e-12 absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?plate ;
                      nltl:averageIcePressure ?pressureQuantity ;
                      nltl:frameSpan ?spanQuantity ;
                      nltl:hullAreaFactor ?hullAreaFactorQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:peakPressureFactor ?peakFactorQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .

                ?plate a nltl:plating ;
                       nltl:framingAngleOmega ?omegaQuantity ;
                       nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
                ?omegaQuantity qudt:numericValue ?omega ;
                               qudt:unit unit:DEG .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                                   qudt:unit unit:MilliM .

                ?pressureQuantity qudt:numericValue ?pressure ;
                                  qudt:unit unit:MegaPA .
                ?spanQuantity qudt:numericValue ?span ;
                              qudt:unit unit:M .
                ?hullAreaFactorQuantity qudt:numericValue ?hullAreaFactor ;
                                        qudt:unit unit:UNITLESS .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit unit:M .
                ?peakFactorQuantity qudt:numericValue ?peakFactor ;
                                    qudt:unit unit:UNITLESS .
                ?yieldQuantity qudt:numericValue ?yieldStrength ;
                               qudt:unit unit:MegaPA .

                ?member a nltl:structuralMember ;
                        nltl:frameSpacing ?spacingQuantity ;
                        nltl:selectedHullAreaFactor ?selectedAreaFactorQuantity .
                ?spacingQuantity qudt:numericValue ?spacing ;
                                 qudt:unit unit:M .
                ?selectedAreaFactorQuantity qudt:numericValue ?selectedAreaFactor ;
                                            qudt:unit unit:UNITLESS .

                FILTER(
                    datatype(?omega) = xsd:decimal &&
                    datatype(?thickness) = xsd:decimal &&
                    datatype(?pressure) = xsd:decimal &&
                    datatype(?span) = xsd:decimal &&
                    datatype(?hullAreaFactor) = xsd:decimal &&
                    datatype(?height) = xsd:decimal &&
                    datatype(?peakFactor) = xsd:decimal &&
                    datatype(?yieldStrength) = xsd:decimal &&
                    datatype(?spacing) = xsd:decimal &&
                    datatype(?selectedAreaFactor) = xsd:decimal &&
                    ?omega <= 20 &&
                    ?yieldStrength > 0 &&
                    ?spacing > 0 &&
                    ?span != 0 &&
                    (1 + ?spacing / (2 * ?span)) != 0 &&
                    (?selectedAreaFactor * ?peakFactor * ?pressure) >= 0 &&
                    (
                        ?height >= ?spacing ||
                        (
                            ?height < ?spacing &&
                            (2 * ?height / ?spacing -
                             (?height / ?spacing) * (?height / ?spacing)) >= 0
                        )
                    )
                )

                BIND(
                    (
                        (500 * ?spacing / (1 + ?spacing / (2 * ?span))) *
                        (500 * ?spacing / (1 + ?spacing / (2 * ?span))) *
                        (?selectedAreaFactor * ?peakFactor * ?pressure / ?yieldStrength) *
                        IF(
                            ?height >= ?spacing,
                            1,
                            2 * ?height / ?spacing -
                            (?height / ?spacing) * (?height / ?spacing)
                        )
                    )
                    AS ?expectedSquared
                )

                FILTER(
                    ?thickness < 0 ||
                    ABS((?thickness * ?thickness) - ?expectedSquared) >
                    (0.000002 * ABS(?expectedSquared) + 0.000000000001)
                )
            }
        """
    ] .
