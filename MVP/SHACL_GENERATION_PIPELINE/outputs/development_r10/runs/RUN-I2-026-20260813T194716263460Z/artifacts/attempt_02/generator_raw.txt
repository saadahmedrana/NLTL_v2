@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        ) ;
        sh:message "Each structural member must have exactly one permitted frame support condition."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The member termination condition must be recorded as one boolean value."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026_MemberSupportShape
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:continuousThroughSupport ;
                        sh:minCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasMemberSupport ;
                        sh:qualifiedValueShape [
                            sh:class nltl:memberSupport ;
                            sh:property [
                                sh:path nltl:connectionBracketPresent ;
                                sh:minCount 1 ;
                                sh:hasValue true
                            ]
                        ] ;
                        sh:qualifiedMinCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasMemberSupport ;
                        sh:qualifiedValueShape [
                            sh:class nltl:memberSupport ;
                            sh:property [
                                sh:path nltl:demonstratedRotationalRestraint ;
                                sh:minCount 1 ;
                                sh:hasValue true
                            ]
                        ] ;
                        sh:qualifiedMinCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:significantRotationalRestraintEvidence ;
                        sh:class nltl:evidenceArtifact ;
                        sh:minCount 1
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:simpleFrameSupport
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:minCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:class nltl:memberSupport ;
                sh:minCount 1 ;
                sh:node gen:I2_026_SimpleMemberSupportShape
            ]
        ]
    ) ;
    sh:message "The frame support condition is inconsistent with the member continuity, support details, restraint evidence, or ice-strengthened-area termination condition." .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .

gen:I2_026_SimpleMemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] .
<END_SHACL>
