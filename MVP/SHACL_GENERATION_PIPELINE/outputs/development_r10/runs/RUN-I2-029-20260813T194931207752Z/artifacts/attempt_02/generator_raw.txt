<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:node gen:I2_029_ApplicableShipShape
        ]
    ) .

gen:I2_029_ApplicableShipShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_029_CalculationCaseShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_029_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:node gen:I2_029_MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:message "Net web thickness in millimetres is required for the calculation."
    ] ;
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:node gen:I2_029_MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:message "Net attached shell plate thickness in millimetres is required for the calculation."
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:I2_029_SectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:message "Section modulus in cubic centimetres is required."
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:node gen:I2_029_ShearAreaQuantityShape ;
        sh:minCount 1 ;
        sh:message "Shear area in square metres is required."
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:node gen:I2_029_MetreQuantityShape ;
        sh:minCount 1 ;
        sh:message "Thickness in metres is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:node [
                    sh:property [
                        sh:path nltl:flangeFitted ;
                        sh:hasValue false
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:flangeFitted ;
                        sh:hasValue true
                    ]
                ] ;
                sh:qualifiedMinCount 1
            ] ;
            sh:property [
                sh:path nltl:netFlangeThickness ;
                sh:node gen:I2_029_MillimetreQuantityShape ;
                sh:minCount 1 ;
                sh:message "Net flange thickness in millimetres is required when a linked structural member has a flange fitted."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Flange material may be included in shear area only when a linked structural member has a flange fitted." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?case
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:flangeMaterialIncludedInShearArea ?included .
    FILTER (?included = true)
    FILTER NOT EXISTS {
        $this nltl:hasStructuralMember ?member .
        ?member nltl:flangeFitted ?fitted .
        FILTER (?fitted = true)
    }
}
"""
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must include attached shell plating in section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The calculation case must exclude attached shell plating from shear area."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean
    ] .

gen:I2_029_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A linked structural member must state whether a flange is fitted."
    ] .

gen:I2_029_MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_029_SectionModulusQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_029_ShearAreaQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_029_MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
