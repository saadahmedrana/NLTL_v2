    sh:sparql [
        sh:message "Framing angle Omega is required when the local-frame area exceeds the attached-plate-flange area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:netLocalFrameFlangeArea ?afnQuantity ;
                      nltl:netAttachedShellPlateThickness ?tpnQuantity ;
                      nltl:frameSpacing ?spacingQuantity .
                ?afnQuantity qudt:numericValue ?afn ;
                             qudt:unit unit:CentiM2 .
                ?tpnQuantity qudt:numericValue ?tpn ;
                             qudt:unit unit:MilliM .
                ?spacingQuantity qudt:numericValue ?spacing ;
                                 qudt:unit unit:M .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                FILTER NOT EXISTS {
                    $this nltl:framingAngleOmega ?omegaQuantity .
                    ?omegaQuantity qudt:numericValue ?omega ;
                                   qudt:unit unit:DEG .
                    FILTER (datatype(?omega) = xsd:decimal)
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "The direct plastic-section calculation requires the complete ship-owned operand set and a linked calculation case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:netLocalFrameFlangeArea ?outerAfnQuantity ;
                      nltl:netAttachedShellPlateThickness ?outerTpnQuantity ;
                      nltl:frameSpacing ?outerSpacingQuantity ;
                      nltl:framingAngleOmega ?outerOmegaQuantity .
                ?outerAfnQuantity qudt:numericValue ?outerAfn ;
                                  qudt:unit unit:CentiM2 .
                ?outerTpnQuantity qudt:numericValue ?outerTpn ;
                                  qudt:unit unit:MilliM .
                ?outerSpacingQuantity qudt:numericValue ?outerSpacing ;
                                      qudt:unit unit:M .
                ?outerOmegaQuantity qudt:numericValue ?outerOmega ;
                                    qudt:unit unit:DEG .
                FILTER ((100 * ?outerAfn) > (1000 * ?outerTpn * ?outerSpacing))
                FILTER (?outerOmega <= 20 || ?outerOmega >= 70)
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?case ;
                          nltl:frameSpacing ?spacingQuantity ;
                          nltl:framingAngleOmega ?omegaQuantity ;
                          nltl:localFrameFlangeCentreHeight ?hfcQuantity ;
                          nltl:netAttachedShellPlateThickness ?tpnQuantity ;

                          nltl:webAngleToShellPlate ?phiQuantity ;
                          nltl:webHeight ?hwQuantity ;
                          nltl:webToFlangeCentreDistance ?bwQuantity .
                    ?case a nltl:calculationCase .
                    ?spacingQuantity qudt:numericValue ?spacing ;

    sh:sparql [
        sh:message "The reported plastic neutral axis and net effective plastic section modulus do not satisfy the direct formulas within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:netLocalFrameFlangeArea ?outerAfnQuantity ;
                      nltl:netAttachedShellPlateThickness ?outerTpnQuantity ;
                      nltl:frameSpacing ?outerSpacingQuantity ;
                      nltl:framingAngleOmega ?outerOmegaQuantity .
                ?outerAfnQuantity qudt:numericValue ?outerAfn ;
                                  qudt:unit unit:CentiM2 .
                ?outerTpnQuantity qudt:numericValue ?outerTpn ;
                                  qudt:unit unit:MilliM .
                ?outerSpacingQuantity qudt:numericValue ?outerSpacing ;
                                      qudt:unit unit:M .
                ?outerOmegaQuantity qudt:numericValue ?outerOmega ;
                                    qudt:unit unit:DEG .
                FILTER ((100 * ?outerAfn) > (1000 * ?outerTpn * ?outerSpacing))
                FILTER (?outerOmega <= 20 || ?outerOmega >= 70)
                FILTER EXISTS {
                    $this nltl:netWebThickness ?availableTwnQuantity .
                    ?availableTwnQuantity qudt:numericValue ?availableTwn ;
                                          qudt:unit unit:MilliM .
                    FILTER (?availableTwn != 0)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?case ;
                          nltl:frameSpacing ?spacingQuantity ;
                          nltl:framingAngleOmega ?omegaQuantity ;
                          nltl:localFrameFlangeCentreHeight ?hfcQuantity ;
                          nltl:netAttachedShellPlateThickness ?tpnQuantity ;
                          nltl:netEffectivePlasticSectionModulus ?zpQuantity ;
                          nltl:netLocalFrameFlangeArea ?afnQuantity ;
                          nltl:netWebThickness ?twnQuantity ;
                          nltl:plasticNeutralAxisHeight ?znaQuantity ;
                          nltl:webAngleToShellPlate ?phiQuantity ;
                          nltl:webHeight ?hwQuantity ;
                          nltl:webToFlangeCentreDistance ?bwQuantity .
                    ?case a nltl:calculationCase .
                    ?spacingQuantity qudt:numericValue ?spacing ;
                                     qudt:unit unit:M .
                    ?omegaQuantity qudt:numericValue ?omega ;
                                   qudt:unit unit:DEG .
                    ?hfcQuantity qudt:numericValue ?hfc ;
                                 qudt:unit unit:MilliM .
                    ?tpnQuantity qudt:numericValue ?tpn ;
                                 qudt:unit unit:MilliM .
                    ?zpQuantity qudt:numericValue ?zp ;
                                qudt:unit unit:CentiM3 .
                    ?afnQuantity qudt:numericValue ?afn ;
                                 qudt:unit unit:CentiM2 .
                    ?twnQuantity qudt:numericValue ?twn ;
                                 qudt:unit unit:MilliM .
                    ?znaQuantity qudt:numericValue ?zna ;
                                 qudt:unit unit:MilliM .
                    ?phiQuantity qudt:numericValue ?phi ;
                                 qudt:unit unit:DEG .
                    ?hwQuantity qudt:numericValue ?hw ;
                                qudt:unit unit:MilliM .
                    ?bwQuantity qudt:numericValue ?bw ;
                                qudt:unit unit:MilliM .
                    FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                    FILTER (?omega <= 20 || ?omega >= 70)
                    FILTER (?twn != 0)
                    BIND (
                        (100 * ?afn + ?hw * ?twn - 1000 * ?tpn * ?spacing)
                        / (2 * ?twn)
                        AS ?expectedZna
                    )
                    BIND (
                        (?phi * 3.1415926535897932384626433832795) / 180
                        AS ?phiRadians
                    )
                    BIND (
                        ?tpn * ?spacing * (?expectedZna + ?tpn / 2)
                            * math:sin(?phiRadians)
                        + (
                            (
                                ((?hw - ?expectedZna) * (?hw - ?expectedZna))
                                + (?expectedZna * ?expectedZna)
                            )
                            * ?twn
                            * math:sin(?phiRadians)
                          ) / 2000
                        + ?afn * (
                            (?hfc - ?expectedZna) * math:sin(?phiRadians)
                            - ?bw * math:cos(?phiRadians)
                          ) / 10
                        AS ?expectedZp
                    )
                    BIND (
                        IF (
                            ABS(?expectedZna) * 0.000001 > 0.000000001,
                            ABS(?expectedZna) * 0.000001,
                            0.000000001
                        )
                        AS ?znaTolerance
                    )
                    BIND (
                        IF (
                            ABS(?expectedZp) * 0.000001 > 0.000000001,
                            ABS(?expectedZp) * 0.000001,
                            0.000000001
                        )
                        AS ?zpTolerance
                    )
                    FILTER (
                        ABS(?zna - ?expectedZna) <= ?znaTolerance
                        && ABS(?zp - ?expectedZp) <= ?zpTolerance
                    )
                }
            }
        """
    ] .
