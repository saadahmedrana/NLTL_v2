gs:ShipI2031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:structuralMember
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:frameProfileTypeValue
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gs:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node gs:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A linked local frame requires structural-location evidence to determine whether I2.6.1 applies." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame .
    FILTER NOT EXISTS {
        ?member nltl:structuralLocation ?location .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable bottom or side local frame requires a midspan plastic collapse load." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame ;
            nltl:structuralLocation ?location .
    FILTER (?location IN (nltl:bottomStructureLocation, nltl:sideStructureLocation))
    FILTER NOT EXISTS {
        ?member nltl:midspanPlasticCollapseLoad ?collapseLoad .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an applicable bottom or side local frame requires plastic-strength evidence." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame ;
            nltl:structuralLocation ?location .
    FILTER (?location IN (nltl:bottomStructureLocation, nltl:sideStructureLocation))
    FILTER NOT EXISTS {
        $this nltl:plasticStrength ?strength .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an applicable bottom or side local frame requires plastic-collapse-mechanism evidence." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame ;
            nltl:structuralLocation ?location .
    FILTER (?location IN (nltl:bottomStructureLocation, nltl:sideStructureLocation))
    FILTER NOT EXISTS {
        $this nltl:plasticCollapseMechanism ?mechanism .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an applicable bottom or side local frame requires combined shear-and-bending effect evidence." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame ;
            nltl:structuralLocation ?location .
    FILTER (?location IN (nltl:bottomStructureLocation, nltl:sideStructureLocation))
    FILTER NOT EXISTS {
        $this nltl:combinedShearBendingEffect ?effect .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an applicable bottom local frame requires patch-load evidence." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:localFrame ;
            nltl:structuralLocation nltl:bottomStructureLocation .

    }
}
"""
    ] .
