                ?heightQuantity qudt:numericValue ?height .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?areaFactorQuantity qudt:numericValue ?areaFactor .
                ?pressureFactorQuantity qudt:numericValue ?pressureFactor .

                ?yieldQuantity qudt:numericValue ?yieldStrength .
                ?requiredQuantity qudt:numericValue ?requiredArea .
                ?actualQuantity qudt:numericValue ?actualArea .

                    IF(?span <= ?height, ?span, ?height)
                    AS ?limitingLength

                BIND(
                    (
                        "10000"^^xsd:decimal *
                        "0.5"^^xsd:decimal *

                    (?actualArea * "10000"^^xsd:decimal) < ?requiredArea
