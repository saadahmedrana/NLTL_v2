gs:I2_032_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:node gs:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:node gs:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gs:PositiveMegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:node gs:SquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gs:TransverseFrameShape
    ] ;
    sh:sparql [
        sh:message "For each linked structural member, required transverse frame shear area must match the I2.6.2 formula within relative tolerance 1e-6 (absolute fallback 1e-9 cm2), and actual shear area must be at least the calculated requirement." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:span ?spanQuantity ;
          nltl:loadPatchHeight ?heightQuantity ;
          nltl:averageIcePressure ?pressureQuantity ;
          nltl:peakPressureFactor ?pressureFactorQuantity ;
          nltl:yieldStrength ?yieldQuantity ;
          nltl:shearArea ?actualAreaQuantity ;
          nltl:hasStructuralMember ?member .

    ?spanQuantity qudt:numericValue ?spanRaw .
    ?heightQuantity qudt:numericValue ?heightRaw .
    ?pressureQuantity qudt:numericValue ?pressureRaw .
    ?pressureFactorQuantity qudt:numericValue ?pressureFactorRaw .
    ?yieldQuantity qudt:numericValue ?yieldRaw .
    ?actualAreaQuantity qudt:numericValue ?actualAreaRaw .

    ?member nltl:frameSpacing ?spacingQuantity ;
            nltl:selectedHullAreaFactor ?areaFactorQuantity ;
            nltl:requiredTransverseFrameShearArea ?requiredAreaQuantity .

    ?spacingQuantity qudt:numericValue ?spacingRaw .
    ?areaFactorQuantity qudt:numericValue ?areaFactorRaw .
    ?requiredAreaQuantity qudt:numericValue ?requiredAreaRaw .

    BIND(xsd:decimal(?spanRaw) AS ?frameSpan)
    BIND(xsd:decimal(?heightRaw) AS ?loadPatchHeight)
    BIND(xsd:decimal(?pressureRaw) AS ?averagePressure)
    BIND(xsd:decimal(?pressureFactorRaw) AS ?peakPressureFactor)
    BIND(xsd:decimal(?yieldRaw) AS ?yieldStrength)
    BIND(xsd:decimal(?actualAreaRaw) AS ?actualAreaM2)
    BIND(xsd:decimal(?spacingRaw) AS ?frameSpacing)
    BIND(xsd:decimal(?areaFactorRaw) AS ?selectedAreaFactor)
    BIND(xsd:decimal(?requiredAreaRaw) AS ?reportedRequiredAreaCm2)

    FILTER(?yieldStrength > 0)

    BIND(
        IF(
            ?frameSpan <= ?loadPatchHeight,
            ?frameSpan,
            ?loadPatchHeight
        )
        AS ?effectiveLength
    )

    BIND(
        (
            "10000"^^xsd:decimal *
            "0.5"^^xsd:decimal *
            ?effectiveLength *
            ?frameSpacing *
            (?selectedAreaFactor * ?peakPressureFactor * ?averagePressure)
        ) /
        ("0.577"^^xsd:decimal * ?yieldStrength)
        AS ?expectedRequiredAreaCm2
    )

    BIND(
        IF(
            ABS(?expectedRequiredAreaCm2) * "0.000001"^^xsd:decimal >
            "0.000000001"^^xsd:decimal,
            ABS(?expectedRequiredAreaCm2) * "0.000001"^^xsd:decimal,
            "0.000000001"^^xsd:decimal
        )
        AS ?formulaToleranceCm2
    )

    FILTER(
        ABS(?reportedRequiredAreaCm2 - ?expectedRequiredAreaCm2) >
            ?formulaToleranceCm2 ||
        (?actualAreaM2 * "10000"^^xsd:decimal) <
            ?expectedRequiredAreaCm2
    )
}
"""
    ] .
