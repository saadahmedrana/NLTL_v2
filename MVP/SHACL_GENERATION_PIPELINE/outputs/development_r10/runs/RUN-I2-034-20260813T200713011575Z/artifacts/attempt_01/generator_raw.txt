<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1
    ] .

gen:DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] .

gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:sideStructureApplicability ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:ForceQuantityShape
    ] .

gen:LoadCaseDemandShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:combinedShearAndBendingDemand ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:DemandQuantityShape
    ] .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A linked longitudinal local frame requires a boolean side-structure applicability value." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame .
    FILTER NOT EXISTS {
        ?member nltl:sideStructureApplicability ?app .
        FILTER(DATATYPE(?app) = xsd:boolean)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable side-structure longitudinal local frame requires a midspan plastic collapse load in kilonewtons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability true .
    FILTER NOT EXISTS {
        ?member nltl:midspanPlasticCollapseLoad ?collapseQuantity .
        ?collapseQuantity qudt:numericValue ?collapseValue ;
            qudt:unit unit:KiloN .
        FILTER(DATATYPE(?collapseValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must equal the applicable member's midspan plastic collapse load within a relative tolerance of 1e-6 and an absolute fallback of 1e-12 kN." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability true ;
        nltl:midspanPlasticCollapseLoad ?collapseQuantity .
    ?collapseQuantity qudt:numericValue ?collapseValue ;
        qudt:unit unit:KiloN .
    FILTER(DATATYPE(?collapseValue) = xsd:decimal)
    FILTER NOT EXISTS {
        $this nltl:plasticStrength ?strengthQuantity .
        ?strengthQuantity qudt:numericValue ?strengthValue ;
            qudt:unit unit:KiloN .
        FILTER(DATATYPE(?strengthValue) = xsd:decimal)
        FILTER(
            ABS(?strengthValue - ?collapseValue) <=
            IF(
                ABS(?collapseValue) * 0.000001 > 0.000000000001,
                ABS(?collapseValue) * 0.000001,
                0.000000000001
            )
        )
    }
}
"""
    ] .
<END_SHACL>
