<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_048_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSteelGradeRequirementCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSteelGradeRequirementCase ;
                sh:minCount 1 ;
                sh:class nltl:steelGradeRequirementCase ;
                sh:node gs:I2_048_SteelGradeCaseShape
            ] ;
            sh:property [
                sh:path nltl:asBuiltPlateThickness ;
                sh:minCount 1 ;
                sh:node gs:I2_048_ThicknessShape
            ] ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                )
            ] ;
            sh:property [
                sh:path nltl:steelMaterialClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:steelMaterialClassOne
                    nltl:steelMaterialClassTwo
                    nltl:steelMaterialClassThree
                )
            ] ;
            sh:property [
                sh:path nltl:steelStrengthCategory ;
                sh:minCount 1 ;
                sh:in (
                    nltl:normalStrengthSteelCategory
                    nltl:highTensileSteelCategory
                )
            ] ;
            sh:property [
                sh:path nltl:requiredHullStructuralSteelGrade ;
                sh:minCount 1 ;
                sh:class nltl:steelGradeValue ;
                sh:node gs:I2_048_GradeValueShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each steel-grade requirement case must be the same case selected for the Table 8 lookup." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual hull structural steel grade must not rank below the required Table 8 grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case ;
                      nltl:requiredHullStructuralSteelGrade ?requiredGrade .
                ?case nltl:actualHullStructuralSteelGrade ?actualGrade .
                ?actualGrade nltl:steelGradeOrderRank ?actualRank .
                ?requiredGrade nltl:steelGradeOrderRank ?requiredRank .
                FILTER (?actualRank < ?requiredRank)
            }
        """
    ] .

gs:I2_048_SteelGradeCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:class nltl:steelGradeValue ;
        sh:node gs:I2_048_GradeValueShape
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:in ( nltl:iacsUrI2Table8 )
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:nodeKind sh:IRI
    ] .

gs:I2_048_GradeValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:minCount 1 ;
        sh:datatype xsd:integer
    ] .

gs:I2_048_ThicknessShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:in ( unit:MilliM )
    ] .
<END_SHACL>
