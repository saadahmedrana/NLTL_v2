        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase
    ] ;
    sh:sparql [
        sh:message "Each steel grade requirement case must also be the ship's corresponding Table 8 lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual hull structural steel grade must not rank below the required Table 8 grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case ?actualGrade ?requiredGrade
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case ;
                      nltl:requiredHullStructuralSteelGrade ?requiredGrade .
                ?case nltl:actualHullStructuralSteelGrade ?actualGrade .
                ?actualGrade nltl:steelGradeOrderRank ?actualRank .
                ?requiredGrade nltl:steelGradeOrderRank ?requiredRank .
                FILTER (xsd:integer(?actualRank) < xsd:integer(?requiredRank))
            }
        """
    ] .
