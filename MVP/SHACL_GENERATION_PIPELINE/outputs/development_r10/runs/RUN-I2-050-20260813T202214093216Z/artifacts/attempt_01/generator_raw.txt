<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MegaNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DesignVerticalIceForceAtBowShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:message "The design vertical ice force at the bow must be provided in MN."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateOne ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:message "The first design vertical ice force candidate must be provided in MN."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateTwo ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:message "The second design vertical ice force candidate must be provided in MN."
    ] ;
    sh:sparql [
        sh:message "The design vertical ice force at the bow must equal the minimum of its two candidates within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 MN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?resultQuantity ;
                      nltl:designVerticalIceForceAtBowCandidateOne ?candidateOneQuantity ;
                      nltl:designVerticalIceForceAtBowCandidateTwo ?candidateTwoQuantity .
                ?resultQuantity qudt:numericValue ?resultRaw .
                ?candidateOneQuantity qudt:numericValue ?candidateOneRaw .
                ?candidateTwoQuantity qudt:numericValue ?candidateTwoRaw .
                BIND(xsd:decimal(?resultRaw) AS ?result)
                BIND(xsd:decimal(?candidateOneRaw) AS ?candidateOne)
                BIND(xsd:decimal(?candidateTwoRaw) AS ?candidateTwo)
                BIND(IF(?candidateOne <= ?candidateTwo, ?candidateOne, ?candidateTwo) AS ?expected)
                BIND(IF(
                    ABS(?expected) * 0.000001 > 0.000000001,
                    ABS(?expected) * 0.000001,
                    0.000000001
                ) AS ?tolerance)
                FILTER(ABS(?result - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
