@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

        sh:maxCount 1
    ] ;
    sh:property [

gs:DesignVerticalIceForceAtBowShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengthCoefficientKI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hullFormCoefficientKh ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:DegreeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateOne ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MegaNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateTwo ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MegaNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MegaNewtonQuantityShape
    ] ;
    sh:sparql [
        sh:message "Design vertical ice force at the bow must equal the minimum of the two candidate forces within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 MN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBowCandidateOne ?candidateOne ;
                      nltl:designVerticalIceForceAtBowCandidateTwo ?candidateTwo ;
                      nltl:designVerticalIceForceAtBow ?result .
                ?candidateOne qudt:numericValue ?candidateOneValue .
                ?candidateTwo qudt:numericValue ?candidateTwoValue .
                ?result qudt:numericValue ?resultValue .
                BIND(
                    IF(
                        ?candidateOneValue <= ?candidateTwoValue,
                        ?candidateOneValue,
                        ?candidateTwoValue
                    )
                    AS ?expected
                )
                BIND(
                    IF(
                        ABS(?expected) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                        ABS(?expected) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?resultValue - ?expected) > ?tolerance)
            }
        """
    ] .
