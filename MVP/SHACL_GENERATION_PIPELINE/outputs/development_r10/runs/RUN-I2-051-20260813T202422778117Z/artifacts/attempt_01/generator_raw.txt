<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_051_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_051_PositiveShearCaseShape
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force in kN must equal the longitudinal distribution factor multiplied by the bow ice force in MN and by 1000, within relative tolerance 1e-6 or absolute tolerance 1e-9 kN near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?bowForceQuantity ;
                      nltl:longitudinalDistributionFactor ?factorQuantity ;
                      nltl:designVerticalIceShearForce ?shearForceQuantity .
                ?bowForceQuantity qudt:numericValue ?bowForceMN .
                ?factorQuantity qudt:numericValue ?factor .
                ?shearForceQuantity qudt:numericValue ?reportedShearForceKN .
                BIND((?factor * ?bowForceMN * 1000.0) AS ?expectedShearForceKN)
                BIND(ABS(?expectedShearForceKN) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000001,
                        ?relativeTolerance,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(?reportedShearForceKN - ?expectedShearForceKN) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "For positive shear force, the longitudinal distribution factor must be 0.0 from the aft end through 0.6 LUI and 1.0 from 0.9 LUI through the forward end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:longitudinalDistributionFactor ?factorQuantity .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity .
                ?luiQuantity qudt:numericValue ?lui .
                ?factorQuantity qudt:numericValue ?factor .
                ?positionQuantity qudt:numericValue ?position .
                FILTER(
                    (?position >= 0.0 &&
                     ?position <= (0.6 * ?lui) &&
                     ?factor != 0.0)
                    ||
                    (?position >= (0.9 * ?lui) &&
                     ?position <= ?lui &&
                     ?factor != 1.0)
                )
            }
        """
    ] .

gen:I2_051_PositiveShearCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:positiveShearForce
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
