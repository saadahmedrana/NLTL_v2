<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_051_MegaNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_051_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_051_MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_051_UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_051_CalculationCaseEvidenceShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:class nltl:shearForceDirectionValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must identify its shear-force direction."
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:node gen:I2_051_MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must provide one longitudinal position from aft in metres."
    ] .

gen:I2_051_DesignVerticalIceShearShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:node gen:I2_051_MegaNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design vertical ice force at bow must be provided in meganewtons."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:node gen:I2_051_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The design vertical ice shear force must be provided in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:node gen:I2_051_UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The longitudinal distribution factor must be provided as one unitless decimal value."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:node gen:I2_051_MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The upper ice waterline length LUI must be provided in metres."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_051_CalculationCaseEvidenceShape
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal Cf multiplied by FIB, with conversion from meganewtons to kilonewtons, within a relative tolerance of 1e-6 and an absolute fallback of 1e-6 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity .
                ?fibQuantity qudt:numericValue ?fib .
                ?fiQuantity qudt:numericValue ?fi .
                ?cfQuantity qudt:numericValue ?cf .
                BIND((?cf * ?fib * "1000"^^xsd:decimal) AS ?expected)
                BIND(ABS(?expected) * "0.000001"^^xsd:decimal AS ?relativeTolerance)
                BIND(
                    IF(
                        ?relativeTolerance > "0.000001"^^xsd:decimal,
                        ?relativeTolerance,
                        "0.000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?fi - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a positive-shear calculation case, Cf must be 0.0 from the aft end through 0.6 LUI and 1.0 from 0.9 LUI through the forward end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity .
                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity .
                ?luiQuantity qudt:numericValue ?lui .
                ?cfQuantity qudt:numericValue ?cf .
                ?positionQuantity qudt:numericValue ?position .
                FILTER(
                    (
                        ?position >= "0"^^xsd:decimal &&
                        ?position <= ("0.6"^^xsd:decimal * ?lui) &&
                        ?cf != "0.0"^^xsd:decimal
                    )
                    ||
                    (
                        ?position >= ("0.9"^^xsd:decimal * ?lui) &&
                        ?position <= ?lui &&
                        ?cf != "1.0"^^xsd:decimal
                    )
                )
            }
        """
    ] .
<END_SHACL>
