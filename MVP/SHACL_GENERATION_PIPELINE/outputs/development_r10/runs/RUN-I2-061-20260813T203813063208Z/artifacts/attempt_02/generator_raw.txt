<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gen:I2_061_CalculationCaseShape ;
        sh:message "Each ship must identify a calculation case using the prescribed analytical procedure."
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:node gen:I2_061_ThicknessValueShape
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:class nltl:calculationMethodValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in ( nltl:prescribedAnalyticalProcedure ) ;
        sh:message "The calculation method must be the prescribed analytical procedure; direct calculation is not permitted as an alternative."
    ] .

gen:I2_061_ThicknessValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
