        sh:message "A ship using a linear calculation method requires yield strength in MPa as an xsd:decimal." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod ?method .
                FILTER (?method IN (
                    nltl:directCalculationMethodValue,
                    nltl:prescribedAnalyticalProcedure
                ))
                FILTER NOT EXISTS {
                    $this nltl:yieldStrength ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """

                    nltl:directCalculationMethodValue,

    sh:sparql [
        sh:message "Nominal shear stress in a member web must be less than yield strength divided by the square root of 3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?case nltl:calculationMethod ?method .
                FILTER (?method IN (
                    nltl:directCalculationMethodValue,
                    nltl:prescribedAnalyticalProcedure
                ))
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (datatype(?yield) = xsd:decimal)
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (datatype(?stress) = xsd:decimal)
                FILTER (

gs:I2_064_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:directCalculationMethodValue
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:prescribedAnalyticalProcedure
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ]
        ]
    ) .
