<BEGIN_SHACL>
@prefix gshape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gshape:I2_065_NonlinearCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "A nonlinear calculation case lacks required analysis information, acceptance evidence, margins, deformation quantities, or the relevant structural dimension." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this ?case
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

                FILTER (
                    !EXISTS {
                        ?case rdf:type nltl:calculationCase .
                    }
                    ||
                    !EXISTS {
                        $this nltl:bucklingResponse ?bucklingResponse .
                        FILTER (DATATYPE(?bucklingResponse) = xsd:string)
                    }
                    ||
                    !EXISTS {
                        $this nltl:plasticDeformation ?plasticDeformation .
                        FILTER (DATATYPE(?plasticDeformation) = xsd:string)
                    }
                    ||
                    !EXISTS {
                        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                        ?evidence rdf:type nltl:evidenceArtifact .
                    }
                    ||
                    !EXISTS {
                        $this nltl:fractureMargin ?fractureMargin .
                        ?fractureMargin rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?fractureMarginValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?fractureMarginValue) = xsd:decimal)
                    }
                    ||
                    !EXISTS {
                        $this nltl:majorBucklingMargin ?majorBucklingMargin .
                        ?majorBucklingMargin rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?majorBucklingMarginValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?majorBucklingMarginValue) = xsd:decimal)
                    }
                    ||
                    !EXISTS {
                        $this nltl:yieldingMargin ?yieldingMargin .
                        ?yieldingMargin rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?yieldingMarginValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?yieldingMarginValue) = xsd:decimal)
                    }
                    ||
                    !EXISTS {
                        ?case nltl:permanentLateralDeformation ?lateralDeformation .
                        ?lateralDeformation rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?lateralDeformationValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?lateralDeformationValue) = xsd:decimal)
                    }
                    ||
                    !EXISTS {
                        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlaneDeformation .
                        ?outOfPlaneDeformation rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?outOfPlaneDeformationValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?outOfPlaneDeformationValue) = xsd:decimal)
                    }
                    ||
                    !EXISTS {
                        ?case nltl:relevantStructuralDimension ?structuralDimension .
                        ?structuralDimension rdf:type qudt:QuantityValue ;
                            qudt:numericValue ?structuralDimensionValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?structuralDimensionValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
