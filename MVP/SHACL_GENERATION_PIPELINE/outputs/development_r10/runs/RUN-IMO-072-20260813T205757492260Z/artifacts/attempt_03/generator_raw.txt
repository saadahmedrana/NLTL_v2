<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_072_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:extendedDarknessOperation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( false )
            ]
        ]
        [
            sh:property [
                sh:path nltl:extendedDarknessOperation ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( true )
            ] ;
            sh:property [
                sh:path nltl:hasAssignedSearchlight ;
                sh:node [
                    sh:property [
                        sh:path nltl:searchlightContinuousUseSuitabilityStatus ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:in ( true ) ;
                        sh:message "Each assigned searchlight must be recorded as suitable for continuous use."
                    ]
                ]
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "A ship operating in extended darkness with linked lifeboats must have an assigned searchlight suitable for continuous use." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:extendedDarknessOperation true ;
                              nltl:hasLifeboat ?lifeboat .
                        FILTER NOT EXISTS {
                            $this nltl:hasAssignedSearchlight ?searchlight .
                            ?searchlight nltl:searchlightContinuousUseSuitabilityStatus ?status .
                            FILTER(
                                datatype(?status) = xsd:boolean &&
                                ?status = true
                            )
                        }
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
