<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:ShipLifeboatTypeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboat ;
        sh:node gen:LifeboatTypeShape
    ] .

gen:LifeboatTypeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:lifeboatTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:partiallyEnclosedLifeboat
            nltl:totallyEnclosedLifeboat
        ) ;
        sh:message "Each lifeboat type shall be partially enclosed or totally enclosed."
    ] .
<END_SHACL>
