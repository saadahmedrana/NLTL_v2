    sh:sparql [
        sh:message "The bending moment, actual area, shear-area conversion, and section modulus must satisfy clause 4.6.2; direct results use relative tolerance 1e-6, and the cross-multiplied squared section-modulus equation uses 2e-6 with a 1e-12 absolute fallback." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:actualWebFrameCrossSectionalArea ?actualAreaNode ;
          nltl:requiredShearArea ?requiredAreaNode ;
          nltl:sectionModulus ?sectionModulusNode ;
          nltl:shearArea ?shearAreaNode ;
          nltl:webFrameFreeFlangeArea ?flangeAreaNode ;
          nltl:webFrameIceLoad ?iceLoadNode ;
          nltl:webFrameMaximumIceLoadBendingMoment ?momentNode ;
          nltl:webFrameSpan ?spanNode ;
          nltl:webFrameWebArea ?webAreaNode ;
          nltl:yieldStrength ?yieldStrengthNode ;
          nltl:hasTableLookupCase ?lookupCase .

    ?lookupCase a nltl:tableLookupCase ;
                nltl:tableReference nltl:traficomTable4Dash8 ;
                nltl:lookupSelectionEvidence ?selectionEvidence ;
                nltl:webFrameTable4Dash8GammaFactor ?gammaNode .

    ?selectionEvidence a nltl:evidenceArtifact .

    ?actualAreaNode qudt:numericValue ?actualArea .
    ?requiredAreaNode qudt:numericValue ?requiredArea .
    ?sectionModulusNode qudt:numericValue ?sectionModulus .
    ?shearAreaNode qudt:numericValue ?shearArea .
    ?flangeAreaNode qudt:numericValue ?flangeArea .
    ?iceLoadNode qudt:numericValue ?iceLoad .
    ?momentNode qudt:numericValue ?moment .
    ?spanNode qudt:numericValue ?span .
    ?webAreaNode qudt:numericValue ?webArea .
    ?yieldStrengthNode qudt:numericValue ?yieldStrength .
    ?gammaNode qudt:numericValue ?gamma .

    FILTER (
        ?yieldStrength <= "0"^^xsd:decimal ||
        ?iceLoad < "0"^^xsd:decimal ||
        ?span < "0"^^xsd:decimal ||
        ?sectionModulus < "0"^^xsd:decimal ||
        (?flangeArea + ?webArea) <= "0"^^xsd:decimal ||
        ((?gamma * ?requiredArea) * (?gamma * ?requiredArea)) >=
            ((?flangeArea + ?webArea) * (?flangeArea + ?webArea)) ||

        ABS(
            ?moment -
            ("0.000193"^^xsd:decimal * ?iceLoad * ?span)
        ) >
        (
            "0.000001"^^xsd:decimal *
            ABS("0.000193"^^xsd:decimal * ?iceLoad * ?span) +
            "0.000000000001"^^xsd:decimal
        ) ||

        ABS(
            ?actualArea - (?flangeArea + ?webArea)
        ) >
        (
            "0.000001"^^xsd:decimal *
            ABS(?flangeArea + ?webArea) +
            "0.000000000001"^^xsd:decimal
        ) ||

        ABS(
            ?requiredArea -
            ("10000"^^xsd:decimal * ?shearArea)
        ) >
        (
            "0.000001"^^xsd:decimal *
            ABS("10000"^^xsd:decimal * ?shearArea) +
            "0.000000000001"^^xsd:decimal
        ) ||

        ABS(
            (
                (
                    ((?flangeArea + ?webArea) * (?flangeArea + ?webArea)) -
                    ((?gamma * ?requiredArea) * (?gamma * ?requiredArea))
                ) *
                ?sectionModulus * ?sectionModulus *
                ?yieldStrength * ?yieldStrength
            ) -
            (
                ((?flangeArea + ?webArea) * (?flangeArea + ?webArea)) *
                ("0.000193"^^xsd:decimal * ?iceLoad * ?span) *
                ("0.000193"^^xsd:decimal * ?iceLoad * ?span) *
                "1000000000000"^^xsd:decimal
            )
        ) >
        (
            "0.000002"^^xsd:decimal *
            ABS(
                ((?flangeArea + ?webArea) * (?flangeArea + ?webArea)) *
                ("0.000193"^^xsd:decimal * ?iceLoad * ?span) *
                ("0.000193"^^xsd:decimal * ?iceLoad * ?span) *
                "1000000000000"^^xsd:decimal
            ) +
            "0.000000000001"^^xsd:decimal
        )
    )
}
"""
