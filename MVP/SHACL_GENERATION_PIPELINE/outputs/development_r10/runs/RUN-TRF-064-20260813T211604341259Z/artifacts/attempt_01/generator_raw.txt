<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF064ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasSidePropellerShafting ;
        sh:class nltl:sidePropellerShaftingAssembly ;
        sh:node gen:TRF064PlatedBossingEnclosureShape ;
        sh:message "Each linked side propeller shafting assembly must be enclosed within a plated bossing."
    ] ;
    sh:property [
        sh:path nltl:hasSternTube ;
        sh:class nltl:sternTube ;
        sh:node gen:TRF064PlatedBossingEnclosureShape ;
        sh:message "Each linked stern tube must be enclosed within a plated bossing."
    ] ;
    sh:property [
        sh:path nltl:hasDetachedStrut ;
        sh:class nltl:detachedStrutStructure ;
        sh:node gen:TRF064DetachedStrutConsiderationShape ;
        sh:message "Each linked detached strut must have evidence covering design, strength, and attachment to the hull."
    ] .

gen:TRF064PlatedBossingEnclosureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:enclosedByPlatedBossing ;
        sh:minCount 1 ;
        sh:class nltl:platedBossingStructure ;
        sh:message "Plated bossing enclosure is required."
    ] .

gen:TRF064DetachedStrutConsiderationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStrutDesignEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutDesignEvidence ;
        sh:message "Detached strut design evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutStrengthEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutStrengthEvidence ;
        sh:message "Detached strut strength evidence is required."
    ] ;
    sh:property [
        sh:path nltl:hasStrutHullAttachmentEvidence ;
        sh:minCount 1 ;
        sh:class nltl:strutHullAttachmentEvidence ;
        sh:message "Evidence concerning detached strut attachment to the hull is required."
    ] .
<END_SHACL>
