    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardThrustKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bollard-thrust availability must be recorded unambiguously."
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:node gen:TRF082BollardThrustQuantityShape ;
        sh:message "Hydrodynamic bollard thrust must be recorded in kilonewtons."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardThrustKnown ;
                sh:hasValue "true"^^xsd:boolean
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardThrustKnown ;
                sh:hasValue "false"^^xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:qualifiedValueShape gen:TRF082TableLookupCaseShape ;
                sh:qualifiedMinCount 1 ;
                sh:message "When bollard thrust is not known, a supported Table 6-8 lookup case is required."
            ]
        ]
    ) .

    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1
    ] .

    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;

        sh:in ( nltl:traficomTable6Dash8 ) ;
        sh:message "The lookup case must reference TRAFICOM Table 6-8."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "Evidence supporting the Table 6-8 selection is required."
    ] .
