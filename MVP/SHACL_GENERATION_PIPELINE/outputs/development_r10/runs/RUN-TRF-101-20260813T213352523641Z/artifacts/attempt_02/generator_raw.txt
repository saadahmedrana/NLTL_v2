<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LoadRadiusRatioQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 0.8 ;
        sh:maxInclusive 0.8 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:node gen:LoadRadiusRatioQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The blade-failure load application radius ratio must be 0.8."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The blade-failure load must be confirmed as acting in the weakest direction."
    ] .

gen:BladeFailureRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:node gen:BladeFailureLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "A blade-failure load case at 0.8R in the weakest direction is required."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedNonlinearBladeStressAnalysis ;
                sh:class nltl:approvalRecord ;
                sh:minCount 1 ;
                sh:message "An approved nonlinear blade stress analysis is required when the formula alternative is not used."
            ]
        ]
        [
            sh:property [
                sh:path nltl:bladeChordLengthAtWeakestSection ;
                sh:node gen:MetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
                sh:node gen:MetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeCylindricalRootRadius ;
                sh:node gen:MetreQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeMinimumYieldStrength ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeUltimateTensileStrength ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeReferenceStrength1 ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The reported reference strength and blade-failure ultimate load must match the prescribed formulas within a relative tolerance of 1e-6 and an absolute fallback of 1e-12." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT DISTINCT $this
                    WHERE {
                        $this nltl:bladeChordLengthAtWeakestSection ?cQuantity ;
                              nltl:bladeMaximumThicknessAtWeakestSection ?tQuantity ;
                              nltl:bladeCylindricalRootRadius ?rQuantity ;
                              nltl:propellerDiameter ?dQuantity ;
                              nltl:bladeMinimumYieldStrength ?yieldQuantity ;
                              nltl:bladeUltimateTensileStrength ?ultimateQuantity ;
                              nltl:bladeReferenceStrength1 ?referenceQuantity ;
                              nltl:bladeFailureUltimateLoad ?loadQuantity .
                        ?cQuantity qudt:numericValue ?c .
                        ?tQuantity qudt:numericValue ?t .
                        ?rQuantity qudt:numericValue ?r .
                        ?dQuantity qudt:numericValue ?diameter .
                        ?yieldQuantity qudt:numericValue ?yieldStrength .
                        ?ultimateQuantity qudt:numericValue ?ultimateStrength .
                        ?referenceQuantity qudt:numericValue ?reportedReference .
                        ?loadQuantity qudt:numericValue ?reportedLoad .
                        BIND ((0.6 * ?yieldStrength) + (0.4 * ?ultimateStrength) AS ?expectedReference)
                        BIND (IF(
                            ABS(?expectedReference) * 0.000001 > 0.000000000001,
                            ABS(?expectedReference) * 0.000001,
                            0.000000000001
                        ) AS ?referenceTolerance)
                        BIND ((0.8 * ?diameter) - (2 * ?r) AS ?denominator)
                        BIND (IF(
                            ?denominator = 0,
                            0,
                            (300 * ?c * ?t * ?t * ?expectedReference) / ?denominator
                        ) AS ?expectedLoad)
                        BIND (IF(
                            ABS(?expectedLoad) * 0.000001 > 0.000000000001,
                            ABS(?expectedLoad) * 0.000001,
                            0.000000000001
                        ) AS ?loadTolerance)
                        FILTER (
                            ?denominator = 0 ||
                            ABS(?reportedReference - ?expectedReference) > ?referenceTolerance ||
                            ABS(?reportedLoad - ?expectedLoad) > ?loadTolerance
                        )
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:bladeTipOffset ?tipQuantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:bladeFailureStatus ?failureStatus .
                ?tipQuantity qudt:numericValue ?tipOffset .
                ?diameterQuantity qudt:numericValue ?diameter .
                FILTER (
                    ?tipOffset > (0.10 * ?diameter) &&
                    ?failureStatus != true
                )
            }
        """
    ] .
<END_SHACL>
