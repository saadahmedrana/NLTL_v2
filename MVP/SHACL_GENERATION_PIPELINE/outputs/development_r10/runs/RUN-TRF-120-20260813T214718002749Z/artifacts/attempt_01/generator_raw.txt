<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF120ThrusterNaturalFrequencyShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A linked thruster natural-frequency calculation case must be typed as a calculation case and identify only a longitudinal or transverse vibration direction." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?thruster .
    ?thruster a nltl:thrusterBody ;
              nltl:hasNaturalFrequencyCalculationCase ?case .
    FILTER (
        NOT EXISTS {
            ?case a nltl:calculationCase ;
                  nltl:vibrationDirection ?direction .
            FILTER (
                ?direction IN (
                    nltl:longitudinalDirection,
                    nltl:transverseDirection
                )
            )
        }
        ||
        EXISTS {
            ?case nltl:vibrationDirection ?otherDirection .
            FILTER (
                ?otherDirection NOT IN (
                    nltl:longitudinalDirection,
                    nltl:transverseDirection
                )
            )
        }
    )
}
""" 
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A thruster natural-frequency estimation must cover both longitudinal and transverse vibration directions." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?thruster .
    ?thruster a nltl:thrusterBody ;
              nltl:hasNaturalFrequencyCalculationCase ?existingCase .
    FILTER (
        NOT EXISTS {
            ?thruster nltl:hasNaturalFrequencyCalculationCase ?longitudinalCase .
            ?longitudinalCase a nltl:calculationCase ;
                              nltl:vibrationDirection nltl:longitudinalDirection .
        }
        ||
        NOT EXISTS {
            ?thruster nltl:hasNaturalFrequencyCalculationCase ?transverseCase .
            ?transverseCase a nltl:calculationCase ;
                            nltl:vibrationDirection nltl:transverseDirection .
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Thruster natural-frequency estimation requires damping, water-added mass in kilograms, natural frequency in hertz, and ship-attachment stiffness in kilonewtons per metre." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?thruster .
    ?thruster a nltl:thrusterBody ;
              nltl:hasNaturalFrequencyCalculationCase ?case .
    FILTER (
        NOT EXISTS {
            $this nltl:damping ?damping .
            FILTER (datatype(?damping) = xsd:string)
        }
        ||
        NOT EXISTS {
            $this nltl:waterAddedMass ?addedMass .
            ?addedMass a qudt:QuantityValue ;
                       qudt:numericValue ?addedMassValue ;
                       qudt:unit unit:KiloGM .
            FILTER (datatype(?addedMassValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            $this nltl:thrusterNaturalFrequency ?naturalFrequency .
            ?naturalFrequency a qudt:QuantityValue ;
                              qudt:numericValue ?naturalFrequencyValue ;
                              qudt:unit unit:HZ .
            FILTER (datatype(?naturalFrequencyValue) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?thruster nltl:shipAttachmentStiffness ?attachmentStiffness .
            ?attachmentStiffness a qudt:QuantityValue ;
                                 qudt:numericValue ?attachmentStiffnessValue ;
                                 qudt:unit unit:KiloN-PER-M .
            FILTER (datatype(?attachmentStiffnessValue) = xsd:decimal)
        }
    )
}
"""
    ] .
<END_SHACL>
