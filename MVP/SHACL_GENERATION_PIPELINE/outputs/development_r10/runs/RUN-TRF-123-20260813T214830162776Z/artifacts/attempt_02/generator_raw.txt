<BEGIN_SHACL>
@prefix shape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:MaterialPropertiesShape
    a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:node shape:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Material properties must provide one component material yield strength in MPa."
    ] .

shape:PropellerShaftLineComponentShape
    a sh:NodeShape ;
    sh:class nltl:propellerShaftLineComponent ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:node shape:MaterialPropertiesShape ;
        sh:minCount 1 ;
        sh:message "Each linked propeller shaft line component must have material properties."
    ] .

shape:OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The occasional-force load case must record that the propeller blade is excluded."
    ] .

shape:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:node shape:PropellerShaftLineComponentShape
    ] ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node shape:OccasionalForceLoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:node shape:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A unitless component yield safety factor is required."
    ] ;
    sh:property [
        sh:path nltl:materialYieldStrength ;
        sh:node shape:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Material yield strength must be provided in MPa."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node shape:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Yield strength must be provided in MPa."
    ] .
<END_SHACL>
