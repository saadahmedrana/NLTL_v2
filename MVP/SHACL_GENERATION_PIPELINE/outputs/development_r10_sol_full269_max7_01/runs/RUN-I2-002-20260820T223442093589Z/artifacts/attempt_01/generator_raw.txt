<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_002_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A Polar Class ship shall have true UR I1 compliance and UR I2 applicability statuses." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:polarClass ?polarClass .
    FILTER (
        NOT EXISTS {
            $this nltl:urI1ComplianceStatus ?i1Status .
            FILTER (?i1Status = "true"^^xsd:boolean)
        }
        ||
        NOT EXISTS {
            $this nltl:urI2Applicability ?i2Status .
            FILTER (?i2Status = "true"^^xsd:boolean)
        }
    )
}
"""
    ] .
<END_SHACL>
