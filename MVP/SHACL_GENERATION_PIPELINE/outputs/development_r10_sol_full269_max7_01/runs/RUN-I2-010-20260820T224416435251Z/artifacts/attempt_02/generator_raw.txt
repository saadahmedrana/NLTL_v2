<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node gen:I2_010_NormalFrameAngleValueShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When at least one normal frame angle is supplied and every supplied angle is between 0 and 10 DEG inclusive, the bow shall be identified as vertical-sided." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER EXISTS {
        $this nltl:normalFrameAngle ?angle .
    }
    FILTER NOT EXISTS {
        $this nltl:normalFrameAngle ?angle .
        FILTER NOT EXISTS {
            ?angle qudt:numericValue ?value ;
                   qudt:unit unit:DEG .
            FILTER(
                datatype(?value) = xsd:decimal &&
                ?value >= "0"^^xsd:decimal &&
                ?value <= "10"^^xsd:decimal
            )
        }
    }
    FILTER NOT EXISTS {
        $this nltl:verticalSidedBowForm true .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A PC6 or PC7 ship with a vertical-sided bow shall identify the bow design force method from I2.3.2.1 (iv) and record a true bow vertical-side status." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:polarClass ?polarClass ;
          nltl:verticalSidedBowForm true .
    FILTER(
        ?polarClass IN (
            nltl:polarClassPc6,
            nltl:polarClassPc7
        )
    )
    FILTER(
        NOT EXISTS {
            $this nltl:bowVerticalSideStatus true .
        } ||
        NOT EXISTS {
            $this nltl:bowDesignForceMethod "I2.3.2.1 (iv)"^^xsd:string .
        }
    )
}
"""
    ] .

gen:I2_010_NormalFrameAngleValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each supplied normal frame angle shall have one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each supplied normal frame angle shall use DEG."
    ] .
<END_SHACL>
