<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:NonBowIceLoadShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The ship must provide one upper ice waterline draught."
    ] ;
    sh:property [
        sh:path nltl:displacementFactorThreshold ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The ship must provide one displacement factor threshold."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:NonBowClassFactorTableLookupCaseShape ;
        sh:message "The ship must provide one table lookup case containing the required class factors."
    ] ;
    sh:property [
        sh:path nltl:hasNonBowIceLoadCalculationCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:nonBowIceLoadCalculationCase ;
        sh:node gen:NonBowIceLoadCalculationCaseShape ;
        sh:message "The ship must provide one non-bow ice-load calculation case."
    ] ;
    sh:sparql [
        sh:message "The non-bow ice force must equal 0.36 times the crushing failure class factor times the ship displacement factor, reported in kN within the specified numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?tableCase ;
                      nltl:hasNonBowIceLoadCalculationCase ?calculationCase .
                ?tableCase nltl:crushingFailureClassFactor ?cfcQuantity .
                ?cfcQuantity qudt:numericValue ?cfc .
                ?calculationCase nltl:shipDisplacementFactor ?dfQuantity ;
                                 nltl:nonBowIceForce ?forceQuantity .
                ?dfQuantity qudt:numericValue ?df .
                ?forceQuantity qudt:numericValue ?force .
                BIND((xsd:decimal("360") * ?cfc * ?df) AS ?expected)
                BIND(IF(ABS(?expected) > xsd:decimal("1"), ABS(?expected) * xsd:decimal("0.000001"), xsd:decimal("0.000001")) AS ?tolerance)
                FILTER(ABS(?force - ?expected) > ?tolerance)
            }
        """
    ] .

gen:NonBowClassFactorTableLookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The table lookup case must provide one crushing failure class factor."
    ] ;
    sh:property [
        sh:path nltl:loadPatchDimensionClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The table lookup case must provide one load patch dimension class factor."
    ] .

gen:NonBowIceLoadCalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:nonBowIceLoadCalculationCase ;
    sh:property [
        sh:path nltl:shipDisplacementFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The calculation case must provide one ship displacement factor."
    ] ;
    sh:property [
        sh:path nltl:nonBowIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:message "The calculation case must provide one non-bow ice force."
    ] ;
    sh:property [
        sh:path nltl:nonBowIceLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonPerMetreQuantityShape ;
        sh:message "The calculation case must provide one non-bow ice line load."
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

gen:MegaNewtonPerMetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaN-M-PER-M2
    ] .
<END_SHACL>
