<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:NonBowIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The ship must provide one upper ice waterline draught in metres."
    ] ;
    sh:property [
        sh:path nltl:displacementFactorThreshold ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The ship must provide one displacement-factor threshold in metres."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:NonBowTableLookupCaseShape ;
        sh:message "The ship must link to one table lookup case containing the required class factors."
    ] ;
    sh:property [
        sh:path nltl:hasNonBowIceLoadCalculationCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:nonBowIceLoadCalculationCase ;
        sh:node gen:NonBowCalculationCaseShape ;
        sh:message "The ship must link to one non-bow ice-load calculation case."
    ] ;
    sh:sparql [
        sh:message "The reported non-bow ice force must equal 0.36*CFC*DF MN, reported in kN, within relative tolerance 1e-6 or absolute tolerance 1e-9 kN near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?tableCase ;
                      nltl:hasNonBowIceLoadCalculationCase ?calculationCase .

                ?tableCase nltl:crushingFailureClassFactor ?cfcQuantity .
                ?cfcQuantity qudt:numericValue ?cfc .

                ?calculationCase nltl:shipDisplacementFactor ?dfQuantity ;
                                 nltl:nonBowIceForce ?forceQuantity .
                ?dfQuantity qudt:numericValue ?df .
                ?forceQuantity qudt:numericValue ?reportedForce .

                BIND (
                    "360"^^xsd:decimal * ?cfc * ?df
                    AS ?expectedForce
                )
                BIND (
                    ABS(?expectedForce) * "0.000001"^^xsd:decimal
                    AS ?relativeTolerance
                )
                BIND (
                    IF(
                        ?relativeTolerance > "0.000000001"^^xsd:decimal,
                        ?relativeTolerance,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER (ABS(?reportedForce - ?expectedForce) > ?tolerance)
            }
        """
    ] .

gen:NonBowTableLookupCaseShape

    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;

    ] ;
    sh:property [
        sh:path nltl:loadPatchDimensionClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The table lookup case must provide one load-patch dimension class factor."
    ] .

gen:NonBowCalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipDisplacementFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The non-bow calculation case must provide one ship displacement factor."
    ] ;
    sh:property [
        sh:path nltl:nonBowIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:message "The non-bow calculation case must provide one non-bow ice force in kN."
    ] ;
    sh:property [
        sh:path nltl:nonBowIceLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeganewtonPerMetreQuantityShape ;
        sh:message "The non-bow calculation case must provide one non-bow ice line load in MN/m."
    ] .
