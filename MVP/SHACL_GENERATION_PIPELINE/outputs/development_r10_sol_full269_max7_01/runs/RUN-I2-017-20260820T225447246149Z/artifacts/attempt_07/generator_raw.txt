gs:ShipNonBowIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreQuantityShape ;
        sh:message "Upper ice waterline draught D_UI is required in metres."
    ] ;
    sh:property [
        sh:path nltl:displacementFactorThreshold ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreQuantityShape ;
        sh:message "Displacement factor threshold CF_DIS is required in metres."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableLookupCase ;
        sh:node gs:NonBowTableLookupCaseShape ;
        sh:message "A table lookup case containing CFC and CFD is required."
    ] ;
    sh:property [
        sh:path nltl:hasNonBowIceLoadCalculationCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:nonBowIceLoadCalculationCase ;
        sh:node gs:NonBowIceLoadCalculationCaseShape ;
        sh:message "A non-bow ice load calculation case is required."
    ] ;
    sh:sparql [
        sh:message "Non-bow ice force must equal 360 times CFC times DF in kN, within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasTableLookupCase ?tableCase ;
                      nltl:hasNonBowIceLoadCalculationCase ?calculationCase .
                ?tableCase nltl:crushingFailureClassFactor ?cfcQuantity .
                ?cfcQuantity qudt:numericValue ?cfc .
                ?calculationCase nltl:shipDisplacementFactor ?dfQuantity ;
                                 nltl:nonBowIceForce ?forceQuantity .
                ?dfQuantity qudt:numericValue ?df .
                ?forceQuantity qudt:numericValue ?force .
                BIND (360.0 * ?cfc * ?df AS ?expectedForce)
                BIND (
                    IF (
                        ABS(?expectedForce) * 0.000001 > 0.000000001,
                        ABS(?expectedForce) * 0.000001,
                        0.000000001
                    ) AS ?tolerance
                )
                FILTER (ABS(?force - ?expectedForce) > ?tolerance)
            }
        """
    ] .
