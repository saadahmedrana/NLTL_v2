<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipHullAreaFactorShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberHullAreaFactorShape
    ] ;
    sh:sparql [
        sh:message "For each structural member with spanned hull areas, the selected hull area factor must equal the largest factor of those areas within the specified numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                {
                    SELECT $this ?member (MAX(?factorValue) AS ?expectedValue)
                    WHERE {
                        $this nltl:hasStructuralMember ?member .
                        ?member nltl:hasSpannedHullArea ?spannedArea .
                        ?spannedArea nltl:hullAreaFactor ?factor .
                        ?factor qudt:numericValue ?factorValue .
                    }
                    GROUP BY $this ?member
                }
                ?member nltl:selectedHullAreaFactor ?selectedFactor .
                ?selectedFactor qudt:numericValue ?selectedValue .
                BIND(
                    ABS(?expectedValue) * "0.000001"^^xsd:decimal
                    AS ?relativeTolerance
                )
                BIND(
                    IF(
                        ?relativeTolerance > "0.000000000001"^^xsd:decimal,
                        ?relativeTolerance,
                        "0.000000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?selectedValue - ?expectedValue) > ?tolerance)
            }
        """
    ] .

gen:StructuralMemberHullAreaFactorShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:minCount 1 ;
                sh:node gen:HullAreaValueShape
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:SelectedHullAreaFactorValueShape
            ]
        ]
    ) .

gen:HullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:HullAreaFactorValueShape
    ] .

gen:HullAreaFactorValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SelectedHullAreaFactorValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
