<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipPlatingThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:node gen:PlatingThicknessCalculationShape ;
        sh:message "The ship must identify each plating item subject to the shell plate thickness calculation."
    ] .

gen:PlatingThicknessCalculationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAllowance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Thickness must equal ice-load net plate thickness plus corrosion and abrasion allowance, within relative tolerance 1e-6 or absolute tolerance 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadRequiredNetPlateThickness ?netQuantity ;
                      nltl:corrosionAbrasionAllowance ?allowanceQuantity ;
                      nltl:thickness ?thicknessQuantity .
                ?netQuantity qudt:numericValue ?net .
                ?allowanceQuantity qudt:numericValue ?allowance .
                ?thicknessQuantity qudt:numericValue ?actual .
                BIND((?net + ?allowance) / xsd:decimal("1000") AS ?expected)
                BIND(ABS(?expected) * xsd:decimal("0.000001") AS ?relativeTolerance)
                BIND(
                    IF(
                        ?relativeTolerance > xsd:decimal("0.000000001"),
                        ?relativeTolerance,
                        xsd:decimal("0.000000001")
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?actual - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
