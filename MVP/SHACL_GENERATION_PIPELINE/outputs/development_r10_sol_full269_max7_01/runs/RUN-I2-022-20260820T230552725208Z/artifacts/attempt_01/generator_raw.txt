<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2-022ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:node gen:I2-022PlatingReadinessShape ;
        sh:message "The ship must identify its plating and provide the applicability and readiness information required by I2.4.2."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasPlating ;
                sh:node gen:I2-022NonApplicableAngleShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasPlating ;
                sh:qualifiedValueShape gen:I2-022ApplicableAngleShape ;
                sh:qualifiedMinCount 1
            ] ;
            sh:property [
                sh:path nltl:peakPressureFactor ;
                sh:minCount 1 ;
                sh:node gen:UnitlessQuantityShape
            ] ;
            sh:property [
                sh:path nltl:averageIcePressure ;
                sh:minCount 1 ;
                sh:node gen:MegaPascalQuantityShape
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:node gen:MegaPascalQuantityShape
            ] ;
            sh:property [
                sh:path nltl:loadPatchHeight ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape
            ]
        ]
    ) ;
    sh:message "When plating has a framing angle of at least 70 degrees, the ship must provide the declared ship-owned calculation inputs." .

gen:I2-022PlatingReadinessShape
    a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:platingHullAreaClassification ;
        sh:minCount 1 ;
        sh:node gen:I2-022HullAreaClassificationShape ;
        sh:message "Each represented plating item must provide its hull-area classification."
    ] ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:node gen:DegreeQuantityShape ;
        sh:message "Each represented plating item must provide its framing angle in degrees."
    ] ;
    sh:or (
        gen:I2-022NonApplicableAngleShape
        gen:I2-022ApplicablePlatingShape
    ) .

gen:I2-022HullAreaClassificationShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:in (
                nltl:bowIntermediateBottomHullArea
                nltl:midbodyBottomHullArea
                nltl:sternBottomHullArea
            )
        ]
        [
            sh:class nltl:hullAreaValue
        ]
    ) .

gen:I2-022NonApplicableAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive 70
            ]
        ]
    ] .

gen:I2-022ApplicableAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 70
            ]
        ]
    ] .

gen:I2-022ApplicablePlatingShape
    a sh:NodeShape ;
    sh:node gen:I2-022ApplicableAngleShape ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "Applicable plating must provide the externally calculated required net plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:platingSupportedByStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:minCount 1 ;
        sh:node gen:I2-022StructuralMemberReadinessShape ;
        sh:message "Applicable plating must identify its supporting structural member and member-owned calculation inputs."
    ] .

gen:I2-022StructuralMemberReadinessShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
