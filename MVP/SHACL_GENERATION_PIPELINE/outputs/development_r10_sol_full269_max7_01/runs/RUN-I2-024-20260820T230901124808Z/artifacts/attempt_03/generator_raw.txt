<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ObliquePlatingInterpolationShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:class nltl:obliquePlatingInterpolationCase ;
        sh:node gen:ObliquePlatingInterpolationCaseShape
    ] .

gen:ObliquePlatingInterpolationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:FramingAngleQuantityShape ;
        sh:message "Each oblique-plating case must have exactly one framing angle in degrees."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:framingAngle qudt:numericValue ) ;
                sh:minExclusive "20"^^xsd:decimal ;
                sh:maxExclusive "70"^^xsd:decimal
            ] ;
            sh:property [
                sh:path nltl:interpolationLowerEndpoint ;
                sh:class nltl:interpolationPoint ;
                sh:minCount 1 ;
                sh:node gen:InterpolationPointShape
            ] ;
            sh:property [
                sh:path nltl:interpolationUpperEndpoint ;
                sh:class nltl:interpolationPoint ;
                sh:minCount 1 ;
                sh:node gen:InterpolationPointShape
            ] ;
            sh:property [
                sh:path nltl:longitudinalFramingNetPlateThickness ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:node gen:MillimetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:transverseFramingNetPlateThickness ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:node gen:MillimetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:interpolatedNetPlateThickness ;
                sh:class qudt:QuantityValue ;
                sh:minCount 1 ;
                sh:node gen:MillimetreQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:framingAngle qudt:numericValue ) ;
                sh:maxInclusive "20"^^xsd:decimal
            ]
        ]
        [
            sh:property [
                sh:path ( nltl:framingAngle qudt:numericValue ) ;
                sh:minInclusive "70"^^xsd:decimal
            ]
        ]
    ) ;
    sh:message "Cases with framing angles strictly between 20 and 70 degrees require complete interpolation inputs and an externally computed result." .

gen:FramingAngleQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:DecimalQuantityShape
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DecimalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .
<END_SHACL>
