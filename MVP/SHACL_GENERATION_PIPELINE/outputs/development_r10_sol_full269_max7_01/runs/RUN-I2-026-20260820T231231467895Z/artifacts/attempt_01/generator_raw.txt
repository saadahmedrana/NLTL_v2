gen:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026_StructuralMemberShape
    ] .

gen:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026_MemberSupportShape ;
        sh:minCount 1 ;
        sh:message "A structural member must identify its member support."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame support condition must be exactly one of fixed or simple support."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path ( nltl:hasMemberSupport nltl:connectionBracketPresent ) ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node gen:I2_026_NoBracketSupportShape
            ] ;
            sh:property [
                sh:path ( nltl:hasMemberSupport nltl:demonstratedRotationalRestraint ) ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:node gen:I2_026_NoBracketOrRestraintSupportShape
            ] ;
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:simpleFrameSupport
            ]
        ]
    ) ;
    sh:message "The support condition must follow the verified fixity, bracket, rotational-restraint, and ice-strengthened-area alternatives." .

gen:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean
    ] .

gen:I2_026_NoBracketSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] .

gen:I2_026_NoBracketOrRestraintSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] .
