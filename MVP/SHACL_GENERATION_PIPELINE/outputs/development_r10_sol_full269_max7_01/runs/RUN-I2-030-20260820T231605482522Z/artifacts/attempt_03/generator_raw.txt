<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_030_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:localFrameSectionCalculationCase ;
        sh:node gs:I2_030_LocalFrameSectionCaseShape ;
        sh:minCount 1 ;
        sh:message "A local-frame section calculation case is required."
    ] .

gs:I2_030_LocalFrameSectionCaseShape
    a sh:NodeShape ;
    sh:class nltl:localFrameSectionCalculationCase ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gs:I2_030_StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify its structural member."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCasePlating ;
        sh:class nltl:plating ;
        sh:node gs:I2_030_PlatingShape ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify its plating."
    ] ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node gs:I2_030_MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_030_InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_030_InterpolationPointShape
    ] ;
    sh:sparql [
        sh:message "Each linked structural member must link back to this local-frame section calculation case." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:sectionCalculationCaseStructuralMember ?member .
    FILTER NOT EXISTS {
        ?member nltl:hasLocalFrameSectionCalculationCase $this .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "When the local-frame area exceeds the attached-plate-flange area, the formula operands and externally computed z_na and Z_p results are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:sectionCalculationCaseStructuralMember ?member .
    ?member nltl:netLocalFrameFlangeArea ?areaQuantity .
    ?areaQuantity qudt:numericValue ?area ;
                  qudt:unit unit:CentiM2 .
    ?member nltl:netAttachedShellPlateThickness ?plateThicknessQuantity .
    ?plateThicknessQuantity qudt:numericValue ?plateThickness ;
                            qudt:unit unit:MilliM .
    ?member nltl:frameSpacing ?spacingQuantity .
    ?spacingQuantity qudt:numericValue ?spacing ;
                     qudt:unit unit:M .
    FILTER (
        (100 * ?area > 1000 * ?plateThickness * ?spacing) &&
        (
            NOT EXISTS { ?member nltl:webHeight ?webHeight . } ||
            NOT EXISTS { ?member nltl:netWebThickness ?webThickness . } ||
            NOT EXISTS { ?member nltl:localFrameFlangeCentreHeight ?flangeHeight . } ||
            NOT EXISTS { ?member nltl:webAngleToShellPlate ?webAngle . } ||
            NOT EXISTS { ?member nltl:webToFlangeCentreDistance ?flangeDistance . } ||
            NOT EXISTS { $this nltl:plasticNeutralAxisHeight ?neutralAxisHeight . } ||
            NOT EXISTS { ?member nltl:netEffectivePlasticSectionModulus ?plasticSectionModulus . }
        )
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "For framing angles strictly between 20 and 70 degrees, both interpolation endpoints are required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:sectionCalculationCasePlating ?plating .
    ?plating nltl:framingAngleOmega ?omegaQuantity .
    ?omegaQuantity qudt:numericValue ?omega ;
                   qudt:unit unit:DEG .
    FILTER (
        (?omega > 20 && ?omega < 70) &&
        (
            NOT EXISTS { $this nltl:interpolationLowerEndpoint ?lowerEndpoint . } ||
            NOT EXISTS { $this nltl:interpolationUpperEndpoint ?upperEndpoint . }
        )
    )
}
"""
    ] .
