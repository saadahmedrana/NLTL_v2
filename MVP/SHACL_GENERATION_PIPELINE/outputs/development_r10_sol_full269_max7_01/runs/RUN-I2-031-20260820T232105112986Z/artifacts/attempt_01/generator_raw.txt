<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_031_StructuralMemberModelShape
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        sh:message "Each applicable local frame requires plastic-strength, midspan-collapse-load, load-case demand, and collapse-mechanism information." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame .
                FILTER (
                    EXISTS {
                        ?member nltl:structuralLocation nltl:bottomStructureLocation .
                    }
                    ||
                    EXISTS {
                        ?member nltl:structuralLocation nltl:sideStructureLocation ;
                                nltl:transverseFrameOrientation true .
                    }
                )
                FILTER NOT EXISTS {
                    ?member nltl:plasticStrength ?strength ;
                            nltl:midspanPlasticCollapseLoad ?collapseLoad ;
                            nltl:hasStructuralMemberLoadCase ?loadCase .
                    ?strength qudt:numericValue ?strengthValue ;
                              qudt:unit unit:KiloN .
                    ?collapseLoad qudt:numericValue ?collapseValue ;
                                  qudt:unit unit:KiloN .
                    ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                    ?demand qudt:numericValue ?demandValue .
                    $this nltl:plasticCollapseMechanism ?mechanism .
                    FILTER(
                        datatype(?strengthValue) = xsd:decimal
                        && datatype(?collapseValue) = xsd:decimal
                        && datatype(?demandValue) = xsd:decimal
                        && datatype(?mechanism) = xsd:string
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a bottom local frame, the ship requires a valid patch load in kilonewtons." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        nltl:structuralLocation nltl:bottomStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:patchLoad ?patchLoad .
                    ?patchLoad qudt:numericValue ?patchValue ;
                               qudt:unit unit:KiloN .
                    FILTER(datatype(?patchValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each bottom-frame load case must specify that the patch-load height is parallel to the frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?loadCase
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        nltl:structuralLocation nltl:bottomStructureLocation ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    ?loadCase nltl:loadPatchOrientation nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every existing load case of an applicable local frame requires a decimal combined shear-and-bending demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?loadCase
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER (
                    EXISTS {
                        ?member nltl:structuralLocation nltl:bottomStructureLocation .
                    }
                    ||
                    EXISTS {
                        ?member nltl:structuralLocation nltl:sideStructureLocation ;
                                nltl:transverseFrameOrientation true .
                    }
                )
                FILTER NOT EXISTS {
                    ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                    ?demand qudt:numericValue ?demandValue .
                    FILTER(datatype(?demandValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Plastic strength and midspan plastic-collapse load must be decimal quantities expressed in kilonewtons." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?quantity
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        ?property ?quantity .
                FILTER(
                    ?property IN (
                        nltl:plasticStrength,
                        nltl:midspanPlasticCollapseLoad
                    )
                )
                FILTER (
                    EXISTS {
                        ?member nltl:structuralLocation nltl:bottomStructureLocation .
                    }
                    ||
                    EXISTS {
                        ?member nltl:structuralLocation nltl:sideStructureLocation ;
                                nltl:transverseFrameOrientation true .
                    }
                )
                FILTER NOT EXISTS {
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:KiloN .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The combined shear-and-bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?loadCase
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        nltl:plasticStrength ?strength ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                ?strength qudt:numericValue ?strengthValue .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                FILTER (
                    EXISTS {
                        ?member nltl:structuralLocation nltl:bottomStructureLocation .
                    }
                    ||
                    EXISTS {
                        ?member nltl:structuralLocation nltl:sideStructureLocation ;
                                nltl:transverseFrameOrientation true .
                    }
                )
                FILTER(
                    datatype(?strengthValue) = xsd:decimal
                    && datatype(?demandValue) = xsd:decimal
                    && ?demandValue > ?strengthValue
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Plastic strength must equal the midspan plastic-collapse load within relative tolerance 1e-6 and absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                        nltl:plasticStrength ?strength ;
                        nltl:midspanPlasticCollapseLoad ?collapseLoad .
                ?strength qudt:numericValue ?strengthValue .
                ?collapseLoad qudt:numericValue ?collapseValue .
                FILTER (
                    EXISTS {
                        ?member nltl:structuralLocation nltl:bottomStructureLocation .
                    }
                    ||
                    EXISTS {
                        ?member nltl:structuralLocation nltl:sideStructureLocation ;
                                nltl:transverseFrameOrientation true .
                    }
                )
                FILTER(
                    datatype(?strengthValue) = xsd:decimal
                    && datatype(?collapseValue) = xsd:decimal
                )
                BIND(
                    IF(
                        ABS(?collapseValue) * 0.000001 > 0.000000000001,
                        ABS(?collapseValue) * 0.000001,
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?strengthValue - ?collapseValue) > ?tolerance)
            }
        """
    ] .

gen:I2_031_StructuralMemberModelShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue
    ] ;
    sh:property [
        sh:path nltl:transverseFrameOrientation ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:class nltl:loadCase ;
        sh:node gen:I2_031_LoadCaseModelShape
    ] .

gen:I2_031_LoadCaseModelShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue
    ] .
<END_SHACL>
