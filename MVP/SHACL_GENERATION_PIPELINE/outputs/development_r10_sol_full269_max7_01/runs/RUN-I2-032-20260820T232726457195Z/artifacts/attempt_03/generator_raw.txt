gen:I2_032Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:NoStructuralMembersShape
        gen:ApplicableShipShape
    ) ;
    sh:sparql [
        sh:message "The selected hull area factor must correspond to a decimal unitless hull area factor on a ship-linked hull area value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:selectedHullAreaFactor ?selectedNode .
                ?selectedNode qudt:numericValue ?selected ;
                              qudt:unit unit:UNITLESS .
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?areaValue .
                    ?areaValue a nltl:hullAreaValue ;
                               nltl:hullAreaFactor ?factorNode .
                    ?factorNode qudt:numericValue ?factor ;
                                qudt:unit unit:UNITLESS .
                    FILTER (
                        DATATYPE(?factor) = xsd:decimal &&
                        ?factor = ?selected
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship span and frame span must represent the same frame-span input within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:span ?spanNode ;
                      nltl:frameSpan ?frameSpanNode .
                ?spanNode qudt:numericValue ?span .
                ?frameSpanNode qudt:numericValue ?frameSpan .
                BIND(
                    IF(
                        ABS(?frameSpan) * 0.000001 > 0.000000001,
                        ABS(?frameSpan) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER (ABS(?span - ?frameSpan) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Yield strength must be non-zero for the transverse-frame shear-area calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:yieldStrength ?yieldNode .
                ?yieldNode qudt:numericValue ?yieldStrength .
                FILTER (?yieldStrength = 0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The reported required transverse frame shear area must match the formula within tolerance, and the actual shear area converted to square centimetres must be at least the formula-derived area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:averageIcePressure ?pressureNode ;
                      nltl:loadPatchHeight ?heightNode ;
                      nltl:peakPressureFactor ?peakFactorNode ;
                      nltl:shearArea ?actualAreaNode ;
                      nltl:span ?spanNode ;
                      nltl:frameSpan ?frameSpanNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:hasComponent ?areaValue .

                ?member nltl:frameSpacing ?spacingNode ;
                        nltl:selectedHullAreaFactor ?selectedFactorNode ;
                        nltl:requiredTransverseFrameShearArea ?reportedAreaNode .

                ?areaValue a nltl:hullAreaValue ;
                           nltl:hullAreaFactor ?hullFactorNode .

                ?pressureNode qudt:numericValue ?averagePressure .
                ?heightNode qudt:numericValue ?loadPatchHeight .
                ?peakFactorNode qudt:numericValue ?peakPressureFactor .
                ?actualAreaNode qudt:numericValue ?actualArea .
                ?spanNode qudt:numericValue ?span .
                ?frameSpanNode qudt:numericValue ?frameSpan .
                ?yieldNode qudt:numericValue ?yieldStrength .
                ?spacingNode qudt:numericValue ?frameSpacing .
                ?selectedFactorNode qudt:numericValue ?selectedFactor .
                ?reportedAreaNode qudt:numericValue ?reportedArea .
                ?hullFactorNode qudt:numericValue ?hullAreaFactor .

                FILTER (?hullAreaFactor = ?selectedFactor)
                FILTER (?yieldStrength != 0)

                BIND(
                    IF(?span <= ?loadPatchHeight, ?span, ?loadPatchHeight)
                    AS ?LL
                )
                BIND(
                    (
                        10000.0 * 0.5 * ?LL * ?frameSpacing *
                        (?hullAreaFactor * ?peakPressureFactor * ?averagePressure)
                    ) / (0.577 * ?yieldStrength)
                    AS ?expectedArea
                )
                BIND(
                    IF(
                        ABS(?expectedArea) * 0.000001 > 0.000000001,
                        ABS(?expectedArea) * 0.000001,
                        0.000000001
                    )
                    AS ?resultTolerance
                )
                FILTER (
                    ABS(?reportedArea - ?expectedArea) > ?resultTolerance ||
                    (?actualArea * 10000.0) < ?expectedArea
                )
            }
        """
    ] .
