gen:StructuralMemberShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMember ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;

                sh:path qudt:unit ;

            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;

                sh:maxCount 1

                sh:path qudt:unit ;

                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredLongitudinalFrameShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:CentiM2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The required longitudinal-frame shear area must match the prescribed calculation, and the actual net effective shear area must be at least the calculated requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasStructuralMember $this ;
                      nltl:averageIcePressure/qudt:numericValue ?averagePressureValue ;
                      nltl:frameSpan/qudt:numericValue ?frameSpanValue ;
                      nltl:loadPatchHeight/qudt:numericValue ?loadPatchHeightValue ;
                      nltl:peakPressureFactor/qudt:numericValue ?peakPressureFactorValue ;
                      nltl:shearArea/qudt:numericValue ?actualShearAreaValue ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrengthValue .

                $this nltl:frameSpacing/qudt:numericValue ?frameSpacingValue ;
                      nltl:selectedHullAreaFactor/qudt:numericValue ?areaFactorValue ;
                      nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reportedRequiredAreaValue .

                BIND(xsd:decimal(?averagePressureValue) AS ?averagePressure)
                BIND(xsd:decimal(?frameSpanValue) AS ?frameSpan)
                BIND(xsd:decimal(?loadPatchHeightValue) AS ?loadPatchHeight)
                BIND(xsd:decimal(?peakPressureFactorValue) AS ?peakPressureFactor)
                BIND(xsd:decimal(?actualShearAreaValue) AS ?actualShearArea)
                BIND(xsd:decimal(?yieldStrengthValue) AS ?yieldStrength)
                BIND(xsd:decimal(?frameSpacingValue) AS ?frameSpacing)
                BIND(xsd:decimal(?areaFactorValue) AS ?areaFactor)
                BIND(xsd:decimal(?reportedRequiredAreaValue) AS ?reportedRequiredArea)

                BIND(IF(?frameSpan = 0, 1, ?frameSpan) AS ?safeFrameSpan)
                BIND(?loadPatchHeight / ?safeFrameSpan AS ?bPrime)
                BIND(IF(?bPrime = 0, 1, ?bPrime) AS ?safeBPrime)
                BIND(
                    IF(
                        ?bPrime < 2,
                        ?loadPatchHeight * (1 - (0.25 * ?bPrime)),
                        ?frameSpan
                    )
                    AS ?b2
                )
                BIND((1 - (0.3 / ?safeBPrime)) AS ?ko)
                BIND((?ko * ?b2) AS ?b1)
                BIND(IF(?yieldStrength = 0, 1, ?yieldStrength) AS ?safeYieldStrength)
                BIND(
                    (
                        10000
                        * (?areaFactor * ?peakPressureFactor * ?averagePressure)
                        * 0.5
                        * ?b1
                        * ?frameSpacing
                    )
                    / (0.577 * ?safeYieldStrength)
                    AS ?expectedRequiredArea
                )
                BIND(ABS(?expectedRequiredArea) AS ?expectedMagnitude)
                BIND(
                    IF(
                        (?expectedMagnitude * 0.000001) > 0.000000001,
                        ?expectedMagnitude * 0.000001,
                        0.000000001
                    )
                    AS ?formulaTolerance
                )

                FILTER(
                    ?frameSpan = 0
                    || ?bPrime = 0
                    || ?yieldStrength = 0
                    || ABS(?reportedRequiredArea - ?expectedRequiredArea) > ?formulaTolerance
                    || (?actualShearArea * 10000) < ?expectedRequiredArea
                )
            }
        """
    ] .
