@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_040_ReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gen:I2_040_YieldStrengthValueShape ;
        sh:message "Yield strength is required for the local web buckling assessment."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_040_StructuralMemberShape
    ] .

gen:I2_040_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralSectionType ;
        sh:minCount 1 ;
        sh:in (
            nltl:flatBarStructuralSection
            nltl:bulbSection
            nltl:teeSection
            nltl:angleSection
        ) ;
        sh:message "The structural member requires a permitted structural section type."
    ] ;
    sh:property [
        sh:path nltl:webHeight ;
        sh:minCount 1 ;
        sh:node gen:I2_040_LengthValueShape ;
        sh:message "Web height in millimetres is required for the local web buckling assessment."
    ] ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:node gen:I2_040_LengthValueShape ;
        sh:message "Net web thickness in millimetres is required as the reported assessment input and result."
    ] .

gen:I2_040_LengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:maxCount 1
    ] .

gen:I2_040_YieldStrengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:maxCount 1
    ] .
<END_SHACL>
