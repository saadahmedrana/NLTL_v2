<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2-048ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:asBuiltThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AsBuiltThicknessShape ;
        sh:message "The ship must provide one as-built thickness in metres."
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:message "The ship must identify one Polar Class."
    ] ;
    sh:property [
        sh:path nltl:materialClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship must provide one material class value."
    ] ;
    sh:property [
        sh:path nltl:steelGrade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship must provide one steel grade value."
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node gen:SteelGradeRequirementCaseShape ;
        sh:message "The ship must identify a steel-grade requirement case."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:TableLookupCaseShape
    ] ;
    sh:sparql [
        sh:message "Each steel-grade requirement case must also be linked as the corresponding table-lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasTableLookupCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual hull structural steel grade must have an order rank not less than the required Table 8 grade rank." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasSteelGradeRequirementCase ?case .
                ?case nltl:actualHullStructuralSteelGrade ?actualGrade ;
                      nltl:requiredHullStructuralSteelGrade ?requiredGrade .
                ?actualGrade nltl:steelGradeOrderRank ?actualRank .
                ?requiredGrade nltl:steelGradeOrderRank ?requiredRank .
                FILTER (?actualRank < ?requiredRank)
            }
        """
    ] .

    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

            nltl:steelMaterialClassOne

    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;

    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;

        sh:message "The steel-grade requirement case must identify one ranked required hull structural steel grade."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;

        sh:message "The steel-grade requirement case must use IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact ;

    ] .
