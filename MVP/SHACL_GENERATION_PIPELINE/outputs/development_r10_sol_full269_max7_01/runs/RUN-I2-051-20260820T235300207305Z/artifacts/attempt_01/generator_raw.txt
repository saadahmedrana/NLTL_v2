<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_051_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:message "Design vertical ice force at bow is required in meganewtons."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:message "Design vertical ice shear force is required in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:message "The longitudinal distribution factor is required."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "Upper ice waterline length LUI is required in metres."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_051_PositiveShearCaseShape ;
        sh:message "A positive-shear calculation case with longitudinal position is required."
    ] ;
    sh:sparql [
        sh:message "For each positive-shear case, FI must equal Cf multiplied by FIB; Cf must be 0.0 from the aft end through 0.6 LUI and 1.0 from 0.9 LUI through the forward end." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?case
            WHERE {
                $this nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity ;
                      nltl:longitudinalDistributionFactor ?cfQuantity ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:hasCalculationCase ?case .

                ?case nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity .

                ?fibQuantity qudt:numericValue ?fib .
                ?fiQuantity qudt:numericValue ?fi .
                ?cfQuantity qudt:numericValue ?cf .
                ?luiQuantity qudt:numericValue ?lui .
                ?positionQuantity qudt:numericValue ?position .

                BIND((?cf * ?fib * "1000"^^xsd:decimal) AS ?expectedFi)
                BIND(
                    IF(
                        ABS(?expectedFi) * "0.000001"^^xsd:decimal
                            > "0.000000001"^^xsd:decimal,
                        ABS(?expectedFi) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?fiTolerance
                )

                FILTER (
                    ABS(?fi - ?expectedFi) > ?fiTolerance
                    ||
                    (
                        ?position >= "0.0"^^xsd:decimal
                        &&
                        ?position <= ("0.6"^^xsd:decimal * ?lui)
                        &&
                        (
                            ?cf < "0.0"^^xsd:decimal
                            ||
                            ?cf > "0.0"^^xsd:decimal
                        )
                    )
                    ||
                    (
                        ?position >= ("0.9"^^xsd:decimal * ?lui)
                        &&
                        ?position <= ?lui
                        &&
                        (
                            ?cf < "1.0"^^xsd:decimal
                            ||
                            ?cf > "1.0"^^xsd:decimal
                        )
                    )
                )
            }
        """
    ] .

gen:I2_051_PositiveShearCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:hasValue nltl:positiveShearForce ;
        sh:message "The calculation case must identify the positive shear-force direction."
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The calculation case requires a longitudinal position from aft in metres."
    ] .

gen:MegaNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
