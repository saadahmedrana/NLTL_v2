<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_061_CalculationCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must identify a calculation case for the prescribed analytical-procedure assessment."
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationMethodValue ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify its calculation method."
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gen:I2_061_AssessedHullStructureShape ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify the assessed hull structure."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A direct calculation method must not replace the prescribed analytical procedure." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:calculationMethod nltl:directCalculationMethodValue .
    FILTER NOT EXISTS {
        $this nltl:calculationMethod nltl:prescribedAnalyticalProcedure .
    }
}
"""
    ] .

gen:I2_061_AssessedHullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node gen:I2_061_ThicknessValueShape ;
        sh:message "The assessed hull structure must provide a thickness quantity."
    ] .

gen:I2_061_ThicknessValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The thickness quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:maxCount 1 ;
        sh:message "The thickness quantity must use metre as its single unit."
    ] .
<END_SHACL>
