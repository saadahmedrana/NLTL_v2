<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal numeric value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must have millimetre as its sole unit."
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal numeric value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must have the unitless unit as its sole unit."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:class nltl:calculationMethodValue ;
        sh:minCount 1 ;
        sh:message "Each calculation case must identify its calculation method."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            sh:property [
                sh:path nltl:permanentLateralDeformation ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "A nonlinear calculation case requires permanent lateral deformation."
            ] ;
            sh:property [
                sh:path nltl:permanentOutOfPlaneDeformation ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "A nonlinear calculation case requires permanent out-of-plane deformation."
            ] ;
            sh:property [
                sh:path nltl:relevantStructuralDimension ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "A nonlinear calculation case requires the relevant structural dimension."
            ] ;
            sh:property [
                sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "A nonlinear calculation case requires acceptance evidence."
            ] ;
            sh:property [
                sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:in ( true ) ;
                sh:minCount 1 ;
                sh:message "The minor permanent deformation assessment must be satisfied."
            ]
        ]
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
}
"""
            ]
        ]
    ) .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:calculationMethod ;
                        sh:hasValue nltl:nonlinearCalculationMethodValue
                    ]
                ] ;
                sh:qualifiedMinCount 1
            ] ;
            sh:property [
                sh:path nltl:fractureMargin ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:message "A ship with a nonlinear calculation case requires a fracture margin."
            ] ;
            sh:property [
                sh:path nltl:majorBucklingMargin ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:message "A ship with a nonlinear calculation case requires a major buckling margin."
            ] ;
            sh:property [
                sh:path nltl:yieldingMargin ;
                sh:node gen:DimensionlessQuantityShape ;
                sh:minCount 1 ;
                sh:message "A ship with a nonlinear calculation case requires a yielding margin."
            ]
        ]
        [
            sh:sparql [
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
}
"""
            ]
        ]
    ) .
<END_SHACL>
