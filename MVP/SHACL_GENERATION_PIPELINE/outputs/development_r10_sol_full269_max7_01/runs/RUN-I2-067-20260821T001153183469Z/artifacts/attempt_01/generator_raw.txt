<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2-067-StrengthContinuityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with structural connections must record strength continuity as true, without conflicting or invalid status values." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:structuralConnection ?connection .
    FILTER (
        NOT EXISTS {
            $this nltl:strengthContinuityStatus ?validStatus .
            FILTER (
                datatype(?validStatus) = xsd:boolean &&
                ?validStatus = true
            )
        }
        ||
        EXISTS {
            $this nltl:strengthContinuityStatus ?invalidStatus .
            FILTER (
                datatype(?invalidStatus) != xsd:boolean ||
                ?invalidStatus != true
            )
        }
    )
}
"""
    ] .
<END_SHACL>
