<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO005Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ice condition is 'ice free waters', total ice present must be recorded as false without conflicting values." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER(?condition = "ice free waters"^^xsd:string)
    FILTER(
        NOT EXISTS {
            $this nltl:totalIcePresent ?present .
            FILTER(
                DATATYPE(?present) = xsd:boolean &&
                STR(?present) IN ("false", "0")
            )
        }
        ||
        EXISTS {
            $this nltl:totalIcePresent ?present .
            FILTER(
                DATATYPE(?present) != xsd:boolean ||
                !(STR(?present) IN ("false", "0"))
            )
        }
    )
}
"""
    ] .
<END_SHACL>
