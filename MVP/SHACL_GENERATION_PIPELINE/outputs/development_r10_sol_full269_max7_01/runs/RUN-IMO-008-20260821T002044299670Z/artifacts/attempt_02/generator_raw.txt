<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:ThinFirstYearIceThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:message "Ice type classifications must be IRIs."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as thin first-year ice must have ice thickness from 0.3 m to 0.7 m, represented as a QUDT QuantityValue with decimal numeric values and metre units." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:iceTypeClassification nltl:thinFirstYearIce .
    FILTER (
        NOT EXISTS {
            $this nltl:iceThickness ?anyThickness .
        }
        ||
        EXISTS {
            $this nltl:iceThickness ?thickness .
            FILTER (
                NOT EXISTS {
                    ?thickness a qudt:QuantityValue .
                }
                ||
                NOT EXISTS {
                    ?thickness qudt:numericValue ?someValue .
                }
                ||
                EXISTS {
                    ?thickness qudt:numericValue ?nonDecimalValue .
                    FILTER (datatype(?nonDecimalValue) != xsd:decimal)
                }
                ||
                EXISTS {
                    ?thickness qudt:numericValue ?decimalValue .
                    FILTER (
                        datatype(?decimalValue) = xsd:decimal
                        &&
                        (
                            ?decimalValue < "0.3"^^xsd:decimal
                            ||
                            ?decimalValue > "0.7"^^xsd:decimal
                        )
                    )
                }
                ||
                NOT EXISTS {
                    ?thickness qudt:unit unit:M .
                }
                ||
                EXISTS {
                    ?thickness qudt:unit ?otherUnit .
                    FILTER (?otherUnit != unit:M)
                }
            )
        }
    )
}
"""
    ] .
<END_SHACL>
