<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BergyWatersApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Ice condition is required to determine whether the bergy-waters limits apply."
    ] .

gen:BergyWatersConcentrationShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:iceCondition ?condition .
                FILTER(?condition = "bergy waters"^^xsd:string)
            }
        """
    ] ;
    sh:property [
        sh:path nltl:landOriginIceConcentration ;
        sh:minCount 1 ;
        sh:message "Land-origin ice concentration is required for bergy waters." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive "1"^^xsd:decimal ;
                sh:message "Land-origin ice concentration must be less than one tenth."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:UNITLESS
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:totalIceConcentration ;
        sh:minCount 1 ;
        sh:message "Total ice concentration is required for bergy waters." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "1"^^xsd:decimal ;
                sh:message "Total ice concentration must not exceed one tenth."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:UNITLESS
            ]
        ]
    ] .
<END_SHACL>
