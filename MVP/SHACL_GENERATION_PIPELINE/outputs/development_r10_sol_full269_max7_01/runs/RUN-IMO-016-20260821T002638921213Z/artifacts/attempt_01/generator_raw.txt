<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PolarShipCertificateRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:node gen:PolarShipCertificateFormShape ;
        sh:message "The ship must have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The certificate language must be recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:in ( "English" "French" "Spanish" )
            ]
        ]
        [
            sh:property [
                sh:path nltl:approvedTranslationPresent ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( true ) ;
                sh:message "A certificate in another language must include an approved translation."
            ]
        ]
    ) .

gen:PolarShipCertificateFormShape
    a sh:NodeShape ;
    sh:class nltl:polarShipCertificateForm ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:class nltl:certificateFormModelValue ;
        sh:in ( nltl:appendixIPolarShipCertificateModel ) ;
        sh:message "The Polar Ship Certificate form must correspond to the Appendix I model."
    ] .
<END_SHACL>
