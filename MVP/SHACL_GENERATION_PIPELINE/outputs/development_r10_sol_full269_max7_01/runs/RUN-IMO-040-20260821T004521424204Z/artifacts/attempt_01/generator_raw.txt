<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_040_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceAccretionLikely ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Ice accretion likelihood is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When ice accretion is likely, every relevant hatch or door must have an associated ice or snow removal or prevention means." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?hatchOrDoor
            WHERE {
                $this nltl:iceAccretionLikely "true"^^xsd:boolean ;
                      nltl:hasRelevantHatchOrDoor ?hatchOrDoor .
                FILTER NOT EXISTS {
                    ?hatchOrDoor
                        nltl:hasIceOrSnowRemovalOrPreventionMeans ?means .
                    FILTER(isIRI(?means))
                }
            }
        """
    ] .
<END_SHACL>
