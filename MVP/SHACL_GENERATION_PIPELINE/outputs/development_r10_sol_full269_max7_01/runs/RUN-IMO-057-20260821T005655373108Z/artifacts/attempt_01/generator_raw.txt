@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO057Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each fire pump must have a containing compartment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?pump
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER NOT EXISTS {
                    ?pump nltl:hasContainingCompartment ?compartment .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each containing compartment of a fire pump must be typed as a compartment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?pump ?compartment
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER NOT EXISTS {
                    ?compartment a nltl:compartment .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each containing compartment of a fire pump must have a maintained temperature." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?pump ?compartment
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER NOT EXISTS {
                    ?compartment nltl:maintainedTemperature ?temperature .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A maintained temperature must be represented as a QUDT QuantityValue." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this ?pump ?compartment ?temperature
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                ?compartment nltl:maintainedTemperature ?temperature .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER NOT EXISTS {
                    ?temperature a qudt:QuantityValue .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A maintained temperature must have a numeric value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this ?pump ?compartment ?temperature
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                ?compartment nltl:maintainedTemperature ?temperature .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER NOT EXISTS {
                    ?temperature qudt:numericValue ?numericValue .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each maintained temperature value must be an xsd:decimal greater than 0 degrees Celsius." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?pump ?compartment ?temperature ?numericValue
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                ?compartment nltl:maintainedTemperature ?temperature .
                ?temperature qudt:numericValue ?numericValue .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER (
                    DATATYPE(?numericValue) != xsd:decimal ||
                    ?numericValue <= "0"^^xsd:decimal
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each maintained temperature must use only the canonical degree Celsius unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this ?pump ?compartment ?temperature
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump a ?pumpType ;
                    nltl:hasContainingCompartment ?compartment .
                ?compartment nltl:maintainedTemperature ?temperature .
                FILTER (?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
                FILTER (
                    NOT EXISTS {
                        ?temperature qudt:unit unit:DEG_C .
                    } ||
                    EXISTS {
                        ?temperature qudt:unit ?otherUnit .
                        FILTER (?otherUnit != unit:DEG_C)
                    }
                )
            }
        """
    ] .
