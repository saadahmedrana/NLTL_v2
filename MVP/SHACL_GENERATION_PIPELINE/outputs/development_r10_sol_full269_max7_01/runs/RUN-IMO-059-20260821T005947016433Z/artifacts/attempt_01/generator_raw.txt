<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:FirefighterOutfitStorageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFirefighterOutfit ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:firefighterOutfitItem ;
        sh:node gen:FirefighterOutfitItemShape ;
        sh:message "Each linked firefighter outfit must be stored in a warm-classified location."
    ] .

gen:FirefighterOutfitItemShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasStorageLocation ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:physicalStorageLocation ;
        sh:node gen:WarmStorageLocationShape ;
        sh:message "A firefighter outfit must have a physical storage location classified as warm."
    ] .

gen:WarmStorageLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:storageTemperatureClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:storageTemperatureClassValue ;
        sh:in ( nltl:warmStorageLocation ) ;
        sh:message "The storage location must be classified as warm."
    ] .
<END_SHACL>
