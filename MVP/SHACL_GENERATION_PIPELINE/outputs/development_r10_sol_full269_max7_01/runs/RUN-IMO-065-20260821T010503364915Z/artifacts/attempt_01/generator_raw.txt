<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DayDurationShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The duration must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DAY ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The duration must have unit day."
    ] .

gen:SurvivalResourceCoverageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:abandonmentEnvironment ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "At least one assessed abandonment environment is required."
    ] ;
    sh:property [
        sh:path nltl:maximumExpectedRescueTime ;
        sh:node gen:DayDurationShape ;
        sh:minCount 1 ;
        sh:message "Maximum expected rescue time is required."
    ] ;
    sh:property [
        sh:path nltl:survivalSupportDuration ;
        sh:node gen:DayDurationShape ;
        sh:minCount 1 ;
        sh:message "Survival support duration is required."
    ] ;
    sh:property [
        sh:path nltl:habitableEnvironmentPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "A habitable environment must be provided."
    ] ;
    sh:property [
        sh:path nltl:weatherProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Protection from cold, wind, and sun must be provided."
    ] ;
    sh:property [
        sh:path nltl:accommodationSpacePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Accommodation space must be provided."
    ] ;
    sh:property [
        sh:path nltl:sustenanceMeansPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Means of sustenance must be provided."
    ] ;
    sh:property [
        sh:path nltl:safeAccessExitPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Safe access and exit points must be provided."
    ] ;
    sh:property [
        sh:path nltl:rescueCommunicationPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Means of communication with rescue assets must be provided."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Survival support duration must be at least the maximum expected rescue time." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:survivalSupportDuration ?supportDuration .
    ?supportDuration qudt:numericValue ?supportValue .
    $this nltl:maximumExpectedRescueTime ?rescueTime .
    ?rescueTime qudt:numericValue ?rescueValue .
    FILTER (xsd:decimal(?supportValue) < xsd:decimal(?rescueValue))
}
"""
    ] .
<END_SHACL>
