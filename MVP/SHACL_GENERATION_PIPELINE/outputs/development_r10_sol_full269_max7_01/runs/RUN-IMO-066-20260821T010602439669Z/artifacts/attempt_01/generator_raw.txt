<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ProtectedItemShape
    a sh:NodeShape ;
    sh:class nltl:protectedItem ;
    sh:property [
        sh:path nltl:protectedItemCategory ;
        sh:minCount 1 ;
        sh:in (
            nltl:escapeRouteProtectedItem
            nltl:musterStationProtectedItem
            nltl:embarkationAreaProtectedItem
            nltl:survivalCraftProtectedItem
            nltl:launchingApplianceProtectedItem
            nltl:survivalCraftAccessProtectedItem
        ) ;
        sh:message "Each protected item must identify at least one applicable protected-item category."
    ] ;
    sh:property [
        sh:path nltl:hasIceSnowRemovalOrPreventionMeans ;
        sh:minCount 1 ;
        sh:class nltl:iceSnowRemovalOrPreventionMeans ;
        sh:message "Each protected item must have associated ice or snow removal or prevention means."
    ] .

gen:IceAccretionProtectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipExposedToIceAccretion ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship's exposure to ice accretion must be recorded as one boolean value."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipExposedToIceAccretion ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:node gen:ProtectedItemShape
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:escapeRouteProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include a protected escape route."
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:musterStationProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include a protected muster station."
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:embarkationAreaProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include a protected embarkation area."
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:survivalCraftProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include protected survival craft."
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:launchingApplianceProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include a protected launching appliance."
            ] ;
            sh:property [
                sh:path nltl:hasProtectedItem ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:protectedItemCategory ;
                        sh:hasValue nltl:survivalCraftAccessProtectedItem
                    ]
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "An ice-exposed ship must include protected access to survival craft."
            ]
        ]
    ) .
<END_SHACL>
