<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:PassengerShipPersonalSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A passenger ship must have exactly one size compatibility status with the value true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:passengerShip .
                FILTER NOT EXISTS {
                    $this nltl:sizeCompatibilityStatus ?status .
                    FILTER(
                        DATATYPE(?status) = xsd:boolean &&
                        ?status = true
                    )
                    FILTER NOT EXISTS {
                        $this nltl:sizeCompatibilityStatus ?otherStatus .
                        FILTER(!sameTerm(?otherStatus, ?status))
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each person linked to a passenger ship must have exactly one assignment containing exactly one immersion suit or thermal protective aid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?value
            WHERE {
                $this a nltl:passengerShip ;
                    nltl:hasPersonOnBoardMember ?person .
                BIND(?person AS ?value)
                FILTER NOT EXISTS {
                    FILTER(isIRI(?person))
                    ?person a nltl:personOnBoardMember ;
                        nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                    FILTER(isIRI(?assignment))
                    ?assignment a nltl:personalSurvivalEquipmentAssignment ;
                        nltl:assignedPersonalSurvivalEquipment ?equipment .
                    FILTER(isIRI(?equipment))
                    FILTER(
                        EXISTS {
                            ?equipment a nltl:immersionSuitEquipment .
                        } ||
                        EXISTS {
                            ?equipment a nltl:thermalProtectiveAidEquipment .
                        }
                    )
                    FILTER NOT EXISTS {
                        ?person nltl:hasPersonalSurvivalEquipmentAssignment ?otherAssignment .
                        FILTER(!sameTerm(?otherAssignment, ?assignment))
                    }
                    FILTER NOT EXISTS {
                        ?assignment nltl:assignedPersonalSurvivalEquipment ?otherEquipment .
                        FILTER(!sameTerm(?otherEquipment, ?equipment))
                    }
                }
            }
        """
    ] .
<END_SHACL>
