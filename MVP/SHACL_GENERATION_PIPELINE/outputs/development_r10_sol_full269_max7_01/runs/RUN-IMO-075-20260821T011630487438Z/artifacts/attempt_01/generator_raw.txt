<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:SurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:SurvivalEquipmentNotRequiredBranch
        gs:SurvivalEquipmentRequiredBranch
    ) .

gs:SurvivalEquipmentNotRequiredBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( false ) ;
        sh:message "Personal or group survival equipment applicability must be explicitly false for this branch."
    ] .

gs:SurvivalEquipmentRequiredBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Personal or group survival equipment applicability must be explicitly true for this branch."
    ] ;
    sh:property [
        sh:path nltl:personsOnBoard ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Persons on board is required when survival equipment is required."
    ] ;
    sh:property [
        sh:path nltl:availableSurvivalEquipmentCapacity ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Available survival equipment capacity is required when survival equipment is required."
    ] ;
    sh:property [
        sh:path nltl:storageAccessibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "Survival equipment storage must be accessible."
    ] ;
    sh:sparql [
        sh:message "Available survival equipment capacity must be at least 110 percent of persons on board, rounded up." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:personalOrGroupSurvivalEquipmentRequired true ;
                      nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (
                    xsd:decimal(?capacity) <
                    CEIL(xsd:decimal(?persons) * 1.10)
                )
            }
        """
    ] .
<END_SHACL>
