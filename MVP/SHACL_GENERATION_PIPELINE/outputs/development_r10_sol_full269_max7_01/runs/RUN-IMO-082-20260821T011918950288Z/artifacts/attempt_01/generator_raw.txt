<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO082ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:in ( "true"^^xsd:boolean ) ;
        sh:minCount 1 ;
        sh:message "The ship shall comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:bridgeConfigurationValue ;
        sh:minCount 1 ;
        sh:message "Bridge configuration information is required to determine whether a clear view astern is required."
    ] ;
    sh:property [
        sh:path nltl:clearViewAsternStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A compliant clear-view-astern status is required for this bridge configuration." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:bridgeConfigurationType
        nltl:asternViewRequiredBridgeConfiguration .
    FILTER (
        NOT EXISTS {
            $this nltl:clearViewAsternStatus ?status .
        }
        ||
        EXISTS {
            $this nltl:clearViewAsternStatus ?status .
            FILTER (
                !isLiteral(?status)
                || DATATYPE(?status) != xsd:boolean
                || ?status != "true"^^xsd:boolean
            )
        }
    )
}
"""
    ] .
<END_SHACL>
