@prefix gen: <urn:nltl:generated-shape:> .

@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LatitudeAtMost80Shape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive "80"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LatitudeOver80Shape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive "80"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PolarNavigationEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:plannedOrActualLatitude ;
                sh:minCount 1 ;
                sh:node gen:LatitudeAtMost80Shape
            ]
        ]
        [
            sh:property [
                sh:path nltl:plannedOrActualLatitude ;
                sh:minCount 1 ;
                sh:node gen:LatitudeOver80Shape
            ] ;
            sh:property [
                sh:path nltl:gnssCompassOrEquivalentCount ;
                sh:minCount 1 ;
                sh:datatype xsd:integer ;
                sh:minInclusive 1 ;
                sh:message "A ship proceeding above 80 degrees latitude must have at least one GNSS compass or equivalent."
            ] ;
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:message "A ship proceeding above 80 degrees latitude must record a connection to main power."
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:message "A ship proceeding above 80 degrees latitude must record a connection to emergency power."
            ]
        ]
    ) ;
    sh:message "The latitude must be stated in degrees; above 80 degrees the required GNSS compass count and power connections must be present." .
