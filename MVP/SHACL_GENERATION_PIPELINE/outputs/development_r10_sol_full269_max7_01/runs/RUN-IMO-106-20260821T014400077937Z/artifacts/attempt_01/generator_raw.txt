<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO106Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "A discharge material classification is required."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Paragraph 1.1.1 must be recorded as not applicable for clean or segregated ballast discharge." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                    }
                    ||
                    EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                        FILTER (
                            DATATYPE(?applicable) = xsd:boolean &&
                            ?applicable = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
