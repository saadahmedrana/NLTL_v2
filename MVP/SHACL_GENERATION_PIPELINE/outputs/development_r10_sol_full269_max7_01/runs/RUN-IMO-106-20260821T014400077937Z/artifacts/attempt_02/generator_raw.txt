<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:CleanOrSegregatedBallastExceptionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean or segregated ballast, paragraph 1.1.1 applicability must be present as boolean false with no conflicting value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                        FILTER (
                            isLiteral(?applicable) &&
                            datatype(?applicable) = xsd:boolean &&
                            ?applicable = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?otherValue .
                        FILTER (
                            !(
                                isLiteral(?otherValue) &&
                                datatype(?otherValue) = xsd:boolean &&
                                ?otherValue = false
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
