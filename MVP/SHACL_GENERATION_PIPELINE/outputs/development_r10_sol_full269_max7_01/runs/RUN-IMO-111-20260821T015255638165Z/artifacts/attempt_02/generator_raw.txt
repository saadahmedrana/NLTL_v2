<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO111Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:polarShipCategoryC ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, each represented oil residue or oily bilge water holding tank with capacity greater than 30 m3 requires an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      a ?tankClass .

                FILTER(
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )
                FILTER(
                    ?tankClass IN (
                        nltl:oilResidueTank,
                        nltl:oilyBilgeWaterHoldingTank
                    )
                )
                FILTER(
                    YEAR(?date) > 2017 ||
                    (
                        YEAR(?date) = 2017 &&
                        (
                            MONTH(?date) > 1 ||
                            (
                                MONTH(?date) = 1 &&
                                DAY(?date) >= 1
                            )
                        )
                    )
                )

                FILTER NOT EXISTS {
                    $this nltl:individualTankCapacity ?capacity .
                    ?capacity a qudt:QuantityValue ;
                              qudt:numericValue ?capacityValue ;
                              qudt:unit unit:M3 .
                    FILTER(DATATYPE(?capacityValue) = xsd:decimal)
                    FILTER NOT EXISTS {
                        ?capacity qudt:unit ?otherCapacityUnit .
                        FILTER(?otherCapacityUnit != unit:M3)
                    }
                    FILTER(
                        ?capacityValue <= 30.0 ||
                        EXISTS {
                            $this nltl:tankOuterShellSeparation ?separation .
                            ?separation a qudt:QuantityValue ;
                                        qudt:numericValue ?separationValue ;
                                        qudt:unit unit:M .
                            FILTER(DATATYPE(?separationValue) = xsd:decimal)
                            FILTER(?separationValue >= 0.76)
                            FILTER NOT EXISTS {
                                ?separation qudt:unit ?otherSeparationUnit .
                                FILTER(?otherSeparationUnit != unit:M)
                            }
                        }
                    )
                }
            }
        """
    ] .
<END_SHACL>
