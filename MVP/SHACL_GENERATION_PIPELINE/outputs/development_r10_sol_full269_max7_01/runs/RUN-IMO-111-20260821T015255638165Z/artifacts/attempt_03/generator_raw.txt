<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO111Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Applicable Category A or B ships with a represented oil residue or oily bilge water holding tank exceeding 30 m3 must have an outer-shell separation of at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:hasComponent ?tank ;
                      nltl:individualTankCapacity ?capacityQuantity .

                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )

                FILTER (
                    DATATYPE(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )

                ?tank a ?tankClass .
                FILTER (
                    ?tankClass IN (
                        nltl:oilResidueTank,
                        nltl:oilyBilgeWaterHoldingTank
                    )
                )

                ?capacityQuantity
                    qudt:numericValue ?capacity ;
                    qudt:unit unit:M3 .

                FILTER (
                    DATATYPE(?capacity) = xsd:decimal &&
                    ?capacity > "30"^^xsd:decimal
                )

                FILTER NOT EXISTS {
                    $this nltl:tankOuterShellSeparation ?separationQuantity .
                    ?separationQuantity
                        qudt:numericValue ?separation ;
                        qudt:unit unit:M .
                    FILTER (
                        DATATYPE(?separation) = xsd:decimal &&
                        ?separation >= "0.76"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
