<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_120_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The operating area is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:arcticWaters ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Arctic-waters value is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The food-waste discharge state is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For food-waste discharge in Arctic waters, the nearest-feature distance must be at least 12 nautical miles and the distance from an area with ice concentration above one tenth must be recorded." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:operatingArea ?operatingArea ;
          nltl:arcticWaters ?arcticWaters ;
          nltl:foodWasteDischargeToSea ?discharge .
    FILTER(
        ?operatingArea = ?arcticWaters &&
        ?discharge = "true"^^xsd:boolean
    )
    FILTER(
        NOT EXISTS {
            $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearestDistance .
            ?nearestDistance qudt:numericValue ?nearestValue ;
                             qudt:unit unit:MI_N .
            FILTER(
                datatype(?nearestValue) = xsd:decimal &&
                ?nearestValue >= "12"^^xsd:decimal
            )
        }
        ||
        NOT EXISTS {
            $this nltl:nearbyIceConcentration ?nearbyConcentration .
            ?nearbyConcentration qudt:numericValue ?concentrationValue ;
                                 qudt:unit unit:UNITLESS .
            FILTER(
                datatype(?concentrationValue) = xsd:decimal &&
                ?concentrationValue > "1"^^xsd:decimal
            )
        }
        ||
        NOT EXISTS {
            $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?iceAreaDistance .
            ?iceAreaDistance qudt:numericValue ?iceAreaDistanceValue ;
                             qudt:unit unit:MI_N .
            FILTER(datatype(?iceAreaDistanceValue) = xsd:decimal)
        }
    )
}
"""
    ] .
<END_SHACL>
