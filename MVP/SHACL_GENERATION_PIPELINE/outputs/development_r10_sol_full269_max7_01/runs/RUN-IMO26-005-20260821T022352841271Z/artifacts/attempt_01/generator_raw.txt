<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:HeadingOrPositionSystemSuitabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:headingOrPositionSystem ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:headingOrPositionSystem ;
                sh:minCount 1 ;
                sh:datatype xsd:string
            ] ;
            sh:property [
                sh:path nltl:intendedAreaSuitabilityApprovalStatus ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:evidenceStateApproved ) ;
                sh:message "Each ship with a heading-reference or position-fixing system must have approved intended-area suitability status."
            ]
        ]
    ) .
<END_SHACL>
