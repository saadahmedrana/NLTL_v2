<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipHeadingMeansShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:node gen:GrossTonnageValueShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:grossTonnage ;
                sh:node gen:GrossTonnageBelow500Shape
            ]
        ]
        [
            sh:property [
                sh:path nltl:grossTonnage ;
                sh:qualifiedValueShape gen:GrossTonnageAtLeast500Shape ;
                sh:qualifiedMinCount 1
            ] ;
            sh:node gen:ApplicableHeadingMeansShape
        ]
    ) .

gen:GrossTonnageValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .

gen:GrossTonnageBelow500Shape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:maxExclusive 500
    ] .

gen:GrossTonnageAtLeast500Shape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minInclusive 500
    ] .

gen:ApplicableHeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node gen:HeadingMeansShape ;
        sh:message "Ships of gross tonnage 500 or more shall have at least two non-magnetic heading means."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means shall identify a distinct independent means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    FILTER (?otherMeans != ?means)
                }
            }
        """
    ] .

gen:HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:message "Each non-magnetic heading means shall be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans
    ] .
<END_SHACL>
