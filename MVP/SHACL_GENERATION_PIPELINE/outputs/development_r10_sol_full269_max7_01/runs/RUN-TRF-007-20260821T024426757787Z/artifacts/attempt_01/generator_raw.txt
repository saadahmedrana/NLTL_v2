<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:minCount 1 ;
                sh:datatype xsd:date ;
                sh:minInclusive "2003-09-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:minCount 1 ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2003-09-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:minCount 1 ;
                sh:datatype xsd:date
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:minCount 1 ;
                sh:datatype xsd:date
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "Once the compliance deadline has been reached, the ship must use the 2017 section 3.2.2 or section 3.2.4 compliance method." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate ;
          nltl:assessmentDate ?assessmentDate .
    FILTER(
        YEAR(?constructionDate) < 2003 ||
        (YEAR(?constructionDate) = 2003 && MONTH(?constructionDate) < 9)
    )
    FILTER(YEAR(?assessmentDate) >= YEAR(?deliveryDate) + 20)
    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod ?method .
        FILTER(
            ?method IN (
                nltl:traficom2017Section3Point2Point2Method,
                nltl:traficom2017Section3Point2Point4Method
            )
        )
    }
}
"""
            ]
        ]
    ) .
<END_SHACL>
