<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF020Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single construction-stage date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A single controlled ice-class value is required to determine applicability."
    ] ;
    sh:or (
        gen:TRF020ConstructionBeforeApplicabilityDate
        gen:TRF020NonApplicableIceClass
        gen:TRF020ApplicableReadiness
    ) .

gen:TRF020ConstructionBeforeApplicabilityDate
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:maxExclusive "2003-09-01"^^xsd:date
    ] .

gen:TRF020NonApplicableIceClass
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF020ApplicableReadiness
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:minCount 1 ;
        sh:node gen:UnitlessDecimalQuantityShape ;
        sh:message "The externally selected engine-output coefficient is required."
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:node gen:DecimalQuantityShape ;
        sh:message "The lower-ice-waterline resistance input is required."
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:node gen:DecimalQuantityShape ;
        sh:message "The upper-ice-waterline resistance input is required."
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:node gen:MetreDecimalQuantityShape ;
        sh:message "Propeller diameter is required."
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Propeller count is required."
    ] ;
    sh:property [
        sh:path nltl:propellerPitchControlType ;
        sh:minCount 1 ;
        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        ) ;
        sh:message "A controlled propeller pitch type is required."
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:minCount 1 ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        ) ;
        sh:message "A controlled propulsion-system type is required."
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:minCount 1 ;
        sh:class nltl:iceWaterline ;
        sh:message "A lower ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:minCount 1 ;
        sh:class nltl:iceWaterline ;
        sh:node gen:TRF020UpperIceWaterlineInputs ;
        sh:message "An upper ice waterline with the required dimensions is required."
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:minCount 1 ;
        sh:node gen:KilowattDecimalQuantityShape ;
        sh:message "The externally calculated minimum power at LIWL is required."
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:minCount 1 ;
        sh:node gen:KilowattDecimalQuantityShape ;
        sh:message "The externally calculated minimum power at UIWL is required."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:node gen:KilowattDecimalQuantityShape ;
        sh:message "Maximum continuous rating power is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be at least both externally calculated minimum powers." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
          nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerQuantity ;
          nltl:minimumRequiredPowerAtUpperIceWaterline ?upperQuantity .
    ?mcrQuantity qudt:numericValue ?mcr .
    ?lowerQuantity qudt:numericValue ?lower .
    ?upperQuantity qudt:numericValue ?upper .
    FILTER(
        xsd:decimal(?mcr) < xsd:decimal(?lower) ||
        xsd:decimal(?mcr) < xsd:decimal(?upper)
    )
}
"""
    ] .

gen:TRF020UpperIceWaterlineInputs
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:upperIceWaterlineLength ;
        sh:minCount 1 ;
        sh:node gen:MetreDecimalQuantityShape ;
        sh:message "Upper ice waterline length is required."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineBreadth ;
        sh:minCount 1 ;
        sh:node gen:MetreDecimalQuantityShape ;
        sh:message "Upper ice waterline breadth is required."
    ] .

gen:DecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

gen:UnitlessDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MetreDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KilowattDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
