<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The power quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloW ;
        sh:message "The power quantity must use kilowatts."
    ] .

gen:TRF023ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassRegulationEdition2002
                    nltl:iceClassRegulationEdition2008
                    nltl:iceClassRegulationEdition2010
                    nltl:iceClassRegulationEdition2017
                    nltl:iceClassRegulationEdition2021
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:message "An applicable ship must identify ice class IB or IC."
            ] ;
            sh:property [
                sh:path nltl:applicableIceClassRegulationEdition ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassRuleEdition1985
                ) ;
                sh:message "An applicable ship must identify the 1985 ice class rules."
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:minCount 1 ;
                sh:node gen:PowerQuantityShape ;
                sh:message "An applicable ship must provide its maximum continuous rating power."
            ] ;
            sh:property [
                sh:path nltl:requiredPowerUnder1985Rules ;
                sh:minCount 1 ;
                sh:node gen:PowerQuantityShape ;
                sh:message "An applicable ship must provide the required power under the 1985 rules."
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "Maximum continuous rating power must be at least the required power under the 1985 rules." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
SELECT $this
WHERE {
    $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
          nltl:requiredPowerUnder1985Rules ?requiredQuantity .
    ?mcrQuantity qudt:numericValue ?mcrValue .
    ?requiredQuantity qudt:numericValue ?requiredValue .
    FILTER (?mcrValue < ?requiredValue)
}
"""
            ]
        ]
    ) ;
    sh:message "The ship must establish a non-applicable alternative or satisfy the 1985 minimum engine-output requirement for ice class IB or IC." .
<END_SHACL>
