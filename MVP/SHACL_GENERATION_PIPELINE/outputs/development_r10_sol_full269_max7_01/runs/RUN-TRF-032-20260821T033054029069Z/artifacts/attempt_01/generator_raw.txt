<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF032Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:regulationRequiredScantling ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:classificationSocietyRequiredScantling ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "When the regulation-required scantling is less than the classification-society-required scantling for a ship that is not ice strengthened, the selected design scantling must be at least the classification-society-required scantling." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:regulationRequiredScantling ?regulationQuantity ;
          nltl:classificationSocietyRequiredScantling ?classificationQuantity .
    ?regulationQuantity qudt:numericValue ?regulationValue .
    ?classificationQuantity qudt:numericValue ?classificationValue .
    FILTER(
        xsd:decimal(?regulationValue) <
        xsd:decimal(?classificationValue)
    )
    FILTER NOT EXISTS {
        $this nltl:selectedDesignScantling ?selectedQuantity .
        ?selectedQuantity a qudt:QuantityValue ;
                          qudt:numericValue ?selectedValue .
        FILTER(
            xsd:decimal(?selectedValue) >=
            xsd:decimal(?classificationValue)
        )
    }
}
"""
            ]
        ]
    ) .
<END_SHACL>
