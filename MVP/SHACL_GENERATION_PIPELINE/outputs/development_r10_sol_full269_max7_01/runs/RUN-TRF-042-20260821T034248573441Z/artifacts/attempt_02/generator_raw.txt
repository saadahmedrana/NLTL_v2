<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:YieldStrengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NormalStrengthYieldShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 235.0 ;
        sh:maxInclusive 235.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:HighStrengthYieldShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 315.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TwoMillimetreAdditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 2.0 ;
        sh:maxInclusive 2.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;

gen:LowerMillimetreAdditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:maxExclusive 2.0 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PlatingRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hullStructuralSteelGrade ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one hull structural steel grade is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:YieldStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:normalStrengthHullStructuralSteel ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:NormalStrengthYieldShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:highStrengthHullStructuralSteel ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:HighStrengthYieldShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:class nltl:steelGradeValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:YieldStrengthQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:TwoMillimetreAdditionShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:LowerMillimetreAdditionShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:specialSurfaceCoatingMaintained ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The reported longitudinal plating factor f2 must match the applicable h/s formula within tolerance max(1e-6*ABS(expected), 1e-12), and h/s must not exceed 1.8." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:spacing ?spacingQuantity ;
                      nltl:longitudinalPlatingFactorF2 ?factorQuantity .
                ?heightQuantity qudt:numericValue ?height .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?factorQuantity qudt:numericValue ?reported .
                BIND(IF(
                    ?spacing = "0"^^xsd:decimal,
                    "0"^^xsd:decimal,
                    ?height / ?spacing
                ) AS ?ratio)
                BIND(IF(
                    ?ratio <= "1"^^xsd:decimal,
                    ("0.4"^^xsd:decimal / ?ratio) + "0.6"^^xsd:decimal,
                    "1.4"^^xsd:decimal - ("0.4"^^xsd:decimal * ?ratio)
                ) AS ?expected)
                BIND(IF(
                    ("0.000001"^^xsd:decimal * ABS(?expected)) > "0.000000000001"^^xsd:decimal,
                    "0.000001"^^xsd:decimal * ABS(?expected),
                    "0.000000000001"^^xsd:decimal
                ) AS ?tolerance)
                FILTER(
                    ?spacing = "0"^^xsd:decimal ||
                    ?ratio > "1.8"^^xsd:decimal ||
                    ABS(?reported - ?expected) > ?tolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Normal-strength steel requires yield strength 235 MPa, and high-strength steel requires at least 315 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade ?grade ;
                      nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER(
                    (?grade = nltl:normalStrengthHullStructuralSteel &&
                     ?yield != "235"^^xsd:decimal) ||
                    (?grade = nltl:highStrengthHullStructuralSteel &&
                     ?yield < "315"^^xsd:decimal)
                )
            }
        """
    ] .
<END_SHACL>
