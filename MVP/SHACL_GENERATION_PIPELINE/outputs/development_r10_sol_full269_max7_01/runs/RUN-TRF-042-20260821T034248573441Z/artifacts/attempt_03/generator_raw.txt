        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;

    ] ;
    sh:property [

        sh:hasValue unit:M ;

    sh:property [
        sh:path qudt:numericValue ;

gen:TRF042PlatingShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:DesignIceLoadHeightValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one design ice load height in metres is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:SpacingValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one spacing value in metres is required."
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:node gen:LongitudinalPlatingFactorValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one dimensionless longitudinal plating factor f2 is required."
    ] ;
    sh:property [
        sh:path nltl:hullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one hull structural steel grade is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:normalStrengthHullStructuralSteel
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:node gen:YieldStrengthValueShape ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive 235 ;
                        sh:maxInclusive 235
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:hasValue nltl:highStrengthHullStructuralSteel
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:node gen:YieldStrengthValueShape ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive 315
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:hullStructuralSteelGrade ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:class nltl:steelGradeValue ;
                sh:not [
                    sh:in (
                        nltl:normalStrengthHullStructuralSteel
                        nltl:highStrengthHullStructuralSteel
                    )
                ]
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:YieldStrengthValueShape
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:class nltl:evidenceState ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:node gen:CorrosionAbrasionAdditionValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one corrosion and abrasion addition in millimetres is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxExclusive 2
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:specialSurfaceCoatingMaintained ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:class nltl:evidenceState ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The reported f2 must satisfy the applicable h/s formula for h/s at most 1.8, within max(1e-6 times the absolute expected value, 1e-12)."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightNode ;
                      nltl:spacing ?spacingNode ;
                      nltl:longitudinalPlatingFactorF2 ?factorNode .
                ?heightNode qudt:numericValue ?height .
                ?spacingNode qudt:numericValue ?spacing .
                ?factorNode qudt:numericValue ?reportedFactor .
                BIND(IF(?spacing != 0, ?height / ?spacing, 0) AS ?ratio)
                BIND(
                    IF(
                        ?spacing != 0 && ?ratio != 0,
                        IF(
                            ?ratio <= 1,
                            0.4 / ?ratio + 0.6,
                            1.4 - 0.4 * ?ratio
                        ),
                        0
                    )
                    AS ?expectedFactor
                )
                BIND(0.000001 * ABS(?expectedFactor) AS ?relativeTolerance)
                BIND(
                    IF(
                        ?relativeTolerance > 0.000000000001,
                        ?relativeTolerance,
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER(
                    ?spacing = 0 ||
                    ?ratio = 0 ||
                    ?ratio > 1.8 ||
                    ABS(?reportedFactor - ?expectedFactor) > ?tolerance
                )
            }
        """
    ] .
<END_SHACL>
