<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node gen:MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An installed bracket must have one thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bracket must be stiffened against buckling."
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] .

gen:PassThroughFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minInclusive 2 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "A frame passing through a supporting structure must have its web plate connected on both sides."
            ]
        ]
    ) .

gen:TerminationAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasConnectionBracket ;
                sh:class nltl:connectionBracket ;
                sh:node gen:ConnectionBracketShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:similarAttachmentConstructionPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) .

gen:ActiveTransverseFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path [
                    sh:alternativePath (
                        nltl:hasFrameAttachment
                        nltl:frameAttachment
                    )
                ] ;
                sh:qualifiedValueShape gen:TerminationAttachmentShape ;
                sh:qualifiedMinCount 1 ;
                sh:message "A transverse frame terminating at a deck or stringer must have a bracket or similar attachment construction."
            ]
        ]
    ) .

gen:TransverseFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node gen:ActiveTransverseFrameShape
        ]
    ) .

gen:FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node gen:MillimetreQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path [
            sh:alternativePath (
                nltl:hasFrameAttachment
                nltl:frameAttachment
            )
        ] ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node gen:PassThroughFrameShape
        ]
    ) ;
    sh:sparql [
        sh:message "Each declared supporting web frame or bulkhead must be paired with an effective frame attachment and the corresponding supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?support
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this ?supportProperty ?support .
                FILTER (
                    ?supportProperty IN (
                        nltl:hasSupportingWebFrame,
                        nltl:hasSupportingBulkhead
                    )
                )
                FILTER NOT EXISTS {
                    $this ?attachmentProperty ?attachment .
                    FILTER (
                        ?attachmentProperty IN (
                            nltl:hasFrameAttachment,
                            nltl:frameAttachment
                        )
                    )
                    ?attachment
                        nltl:hasAttachedSupportingStructure ?support ;
                        nltl:effectiveAttachmentConfirmed true .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every supporting structure represented on a frame attachment must have the reciprocal supporting-structure attachment and confirmed effective attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?attachment ?support
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this ?attachmentProperty ?attachment .
                FILTER (
                    ?attachmentProperty IN (
                        nltl:hasFrameAttachment,
                        nltl:frameAttachment
                    )
                )
                ?attachment nltl:hasAttachedSupportingStructure ?support .
                FILTER NOT EXISTS {
                    ?attachment nltl:effectiveAttachmentConfirmed true .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every supporting-structure attachment that points to this frame's attachment must identify the same supporting structure on that attachment and confirm effective attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?attachment ?support
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this ?attachmentProperty ?attachment .
                FILTER (
                    ?attachmentProperty IN (
                        nltl:hasFrameAttachment,
                        nltl:frameAttachment
                    )
                )
                ?support nltl:hasSupportingStructureAttachment ?attachment .
                FILTER NOT EXISTS {
                    ?attachment
                        nltl:hasAttachedSupportingStructure ?support ;
                        nltl:effectiveAttachmentConfirmed true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A longitudinal frame must have a bracketed attachment to each declared supporting web frame and bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?support
            WHERE {
                $this
                    a nltl:longitudinalFrame ;
                    nltl:withinIceStrengthenedArea true ;
                    ?supportProperty ?support .
                FILTER (
                    ?supportProperty IN (
                        nltl:hasSupportingWebFrame,
                        nltl:hasSupportingBulkhead
                    )
                )
                FILTER NOT EXISTS {
                    $this ?attachmentProperty ?attachment .
                    FILTER (
                        ?attachmentProperty IN (
                            nltl:hasFrameAttachment,
                            nltl:frameAttachment
                        )
                    )
                    ?attachment
                        nltl:hasAttachedSupportingStructure ?support ;
                        nltl:hasConnectionBracket ?bracket .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame with an installed bracket must provide its web-plate thickness in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this ?attachmentProperty ?attachment .
                FILTER (
                    ?attachmentProperty IN (
                        nltl:hasFrameAttachment,
                        nltl:frameAttachment
                    )
                )
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?frameThickness .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An installed bracket must be at least as thick as the frame web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this ?bracket
            WHERE {
                $this
                    nltl:withinIceStrengthenedArea true ;
                    nltl:frameWebThickness ?frameThickness ;
                    ?attachmentProperty ?attachment .
                FILTER (
                    ?attachmentProperty IN (
                        nltl:hasFrameAttachment,
                        nltl:frameAttachment
                    )
                )
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?frameThickness
                    qudt:numericValue ?frameValue ;
                    qudt:unit unit:MilliM .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?bracketThickness
                    qudt:numericValue ?bracketValue ;
                    qudt:unit unit:MilliM .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .
<END_SHACL>
