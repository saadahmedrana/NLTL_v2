<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node gen:ApplicableFrameShape
        ]
    ) .

gen:LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The longitudinal frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node gen:ApplicableFrameShape ;
            sh:node gen:ApplicableLongitudinalFrameShape
        ]
    ) .

        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;

        sh:message "The transverse frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [

            ] ;

        ]
    ) .

gen:ApplicableFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable frame must state whether it passes through a supporting structure."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2 ;
                sh:message "A frame passing through a supporting structure must have its web plate connected on exactly two sides."
            ]
        ]
    ) ;
    sh:property [
        sh:path [
            sh:alternativePath (
                nltl:hasFrameAttachment
                nltl:frameAttachment
            )
        ] ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead must be paired with a frame attachment." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this (nltl:hasFrameAttachment|nltl:frameAttachment) ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Each frame attachment and supporting structure pair must have the reciprocal supporting-structure attachment relationship." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this (nltl:hasFrameAttachment|nltl:frameAttachment) ?attachment .
    ?attachment nltl:hasAttachedSupportingStructure ?support .
    FILTER NOT EXISTS {
        ?support nltl:hasSupportingStructureAttachment ?attachment .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A frame with an installed bracket must have one decimal web thickness in millimetres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this (nltl:hasFrameAttachment|nltl:frameAttachment) ?attachment .
    ?attachment nltl:hasConnectionBracket ?bracket .
    FILTER (
        !EXISTS {
            $this nltl:frameWebThickness ?quantity .
            ?quantity qudt:numericValue ?value ;
                      qudt:unit unit:MilliM .
            FILTER(datatype(?value) = xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:frameWebThickness ?quantity1, ?quantity2 .
            FILTER(?quantity1 != ?quantity2)
        }
        ||
        EXISTS {
            $this nltl:frameWebThickness ?quantity .
            ?quantity qudt:numericValue ?value1, ?value2 .
            FILTER(?value1 != ?value2)
        }
        ||
        EXISTS {
            $this nltl:frameWebThickness ?quantity .
            ?quantity qudt:numericValue ?value .
            FILTER(datatype(?value) != xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:frameWebThickness ?quantity .
            ?quantity qudt:unit ?actualUnit .
            FILTER(?actualUnit != unit:MilliM)
        }
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:frameWebThickness ?frameQuantity .
    ?frameQuantity qudt:numericValue ?frameThickness ;
                   qudt:unit unit:MilliM .
    FILTER(datatype(?frameThickness) = xsd:decimal)

    $this (nltl:hasFrameAttachment|nltl:frameAttachment) ?attachment .
    ?attachment nltl:hasConnectionBracket ?bracket .
    ?bracket nltl:bracketThickness ?bracketQuantity .
    ?bracketQuantity qudt:numericValue ?bracketThickness ;
                     qudt:unit unit:MilliM .
    FILTER(datatype(?bracketThickness) = xsd:decimal)
    FILTER(?bracketThickness < ?frameThickness)
}
"""
    ] .

gen:ApplicableLongitudinalFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead of a longitudinal frame must be paired with an attachment having a connection bracket." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this (nltl:hasFrameAttachment|nltl:frameAttachment) ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:hasConnectionBracket ?bracket .
    }
}
"""
    ] .

gen:ApplicableTransverseFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable transverse frame must state whether it terminates at a deck or ice stringer."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path [
                    sh:alternativePath (
                        nltl:hasFrameAttachment
                        nltl:frameAttachment
                    )
                ] ;
                sh:qualifiedValueShape [
                    sh:or (
                        [
                            sh:property [
                                sh:path nltl:hasConnectionBracket ;
                                sh:minCount 1
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:similarAttachmentConstructionPresent ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue true
                            ]
                        ]
                    )
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar attachment construction."
            ]
        ]
    ) .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure ;
        sh:minCount 1 ;
        sh:message "Each frame attachment must identify its attached supporting structure."
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame attachment to its supporting structure must be confirmed effective."
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "A connection bracket must have one decimal thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The connection bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The connection bracket must be stiffened against buckling."
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .
<END_SHACL>
