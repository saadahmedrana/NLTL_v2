<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:node gen:ApplicableFrameShape
        ]
    ) .

gen:ApplicableFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:minCount 1 ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2 ;
                sh:message "A frame passing through a supporting structure must have its web plate connected on exactly two sides."
            ]
        ]
    ) ;
    sh:or (
        [
            sh:property [
                sh:path (
                    nltl:hasFrameAttachment
                    nltl:hasConnectionBracket
                ) ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameWebThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:MillimetreQuantityShape
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each listed supporting web frame or bulkhead must be paired with an attachment to that supporting structure." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?support
WHERE {
    $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each supporting web frame or bulkhead of a longitudinal frame must be paired with an attachment having a connection bracket." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?support
WHERE {
    $this a nltl:longitudinalFrame .
    $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support .
        ?attachment nltl:hasConnectionBracket ?bracket .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A transverse frame must state whether it terminates at a deck or ice stringer." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this a nltl:transverseFrame .
    FILTER NOT EXISTS {
        $this nltl:terminatesAtDeckOrIceStringer ?termination .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each attachment of a transverse frame terminating at a deck or ice stringer must have a bracket or confirmed similar construction." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?attachment
WHERE {
    $this a nltl:transverseFrame ;
          nltl:terminatesAtDeckOrIceStringer true ;
          nltl:hasFrameAttachment ?attachment .
    FILTER NOT EXISTS {
        ?attachment nltl:hasConnectionBracket ?bracket .
    }
    FILTER NOT EXISTS {
        ?attachment nltl:similarAttachmentConstructionPresent true .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?bracket
WHERE {
    $this nltl:frameWebThickness ?frameThicknessQuantity ;
          nltl:hasFrameAttachment ?attachment .
    ?frameThicknessQuantity qudt:numericValue ?frameThickness .
    ?attachment nltl:hasConnectionBracket ?bracket .
    ?bracket nltl:bracketThickness ?bracketThicknessQuantity .
    ?bracketThicknessQuantity qudt:numericValue ?bracketThickness .
    FILTER(
        xsd:decimal(?bracketThickness) <
        xsd:decimal(?frameThickness)
    )
}
"""
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The frame attachment must be confirmed as effective."
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each attached supporting structure must reciprocally identify the same frame attachment." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?support
WHERE {
    $this nltl:hasAttachedSupportingStructure ?support .
    FILTER NOT EXISTS {
        ?support nltl:hasSupportingStructureAttachment $this .
    }
}
"""
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened against buckling."
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .
<END_SHACL>
