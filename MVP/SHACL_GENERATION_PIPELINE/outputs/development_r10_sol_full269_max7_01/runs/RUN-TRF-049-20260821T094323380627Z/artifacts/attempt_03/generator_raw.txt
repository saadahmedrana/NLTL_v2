<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthMilliMShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:class nltl:frameAttachment
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:node gen:LengthMilliMShape ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue false ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:passesThroughSupportingStructure ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:webPlateConnectionSideCount ;
                        sh:datatype xsd:integer ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive 2 ;
                        sh:maxInclusive 2
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead in the ice-strengthened area must be paired with an effectively attached frame attachment and the reciprocal support-to-attachment link." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?support
WHERE {
    $this nltl:withinIceStrengthenedArea "true"^^xsd:boolean ;
          (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:effectiveAttachmentConfirmed "true"^^xsd:boolean .
        ?support nltl:hasSupportingStructureAttachment ?attachment .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead of a longitudinal frame in the ice-strengthened area must be paired with an attachment carrying a connection bracket." ;
        sh:select """
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?support
WHERE {
    $this rdf:type nltl:longitudinalFrame ;
          nltl:withinIceStrengthenedArea "true"^^xsd:boolean ;
          (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:hasConnectionBracket ?bracket .
        ?support nltl:hasSupportingStructureAttachment ?attachment .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A transverse frame in the ice-strengthened area must state whether it terminates at a deck or ice stringer." ;
        sh:select """
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this rdf:type nltl:transverseFrame ;
          nltl:withinIceStrengthenedArea "true"^^xsd:boolean .
    FILTER NOT EXISTS {
        $this nltl:terminatesAtDeckOrIceStringer ?termination .
        FILTER(datatype(?termination) = xsd:boolean)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Each relevant attachment of a terminating transverse frame must carry a bracket or record similar attachment construction." ;
        sh:select """
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?support
WHERE {
    $this rdf:type nltl:transverseFrame ;
          nltl:withinIceStrengthenedArea "true"^^xsd:boolean ;
          nltl:terminatesAtDeckOrIceStringer "true"^^xsd:boolean ;
          (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?support .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support .
        ?support nltl:hasSupportingStructureAttachment ?attachment .
        {
            ?attachment nltl:hasConnectionBracket ?bracket .
        }
        UNION
        {
            ?attachment nltl:similarAttachmentConstructionPresent "true"^^xsd:boolean .
        }
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A frame carrying a connection bracket must provide its web thickness in millimetres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?bracket
WHERE {
    $this nltl:hasFrameAttachment ?attachment .
    ?attachment nltl:hasConnectionBracket ?bracket .
    FILTER NOT EXISTS {
        $this nltl:frameWebThickness ?frameThickness .
        ?frameThickness qudt:numericValue ?frameValue ;
                        qudt:unit unit:MilliM .
        FILTER(datatype(?frameValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?bracket
WHERE {
    $this nltl:hasFrameAttachment ?attachment ;
          nltl:frameWebThickness ?frameThickness .
    ?attachment nltl:hasConnectionBracket ?bracket .
    ?frameThickness qudt:numericValue ?frameValue ;
                    qudt:unit unit:MilliM .
    ?bracket nltl:bracketThickness ?bracketThickness .
    ?bracketThickness qudt:numericValue ?bracketValue ;
                      qudt:unit unit:MilliM .
    FILTER(datatype(?frameValue) = xsd:decimal)
    FILTER(datatype(?bracketValue) = xsd:decimal)
    FILTER(?bracketValue < ?frameValue)
}
"""
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasFrameAttachment ;
    sh:class nltl:frameAttachment ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .

gen:SupportingStructureShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasAttachedSupportingStructure ;
    sh:class nltl:supportingStructure ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:class nltl:frameAttachment
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasConnectionBracket ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node gen:LengthMilliMShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
