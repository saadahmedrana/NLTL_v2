<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "The frame-to-shell attachment shall use a double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "The attachment shall state whether it crosses a shell plate butt."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:message "Scalloping is not allowed when the attachment does not cross a shell plate butt."
            ]
        ]
    ) .

gen:FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachmentRecord ;
        sh:node gen:FrameAttachmentRecordShape ;
        sh:message "Each frame shall have a conforming frame-to-shell attachment record."
    ] .
<END_SHACL>
