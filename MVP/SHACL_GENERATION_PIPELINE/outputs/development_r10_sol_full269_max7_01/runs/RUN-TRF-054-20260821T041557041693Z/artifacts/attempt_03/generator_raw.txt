gen:TRF054IceStringerShape
    a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:locatedOutsideIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must state whether it is outside the ice belt."
    ] ;
    sh:property [
        sh:path nltl:supportsIceStrengthenedFrames ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ice stringer must state whether it supports ice-strengthened frames."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:locatedOutsideIceBelt ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:supportsIceStrengthenedFrames ;
                sh:hasValue true
            ] ;

                sh:path nltl:stringerSpan ;
                sh:node gen:LengthMValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToAdjacentIceStringer ;
                sh:node gen:LengthMValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:distanceToIceBelt ;
                sh:node gen:LengthMValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node gen:UnitlessValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:node gen:PressureMPaValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:outsideIceBeltDistributionFactorF9 ;
                sh:node gen:FactorF9ValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:outsideIceBeltSafetyFactorF10 ;

            sh:property [
                sh:path nltl:outsideIceBeltShearFactorF11 ;
                sh:node gen:FactorF11ValueShape ;

                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:outsideIceBeltFactor ;
                sh:node gen:UnitlessValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:sectionModulus ;
                sh:node gen:SectionModulusValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:requiredShearArea ;
                sh:node gen:ShearAreaValueShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "For an applicable ice stringer, design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy clause 4.5.2." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:locatedOutsideIceBelt true ;
                              nltl:supportsIceStrengthenedFrames true ;
                              nltl:icePressure/qudt:numericValue ?pressureRaw ;
                              nltl:designIceLoadHeight/qudt:numericValue ?heightRaw ;
                              nltl:designLineLoad/qudt:numericValue ?reportedLineLoadRaw ;
                              nltl:stringerSpan/qudt:numericValue ?spanRaw ;
                              nltl:distanceToAdjacentIceStringer/qudt:numericValue ?adjacentRaw ;
                              nltl:distanceToIceBelt/qudt:numericValue ?distanceRaw ;
                              nltl:frameMomentFactorM/qudt:numericValue ?momentRaw ;
                              nltl:yieldStrength/qudt:numericValue ?yieldRaw ;
                              nltl:outsideIceBeltDistributionFactorF9/qudt:numericValue ?f9Raw ;
                              nltl:outsideIceBeltSafetyFactorF10/qudt:numericValue ?f10Raw ;
                              nltl:outsideIceBeltShearFactorF11/qudt:numericValue ?f11Raw ;
                              nltl:outsideIceBeltFactor/qudt:numericValue ?reportedOutsideRaw ;
                              nltl:sectionModulus/qudt:numericValue ?reportedZRaw ;
                              nltl:requiredShearArea/qudt:numericValue ?reportedAreaRaw .

                        BIND(xsd:decimal(?pressureRaw) AS ?pressure)
                        BIND(xsd:decimal(?heightRaw) AS ?height)
                        BIND(xsd:decimal(?reportedLineLoadRaw) AS ?reportedLineLoad)
                        BIND(xsd:decimal(?spanRaw) AS ?span)
                        BIND(xsd:decimal(?adjacentRaw) AS ?adjacent)
                        BIND(xsd:decimal(?distanceRaw) AS ?distance)
                        BIND(xsd:decimal(?momentRaw) AS ?moment)
                        BIND(xsd:decimal(?yieldRaw) AS ?yieldStrength)
                        BIND(xsd:decimal(?f9Raw) AS ?f9)
                        BIND(xsd:decimal(?f10Raw) AS ?f10)
                        BIND(xsd:decimal(?f11Raw) AS ?f11)
                        BIND(xsd:decimal(?reportedOutsideRaw) AS ?reportedOutside)
                        BIND(xsd:decimal(?reportedZRaw) AS ?reportedZ)
                        BIND(xsd:decimal(?reportedAreaRaw) AS ?reportedArea)

                        BIND(?pressure * ?height AS ?calculatedLineLoadMNPerM)
                        BIND(IF(?calculatedLineLoadMNPerM < 0.15,
                                0.15,
                                ?calculatedLineLoadMNPerM) AS ?expectedLineLoadMNPerM)
                        BIND(?expectedLineLoadMNPerM * 1000 AS ?expectedLineLoadKNPerM)
                        BIND(IF(ABS(?expectedLineLoadKNPerM) * 0.000001 > 0.000000001,
                                ABS(?expectedLineLoadKNPerM) * 0.000001,
                                0.000000001) AS ?lineLoadTolerance)

                        BIND(IF(?adjacent = 0,
                                0,
                                1 - (?distance / ?adjacent)) AS ?expectedOutside)
                        BIND(IF(ABS(?expectedOutside) * 0.000001 > 0.000000001,
                                ABS(?expectedOutside) * 0.000001,
                                0.000000001) AS ?outsideTolerance)

                        BIND(?reportedLineLoad / 1000 AS ?reportedLineLoadMNPerM)
                        BIND(IF(?moment = 0 || ?yieldStrength = 0,
                                0,
                                ?f9 * ?f10 * ?reportedLineLoadMNPerM *
                                ?span * ?span * ?reportedOutside * 1000000 /
                                (?moment * ?yieldStrength)) AS ?expectedZ)
                        BIND(IF(ABS(?expectedZ) * 0.000001 > 0.000000001,
                                ABS(?expectedZ) * 0.000001,
                                0.000000001) AS ?zTolerance)

                        BIND(IF(?yieldStrength = 0,
                                0,
                                1.7320508075688772 * ?f9 * ?f10 * ?f11 *
                                ?reportedLineLoadMNPerM * ?span *
                                ?reportedOutside * 10000 /
                                (2 * ?yieldStrength)) AS ?expectedArea)
                        BIND(IF(ABS(?expectedArea) * 0.000001 > 0.000000001,
                                ABS(?expectedArea) * 0.000001,
                                0.000000001) AS ?areaTolerance)

                        FILTER(
                            ?adjacent = 0 ||
                            ?moment = 0 ||
                            ?yieldStrength = 0 ||
                            ABS(?reportedLineLoad - ?expectedLineLoadKNPerM) > ?lineLoadTolerance ||
                            ABS(?reportedOutside - ?expectedOutside) > ?outsideTolerance ||
                            ABS(?reportedZ - ?expectedZ) > ?zTolerance ||
                            ABS(?reportedArea - ?expectedArea) > ?areaTolerance
                        )
                    }
                """
            ]
        ]
    ) .
