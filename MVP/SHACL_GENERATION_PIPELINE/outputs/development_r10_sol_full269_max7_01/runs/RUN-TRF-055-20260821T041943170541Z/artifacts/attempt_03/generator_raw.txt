    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:requiredSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SectionModulusQuantityShape
    ] ;
    sh:property [
        sh:path nltl:actualShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ShearAreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:or (
        [
            sh:property [

            sh:property [
                sh:path nltl:servesAsIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;

            sh:property [
                sh:path nltl:designLineLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:StandardLineLoadQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:abreastOfHatch ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [

            sh:property [

    sh:sparql [
        sh:message "Actual section modulus and actual shear area must each be at least their respective required values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:actualSectionModulus ?actualSectionNode ;
                      nltl:requiredSectionModulus ?requiredSectionNode ;
                      nltl:actualShearArea ?actualShearNode ;
                      nltl:requiredShearArea ?requiredShearNode .

                ?actualShearNode qudt:numericValue ?actualShear .
                ?requiredShearNode qudt:numericValue ?requiredShear .

                    xsd:decimal(?actualSection) < xsd:decimal(?requiredSection) ||
                    xsd:decimal(?actualShear) < xsd:decimal(?requiredShear)
                )
            }
        """
