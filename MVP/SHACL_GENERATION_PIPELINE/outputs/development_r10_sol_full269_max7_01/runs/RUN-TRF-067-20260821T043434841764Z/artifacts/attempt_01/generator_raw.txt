<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF067Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "An ice class is required to determine applicability of the rudder load protection requirement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IA or IA Super shall have a linked rudder load absorbing arrangement." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
    FILTER NOT EXISTS {
        $this nltl:hasRudderProtectionArrangement ?arrangement .
        ?arrangement a nltl:rudderLoadAbsorbingArrangement .
        FILTER (isIRI(?arrangement))
    }
}
"""
    ] .
<END_SHACL>
