<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF069Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice-class selector is required."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller-type selector is required."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A pitch-type selector is required."
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Maximum ice torque is required as a decimal quantity in kilonewton-metres."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:node [
            sh:property [
                sh:path nltl:normalOperationalConditions ;
                sh:datatype xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
                sh:datatype xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
                sh:datatype xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
                sh:datatype xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:radialIceEntryOutsideLoadModel ;
                sh:datatype xsd:boolean
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship must have a conforming design condition covering normal service-life ice loads and the prescribed load-model exclusions; fixed-pitch ships must include reversal loads on that same condition." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:propellerType ?propellerType ;
          nltl:pitchType ?pitchType .

    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        )
    )
    FILTER (?propellerType IN ("open", "ducted"))
    FILTER (?pitchType IN ("controllable", "fixed"))

    FILTER NOT EXISTS {
        $this nltl:hasDesignCondition ?condition .
        ?condition a nltl:designCondition ;
            nltl:normalOperationalConditions ?normalOperations ;
            nltl:normalServiceLifePropellerIceLoadCondition ?serviceLifeLoads ;
            nltl:stoppedPropellerDraggingOutsideLoadModel ?stoppedDraggingOutside ;
            nltl:radialIceEntryOutsideLoadModel ?radialEntryOutside .

        FILTER (isIRI(?condition))
        FILTER (
            datatype(?normalOperations) = xsd:boolean &&
            datatype(?serviceLifeLoads) = xsd:boolean &&
            datatype(?stoppedDraggingOutside) = xsd:boolean &&
            datatype(?radialEntryOutside) = xsd:boolean &&
            ?normalOperations = true &&
            ?serviceLifeLoads = true &&
            ?stoppedDraggingOutside = true &&
            ?radialEntryOutside = true
        )
        FILTER (
            ?pitchType != "fixed" ||
            EXISTS {
                ?condition nltl:fixedPitchPropellerReversalLoadIncluded ?reversalLoads .
                FILTER (
                    datatype(?reversalLoads) = xsd:boolean &&
                    ?reversalLoads = true
                )
            }
        )
    }
}
"""
    ] .
<END_SHACL>
