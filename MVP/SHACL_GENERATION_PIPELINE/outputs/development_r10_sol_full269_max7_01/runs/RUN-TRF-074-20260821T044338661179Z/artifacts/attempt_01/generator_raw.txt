<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF074Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Ice class is required to determine whether the propeller-submersion rule applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, ballast-condition propeller-top depth must be supplied as a decimal quantity in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (
        NOT EXISTS {
            $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
        }
        ||
        EXISTS {
            $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
            FILTER (
                NOT EXISTS {
                    ?depthQuantity qudt:numericValue ?depthValue .
                    FILTER (datatype(?depthValue) = xsd:decimal)
                }
                ||
                EXISTS {
                    ?depthQuantity qudt:numericValue ?invalidDepthValue .
                    FILTER (datatype(?invalidDepthValue) != xsd:decimal)
                }
                ||
                NOT EXISTS {
                    ?depthQuantity qudt:unit unit:M .
                }
                ||
                EXISTS {
                    ?depthQuantity qudt:unit ?otherDepthUnit .
                    FILTER (?otherDepthUnit != unit:M)
                }
            )
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, required propeller submersion depth hi must be supplied as a decimal quantity in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (
        NOT EXISTS {
            $this nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .
        }
        ||
        EXISTS {
            $this nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .
            FILTER (
                NOT EXISTS {
                    ?requiredQuantity qudt:numericValue ?requiredValue .
                    FILTER (datatype(?requiredValue) = xsd:decimal)
                }
                ||
                EXISTS {
                    ?requiredQuantity qudt:numericValue ?invalidRequiredValue .
                    FILTER (datatype(?invalidRequiredValue) != xsd:decimal)
                }
                ||
                NOT EXISTS {
                    ?requiredQuantity qudt:unit unit:M .
                }
                ||
                EXISTS {
                    ?requiredQuantity qudt:unit ?otherRequiredUnit .
                    FILTER (?otherRequiredUnit != unit:M)
                }
            )
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, a propeller-top depth below hi requires the propulsion system to be designed for ice class IA." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
          nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .

    ?depthQuantity qudt:numericValue ?depthValue ;
                   qudt:unit unit:M .

    ?requiredQuantity qudt:numericValue ?requiredValue ;
                      qudt:unit unit:M .

    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (datatype(?depthValue) = xsd:decimal)
    FILTER (datatype(?requiredValue) = xsd:decimal)
    FILTER (?depthValue < ?requiredValue)
    FILTER NOT EXISTS {
        $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
    }
}
"""
    ] .
<END_SHACL>
