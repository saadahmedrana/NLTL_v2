<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF074Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IB or IC must provide a valid ballast-condition propeller-top depth in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (
        NOT EXISTS {
            $this nltl:ballastConditionPropellerTopDepth ?anyDepth .
        }
        ||
        EXISTS {
            $this nltl:ballastConditionPropellerTopDepth ?depthQuantity .
            FILTER NOT EXISTS {
                ?depthQuantity qudt:numericValue ?depth ;
                               qudt:unit unit:M .
                FILTER (DATATYPE(?depth) = xsd:decimal)
                FILTER NOT EXISTS {
                    ?depthQuantity qudt:unit ?otherUnit .
                    FILTER (?otherUnit != unit:M)
                }
                FILTER NOT EXISTS {
                    ?depthQuantity qudt:numericValue ?otherDepth .
                    FILTER (DATATYPE(?otherDepth) != xsd:decimal)
                }
            }
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships of ice class IB or IC must provide a valid required propeller submersion depth hi in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (
        NOT EXISTS {
            $this nltl:requiredPropellerSubmersionDepthHi ?anyHi .
        }
        ||
        EXISTS {
            $this nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
            FILTER NOT EXISTS {
                ?hiQuantity qudt:numericValue ?hi ;
                            qudt:unit unit:M .
                FILTER (DATATYPE(?hi) = xsd:decimal)
                FILTER NOT EXISTS {
                    ?hiQuantity qudt:unit ?otherUnit .
                    FILTER (?otherUnit != unit:M)
                }
                FILTER NOT EXISTS {
                    ?hiQuantity qudt:numericValue ?otherHi .
                    FILTER (DATATYPE(?otherHi) != xsd:decimal)
                }
            }
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IB or IC, when the ballast-condition propeller-top depth is less than hi, the propulsion system must be designed according to ice class IA." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
          nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
    ?depthQuantity qudt:numericValue ?depth ;
                   qudt:unit unit:M .
    ?hiQuantity qudt:numericValue ?hi ;
                qudt:unit unit:M .
    FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
    FILTER (DATATYPE(?depth) = xsd:decimal)
    FILTER (DATATYPE(?hi) = xsd:decimal)
    FILTER (?depth < ?hi)
    FILTER NOT EXISTS {
        $this nltl:propulsionSystemDesignedForIceClass nltl:iceClassIa .
    }
}
"""
    ] .
<END_SHACL>
