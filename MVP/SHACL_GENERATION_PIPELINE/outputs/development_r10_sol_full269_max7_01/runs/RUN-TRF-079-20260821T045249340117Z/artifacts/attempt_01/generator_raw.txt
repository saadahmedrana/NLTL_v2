    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerShaftCentrelineDepth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerImmersionFunction ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The immersion function and k2 must satisfy equations 6.16 and 6.15 within relative tolerance 1e-6 and absolute fallback 1e-9; propeller diameter must be non-zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceThickness/qudt:numericValue ?hIce ;
                      nltl:propellerShaftCentrelineDepth/qudt:numericValue ?h0 ;
                      nltl:propellerDiameter/qudt:numericValue ?diameter ;
                      nltl:propellerImmersionFunction/qudt:numericValue ?reportedF ;
                      nltl:iceLoadCycleCoefficientK2/qudt:numericValue ?reportedK2 .

                BIND(
                    IF(
                        ?diameter = 0,
                        0.0,
                        ((?h0 - ?hIce) / (?diameter / 2.0)) - 1.0
                    )
                    AS ?expectedF
                )
                BIND(
                    IF(
                        ABS(?expectedF) * 0.000001 > 0.000000001,
                        ABS(?expectedF) * 0.000001,
                        0.000000001
                    )
                    AS ?fTolerance
                )
                BIND(
                    IF(
                        ?expectedF < 0,
                        0.8 - ?expectedF,
                        IF(
                            ?expectedF <= 1,
                            0.8 - (0.4 * ?expectedF),
                            IF(
                                ?expectedF <= 2.5,
                                0.6 - (0.2 * ?expectedF),
                                0.1
                            )
                        )
                    )
                    AS ?expectedK2
                )
                BIND(
                    IF(
                        ABS(?expectedK2) * 0.000001 > 0.000000001,
                        ABS(?expectedK2) * 0.000001,
                        0.000000001
                    )
                    AS ?k2Tolerance
                )
                FILTER(
                    ?diameter = 0 ||
                    ABS(?reportedF - ?expectedF) > ?fTolerance ||
                    ABS(?reportedK2 - ?expectedK2) > ?k2Tolerance
                )
            }
        """
    ] .
