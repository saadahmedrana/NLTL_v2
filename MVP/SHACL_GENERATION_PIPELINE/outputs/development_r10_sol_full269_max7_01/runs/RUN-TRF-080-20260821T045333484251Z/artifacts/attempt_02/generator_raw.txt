<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PropellerIceInteractionLoadCycleShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:hasComponent ?component .
                ?component a nltl:propellerIceInteractionComponent .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must provide one integer ice load cycle count."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must provide one integer propeller blade count."
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "An applicable ship must provide one unitless effective ice load cycle count."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The effective ice load cycle count must equal the ice load cycle count multiplied by the propeller blade count, within relative tolerance 1e-6 and absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?result
            WHERE {
                $this nltl:iceLoadCycleCount ?cycleCount ;
                      nltl:propellerBladeCount ?bladeCount ;
                      nltl:effectiveIceLoadCycleCount ?result .
                ?result qudt:numericValue ?reportedValue ;
                        qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                BIND(
                    xsd:decimal(?cycleCount) * xsd:decimal(?bladeCount)
                    AS ?expectedValue
                )
                BIND(
                    IF(
                        ABS(?expectedValue) * 0.000001 > 0.000000000001,
                        ABS(?expectedValue) * 0.000001,
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER(
                    ABS(xsd:decimal(?reportedValue) - ?expectedValue) > ?tolerance
                )
            }
        """
    ] .
<END_SHACL>
