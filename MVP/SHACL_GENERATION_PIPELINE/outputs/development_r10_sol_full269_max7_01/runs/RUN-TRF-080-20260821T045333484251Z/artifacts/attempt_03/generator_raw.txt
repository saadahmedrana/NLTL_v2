<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF080Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship must provide the ice load cycle count, propeller blade count, and effective ice load cycle count." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?component .
    ?component a nltl:propellerIceInteractionComponent .
    FILTER (
        NOT EXISTS {
            $this nltl:iceLoadCycleCount ?iceLoadCycleCount .
        }
        ||
        NOT EXISTS {
            $this nltl:propellerBladeCount ?propellerBladeCount .
        }
        ||
        NOT EXISTS {
            $this nltl:effectiveIceLoadCycleCount ?effectiveIceLoadCycleCount .
        }
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ice load cycle count and propeller blade count must be xsd:integer values." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?component .
    ?component a nltl:propellerIceInteractionComponent .
    $this ?property ?value .
    FILTER (
        ?property IN (
            nltl:iceLoadCycleCount,
            nltl:propellerBladeCount
        )
    )
    FILTER (datatype(?value) != xsd:integer)
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each effective ice load cycle count must have a decimal numeric value and the unit UNITLESS." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?component .
    ?component a nltl:propellerIceInteractionComponent .
    $this nltl:effectiveIceLoadCycleCount ?result .
    FILTER (
        NOT EXISTS {
            ?result qudt:numericValue ?numericValue .
            FILTER (datatype(?numericValue) = xsd:decimal)
        }
        ||
        EXISTS {
            ?result qudt:numericValue ?invalidNumericValue .
            FILTER (datatype(?invalidNumericValue) != xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?result qudt:unit unit:UNITLESS .
        }
        ||
        EXISTS {
            ?result qudt:unit ?invalidUnit .
            FILTER (?invalidUnit != unit:UNITLESS)
        }
    )
}
"""
    ] .
<END_SHACL>
