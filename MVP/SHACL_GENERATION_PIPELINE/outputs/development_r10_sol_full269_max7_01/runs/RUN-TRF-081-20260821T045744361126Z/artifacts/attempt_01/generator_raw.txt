<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ForceQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF081DesignThrustShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:forwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:backwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:forwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:backwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:propellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:sparql [
        sh:message "Ice thrust and propeller shaft design thrust values must satisfy the prescribed formulas within relative tolerance 1e-6 or absolute tolerance 1e-9 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce/qudt:numericValue ?forwardBlade ;
                      nltl:backwardBladeForce/qudt:numericValue ?backwardBlade ;
                      nltl:hydrodynamicBollardThrust/qudt:numericValue ?bollard ;
                      nltl:forwardIceThrust/qudt:numericValue ?forwardIce ;
                      nltl:backwardIceThrust/qudt:numericValue ?backwardIce ;
                      nltl:forwardPropellerShaftDesignThrust/qudt:numericValue ?forwardDesign ;
                      nltl:backwardPropellerShaftDesignThrust/qudt:numericValue ?backwardDesign ;
                      nltl:propellerShaftDesignThrust/qudt:numericValue ?design .

                BIND ("1.1"^^xsd:decimal * ?forwardBlade AS ?expectedForwardIce)
                BIND ("1.1"^^xsd:decimal * ?backwardBlade AS ?expectedBackwardIce)
                BIND (?bollard + "2.2"^^xsd:decimal * ?forwardIce AS ?expectedForwardDesign)
                BIND ("1.5"^^xsd:decimal * ?backwardIce AS ?expectedBackwardDesign)
                BIND (
                    IF(
                        ?forwardDesign >= ?backwardDesign,
                        ?forwardDesign,
                        ?backwardDesign
                    )
                    AS ?expectedDesign
                )

                FILTER (
                    ABS(?forwardIce - ?expectedForwardIce) >
                        IF(
                            ABS(?expectedForwardIce) * "0.000001"^^xsd:decimal >
                                "0.000000001"^^xsd:decimal,
                            ABS(?expectedForwardIce) * "0.000001"^^xsd:decimal,
                            "0.000000001"^^xsd:decimal
                        )
                    ||
                    ABS(?backwardIce - ?expectedBackwardIce) >
                        IF(
                            ABS(?expectedBackwardIce) * "0.000001"^^xsd:decimal >
                                "0.000000001"^^xsd:decimal,
                            ABS(?expectedBackwardIce) * "0.000001"^^xsd:decimal,
                            "0.000000001"^^xsd:decimal
                        )
                    ||
                    ABS(?forwardDesign - ?expectedForwardDesign) >
                        IF(
                            ABS(?expectedForwardDesign) * "0.000001"^^xsd:decimal >
                                "0.000000001"^^xsd:decimal,
                            ABS(?expectedForwardDesign) * "0.000001"^^xsd:decimal,
                            "0.000000001"^^xsd:decimal
                        )
                    ||
                    ABS(?backwardDesign - ?expectedBackwardDesign) >
                        IF(
                            ABS(?expectedBackwardDesign) * "0.000001"^^xsd:decimal >
                                "0.000000001"^^xsd:decimal,
                            ABS(?expectedBackwardDesign) * "0.000001"^^xsd:decimal,
                            "0.000000001"^^xsd:decimal
                        )
                    ||
                    ABS(?design - ?expectedDesign) >
                        IF(
                            ABS(?expectedDesign) * "0.000001"^^xsd:decimal >
                                "0.000000001"^^xsd:decimal,
                            ABS(?expectedDesign) * "0.000001"^^xsd:decimal,
                            "0.000000001"^^xsd:decimal
                        )
                )
            }
        """
    ] .
<END_SHACL>
