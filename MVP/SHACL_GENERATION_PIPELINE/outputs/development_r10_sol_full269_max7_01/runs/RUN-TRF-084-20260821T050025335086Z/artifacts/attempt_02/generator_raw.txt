<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PitchQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PropellerPitchAtMcrBollardShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:node gen:PitchQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrFreeRunning ;
        sh:node gen:PitchQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A CP propeller requires a boolean indication of whether the bollard-condition pitch is known." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER(?propellerType = "CP"^^xsd:string)
                FILTER NOT EXISTS {
                    $this nltl:pitchValueKnown ?known .
                    FILTER(DATATYPE(?known) = xsd:boolean)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A known CP-propeller pitch at MCR in bollard condition must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType ;
                      nltl:pitchValueKnown ?known .
                FILTER(?propellerType = "CP"^^xsd:string)
                FILTER(?known = "true"^^xsd:boolean)
                FILTER NOT EXISTS {
                    $this nltl:propellerPitchAtMcrBollard ?bollardPitch .
                }
            }
        """
    ] .
<END_SHACL>
