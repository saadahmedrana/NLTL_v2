<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF084Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A CP propeller requires a boolean indication of whether its bollard-condition pitch is known." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propellerType "CP"^^xsd:string .
    FILTER NOT EXISTS {
        $this nltl:pitchValueKnown ?known .
        FILTER(DATATYPE(?known) = xsd:boolean)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the bollard-condition pitch is known for a CP propeller, it must be recorded as a length in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:propellerType "CP"^^xsd:string ;
          nltl:pitchValueKnown true .
    FILTER NOT EXISTS {
        $this nltl:propellerPitchAtMcrBollard ?bollardPitch .
        ?bollardPitch a qudt:QuantityValue ;
                      qudt:numericValue ?bollardValue ;
                      qudt:unit unit:M .
        FILTER(DATATYPE(?bollardValue) = xsd:decimal)
    }
}
"""
    ] .
<END_SHACL>
