@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:CpPropellerPitchSelectorShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT ?this
WHERE {
    ?this a nltl:ship ;
          nltl:propellerType ?propellerType .
    FILTER(?propellerType = "CP"^^xsd:string)
}
"""
    ] ;
    sh:property [
        sh:path nltl:pitchValueKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A CP propeller requires one boolean indication of whether its bollard-condition pitch is known."
    ] .

gen:CpPropellerKnownBollardPitchShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT ?this
WHERE {
    ?this a nltl:ship ;
          nltl:propellerType ?propellerType ;
          nltl:pitchValueKnown ?pitchKnown .
    FILTER(?propellerType = "CP"^^xsd:string)
    FILTER(?pitchKnown = "true"^^xsd:boolean)
}
"""
    ] ;
    sh:property [
        sh:path nltl:propellerPitchAtMcrBollard ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A known CP propeller pitch at MCR bollard condition must be recorded in metres."
    ] .
<END_SHACL>
