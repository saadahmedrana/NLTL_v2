<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF084Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A CP propeller requires an explicit boolean pitchValueKnown value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:propellerType ?propellerType .
                    FILTER(
                        DATATYPE(?propellerType) = xsd:string &&
                        STR(?propellerType) = "CP"
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:pitchValueKnown ?known .
                    FILTER(
                        DATATYPE(?known) = xsd:boolean &&
                        (?known = true || ?known = false)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A CP propeller requires a valid bollard-condition pitch quantity in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:propellerType ?propellerType .
                    FILTER(
                        DATATYPE(?propellerType) = xsd:string &&
                        STR(?propellerType) = "CP"
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:propellerPitchAtMcrBollard ?bollardPitch .
                    ?bollardPitch a qudt:QuantityValue ;
                        qudt:numericValue ?bollardValue ;
                        qudt:unit unit:M .
                    FILTER(DATATYPE(?bollardValue) = xsd:decimal)
                    FILTER NOT EXISTS {
                        ?bollardPitch qudt:unit ?otherBollardUnit .
                        FILTER(?otherBollardUnit != unit:M)
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the CP bollard pitch is not known, a valid free-running pitch quantity in metres is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType ;
                    nltl:pitchValueKnown ?known .
                FILTER(
                    DATATYPE(?propellerType) = xsd:string &&
                    STR(?propellerType) = "CP" &&
                    DATATYPE(?known) = xsd:boolean &&
                    ?known = false
                )
                FILTER NOT EXISTS {
                    $this nltl:propellerPitchAtMcrFreeRunning ?freeRunningPitch .
                    ?freeRunningPitch a qudt:QuantityValue ;
                        qudt:numericValue ?freeRunningValue ;
                        qudt:unit unit:M .
                    FILTER(DATATYPE(?freeRunningValue) = xsd:decimal)
                    FILTER NOT EXISTS {
                        ?freeRunningPitch qudt:unit ?otherFreeRunningUnit .
                        FILTER(?otherFreeRunningUnit != unit:M)
                    }
                }
            }
        """
    ] .
<END_SHACL>
