<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF086Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For a CP propeller, pitchValueKnown and the required well-formed pitch values must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType "CP"^^xsd:string .
                FILTER NOT EXISTS {
                    $this nltl:pitchValueKnown ?known .
                    FILTER(DATATYPE(?known) = xsd:boolean)
                    FILTER(
                        (
                            ?known = "true"^^xsd:boolean
                            &&
                            EXISTS {
                                $this nltl:propellerPitchAtMcrBollard ?bollardPitch .
                                ?bollardPitch a qudt:QuantityValue ;
                                    qudt:numericValue ?bollardValue ;
                                    qudt:unit unit:M .
                                FILTER(DATATYPE(?bollardValue) = xsd:decimal)
                                FILTER NOT EXISTS {
                                    ?bollardPitch qudt:unit ?otherBollardUnit .
                                    FILTER(?otherBollardUnit != unit:M)
                                }
                            }
                        )
                        ||
                        (
                            ?known = "false"^^xsd:boolean
                            &&
                            EXISTS {
                                $this nltl:propellerPitchAtMcrBollard ?fallbackPitch .
                                ?fallbackPitch a qudt:QuantityValue ;
                                    qudt:numericValue ?fallbackValue ;
                                    qudt:unit unit:M .
                                FILTER(DATATYPE(?fallbackValue) = xsd:decimal)
                                FILTER NOT EXISTS {
                                    ?fallbackPitch qudt:unit ?otherFallbackUnit .
                                    FILTER(?otherFallbackUnit != unit:M)
                                }
                            }
                            &&
                            EXISTS {
                                $this nltl:propellerPitchAtMcrFreeRunning ?freeRunningPitch .
                                ?freeRunningPitch a qudt:QuantityValue ;
                                    qudt:numericValue ?freeRunningValue ;
                                    qudt:unit unit:M .
                                FILTER(DATATYPE(?freeRunningValue) = xsd:decimal)
                                FILTER NOT EXISTS {
                                    ?freeRunningPitch qudt:unit ?otherFreeRunningUnit .
                                    FILTER(?otherFreeRunningUnit != unit:M)
                                }
                            }
                        )
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a CP propeller, every pitchValueKnown value must be an xsd:boolean." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?value
            WHERE {
                $this nltl:propellerType "CP"^^xsd:string ;
                    nltl:pitchValueKnown ?value .
                FILTER(!ISLITERAL(?value) || DATATYPE(?value) != xsd:boolean)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A CP propeller must not have contradictory pitchValueKnown values." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerType "CP"^^xsd:string ;
                    nltl:pitchValueKnown "true"^^xsd:boolean ;
                    nltl:pitchValueKnown "false"^^xsd:boolean .
            }
        """
    ] .
<END_SHACL>
