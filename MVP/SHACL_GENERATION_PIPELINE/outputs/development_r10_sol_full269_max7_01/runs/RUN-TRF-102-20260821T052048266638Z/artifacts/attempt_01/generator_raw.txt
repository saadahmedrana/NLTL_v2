<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SpindleTorqueShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:ApprovedStressAnalysisBranch
        gen:FormulaCalculationBranch
    ) ;
    sh:message "Maximum spindle torque must be determined by an approved stress analysis or by the prescribed blade-failure torque calculation." .

gen:ApprovedStressAnalysisBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:approvedSpindleTorqueStressAnalysis ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:message "An approved spindle torque stress analysis is required for the stress-analysis alternative."
    ] ;
    sh:property [
        sh:path nltl:maximumSpindleTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TorqueQuantityValueShape ;
        sh:message "The stress-analysis alternative must provide the determined maximum spindle torque."
    ] .

gen:FormulaCalculationBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeExpandedAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:leadingEdgeChordPortionAtZeroPointEightRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:trailingEdgeChordPortionAtZeroPointEightRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureSpindleTorqueFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureSpindleTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TorqueQuantityValueShape
    ] ;
    sh:property [
        sh:path nltl:maximumSpindleTorque ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TorqueQuantityValueShape
    ] ;
    sh:sparql [
        sh:message "The blade count must be nonzero, and the reported spindle torque factor and torque results must match the prescribed formulas within a relative tolerance of 1e-6 with an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeExpandedAreaRatio/qudt:numericValue ?ear ;
                      nltl:propellerBladeCount ?z ;
                      nltl:leadingEdgeChordPortionAtZeroPointEightRadius/qudt:numericValue ?leading ;
                      nltl:trailingEdgeChordPortionAtZeroPointEightRadius/qudt:numericValue ?trailing ;
                      nltl:extremeIceForce/qudt:numericValue ?force ;
                      nltl:bladeFailureSpindleTorqueFactor/qudt:numericValue ?reportedFactor ;
                      nltl:bladeFailureSpindleTorque/qudt:numericValue ?reportedBladeTorque ;
                      nltl:maximumSpindleTorque/qudt:numericValue ?reportedMaximumTorque .

                BIND(IF(?z = 0, 1, ?z) AS ?safeBladeCount)
                BIND((4 * ?ear / ?safeBladeCount) AS ?ratio)
                BIND((0.7 * (1 - (?ratio * ?ratio * ?ratio))) AS ?rawFactor)
                BIND(IF(?rawFactor < 0.3, 0.3, ?rawFactor) AS ?expectedFactor)
                BIND(IF(?leading >= (0.8 * ?trailing), ?leading, (0.8 * ?trailing)) AS ?effectiveChord)
                BIND((?effectiveChord * ?expectedFactor * ?force) AS ?expectedTorque)

                BIND(
                    IF(
                        ABS(?expectedFactor) * 0.000001 > 0.000000001,
                        ABS(?expectedFactor) * 0.000001,
                        0.000000001
                    ) AS ?factorTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedTorque) * 0.000001 > 0.000000001,
                        ABS(?expectedTorque) * 0.000001,
                        0.000000001
                    ) AS ?torqueTolerance
                )

                FILTER (
                    ?z = 0 ||
                    ABS(?reportedFactor - ?expectedFactor) > ?factorTolerance ||
                    ABS(?reportedBladeTorque - ?expectedTorque) > ?torqueTolerance ||
                    ABS(?reportedMaximumTorque - ?expectedTorque) > ?torqueTolerance
                )
            }
        """
    ] .

gen:UnitlessQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ForceQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TorqueQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN-M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
