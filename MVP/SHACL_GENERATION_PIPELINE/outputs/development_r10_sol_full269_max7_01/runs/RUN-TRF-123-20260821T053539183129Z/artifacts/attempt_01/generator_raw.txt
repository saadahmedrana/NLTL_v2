<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipOccasionalForceReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:minCount 1 ;
        sh:node gen:OccasionalForceLoadCaseShape ;
        sh:message "The ship must provide an occasional-force load case."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Component stress multiplied by the reported yield safety factor must not exceed the component material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?component
            WHERE {
                $this nltl:hasOccasionalForceLoadCase ?loadCase .
                ?loadCase
                    nltl:occasionalForceCaseAssessedComponent ?component ;
                    nltl:componentYieldSafetyFactor ?factorQuantity .
                ?component
                    nltl:occasionalForceComponentStress ?stressQuantity ;
                    nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties
                    nltl:componentMaterialYieldStrength ?yieldQuantity .
                ?factorQuantity qudt:numericValue ?factor .
                ?stressQuantity qudt:numericValue ?stress .
                ?yieldQuantity qudt:numericValue ?yieldStrength .
                FILTER(
                    (xsd:decimal(?stress) * xsd:decimal(?factor))
                    > xsd:decimal(?yieldStrength)
                )
            }
        """
    ] .

gen:OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:occasionalForceLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The load case must confirm that the propeller blade is excluded from the occasional-force scope."
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:UnitlessDecimalQuantityShape ;
        sh:message "The load case must provide the externally determined component yield safety factor."
    ] ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:class nltl:shipComponent ;
        sh:minCount 1 ;
        sh:node gen:AssessedComponentShape ;
        sh:message "The load case must identify each assessed component."
    ] .

gen:AssessedComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:MegapascalDecimalQuantityShape ;
        sh:message "The assessed component must provide its externally calculated occasional-force stress."
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:class nltl:materialProperties ;
        sh:minCount 1 ;
        sh:node gen:MaterialPropertiesShape ;
        sh:message "The assessed component must link to its material properties."
    ] .

gen:MaterialPropertiesShape
    a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:MegapascalDecimalQuantityShape ;
        sh:message "The material properties must provide the component material yield strength."
    ] .

gen:MegapascalDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
