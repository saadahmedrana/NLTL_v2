<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:KilowattQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnspecifiedUnitQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:EngineOutputRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one ice class is required to determine the applicable engine-output minimum."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum continuous rating power must be provided once in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula base power P0 must be provided once in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:node gen:UnspecifiedUnitQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula displacement Delta must be provided once."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f1 must be provided once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f2 must be provided once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f3 must be provided once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:node gen:UnspecifiedUnitQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f4 must be provided once."
    ] ;
    sh:sparql [
        sh:message "Engine output must be at least f1*f2*f3*(f4*Delta + P0), and at least 740 kW for ice classes IB and IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:maximumContinuousRatingPower ?powerNode ;
                      nltl:engineOutputFormulaBasePowerP0 ?p0Node ;
                      nltl:engineOutputFormulaDisplacement ?displacementNode ;
                      nltl:engineOutputFormulaFactorF1 ?f1Node ;
                      nltl:engineOutputFormulaFactorF2 ?f2Node ;
                      nltl:engineOutputFormulaFactorF3 ?f3Node ;
                      nltl:engineOutputFormulaFactorF4 ?f4Node .

                ?powerNode qudt:numericValue ?power .
                ?p0Node qudt:numericValue ?p0 .
                ?displacementNode qudt:numericValue ?displacement .
                ?f1Node qudt:numericValue ?f1 .
                ?f2Node qudt:numericValue ?f2 .
                ?f3Node qudt:numericValue ?f3 .
                ?f4Node qudt:numericValue ?f4 .

                FILTER (
                    ?power < (?f1 * ?f2 * ?f3 * ((?f4 * ?displacement) + ?p0))
                    ||
                    (
                        ?iceClass IN (nltl:iceClassIb, nltl:iceClassIc)
                        &&
                        ?power < "740"^^xsd:decimal
                    )
                )
            }
        """
    ] .
<END_SHACL>
