<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:DecimalMetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:DecimalMillimetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:DecimalSquareCentimetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:ApplicableAreaSelectorShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:netLocalFrameFlangeArea ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:DecimalSquareCentimetreQuantityShape> ;
        sh:message "Net local-frame flange area is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:DecimalMillimetreQuantityShape> ;
        sh:message "Net attached shell-plate thickness is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:DecimalMetreQuantityShape> ;
        sh:message "Frame spacing is required to determine applicability."
    ] .

<urn:nltl:generated-shape:I2_030ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node <urn:nltl:generated-shape:ApplicableAreaSelectorShape>
    ] ;
    sh:sparql [
        sh:message "Where the local-frame area exceeds the attached-plate-flange area, the linked calculation case, required inputs, plastic neutral-axis result, and net effective plastic section modulus must be provided with their declared units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit unit:CentiM2 .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                FILTER(
                    datatype(?area) = xsd:decimal &&
                    datatype(?plateThickness) = xsd:decimal &&
                    datatype(?spacing) = xsd:decimal &&
                    (100 * ?area > 1000 * ?plateThickness * ?spacing)
                )
                FILTER NOT EXISTS {
                    ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                            nltl:webHeight ?webHeightQ ;
                            nltl:netWebThickness ?webThicknessQ ;
                            nltl:localFrameFlangeCentreHeight ?flangeHeightQ ;
                            nltl:webAngleToShellPlate ?webAngleQ ;
                            nltl:webToFlangeCentreDistance ?webDistanceQ ;
                            nltl:netEffectivePlasticSectionModulus ?sectionModulusQ .
                    $this nltl:hasCalculationCase ?case .
                    ?case a nltl:calculationCase , nltl:localFrameSectionCalculationCase ;
                          nltl:sectionCalculationCaseStructuralMember ?member ;
                          nltl:sectionCalculationCasePlating ?plating ;
                          nltl:plasticNeutralAxisHeight ?neutralAxisQ .
                    ?plating a nltl:plating ;
                             nltl:framingAngleOmega ?framingAngleQ .
                    ?webHeightQ qudt:numericValue ?webHeight ;
                                qudt:unit unit:MilliM .
                    ?webThicknessQ qudt:numericValue ?webThickness ;
                                   qudt:unit unit:MilliM .
                    ?flangeHeightQ qudt:numericValue ?flangeHeight ;
                                  qudt:unit unit:MilliM .
                    ?webAngleQ qudt:numericValue ?webAngle ;
                               qudt:unit unit:DEG .
                    ?webDistanceQ qudt:numericValue ?webDistance ;
                                  qudt:unit unit:MilliM .
                    ?neutralAxisQ qudt:numericValue ?neutralAxis ;
                                  qudt:unit unit:MilliM .
                    ?sectionModulusQ qudt:numericValue ?sectionModulus ;
                                     qudt:unit unit:CentiM3 .
                    ?framingAngleQ qudt:numericValue ?framingAngle ;
                                   qudt:unit unit:DEG .
                    FILTER(
                        datatype(?webHeight) = xsd:decimal &&
                        datatype(?webThickness) = xsd:decimal &&
                        datatype(?flangeHeight) = xsd:decimal &&
                        datatype(?webAngle) = xsd:decimal &&
                        datatype(?webDistance) = xsd:decimal &&
                        datatype(?neutralAxis) = xsd:decimal &&
                        datatype(?sectionModulus) = xsd:decimal &&
                        datatype(?framingAngle) = xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For applicable oblique framing with 20 < Omega < 70 degrees, the calculation case must provide lower and upper interpolation endpoints with decimal coordinates and results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ ;
                        nltl:hasLocalFrameSectionCalculationCase ?case .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit unit:CentiM2 .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                ?case nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleQ .
                ?angleQ qudt:numericValue ?angle ;
                        qudt:unit unit:DEG .
                FILTER(
                    datatype(?area) = xsd:decimal &&
                    datatype(?plateThickness) = xsd:decimal &&
                    datatype(?spacing) = xsd:decimal &&
                    datatype(?angle) = xsd:decimal &&
                    (100 * ?area > 1000 * ?plateThickness * ?spacing) &&
                    (?angle > 20) &&
                    (?angle < 70)
                )
                FILTER NOT EXISTS {
                    ?case nltl:interpolationLowerEndpoint ?lower ;
                          nltl:interpolationUpperEndpoint ?upper .
                    ?lower a nltl:interpolationPoint ;
                           nltl:interpolationPointCoordinate ?lowerCoordinateQ ;
                           nltl:interpolationPointResult ?lowerResultQ .
                    ?upper a nltl:interpolationPoint ;
                           nltl:interpolationPointCoordinate ?upperCoordinateQ ;
                           nltl:interpolationPointResult ?upperResultQ .
                    ?lowerCoordinateQ qudt:numericValue ?lowerCoordinate ;
                                      qudt:unit unit:UNITLESS .
                    ?upperCoordinateQ qudt:numericValue ?upperCoordinate ;
                                      qudt:unit unit:UNITLESS .
                    ?lowerResultQ qudt:numericValue ?lowerResult .
                    ?upperResultQ qudt:numericValue ?upperResult .
                    FILTER(
                        datatype(?lowerCoordinate) = xsd:decimal &&
                        datatype(?upperCoordinate) = xsd:decimal &&
                        datatype(?lowerResult) = xsd:decimal &&
                        datatype(?upperResult) = xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
