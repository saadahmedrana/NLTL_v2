<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF028AlternativeKeOrRchApprovalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:approvalStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:approvalRevocationStatus ;
        sh:class nltl:evidenceState
    ] ;
    sh:property [
        sh:path nltl:hasEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceRch ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/N>
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "An approved alternative Ke or RCH value requires revocation status, calculation or model-test evidence, and a valid alternative Ke or RCH result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:approvalStatus true .
                FILTER NOT EXISTS {
                    $this nltl:approvalRevocationStatus ?revocationStatus ;
                          nltl:hasEvidence ?evidence ;
                          ?resultProperty ?quantity .
                    ?revocationStatus a nltl:evidenceState .
                    ?evidence a nltl:evidenceArtifact ;
                              a ?evidenceType .
                    FILTER(?evidenceType IN (
                        nltl:alternativeCalculationEvidence,
                        nltl:modelTestEvidence
                    ))
                    FILTER(?resultProperty IN (
                        nltl:engineOutputCoefficientKe,
                        nltl:brashIceChannelResistanceRch
                    ))
                    ?quantity qudt:numericValue ?numericValue ;
                              qudt:unit ?unit .
                    FILTER(datatype(?numericValue) = xsd:decimal)
                    FILTER(
                        (?resultProperty = nltl:engineOutputCoefficientKe &&
                         ?unit = <http://qudt.org/vocab/unit/UNITLESS>) ||
                        (?resultProperty = nltl:brashIceChannelResistanceRch &&
                         ?unit = <http://qudt.org/vocab/unit/N>)
                    )
                }
            }
        """
    ] .
<END_SHACL>
