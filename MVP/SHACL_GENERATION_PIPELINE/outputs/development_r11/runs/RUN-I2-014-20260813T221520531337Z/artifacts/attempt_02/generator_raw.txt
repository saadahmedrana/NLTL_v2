        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:BowSubregionRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:node gen:BowSubregionCalculationCaseShape ;
        sh:minCount 4 ;
        sh:maxCount 4
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:node gen:LineLoadQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:totalGlancingImpactForce ;
        sh:node gen:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The four bow-subregion mid-length positions must be distinct." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    SELECT $this (COUNT(DISTINCT ?mid) AS ?midCount)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionMidLengthPosition ?midQuantity .
                        ?midQuantity qudt:numericValue ?mid .
                    }
                    GROUP BY $this
                }
                FILTER (?midCount != 4)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each bow-subregion mid-length position must equal 1/8, 3/8, 5/8, or 7/8 of bow length within relative tolerance 1e-6 and absolute fallback 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?bowLengthQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?bowLengthQuantity qudt:numericValue ?bowLength .
                ?case nltl:bowSubregionMidLengthPosition ?midQuantity .
                ?midQuantity qudt:numericValue ?mid .
                BIND (?bowLength / 8 AS ?expected1)
                BIND ((3 * ?bowLength) / 8 AS ?expected2)
                BIND ((5 * ?bowLength) / 8 AS ?expected3)
                BIND ((7 * ?bowLength) / 8 AS ?expected4)
                BIND (IF(ABS(?expected1) * 0.000001 > 0.000000001, ABS(?expected1) * 0.000001, 0.000000001) AS ?tolerance1)
                BIND (IF(ABS(?expected2) * 0.000001 > 0.000000001, ABS(?expected2) * 0.000001, 0.000000001) AS ?tolerance2)
                BIND (IF(ABS(?expected3) * 0.000001 > 0.000000001, ABS(?expected3) * 0.000001, 0.000000001) AS ?tolerance3)
                BIND (IF(ABS(?expected4) * 0.000001 > 0.000000001, ABS(?expected4) * 0.000001, 0.000000001) AS ?tolerance4)
                FILTER (
                    ABS(?mid - ?expected1) > ?tolerance1 &&
                    ABS(?mid - ?expected2) > ?tolerance2 &&
                    ABS(?mid - ?expected3) > ?tolerance3 &&
                    ABS(?mid - ?expected4) > ?tolerance4
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Selected bow force, line load, and pressure must equal their respective maxima across the four subregion calculations within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForceQuantity ;
                      nltl:selectedMaximumBowLineLoad ?selectedLineLoadQuantity ;
                      nltl:selectedMaximumBowPressure ?selectedPressureQuantity .
                ?selectedForceQuantity qudt:numericValue ?selectedForce .
                ?selectedLineLoadQuantity qudt:numericValue ?selectedLineLoad .
                ?selectedPressureQuantity qudt:numericValue ?selectedPressure .
                {
                    SELECT $this
                           (MAX(?force) AS ?maximumForce)
                           (MAX(?lineLoad) AS ?maximumLineLoad)
                           (MAX(?pressure) AS ?maximumPressure)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?forceQuantity ;
                              nltl:bowSubregionCalculatedLineLoad ?lineLoadQuantity ;
                              nltl:bowSubregionCalculatedPressure ?pressureQuantity .
                        ?forceQuantity qudt:numericValue ?force .
                        ?lineLoadQuantity qudt:numericValue ?lineLoad .
                        ?pressureQuantity qudt:numericValue ?pressure .
                    }
                    GROUP BY $this
                }
                BIND (IF(ABS(?maximumForce) * 0.000001 > 0.000000001, ABS(?maximumForce) * 0.000001, 0.000000001) AS ?forceTolerance)
                BIND (IF(ABS(?maximumLineLoad) * 0.000001 > 0.000000001, ABS(?maximumLineLoad) * 0.000001, 0.000000001) AS ?lineLoadTolerance)
                BIND (IF(ABS(?maximumPressure) * 0.000001 > 0.000000001, ABS(?maximumPressure) * 0.000001, 0.000000001) AS ?pressureTolerance)
                FILTER (
                    ABS(?selectedForce - ?maximumForce) > ?forceTolerance ||
                    ABS(?selectedLineLoad - ?maximumLineLoad) > ?lineLoadTolerance ||
                    ABS(?selectedPressure - ?maximumPressure) > ?pressureTolerance
                )
            }
        """
    ] .
