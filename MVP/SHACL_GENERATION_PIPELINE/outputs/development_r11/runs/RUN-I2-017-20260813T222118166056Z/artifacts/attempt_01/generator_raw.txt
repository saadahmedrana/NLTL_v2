        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER NOT EXISTS {
        $this nltl:upperIceWaterlineDraughtDUI ?duiQuantity ;
              nltl:displacementFactorThreshold ?thresholdQuantity ;
              nltl:hasNonBowIceLoadCalculationCase ?calculationCase ;
              nltl:hasTableLookupCase ?tableCase .

        ?duiQuantity qudt:numericValue ?duiRaw ;
                     qudt:unit unit:M .
        ?thresholdQuantity qudt:numericValue ?thresholdRaw ;
                           qudt:unit unit:M .

        ?calculationCase
            nltl:shipDisplacementFactor ?dfQuantity ;
            nltl:nonBowIceForce ?forceQuantity ;
            nltl:nonBowIceLineLoad ?lineLoadQuantity .

        ?dfQuantity qudt:numericValue ?dfRaw ;
                    qudt:unit unit:UNITLESS .
        ?forceQuantity qudt:numericValue ?forceRaw ;
                       qudt:unit unit:KiloN .
        ?lineLoadQuantity qudt:numericValue ?lineLoadRaw ;
                          qudt:unit unit:MegaN-M-PER-M2 .

        ?tableCase
            nltl:crushingFailureClassFactor ?cfcQuantity ;
            nltl:loadPatchDimensionClassFactor ?cfdQuantity .

        ?cfcQuantity qudt:numericValue ?cfcRaw ;
                     qudt:unit unit:UNITLESS .
        ?cfdQuantity qudt:numericValue ?cfdRaw ;
                     qudt:unit unit:UNITLESS .

        BIND(xsd:decimal(?duiRaw) AS ?dui)
        BIND(xsd:decimal(?thresholdRaw) AS ?threshold)
        BIND(xsd:decimal(?dfRaw) AS ?df)
        BIND(xsd:decimal(?forceRaw) AS ?forceKN)
        BIND(xsd:decimal(?lineLoadRaw) AS ?lineLoad)
        BIND(xsd:decimal(?cfcRaw) AS ?cfc)
        BIND(xsd:decimal(?cfdRaw) AS ?cfd)

        BIND(IF(?dui <= ?threshold, ?dui, ?threshold) AS ?dfPowerBase)
        BIND(
            IF(
                ?dui <= ?threshold,
                ?df,
                ?df - (0.10 * (?dui - ?threshold))
            )
            AS ?dfAdjusted
        )

        BIND(?dfPowerBase * ?dfPowerBase AS ?base2)
        BIND(?base2 * ?base2 AS ?base4)
        BIND(?base4 * ?base4 AS ?base8)
        BIND(?base8 * ?base8 AS ?base16)

        BIND(?dfAdjusted * ?dfAdjusted AS ?df2)
        BIND(?df2 * ?df2 AS ?df4)
        BIND(?df4 * ?df4 AS ?df8)
        BIND(?df8 * ?df8 AS ?df16)
        BIND(?df16 * ?df8 AS ?df24)
        BIND(?df24 * ?dfAdjusted AS ?df25)

        BIND(360.0 * ?cfc * ?df AS ?expectedForceKN)
        BIND(?forceKN / 1000.0 AS ?forceMN)

        BIND(?forceMN * ?forceMN AS ?force2)
        BIND(?force2 * ?force2 AS ?force4)
        BIND(?force4 * ?force4 AS ?force8)
        BIND(?force8 * ?force8 AS ?force16)
        BIND(?force16 * ?force16 AS ?force32)
        BIND(?force32 * ?force16 AS ?force48)
        BIND(?force48 * ?force8 AS ?force56)
        BIND(?force56 * ?force4 AS ?force60)
        BIND(?force60 * ?forceMN AS ?force61)

        BIND(ABS(?lineLoad) AS ?lineLoadAbs)
        BIND(?lineLoadAbs * ?lineLoadAbs AS ?lineLoad2)
        BIND(?lineLoad2 * ?lineLoad2 AS ?lineLoad4)
        BIND(?lineLoad4 * ?lineLoad4 AS ?lineLoad8)
        BIND(?lineLoad8 * ?lineLoad8 AS ?lineLoad16)
        BIND(?lineLoad16 * ?lineLoad16 AS ?lineLoad32)
        BIND(?lineLoad32 * ?lineLoad32 AS ?lineLoad64)
        BIND(?lineLoad64 * ?lineLoad32 AS ?lineLoad96)
        BIND(?lineLoad96 * ?lineLoad4 AS ?lineLoad100)

        BIND(ABS(0.639 * ?cfd) AS ?lineCoefficient)
        BIND(?lineCoefficient * ?lineCoefficient AS ?coefficient2)
        BIND(?coefficient2 * ?coefficient2 AS ?coefficient4)
        BIND(?coefficient4 * ?coefficient4 AS ?coefficient8)
        BIND(?coefficient8 * ?coefficient8 AS ?coefficient16)
        BIND(?coefficient16 * ?coefficient16 AS ?coefficient32)
        BIND(?coefficient32 * ?coefficient32 AS ?coefficient64)
        BIND(?coefficient64 * ?coefficient32 AS ?coefficient96)
        BIND(?coefficient96 * ?coefficient4 AS ?coefficient100)
        BIND(?coefficient100 * ?force61 AS ?expectedLineLoad100)

        FILTER(
            ?dfPowerBase >= 0 &&
            ?dfAdjusted >= 0 &&
            ABS(?df25 - ?base16)
                <= ((0.000025 * ABS(?base16)) + 0.000000000001) &&
            ABS(?forceKN - ?expectedForceKN)
                <= ((0.000001 * ABS(?expectedForceKN)) + 0.000000001) &&
            ?forceMN >= 0 &&
            (?lineLoad * ?cfd) >= 0 &&
            ABS(?lineLoad100 - ?expectedLineLoad100)
                <= ((0.0001 * ABS(?expectedLineLoad100)) + 0.000000000001)
        )
    }
}
