<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "0"^^xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "0"^^xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    ] .

        sh:datatype xsd:decimal ;

        sh:maxCount 1

    ] .

    sh:class nltl:tableLookupCase ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchDimensionClassFactor ;

        sh:maxCount 1

    sh:class nltl:nonBowIceLoadCalculationCase ;
    sh:property [
        sh:path nltl:shipDisplacementFactor ;

    ] ;
    sh:property [

    ] ;
    sh:property [

gen:NonBowIceLoadShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:displacementFactorThreshold ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasNonBowIceLoadCalculationCase ;
        sh:class nltl:nonBowIceLoadCalculationCase ;
        sh:node gen:NonBowIceLoadCalculationCaseShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:TableLookupCaseShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "DF, non-bow force, or non-bow line load does not satisfy the prescribed formula within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDraughtDUI ?duiQuantity ;
                      nltl:displacementFactorThreshold ?thresholdQuantity ;
                      nltl:hasNonBowIceLoadCalculationCase ?calculationCase ;
                      nltl:hasTableLookupCase ?tableCase .

                ?duiQuantity qudt:numericValue ?dui .
                ?thresholdQuantity qudt:numericValue ?threshold .

                ?tableCase nltl:crushingFailureClassFactor ?cfcQuantity ;
                           nltl:loadPatchDimensionClassFactor ?cfdQuantity .
                ?cfcQuantity qudt:numericValue ?cfc .
                ?cfdQuantity qudt:numericValue ?cfd .

                ?calculationCase nltl:shipDisplacementFactor ?dfQuantity ;
                                 nltl:nonBowIceForce ?forceQuantity ;
                                 nltl:nonBowIceLineLoad ?lineLoadQuantity .
                ?dfQuantity qudt:numericValue ?df .
                ?forceQuantity qudt:numericValue ?forceKN .
                ?lineLoadQuantity qudt:numericValue ?lineLoad .

                BIND("0.000001"^^xsd:decimal AS ?relativeTolerance)
                BIND("0.000000001"^^xsd:decimal AS ?absoluteTolerance)

                BIND(IF(?dui <= ?threshold, ?dui, ?threshold) AS ?dfRootBase)
                BIND(IF(?dui <= ?threshold,
                        "0"^^xsd:decimal,
                        "0.10"^^xsd:decimal * (?dui - ?threshold))
                     AS ?dfIncrement)

                BIND(?dfRootBase * ?dfRootBase AS ?dfBase2)
                BIND(?dfBase2 * ?dfBase2 AS ?dfBase4)
                BIND(?dfBase4 * ?dfBase4 AS ?dfBase8)
                BIND(?dfBase8 * ?dfBase8 AS ?dfBase16)

                BIND((?df / ("1"^^xsd:decimal + ?relativeTolerance)) - ?dfIncrement AS ?dfRelHighBase)
                BIND((?df / ("1"^^xsd:decimal - ?relativeTolerance)) - ?dfIncrement AS ?dfRelLowBase)
                BIND((?df - ?absoluteTolerance) - ?dfIncrement AS ?dfAbsHighBase)
                BIND((?df + ?absoluteTolerance) - ?dfIncrement AS ?dfAbsLowBase)

                BIND(?dfRelHighBase * ?dfRelHighBase AS ?dfRelHigh2)
                BIND(?dfRelHigh2 * ?dfRelHigh2 AS ?dfRelHigh4)
                BIND(?dfRelHigh4 * ?dfRelHigh4 AS ?dfRelHigh8)
                BIND(?dfRelHigh8 * ?dfRelHigh8 AS ?dfRelHigh16)
                BIND(?dfRelHigh16 * ?dfRelHigh8 * ?dfRelHighBase AS ?dfRelHigh25)

                BIND(?dfRelLowBase * ?dfRelLowBase AS ?dfRelLow2)
                BIND(?dfRelLow2 * ?dfRelLow2 AS ?dfRelLow4)
                BIND(?dfRelLow4 * ?dfRelLow4 AS ?dfRelLow8)
                BIND(?dfRelLow8 * ?dfRelLow8 AS ?dfRelLow16)
                BIND(?dfRelLow16 * ?dfRelLow8 * ?dfRelLowBase AS ?dfRelLow25)

                BIND(?dfAbsHighBase * ?dfAbsHighBase AS ?dfAbsHigh2)
                BIND(?dfAbsHigh2 * ?dfAbsHigh2 AS ?dfAbsHigh4)
                BIND(?dfAbsHigh4 * ?dfAbsHigh4 AS ?dfAbsHigh8)
                BIND(?dfAbsHigh8 * ?dfAbsHigh8 AS ?dfAbsHigh16)
                BIND(?dfAbsHigh16 * ?dfAbsHigh8 * ?dfAbsHighBase AS ?dfAbsHigh25)

                BIND(?dfAbsLowBase * ?dfAbsLowBase AS ?dfAbsLow2)
                BIND(?dfAbsLow2 * ?dfAbsLow2 AS ?dfAbsLow4)
                BIND(?dfAbsLow4 * ?dfAbsLow4 AS ?dfAbsLow8)
                BIND(?dfAbsLow8 * ?dfAbsLow8 AS ?dfAbsLow16)
                BIND(?dfAbsLow16 * ?dfAbsLow8 * ?dfAbsLowBase AS ?dfAbsLow25)

                BIND("360"^^xsd:decimal * ?cfc * ?df AS ?expectedForceKN)
                BIND(ABS(?forceKN - ?expectedForceKN) AS ?forceDifference)
                BIND(?relativeTolerance * ABS(?expectedForceKN) AS ?forceRelativeLimit)

                BIND("0.639"^^xsd:decimal * ?cfd AS ?lineCoefficient)
                BIND(?forceKN / "1000"^^xsd:decimal AS ?forceMN)

                BIND(?lineCoefficient * ?lineCoefficient AS ?coefficient2)
                BIND(?coefficient2 * ?coefficient2 AS ?coefficient4)
                BIND(?coefficient4 * ?coefficient4 AS ?coefficient8)
                BIND(?coefficient8 * ?coefficient8 AS ?coefficient16)
                BIND(?coefficient16 * ?coefficient16 AS ?coefficient32)
                BIND(?coefficient32 * ?coefficient32 AS ?coefficient64)
                BIND(?coefficient64 * ?coefficient32 * ?coefficient4 AS ?coefficient100)

                BIND(?forceMN * ?forceMN AS ?forceMN2)
                BIND(?forceMN2 * ?forceMN2 AS ?forceMN4)
                BIND(?forceMN4 * ?forceMN4 AS ?forceMN8)
                BIND(?forceMN8 * ?forceMN8 AS ?forceMN16)
                BIND(?forceMN16 * ?forceMN16 AS ?forceMN32)
                BIND(?forceMN32 * ?forceMN16 * ?forceMN8 * ?forceMN4 * ?forceMN AS ?forceMN61)
                BIND(?coefficient100 * ?forceMN61 AS ?lineFormulaPower100)

                BIND(?lineLoad / ("1"^^xsd:decimal + ?relativeTolerance) AS ?lineRelHighBase)
                BIND(?lineLoad / ("1"^^xsd:decimal - ?relativeTolerance) AS ?lineRelLowBase)
                BIND(?lineLoad - ?absoluteTolerance AS ?lineAbsHighBase)
                BIND(?lineLoad + ?absoluteTolerance AS ?lineAbsLowBase)

                BIND(?lineRelHighBase * ?lineRelHighBase AS ?lineRelHigh2)
                BIND(?lineRelHigh2 * ?lineRelHigh2 AS ?lineRelHigh4)
                BIND(?lineRelHigh4 * ?lineRelHigh4 AS ?lineRelHigh8)
                BIND(?lineRelHigh8 * ?lineRelHigh8 AS ?lineRelHigh16)
                BIND(?lineRelHigh16 * ?lineRelHigh16 AS ?lineRelHigh32)
                BIND(?lineRelHigh32 * ?lineRelHigh32 AS ?lineRelHigh64)
                BIND(?lineRelHigh64 * ?lineRelHigh32 * ?lineRelHigh4 AS ?lineRelHigh100)

                BIND(?lineRelLowBase * ?lineRelLowBase AS ?lineRelLow2)
                BIND(?lineRelLow2 * ?lineRelLow2 AS ?lineRelLow4)
                BIND(?lineRelLow4 * ?lineRelLow4 AS ?lineRelLow8)
                BIND(?lineRelLow8 * ?lineRelLow8 AS ?lineRelLow16)
                BIND(?lineRelLow16 * ?lineRelLow16 AS ?lineRelLow32)
                BIND(?lineRelLow32 * ?lineRelLow32 AS ?lineRelLow64)
                BIND(?lineRelLow64 * ?lineRelLow32 * ?lineRelLow4 AS ?lineRelLow100)

                BIND(?lineAbsHighBase * ?lineAbsHighBase AS ?lineAbsHigh2)
                BIND(?lineAbsHigh2 * ?lineAbsHigh2 AS ?lineAbsHigh4)
                BIND(?lineAbsHigh4 * ?lineAbsHigh4 AS ?lineAbsHigh8)
                BIND(?lineAbsHigh8 * ?lineAbsHigh8 AS ?lineAbsHigh16)
                BIND(?lineAbsHigh16 * ?lineAbsHigh16 AS ?lineAbsHigh32)
                BIND(?lineAbsHigh32 * ?lineAbsHigh32 AS ?lineAbsHigh64)
                BIND(?lineAbsHigh64 * ?lineAbsHigh32 * ?lineAbsHigh4 AS ?lineAbsHigh100)

                BIND(?lineAbsLowBase * ?lineAbsLowBase AS ?lineAbsLow2)
                BIND(?lineAbsLow2 * ?lineAbsLow2 AS ?lineAbsLow4)
                BIND(?lineAbsLow4 * ?lineAbsLow4 AS ?lineAbsLow8)
                BIND(?lineAbsLow8 * ?lineAbsLow8 AS ?lineAbsLow16)
                BIND(?lineAbsLow16 * ?lineAbsLow16 AS ?lineAbsLow32)
                BIND(?lineAbsLow32 * ?lineAbsLow32 AS ?lineAbsLow64)
                BIND(?lineAbsLow64 * ?lineAbsLow32 * ?lineAbsLow4 AS ?lineAbsLow100)

                FILTER (
                    (
                        (
                            ?dfRelHighBase > "0"^^xsd:decimal &&
                            ?dfRelHigh25 > ?dfBase16
                        ) &&
                        (
                            ?dfAbsHighBase > "0"^^xsd:decimal &&
                            ?dfAbsHigh25 > ?dfBase16
                        )
                    ) ||
                    (
                        ?dfRelLow25 < ?dfBase16 &&
                        ?dfAbsLow25 < ?dfBase16
                    ) ||
                    (
                        ?forceDifference > ?forceRelativeLimit &&
                        ?forceDifference > ?absoluteTolerance
                    ) ||
                    (
                        (
                            ?lineRelHigh100 > ?lineFormulaPower100
                        ) &&
                        (
                            ?lineAbsHighBase > "0"^^xsd:decimal &&
                            ?lineAbsHigh100 > ?lineFormulaPower100
                        )
                    ) ||
                    (
                        ?lineRelLow100 < ?lineFormulaPower100 &&
                        ?lineAbsLow100 < ?lineFormulaPower100
                    )
                )
            }
        """
    ] .
<END_SHACL>
