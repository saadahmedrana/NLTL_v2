<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:HullAreaFactorValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:StructuralMemberHullAreaShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:spansHullArea ;
        sh:class nltl:hullRegionValue
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:class nltl:hullAreaValue ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:class nltl:hullAreaValue ;
                sh:minCount 2
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:HullAreaFactorValueShape
            ] ;
            sh:property [
                sh:path (
                    [ sh:inversePath nltl:hasStructuralMember ]
                    nltl:hullAreaFactor
                ) ;
                sh:minCount 1 ;
                sh:node gen:HullAreaFactorValueShape
            ]
        ]
    ) .

gen:HullAreaBoundaryFactorShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberHullAreaShape
    ] ;
    sh:sparql [
        sh:message "Each boundary-spanning member must have one valid selected factor equal to the largest valid ship-owned hull area factor, within relative tolerance 1e-6 and absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasSpannedHullArea ?area1, ?area2 .
                FILTER (?area1 != ?area2)

                OPTIONAL {
                    ?member nltl:selectedHullAreaFactor ?selectedNode .
                    ?selectedNode a qudt:QuantityValue ;
                        qudt:numericValue ?selected ;
                        qudt:unit unit:UNITLESS .
                    FILTER (DATATYPE(?selected) = xsd:decimal)
                }

                OPTIONAL {
                    $this nltl:hullAreaFactor ?candidateNode .
                    ?candidateNode a qudt:QuantityValue ;
                        qudt:numericValue ?candidate ;
                        qudt:unit unit:UNITLESS .
                    FILTER (DATATYPE(?candidate) = xsd:decimal)
                }
            }
            GROUP BY $this ?member
            HAVING (
                COUNT(DISTINCT ?selectedNode) != 1 ||
                COUNT(DISTINCT ?candidateNode) = 0 ||
                ABS(MAX(?selected) - MAX(?candidate)) >
                    ((0.000001 * ABS(MAX(?candidate))) + 0.000000000001)
            )
        """
    ] .
<END_SHACL>
