<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_022_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node gen:I2_022_PlatingConditionalShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasPlating ;
                sh:node gen:I2_022_NonApplicablePlatingShape
            ]
        ]
        [
            sh:node gen:I2_022_ShipInputsShape ;
            sh:property [
                sh:path nltl:hasPlating ;
                sh:qualifiedValueShape gen:I2_022_ApplicablePlatingShape ;
                sh:qualifiedMinCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For plating with Omega at least 70 degrees, the selected hull area factor must correspond to AF and the reported net thickness must satisfy the formula within a 1e-6 relative thickness tolerance, evaluated by its squared equivalent with a 2e-6 relative tolerance and a 1e-12 absolute-scale fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:averageIcePressure ?averagePressureQuantity ;
                      nltl:hullAreaFactor ?hullAreaFactorQuantity ;
                      nltl:loadPatchHeight ?loadPatchHeightQuantity ;
                      nltl:peakPressureFactor ?peakPressureFactorQuantity ;
                      nltl:yieldStrength ?yieldStrengthQuantity .

                ?plating nltl:framingAngleOmega ?angleQuantity ;
                         nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity ;
                         nltl:platingSupportedByStructuralMember ?member .

                ?member nltl:frameSpacing ?spacingQuantity ;
                        nltl:selectedHullAreaFactor ?selectedAreaFactorQuantity .

                ?angleQuantity qudt:numericValue ?angle ;
                               qudt:unit unit:DEG .
                ?thicknessQuantity qudt:numericValue ?reportedThickness ;
                                   qudt:unit unit:MilliM .
                ?spacingQuantity qudt:numericValue ?spacing ;
                                 qudt:unit unit:M .
                ?selectedAreaFactorQuantity qudt:numericValue ?selectedAreaFactor ;
                                            qudt:unit unit:UNITLESS .
                ?averagePressureQuantity qudt:numericValue ?averagePressure ;
                                         qudt:unit unit:MegaPA .
                ?hullAreaFactorQuantity qudt:numericValue ?hullAreaFactor ;
                                        qudt:unit unit:UNITLESS .
                ?loadPatchHeightQuantity qudt:numericValue ?loadPatchHeight ;
                                         qudt:unit unit:M .
                ?peakPressureFactorQuantity qudt:numericValue ?peakPressureFactor ;
                                            qudt:unit unit:UNITLESS .
                ?yieldStrengthQuantity qudt:numericValue ?yieldStrength ;
                                       qudt:unit unit:MegaPA .

                FILTER (?angle >= 70)
                BIND(
                    ((500 * ?spacing) / (1 + (?spacing / (2 * ?loadPatchHeight))))
                    AS ?thicknessCoefficient
                )
                BIND(
                    (?thicknessCoefficient * ?thicknessCoefficient *
                     ((?selectedAreaFactor * ?peakPressureFactor * ?averagePressure) /
                      ?yieldStrength))
                    AS ?expectedThicknessSquared
                )
                BIND(
                    IF(
                        ABS(?expectedThicknessSquared) > 0.000000000001,
                        ABS(?expectedThicknessSquared),
                        0.000000000001
                    )
                    AS ?formulaScale
                )
                BIND(
                    IF(
                        ABS(?hullAreaFactor) > 0.000000000001,
                        ABS(?hullAreaFactor),
                        0.000000000001
                    )
                    AS ?areaFactorScale
                )
                FILTER (
                    ABS((?reportedThickness * ?reportedThickness) -
                        ?expectedThicknessSquared) >
                        (0.000002 * ?formulaScale)
                    ||
                    ABS(?selectedAreaFactor - ?hullAreaFactor) >
                        (0.000001 * ?areaFactorScale)
                )
            }
        """
    ] .

        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_022_NonNegativeUnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

        sh:path nltl:yieldStrength ;

        sh:node gen:I2_022_PositiveMegaPAQuantityShape
    ] .
