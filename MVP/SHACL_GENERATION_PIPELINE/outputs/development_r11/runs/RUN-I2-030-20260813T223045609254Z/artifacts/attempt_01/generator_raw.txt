gen:I2_030_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberShape ;
        sh:minCount 1 ;
        sh:message "A structural member and its applicability inputs are required for the calculation."
    ] ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:node gen:PlatingShape
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape

        sh:path nltl:sectionModulus ;

    sh:sparql [
        sh:message "When the local-frame area exceeds the attached-plate-flange area, the calculation inputs, result, framing angle, calculation case, and section modulus must be available with the prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit unit:CentiM2 .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                BIND(100.0 * xsd:decimal(?area) AS ?localFrameArea)
                BIND(1000.0 * xsd:decimal(?plateThickness) * xsd:decimal(?spacing) AS ?attachedPlateArea)
                FILTER(?localFrameArea > ?attachedPlateArea)
                FILTER NOT EXISTS {
                    ?member nltl:webHeight ?webHeightQ ;
                            nltl:netWebThickness ?webThicknessQ ;
                            nltl:localFrameFlangeCentreHeight ?flangeHeightQ ;
                            nltl:webToFlangeCentreDistance ?flangeDistanceQ ;
                            nltl:webAngleToShellPlate ?webAngleQ ;
                            nltl:netEffectivePlasticSectionModulus ?plasticModulusQ .
                    ?webHeightQ qudt:numericValue ?webHeight ;
                                qudt:unit unit:MilliM .
                    ?webThicknessQ qudt:numericValue ?webThickness ;
                                   qudt:unit unit:MilliM .
                    ?flangeHeightQ qudt:numericValue ?flangeHeight ;
                                   qudt:unit unit:MilliM .
                    ?flangeDistanceQ qudt:numericValue ?flangeDistance ;
                                     qudt:unit unit:MilliM .
                    ?webAngleQ qudt:numericValue ?webAngle ;
                               qudt:unit unit:DEG .
                    ?plasticModulusQ qudt:numericValue ?plasticModulus ;
                                     qudt:unit unit:CentiM3 .
                    $this nltl:hasCalculationCase ?case ;
                          nltl:hasPlating ?plating ;
                          nltl:sectionModulus ?sectionModulusQ .
                    ?case nltl:plasticNeutralAxisHeight ?axisQ .
                    ?axisQ qudt:numericValue ?axis ;
                           qudt:unit unit:MilliM .
                    ?plating nltl:framingAngleOmega ?omegaQ .
                    ?omegaQ qudt:numericValue ?omega ;
                            qudt:unit unit:DEG .
                    ?sectionModulusQ qudt:numericValue ?sectionModulus ;
                                     qudt:unit unit:CentiM3 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Plastic neutral axis height must satisfy the prescribed formula within relative tolerance 1e-6, with absolute fallback 1e-9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:webHeight ?webHeightQ ;
                        nltl:netWebThickness ?webThicknessQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit unit:CentiM2 .
                ?webHeightQ qudt:numericValue ?webHeight ;
                            qudt:unit unit:MilliM .
                ?webThicknessQ qudt:numericValue ?webThickness ;
                               qudt:unit unit:MilliM .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                ?case nltl:plasticNeutralAxisHeight ?axisQ .
                ?axisQ qudt:numericValue ?reportedAxis ;
                       qudt:unit unit:MilliM .
                BIND(xsd:decimal(?area) AS ?a)
                BIND(xsd:decimal(?webHeight) AS ?hw)
                BIND(xsd:decimal(?webThickness) AS ?tw)
                BIND(xsd:decimal(?plateThickness) AS ?tp)
                BIND(xsd:decimal(?spacing) AS ?s)
                FILTER((100.0 * ?a) > (1000.0 * ?tp * ?s))
                FILTER(?tw != 0.0)
                BIND((100.0 * ?a + ?hw * ?tw - 1000.0 * ?tp * ?s) / (2.0 * ?tw) AS ?expectedAxis)
                BIND(IF(ABS(?expectedAxis) * 0.000001 > 0.000000001,
                        ABS(?expectedAxis) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(xsd:decimal(?reportedAxis) - ?expectedAxis) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Outside the oblique-framing interval, net effective plastic section modulus must satisfy the prescribed formula within relative tolerance 1e-6, with absolute fallback 1e-9 cm3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case ;
                      nltl:hasPlating ?plating .
                ?member nltl:frameSpacing ?spacingQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:webHeight ?webHeightQ ;
                        nltl:netWebThickness ?webThicknessQ ;
                        nltl:netLocalFrameFlangeArea ?flangeAreaQ ;
                        nltl:localFrameFlangeCentreHeight ?flangeHeightQ ;
                        nltl:webToFlangeCentreDistance ?flangeDistanceQ ;
                        nltl:webAngleToShellPlate ?webAngleQ ;
                        nltl:netEffectivePlasticSectionModulus ?modulusQ .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?webHeightQ qudt:numericValue ?webHeight ;
                            qudt:unit unit:MilliM .
                ?webThicknessQ qudt:numericValue ?webThickness ;
                               qudt:unit unit:MilliM .
                ?flangeAreaQ qudt:numericValue ?flangeArea ;
                              qudt:unit unit:CentiM2 .
                ?flangeHeightQ qudt:numericValue ?flangeHeight ;
                                qudt:unit unit:MilliM .
                ?flangeDistanceQ qudt:numericValue ?flangeDistance ;
                                  qudt:unit unit:MilliM .
                ?webAngleQ qudt:numericValue ?webAngle ;
                           qudt:unit unit:DEG .
                ?modulusQ qudt:numericValue ?reportedModulus ;
                          qudt:unit unit:CentiM3 .
                ?case nltl:plasticNeutralAxisHeight ?axisQ .
                ?axisQ qudt:numericValue ?axis ;
                       qudt:unit unit:MilliM .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit unit:DEG .
                BIND(xsd:decimal(?spacing) AS ?s)
                BIND(xsd:decimal(?plateThickness) AS ?tp)
                BIND(xsd:decimal(?webHeight) AS ?hw)
                BIND(xsd:decimal(?webThickness) AS ?tw)
                BIND(xsd:decimal(?flangeArea) AS ?af)
                BIND(xsd:decimal(?flangeHeight) AS ?hfc)
                BIND(xsd:decimal(?flangeDistance) AS ?bw)
                BIND(xsd:decimal(?axis) AS ?zna)
                BIND(xsd:decimal(?webAngle) * 3.14159265358979323846 / 180.0 AS ?phi)
                FILTER((100.0 * ?af) > (1000.0 * ?tp * ?s))
                FILTER(xsd:decimal(?omega) <= 20.0 || xsd:decimal(?omega) >= 70.0)
                BIND(
                    ?tp * ?s * (?zna + ?tp / 2.0) * math:sin(?phi)
                    + (((?hw - ?zna) * (?hw - ?zna) + ?zna * ?zna) * ?tw * math:sin(?phi)) / 2000.0
                    + ?af * ((?hfc - ?zna) * math:sin(?phi) - ?bw * math:cos(?phi)) / 10.0
                    AS ?expectedModulus
                )
                BIND(IF(ABS(?expectedModulus) * 0.000001 > 0.000000001,
                        ABS(?expectedModulus) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(xsd:decimal(?reportedModulus) - ?expectedModulus) > ?tolerance)
            }
        """
    ] ;

            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

    sh:sparql [
        sh:message "For oblique framing, net effective plastic section modulus must equal the linear interpolation between the 20-degree and 70-degree endpoint results within relative tolerance 1e-6, with absolute fallback 1e-9 cm3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasPlating ?plating ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ ;
                        nltl:netEffectivePlasticSectionModulus ?modulusQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit unit:CentiM2 .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit unit:MilliM .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit unit:M .
                ?modulusQ qudt:numericValue ?reportedModulus ;
                          qudt:unit unit:CentiM3 .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit unit:DEG .
                ?case nltl:interpolationLowerEndpoint ?lower ;
                      nltl:interpolationUpperEndpoint ?upper .
                ?lower nltl:interpolationPointCoordinate ?lowerCoordinateQ ;
                       nltl:interpolationPointResult ?lowerResultQ .
                ?upper nltl:interpolationPointCoordinate ?upperCoordinateQ ;
                       nltl:interpolationPointResult ?upperResultQ .
                ?lowerCoordinateQ qudt:numericValue ?lowerCoordinate ;
                                  qudt:unit unit:UNITLESS .
                ?upperCoordinateQ qudt:numericValue ?upperCoordinate ;
                                  qudt:unit unit:UNITLESS .
                ?lowerResultQ qudt:numericValue ?lowerResult ;
                              qudt:unit unit:CentiM3 .
                ?upperResultQ qudt:numericValue ?upperResult ;
                              qudt:unit unit:CentiM3 .
                FILTER((100.0 * xsd:decimal(?area)) >
                       (1000.0 * xsd:decimal(?plateThickness) * xsd:decimal(?spacing)))
                FILTER(xsd:decimal(?omega) > 20.0 && xsd:decimal(?omega) < 70.0)
                FILTER(xsd:decimal(?upperCoordinate) != xsd:decimal(?lowerCoordinate))
                BIND(
                    xsd:decimal(?lowerResult)
                    + ((xsd:decimal(?omega) - xsd:decimal(?lowerCoordinate))
                    * (xsd:decimal(?upperResult) - xsd:decimal(?lowerResult))
                    / (xsd:decimal(?upperCoordinate) - xsd:decimal(?lowerCoordinate)))
                    AS ?expectedModulus
                )
                BIND(IF(ABS(?expectedModulus) * 0.000001 > 0.000000001,
                        ABS(?expectedModulus) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(xsd:decimal(?reportedModulus) - ?expectedModulus) > ?tolerance)
            }
        """
    ] .
