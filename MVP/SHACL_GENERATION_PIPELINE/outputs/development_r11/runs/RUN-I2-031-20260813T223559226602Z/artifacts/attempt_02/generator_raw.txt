<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable local frame must have a midspan plastic collapse load expressed as an xsd:decimal quantity in kilonewtons." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                    ?collapseQuantity
                        qudt:numericValue ?collapseValue ;
                        qudt:unit unit:KiloN .
                    FILTER(datatype(?collapseValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable local frame must have a structural-member load case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each load case of an applicable local frame must provide a decimal combined shear and bending demand in the collapse-load unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location ;
                    nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER NOT EXISTS {
                    ?member nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                    ?collapseQuantity
                        qudt:numericValue ?collapseValue ;
                        qudt:unit ?commonUnit .
                    ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                    ?demandQuantity
                        qudt:numericValue ?demandValue ;
                        qudt:unit ?commonUnit .
                    FILTER(
                        datatype(?collapseValue) = xsd:decimal &&
                        datatype(?demandValue) = xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed the applicable member's midspan plastic collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location ;
                    nltl:midspanPlasticCollapseLoad ?collapseQuantity ;
                    nltl:hasStructuralMemberLoadCase ?loadCase .
                ?collapseQuantity
                    qudt:numericValue ?collapseValue ;
                    qudt:unit ?commonUnit .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity
                    qudt:numericValue ?demandValue ;
                    qudt:unit ?commonUnit .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER(
                    datatype(?collapseValue) = xsd:decimal &&
                    datatype(?demandValue) = xsd:decimal &&
                    ?demandValue > ?collapseValue
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must be recorded as an xsd:decimal quantity in kilonewtons when an applicable local frame is present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:plasticStrength ?strengthQuantity .
                    ?strengthQuantity
                        qudt:numericValue ?strengthValue ;
                        qudt:unit unit:KiloN .
                    FILTER(datatype(?strengthValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must equal the applicable member's midspan plastic collapse load within a relative tolerance of 1e-6 and an absolute fallback of 1e-12 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this
                    nltl:hasStructuralMember ?member ;
                    nltl:plasticStrength ?strengthQuantity .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation ?location ;
                    nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                ?strengthQuantity
                    qudt:numericValue ?strengthValue ;
                    qudt:unit unit:KiloN .
                ?collapseQuantity
                    qudt:numericValue ?collapseValue ;
                    qudt:unit unit:KiloN .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (
                        ?location = nltl:sideStructureLocation &&
                        EXISTS {
                            ?member nltl:transverseFrameOrientation true .
                        }
                    )
                )
                FILTER(
                    datatype(?strengthValue) = xsd:decimal &&
                    datatype(?collapseValue) = xsd:decimal
                )
                BIND(
                    IF(
                        ABS(?collapseValue) * 0.000001 > 0.000000000001,
                        ABS(?collapseValue) * 0.000001,
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?strengthValue - ?collapseValue) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bottom local frame requires a patch load expressed as an xsd:decimal quantity in kilonewtons." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation nltl:bottomStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:patchLoad ?patchQuantity .
                    ?patchQuantity
                        qudt:numericValue ?patchValue ;
                        qudt:unit unit:KiloN .
                    FILTER(datatype(?patchValue) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every load case of a bottom local frame must specify only the patch orientation with dimension b parallel to the frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a nltl:localFrame ;
                    nltl:structuralLocation nltl:bottomStructureLocation ;
                    nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER(
                    !EXISTS {
                        ?loadCase
                            nltl:loadPatchOrientation
                            nltl:loadPatchHeightParallelToFrame .
                    } ||
                    EXISTS {
                        ?loadCase nltl:loadPatchOrientation ?orientation .
                        FILTER(
                            ?orientation !=
                            nltl:loadPatchHeightParallelToFrame
                        )
                    }
                )
            }
        """
    ] .
