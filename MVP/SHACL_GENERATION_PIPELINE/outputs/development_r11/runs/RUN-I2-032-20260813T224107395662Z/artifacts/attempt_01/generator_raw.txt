@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .

@prefix unit: <http://qudt.org/vocab/unit/> .

    sh:class qudt:QuantityValue ;

    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_032_ShearAreaShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveMegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberShearAreaShape
    ] ;
    sh:sparql [
        sh:message "For each linked structural member, the reported required shear area must match the I2.6.2 formula within relative tolerance 1e-6 (absolute fallback 1e-9 cm2), and the actual shear area must be at least the calculated requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressureQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:peakPressureFactor ?peakFactorQuantity ;
                      nltl:shearArea ?actualAreaQuantity ;
                      nltl:span ?spanQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:frameSpacing ?spacingQuantity ;
                        nltl:selectedHullAreaFactor ?areaFactorQuantity ;
                        nltl:requiredTransverseFrameShearArea ?requiredAreaQuantity .
                ?pressureQuantity qudt:numericValue ?pressure .
                ?heightQuantity qudt:numericValue ?height .
                ?peakFactorQuantity qudt:numericValue ?peakFactor .
                ?actualAreaQuantity qudt:numericValue ?actualArea .
                ?spanQuantity qudt:numericValue ?span .
                ?yieldQuantity qudt:numericValue ?yieldStrength .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?areaFactorQuantity qudt:numericValue ?areaFactor .
                ?requiredAreaQuantity qudt:numericValue ?reportedRequiredArea .
                BIND(IF(xsd:decimal(?span) < xsd:decimal(?height),
                        xsd:decimal(?span),
                        xsd:decimal(?height)) AS ?LL)
                BIND(
                    (xsd:decimal("100") * xsd:decimal("100") *
                     xsd:decimal("0.5") * ?LL *
                     xsd:decimal(?spacing) * xsd:decimal(?areaFactor) *
                     xsd:decimal(?peakFactor) * xsd:decimal(?pressure)) /
                    (xsd:decimal("0.577") * xsd:decimal(?yieldStrength))
                    AS ?expectedRequiredArea
                )
                BIND(
                    IF(
                        ABS(?expectedRequiredArea) * xsd:decimal("0.000001") >
                        xsd:decimal("0.000000001"),
                        ABS(?expectedRequiredArea) * xsd:decimal("0.000001"),
                        xsd:decimal("0.000000001")
                    )
                    AS ?formulaTolerance
                )
                FILTER(
                    ABS(xsd:decimal(?reportedRequiredArea) - ?expectedRequiredArea) >
                        ?formulaTolerance ||
                    xsd:decimal(?actualArea) * xsd:decimal("10000") <
                        ?expectedRequiredArea
                )
            }
        """
    ] .
<END_SHACL>
