gs:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked longitudinal local frame must state whether the side-structure requirement applies." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame .
    FILTER NOT EXISTS {
        ?member nltl:sideStructureApplicability ?applicable .
        FILTER(
            datatype(?applicable) = xsd:boolean &&
            (?applicable = "true"^^xsd:boolean ||
             ?applicable = "false"^^xsd:boolean)
        )
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable longitudinal local frame must have a midspan plastic-collapse load in kilonewtons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean .
    FILTER NOT EXISTS {
        ?member nltl:midspanPlasticCollapseLoad ?collapseLoad .
        ?collapseLoad qudt:numericValue ?collapseValue ;
            qudt:unit unit:KiloN .
        FILTER(datatype(?collapseValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable longitudinal local frame must be linked to a structural-member load case." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean .
    FILTER NOT EXISTS {
        ?member nltl:hasStructuralMemberLoadCase ?loadCase .
        ?loadCase a nltl:loadCase .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each load case of an applicable longitudinal local frame must provide combined shear-and-bending demand in kilonewtons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean ;
        nltl:hasStructuralMemberLoadCase ?loadCase .
    FILTER NOT EXISTS {
        ?loadCase nltl:combinedShearAndBendingDemand ?demand .
        ?demand qudt:numericValue ?demandValue ;
            qudt:unit unit:KiloN .
        FILTER(datatype(?demandValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with an applicable longitudinal local frame must provide plastic strength in kilonewtons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean .
    FILTER NOT EXISTS {
        $this nltl:plasticStrength ?plasticStrength .
        ?plasticStrength qudt:numericValue ?strengthValue ;
            qudt:unit unit:KiloN .
        FILTER(datatype(?strengthValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must equal the applicable member's midspan plastic-collapse load within relative tolerance 1e-6 and absolute fallback 1e-9 kN." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member ;
        nltl:plasticStrength ?plasticStrength .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean ;
        nltl:midspanPlasticCollapseLoad ?collapseLoad .
    ?plasticStrength qudt:numericValue ?strengthValue ;
        qudt:unit unit:KiloN .
    ?collapseLoad qudt:numericValue ?collapseValue ;
        qudt:unit unit:KiloN .
    FILTER(
        datatype(?strengthValue) = xsd:decimal &&
        datatype(?collapseValue) = xsd:decimal
    )
    FILTER(
        ABS(?strengthValue - ?collapseValue) >
        IF(
            ABS(?collapseValue) * 0.000001 > 0.000000001,
            ABS(?collapseValue) * 0.000001,
            0.000000001
        )
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combined shear-and-bending demand must not exceed plastic strength." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member ;
        nltl:plasticStrength ?plasticStrength .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability "true"^^xsd:boolean ;
        nltl:hasStructuralMemberLoadCase ?loadCase .
    ?loadCase nltl:combinedShearAndBendingDemand ?demand .
    ?demand qudt:numericValue ?demandValue ;
        qudt:unit unit:KiloN .
    ?plasticStrength qudt:numericValue ?strengthValue ;
        qudt:unit unit:KiloN .
    FILTER(
        datatype(?demandValue) = xsd:decimal &&
        datatype(?strengthValue) = xsd:decimal &&
        ?demandValue > ?strengthValue
    )
}
"""
    ] .
