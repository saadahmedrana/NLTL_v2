<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_040_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:I2_040_YieldStrengthShape ;
                sh:message "Yield strength in MPa is required when structural members are assessed."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:node gen:I2_040_StructuralMemberShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The structural member web height to net web thickness ratio exceeds the limit for its section type and yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?yieldQuantity qudt:numericValue ?yield .
                ?member nltl:structuralSectionType ?section ;
                        nltl:webHeight ?heightQuantity ;
                        nltl:netWebThickness ?thicknessQuantity .
                ?heightQuantity qudt:numericValue ?height .
                ?thicknessQuantity qudt:numericValue ?thickness .
                FILTER (
                    ?section IN (
                        nltl:flatBarStructuralSection,
                        nltl:bulbSection,
                        nltl:teeSection,
                        nltl:angleSection
                    )
                )
                BIND(
                    IF(
                        ?section = nltl:flatBarStructuralSection,
                        "79524"^^xsd:decimal,
                        "648025"^^xsd:decimal
                    )
                    AS ?coefficientSquared
                )
                FILTER (
                    (?height * ?height * ?yield) >
                    (?coefficientSquared * ?thickness * ?thickness)
                )
            }
        """
    ] .

gen:I2_040_YieldStrengthShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive "0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_040_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralSectionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:flatBarStructuralSection
            nltl:bulbSection
            nltl:teeSection
            nltl:angleSection
        ) ;
        sh:message "Structural section type must identify a flat-bar, bulb, tee, or angle section."
    ] ;
    sh:property [
        sh:path nltl:webHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_040_PositiveMillimetreShape ;
        sh:message "Web height must be a positive decimal quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_040_PositiveMillimetreShape ;
        sh:message "Net web thickness must be a positive decimal quantity in millimetres."
    ] .

gen:I2_040_PositiveMillimetreShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minExclusive "0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
