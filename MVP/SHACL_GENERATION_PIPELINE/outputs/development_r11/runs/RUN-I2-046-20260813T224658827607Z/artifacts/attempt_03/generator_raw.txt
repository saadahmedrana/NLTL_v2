<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_046_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasInternalIceStrengthenedStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:internalStructure ;
        sh:message "Every identified internal ice-strengthened member must be an internal structure."
    ] ;
    sh:sparql [
        sh:message "A Polar Class ship must identify its internal ice-strengthened structures." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass .
                FILTER NOT EXISTS {
                    $this nltl:hasInternalIceStrengthenedStructure ?member .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each plated internal ice-strengthened member must have a corrosion/abrasion allowance of at least 1.0 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:hasInternalIceStrengthenedStructure ?member .
                ?member a nltl:plating .
                FILTER NOT EXISTS {
                    ?member nltl:corrosionAbrasionAllowance ?allowance .
                    ?allowance qudt:numericValue ?value ;
                               qudt:unit unit:MilliM .
                    FILTER (
                        DATATYPE(?value) = xsd:decimal &&
                        ?value >= 1.0
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each reported corrosion/abrasion allowance for a plated internal ice-strengthened member must be a decimal value of at least 1.0 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member ?allowance
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:hasInternalIceStrengthenedStructure ?member .
                ?member a nltl:plating ;
                        nltl:corrosionAbrasionAllowance ?allowance .
                FILTER NOT EXISTS {
                    ?allowance qudt:numericValue ?value ;
                               qudt:unit unit:MilliM .
                    FILTER (
                        DATATYPE(?value) = xsd:decimal &&
                        ?value >= 1.0
                    )
                }
            }
        """
    ] .
<END_SHACL>
