gs:I2_054_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:AngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasHullGirderBendingMomentCalculationCase ;
        sh:minCount 1 ;
        sh:node gs:I2_054_CaseShape
    ] ;
    sh:sparql [
        sh:message "Each hull-girder bending-moment case must also be linked as the ship's calculation case so that its interpolation evidence belongs to the calculation being checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasHullGirderBendingMomentCalculationCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The case position and distribution factor must follow the stated piecewise-linear envelope, and the reported bending moment must satisfy the formula. Derived comparisons use relative tolerance 1e-6 with absolute fallback 1e-9; the equivalent fifth-power comparison is used for the -0.2 exponent." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this ?case
            WHERE {
                $this nltl:upperIceWaterlineLengthLUI ?lq ;
                      nltl:stemAngle ?gq ;
                      nltl:designVerticalIceForceAtBow ?fq ;
                      nltl:hasHullGirderBendingMomentCalculationCase ?case ;
                      nltl:hasCalculationCase ?case .
                ?lq qudt:numericValue ?l ;
                    qudt:unit unit:M .
                ?gq qudt:numericValue ?gamma ;
                    qudt:unit unit:DEG .
                ?fq qudt:numericValue ?force ;
                    qudt:unit unit:MegaN .
                ?case nltl:caseHullGirderLongitudinalPositionFromAft ?xq ;
                      nltl:caseBendingMomentDistributionFactor ?cmq ;
                      nltl:caseDesignVerticalIceBendingMoment ?miq .
                ?xq qudt:numericValue ?x ;
                    qudt:unit unit:M .
                ?cmq qudt:numericValue ?cm ;
                     qudt:unit unit:UNITLESS .
                ?miq qudt:numericValue ?mi ;
                     qudt:unit unit:MegaN-M .

                BIND(xsd:double(?x) / xsd:double(?l) AS ?r)
                BIND(
                    IF(?r <= 0.5e0,
                       2.0e0 * ?r,
                       IF(?r <= 0.7e0,
                          1.0e0,
                          IF(?r <= 0.95e0,
                             2.96e0 - 2.8e0 * ?r,
                             6.0e0 - 6.0e0 * ?r)))
                    AS ?expectedCm
                )
                BIND(
                    IF(ABS(?expectedCm) > 1.0e-9,
                       ABS(?expectedCm) * 1.0e-6,
                       1.0e-9)
                    AS ?cmTolerance
                )
                BIND(
                    math:sin(
                        xsd:double(?gamma) *
                        3.141592653589793e0 /
                        180.0e0
                    )
                    AS ?sinGamma
                )
                BIND(
                    0.1e0 *
                    ?expectedCm *
                    xsd:double(?l) *
                    xsd:double(?force)
                    AS ?momentBase
                )
                BIND(
                    ?momentBase * ?momentBase * ?momentBase *
                    ?momentBase * ?momentBase
                    AS ?expectedPower
                )
                BIND(
                    xsd:double(?mi) * xsd:double(?mi) *
                    xsd:double(?mi) * xsd:double(?mi) *
                    xsd:double(?mi) * ?sinGamma
                    AS ?reportedPower
                )
                BIND(
                    IF(ABS(?expectedPower) > 1.0e-9,
                       ABS(?expectedPower) * 1.0e-6,
                       1.0e-9)
                    AS ?powerTolerance
                )

                FILTER(
                    xsd:double(?l) <= 0.0e0 ||
                    xsd:double(?x) < 0.0e0 ||
                    xsd:double(?x) > xsd:double(?l) ||
                    ?sinGamma <= 0.0e0 ||
                    ABS(xsd:double(?cm) - ?expectedCm) > ?cmTolerance ||
                    IF(
                        ABS(?momentBase) <= 1.0e-9,
                        ABS(xsd:double(?mi)) > 1.0e-9,
                        ABS(?reportedPower - ?expectedPower) > ?powerTolerance
                    )
                )
            }
        """
    ] .
