gen:I2_054_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositiveLengthQuantityShape ;
        sh:message "Upper ice waterline length L_UI must be provided in metres."
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AngleQuantityShape ;
        sh:message "Stem angle must be provided in degrees."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceQuantityShape ;
        sh:message "Design vertical ice force at bow must be provided in meganewtons."
    ] ;
    sh:property [
        sh:path nltl:hasHullGirderBendingMomentCalculationCase ;
        sh:minCount 1 ;
        sh:class nltl:hullGirderBendingMomentCalculationCase ;
        sh:node gen:I2_054_CalculationCaseShape ;
        sh:message "At least one complete hull-girder bending-moment calculation case is required."
    ] ;
    sh:sparql [
        sh:message "Each hull-girder bending-moment calculation case must also be linked as the ship's calculation case so its interpolation evidence is associated with that calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasHullGirderBendingMomentCalculationCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasCalculationCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The case bending-moment distribution factor must follow the prescribed piecewise-linear envelope, using relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case
            WHERE {
                $this nltl:upperIceWaterlineLengthLUI ?lengthQuantity ;
                      nltl:hasHullGirderBendingMomentCalculationCase ?case .
                ?lengthQuantity qudt:numericValue ?length .
                ?case nltl:caseHullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:caseBendingMomentDistributionFactor ?factorQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?factorQuantity qudt:numericValue ?factor .
                BIND (?position / ?length AS ?coordinate)
                BIND (
                    IF(
                        ?coordinate <= 0.5,
                        2.0 * ?coordinate,
                        IF(
                            ?coordinate <= 0.7,
                            1.0,
                            IF(
                                ?coordinate <= 0.95,
                                2.96 - (2.8 * ?coordinate),
                                6.0 - (6.0 * ?coordinate)
                            )
                        )
                    )
                    AS ?expectedFactor
                )
                BIND (
                    IF(
                        ABS(?expectedFactor) * 0.000001 > 0.000000001,
                        ABS(?expectedFactor) * 0.000001,
                        0.000000001
                    )
                    AS ?factorTolerance
                )
                FILTER (
                    ?coordinate < 0.0 ||
                    ?coordinate > 1.0 ||
                    ABS(?factor - ?expectedFactor) > ?factorTolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Case M_I must equal 0.1*C_m*L_UI*sin(gamma_stem)^(-0.2)*F_IB in MN m, within relative tolerance 1e-6 and absolute fallback 1e-9 MN m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this ?case
            WHERE {
                $this nltl:upperIceWaterlineLengthLUI ?lengthQuantity ;
                      nltl:stemAngle ?angleQuantity ;
                      nltl:designVerticalIceForceAtBow ?forceQuantity ;
                      nltl:hasHullGirderBendingMomentCalculationCase ?case .
                ?lengthQuantity qudt:numericValue ?length .
                ?angleQuantity qudt:numericValue ?angleDegrees .
                ?forceQuantity qudt:numericValue ?force .
                ?case nltl:caseBendingMomentDistributionFactor ?factorQuantity ;
                      nltl:caseDesignVerticalIceBendingMoment ?momentQuantity .
                ?factorQuantity qudt:numericValue ?factor .
                ?momentQuantity qudt:numericValue ?moment .
                BIND (math:sin(?angleDegrees * 0.017453292519943295) AS ?sine)
                BIND (0.1 * ?factor * ?length * ?force AS ?baseMoment)
                BIND (
                    ?moment * ?moment * ?moment * ?moment * ?moment * ?sine
                    AS ?actualPower
                )
                BIND (
                    ?baseMoment * ?baseMoment * ?baseMoment * ?baseMoment * ?baseMoment
                    AS ?expectedPower
                )
                BIND (
                    IF(
                        ?expectedPower != 0.0,
                        ?actualPower / ?expectedPower,
                        1.0
                    )
                    AS ?powerRatio
                )
                BIND (
                    (1.0 - 0.000001) * (1.0 - 0.000001) *
                    (1.0 - 0.000001) * (1.0 - 0.000001) *
                    (1.0 - 0.000001)
                    AS ?lowerPowerRatio
                )
                BIND (
                    (1.0 + 0.000001) * (1.0 + 0.000001) *
                    (1.0 + 0.000001) * (1.0 + 0.000001) *
                    (1.0 + 0.000001)
                    AS ?upperPowerRatio
                )
                FILTER (
                    ?sine <= 0.0 ||
                    (
                        ?expectedPower = 0.0 &&
                        ABS(?moment) > 0.000000001
                    ) ||
                    (
                        ?expectedPower != 0.0 &&
                        (
                            ?powerRatio < ?lowerPowerRatio ||
                            ?powerRatio > ?upperPowerRatio
                        )
                    )
                )
            }
        """
    ] .
