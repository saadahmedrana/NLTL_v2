<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_065_CalculationCaseMethodShape
    ] ;
    sh:sparql [
        sh:message "Each nonlinear calculation case requires acceptance evidence, dimensionless fracture, major-buckling and yielding margins, permanent deformation measurements, a relevant structural dimension, and a satisfied minor-deformation assessment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
                FILTER (
                    NOT EXISTS {
                        ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
                        ?evidence a nltl:evidenceArtifact .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:fractureMargin ?fractureQuantity .
                        ?fractureQuantity
                            qudt:numericValue ?fractureValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?fractureValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:majorBucklingMargin ?bucklingQuantity .
                        ?bucklingQuantity
                            qudt:numericValue ?bucklingValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?bucklingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldingMargin ?yieldingQuantity .
                        ?yieldingQuantity
                            qudt:numericValue ?yieldingValue ;
                            qudt:unit unit:UNITLESS .
                        FILTER (DATATYPE(?yieldingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentLateralDeformation ?lateralQuantity .
                        ?lateralQuantity
                            qudt:numericValue ?lateralValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?lateralValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:permanentOutOfPlaneDeformation ?outOfPlaneQuantity .
                        ?outOfPlaneQuantity
                            qudt:numericValue ?outOfPlaneValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?outOfPlaneValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:relevantStructuralDimension ?dimensionQuantity .
                        ?dimensionQuantity
                            qudt:numericValue ?dimensionValue ;
                            qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?dimensionValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:minorPermanentDeformationAssessmentSatisfied ?assessment .
                        FILTER (
                            DATATYPE(?assessment) = xsd:boolean
                            &&
                            ?assessment = "true"^^xsd:boolean
                        )
                    }
                )
            }
        """
    ] .

gen:I2_065_CalculationCaseMethodShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Each calculation case requires a calculation method to determine applicability."
    ] .
<END_SHACL>
