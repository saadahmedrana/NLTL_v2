<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_065_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_065_CalculationCaseSelectorShape ;
        sh:message "Each linked calculation case must identify a typed calculation method."
    ] .

gen:I2_065_CalculationCaseSelectorShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationMethodValue ;
        sh:message "The calculation method must be one calculationMethodValue instance."
    ] .

gen:I2_065_NonlinearCalculationCaseShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasCalculationCase ?this .
                ?this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
            }
        """
    ] ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:hasValue nltl:nonlinearCalculationMethodValue
    ] ;
    sh:property [
        sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "A nonlinear calculation case must provide nonlinear analysis acceptance evidence."
    ] ;
    sh:property [
        sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:message "The nonlinear calculation case must record a satisfied minor permanent deformation assessment."
    ] ;
    sh:property [
        sh:path nltl:permanentLateralDeformation ;
        sh:minCount 1 ;
        sh:node gen:I2_065_MillimetreQuantityShape ;
        sh:message "Permanent lateral deformation must be supplied in millimetres."
    ] ;
    sh:property [
        sh:path nltl:permanentOutOfPlaneDeformation ;
        sh:minCount 1 ;
        sh:node gen:I2_065_MillimetreQuantityShape ;
        sh:message "Permanent out-of-plane deformation must be supplied in millimetres."
    ] ;
    sh:property [
        sh:path nltl:relevantStructuralDimension ;
        sh:minCount 1 ;
        sh:node gen:I2_065_MillimetreQuantityShape ;
        sh:message "The relevant structural dimension must be supplied in millimetres."
    ] .

gen:I2_065_NonlinearShipMarginShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:fractureMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_UnitlessQuantityShape ;
        sh:message "A fracture margin is required for nonlinear evaluation."
    ] ;
    sh:property [
        sh:path nltl:majorBucklingMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_UnitlessQuantityShape ;
        sh:message "A major buckling margin is required for nonlinear evaluation."
    ] ;
    sh:property [
        sh:path nltl:yieldingMargin ;
        sh:minCount 1 ;
        sh:node gen:I2_065_UnitlessQuantityShape ;
        sh:message "A yielding margin is required for nonlinear evaluation."
    ] .

gen:I2_065_MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:MilliM )
    ] .

gen:I2_065_UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:UNITLESS )
    ] .
<END_SHACL>
