<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PolarCertificateScheduleConsistencyShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:minCount 1 ;
        sh:node gen:SolasCertificateScheduleShape ;
        sh:message "An applicable SOLAS certificate schedule is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The Polar Ship Certificate supplement shall be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:minCount 1 ;
        sh:node gen:PolarCertificateSupplementShape ;
        sh:message "A Polar Ship Certificate supplement with required equipment records is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each Polar Ship Certificate validity, survey, or endorsement date shall match a date in every linked applicable SOLAS certificate schedule." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?polarProperty ?polarDate ?schedule
WHERE {
    $this nltl:hasSolasCertificateSchedule ?schedule .
    $this ?polarProperty ?polarDate .
    FILTER (
        ?polarProperty IN (
            nltl:polarCertificateValidityDates,
            nltl:surveyDates,
            nltl:endorsementDates
        )
    )
    FILTER NOT EXISTS {
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDate ?scheduleDate .
        FILTER (
            YEAR(?polarDate) = YEAR(?scheduleDate) &&
            MONTH(?polarDate) = MONTH(?scheduleDate) &&
            DAY(?polarDate) = DAY(?scheduleDate)
        )
    }
}
"""
    ] .

gen:SolasCertificateScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:node gen:CertificateScheduleDateRecordShape ;
        sh:message "The SOLAS certificate schedule requires certificate schedule date records."
    ] .

gen:CertificateScheduleDateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A certificate schedule date record requires a date."
    ] .

gen:PolarCertificateSupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate supplement shall include required equipment records."
    ] .
<END_SHACL>
