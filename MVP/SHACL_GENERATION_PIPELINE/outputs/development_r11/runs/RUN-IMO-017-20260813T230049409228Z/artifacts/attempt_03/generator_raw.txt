<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasCertificateScheduleShape
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The Polar Ship Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017PolarCertificateSupplementShape ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate must include a supplement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each Polar Certificate validity, survey, or endorsement date must match a SOLAS schedule record of the corresponding date type." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this ?polarProperty ?polarDate
WHERE {
    $this ?polarProperty ?polarDate .
    FILTER (
        ?polarProperty IN (
            nltl:polarCertificateValidityDates,
            nltl:surveyDates,
            nltl:endorsementDates
        )
    )
    FILTER (datatype(?polarDate) = xsd:date)
    FILTER NOT EXISTS {
        $this nltl:hasSolasCertificateSchedule ?schedule .
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDateType ?dateType ;
                nltl:certificateScheduleDate ?scheduleDate .
        FILTER (
            (?polarProperty = nltl:polarCertificateValidityDates &&
             ?dateType = "validity" &&
             ?scheduleDate = ?polarDate)
            ||
            (?polarProperty = nltl:surveyDates &&
             ?dateType = "survey" &&
             ?scheduleDate = ?polarDate)
            ||
            (?polarProperty = nltl:endorsementDates &&
             ?dateType = "endorsement" &&
             ?scheduleDate = ?polarDate)
        )
    }
}
"""
    ] .

gen:IMO017SolasCertificateScheduleShape
    a sh:NodeShape ;
    sh:class nltl:solasCertificateSchedule ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017CertificateScheduleDateRecordShape
    ] .

gen:IMO017CertificateScheduleDateRecordShape
    a sh:NodeShape ;
    sh:class nltl:certificateScheduleDateRecord ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:datatype xsd:string ;
        sh:in ( "validity" "survey" "endorsement" )
    ] .

gen:IMO017PolarCertificateSupplementShape
    a sh:NodeShape ;
    sh:class nltl:polarCertificateSupplement ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate supplement must contain the required equipment records."
    ] .
<END_SHACL>
