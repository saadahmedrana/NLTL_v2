<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:KilonewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1
    ] .

gen:MegapascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1
    ] .

gen:YearQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/YR> ;
        sh:minCount 1
    ] .

gen:NormalDesignConditionShape
    a sh:NodeShape ;
    sh:class nltl:designCondition ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:designConditionExpectedLoad ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:designConditionLocalStrengthCapacity ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1
    ] .

gen:ThrusterApplicabilityReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Main-propulsion-thruster status is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Thruster type is required to determine applicability."
    ] .

gen:MainPropulsionThrusterDesignShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:mainPropulsionThruster ?main ;
                      nltl:thrusterType ?type .
                FILTER (
                    ?main = true &&
                    LCASE(STR(?type)) IN ("azimuthing", "fixed")
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:extremeLoad ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:localIcePressure ;
        sh:node gen:MegapascalQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:serviceLife ;
        sh:node gen:YearQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:qualifiedValueShape gen:NormalDesignConditionShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "A normal operational design condition with expected load and local-strength capacity is required."
    ] ;
    sh:sparql [
        sh:message "The normal-condition expected load must match the thruster-body ice-interaction load within tolerance, and local-strength capacity must not be less than either that expected load or the extreme load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:extremeLoad ?extremeQuantity ;
                      nltl:localIcePressure ?pressureQuantity ;
                      nltl:propellerIceInteractionLoad ?propellerQuantity ;
                      nltl:serviceLife ?serviceLifeQuantity ;
                      nltl:thrusterBodyIceInteractionLoad ?bodyQuantity ;
                      nltl:hasDesignCondition ?condition .

                ?extremeQuantity qudt:numericValue ?extreme ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?pressureQuantity qudt:numericValue ?pressure ;
                                  qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?propellerQuantity qudt:numericValue ?propeller ;
                                   qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?serviceLifeQuantity qudt:numericValue ?serviceLife ;
                                     qudt:unit <http://qudt.org/vocab/unit/YR> .
                ?bodyQuantity qudt:numericValue ?body ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                ?condition nltl:normalOperationalConditions true ;
                           nltl:designConditionExpectedLoad ?expectedQuantity ;
                           nltl:designConditionLocalStrengthCapacity ?capacityQuantity .

                ?expectedQuantity qudt:numericValue ?expected ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?capacityQuantity qudt:numericValue ?capacity ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                FILTER (
                    ABS(?expected - ?body) >
                        IF(
                            ABS(?body) * 0.000001 > 0.000000001,
                            ABS(?body) * 0.000001,
                            0.000000001
                        )
                    ||
                    ?capacity < ?expected
                    ||
                    ?capacity < ?extreme
                )
            }
        """
    ] .
<END_SHACL>
