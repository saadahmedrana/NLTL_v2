gs:ThrusterApplicabilityReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        sh:message "A main-propulsion thruster requires a thruster type to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:mainPropulsionThruster ?main .
                FILTER (?main = true)
                FILTER NOT EXISTS {
                    $this nltl:thrusterType ?type .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An azimuthing or fixed thruster requires its main-propulsion status to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterType ?type .
                FILTER (STR(?type) IN ("azimuthing", "fixed"))
                FILTER NOT EXISTS {
                    $this nltl:mainPropulsionThruster ?main .
                }
            }
        """
    ] .
