gen:I2_014Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:BowLengthValueShape ;
        sh:message "The ship must report one bow waterline length in metres."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The bow subregion count must be 4."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:node gen:BowSubregionCalculationCaseShape ;
        sh:message "The ship must have exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:ForceValueShape ;
        sh:message "The ship must report the selected maximum bow force."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LineLoadValueShape ;
        sh:message "The ship must report the selected maximum bow line load."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureValueShape ;
        sh:message "The ship must report the selected maximum bow pressure."
    ] ;
    sh:sparql [
        sh:message "The four bow subregion midpoint positions must correspond to 1/8, 3/8, 5/8 and 7/8 of the bow length within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                FILTER (
                    !EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case1 .
                        ?case1 nltl:bowSubregionMidLengthPosition ?positionQuantity1 .
                        ?positionQuantity1 qudt:numericValue ?position1 .
                        BIND((?length / "8"^^xsd:decimal) AS ?expected1)
                        BIND(IF(ABS(?expected1) > "0.001"^^xsd:decimal,
                                ABS(?expected1) * "0.000001"^^xsd:decimal,
                                "0.000000001"^^xsd:decimal) AS ?tolerance1)
                        FILTER(ABS(?position1 - ?expected1) <= ?tolerance1)
                    }
                    ||
                    !EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case2 .
                        ?case2 nltl:bowSubregionMidLengthPosition ?positionQuantity2 .
                        ?positionQuantity2 qudt:numericValue ?position2 .
                        BIND((("3"^^xsd:decimal * ?length) / "8"^^xsd:decimal) AS ?expected2)
                        BIND(IF(ABS(?expected2) > "0.001"^^xsd:decimal,
                                ABS(?expected2) * "0.000001"^^xsd:decimal,
                                "0.000000001"^^xsd:decimal) AS ?tolerance2)
                        FILTER(ABS(?position2 - ?expected2) <= ?tolerance2)
                    }
                    ||
                    !EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case3 .
                        ?case3 nltl:bowSubregionMidLengthPosition ?positionQuantity3 .
                        ?positionQuantity3 qudt:numericValue ?position3 .
                        BIND((("5"^^xsd:decimal * ?length) / "8"^^xsd:decimal) AS ?expected3)
                        BIND(IF(ABS(?expected3) > "0.001"^^xsd:decimal,
                                ABS(?expected3) * "0.000001"^^xsd:decimal,
                                "0.000000001"^^xsd:decimal) AS ?tolerance3)
                        FILTER(ABS(?position3 - ?expected3) <= ?tolerance3)
                    }
                    ||
                    !EXISTS {
                        $this nltl:hasBowSubregionCalculationCase ?case4 .
                        ?case4 nltl:bowSubregionMidLengthPosition ?positionQuantity4 .
                        ?positionQuantity4 qudt:numericValue ?position4 .
                        BIND((("7"^^xsd:decimal * ?length) / "8"^^xsd:decimal) AS ?expected4)
                        BIND(IF(ABS(?expected4) > "0.001"^^xsd:decimal,
                                ABS(?expected4) * "0.000001"^^xsd:decimal,
                                "0.000000001"^^xsd:decimal) AS ?tolerance4)
                        FILTER(ABS(?position4 - ?expected4) <= ?tolerance4)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected bow force, line load and pressure must equal the respective maxima from the four subregion cases within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForceQuantity ;
                      nltl:selectedMaximumBowLineLoad ?selectedLineLoadQuantity ;
                      nltl:selectedMaximumBowPressure ?selectedPressureQuantity .
                ?selectedForceQuantity qudt:numericValue ?selectedForce .
                ?selectedLineLoadQuantity qudt:numericValue ?selectedLineLoad .
                ?selectedPressureQuantity qudt:numericValue ?selectedPressure .
                {
                    SELECT $this
                           (MAX(?force) AS ?maximumForce)
                           (MAX(?lineLoad) AS ?maximumLineLoad)
                           (MAX(?pressure) AS ?maximumPressure)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?forceQuantity ;
                              nltl:bowSubregionCalculatedLineLoad ?lineLoadQuantity ;
                              nltl:bowSubregionCalculatedPressure ?pressureQuantity .
                        ?forceQuantity qudt:numericValue ?force .
                        ?lineLoadQuantity qudt:numericValue ?lineLoad .
                        ?pressureQuantity qudt:numericValue ?pressure .
                    }
                    GROUP BY $this
                }
                BIND(IF(ABS(?maximumForce) > "0.001"^^xsd:decimal,
                        ABS(?maximumForce) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal) AS ?forceTolerance)
                BIND(IF(ABS(?maximumLineLoad) > "0.001"^^xsd:decimal,
                        ABS(?maximumLineLoad) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal) AS ?lineLoadTolerance)
                BIND(IF(ABS(?maximumPressure) > "0.001"^^xsd:decimal,
                        ABS(?maximumPressure) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal) AS ?pressureTolerance)
                FILTER(
                    ABS(?selectedForce - ?maximumForce) > ?forceTolerance
                    ||
                    ABS(?selectedLineLoad - ?maximumLineLoad) > ?lineLoadTolerance
                    ||
                    ABS(?selectedPressure - ?maximumPressure) > ?pressureTolerance
                )
            }
        """
    ] .
