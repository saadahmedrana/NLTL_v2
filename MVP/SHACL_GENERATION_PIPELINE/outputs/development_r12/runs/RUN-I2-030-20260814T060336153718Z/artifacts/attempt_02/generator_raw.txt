gen:I2_030Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:StructuralMemberCaseLinkShape
    ] ;
    sh:sparql [
        sh:message "Each local-frame calculation case must be linked to the same ship, structural member, and plating." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case .
                ?case nltl:sectionCalculationCaseStructuralMember ?declaredMember ;
                      nltl:sectionCalculationCasePlating ?plating .
                FILTER (
                    ?declaredMember != ?member ||
                    !EXISTS { $this nltl:hasCalculationCase ?case . } ||
                    !EXISTS { $this nltl:hasPlating ?plating . }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable local-frame calculation requires the formula operands and reported results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                        nltl:netLocalFrameFlangeArea/qudt:numericValue ?afn ;
                        nltl:netAttachedShellPlateThickness/qudt:numericValue ?tpn ;
                        nltl:frameSpacing/qudt:numericValue ?spacing .
                ?case nltl:sectionCalculationCaseStructuralMember ?member .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                FILTER (
                    !EXISTS { ?member nltl:webHeight ?webHeight . } ||
                    !EXISTS { ?member nltl:netWebThickness ?webThickness . } ||
                    !EXISTS { ?member nltl:localFrameFlangeCentreHeight ?flangeHeight . } ||
                    !EXISTS { ?member nltl:webAngleToShellPlate ?webAngle . } ||
                    !EXISTS { ?member nltl:webToFlangeCentreDistance ?webDistance . } ||
                    !EXISTS { ?member nltl:netEffectivePlasticSectionModulus ?sectionModulus . } ||
                    !EXISTS { ?case nltl:plasticNeutralAxisHeight ?neutralAxis . }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Oblique framing between 20 and 70 degrees requires linked 20-degree and 70-degree interpolation endpoints." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                        nltl:netLocalFrameFlangeArea/qudt:numericValue ?afn ;
                        nltl:netAttachedShellPlateThickness/qudt:numericValue ?tpn ;
                        nltl:frameSpacing/qudt:numericValue ?spacing .
                ?case nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega/qudt:numericValue ?omega .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                FILTER (?omega > 20 && ?omega < 70)
                FILTER (
                    !EXISTS { ?case nltl:interpolationLowerEndpoint ?lower . } ||
                    !EXISTS { ?case nltl:interpolationUpperEndpoint ?upper . }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The plastic neutral-axis height does not match the prescribed formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                        nltl:netLocalFrameFlangeArea/qudt:numericValue ?afn ;
                        nltl:webHeight/qudt:numericValue ?hw ;
                        nltl:netWebThickness/qudt:numericValue ?twn ;
                        nltl:netAttachedShellPlateThickness/qudt:numericValue ?tpn ;
                        nltl:frameSpacing/qudt:numericValue ?spacing .
                ?case nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:plasticNeutralAxisHeight/qudt:numericValue ?reportedZna .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                BIND (((100 * ?afn + ?hw * ?twn - 1000 * ?tpn * ?spacing) / (2 * ?twn)) AS ?expectedZna)
                BIND (IF(ABS(?expectedZna) > 0.001,
                         ABS(?expectedZna) * 0.000001,
                         0.000000001) AS ?tolerance)
                FILTER (ABS(?reportedZna - ?expectedZna) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The net effective plastic section modulus does not match the prescribed direct formula." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                        nltl:netLocalFrameFlangeArea/qudt:numericValue ?afn ;
                        nltl:webHeight/qudt:numericValue ?hw ;
                        nltl:netWebThickness/qudt:numericValue ?twn ;
                        nltl:netAttachedShellPlateThickness/qudt:numericValue ?tpn ;
                        nltl:frameSpacing/qudt:numericValue ?spacing ;
                        nltl:localFrameFlangeCentreHeight/qudt:numericValue ?hfc ;
                        nltl:webAngleToShellPlate/qudt:numericValue ?phi ;
                        nltl:webToFlangeCentreDistance/qudt:numericValue ?bw ;
                        nltl:netEffectivePlasticSectionModulus/qudt:numericValue ?reportedZp .
                ?case nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega/qudt:numericValue ?omega .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                FILTER (?omega <= 20 || ?omega >= 70)
                BIND (((100 * ?afn + ?hw * ?twn - 1000 * ?tpn * ?spacing) / (2 * ?twn)) AS ?zna)
                BIND ((?phi * 3.141592653589793 / 180) AS ?phiRadians)
                BIND (
                    ?tpn * ?spacing * (?zna + ?tpn / 2) * math:sin(?phiRadians)
                    + (((?hw - ?zna) * (?hw - ?zna) + ?zna * ?zna)
                       * ?twn * math:sin(?phiRadians)) / 2000
                    + ?afn * ((?hfc - ?zna) * math:sin(?phiRadians)
                              - ?bw * math:cos(?phiRadians)) / 10
                    AS ?expectedZp
                )
                BIND (IF(ABS(?expectedZp) > 0.001,
                         ABS(?expectedZp) * 0.000001,
                         0.000000001) AS ?tolerance)
                FILTER (ABS(?reportedZp - ?expectedZp) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "For oblique framing, the net effective plastic section modulus must equal the linear interpolation between the 20-degree and 70-degree endpoint results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:hasCalculationCase ?case .
                ?member nltl:hasLocalFrameSectionCalculationCase ?case ;
                        nltl:netLocalFrameFlangeArea/qudt:numericValue ?afn ;
                        nltl:netAttachedShellPlateThickness/qudt:numericValue ?tpn ;
                        nltl:frameSpacing/qudt:numericValue ?spacing ;
                        nltl:netEffectivePlasticSectionModulus/qudt:numericValue ?reportedZp .
                ?case nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating ;
                      nltl:interpolationLowerEndpoint ?lower ;
                      nltl:interpolationUpperEndpoint ?upper .
                ?plating nltl:framingAngleOmega/qudt:numericValue ?omega .
                ?lower nltl:interpolationPointCoordinate/qudt:numericValue ?lowerCoordinate ;
                       nltl:interpolationPointResult/qudt:numericValue ?lowerResult .
                ?upper nltl:interpolationPointCoordinate/qudt:numericValue ?upperCoordinate ;
                       nltl:interpolationPointResult/qudt:numericValue ?upperResult .
                FILTER ((100 * ?afn) > (1000 * ?tpn * ?spacing))
                FILTER (?omega > 20 && ?omega < 70)
                FILTER (?lowerCoordinate >= 20 && ?lowerCoordinate <= 20)
                FILTER (?upperCoordinate >= 70 && ?upperCoordinate <= 70)
                BIND (
                    ?lowerResult
                    + ((?omega - ?lowerCoordinate)
                       / (?upperCoordinate - ?lowerCoordinate))
                      * (?upperResult - ?lowerResult)
                    AS ?expectedZp
                )
                BIND (IF(ABS(?expectedZp) > 0.001,
                         ABS(?expectedZp) * 0.000001,
                         0.000000001) AS ?tolerance)
                FILTER (ABS(?reportedZp - ?expectedZp) > ?tolerance)
            }
        """
    ] .
