        a sh:SPARQLConstraint ;
        sh:message "Plastic strength must equal the applicable member's midspan plastic collapse load within relative tolerance 1e-6 and absolute fallback 1e-12 kN." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member ;
          nltl:plasticStrength ?strength .
    ?member nltl:longitudinalFrameOrientation true ;
            nltl:structuralLocation nltl:sideStructureLocation ;
            nltl:midspanPlasticCollapseLoad ?collapseLoad .
    ?strength qudt:numericValue ?strengthValue ;
              qudt:unit unit:KiloN .
    ?collapseLoad qudt:numericValue ?collapseValue ;
                  qudt:unit unit:KiloN .
    FILTER(
        datatype(?strengthValue) = xsd:decimal &&
        datatype(?collapseValue) = xsd:decimal
    )
    BIND(ABS(?collapseValue) * 0.000001 AS ?relativeTolerance)
    BIND(
        IF(
            ?relativeTolerance > 0.000000000001,
            ?relativeTolerance,
            0.000000000001
        ) AS ?tolerance
    )
    FILTER(ABS(?strengthValue - ?collapseValue) > ?tolerance)
}
"""
    ] ;
