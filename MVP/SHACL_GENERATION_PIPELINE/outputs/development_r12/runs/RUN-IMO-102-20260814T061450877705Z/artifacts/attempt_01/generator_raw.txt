<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PolarTrainingRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:class nltl:iceConditionValue ;
        sh:message "Ice condition classification is required to determine polar training applicability."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Ship type is required to determine the applicable polar training level."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gen:CrewMemberTrainingInputShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In open waters, a master, chief mate, or officer in charge of a navigational watch on a tanker or passenger ship must have a valid matching basic polar training record." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?crew
WHERE {
    $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
          nltl:shipType ?shipType ;
          nltl:hasCrewMemberInventory ?crew .
    ?crew nltl:crewRole ?role .
    FILTER(datatype(?shipType) = xsd:string)
    FILTER(datatype(?role) = xsd:string)
    FILTER(LCASE(STR(?shipType)) IN ("tankers", "passenger ships"))
    FILTER(LCASE(STR(?role)) IN (
        "master",
        "chief mate",
        "officers in charge of a navigational watch"
    ))
    FILTER NOT EXISTS {
        ?crew (nltl:hasPolarTrainingRecord|nltl:hasCrewTrainingRecord) ?record .
        ?record a nltl:polarTrainingRecord ;
                nltl:requiredTrainingLevel nltl:basicPolarTrainingLevel ;
                nltl:trainingRecordLevel nltl:basicPolarTrainingLevel ;
                nltl:stcwQualificationValid ?valid .
        FILTER(?valid = true)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In open waters, the required training level for the specified bridge personnel on other ships must be not applicable." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?crew ?record
WHERE {
    $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
          nltl:shipType ?shipType ;
          nltl:hasCrewMemberInventory ?crew .
    ?crew nltl:crewRole ?role ;
          (nltl:hasPolarTrainingRecord|nltl:hasCrewTrainingRecord) ?record .
    ?record nltl:requiredTrainingLevel ?requiredLevel .
    FILTER(datatype(?shipType) = xsd:string)
    FILTER(datatype(?role) = xsd:string)
    FILTER(LCASE(STR(?shipType)) = "other")
    FILTER(LCASE(STR(?role)) IN (
        "master",
        "chief mate",
        "officers in charge of a navigational watch"
    ))
    FILTER(?requiredLevel != nltl:polarTrainingNotApplicable)
}
"""
    ] .

gen:CrewMemberTrainingInputShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:crewRole ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "Crew role is required to determine polar training applicability."
    ] .
<END_SHACL>
