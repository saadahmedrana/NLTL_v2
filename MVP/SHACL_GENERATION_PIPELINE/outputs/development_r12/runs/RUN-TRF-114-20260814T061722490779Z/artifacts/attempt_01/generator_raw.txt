<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF114ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:node gen:TRF114ImpactLoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:node gen:TRF114SpeedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node gen:TRF114LengthQuantityShape
    ] ;
    sh:sparql [
        sh:message "A ship with a linked thruster ice impact load case must provide its typical ice operating speed in knots." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasThrusterIceImpactLoadCase ?case .
    FILTER NOT EXISTS {
        $this nltl:iceOperatingSpeed ?speed .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A non-hemispherical impact case requires correspondence evidence on that case and an equivalent impact sphere radius on the ship." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?case
WHERE {
    $this nltl:hasThrusterIceImpactLoadCase ?case .
    ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
    FILTER (
        NOT EXISTS {
            ?case nltl:contactGeometryCorrespondenceEvidence ?evidence .
        }
        ||
        NOT EXISTS {
            $this nltl:equivalentImpactSphereRadius ?radius .
        }
    )
}
"""
    ] .

gen:TRF114ImpactLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:minCount 1 ;
        sh:message "The design ice block must reference TRAFICOM Table 6-3."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:minCount 1 ;
        sh:message "The impact load case must reference TRAFICOM Table 6-16."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:node gen:TRF114ThrusterBodyShape ;
        sh:minCount 1 ;
        sh:message "The impact case must identify the applicable thruster body."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:node gen:TRF114ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The impact case must provide the thruster ice impact load in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:node gen:TRF114ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The impact case must provide the thruster ice impact demand in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case must provide loaded-area evidence."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:minCount 1 ;
        sh:in (
            nltl:hemisphericalContactGeometry
            nltl:nonHemisphericalContactGeometry
        ) ;
        sh:message "Contact geometry must be classified as hemispherical or non-hemispherical."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryCorrespondenceEvidence ;
        sh:class nltl:evidenceArtifact
    ] .

gen:TRF114ThrusterBodyShape
    a sh:NodeShape ;
    sh:class nltl:thrusterBody ;
    sh:property [
        sh:path nltl:thrusterResistanceCapacity ;
        sh:node gen:TRF114ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The applicable thruster body must provide its resistance capacity in kilonewtons."
    ] .

gen:TRF114ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1
    ] .

gen:TRF114SpeedQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KN ;
        sh:minCount 1
    ] .

gen:TRF114LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1
    ] .
<END_SHACL>
