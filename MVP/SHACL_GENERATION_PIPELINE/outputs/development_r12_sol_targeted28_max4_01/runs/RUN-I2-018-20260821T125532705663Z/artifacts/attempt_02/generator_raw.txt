    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableLookupCase
    ] ;
    sh:sparql [
        sh:message "Average ice pressure must equal ice force divided by 1000 times the load-patch height and length, within relative tolerance 1e-6 and absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceForce/qudt:numericValue ?forceValue ;
                      nltl:loadPatchHeight/qudt:numericValue ?heightValue ;
                      nltl:loadPatchLength/qudt:numericValue ?lengthValue ;
                      nltl:averageIcePressure/qudt:numericValue ?pressureValue .
                BIND(xsd:decimal(?forceValue) AS ?force)
                BIND(xsd:decimal(?heightValue) AS ?height)
                BIND(xsd:decimal(?lengthValue) AS ?length)
                BIND(xsd:decimal(?pressureValue) AS ?pressure)
                BIND(IF(?height = 0 || ?length = 0,
                        0,
                        ?force / (1000 * ?height * ?length)) AS ?expected)
                BIND(IF(ABS(?expected) * 0.000001 > 0.000000000001,
                        ABS(?expected) * 0.000001,
                        0.000000000001) AS ?tolerance)
                FILTER(
                    ?height = 0 ||
                    ?length = 0 ||
                    ABS(?pressure - ?expected) > ?tolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The ship's peak pressure factor must exactly equal the Table 3 factor selected by its lookup case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor/qudt:numericValue ?shipFactorValue ;
                      nltl:hasPeakPressureFactorLookupCase ?lookupCase .
                ?lookupCase nltl:selectedPeakPressureFactor/qudt:numericValue ?selectedFactorValue .
                FILTER(
                    xsd:decimal(?shipFactorValue) !=
                    xsd:decimal(?selectedFactorValue)
                )
            }
        """
    ] .
