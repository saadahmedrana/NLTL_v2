@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

    sh:class qudt:QuantityValue ;

        sh:path qudt:numericValue ;

        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:LongitudinalFrameShape a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:LengthQuantityShape ;
        sh:message "Each represented longitudinal frame must have one frame spacing in metres."
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:UnitlessQuantityShape ;
        sh:message "Each represented longitudinal frame must have one selected hull area factor."
    ] ;
    sh:property [
        sh:path nltl:requiredLongitudinalFrameShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:SquareCentimetreQuantityShape ;
        sh:message "Each represented longitudinal frame must have one required shear area in square centimetres."
    ] .

gs:LongitudinalFrameShearAreaShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:minCount 1 ;
                sh:class nltl:structuralMember ;
                sh:node gs:LongitudinalFrameShape
            ] ;
            sh:property [
                sh:path nltl:averageIcePressure ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:PressureQuantityShape ;
                sh:message "Average ice pressure is required for the longitudinal-frame calculation."
            ] ;
            sh:property [
                sh:path nltl:peakPressureFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:UnitlessQuantityShape ;
                sh:message "Peak pressure factor is required for the longitudinal-frame calculation."
            ] ;
            sh:property [
                sh:path nltl:frameSpan ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:LengthQuantityShape ;
                sh:message "Frame span is required in metres for the longitudinal-frame calculation."
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:PressureQuantityShape ;
                sh:message "Yield strength is required in megapascals for the longitudinal-frame calculation."
            ] ;
            sh:property [
                sh:path nltl:shearArea ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:SquareMetreQuantityShape ;
                sh:message "Actual net effective shear area is required in square metres."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The required longitudinal-frame shear area must match the prescribed formula within relative tolerance 1e-6 or absolute tolerance 1e-9 cm2; formula divisors must be nonzero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:averageIcePressure/qudt:numericValue ?pAvg ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:frameSpan/qudt:numericValue ?b ;
                      nltl:yieldStrength/qudt:numericValue ?sigmaY ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:frameSpacing/qudt:numericValue ?s ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reported .
                BIND((?s = 0 || ?b = 0 || ?sigmaY = 0) AS ?invalid)
                BIND(IF(?invalid, 0, ?b / ?s) AS ?bPrime)
                BIND(
                    IF(
                        ?invalid,
                        0,
                        IF(
                            ?bPrime < 2,
                            ?b * (1 - (0.25 * ?bPrime)),
                            ?s
                        )
                    )
                    AS ?b2
                )
                BIND(IF(?invalid, 0, 1 - (0.3 / ?bPrime)) AS ?ko)
                BIND(IF(?invalid, 0, ?ko * ?b2) AS ?b1)
                BIND(
                    IF(
                        ?invalid,
                        0,
                        10000 * (?af * ?ppf * ?pAvg) * 0.5 * ?b1 * ?s
                        / (0.577 * ?sigmaY)
                    )
                    AS ?expected
                )
                BIND(
                    IF(
                        ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(?invalid || ABS(?reported - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual net effective shear area must be at least the required longitudinal-frame shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:shearArea/qudt:numericValue ?actual ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?required .
                FILTER((?actual * 10000) < ?required)
            }
        """
    ] .
<END_SHACL>
