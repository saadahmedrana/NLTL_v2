<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

        sh:maxCount 1
    ] ;
    sh:property [

gen:LongitudinalFrameShape a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;

        sh:path nltl:requiredLongitudinalFrameShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    ] .

gen:LongitudinalFrameShearAreaShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:LongitudinalFrameShape
    ] ;
    sh:sparql [
        sh:message "Load patch height, frame span, and yield strength must be non-zero for the shear-area calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:loadPatchHeight ?bNode ;
                      nltl:frameSpan ?sNode ;
                      nltl:yieldStrength ?yieldNode .
                ?bNode qudt:numericValue ?b .
                ?sNode qudt:numericValue ?s .
                ?yieldNode qudt:numericValue ?yieldStrength .
                FILTER (
                    ?b = 0.0 ||
                    ?s = 0.0 ||
                    ?yieldStrength = 0.0
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The required frame shear area must match the formula within relative tolerance 1e-6 (absolute fallback 1e-9 cm2), and the actual shear area must not be less than the calculated requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:averageIcePressure ?pressureNode ;
                      nltl:peakPressureFactor ?pressureFactorNode ;
                      nltl:loadPatchHeight ?bNode ;
                      nltl:frameSpan ?sNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:shearArea ?actualAreaNode ;
                      nltl:hasStructuralMember ?member .

                ?pressureNode qudt:numericValue ?averagePressure .
                ?pressureFactorNode qudt:numericValue ?peakPressureFactor .
                ?bNode qudt:numericValue ?b .
                ?sNode qudt:numericValue ?s .
                ?yieldNode qudt:numericValue ?yieldStrength .
                ?actualAreaNode qudt:numericValue ?actualAreaM2 .

                ?member nltl:frameSpacing ?spacingNode ;
                        nltl:selectedHullAreaFactor ?areaFactorNode ;
                        nltl:requiredLongitudinalFrameShearArea ?requiredAreaNode .

                ?spacingNode qudt:numericValue ?a .
                ?areaFactorNode qudt:numericValue ?areaFactor .
                ?requiredAreaNode qudt:numericValue ?reportedRequiredArea .

                FILTER (
                    ?b != 0.0 &&
                    ?s != 0.0 &&
                    ?yieldStrength != 0.0
                )

                BIND (?b / ?s AS ?bPrime)
                BIND (
                    IF(
                        ?bPrime < 2.0,
                        ?b * (1.0 - 0.25 * ?bPrime),
                        ?s
                    ) AS ?b2
                )
                BIND ((1.0 - 0.3 / ?bPrime) * ?b2 AS ?b1)
                BIND (
                    10000.0 *
                    (?areaFactor * ?peakPressureFactor * ?averagePressure) *
                    0.5 * ?b1 * ?a /
                    (0.577 * ?yieldStrength)
                    AS ?expectedRequiredArea
                )
                BIND (
                    IF(
                        ABS(?expectedRequiredArea) * 0.000001 > 0.000000001,
                        ABS(?expectedRequiredArea) * 0.000001,
                        0.000000001
                    ) AS ?tolerance
                )

                FILTER (
                    ABS(?reportedRequiredArea - ?expectedRequiredArea) > ?tolerance ||
                    (?actualAreaM2 * 10000.0) < ?expectedRequiredArea
                )
            }
        """
    ] .
<END_SHACL>
