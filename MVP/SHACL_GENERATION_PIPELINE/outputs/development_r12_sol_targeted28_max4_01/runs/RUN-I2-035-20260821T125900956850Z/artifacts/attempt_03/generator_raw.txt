@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LongitudinalFrameShearAreaShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        sh:message "Load patch height, frame span, and yield strength must be non-zero for the shear-area calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:loadPatchHeight/qudt:numericValue ?b ;
                      nltl:frameSpan/qudt:numericValue ?s ;
                      nltl:yieldStrength/qudt:numericValue ?sigma .
                FILTER (?b = 0 || ?s = 0 || ?sigma = 0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The required longitudinal frame shear area must match the prescribed calculation within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:averageIcePressure/qudt:numericValue ?pavg ;
                      nltl:loadPatchHeight/qudt:numericValue ?b ;
                      nltl:frameSpan/qudt:numericValue ?s ;
                      nltl:peakPressureFactor/qudt:numericValue ?ppf ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:frameSpacing/qudt:numericValue ?a ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?af ;
                        nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reported .
                FILTER (?b != 0 && ?s != 0 && ?sigma != 0)
                BIND (?b / ?s AS ?bPrime)
                BIND (IF(?bPrime < 2, ?b * (1 - 0.25 * ?bPrime), ?s) AS ?b2)
                BIND ((1 - 0.3 / ?bPrime) * ?b2 AS ?b1)
                BIND (
                    10000 * (?af * ?ppf * ?pavg) * 0.5 * ?b1 * ?a /
                    (0.577 * ?sigma)
                    AS ?expected
                )
                BIND (
                    IF(
                        ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER (ABS(?reported - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The actual net effective shear area must be at least the required longitudinal frame shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shearArea/qudt:numericValue ?actualSquareMetres ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?requiredSquareCentimetres .
                FILTER (?actualSquareMetres * 10000 < ?requiredSquareCentimetres)
            }
        """
    ] .
<END_SHACL>
