gen:I2_048_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:class nltl:polarClassValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Polar Class is required for the Table 8 steel-grade selection."
    ] ;
    sh:property [
        sh:path nltl:materialClass ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Material class is required for the Table 8 steel-grade selection."
    ] ;
    sh:property [
        sh:path nltl:asBuiltThickness ;
        sh:node gen:I2_048_AsBuiltThicknessShape ;
        sh:minCount 1 ;
        sh:message "Ship as-built thickness is required."
    ] ;
    sh:property [
        sh:path nltl:steelGrade ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_048_SteelGradeRequirementCaseShape ;
        sh:minCount 1 ;
        sh:message "A steel-grade requirement case is required."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_048_TableLookupCaseShape ;
        sh:minCount 1 ;
        sh:message "A Table 8 lookup case is required."
    ] .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:maxCount 1
    ] .

gen:I2_048_AsBuiltPlateThicknessShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:maxCount 1
    ] .

gen:I2_048_SteelGradeValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeOrderRank ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "A steel-grade order rank is required for grade comparison."
    ] .

gen:I2_048_TableLookupCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:in ( nltl:iacsUrI2Table8 ) ;
        sh:minCount 1 ;
        sh:message "The lookup must reference IACS UR I2 Table 8."
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Evidence of the Table 8 lookup selection is required."
    ] .

gen:I2_048_SteelGradeRequirementCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:steelGradeRequirementCasePlating ;
        sh:class nltl:plating ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The plating assessed by the steel-grade case is required."
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:node gen:I2_048_AsBuiltPlateThicknessShape ;
        sh:minCount 1 ;
        sh:message "As-built plate thickness in millimetres is required."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:message "Steel material class must be Class One, Two, or Three."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:message "Steel strength category is required."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:class nltl:steelGradeValue ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_048_SteelGradeValueShape ;
        sh:minCount 1 ;
        sh:message "The steel grade required by Table 8 is required."
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:class nltl:steelGradeValue ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_048_SteelGradeValueShape ;
        sh:minCount 1 ;
        sh:message "The actual hull structural steel grade is required."
    ] ;
    sh:node gen:I2_048_TableLookupCaseShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The actual steel grade must not rank below the Table 8 required grade." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?actualGrade ?requiredGrade
WHERE {
    $this nltl:actualHullStructuralSteelGrade ?actualGrade ;
          nltl:requiredHullStructuralSteelGrade ?requiredGrade .
    ?actualGrade nltl:steelGradeOrderRank ?actualRank .
    ?requiredGrade nltl:steelGradeOrderRank ?requiredRank .
    FILTER(xsd:integer(?actualRank) < xsd:integer(?requiredRank))
}
"""
    ] .
