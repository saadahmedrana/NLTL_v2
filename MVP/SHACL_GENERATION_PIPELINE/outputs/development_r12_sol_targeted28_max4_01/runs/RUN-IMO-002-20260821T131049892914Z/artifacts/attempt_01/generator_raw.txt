<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO002ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:message "The ship must declare a design ice condition."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:message "The ship must declare its Polar Code category."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for thin first-year ice with possible old-ice inclusions that is not category A must be category B." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this nltl:designIceCondition
        nltl:thinFirstYearIceWithPossibleOldIceInclusions .
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryA .
    }
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryB .
    }
}
"""
    ] .
<END_SHACL>
