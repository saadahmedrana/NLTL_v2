<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO002CategoryBShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for at least thin first-year ice with possible old-ice inclusions, and not assigned category A, must have category B and no non-B category value." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:designIceCondition nltl:thinFirstYearIceWithPossibleOldIceInclusions .
    FILTER NOT EXISTS {
        $this nltl:shipCategory nltl:polarShipCategoryA .
    }
    FILTER (
        NOT EXISTS {
            $this nltl:shipCategory nltl:polarShipCategoryB .
        }
        ||
        EXISTS {
            $this nltl:shipCategory ?category .
            FILTER (?category != nltl:polarShipCategoryB)
        }
    )
}
"""
    ] .
<END_SHACL>
