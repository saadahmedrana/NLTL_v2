                sh:path nltl:shipCategory ;

            ]

            sh:property [
                sh:path nltl:shipCategory ;
                sh:hasValue nltl:polarShipCategoryC

        sh:message "Each machinery component of a category A or B ship must have an approved scantling record identifying the approving authority and approval standard." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:hasComponent ?component .
    ?component a nltl:machineryComponent .
    FILTER (?category IN (
        nltl:polarShipCategoryA,
        nltl:polarShipCategoryB
    ))

        ?approval a nltl:approvalRecord ;
                  nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                  nltl:approvingAuthority ?authority ;

        )
    }
}
"""

PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:shipCategory nltl:polarShipCategoryC ;
          nltl:shipIceStrengthened true ;
          nltl:hasComponent ?component .
    ?component a nltl:machineryComponent .
    FILTER NOT EXISTS {
        ?component nltl:hasApprovalRecord ?approval .
        ?approval a nltl:approvalRecord ;
                  nltl:scantlingApprovalStatus nltl:evidenceStateApproved ;
                  nltl:approvingAuthority ?authority ;
                  nltl:approvalStandard ?standard ;
                  nltl:operatingIceType ?iceType ;
                  nltl:operatingIceConcentration ?concentration .

        FILTER (
            DATATYPE(?authority) = xsd:string &&
            DATATYPE(?standard) = xsd:string &&
            DATATYPE(?iceType) = xsd:string &&
            DATATYPE(?concentrationValue) = xsd:decimal
        )
        FILTER NOT EXISTS {
            ?concentration qudt:unit ?otherUnit .
            FILTER (?otherUnit != <http://qudt.org/vocab/unit/UNITLESS>)
        }
    }
}
"""
