<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ClassCertificateShape
    a sh:NodeShape ;
    sh:class nltl:classCertificate ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAmidships ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAmidships ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF014Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must identify its ice class."
    ] ;
    sh:property [
        sh:path nltl:hasClassCertificate ;
        sh:node gen:ClassCertificateShape ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A class certificate indicating all six maximum and minimum ice-class draughts is required."
    ] ;
    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:class nltl:iceDraughtRestrictionDocument ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Restrictions on draught when operating in ice must be documented."
    ] ;
    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-draught restriction documentation must be retained on board."
    ] ;
    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ice-draught restriction documentation must be readily available to the master."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date is required to determine the applicable marking rule."
    ] ;
    sh:property [
        sh:path nltl:firstScheduledDryDockingDate ;
        sh:datatype xsd:date ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The summer load-line fresh-water draught is required to determine marking applicability."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:node gen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The upper ice waterline draught is required to determine marking applicability."
    ] ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node gen:LengthQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship built on or after 1 July 2007 with the summer load line above the UIWL must have a warning triangle and an ice-class draught mark at the certificate maximum draught amidships." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionDate ?constructionDate ;
          nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
          nltl:upperIceWaterlineDraught ?uiwlQuantity .
    ?summerQuantity qudt:numericValue ?summerDraught ;
                    qudt:unit unit:M .
    ?uiwlQuantity qudt:numericValue ?uiwlDraught ;
                  qudt:unit unit:M .
    FILTER(
        ?constructionDate >= "2007-07-01"^^xsd:date &&
        ?summerDraught < ?uiwlDraught
    )
    FILTER NOT EXISTS {
        $this nltl:warningTrianglePresent true ;
              nltl:iceClassDraughtMarkPresent true ;
              nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity ;
              nltl:hasClassCertificate ?certificate .
        ?markQuantity qudt:numericValue ?markDraught ;
                      qudt:unit unit:M .
        ?certificate nltl:maximumIceClassDraughtAmidships ?maximumQuantity .
        ?maximumQuantity qudt:numericValue ?maximumDraught ;
                         qudt:unit unit:M .
        FILTER(?markDraught = ?maximumDraught)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship built before 1 July 2007 with the UIWL below the summer load line, complete marking is required when the first scheduled dry docking after the cutoff is due or past; otherwise that docking date is required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:constructionDate ?constructionDate ;
          nltl:summerLoadLineFreshWaterDraught ?summerQuantity ;
          nltl:upperIceWaterlineDraught ?uiwlQuantity .
    ?summerQuantity qudt:numericValue ?summerDraught ;
                    qudt:unit unit:M .
    ?uiwlQuantity qudt:numericValue ?uiwlDraught ;
                  qudt:unit unit:M .
    OPTIONAL {
        $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
    }
    FILTER(
        ?constructionDate < "2007-07-01"^^xsd:date &&
        ?summerDraught < ?uiwlDraught
    )
    FILTER NOT EXISTS {
        $this nltl:warningTrianglePresent true ;
              nltl:iceClassDraughtMarkPresent true ;
              nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity ;
              nltl:hasClassCertificate ?certificate .
        ?markQuantity qudt:numericValue ?markDraught ;
                      qudt:unit unit:M .
        ?certificate nltl:maximumIceClassDraughtAmidships ?maximumQuantity .
        ?maximumQuantity qudt:numericValue ?maximumDraught ;
                         qudt:unit unit:M .
        FILTER(?markDraught = ?maximumDraught)
    }
    FILTER(
        !BOUND(?dryDockingDate) ||
        YEAR(?dryDockingDate) < YEAR(NOW()) ||
        (
            YEAR(?dryDockingDate) = YEAR(NOW()) &&
            MONTH(?dryDockingDate) < MONTH(NOW())
        ) ||
        (
            YEAR(?dryDockingDate) = YEAR(NOW()) &&
            MONTH(?dryDockingDate) = MONTH(NOW()) &&
            DAY(?dryDockingDate) <= DAY(NOW())
        )
    )
}
"""
    ] .
<END_SHACL>
