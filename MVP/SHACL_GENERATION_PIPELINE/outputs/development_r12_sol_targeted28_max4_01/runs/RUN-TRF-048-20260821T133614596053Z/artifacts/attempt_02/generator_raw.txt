gen:TRF048LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:class nltl:frameBoundaryConditionTypeValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue true
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:CubicCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:SquareCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:grossFrameShearArea ;
        sh:node gen:SquareCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bracketArea ;
        sh:node gen:SquareCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualFrameShearArea ;
        sh:node gen:SquareCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The reported load-distribution factor, section modulus, and required shear area must satisfy the prescribed formulas within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?pressure ;
                      nltl:designIceLoadHeight/qudt:numericValue ?height ;
                      nltl:spacing/qudt:numericValue ?spacing ;
                      nltl:span/qudt:numericValue ?span ;
                      nltl:frameMomentFactorM/qudt:numericValue ?momentFactor ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
                      nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?reportedF4 ;
                      nltl:sectionModulus/qudt:numericValue ?reportedSectionModulus ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedShearArea .

                BIND(1.0 - (0.2 * ?height / ?spacing) AS ?expectedF4)
                BIND(
                    (?expectedF4 * ?pressure * ?height * ?span * ?span
                     / (4.0 * ?momentFactor * ?yieldStrength)) * 1000000.0
                    AS ?expectedSectionModulus
                )
                BIND(
                    (1.7320508075688772 * ?expectedF4 * 2.16 * ?pressure
                     * ?height * ?span / (2.0 * ?yieldStrength)) * 10000.0
                    AS ?expectedShearArea
                )
                BIND(
                    IF(
                        ABS(?expectedF4) * 0.000001 > 0.000000001,
                        ABS(?expectedF4) * 0.000001,
                        0.000000001
                    )
                    AS ?f4Tolerance
                )
                BIND(
                    IF(
                        ABS(?expectedSectionModulus) * 0.000001 > 0.000000001,
                        ABS(?expectedSectionModulus) * 0.000001,
                        0.000000001
                    )
                    AS ?sectionModulusTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedShearArea) * 0.000001 > 0.000000001,
                        ABS(?expectedShearArea) * 0.000001,
                        0.000000001
                    )
                    AS ?shearAreaTolerance
                )

                FILTER(
                    ?spacing = 0.0 ||
                    ?yieldStrength = 0.0 ||
                    ABS(?reportedF4 - ?expectedF4) > ?f4Tolerance ||
                    ABS(?reportedSectionModulus - ?expectedSectionModulus) > ?sectionModulusTolerance ||
                    ABS(?reportedShearArea - ?expectedShearArea) > ?shearAreaTolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a continuous beam with brackets without significantly different boundary conditions, frame moment factor m must equal 13.3." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionType nltl:continuousBeamWithBracketsFrameCondition ;
                      nltl:significantlyDifferentBoundaryConditions ?different ;
                      nltl:frameMomentFactorM/qudt:numericValue ?momentFactor .
                FILTER(
                    ?different = false &&
                    (?momentFactor < 13.3 || ?momentFactor > 13.3)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Actual frame shear area must equal gross frame shear area minus bracket area within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:grossFrameShearArea/qudt:numericValue ?grossArea ;
                      nltl:bracketArea/qudt:numericValue ?bracketArea ;
                      nltl:actualFrameShearArea/qudt:numericValue ?actualArea .

                BIND(?grossArea - ?bracketArea AS ?expectedActualArea)
                BIND(
                    IF(
                        ABS(?expectedActualArea) * 0.000001 > 0.000000001,
                        ABS(?expectedActualArea) * 0.000001,
                        0.000000001
                    )
                    AS ?actualAreaTolerance
                )
                FILTER(
                    ABS(?actualArea - ?expectedActualArea) > ?actualAreaTolerance
                )
            }
        """
    ] .
