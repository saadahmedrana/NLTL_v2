<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:DesignConditionReadinessShape
    a sh:NodeShape ;
    sh:class nltl:designCondition ;
    sh:property [
        sh:path nltl:designConditionLocalStrengthCapacity ;
        sh:node gs:KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:message "Each applicable design condition requires an externally computed local-strength capacity in kN."
    ] .

gs:TRF070Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Main-propulsion-thruster applicability must be stated once."
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thruster type must be stated once."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "This branch applies only when the supplied thruster type is outside the azimuthing and fixed set." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:thrusterType ?thrusterType .
                        FILTER(?thrusterType IN ("azimuthing", "fixed"))
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:in ("azimuthing" "fixed")
            ] ;
            sh:property [
                sh:path nltl:propellerIceInteractionLoad ;
                sh:node gs:KiloNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:message "An applicable main-propulsion thruster requires its propeller/ice interaction load in kN."
            ] ;
            sh:property [
                sh:path nltl:thrusterBodyIceInteractionLoad ;
                sh:node gs:KiloNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:message "An applicable main-propulsion thruster requires its thruster-body/ice interaction load in kN."
            ] ;
            sh:property [
                sh:path nltl:localIcePressure ;
                sh:node gs:KiloNewtonQuantityShape ;
                sh:minCount 1 ;
                sh:message "An applicable main-propulsion thruster requires the local ice pressure value in kN."
            ] ;
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:class nltl:designCondition ;
                sh:node gs:DesignConditionReadinessShape ;
                sh:minCount 1 ;
                sh:message "An applicable main-propulsion thruster requires a design condition with local-strength capacity."
            ] ;
            sh:sparql [
                sh:message "Design-condition local-strength capacity must be greater than or equal to the applicable local ice pressure." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:localIcePressure ?pressureQuantity ;
                              nltl:hasDesignCondition ?condition .
                        ?pressureQuantity qudt:numericValue ?pressure .
                        ?condition nltl:designConditionLocalStrengthCapacity ?capacityQuantity .
                        ?capacityQuantity qudt:numericValue ?capacity .
                        FILTER(?capacity < ?pressure)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
