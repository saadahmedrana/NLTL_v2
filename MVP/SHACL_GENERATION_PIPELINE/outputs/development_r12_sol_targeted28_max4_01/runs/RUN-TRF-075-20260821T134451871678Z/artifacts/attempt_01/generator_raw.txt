@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF075ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:node gen:TRF075PropellerBladeLoadCaseShape
    ] .

gen:TRF075PropellerBladeLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseBlade ;
        sh:class nltl:propellerBlade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each propeller-blade load case must identify exactly one assessed propeller blade."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseIceBlock ;
        sh:class nltl:iceBlock ;
        sh:minCount 1 ;
        sh:message "Each propeller-blade load case must identify its ice block."
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:backwardBladeForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF075ForceQuantityShape
            ] ;
            sh:property [
                sh:path nltl:forwardBladeForce ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:forwardBladeForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:TRF075ForceQuantityShape
            ] ;
            sh:property [
                sh:path nltl:backwardBladeForce ;
                sh:maxCount 0
            ]
        ]
    ) ;
    sh:message "Each propeller-blade load case must contain exactly one backward or forward blade force, not both." .

gen:TRF075ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
