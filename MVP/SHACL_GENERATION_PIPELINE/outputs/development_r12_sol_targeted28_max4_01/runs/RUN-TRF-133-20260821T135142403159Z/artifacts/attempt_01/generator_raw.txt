gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship's ice class must be provided."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Timber load line applicability must be stated."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:class nltl:warningTriangleMarking ;
        sh:node gen:TRF133WarningTriangleMarkingShape ;
        sh:minCount 1 ;
        sh:message "A conforming warning triangle marking is required."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:class nltl:iceClassDraughtMarking ;
        sh:node gen:TRF133IceClassDraughtMarkingShape ;
        sh:minCount 1 ;
        sh:message "A conforming ice-class draught marking is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:in ( true ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path (
                    nltl:hasIceClassDraughtMarking
                    nltl:draughtMarkAftReferencePoint
                ) ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:timberLoadLineVerticalReference ) ;
                sh:minCount 1 ;
                sh:message "When the timber load line mark applies, the draught mark must be referenced to its vertical line."
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:datatype xsd:boolean ;
                sh:in ( false ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path (
                    nltl:hasIceClassDraughtMarking
                    nltl:draughtMarkAftReferencePoint
                ) ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:loadLineRingCentreReference ) ;
                sh:minCount 1 ;
                sh:message "When the timber load line mark does not apply, the draught mark must be referenced to the centre of the load line ring."
            ]
        ]
    ) .
