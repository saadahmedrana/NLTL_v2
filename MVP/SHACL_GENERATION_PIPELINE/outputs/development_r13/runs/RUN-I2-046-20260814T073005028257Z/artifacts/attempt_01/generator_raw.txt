<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_046_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClassRequirementsApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one Polar Class applicability value is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClassRequirementsApplicable ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                ) ;
                sh:message "An applicable ship must have valid Polar Class evidence."
            ] ;
            sh:property [
                sh:path nltl:hasInternalIceStrengthenedStructure ;
                sh:class nltl:internalIceStrengthenedStructure ;
                sh:node gen:I2_046_InternalStructureShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClassRequirementsApplicable ;
                sh:hasValue false
            ]
        ]
    ) .

gen:I2_046_InternalStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:internalStructureCorrosionAbrasionAddition ;
        sh:class qudt:QuantityValue ;
        sh:node gen:I2_046_CorrosionAbrasionAdditionShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked internal ice-strengthened structure must have one corrosion/abrasion addition."
    ] .

gen:I2_046_CorrosionAbrasionAdditionShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The corrosion/abrasion addition must be at least 1.0 mm."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The corrosion/abrasion addition unit must be millimetres."
    ] .
<END_SHACL>
