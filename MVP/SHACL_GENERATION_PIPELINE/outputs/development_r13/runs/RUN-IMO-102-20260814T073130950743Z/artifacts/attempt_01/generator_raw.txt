<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PolarTrainingRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        ) ;
        sh:message "The ship must have exactly one controlled ice-condition classification."
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        ) ;
        sh:message "The ship must have exactly one controlled polar-training ship type."
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:class nltl:crewMember ;
        sh:node gen:PolarTrainingCrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A represented crew member lacks the matrix-required training level or a valid matching STCW qualification." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this ?crew
WHERE {
    $this nltl:iceConditionClassification ?ice ;
          nltl:polarTrainingShipTypeClassification ?shipType ;
          nltl:hasCrewMemberInventory ?crew .
    ?crew nltl:polarTrainingCrewRoleClassification ?role .

    BIND(
        IF(
            ?ice = nltl:iceFreeIceCondition,
            nltl:polarTrainingNotApplicable,
            IF(
                ?ice = nltl:openWaterIceCondition &&
                ?shipType = nltl:otherShipPolarTrainingType,
                nltl:polarTrainingNotApplicable,
                IF(
                    ?ice = nltl:openWaterIceCondition,
                    nltl:basicPolarTrainingLevel,
                    IF(
                        ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole,
                        nltl:basicPolarTrainingLevel,
                        nltl:advancedPolarTrainingLevel
                    )
                )
            )
        )
        AS ?expected
    )

    FILTER(
        NOT EXISTS {
            ?crew nltl:hasCrewTrainingRecord ?record .
            ?record nltl:requiredTrainingLevel ?expected .
            FILTER(
                ?expected = nltl:polarTrainingNotApplicable ||
                EXISTS {
                    ?crew nltl:hasPolarTrainingRecord ?record .
                    ?record nltl:trainingRecordLevel ?expected ;
                            nltl:stcwQualificationValid true .
                }
            )
        }
        ||
        EXISTS {
            ?crew (nltl:hasCrewTrainingRecord|nltl:hasPolarTrainingRecord) ?linkedRecord .
            FILTER(
                NOT EXISTS {
                    ?linkedRecord nltl:requiredTrainingLevel ?expected .
                }
                ||
                EXISTS {
                    ?linkedRecord nltl:requiredTrainingLevel ?actualRequired .
                    FILTER(?actualRequired != ?expected)
                }
                ||
                (
                    ?expected != nltl:polarTrainingNotApplicable &&
                    (
                        NOT EXISTS {
                            ?linkedRecord nltl:trainingRecordLevel ?expected .
                        }
                        ||
                        EXISTS {
                            ?linkedRecord nltl:trainingRecordLevel ?actualLevel .
                            FILTER(?actualLevel != ?expected)
                        }
                        ||
                        NOT EXISTS {
                            ?linkedRecord nltl:stcwQualificationValid true .
                        }
                        ||
                        EXISTS {
                            ?linkedRecord nltl:stcwQualificationValid ?qualificationState .
                            FILTER(?qualificationState != true)
                        }
                    )
                )
            )
        }
    )
}
"""
    ] .

gen:PolarTrainingCrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        ) ;
        sh:message "Each represented crew member must have exactly one controlled polar-training role."
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:class nltl:polarTrainingRecord ;
        sh:node gen:PolarTrainingRecordShape
    ] ;
    sh:property [
        sh:path nltl:hasPolarTrainingRecord ;
        sh:class nltl:polarTrainingRecord ;
        sh:node gen:PolarTrainingRecordShape
    ] .

gen:PolarTrainingRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        )
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        )
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .
<END_SHACL>
