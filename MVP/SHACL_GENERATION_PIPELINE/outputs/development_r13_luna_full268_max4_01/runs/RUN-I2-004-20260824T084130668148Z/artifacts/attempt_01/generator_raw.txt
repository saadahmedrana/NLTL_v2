<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-004-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterline ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify the upper ice waterline." ;
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloTON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The ship displacement at the upper ice waterline must be recorded in metric kilotonnes." ;
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:upperIceWaterlineCase ;
            sh:property [
                sh:path nltl:waterlineCaseDisplacement ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloTON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                    ]
                ] ;
            ]
        ] ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline displacement must equal the greatest displacement among the declared upper ice waterline cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperIceWaterlineDisplacement ?duiQuantity .
                ?duiQuantity qudt:numericValue ?duiValue .
                ?duiQuantity qudt:unit unit:KiloTON_Metric .

                $this nltl:hasUpperIceWaterlineCase ?case .
                ?case nltl:waterlineCaseDisplacement ?caseQuantity .
                ?caseQuantity qudt:numericValue ?caseValue .
                ?caseQuantity qudt:unit unit:KiloTON_Metric .

                FILTER NOT EXISTS {
                    $this nltl:hasUpperIceWaterlineCase ?otherCase .
                    ?otherCase nltl:waterlineCaseDisplacement ?otherQuantity .
                    ?otherQuantity qudt:numericValue ?otherValue .
                    ?otherQuantity qudt:unit unit:KiloTON_Metric .
                    FILTER (?otherValue > ?caseValue)
                }

                FILTER (
                    ABS(?duiValue - ?caseValue) >
                    (0.000001 * ABS(?caseValue) + 0.000001)
                )
            }
        """
    ] .
<END_SHACL>
