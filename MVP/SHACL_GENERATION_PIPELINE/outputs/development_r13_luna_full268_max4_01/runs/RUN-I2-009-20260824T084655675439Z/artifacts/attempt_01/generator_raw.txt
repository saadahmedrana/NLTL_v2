<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hullArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A hull area value is required." ;
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureQuantityShape ;
        sh:message "Average ice pressure is required in MPa." ;
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DegreeQuantityShape ;
        sh:message "The buttock angle is required in degrees." ;
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AspectRatioShape ;
        sh:message "The load patch aspect ratio must equal 3.6." ;
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "The non-bow load patch height is required in metres." ;
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthQuantityShape ;
        sh:message "The non-bow load patch length is required in metres." ;
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DegreeQuantityShape ;
        sh:message "The normal frame angle is required in degrees." ;
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability must be recorded." ;
    ] ;
    sh:property [
        sh:path nltl:hasHullAreaAssignment ;
        sh:minCount 1 ;
        sh:node gen:I2_009_HullAreaAssignmentShape ;
        sh:message "At least one hull area assignment is required." ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Bow-form applicability must be true exactly when the buttock angle is greater than 0 and less than 80 degrees and the normal frame angle is greater than 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:buttockAngle ?buttockNode .
                ?buttockNode q:numericValue ?buttock .
                $this nltl:normalFrameAngle ?frameNode .
                ?frameNode q:numericValue ?frame .
                $this nltl:bowFormApplicability ?applicable .
                FILTER (
                    (
                        ?buttock > 0 &&
                        ?buttock < 80 &&
                        ?frame > 10 &&
                        ?applicable = true
                    )
                    ||
                    (
                        (
                            ?buttock <= 0 ||
                            ?buttock >= 80 ||
                            ?frame <= 10
                        ) &&
                        ?applicable = false
                    )
                )
            }
        """ ;
    ] .

gen:I2_009_HullAreaAssignmentShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaAssignment ;
    sh:property [
        sh:path nltl:assignedHullArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowIntermediateBottomHullArea
            nltl:midbodyBottomHullArea
            nltl:sternBottomHullArea
            nltl:otherIceStrengthenedArea
        ) ;
        sh:class nltl:hullAreaValue ;
        sh:message "Each hull area assignment must identify an allowed hull area value." ;
    ] .

gen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .

gen:AspectRatioShape
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "3.6"^^xsd:decimal ;
        sh:maxInclusive "3.6"^^xsd:decimal ;
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
<END_SHACL>
