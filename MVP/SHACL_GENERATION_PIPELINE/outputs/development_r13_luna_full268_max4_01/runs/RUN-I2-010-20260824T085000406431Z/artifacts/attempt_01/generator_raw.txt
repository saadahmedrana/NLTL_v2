<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-010> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For a ship assigned Polar Class PC6 or PC7, a string-valued bow design force method and evidence of a vertical-sided bow or a normal frame angle between 0 and 10 degrees are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass .
                FILTER (
                    ?polarClass = nltl:polarClassPc6 ||
                    ?polarClass = nltl:polarClassPc7
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:bowDesignForceMethod ?method .
                        FILTER (datatype(?method) = xsd:string)
                    }
                    ||
                    (
                        NOT EXISTS {
                            $this nltl:verticalSidedBowForm ?verticalForm .
                            FILTER (?verticalForm = true)
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:bowVerticalSideStatus ?verticalStatus .
                            FILTER (?verticalStatus = true)
                        }
                        &&
                        NOT EXISTS {
                            $this nltl:normalFrameAngle ?angle .
                            ?angle qudt:numericValue ?value ;
                                   qudt:unit <http://qudt.org/vocab/unit/DEG> .
                            FILTER (
                                datatype(?value) = xsd:decimal &&
                                ?value >= 0 &&
                                ?value <= 10
                            )
                        }
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each represented normal frame angle must be a decimal quantity in degrees between 0 and 10." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:normalFrameAngle ?angle .
                FILTER NOT EXISTS {
                    ?angle qudt:numericValue ?value ;
                           qudt:unit <http://qudt.org/vocab/unit/DEG> .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= 0 &&
                        ?value <= 10
                    )
                }
            }
        """
    ] .
<END_SHACL>
