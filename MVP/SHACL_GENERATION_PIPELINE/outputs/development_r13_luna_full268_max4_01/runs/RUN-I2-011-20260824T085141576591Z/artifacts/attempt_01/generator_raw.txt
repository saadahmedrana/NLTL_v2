<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_011_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bulbous-bow applicability information is required."
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A valid Polar Class value is required to determine applicability."
    ] ;
    sh:or (
        gs:I2_011_ApplicableShape
        gs:I2_011_NonApplicableShape
    ) .

gs:I2_011_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:in (
            nltl:polarClassPc6
            nltl:polarClassPc7
        )
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The I2.3.2.1(iii) bow design force is required in MN."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The I2.3.2.1(iv) bow design force is required in MN."
    ] ;
    sh:property [
        sh:path nltl:shapeCoefficient ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0.6"^^xsd:decimal ;
                sh:maxInclusive "0.6"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The assumed shape coefficient fa must equal 0.6."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "1.3"^^xsd:decimal ;
                sh:maxInclusive "1.3"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The assumed load-patch aspect ratio AR must equal 1.3."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The applicable bow design force is required in kN."
    ] ;
    sh:sparql [
        sh:message "The bow design force must not be less than either referenced design force." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:bowDesignForce ?resultValue ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiValue ;
                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivValue .

                ?resultValue qudt:numericValue ?result .
                ?iiiValue qudt:numericValue ?iii .
                ?ivValue qudt:numericValue ?iv .

                FILTER (?result < ?iii || ?result < ?iv)
            }
        """
    ] .

gs:I2_011_NonApplicableShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                )
            ]
        ]
    ) .
