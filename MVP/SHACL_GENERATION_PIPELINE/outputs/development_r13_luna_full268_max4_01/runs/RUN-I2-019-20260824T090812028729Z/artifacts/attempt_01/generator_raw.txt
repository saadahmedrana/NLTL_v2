<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:HullAreaQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:HullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:node gs:HullAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each hull area reached by a structural member must have exactly one dimensionless hull area factor."
    ] .

gs:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberShape
    ] .

gs:StructuralMemberShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasStructuralMember ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasSpannedHullArea ;
        sh:node gs:HullAreaValueShape
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:node gs:HullAreaQuantityShape ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "A structural member spanning one or more hull areas must specify a selected hull area factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSpannedHullArea ?area .
                FILTER NOT EXISTS {
                    $this nltl:selectedHullAreaFactor ?selected .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected hull area factor must equal the largest hull area factor among all hull areas spanned by the structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedHullAreaFactor ?selectedQuantity ;
                      nltl:hasSpannedHullArea ?area .
                ?selectedQuantity qudt:numericValue ?selectedValue .
                ?area nltl:hullAreaFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factorValue .
                FILTER NOT EXISTS {
                    $this nltl:hasSpannedHullArea ?greaterArea .
                    ?greaterArea nltl:hullAreaFactor ?greaterQuantity .
                    ?greaterQuantity qudt:numericValue ?greaterValue .
                    FILTER (?greaterValue > ?selectedValue)
                }
                FILTER NOT EXISTS {
                    $this nltl:hasSpannedHullArea ?equalArea .
                    ?equalArea nltl:hullAreaFactor ?equalQuantity .
                    ?equalQuantity qudt:numericValue ?equalValue .
                    FILTER (?equalValue = ?selectedValue)
                }
            }
        """
    ] .
<END_SHACL>
