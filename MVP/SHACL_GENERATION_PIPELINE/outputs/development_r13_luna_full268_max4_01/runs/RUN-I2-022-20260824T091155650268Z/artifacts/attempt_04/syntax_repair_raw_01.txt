<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_022_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:platingHullAreaClassification ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:in (
                            nltl:bowIntermediateBottomHullArea
                            nltl:midbodyBottomHullArea
                            nltl:sternBottomHullArea
                        )
                    ] ;
                    sh:property [
                        sh:path nltl:framingAngleOmega ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1 ;
                                sh:minInclusive 70
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue <http://qudt.org/vocab/unit/DEG>
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:platingSupportedByStructuralMember ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:structuralMember ;
                            sh:property [
                                sh:path nltl:frameSpacing ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1 ;
                                        sh:hasValue <http://qudt.org/vocab/unit/M>
                                    ]
                                ]
                            ] ;
                            sh:property [
                                sh:path nltl:selectedHullAreaFactor ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minCount 1
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1 ;
                                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
                                    ]
                                ]
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:iceLoadRequiredNetPlateThickness ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:minCount 1 ;
                                sh:maxCount 1 ;
                                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
                            ]
                        ]
                    ]
                ]
                [
                    sh:or (
                        [
                            sh:property [
                                sh:path nltl:framingAngleOmega ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minCount 1 ;
                                        sh:maxExclusive 70
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1 ;
                                        sh:hasValue <http://qudt.org/vocab/unit/DEG>
                                    ]
                                ]
                            ]
                        ]
                        [
                            sh:property [
                                sh:path nltl:platingHullAreaClassification ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:sparql [
                                sh:message "Non-applicable plating must have a hull-area classification outside BI_b, M_b, and S_b." ;
                                sh:select """
                                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                                    SELECT $this
                                    WHERE {
                                        $this nltl:platingHullAreaClassification ?area .
                                        FILTER (
                                            ?area IN (
                                                nltl:bowIntermediateBottomHullArea,
                                                nltl:midbodyBottomHullArea,
                                                nltl:sternBottomHullArea
                                            )
                                        )
                                    }
                                """
                            ]
                        ]
                    )
                ]
            )
        ]
    ] ;
    sh:sparql [
        sh:message "Applicable plating requires the declared ship-owned inputs and a ship-to-structural-member operand path." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:platingHullAreaClassification ?area ;
                         nltl:framingAngleOmega ?angle .
                FILTER (
                    ?area IN (
                        nltl:bowIntermediateBottomHullArea,
                        nltl:midbodyBottomHullArea,
                        nltl:sternBottomHullArea
                    )
                )
                ?angle qudt:numericValue ?angleValue ;
                       qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (?angleValue >= 70)

                FILTER (
                    NOT EXISTS {
                        $this nltl:averageIcePressure ?avg .
                        ?avg qudt:numericValue ?avgValue ;
                            qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:peakPressureFactor ?peak .
                        ?peak qudt:numericValue ?peakValue ;
                             qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldStrength ?yield .
                        ?yield qudt:numericValue ?yieldValue ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:loadPatchHeight ?height .
                        ?height qudt:numericValue ?heightValue ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:hasStructuralMember ?member .
                        ?member nltl:frameSpacing ?spacing ;
                               nltl:selectedHullAreaFactor ?factor .
                        ?spacing qudt:numericValue ?spacingValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?factor qudt:numericValue ?factorValue ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    }
                )
            }
        """
    ] .

        ]
    ] .
