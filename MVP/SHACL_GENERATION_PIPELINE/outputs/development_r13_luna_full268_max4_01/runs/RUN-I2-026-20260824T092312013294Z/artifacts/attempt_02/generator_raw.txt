<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_026_StructuralMemberShape
    ] .

gs:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gs:I2_026_MemberSupportShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:fixedFrameSupport
            ]
            [
                sh:hasValue nltl:simpleFrameSupport
            ]
        )
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The frame support condition does not match the declared continuity, connection bracket, rotational restraint, or termination state." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:continuousThroughSupport ?continuous ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates ;
                      nltl:hasMemberSupport ?support .
                ?support nltl:connectionBracketPresent ?bracket ;
                         nltl:demonstratedRotationalRestraint ?restraint .
                FILTER (
                    (
                        ?terminates = true &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:fixedFrameSupport
                        }
                    )
                    ||
                    (
                        ?terminates = false &&
                        ?continuous = true &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:fixedFrameSupport
                        }
                    )
                    ||
                    (
                        ?terminates = false &&
                        ?continuous = false &&
                        ?bracket = true &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:fixedFrameSupport
                        }
                    )
                    ||
                    (
                        ?terminates = false &&
                        ?continuous = false &&
                        ?bracket = false &&
                        ?restraint = true &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:fixedFrameSupport
                        }
                    )
                    ||
                    (
                        ?terminates = false &&
                        ?continuous = false &&
                        ?bracket = false &&
                        ?restraint = false &&
                        NOT EXISTS {
                            $this nltl:frameSupportCondition nltl:simpleFrameSupport
                        }
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A demonstrated significant rotational restraint must have supporting evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence
                }
            }
        """
    ] .

gs:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
