gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_029_CalculationCaseShape ;
        sh:message "Each linked calculation case must satisfy the section modulus and shear area constraint."
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be included in the section modulus calculation."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be excluded from the shear area calculation."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Flange material inclusion in the shear area must be explicitly represented."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:class nltl:structuralMember
        ] ;
        sh:message "The calculation case must identify its structural member."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:flangeMaterialIncludedInShearArea ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path (
                    nltl:sectionCalculationCaseStructuralMember
                    nltl:flangeFitted
                ) ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:flangeMaterialIncludedInShearArea ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ]
        ]
    ) .
