<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_029_CalculationCaseShape ;
        sh:message "Each represented calculation case must conform to the section calculation requirements."
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be included in the section modulus calculation."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be excluded from the shear area calculation."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:hasValue true ]
            [ sh:hasValue false ]
        ) ;
        sh:message "Flange-material inclusion in the shear area must be explicitly recorded as true or false."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:I2_029_StructuralMemberShape ;
        sh:message "The calculation case must identify its evaluated structural member."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When flange material is included in the shear area, the linked structural member must have a fitted flange." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:flangeMaterialIncludedInShearArea true .
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                FILTER NOT EXISTS {
                    ?member nltl:flangeFitted true .
                }
                FILTER NOT EXISTS {
                    ?member nltl:flangeFitted false .
                }
            }
        """
    ] .

gen:I2_029_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember .
<END_SHACL>
