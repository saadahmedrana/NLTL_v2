<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_029_CalculationCaseShape
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each calculation case must include attached shell plating in the section modulus."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each calculation case must exclude attached shell plating from the shear area."
    ] ;
    sh:or (
        gen:I2_029_FlangeIncludedCaseShape
        gen:I2_029_FlangeExcludedCaseShape
    ) .

gen:I2_029_FlangeIncludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "When flange material is included in the shear area, the value must be explicitly true."
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:node gen:I2_029_FittedStructuralMemberShape
    ] .

gen:I2_029_FlangeExcludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The calculation case must explicitly represent whether flange material is included in the shear area."
    ] .

gen:I2_029_FittedStructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each structural member linked to a calculation case including flange material in shear area must have a fitted flange."
    ] .
<END_SHACL>
