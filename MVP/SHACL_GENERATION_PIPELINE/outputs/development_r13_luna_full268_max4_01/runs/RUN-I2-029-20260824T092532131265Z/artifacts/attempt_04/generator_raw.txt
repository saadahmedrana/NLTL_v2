<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_029_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node gen:I2_029_CalculationCaseShape
    ] .

gen:I2_029_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be included in the section modulus calculation."
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Attached shell plating must be excluded from the shear area calculation."
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Flange-material inclusion in the shear area must be represented by exactly one boolean value."
    ] ;
    sh:or (
        [
            sh:node gen:I2_029_FlangeIncludedCaseShape
        ]
        [
            sh:node gen:I2_029_FlangeNotIncludedCaseShape
        ]
    ) .

gen:I2_029_FlangeIncludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:hasValue true ;
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:node gen:I2_029_FittedStructuralMemberShape ;
        sh:message "When flange material is included in shear area, every linked structural member must have a fitted flange."
    ] .

gen:I2_029_FlangeNotIncludedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:hasValue false ;
    ] .

gen:I2_029_FittedStructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:flangeFitted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The structural member linked to a flange-included calculation case must have a fitted flange."
    ] .
<END_SHACL>
