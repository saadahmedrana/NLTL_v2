<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:minCount 1 ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:minCount 1 ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an applicable area-exceeds branch, the calculation case shall provide the externally computed plastic neutral axis result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?member nltl:netLocalFrameFlangeArea ?areaNode .
                ?areaNode qudt:numericValue ?area .
                ?member nltl:netAttachedShellPlateThickness ?thicknessNode .
                ?thicknessNode qudt:numericValue ?thickness .
                ?member nltl:frameSpacing ?spacingNode .
                ?spacingNode qudt:numericValue ?spacing .
                FILTER (
                    (?area * 100) >
                    (1000 * ?thickness * ?spacing)
                )
                FILTER NOT EXISTS {
                    ?case nltl:plasticNeutralAxisHeight ?neutralAxis .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For oblique framing with 20 degrees less than Omega less than 70 degrees, the calculation case shall provide both interpolation endpoints." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?angleNode .
                ?angleNode qudt:numericValue ?angle .
                FILTER (?angle > 20 && ?angle < 70)
                {
                    FILTER NOT EXISTS {
                        ?case nltl:interpolationLowerEndpoint ?lower .
                    }
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        ?case nltl:interpolationUpperEndpoint ?upper .
                    }
                }
            }
        """
    ] .

gen:CalculationCaseShape a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape
    ] .

gen:StructuralMemberShape a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasLocalFrameSectionCalculationCase ;
        sh:minCount 1 ;
        sh:class nltl:localFrameSectionCalculationCase ;
        sh:node gen:LocalFrameSectionCalculationCaseShape
    ] ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:localFrameFlangeCentreHeight ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:netEffectivePlasticSectionModulus ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:CubicCentimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:netLocalFrameFlangeArea ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:SquareCentimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:webAngleToShellPlate ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DegreeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:webHeight ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:webToFlangeCentreDistance ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] .

gen:LocalFrameSectionCalculationCaseShape a sh:NodeShape ;
    sh:class nltl:localFrameSectionCalculationCase ;
    sh:property [
        sh:path nltl:sectionCalculationCasePlating ;
        sh:minCount 1 ;
        sh:class nltl:plating ;
        sh:node gen:PlatingShape
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:class nltl:structuralMember
    ] .

gen:PlatingShape a sh:NodeShape ;
    sh:class nltl:plating ;
    sh:property [
        sh:path nltl:framingAngleOmega ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DegreeQuantityShape
    ] .

gen:InterpolationPointShape a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue
    ] .

gen:DecimalQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] .

gen:MetreQuantityShape a sh:NodeShape ;
    sh:node gen:DecimalQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

<END_SHACL>
