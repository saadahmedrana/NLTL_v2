<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_031_StructuralMemberShape ;
        sh:message "Each represented structural member must satisfy the I2.6.1 local-frame constraints."
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:minCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:message "The ship must provide the patch load used for the applicable structural assessment."
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:class nltl:frameProfileTypeValue ;
        sh:message "The ship must identify the applicable frame profile type."
    ] .

gen:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:class nltl:structuralLocationValue ;
        sh:or (
            [
                sh:hasValue nltl:bottomStructureLocation ;
                sh:property [
                    sh:path (
                        nltl:hasStructuralMemberLoadCase
                        nltl:loadPatchOrientation
                    ) ;
                    sh:minCount 1 ;
                    sh:hasValue nltl:loadPatchHeightParallelToFrame ;
                    sh:message "For a bottom-structure member, the patch-load dimension b must be parallel with the frame direction."
                ]
            ]
            [
                sh:hasValue nltl:sideStructureLocation
            ]
        ) ;
        sh:message "A structural member must be identified as being in a bottom or side structure."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:node gen:I2_031_LoadCaseShape ;
        sh:message "The structural member must have a load case establishing its combined shear and bending demand."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:message "The structural member must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:node gen:I2_031_KiloNewtonQuantityShape ;
        sh:message "The structural member must provide the midspan load defining its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The structural member must identify the plastic collapse mechanism associated with the plastic-strength result."
    ] ;
    sh:sparql [
        sh:message "For every structural-member load case, combined shear and bending demand must not exceed plastic strength, and the midspan plastic-collapse load must agree with the stated plastic strength within a relative tolerance of 1e-6 plus an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                $this nltl:plasticStrength ?strengthQuantity .
                $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?strengthQuantity qudt:numericValue ?strength .
                ?collapseQuantity qudt:numericValue ?collapse .
                ?demandQuantity qudt:numericValue ?demand .
                FILTER (
                    ?demand > ?strength ||
                    ABS(?collapse - ?strength) >
                    (1e-6 * ABS(?strength) + 1e-9)
                )
            }
        """
    ] .

gen:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:node gen:I2_031_QuantityShape ;
        sh:message "The load case must provide the combined shear and bending demand."
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue ;
        sh:maxCount 1
    ] .

gen:I2_031_QuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] .

gen:I2_031_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .
<END_SHACL>
