<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-031-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must identify its frame profile type."
    ] ;
    sh:property [
        sh:path nltl:plasticCollapseMechanism ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record the plastic collapse mechanism."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node <urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A bottom structural member requires a ship-owned patch load with a decimal numeric value in kN." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:structuralLocation nltl:bottomStructureLocation .
    FILTER NOT EXISTS {
        $this nltl:patchLoad ?patchLoad .
        ?patchLoad qudt:numericValue ?value ;
                   qudt:unit qudt:KiloN .
        FILTER(datatype(?value) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The midspan plastic collapse load must define the plastic strength associated with the recorded collapse mechanism." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:plasticCollapseMechanism ?mechanism ;
          nltl:hasStructuralMember ?member .
    ?member nltl:midspanPlasticCollapseLoad ?midspanLoad ;
            nltl:plasticStrength ?strength .
    ?midspanLoad qudt:numericValue ?midspanValue ;
                 qudt:unit qudt:KiloN .
    ?strength qudt:numericValue ?strengthValue ;
              qudt:unit qudt:KiloN .
    FILTER(datatype(?midspanValue) = xsd:decimal)
    FILTER(datatype(?strengthValue) = xsd:decimal)
    FILTER(ABS(?midspanValue - ?strengthValue) >
           ((ABS(?strengthValue) * 0.000001) + 0.000001))
}
"""
    ] .

<urn:nltl:generated-shape:I2-031-QuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-031-StructuralMemberShape>
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must identify its structural location."
    ] ;
    sh:property [
        sh:path nltl:transverseFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must record its transverse frame orientation."
    ] ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must provide its combined shear and bending demand in kN."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must provide its plastic strength in kN."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node <urn:nltl:generated-shape:I2-031-QuantityValueShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each applicable structural member must provide its midspan plastic collapse load in kN."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:class nltl:loadCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-031-LoadCaseShape>
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:loadPatchOrientation ;
                        sh:hasValue nltl:loadPatchHeightParallelToFrame ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:loadPatchOrientation ;
                        sh:maxCount 0
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand must not exceed the plastic strength of the structural member." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:combinedShearAndBendingDemand ?demand ;
          nltl:plasticStrength ?strength .
    ?demand qudt:numericValue ?demandValue ;
            qudt:unit qudt:KiloN .
    ?strength qudt:numericValue ?strengthValue ;
              qudt:unit qudt:KiloN .
    FILTER(datatype(?demandValue) = xsd:decimal)
    FILTER(datatype(?strengthValue) = xsd:decimal)
    FILTER(?demandValue > ?strengthValue)
}
"""
    ] .

<urn:nltl:generated-shape:I2-031-LoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:class nltl:loadPatchOrientationValue ;
        sh:maxCount 1
    ] .
<END_SHACL>
