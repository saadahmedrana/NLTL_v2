<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue ;
        sh:minCount 1 ;
        sh:message "A frame profile type is required for the represented ship."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_StructuralMemberShape ;
        sh:message "Each represented structural member must provide the applicability information and applicable load-case checks."
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The longitudinal-frame orientation is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:message "The structural location is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:I2_034_PlasticStrengthQuantityShape ;
        sh:minCount 1 ;
        sh:message "An applicable structural member must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:I2_034_MidspanLoadQuantityShape ;
        sh:minCount 1 ;
        sh:message "An applicable structural member must provide the midspan plastic-collapse load defining its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_034_LoadCaseShape ;
        sh:message "Each represented structural-member load case must provide a combined shear-and-bending demand."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a longitudinal local frame in a side structure, the combined shear-and-bending demand must not exceed the member plastic strength or its midspan plastic-collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation ;
                      nltl:plasticStrength ?plasticStrength ;
                      nltl:midspanPlasticCollapseLoad ?midspanLoad ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .

                ?plasticStrength qudt:numericValue ?plasticStrengthValue .
                ?midspanLoad qudt:numericValue ?midspanLoadValue .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .

                FILTER (
                    ?demandValue > ?plasticStrengthValue ||
                    ?demandValue > ?midspanLoadValue
                )
            }
        """
    ] .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_034_DemandQuantityShape ;
        sh:minCount 1 ;
        sh:message "A represented load case must provide a combined shear-and-bending demand."
    ] .

gen:I2_034_PlasticStrengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_034_MidspanLoadQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .
<END_SHACL>
