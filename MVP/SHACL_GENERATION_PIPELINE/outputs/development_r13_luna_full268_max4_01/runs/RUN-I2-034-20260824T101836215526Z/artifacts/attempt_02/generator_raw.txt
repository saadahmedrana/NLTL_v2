<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_034_MemberDispatchShape ;
        sh:message "Each represented structural member must provide the applicability information needed to determine whether the side-structure rule applies."
    ] .

gen:I2_034_MemberDispatchShape
    a sh:NodeShape ;
    sh:and (
        gen:I2_034_MemberApplicabilityShape
        gen:I2_034_MemberBranchShape
    ) .

gen:I2_034_MemberApplicabilityShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A represented structural member must state whether it is longitudinally framed."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:class nltl:structuralLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A represented structural member must state its structural location."
    ] .

gen:I2_034_MemberBranchShape
    a sh:NodeShape ;
    sh:or (
        gen:I2_034_ApplicableMemberShape
        gen:I2_034_NonApplicableOrientationShape
        gen:I2_034_NonApplicableLocationShape
    ) .

gen:I2_034_ApplicableMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:sideStructureLocation
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:node gen:I2_034_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable side-structure longitudinal frame must provide its plastic strength."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:node gen:I2_034_KiloNewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable side-structure longitudinal frame must provide its midspan plastic-collapse load."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gen:I2_034_LoadCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For every represented load case of an applicable member, combined shear and bending demand must not exceed plastic strength, and the plastic strength must agree with the midspan plastic-collapse load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?plasticQuantity .
                ?plasticQuantity q:numericValue ?plastic .
                $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
                ?collapseQuantity q:numericValue ?collapse .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity q:numericValue ?demand .
                FILTER (
                    ?demand > ?plastic
                    ||
                    ABS(?plastic - ?collapse) >
                    (0.000001 * ABS(?plastic) + 0.000001)
                )
            }
        """
    ] .

gen:I2_034_NonApplicableOrientationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:hasValue false
    ] .

gen:I2_034_NonApplicableLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:hasValue nltl:bottomStructureLocation
    ] .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:node gen:I2_034_DemandQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Every represented load case of an applicable member must provide combined shear and bending demand."
    ] .

gen:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_034_KiloNewtonQuantityShape
    a sh:NodeShape ;
    sh:class q:QuantityValue ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
