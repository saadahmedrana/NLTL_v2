        sh:message "The actual net effective shear area must satisfy shearArea >= the calculated required longitudinal frame shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressureNode ;
                      nltl:peakPressureFactor ?peakNode ;
                      nltl:frameSpan ?spanNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:shearArea ?actualNode ;
                      nltl:hasStructuralMember ?member .
                ?pressureNode qudt:numericValue ?pressure .
                ?peakNode qudt:numericValue ?peak .
                ?spanNode qudt:numericValue ?span .
                ?yieldNode qudt:numericValue ?yieldStrength .
                ?actualNode qudt:numericValue ?actual .
                ?member nltl:frameSpacing ?spacingNode ;
                        nltl:selectedHullAreaFactor ?factorNode .
                ?spacingNode qudt:numericValue ?spacing .
                ?factorNode qudt:numericValue ?factor .
                BIND(?span AS ?a)
                BIND(?span AS ?b)
                BIND(?b / ?spacing AS ?bPrime)
                BIND(1 - (0.3 / ?bPrime) AS ?ko)
                BIND(IF(?bPrime < 2, ?b * (1 - (0.25 * ?bPrime)), ?spacing) AS ?b2)
                BIND(?ko * ?b2 AS ?b1)
                BIND((100 * 100 * (?factor * ?peak * ?pressure) * 0.5 * ?b1 * ?a) / (0.577 * ?yieldStrength) AS ?alCm2)
                BIND(?alCm2 * 0.0001 AS ?alM2)
                FILTER(?actual < ?alM2)
            }
        """
    ] .

gen:I2_035_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:frameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedHullAreaFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:requiredLongitudinalFrameShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:CentiM2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "requiredLongitudinalFrameShearArea must equal the calculated A_L within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameSpacing ?spacingNode ;
                      nltl:selectedHullAreaFactor ?factorNode ;
                      nltl:requiredLongitudinalFrameShearArea ?resultNode .
                ?spacingNode qudt:numericValue ?spacing .
                ?factorNode qudt:numericValue ?factor .
                ?resultNode qudt:numericValue ?reported .
                ^nltl:hasStructuralMember ?ship .
                ?ship nltl:averageIcePressure ?pressureNode ;
                      nltl:peakPressureFactor ?peakNode ;
                      nltl:frameSpan ?spanNode ;
                      nltl:yieldStrength ?yieldNode .
                ?pressureNode qudt:numericValue ?pressure .
                ?peakNode qudt:numericValue ?peak .
                ?spanNode qudt:numericValue ?span .
                ?yieldNode qudt:numericValue ?yieldStrength .
                BIND(?span AS ?a)
                BIND(?span AS ?b)
                BIND(?b / ?spacing AS ?bPrime)
                BIND(1 - (0.3 / ?bPrime) AS ?ko)
                BIND(IF(?bPrime < 2, ?b * (1 - (0.25 * ?bPrime)), ?spacing) AS ?b2)
                BIND(?ko * ?b2 AS ?b1)
                BIND((100 * 100 * (?factor * ?peak * ?pressure) * 0.5 * ?b1 * ?a) / (0.577 * ?yieldStrength) AS ?alCm2)
                BIND(?alCm2 * 0.0001 AS ?alM2)
                BIND(?alCm2 * 0.000001 + 0.000000001 AS ?tolerance)
                FILTER(?reported < (?alCm2 - ?tolerance) || ?reported > (?alCm2 + ?tolerance))
            }
        """
    ] .
