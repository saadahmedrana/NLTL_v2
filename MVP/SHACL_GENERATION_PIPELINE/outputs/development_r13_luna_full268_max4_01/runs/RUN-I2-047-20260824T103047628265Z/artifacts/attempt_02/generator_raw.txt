<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:SteelRenewalForIceStrengthenedStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path ( nltl:hasPlating nltl:iceStrengthenedStructureStatus ) ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "steelRenewalRequired must be true exactly when the ice-strengthened status is true and gaugedThickness is less than netThickness plus 0.0005 M; otherwise it must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:gaugedThickness ?gaugedQuantity ;
                      nltl:netThickness ?netQuantity ;
                      nltl:hasPlating ?plating ;
                      nltl:steelRenewalRequired ?renewal .

                ?gaugedQuantity qudt:numericValue ?gauged ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?netQuantity qudt:numericValue ?net ;
                             qudt:unit <http://qudt.org/vocab/unit/M> .
                ?plating nltl:iceStrengthenedStructureStatus ?iceStatus .

                FILTER (
                    (?iceStatus = true &&
                     ?gauged < (?net + 0.0005) &&
                     ?renewal != true)
                    ||
                    (?iceStatus = true &&
                     ?gauged >= (?net + 0.0005) &&
                     ?renewal != false)
                    ||
                    (?iceStatus = false &&
                     ?renewal != false)
                )
            }
        """
    ] .
<END_SHACL>
