<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_047_SteelRenewalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path (
            nltl:hasPlating
            nltl:iceStrengthenedStructureStatus
        ) ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice-strengthened structure status is required to determine steel renewal applicability."
    ] ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Gauged thickness must be a single metre-valued QUDT quantity."
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Net thickness must be a single metre-valued QUDT quantity."
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Steel renewal status must be recorded as a single boolean value."
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be true exactly when the ice-strengthened structure status is true and gauged thickness is less than net thickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating/nltl:iceStrengthenedStructureStatus ?iceStatus ;
                       nltl:gaugedThickness ?gaugedQuantity ;
                       nltl:netThickness ?netQuantity ;
                       nltl:steelRenewalRequired ?renewal .

                ?gaugedQuantity qudt:numericValue ?gauged ;
                                 qudt:unit <http://qudt.org/vocab/unit/M> .
                ?netQuantity qudt:numericValue ?net ;
                              qudt:unit <http://qudt.org/vocab/unit/M> .

                FILTER (
                    (?iceStatus = true &&
                     (?renewal != true ||
                      ! (?gauged < (?net + 0.0005))))
                    ||
                    (?iceStatus = false &&
                     (?renewal != false ||
                      (?gauged < (?net + 0.0005))))
                )
            }
        """
    ] .
<END_SHACL>
