                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;

gen:I2_051_CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:shearForceDirectionValue ;
        sh:hasValue nltl:positiveShearForce
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasCalculationCase
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:sparql [
                sh:message "For positive shear, a calculation case in the aft applicable interval must have a longitudinal distribution factor of exactly 0." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        ?ship nltl:hasCalculationCase $this ;
                              nltl:upperIceWaterlineLengthLUI ?lui .
                        $this nltl:hullGirderLongitudinalPositionFromAft ?position .
                        ?position qudt:numericValue ?p .
                        ?lui qudt:numericValue ?l .
                        BIND ((0.6 * ?l) AS ?upper)
                        FILTER (!(?p >= 0 && ?p <= ?upper))
                    }
                """
            ] ;
            sh:property [
                sh:path (
                    [
                        sh:inversePath nltl:hasCalculationCase
                    ]
                    nltl:longitudinalDistributionFactor
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "0"^^xsd:decimal ;
                        sh:maxInclusive "0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "For positive shear, a calculation case in the forward applicable interval must have a longitudinal distribution factor of exactly 1." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        ?ship nltl:hasCalculationCase $this ;
                              nltl:upperIceWaterlineLengthLUI ?lui .
                        $this nltl:hullGirderLongitudinalPositionFromAft ?position .
                        ?position qudt:numericValue ?p .
                        ?lui qudt:numericValue ?l .
                        BIND ((0.9 * ?l) AS ?lower)
                        FILTER (!(?p >= ?lower && ?p <= ?l))
                    }
                """
            ] ;
            sh:property [
                sh:path (
                    [
                        sh:inversePath nltl:hasCalculationCase
                    ]
                    nltl:longitudinalDistributionFactor
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "1"^^xsd:decimal ;
                        sh:maxInclusive "1"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "A positive-shear calculation case outside the two specified longitudinal intervals is not subject to those longitudinal distribution factor alternatives." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        ?ship nltl:hasCalculationCase $this ;
                              nltl:upperIceWaterlineLengthLUI ?lui .
                        $this nltl:hullGirderLongitudinalPositionFromAft ?position .
                        ?position qudt:numericValue ?p .
                        ?lui qudt:numericValue ?l .
                        BIND ((0.6 * ?l) AS ?aft)
                        BIND ((0.9 * ?l) AS ?forward)
                        FILTER (?p >= 0 && ?p <= ?aft)
                        FILTER (?p >= ?forward && ?p <= ?l)
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal the positive-shear longitudinal distribution factor times the design vertical ice force at bow, with MN converted to kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:designVerticalIceForceAtBow ?fib ;
                      nltl:designVerticalIceShearForce ?fi ;
                      nltl:longitudinalDistributionFactor ?cf .
                $this nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:hullGirderLongitudinalPositionFromAft ?position .
                ?fib qudt:numericValue ?fibValue .
                ?fi qudt:numericValue ?fiValue .
                ?cf qudt:numericValue ?cfValue .
                BIND ((?cfValue * ?fibValue * 1000) AS ?expected)
                BIND (IF((?expected * 0.000001) > 0.000001,
                         (?expected * 0.000001),
                         0.000001) AS ?tolerance)
                FILTER (?fiValue < (?expected - ?tolerance) ||
                        ?fiValue > (?expected + ?tolerance))
            }
        """
    ] .
</END_SHACL>
