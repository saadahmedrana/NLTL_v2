<urn:nltl:generated-shape:I2-055-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:MegaPA

        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

                sh:path qudt:numericValue ;

                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;

            ]

        a sh:SPARQLConstraint ;
        sh:message "Design stress must not exceed permissible stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

                    datatype(?designValue) = xsd:decimal &&
                    datatype(?permissibleValue) = xsd:decimal &&
                    ?designValue > ?permissibleValue
                )
            }
        """
    ] .
