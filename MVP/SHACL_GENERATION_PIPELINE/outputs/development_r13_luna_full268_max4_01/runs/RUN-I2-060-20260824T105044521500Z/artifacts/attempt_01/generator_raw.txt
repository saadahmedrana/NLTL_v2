<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_060_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_060_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A structural member in way of a cut-out marked as requiring stiffening must have structural stiffening recorded on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout ;
                        nltl:memberInWayOfCutout true ;
                        nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening
                }
            }
        """
    ] .

gen:I2_060_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCutout ;
        sh:node [
            sh:class nltl:cutout
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:memberInstability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ] ;
    sh:message "Each linked structural member must state whether it is in way of a cut-out; applicable members must not cause instability and must state whether stiffening is necessary." .
<END_SHACL>
