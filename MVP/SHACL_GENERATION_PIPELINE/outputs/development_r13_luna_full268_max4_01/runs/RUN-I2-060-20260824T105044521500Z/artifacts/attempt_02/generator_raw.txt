<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "The ship must record that applicable members do not cause instability."
    ] ;
    sh:sparql [
        sh:message "A member in way of a cut-out that requires stiffening must have structural stiffening recorded on the ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout ;
                        nltl:memberInWayOfCutout true ;
                        nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-StructuralMemberShape>
    a sh:NodeShape ;
    sh:targetClass nltl:structuralMember ;
    sh:property [
        sh:path nltl:memberInWayOfCutout ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Each structural member must explicitly record whether it is in way of a cut-out."
    ] ;
    sh:property [
        sh:path nltl:stiffeningNecessary ;
        sh:datatype xsd:boolean ;
        sh:message "Stiffening necessity must be recorded as a boolean."
    ] ;
    sh:sparql [
        sh:message "A member in way of a cut-out must record whether stiffening is necessary."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasCutout ?cutout ;
                      nltl:memberInWayOfCutout true .
                FILTER NOT EXISTS {
                    $this nltl:stiffeningNecessary ?necessary .
                    FILTER(?necessary = true)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2-060-ShipStiffeningShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string ;
        sh:message "Structural stiffening, when required by an applicable member, must be recorded on the ship."
    ] .
<END_SHACL>
