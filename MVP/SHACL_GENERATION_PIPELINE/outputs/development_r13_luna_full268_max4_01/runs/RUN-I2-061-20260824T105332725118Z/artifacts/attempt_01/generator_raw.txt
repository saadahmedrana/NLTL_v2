<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseShape ;
        sh:message "Each linked calculation case must identify a calculation method and its assessed hull structure."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationMethodValue ;
        sh:in (
            nltl:directCalculationMethodValue
            nltl:prescribedAnalyticalProcedure
        ) ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify a permitted calculation method."
    ] ;
    sh:property [
        sh:path nltl:prescribedAnalyticalProcedure ;
        sh:nodeKind sh:IRI ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:minCount 1 ;
        sh:message "A prescribed analytical procedure must be present; direct calculation must not replace it."
    ] ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gen:AssessedHullStructureShape ;
        sh:minCount 1 ;
        sh:message "The calculation case must identify its assessed hull structure."
    ] .

gen:AssessedHullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:node gen:ThicknessQuantityShape ;
        sh:minCount 1 ;
        sh:message "The assessed hull structure must provide its thickness."
    ] .

gen:ThicknessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
