<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_062_ComplexReadinessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The direct calculation result must be recorded as used when an applicable load carrying stringer or web frame forms part of a grillage system."
    ] ;
    sh:sparql [
        sh:message "A direct calculation must be recorded for load carrying stringers and web frames forming part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                ?member nltl:partOfGrillageSystem ?grillage .
                ?grillage a nltl:grillageSystem .
                FILTER (NOT EXISTS {
                    $this nltl:directCalculationUsed ?used .
                    FILTER (?used = true^^xsd:boolean)
                })
            }
        """
    ] .
<END_SHACL>
