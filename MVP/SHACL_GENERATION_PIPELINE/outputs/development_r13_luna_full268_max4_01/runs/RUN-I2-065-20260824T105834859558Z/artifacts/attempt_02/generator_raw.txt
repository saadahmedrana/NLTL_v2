<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:I2-065-ComplexReadiness> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each linked nonlinear calculation case must provide the required analysis evidence, acceptance result, deformation and structural-dimension inputs, and margins; the evidence must cover reliable buckling and plastic-deformation capture and suitable fracture, major-buckling, and yielding margins." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:hasCalculationCase ?case .
    ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .

    FILTER (
        NOT EXISTS {
            $this nltl:bucklingResponse ?bucklingResponse .
            FILTER (datatype(?bucklingResponse) = xsd:string)
        }
        ||
        NOT EXISTS {
            $this nltl:plasticDeformation ?plasticDeformation .
            FILTER (datatype(?plasticDeformation) = xsd:string)
        }
        ||
        NOT EXISTS {
            ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
            ?evidence a nltl:evidenceArtifact .
        }
        ||
        EXISTS {
            ?case nltl:nonlinearAnalysisAcceptanceEvidence ?evidence .
            FILTER NOT EXISTS { ?evidence a nltl:evidenceArtifact }
        }
        ||
        NOT EXISTS {
            ?case nltl:minorPermanentDeformationAssessmentSatisfied ?satisfied .
            FILTER (datatype(?satisfied) = xsd:boolean && ?satisfied = true)
        }
        ||
        NOT EXISTS {
            ?case nltl:fractureMargin ?fracture .
            ?fracture qudt:numericValue ?fractureNumber ;
                     qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
            FILTER (datatype(?fractureNumber) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?case nltl:majorBucklingMargin ?majorBuckling .
            ?majorBuckling qudt:numericValue ?majorBucklingNumber ;
                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
            FILTER (datatype(?majorBucklingNumber) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?case nltl:yieldingMargin ?yielding .
            ?yielding qudt:numericValue ?yieldingNumber ;
                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
            FILTER (datatype(?yieldingNumber) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?case nltl:permanentLateralDeformation ?lateral .
            ?lateral qudt:numericValue ?lateralNumber ;
                     qudt:unit <http://qudt.org/vocab/unit/MilliM> .
            FILTER (datatype(?lateralNumber) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?case nltl:permanentOutOfPlaneDeformation ?outOfPlane .
            ?outOfPlane qudt:numericValue ?outOfPlaneNumber ;
                        qudt:unit <http://qudt.org/vocab/unit/MilliM> .
            FILTER (datatype(?outOfPlaneNumber) = xsd:decimal)
        }
        ||
        NOT EXISTS {
            ?case nltl:relevantStructuralDimension ?dimension .
            ?dimension qudt:numericValue ?dimensionNumber ;
                       qudt:unit <http://qudt.org/vocab/unit/MilliM> .
            FILTER (datatype(?dimensionNumber) = xsd:decimal)
        }
    )
}
"""
    ] .
<END_SHACL>
