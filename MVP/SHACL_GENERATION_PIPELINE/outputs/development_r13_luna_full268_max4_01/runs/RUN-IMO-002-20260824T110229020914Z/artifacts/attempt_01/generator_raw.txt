<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO002ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare exactly one design ice condition."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must declare exactly one Polar Code ship category."
    ] ;
    sh:or (
        gen:IMO002NonApplicableShape
        gen:IMO002CategoryAShape
        gen:IMO002CategoryBShape
    ) .

gen:IMO002NonApplicableShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The conditional category-B rule is not applicable when the declared design ice condition is not thin first-year ice with possible old-ice inclusions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (?condition != nltl:thinFirstYearIceWithPossibleOldIceInclusions)
            }
        """
    ] .

gen:IMO002CategoryAShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:hasValue nltl:thinFirstYearIceWithPossibleOldIceInclusions
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryA
    ] .

gen:IMO002CategoryBShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:hasValue nltl:thinFirstYearIceWithPossibleOldIceInclusions
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:hasValue nltl:polarShipCategoryB
    ] ;
    sh:message "A ship not in category A that is designed to operate in at least thin first-year ice with possible old-ice inclusions must be category B." .
<END_SHACL>
