<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-002_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its design ice condition."
    ] ;
    sh:or ( _:applicableBranch _:nonApplicableBranch ) .

_:applicableBranch
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:in (
            nltl:thinFirstYearIceWithPossibleOldIceInclusions
            nltl:mediumFirstYearIceWithPossibleOldIceInclusions
        )
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:or ( _:categoryA _:categoryB ) ;
        sh:message "A ship designed for at least thin first-year ice must have category A or category B."
    ] .

_:categoryA
    sh:hasValue nltl:polarShipCategoryA .

_:categoryB
    sh:hasValue nltl:polarShipCategoryB .

_:nonApplicableBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The non-applicable branch cannot be used for a ship designed for at least thin first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?condition .
                FILTER (
                    ?condition IN (
                        nltl:thinFirstYearIceWithPossibleOldIceInclusions,
                        nltl:mediumFirstYearIceWithPossibleOldIceInclusions
                    )
                )
            }
        """
    ] .
<END_SHACL>
