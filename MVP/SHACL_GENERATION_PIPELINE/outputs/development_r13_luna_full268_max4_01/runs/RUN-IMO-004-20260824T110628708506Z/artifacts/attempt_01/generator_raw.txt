<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_004_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:node [
            sh:in ( nltl:seaIceOrigin )
        ]
    ] ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:node [
            sh:in ( nltl:firstYearIce nltl:mediumFirstYearIce nltl:thinFirstYearIce )
        ]
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A ship meeting the sea-ice, winter-growth, and thickness conditions must be classified as first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin .
                $this nltl:winterGrowthCount ?growthCount .
                FILTER ( ?growthCount <= 1 ) .
                $this nltl:iceThickness ?thicknessNode .
                ?thicknessNode qudt:numericValue ?thickness .
                FILTER ( ?thickness >= 0.3 && ?thickness <= 2.0 ) .
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
<END_SHACL>
