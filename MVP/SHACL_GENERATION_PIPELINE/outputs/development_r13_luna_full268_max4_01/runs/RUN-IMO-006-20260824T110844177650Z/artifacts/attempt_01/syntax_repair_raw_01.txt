@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_006_MediumFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:mediumFirstYearIce ;
                sh:property [
                    sh:path nltl:iceThickness ;
                    sh:minCount 1 ;
                    sh:node [
                        sh:class qudt:QuantityValue ;
                        sh:property [
                            sh:path qudt:numericValue ;
                            sh:datatype xsd:decimal ;
                            sh:minCount 1 ;
                            sh:maxCount 1 ;
                            sh:minInclusive "0.7"^^xsd:decimal ;
                            sh:maxInclusive "1.2"^^xsd:decimal
                        ] ;
                        sh:property [
                            sh:path qudt:unit ;
                            sh:hasValue unit:M ;
                            sh:minCount 1 ;
                            sh:maxCount 1
                        ]
                    ]
                ]
            ]
            [
                sh:in (
                    nltl:firstYearIce
                    nltl:thinFirstYearIce
                )
            ]
        ) ;
        sh:message "Ice type classification must identify a permitted ice type, with medium first-year ice requiring thickness from 0.7 m through 1.2 m."
    ] ;
    sh:sparql [
        sh:message "Medium first-year ice requires an ice thickness between 0.7 m and 1.2 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
                FILTER NOT EXISTS {
                    $this nltl:iceThickness ?quantity .
                    ?quantity qudt:numericValue ?value .
                    ?quantity qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.7"^^xsd:decimal &&
                        ?value <= "1.2"^^xsd:decimal
                    )
                }
            }
        """
    ] .
