<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice condition classification."
    ] ;
    sh:sparql [
        sh:message "When the ice condition is open water, sea ice concentration must be a decimal less than 1 with unit UNITLESS, and ice of land origin must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:iceConditionClassification
                        nltl:openWaterIceCondition .
                }
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?quantity .
                        ?quantity qudt:numericValue ?value ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?value) = xsd:decimal &&
                            ?value < 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?badQuantity .
                        FILTER NOT EXISTS {
                            ?badQuantity qudt:numericValue ?validValue ;
                                         qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                            FILTER (
                                datatype(?validValue) = xsd:decimal &&
                                ?validValue < 1
                            )
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent ?landOriginIce .
                        FILTER (
                            datatype(?landOriginIce) = xsd:boolean &&
                            ?landOriginIce = false
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
