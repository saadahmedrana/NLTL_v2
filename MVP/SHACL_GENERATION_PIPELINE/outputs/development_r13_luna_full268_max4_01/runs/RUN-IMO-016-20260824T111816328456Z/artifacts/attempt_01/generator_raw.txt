<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-016> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:polarShipCertificateForm ;
            sh:property [
                sh:path nltl:certificateFormModel ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in ( nltl:appendixIPolarShipCertificateModel )
            ]
        ] ;
        sh:message "Each ship shall have a Polar Ship Certificate corresponding to the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The certificate language is required to determine whether an approved translation is required."
    ] ;
    sh:sparql [
        sh:message "If the certificate language is neither English, French, nor Spanish, an approved translation shall be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:certificateLanguage ?language .
                FILTER (
                    ?language NOT IN (
                        "English"^^xsd:string,
                        "French"^^xsd:string,
                        "Spanish"^^xsd:string
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:approvedTranslationPresent ?translation .
                    FILTER ( ?translation = true )
                }
            }
        """
    ] .
<END_SHACL>
