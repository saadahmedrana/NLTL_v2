<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate validity date is required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A survey date is required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An endorsement date is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:class nltl:polarShipCertificateForm ;
        sh:message "A Polar Ship Certificate is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:minCount 1 ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017SupplementShape ;
        sh:message "A Polar Certificate supplement with required equipment records is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:minCount 1 ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasScheduleShape ;
        sh:message "The applicable SOLAS certificate schedule is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate validity, survey, and endorsement dates must match the corresponding SOLAS certificate schedule dates." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER (
                    EXISTS {
                        $this nltl:polarCertificateValidityDates ?polarDate .
                        FILTER NOT EXISTS {
                            $this nltl:hasSolasCertificateSchedule ?schedule .
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                    nltl:certificateScheduleDate ?polarDate .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:surveyDates ?polarDate .
                        FILTER NOT EXISTS {
                            $this nltl:hasSolasCertificateSchedule ?schedule .
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                    nltl:certificateScheduleDate ?polarDate .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:endorsementDates ?polarDate .
                        FILTER NOT EXISTS {
                            $this nltl:hasSolasCertificateSchedule ?schedule .
                            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                            ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                    nltl:certificateScheduleDate ?polarDate .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                                nltl:certificateScheduleDate ?scheduleDate .
                        FILTER NOT EXISTS {
                            $this nltl:polarCertificateValidityDates ?scheduleDate .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                                nltl:certificateScheduleDate ?scheduleDate .
                        FILTER NOT EXISTS {
                            $this nltl:surveyDates ?scheduleDate .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:hasSolasCertificateSchedule ?schedule .
                        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                                nltl:certificateScheduleDate ?scheduleDate .
                        FILTER NOT EXISTS {
                            $this nltl:endorsementDates ?scheduleDate .
                        }
                    }
                )
            }
        """
    ] .

gen:IMO017SupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:message "The Polar Certificate supplement must record the required equipment."
    ] .

gen:IMO017SolasScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:minCount 1 ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017ScheduleDateRecordShape ;
        sh:message "The SOLAS certificate schedule must contain certificate schedule date records."
    ] .

gen:IMO017ScheduleDateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record must contain an xsd:date value."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "Each certificate schedule date record must have a permitted date category."
    ] .
<END_SHACL>
