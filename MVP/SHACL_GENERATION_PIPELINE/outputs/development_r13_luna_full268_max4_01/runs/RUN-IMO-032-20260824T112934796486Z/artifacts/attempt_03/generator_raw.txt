<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO032ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO032CategoryABBranch
        gen:IMO032CategoryCStrengthenedBranch
        gen:IMO032CategoryCNotStrengthenedBranch
    ) .

gen:IMO032CategoryABBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:class nltl:polarShipCategoryValue ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
        ) ;
        sh:minCount 1 ;
        sh:message "A category A or B ship is required for this approval branch."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:node gen:IMO032CategoryABApprovalRecordShape ;
        sh:minCount 1 ;
        sh:message "Category A or B requires an approval record with an authorized authority and an acceptable or equivalent-safety standard."
    ] .

gen:IMO032CategoryCStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:class nltl:polarShipCategoryValue ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
        sh:message "This branch applies to a category C ship."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "Category C applicability requires recorded ice strengthening."
    ] ;
    sh:property [
        sh:path nltl:hasApprovalRecord ;
        sh:node gen:IMO032CategoryCApprovalRecordShape ;
        sh:minCount 1 ;
        sh:message "An ice-strengthened category C ship requires an approval record addressing operating ice type and concentration."
    ] .

gen:IMO032CategoryCNotStrengthenedBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:class nltl:polarShipCategoryValue ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:minCount 1 ;
        sh:message "This branch applies to a category C ship."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:message "The non-applicable category C branch requires recorded absence of ice strengthening."
    ] .

gen:IMO032CategoryABApprovalRecordShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval must be by the Administration or an accepted recognized organization."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "standard acceptable to the Organization"
            "another standard offering an equivalent level of safety"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval standard must be acceptable to the Organization or offer an equivalent level of safety."
    ] .

gen:IMO032CategoryCApprovalRecordShape
    a sh:NodeShape ;
    sh:class nltl:approvalRecord ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:hasValue nltl:evidenceStateApproved ;
        sh:minCount 1 ;
        sh:message "The scantling approval status must be approved."
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval must be by the Administration or an accepted recognized organization."
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:in (
            "acceptable standard adequate for operating ice type and concentration"
        ) ;
        sh:minCount 1 ;
        sh:message "The approval standard must be acceptable and adequate for the operating ice type and concentration."
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The approval record must specify the operating ice type."
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:node gen:IMO032IceConcentrationQuantityShape ;
        sh:message "The approval record must specify the operating ice concentration as a unitless quantity."
    ] .

gen:IMO032IceConcentrationQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
