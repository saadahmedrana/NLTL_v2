gs:IMO_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:class nltl:polarShipCategoryValue ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one permitted polar ship category."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one construction date."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date
            ] ;
            sh:sparql [
                sh:message "For category A or B, the construction date must be on or after 1 January 2017." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionDate ?date .
                        FILTER (
                            YEAR(?date) < 2017
                            ||
                            (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) < 1)
                            ||
                            (YEAR(?date) = 2017 && MONTH(?date) < 1)
                        )
                    }
                """
            ] ;
            sh:property [
                sh:path nltl:hasLoadingConditionCase ;
                sh:class nltl:loadingConditionCase ;
                sh:minCount 1 ;
                sh:node gs:IMO_037_LoadingConditionCaseShape ;
                sh:message "Each applicable ship must provide at least one loading condition case."
            ]
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:shipCategory ;
                        sh:hasValue nltl:polarShipCategoryC
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:shipCategory ;
                        sh:in (
                            nltl:polarShipCategoryA
                            nltl:polarShipCategoryB
                        )
                    ] ;
                    sh:property [
                        sh:path nltl:constructionDate ;
                        sh:datatype xsd:date
                    ] ;
                    sh:sparql [
                        sh:message "For category A or B, a construction date before 1 January 2017 is outside this requirement's applicability." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl/vocab#>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                $this nltl:constructionDate ?date .
                                FILTER (
                                    YEAR(?date) > 2017
                                    ||
                                    (YEAR(?date) = 2017 && MONTH(?date) > 1)
                                    ||
                                    (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
                                )
                            }
                        """
                    ]
                )
            )
        ]
    ) .
