<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-047-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasEssentialEngine ;
        sh:class nltl:essentialEngine ;
        sh:minCount 1 ;
        sh:message "The ship must identify its essential engine machinery."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Combustion air temperature must be supplied as one degree-Celsius quantity."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:DEG_C ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion must be supplied in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion air temperature must be within the manufacturer's minimum and maximum criteria." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:combustionAirTemperature ?temperatureNode .
    ?temperatureNode qudt:numericValue ?temperature .
    $this nltl:manufacturerMinAirTemperature ?minimumNode .
    ?minimumNode qudt:numericValue ?minimum .
    $this nltl:manufacturerMaxAirTemperature ?maximumNode .
    ?maximumNode qudt:numericValue ?maximum .
    FILTER (?temperature < ?minimum || ?temperature > ?maximum)
}
"""
    ] .
<END_SHACL>
