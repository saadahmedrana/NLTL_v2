<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

<urn:nltl:generated-shape:IMO-049-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall identify the machinery component scope for the scantling approval requirement."
    ] ;
    sh:or (
        <urn:nltl:generated-shape:IMO-049-CategoryA>
        <urn:nltl:generated-shape:IMO-049-CategoryB>
        <urn:nltl:generated-shape:IMO-049-CategoryCStrengthened>
    ) .

<urn:nltl:generated-shape:IMO-049-CategoryA>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryA ;
        sh:class nltl:polarShipCategoryValue ;
        sh:message "A Category A ship shall identify its ship category as Category A."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:message "An applicable Category A ship shall have machinery components covered by scantling approval evidence."
    ] ;
    sh:sparql [
        sh:message "Each machinery component of a Category A ship shall have an approved scantling approval record with an authorized authority and an acceptable or equivalent-safety standard." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record nltl:scantlingApprovalStatus nltl:evidenceStateApproved .
                    ?record nltl:approvingAuthority ?authority .
                    ?record nltl:approvalStandard ?standard .
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER (
                        CONTAINS(LCASE(STR(?standard)), "acceptable") ||
                        CONTAINS(LCASE(STR(?standard)), "equivalent") ||
                        CONTAINS(LCASE(STR(?standard)), "equivalent level of safety")
                    )
                }
            }
        """
    ] .

<urn:nltl:generated-shape:IMO-049-CategoryCStrengthened>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryC ;
        sh:class nltl:polarShipCategoryValue ;
        sh:message "An applicable ship shall identify its ship category as Category C."
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Category C applicability requires the ship to be identified as ice-strengthened."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:message "An applicable ice-strengthened Category C ship shall have machinery components covered by scantling approval evidence."
    ] ;
    sh:sparql [
        sh:message "Each machinery component of an ice-strengthened Category C ship shall have approved scantling evidence from an authorized authority, using an acceptable standard adequate for the operating ice type and concentration." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                FILTER NOT EXISTS {
                    ?component nltl:hasApprovalRecord ?record .
                    ?record nltl:scantlingApprovalStatus nltl:evidenceStateApproved .
                    ?record nltl:approvingAuthority ?authority .
                    ?record nltl:approvalStandard ?standard .
                    ?record nltl:operatingIceType ?iceType .
                    ?record nltl:operatingIceConcentration ?concentration .
                    ?concentration qudt:numericValue ?numericValue ;
                                   qudt:unit unit:UNITLESS .
                    FILTER (DATATYPE(?numericValue) = xsd:decimal)
                    FILTER (
                        STR(?authority) = "Administration" ||
                        STR(?authority) = "recognized organization accepted by the Administration"
                    )
                    FILTER (
                        CONTAINS(LCASE(STR(?standard)), "acceptable") &&
                        CONTAINS(LCASE(STR(?standard)), "adequate")
                    )
                }
            }
        """
    ] .
