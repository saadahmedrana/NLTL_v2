<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .

nltl:IMO049Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in (
            "propeller blade"
            "propulsion line"
            "steering equipment"
            "appendage"
        )
    ] ;
    sh:or (
        _:imo049BranchA
        _:imo049BranchB
        _:imo049BranchC
    ) ;
    sh:message "Each applicable machinery component shall have approved scantling evidence under the applicable polar ship category policy." .

_:imo049BranchA
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryA
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:qualifiedValueShape [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node nltl:IMO049CategoryABComponentApprovalShape
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] .

_:imo049BranchB
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryB
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:qualifiedValueShape [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node nltl:IMO049CategoryABComponentApprovalShape
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] .

_:imo049BranchC
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:polarShipCategoryC
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:qualifiedValueShape [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:hasApprovalRecord ;
                sh:minCount 1 ;
                sh:node nltl:IMO049CategoryCComponentApprovalShape
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] .

nltl:IMO049CategoryABComponentApprovalShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:machineryComponentType ;
        sh:datatype xsd:string ;
        sh:minCount 0 ;
        sh:in (
            "propeller blade"
            "propulsion line"
            "steering equipment"
            "appendage"
        )
    ] ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        )
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:pattern "acceptable" ;
                sh:flags "i"
            ]
            [
                sh:pattern "equivalent" ;
                sh:flags "i"
            ]
        )
    ] ;
    sh:message "Category A or B machinery component approval shall identify an approved status, authorized authority, and an acceptable or equivalent-safety standard." .

nltl:IMO049CategoryCComponentApprovalShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:scantlingApprovalStatus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:evidenceStateApproved
    ] ;
    sh:property [
        sh:path nltl:approvingAuthority ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            "Administration"
            "recognized organization accepted by the Administration"
        )
    ] ;
    sh:property [
        sh:path nltl:approvalStandard ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:pattern "acceptable" ;
        sh:flags "i" ;
        sh:not [
            sh:pattern "equivalent" ;
            sh:flags "i"
        ]
    ] ;
    sh:property [
        sh:path nltl:operatingIceType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingIceConcentration ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:UNITLESS
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Category C approvalStandard shall be acceptable and adequate for the stated operating ice type and concentration, without an equivalent-safety alternative." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:approvalStandard ?standard ;
                      nltl:operatingIceType ?iceType ;
                      nltl:operatingIceConcentration ?quantity .
                ?quantity qudt:numericValue ?concentration ;
                          qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (
                    !CONTAINS(LCASE(STR(?standard)), "acceptable") ||
                    !CONTAINS(LCASE(STR(?standard)), "adequate") ||
                    !CONTAINS(LCASE(STR(?standard)), LCASE(STR(?iceType))) ||
                    !CONTAINS(LCASE(STR(?standard)), STR(?concentration))
                )
            }
        """
    ] ;
    sh:message "Category C machinery component approval shall include approved status, authorized authority, and operating ice context." .
<END_SHACL>
