<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO_070_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each person on a passenger ship must have an assigned immersion suit or thermal protective aid with size compatibility status true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:passengerShip ;
                      nltl:hasPersonOnBoardMember ?person .
                FILTER NOT EXISTS {
                    ?person nltl:hasPersonalSurvivalEquipmentAssignment ?assignment .
                    ?assignment a nltl:personalSurvivalEquipmentAssignment ;
                               nltl:assignedPersonalSurvivalEquipment ?equipment ;
                               nltl:sizeCompatibilityStatus ?status .
                    ?equipment a nltl:personalSurvivalEquipment ;
                               a ?equipmentType .
                    FILTER (
                        ?equipmentType IN (
                            nltl:immersionSuitEquipment,
                            nltl:thermalProtectiveAidEquipment
                        )
                    )
                    FILTER (
                        datatype(?status) = xsd:boolean &&
                        ?status = true
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
