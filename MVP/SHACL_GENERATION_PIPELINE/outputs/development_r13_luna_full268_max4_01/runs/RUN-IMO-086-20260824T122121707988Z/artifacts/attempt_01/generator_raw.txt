<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO086ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node gen:IMO086HeadingMeansShape ;
        sh:message "The ship must have at least two non-magnetic means to determine and display heading."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means must be independent from another distinct non-magnetic heading means of the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:hasNonMagneticHeadingMeans ?means .
                    FILTER NOT EXISTS {
                        ?means nltl:independentFromHeadingMeans ?other .
                        FILTER (?other != ?means)
                        $this nltl:hasNonMagneticHeadingMeans ?other .
                    }
                }
            }
        """
    ] .

gen:IMO086HeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
    ] .
<END_SHACL>
