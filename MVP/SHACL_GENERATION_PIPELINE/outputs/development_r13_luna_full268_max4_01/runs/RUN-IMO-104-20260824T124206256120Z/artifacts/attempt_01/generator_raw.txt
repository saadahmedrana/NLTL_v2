<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO104ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization." 
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO104CrewMemberShape
    ] ;
    sh:property [
        sh:path nltl:hasCrewMember ;
        sh:node gen:IMO104CrewMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each crew member must have familiarization records covering every PWOM procedure or equipment item relevant to assigned duties." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this (nltl:hasCrewMemberInventory|nltl:hasCrewMember) ?crew .
                ?crew nltl:hasAssignedDuty ?duty .
                ?duty nltl:hasRelevantPolarWaterOperationalManualItem ?item .
                FILTER NOT EXISTS {
                    ?crew nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?item .
                }
            }
        """
    ] .

gen:IMO104CrewMemberShape
    a sh:NodeShape ;
    sh:class nltl:crewMember ;
    sh:property [
        sh:path nltl:hasAssignedDuty ;
        sh:minCount 1 ;
        sh:node gen:IMO104AssignedDutyShape ;
        sh:message "Each crew member must have an assigned duty." 
    ] ;
    sh:property [
        sh:path nltl:hasFamiliarizationRecord ;
        sh:minCount 1 ;
        sh:node gen:IMO104FamiliarizationRecordShape ;
        sh:message "Each crew member must have a PWOM familiarization record." 
    ] .

gen:IMO104AssignedDutyShape
    a sh:NodeShape ;
    sh:class nltl:assignedDuty ;
    sh:property [
        sh:path nltl:hasRelevantPolarWaterOperationalManualItem ;
        sh:minCount 1 ;
        sh:node gen:IMO104ManualItemShape ;
        sh:message "Each assigned duty must identify its relevant PWOM procedure or equipment." 
    ] .

gen:IMO104FamiliarizationRecordShape
    a sh:NodeShape ;
    sh:class nltl:familiarizationRecord ;
    sh:property [
        sh:path nltl:familiarizationRecordItem ;
        sh:minCount 1 ;
        sh:node gen:IMO104ManualItemShape ;
        sh:message "A familiarization record must identify the covered PWOM procedure or equipment item." 
    ] .

gen:IMO104ManualItemShape
    a sh:NodeShape ;
    sh:class nltl:polarWaterOperationalManualItem .

<END_SHACL>
