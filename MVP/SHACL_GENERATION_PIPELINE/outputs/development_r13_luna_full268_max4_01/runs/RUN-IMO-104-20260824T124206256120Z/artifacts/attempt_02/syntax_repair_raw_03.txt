<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-104-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarWaterOperationalManualFamiliarizationStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ship must record completed PWOM familiarization status."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each crew member's assigned duty must have familiarization coverage for every relevant PWOM procedure or equipment item." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:hasCrewMemberInventory ?crewMember .
                ?crewMember nltl:hasAssignedDuty ?assignedDuty .
                ?assignedDuty nltl:hasRelevantPolarWaterOperationalManualItem ?manualItem .
                FILTER NOT EXISTS {
                    ?crewMember nltl:hasFamiliarizationRecord ?record .
                    ?record nltl:familiarizationRecordItem ?manualItem .
                }
            }
        """
    ] .
<END_SHACL>
