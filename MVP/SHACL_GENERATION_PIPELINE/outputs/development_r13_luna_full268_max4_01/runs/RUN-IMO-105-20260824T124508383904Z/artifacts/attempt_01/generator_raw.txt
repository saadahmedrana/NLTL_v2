<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_105_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "An operating area is required to determine whether the Arctic-waters discharge prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, discharge of oil or oily mixtures to the sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER (STR(?area) = "Arctic_waters")
                FILTER (
                    NOT EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                        FILTER (?discharge = false)
                    }
                    ||
                    EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?nonConformingDischarge .
                        FILTER (?nonConformingDischarge != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
