<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-106> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:minCount 1 ;
        sh:class nltl:dischargedMaterialTypeValue ;
        sh:message "A ship must provide its discharged material classification to evaluate the Polar Code oil-discharge exception."
    ] ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For clean ballast or segregated ballast discharge, the paragraph 1.1.1 oil-discharge prohibition must be recorded as not applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material ;
                      nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER (
                    !(
                        ?applicable = false
                        && DATATYPE(?applicable) = xsd:boolean
                    )
                )
            }
        """
    ] .
<END_SHACL>
