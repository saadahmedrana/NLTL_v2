<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-106-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The oil-discharge prohibition applicability result must be recorded as a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:dischargedMaterialClassification ;
        sh:class nltl:dischargedMaterialTypeValue
    ] ;
    sh:sparql [
        sh:message "When the discharged material is clean ballast or segregated ballast, the oil-discharge prohibition applicability result must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material = nltl:cleanBallastMaterial ||
                    ?material = nltl:segregatedBallastMaterial
                )
                $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable ?applicable .
                FILTER (?applicable != "false"^^xsd:boolean)
            }
        """
    ] .
<END_SHACL>
