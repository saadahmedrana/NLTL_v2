<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO108Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:oilFuelTankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For applicable Category A or B ships, every reported oil-fuel tank separation must be at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date ;
                      nltl:aggregateOilFuelCapacity ?aggregate ;
                      nltl:individualOilFuelTankCapacity ?individual .

                FILTER (?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))

                FILTER (
                    YEAR(?date) > 2017
                    || (YEAR(?date) = 2017 && MONTH(?date) > 1)
                    || (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
                )

                ?aggregate qudt:numericValue ?aggregateValue .
                ?individual qudt:numericValue ?individualValue .

                FILTER (?aggregateValue < 600)
                FILTER (?individualValue > 30)

                FILTER (
                    NOT EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?anySeparation .
                    }
                    ||
                    EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?reportedSeparation .
                        ?reportedSeparation qudt:numericValue ?separationValue .
                        FILTER (?separationValue < 0.76)
                    }
                )
            }
        """
    ] .
<END_SHACL>
