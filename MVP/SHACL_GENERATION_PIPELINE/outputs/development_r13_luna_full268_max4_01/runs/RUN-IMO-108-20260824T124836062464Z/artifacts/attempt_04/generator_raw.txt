<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-108> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:aggregateOilFuelCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:individualOilFuelTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:oilFuelTankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B ship, every reported oil fuel tank outer shell separation must be at least 0.76 m, and a separation value must be reported." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship .
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category IN (
                        nltl:polarShipCategoryA,
                        nltl:polarShipCategoryB
                    )
                )

                $this nltl:constructionDate ?constructionDate .
                FILTER (datatype(?constructionDate) = xsd:date)
                FILTER (YEAR(?constructionDate) > 2017 ||
                        YEAR(?constructionDate) = 2017)

                $this nltl:aggregateOilFuelCapacity ?aggregateQuantity .
                ?aggregateQuantity qudt:numericValue ?aggregateValue .
                ?aggregateQuantity qudt:unit unit:M3 .
                FILTER (?aggregateValue < "600"^^xsd:decimal)

                $this nltl:individualOilFuelTankCapacity ?individualQuantity .
                ?individualQuantity qudt:numericValue ?individualValue .
                ?individualQuantity qudt:unit unit:M3 .
                FILTER (?individualValue > "30"^^xsd:decimal)

                FILTER (
                    NOT EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?reportedSeparation
                    }
                    ||
                    EXISTS {
                        $this nltl:oilFuelTankOuterShellSeparation ?inadequateSeparation .
                        ?inadequateSeparation qudt:numericValue ?separationValue .
                        ?inadequateSeparation qudt:unit unit:M .
                        FILTER (?separationValue < "0.76"^^xsd:decimal)
                    }
                )
            }
        """
    ] .
<END_SHACL>
