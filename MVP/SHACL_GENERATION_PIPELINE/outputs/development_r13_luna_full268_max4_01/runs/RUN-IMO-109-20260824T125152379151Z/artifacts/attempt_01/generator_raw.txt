<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-109-DirectStatic> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:cargoTankCarriesOil ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "For an applicable Category A or B non-oil-tanker ship constructed on or after 1 January 2017 with cargo tanks carrying oil, cargo tank outer shell separation must be at least 0.76 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )

                $this nltl:shipType ?shipType .
                FILTER (datatype(?shipType) = xsd:string)

                FILTER NOT EXISTS {
                    $this a nltl:oilTanker .
                }

                $this nltl:constructionDate ?constructionDate .
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )

                $this nltl:cargoTankCarriesOil ?carriesOil .
                FILTER (
                    datatype(?carriesOil) = xsd:boolean &&
                    ?carriesOil = true
                )

                FILTER NOT EXISTS {
                    $this nltl:cargoTankOuterShellSeparation ?separation .
                    ?separation a qudt:QuantityValue ;
                        qudt:numericValue ?value ;
                        qudt:unit unit:M .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= "0.76"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
