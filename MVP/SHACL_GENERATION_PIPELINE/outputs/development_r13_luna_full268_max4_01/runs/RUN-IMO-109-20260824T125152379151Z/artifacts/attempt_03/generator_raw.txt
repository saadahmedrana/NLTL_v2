<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-109-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship type is required and must be an xsd:string."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The construction date must be on or after 1 January 2017."
    ] ;
    sh:property [
        sh:path nltl:cargoTankCarriesOil ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The cargo tank must be recorded as carrying oil."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An oil tanker is not within the applicability of this requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true .
                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS { $this a nltl:oilTanker . }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable ship must have at least one cargo-tank outer-shell separation measurement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true .
                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS { $this a nltl:oilTanker . }
                FILTER NOT EXISTS {
                    $this nltl:cargoTankOuterShellSeparation ?separation .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every applicable cargo-tank separation measurement must be a decimal quantity in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true ;
                      nltl:cargoTankOuterShellSeparation ?separation .
                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS { $this a nltl:oilTanker . }
                ?separation ?predicate ?value .
                FILTER (
                    (?predicate = qudt:unit && ?value != unit:M) ||
                    (?predicate = qudt:numericValue && datatype(?value) != xsd:decimal)
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every applicable cargo-tank separation measurement must be at least 0.76 metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:shipType ?shipType ;
                      nltl:constructionDate ?constructionDate ;
                      nltl:cargoTankCarriesOil true ;
                      nltl:cargoTankOuterShellSeparation ?separation .
                FILTER (?category IN (nltl:polarShipCategoryA, nltl:polarShipCategoryB))
                FILTER (?constructionDate >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS { $this a nltl:oilTanker . }
                ?separation qudt:numericValue ?value ;
                             qudt:unit unit:M .
                FILTER (?value < "0.76"^^xsd:decimal)
            }
        """
    ] .
<END_SHACL>
