<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship category must be exactly one canonical Polar Ship Category value."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single valid construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A single deadweight quantity in metric tonnes is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:class nltl:oilTanker ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2017-01-01"^^xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:deadweight ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:maxExclusive "5000"^^xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:TON_Metric ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must have exactly one true double-bottom protection status."
            ] ;
            sh:property [
                sh:path nltl:wingTankProtectionStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "An applicable ship must have exactly one true wing-tank protection status."
            ]
        ]
        [
            sh:sparql [
                sh:message "Protection-status requirements apply only when the ship is a Category A or B oil tanker of less than 5,000 tonnes deadweight constructed on or after 1 January 2017." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this a nltl:oilTanker ;
                              nltl:shipCategory ?category ;
                              nltl:constructionDate ?date ;
                              nltl:deadweight ?deadweight .
                        ?deadweight qudt:numericValue ?value ;
                                    qudt:unit unit:TON_Metric .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                        )
                        FILTER (datatype(?date) = xsd:date)
                        FILTER (?date >= "2017-01-01"^^xsd:date)
                        FILTER (datatype(?value) = xsd:decimal)
                        FILTER (?value < "5000"^^xsd:decimal)
                    }
                """
            ]
        ]
    ) .

</END_SHACL>
