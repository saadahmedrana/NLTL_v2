<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_110_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ship category must be exactly one of Category A or Category B."
    ] ;
    sh:property [
        sh:path nltl:shipType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ship type must be supplied as exactly one string value."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Construction date must be supplied as exactly one date value."
    ] ;
    sh:property [
        sh:path nltl:deadweight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:TON_Metric ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Deadweight must be exactly one metric-tonne quantity value."
    ] ;
    sh:or (
        gen:IMO_110_Applicable
        gen:IMO_110_NonApplicable
    ) .

gen:IMO_110_Applicable
    a sh:NodeShape ;
    sh:class nltl:oilTanker ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB ) ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minInclusive "2017-01-01"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:entireCargoTankLengthDoubleBottomProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable oil tankers must have exactly one true entire-cargo-tank-length double-bottom protection status."
    ] ;
    sh:property [
        sh:path nltl:wingTankProtectionStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable oil tankers must have exactly one true wing-tank protection status."
    ] ;
    sh:sparql [
        sh:message "An applicable oil tanker must have deadweight strictly less than 5000 metric tonnes." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this
            WHERE {
                $this nltl:deadweight ?quantity .
                ?quantity qudt:numericValue ?value ;
                          qudt:unit unit:TON_Metric .
                FILTER (?value >= 5000)
            }
        """
    ] .

gen:IMO_110_NonApplicable
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:not [
                    sh:in ( nltl:polarShipCategoryA nltl:polarShipCategoryB )
                ]
            ]
        ]
        [
            sh:not [
                sh:class nltl:oilTanker
            ]
        ]
        [
            sh:property [
                sh:path nltl:deadweight ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "5000"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:TON_Metric
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionDate ;
                sh:maxInclusive "2016-12-31"^^xsd:date
            ]
        ]
    ) .
<END_SHACL>
