<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO115Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:IMO115Applicable
        gen:IMO115NotApplicable
    ) ;
    sh:sparql [
        sh:message "The sewage discharge type and comminuted/disinfected applicability indicator must agree." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:comminutedAndDisinfected ?b ;
                      nltl:sewageDischargeType ?t .
                FILTER (
                    !(
                        (?b = true && STR(?t) = "comminuted_and_disinfected")
                        ||
                        (?b = false && STR(?t) != "comminuted_and_disinfected")
                    )
                )
            }
        """
    ] .
