<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:Chapter9Dash1ApplicabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:FishingVesselChapter9Dash1Inputs
        gen:PleasureYachtChapter9Dash1Inputs
        gen:CargoShipChapter9Dash1Inputs
    ) ;
    sh:property [
        sh:path nltl:polarCodeChapter9Dash1Applicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Chapter 9-1 applicability must be recorded as a single boolean value."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The Chapter 9-1 applicability result does not match the ship type, dimensions, gross tonnage, and trade status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                BIND(
                    (
                        EXISTS {
                            $this rdf:type nltl:fishingVessel .
                            $this nltl:lengthOverall ?lengthQuantity .
                            ?lengthQuantity qudt:numericValue ?lengthValue .
                            FILTER(?lengthValue >= 24)
                        }
                        ||
                        EXISTS {
                            $this rdf:type nltl:pleasureYacht .
                            $this nltl:grossTonnage ?pleasureTonnage .
                            ?pleasureTonnage qudt:numericValue ?pleasureValue .
                            $this nltl:engagedInTrade false .
                            FILTER(?pleasureValue >= 300)
                        }
                        ||
                        EXISTS {
                            $this rdf:type nltl:cargoShip .
                            $this nltl:grossTonnage ?cargoTonnage .
                            ?cargoTonnage qudt:numericValue ?cargoValue .
                            FILTER(?cargoValue >= 300 && ?cargoValue < 500)
                        }
                    )
                    AS ?expected
                )
                $this nltl:polarCodeChapter9Dash1Applicable ?actual
                FILTER(?actual != ?expected)
            }
        """
    ] .

gen:FishingVesselChapter9Dash1Inputs
    a sh:NodeShape ;
    sh:class nltl:fishingVessel ;
    sh:property [
        sh:path nltl:lengthOverall ;
        sh:node gen:LengthOverallQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A fishing vessel requires one length-overall value for Chapter 9-1 applicability."
    ] .

gen:PleasureYachtChapter9Dash1Inputs
    a sh:NodeShape ;
    sh:class nltl:pleasureYacht ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:node gen:GrossTonnageQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A pleasure yacht requires one gross-tonnage value for Chapter 9-1 applicability."
    ] ;
    sh:property [
        sh:path nltl:engagedInTrade ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A pleasure yacht requires one trade-status value for Chapter 9-1 applicability."
    ] .

gen:CargoShipChapter9Dash1Inputs
    a sh:NodeShape ;
    sh:class nltl:cargoShip ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:node gen:GrossTonnageQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A cargo ship requires one gross-tonnage value for Chapter 9-1 applicability."
    ] .

gen:LengthOverallQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:GrossTonnageQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
