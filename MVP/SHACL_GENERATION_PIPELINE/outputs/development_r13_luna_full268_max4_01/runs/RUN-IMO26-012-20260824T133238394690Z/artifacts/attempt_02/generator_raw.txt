<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26_012_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:IMO26_012_BelowThresholdShape
        gs:IMO26_012_ApplicableShape
    ) .

gs:IMO26_012_BelowThresholdShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:maxExclusive 500
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:UNITLESS
            ]
        ]
    ] .

gs:IMO26_012_ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:minInclusive 500
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue qudt:UNITLESS
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:qualifiedValueShape gs:IMO26_012_HeadingMeansShape ;
        sh:qualifiedMinCount 2 ;
        sh:message "At least two typed non-magnetic heading means are required for ships of 500 gross tonnage and upwards."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every typed non-magnetic heading means shall be connected to both main and emergency power." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                ?means a nltl:nonMagneticHeadingMeans .
                FILTER (
                    NOT EXISTS {
                        ?means nltl:connectedToMainPower ?mainPower .
                        ?mainPower a nltl:benchmarkEntity .
                    }
                    ||
                    NOT EXISTS {
                        ?means nltl:connectedToEmergencyPower ?emergencyPower .
                        ?emergencyPower a nltl:benchmarkEntity .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every typed non-magnetic heading means shall be independent from another distinct typed means belonging to the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                ?means a nltl:nonMagneticHeadingMeans .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    FILTER (?otherMeans != ?means)
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                    ?otherMeans a nltl:nonMagneticHeadingMeans .
                }
            }
        """
    ] .

gs:IMO26_012_HeadingMeansShape
    a sh:NodeShape ;
    sh:class nltl:nonMagneticHeadingMeans ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity
    ] .
<END_SHACL>
