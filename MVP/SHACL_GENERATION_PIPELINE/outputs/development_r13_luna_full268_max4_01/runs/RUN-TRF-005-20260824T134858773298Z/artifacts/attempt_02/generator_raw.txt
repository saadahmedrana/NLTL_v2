<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF005Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:sparql [
        sh:message "Ships with a construction stage date on or after 2003-09-01 and a construction contract date before 2010-01-01 must specify exactly the 2002 Ice Class Regulations edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate .
                $this nltl:constructionContractDate ?contractDate .
                FILTER (
                    ?stageDate >= "2003-09-01"^^xsd:date &&
                    ?contractDate < "2010-01-01"^^xsd:date
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:applicableIceClassRegulationEdition
                            nltl:iceClassRegulationEdition2002 .
                    }
                    ||
                    EXISTS {
                        $this nltl:applicableIceClassRegulationEdition ?edition .
                        FILTER (
                            ?edition != nltl:iceClassRegulationEdition2002
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
