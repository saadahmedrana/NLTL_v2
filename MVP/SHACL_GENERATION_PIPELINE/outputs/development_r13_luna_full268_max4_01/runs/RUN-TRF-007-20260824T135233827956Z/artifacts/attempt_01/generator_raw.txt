<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-007> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A ship must provide a permitted Finnish-Swedish ice-class value."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction-stage date is required to determine applicability of this rule."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA Super or IA ship constructed before 1 September 2003, the ship must provide a permitted 2017 compliance method and an assessment date no later than 1 January of the twentieth year after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .

                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa
                    )
                )
                FILTER (
                    datatype(?constructionDate) = xsd:date &&
                    ?constructionDate < "2003-09-01"^^xsd:date
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:deliveryDate ?deliveryDate .
                        FILTER (datatype(?deliveryDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:assessmentDate ?assessmentDate .
                        FILTER (datatype(?assessmentDate) = xsd:date)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:engineOutputComplianceMethod ?method .
                        FILTER (
                            ?method IN (
                                nltl:traficom2017Section3Point2Point2Method,
                                nltl:traficom2017Section3Point2Point4Method
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:deliveryDate ?deliveryDate ;
                              nltl:assessmentDate ?assessmentDate .

                        FILTER (
                            datatype(?deliveryDate) = xsd:date &&
                            datatype(?assessmentDate) = xsd:date
                        )

                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20
                            ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                (
                                    MONTH(?assessmentDate) > 1
                                    ||
                                    (
                                        MONTH(?assessmentDate) = 1 &&
                                        DAY(?assessmentDate) > 1
                                    )
                                )
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
