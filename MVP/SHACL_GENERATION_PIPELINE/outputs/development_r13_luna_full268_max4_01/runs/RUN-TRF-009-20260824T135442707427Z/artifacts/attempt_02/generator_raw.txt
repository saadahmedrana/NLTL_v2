<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF009ApplicableShape
        gen:TRF009NonApplicableShape
    ) ;
    sh:message "The ship must satisfy the applicable or non-applicable TRF-009 ice-class branch." .

gen:TRF009ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must have ice class IA or IA Super." 
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:maxInclusive "2003-08-31"^^xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An IA or IA Super ship must have a construction stage date before 1 September 2003."
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must have a delivery date for the twenty-year deadline calculation."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must have an assessment date."
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An applicable ship must identify compliance with section 3.2.2 or 3.2.4."
    ] ;
    sh:sparql [
        sh:message "The assessment date must not be later than 1 January in the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:deliveryDate ?delivery ;
                      nltl:assessmentDate ?assessment .
                FILTER (
                    YEAR(?assessment) > YEAR(?delivery) + 20
                    ||
                    (
                        YEAR(?assessment) = YEAR(?delivery) + 20
                        &&
                        (
                            MONTH(?assessment) > 1
                            ||
                            (
                                MONTH(?assessment) = 1
                                &&
                                DAY(?assessment) > 1
                            )
                        )
                    )
                )
            }
        """
    ] .

gen:TRF009NonApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A ship must identify its ice class to determine TRF-009 applicability."
    ] ;
    sh:sparql [
        sh:message "The non-applicable branch requires an ice class other than IA or IA Super." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIa,
                        nltl:iceClassIaSuper
                    )
                )
            }
        """
    ] .
<END_SHACL>
