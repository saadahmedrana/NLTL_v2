<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-011-PointShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    ] .

<urn:nltl:generated-shape:TRF-011-LengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-PointShape>
    ] .

<urn:nltl:generated-shape:TRF-011-Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF-011-IceWaterlineShape>
    ] ;
    sh:sparql [
        sh:message "The upper ice waterline must equal, pointwise, the maximum vertical coordinate of the intended ice-operating waterlines at every longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                {
                    {
                        SELECT ?ship ?x (MAX(?intendedY) AS ?maximumY)
                        WHERE {
                            ?ship nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .
                            ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                            ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                                           nltl:verticalCoordinate ?intendedYQuantity .
                            ?intendedXQuantity qudt:numericValue ?x ;
                                               qudt:unit <http://qudt.org/vocab/unit/M> .
                            ?intendedYQuantity qudt:numericValue ?intendedY ;
                                               qudt:unit <http://qudt.org/vocab/unit/M> .
                        }
                        GROUP BY ?ship ?x
                    }
                    FILTER (?ship = $this)
                    FILTER NOT EXISTS {
                        $this nltl:hasUpperIceWaterline ?upperWaterline .
                        ?upperWaterline nltl:hasWaterlineProfilePoint ?upperPoint .
                        ?upperPoint nltl:longitudinalPosition ?upperXQuantity ;
                                    nltl:verticalCoordinate ?upperYQuantity .
                        ?upperXQuantity qudt:numericValue ?upperX ;
                                        qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?upperYQuantity qudt:numericValue ?upperY ;
                                        qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (?upperX = ?x && ?upperY = ?maximumY)
                    }
                }
                UNION
                {
                    $this nltl:hasUpperIceWaterline ?upperWaterline2 .
                    ?upperWaterline2 nltl:hasWaterlineProfilePoint ?upperPoint2 .
                    ?upperPoint2 nltl:longitudinalPosition ?upperXQuantity2 ;
                                  nltl:verticalCoordinate ?upperYQuantity2 .
                    ?upperXQuantity2 qudt:numericValue ?upperX2 ;
                                    qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?upperYQuantity2 qudt:numericValue ?upperY2 ;
                                    qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER NOT EXISTS {
                        $this nltl:hasIntendedIceOperatingWaterline ?intendedWaterline2 .
                        ?intendedWaterline2 nltl:hasWaterlineProfilePoint ?intendedPoint2 .
                        ?intendedPoint2 nltl:longitudinalPosition ?intendedXQuantity2 ;
                                        nltl:verticalCoordinate ?intendedYQuantity2 .
                        ?intendedXQuantity2 qudt:numericValue ?intendedX2 ;
                                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        ?intendedYQuantity2 qudt:numericValue ?intendedY3 ;
                                            qudt:unit <http://qudt.org/vocab/unit/M> .
                        FILTER (?intendedX2 = ?upperX2 && ?intendedY3 = ?upperY2)
                    }
                }
            }
        """
    ] .
<END_SHACL>
