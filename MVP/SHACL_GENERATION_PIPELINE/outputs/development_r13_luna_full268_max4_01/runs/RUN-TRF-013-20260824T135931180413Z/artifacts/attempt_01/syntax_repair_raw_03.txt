<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:LengthQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF013ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The ship must explicitly be recorded as ice-strengthened."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "An ice class is required for an ice-strengthened ship."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:message "The upper ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:message "The lower ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The operating fore draught is required in metres."
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The operating aft draught is required in metres."
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtFore ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The maximum ice-class fore draught is required in metres."
    ] ;
    sh:property [
        sh:path nltl:maximumIceClassDraughtAft ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The maximum ice-class aft draught is required in metres."
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtFore ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The minimum ice-class fore draught is required in metres."
    ] ;
    sh:property [
        sh:path nltl:minimumIceClassDraughtAft ;
        sh:minCount 1 ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:message "The minimum ice-class aft draught is required in metres."
    ] ;
    sh:sparql [
        sh:message "Operating fore and aft draughts must each be between their corresponding minimum and maximum ice-class draughts." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingForeDraught ?opF ;
                      nltl:operatingAftDraught ?opA ;
                      nltl:maximumIceClassDraughtFore ?maxF ;
                      nltl:maximumIceClassDraughtAft ?maxA ;
                      nltl:minimumIceClassDraughtFore ?minF ;
                      nltl:minimumIceClassDraughtAft ?minA .
                ?opF qudt:numericValue ?opFN .
                ?opA qudt:numericValue ?opAN .
                ?maxF qudt:numericValue ?maxFN .
                ?maxA qudt:numericValue ?maxAN .
                ?minF qudt:numericValue ?minFN .
                ?minA qudt:numericValue ?minAN .
                FILTER (
                    ?opFN < ?minFN ||
                    ?opFN > ?maxFN ||
                    ?opAN < ?minAN ||
                    ?opAN > ?maxAN
                )
            }
        """
    ] .
</END_SHACL>
