        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A class certificate with all maximum and minimum ice-class draughts is required."
    ] ;
    sh:property [
        sh:path nltl:hasIceDraughtRestrictionDocument ;
        sh:class nltl:iceDraughtRestrictionDocument ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice-draught restriction document is required."
    ] ;
    sh:property [
        sh:path nltl:readilyAvailableToMaster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice-draught restriction document must be readily available to the master."
    ] ;
    sh:property [
        sh:path nltl:retainedOnBoard ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The ice-draught restriction document must be retained on board."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The required warning triangle and ice-class draught mark must be present when the applicable date and draught conditions require them." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:assessmentDate ?assessment .
                $this nltl:constructionDate ?construction .
                $this nltl:summerLoadLineFreshWaterDraught ?summerNode .
                ?summerNode qudt:numericValue ?summer .
                $this nltl:upperIceWaterlineDraught ?uiwlNode .
                ?uiwlNode qudt:numericValue ?uiwl .

                FILTER (
                    (
                        ?construction >= "2007-07-01"^^xsd:date &&
                        ?summer > ?uiwl &&
                        NOT EXISTS {
                            $this nltl:warningTrianglePresent true .
                            $this nltl:iceClassDraughtMarkPresent true .
                            $this nltl:iceClassDraughtMarkDraughtAmidships ?markNode .
                            ?markNode qudt:numericValue ?markValue .
                            ?markNode qudt:unit unit:M
                        }
                    )
                    ||
                    (
                        ?construction < "2007-07-01"^^xsd:date &&
                        NOT EXISTS {
                            $this nltl:warningTrianglePresent true .
                            $this nltl:iceClassDraughtMarkPresent true .
                            $this nltl:iceClassDraughtMarkDraughtAmidships ?markNode2 .
                            ?markNode2 qudt:numericValue ?markValue2 .
                            ?markNode2 qudt:unit unit:M
                        } &&
                        (
                            NOT EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDocking .
                            }
                            ||
                            EXISTS {
                                $this nltl:firstScheduledDryDockingDate ?dryDockingDue .
                                FILTER (?dryDockingDue <= ?assessment)
                            }
                        )
                    )
                )
            }
        """
    ] .
