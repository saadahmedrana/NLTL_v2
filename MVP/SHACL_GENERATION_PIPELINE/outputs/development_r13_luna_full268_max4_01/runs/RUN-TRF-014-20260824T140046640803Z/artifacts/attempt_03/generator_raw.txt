gen:TRF014PostCutoffMarking
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a ship built on or after 1 July 2007 with the summer load line above the UIWL, the warning triangle and ice-class draught mark must be present and the amidships mark must equal the certificate maximum amidships draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summer ;
                      nltl:upperIceWaterlineDraught ?uiwl .
                ?summer qudt:numericValue ?summerValue .
                ?uiwl qudt:numericValue ?uiwlValue .
                FILTER (
                    ?constructionDate >= "2007-07-01"^^xsd:date &&
                    ?summerValue > ?uiwlValue
                )
                FILTER NOT EXISTS {
                    $this nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?mark ;
                          nltl:hasClassCertificate ?certificate .
                    ?mark qudt:numericValue ?markValue ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?certificate nltl:maximumIceClassDraughtAmidships ?certificateMark .
                    ?certificateMark qudt:numericValue ?certificateValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (?markValue = ?certificateValue)
                }
            }
        """
    ] .

        sh:message "For a pre-1 July 2007 ship with the summer load line above the UIWL, complete marking evidence is required unless the docking-date alternative is satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summer ;
                      nltl:upperIceWaterlineDraught ?uiwl .
                ?summer qudt:numericValue ?summerValue .
                ?uiwl qudt:numericValue ?uiwlValue .
                FILTER (
                    ?constructionDate < "2007-07-01"^^xsd:date &&
                    ?summerValue > ?uiwlValue
                )
                FILTER NOT EXISTS {
                    $this nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?mark ;
                          nltl:hasClassCertificate ?certificate .
                    ?mark qudt:numericValue ?markValue ;

    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the first scheduled dry docking date is on or before the supplied assessment date, the required marking evidence must be present and match the certificate maximum amidships draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summer ;
                      nltl:upperIceWaterlineDraught ?uiwl ;
                      nltl:firstScheduledDryDockingDate ?dockingDate ;
                      nltl:assessmentDate ?assessmentDate .
                ?summer qudt:numericValue ?summerValue .
                ?uiwl qudt:numericValue ?uiwlValue .
                FILTER (
                    ?constructionDate < "2007-07-01"^^xsd:date &&
                    ?summerValue > ?uiwlValue &&
                    ?dockingDate <= ?assessmentDate
                )
                FILTER NOT EXISTS {
                    $this nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?mark ;
                          nltl:hasClassCertificate ?certificate .
