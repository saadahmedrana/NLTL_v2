<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF016Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:draughtAmidships ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:liwlDraughtAmidships ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:forwardDraught ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:levelIceThickness ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:displacementAtUpperIceWaterline ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerHighestPointSubmerged ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path (
                    nltl:hasUpperIceWaterline
                    nltl:waterlineDisplacement
                ) ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/TON_Metric> ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasBallastTank ;
                sh:class nltl:ballastTank ;
                sh:node gen:TRF016BallastTankShape
            ] ;
            sh:sparql [
                sh:message "Draught and forward-draught constraints must use the greatest-displacement upper ice waterline." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:hasUpperIceWaterline ?uiwl ;
                              nltl:displacementAtUpperIceWaterline ?shipDisplacement ;
                              nltl:levelIceThickness ?iceThickness ;
                              nltl:draughtAmidships ?draught ;
                              nltl:liwlDraughtAmidships ?liwlDraught ;
                              nltl:forwardDraught ?forward ;
                              nltl:propellerHighestPointSubmerged ?submerged .

                        ?uiwl nltl:waterlineDisplacement ?uiwlDisplacement .

                        ?shipDisplacement qudt:numericValue ?delta .
                        ?iceThickness qudt:numericValue ?hi .
                        ?draught qudt:numericValue ?draughtValue .
                        ?liwlDraught qudt:numericValue ?liwlValue .
                        ?forward qudt:numericValue ?forwardValue .
                        ?uiwlDisplacement qudt:numericValue ?uiwlDelta .

                        FILTER NOT EXISTS {
                            $this nltl:hasUpperIceWaterline ?otherUiwl .
                            ?otherUiwl nltl:waterlineDisplacement ?otherDisplacement .
                            ?otherDisplacement qudt:numericValue ?otherDelta .
                            FILTER (?otherDelta > ?uiwlDelta)
                        } .

                        FILTER (
                            ABS(?delta - ?uiwlDelta) >
                            (0.000001 * ABS(?uiwlDelta) + 0.000001)
                            ||
                            ?draughtValue < ?liwlValue
                            ||
                            ?submerged != true
                            ||
                            ?forwardValue +
                            (
                                0.000001 *
                                ABS(
                                    MIN(
                                        ((2 + (0.00025 * ?delta)) * ?hi),
                                        (4 * ?hi)
                                    )
                                ) + 0.000001
                            )
                            <
                            MIN(
                                ((2 + (0.00025 * ?delta)) * ?hi),
                                (4 * ?hi)
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:navigatingInIce ;
                sh:hasValue false
            ]
        ]
    ) .

    sh:property [

        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:usedToReachLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:situatedAboveLowerIceWaterline ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:usedToReachLowerIceWaterline ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:freezingPreventionPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:situatedAboveLowerIceWaterline ;
                sh:hasValue false ;
            ]
        ]
        [
            sh:property [
                sh:path nltl:usedToReachLowerIceWaterline ;
                sh:hasValue false ;
            ]
        ]
    ) .
<END_SHACL>
