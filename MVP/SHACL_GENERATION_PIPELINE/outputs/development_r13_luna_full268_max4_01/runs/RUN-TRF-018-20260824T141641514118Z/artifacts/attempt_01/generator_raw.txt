<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF018> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceClass must be one of IA Super, IA, IB, or IC."
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "calculatedRequiredPower must be a single kW quantity with a decimal numeric value."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "maximumContinuousRatingPower must be a single kW quantity with a decimal numeric value."
    ] ;
    sh:sparql [
        sh:message "maximumContinuousRatingPower must be at least calculatedRequiredPower and at least 1000 kW for IA, IB, or IC, or at least 2800 kW for IA Super." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedQuantity ;
                      nltl:maximumContinuousRatingPower ?ratingQuantity .

                ?calculatedQuantity qudt:numericValue ?calculatedPower ;
                                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?ratingQuantity qudt:numericValue ?ratingPower ;
                                qudt:unit <http://qudt.org/vocab/unit/KiloW> .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        (?ratingPower < ?calculatedPower || ?ratingPower < 2800)
                    )
                    ||
                    (
                        (
                            ?iceClass = nltl:iceClassIa ||
                            ?iceClass = nltl:iceClassIb ||
                            ?iceClass = nltl:iceClassIc
                        ) &&
                        (?ratingPower < ?calculatedPower || ?ratingPower < 1000)
                    )
                )
            }
        """
    ] .
<END_SHACL>
