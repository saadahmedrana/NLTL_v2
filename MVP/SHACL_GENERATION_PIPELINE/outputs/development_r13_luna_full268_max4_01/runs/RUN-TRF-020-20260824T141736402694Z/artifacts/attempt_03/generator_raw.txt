<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlg: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .

nltlg:TRF020ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltlg:TRF020ApplicableBranch
        nltlg:TRF020NonApplicableBranch
    ) .

nltlg:TRF020ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:property [
        sh:path nltl:engineOutputCoefficientKe ;
        sh:node nltlg:TRF020DimensionlessQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
        sh:node nltlg:TRF020UnspecifiedUnitQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
        sh:node nltlg:TRF020UnspecifiedUnitQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node nltlg:TRF020LengthQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in ( 1 2 3 )
    ] ;
    sh:property [
        sh:path nltl:propellerPitchControlType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:controllablePitch
            nltl:fixedPitch
        )
    ] ;
    sh:property [
        sh:path nltl:propulsionSystemType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:conventionalPropulsionSystem
            nltl:electricPropulsionMachinery
            nltl:hydraulicPropulsionMachinery
        )
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node [
            sh:class nltl:iceWaterline
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node nltlg:TRF020UpperWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node nltlg:TRF020PowerQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
        sh:node nltlg:TRF020PowerQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
        sh:node nltlg:TRF020PowerQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For an applicable ship, maximum continuous rating power must be at least both externally calculated minimum powers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrNode .
                ?mcrNode qudt:numericValue ?mcr .
                $this nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerNode .
                ?lowerNode qudt:numericValue ?lower .
                $this nltl:minimumRequiredPowerAtUpperIceWaterline ?upperNode .
                ?upperNode qudt:numericValue ?upper .
                FILTER (
                    ?mcr < ?lower ||
                    ?mcr < ?upper
                )
            }
        """
    ] .

nltlg:TRF020NonApplicableBranch
    a sh:NodeShape ;
    sh:or (
        nltlg:TRF020NonApplicableByDate
        nltlg:TRF020NonApplicableByIceClass
    ) .

    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxExclusive "2003-09-01"^^xsd:date
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlg:TRF020NonApplicableByIceClass
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:not [
            sh:in (
                nltl:iceClassIaSuper
                nltl:iceClassIa
                nltl:iceClassIb
                nltl:iceClassIc
            )
        ]
    ] .
