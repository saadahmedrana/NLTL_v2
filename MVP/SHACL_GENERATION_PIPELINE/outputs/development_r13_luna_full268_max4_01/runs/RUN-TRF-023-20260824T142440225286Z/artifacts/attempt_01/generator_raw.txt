<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-023-ship>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IB or IC ship under the 1985 ice class rules, maximum continuous rating power is required in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition nltl:iceClassRuleEdition1985 .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER NOT EXISTS {
                    $this nltl:maximumContinuousRatingPower ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    FILTER (DATATYPE(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IB or IC ship under the 1985 ice class rules, required power under the 1985 rules is required in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition nltl:iceClassRuleEdition1985 .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                FILTER NOT EXISTS {
                    $this nltl:requiredPowerUnder1985Rules ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                    FILTER (DATATYPE(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Maximum continuous rating power must be at least the required power under the 1985 ice class rules." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition nltl:iceClassRuleEdition1985 ;
                      nltl:maximumContinuousRatingPower ?maximumQuantity ;
                      nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                FILTER (?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                ?maximumQuantity qudt:numericValue ?maximumValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?requiredQuantity qudt:numericValue ?requiredValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                FILTER (DATATYPE(?maximumValue) = xsd:decimal)
                FILTER (DATATYPE(?requiredValue) = xsd:decimal)
                FILTER (?maximumValue < ?requiredValue)
            }
        """
    ] .
<END_SHACL>
