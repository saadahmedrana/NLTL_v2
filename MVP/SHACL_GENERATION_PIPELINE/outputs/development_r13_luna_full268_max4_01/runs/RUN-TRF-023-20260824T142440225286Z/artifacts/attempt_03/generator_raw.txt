<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF023ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IB or IC ship subject to the 1985 ice class rules, maximum continuous rating power must be present in kW and must be at least the required power under the 1985 rules, also in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRuleEdition1985 .

                FILTER (
                    ?iceClass = nltl:iceClassIb ||
                    ?iceClass = nltl:iceClassIc
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?maximumQuantity .
                        ?maximumQuantity qudt:numericValue ?maximumValue ;
                                           qudt:unit unit:KiloW .
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                        ?requiredQuantity qudt:numericValue ?requiredValue ;
                                          qudt:unit unit:KiloW .
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?maximumQuantity .
                        ?maximumQuantity qudt:numericValue ?maximumValue ;
                                           qudt:unit unit:KiloW .
                        $this nltl:requiredPowerUnder1985Rules ?requiredQuantity .
                        ?requiredQuantity qudt:numericValue ?requiredValue ;
                                          qudt:unit unit:KiloW .
                        FILTER ( ?maximumValue < ?requiredValue )
                    }
                )
            }
        """
    ] .
<END_SHACL>
