<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF024ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "The ship must have an allowed Finnish-Swedish ice class."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:deliveryDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:assessmentDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "For an IA Super or IA ship constructed before 1 September 2003, the assessment date must not be later than 1 January of the year twenty years after delivery." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER (
                            ?iceClass = nltl:iceClassIaSuper ||
                            ?iceClass = nltl:iceClassIa
                        )
                        $this nltl:constructionStageDate ?stageDate .
                        $this nltl:deliveryDate ?deliveryDate .
                        $this nltl:assessmentDate ?assessmentDate .
                        FILTER (
                            YEAR(?stageDate) < 2003 ||
                            (
                                YEAR(?stageDate) = 2003 &&
                                MONTH(?stageDate) < 9
                            )
                        )
                        FILTER (
                            YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                MONTH(?assessmentDate) > 1
                            ) ||
                            (
                                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                )
            ] ;
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "An IA Super or IA ship constructed on or after 1 September 2003 is outside the conditional deadline in this requirement." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER (
                            ?iceClass = nltl:iceClassIaSuper ||
                            ?iceClass = nltl:iceClassIa
                        )
                        $this nltl:constructionStageDate ?stageDate .
                        FILTER (
                            YEAR(?stageDate) < 2003 ||
                            (
                                YEAR(?stageDate) = 2003 &&
                                MONTH(?stageDate) < 9
                            )
                        )
                    }
                """
            ]
        )
    ] .
<END_SHACL>
