<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix nltlgen: <urn:nltl:generated-shape:> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltlgen:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltlgen:TRF031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldPoint ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:combinedBendingShearStress ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldShearStress ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:vonMisesYieldCriterion ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "von Mises yield criterion"
    ] ;
    sh:property [
        sh:path nltl:beamTheoryUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:beamTheoryUsed ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:allowableShearStress ;
                sh:node nltlgen:PressureQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Combined bending and shear stress must be lower than the yield point." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:combinedBendingShearStress ?combinedNode ;
                      nltl:yieldPoint ?yieldNode .
                ?combinedNode qudt:numericValue ?combined .
                ?yieldNode qudt:numericValue ?yield .
                FILTER (?combined >= ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Yield shear stress must equal yield point divided by square root of three within the permitted numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldPoint ?yieldNode ;
                      nltl:yieldShearStress ?shearNode .
                ?yieldNode qudt:numericValue ?yield .
                ?shearNode qudt:numericValue ?shear .
                BIND (?yield / 1.7320508075688772 AS ?expected)
                FILTER (ABS(?shear - ?expected) >
                        (ABS(?expected) * 0.000001 + 0.000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "When beam theory is used, allowable shear stress must not exceed 0.9 times yield shear stress." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:beamTheoryUsed true ;
                      nltl:allowableShearStress ?allowableNode ;
                      nltl:yieldPoint ?yieldNode .
                ?allowableNode qudt:numericValue ?allowable .
                ?yieldNode qudt:numericValue ?yield .
                BIND ((?yield / 1.7320508075688772) * 0.9 AS ?limit)
                FILTER (?allowable > ?limit)
            }
        """
    ] .
<END_SHACL>
