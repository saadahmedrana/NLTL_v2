<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ScantlingQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

gen:TRF032Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:regulationRequiredScantling ;
                sh:node gen:ScantlingQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:classificationSocietyRequiredScantling ;
                sh:node gen:ScantlingQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:selectedDesignScantling ;
                sh:node gen:ScantlingQuantityShape ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "When the regulation-required scantling is less than the classification-society-required scantling for a non-ice-strengthened ship, the selected design scantling must equal the classification-society-required scantling." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:iceStrengtheningStatus false ;
                              nltl:regulationRequiredScantling ?regulation ;
                              nltl:classificationSocietyRequiredScantling ?classification ;
                              nltl:selectedDesignScantling ?selected .
                        ?regulation qudt:numericValue ?regulationValue .
                        ?classification qudt:numericValue ?classificationValue .
                        ?selected qudt:numericValue ?selectedValue .
                        FILTER (
                            ?regulationValue < ?classificationValue &&
                            ?selectedValue != ?classificationValue
                        )
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
