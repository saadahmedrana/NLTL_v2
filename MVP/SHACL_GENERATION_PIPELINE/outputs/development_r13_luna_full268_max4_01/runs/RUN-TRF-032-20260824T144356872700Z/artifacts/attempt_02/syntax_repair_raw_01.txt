<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-032-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:sparql [
                sh:message "When the ship is not ice strengthened and the regulation-required scantling is less than the classification-society-required scantling, the selected design scantling must equal the classification-society-required scantling." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                    SELECT $this
                    WHERE {
                        $this nltl:regulationRequiredScantling ?regulationScantling .
                        ?regulationScantling qudt:numericValue ?regulationValue .
                        $this nltl:classificationSocietyRequiredScantling ?classificationScantling .
                        ?classificationScantling qudt:numericValue ?classificationValue .
                        FILTER (?regulationValue < ?classificationValue)
                        FILTER NOT EXISTS {
                            $this nltl:selectedDesignScantling ?selectedScantling .
                            ?selectedScantling qudt:numericValue ?selectedValue .
                            FILTER (?selectedValue = ?classificationValue)
                        }
                    }
                """
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
