<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_032_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceStrengtheningStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Ice strengthening status is required to determine applicability of the scantling rule."
    ] ;
    sh:property [
        sh:path nltl:regulationRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRequiredScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:selectedDesignScantling ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "For a non-ice-strengthened ship, regulation scantling and classification-society scantling are required for comparison; when the regulation scantling is lower, the selected design scantling must equal the classification-society scantling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT ?this
            WHERE {
                ?this nltl:iceStrengtheningStatus false .

                OPTIONAL {
                    ?this nltl:regulationRequiredScantling ?regulationQuantity .
                    ?regulationQuantity qudt:numericValue ?regulationValue .
                }

                OPTIONAL {
                    ?this nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                    ?classificationQuantity qudt:numericValue ?classificationValue .
                }

                OPTIONAL {
                    ?this nltl:selectedDesignScantling ?selectedQuantity .
                    ?selectedQuantity qudt:numericValue ?selectedValue .
                }

                FILTER (
                    !BOUND(?regulationValue) ||
                    !BOUND(?classificationValue) ||
                    (
                        ?regulationValue < ?classificationValue &&
                        (
                            !BOUND(?selectedValue) ||
                            ?selectedValue != ?classificationValue
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
