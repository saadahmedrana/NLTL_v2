<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-036-ship> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceClassDesignParameterSet ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDesignParameterSet ;
        sh:minCount 1 ;
        sh:message "The ship must link to an ice class design parameter set."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one of the Table 4-1 ice classes."
    ] .

<urn:nltl:generated-shape:TRF-036-parameter-set> a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasIceClassDesignParameterSet
        ] ;
        sh:class nltl:ship ;
        sh:minCount 1 ;
        sh:message "The ice class design parameter set must be linked from a ship."
    ] ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The level ice thickness and design ice load height must conform to the Table 4-1 values for the linked ship's ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                       nltl:iceClass ?iceClass .
                $this nltl:levelIceThickness ?levelQuantity ;
                       nltl:designIceLoadHeight ?heightQuantity .
                ?levelQuantity qudt:numericValue ?level ;
                               qudt:unit <http://qudt.org/vocab/unit/M> .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                FILTER (
                    !(
                        (
                            ?iceClass = nltl:iceClassIaSuper &&
                            ?level <= 1.0 &&
                            ?height >= 0.35 &&
                            ?height <= 0.35
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIa &&
                            ?level <= 0.8 &&
                            ?height >= 0.30 &&
                            ?height <= 0.30
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIb &&
                            ?level <= 0.6 &&
                            ?height >= 0.25 &&
                            ?height <= 0.25
                        ) ||
                        (
                            ?iceClass = nltl:iceClassIc &&
                            ?level <= 0.4 &&
                            ?height >= 0.22 &&
                            ?height <= 0.22
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
