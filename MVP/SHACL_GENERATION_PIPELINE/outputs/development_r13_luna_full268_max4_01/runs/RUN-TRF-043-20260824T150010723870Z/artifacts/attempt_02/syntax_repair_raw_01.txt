<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix q: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_043_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:node <urn:nltl:generated-shape:TRF_043_TerminationCaseShape>
    ] .

<urn:nltl:generated-shape:TRF_043_TerminationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:datatype q:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF_043_ExtensionQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary must be permitted when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?extension .
                ?extension q:numericValue ?extensionValue .
                FILTER (?extensionValue <= 250.0)
                FILTER (NOT EXISTS {
                    $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted
                        "true"^^xsd:boolean .
                })
            }
        """
    ] .

<urn:nltl:generated-shape:TRF_043_ExtensionQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

<urn:nltl:generated-shape:TRF_043_FrameShape>
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT ?this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasComponent ?this .
                ?this a nltl:frame .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesUpperBowIceBeltTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF_043_MetreQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF_043_MetreQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        )
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        )
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing vertical extents must satisfy the applicable Table 4-6 minimums." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX q: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?above ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?below .

                ?above q:numericValue ?aboveValue .
                ?below q:numericValue ?belowValue .

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?aboveValue >= 1.2
                    )
                    ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?aboveValue >= 1.0
                    )
                )

                FILTER (
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:bowRegion &&
                        ?reachesTankTop = true &&
                        ?belowValue >= 2.0
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:midbodyRegion &&
                        ?belowValue >= 2.0
                    )
                    ||
                    (
                        ?iceClass = nltl:iceClassIaSuper &&
                        ?region = nltl:sternRegion &&
                        ?belowValue >= 1.6
                    )
                    ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:bowRegion &&
                        ?belowValue >= 1.6
                    )
                    ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:midbodyRegion &&
                        ?belowValue >= 1.3
                    )
                    ||
                    (
                        ?iceClass IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        ?region = nltl:sternRegion &&
                        ?belowValue >= 1.0
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When an upper bow ice belt is required, the framing must reach its top." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:upperBowIceBeltRequired "true"^^xsd:boolean .
                FILTER NOT EXISTS {
                    $this nltl:reachesUpperBowIceBeltTop "true"^^xsd:boolean .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF_043_MetreQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path q:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path q:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .
<END_SHACL>
