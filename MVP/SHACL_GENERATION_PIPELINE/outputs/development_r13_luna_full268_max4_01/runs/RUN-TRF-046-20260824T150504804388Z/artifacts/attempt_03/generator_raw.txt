<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF046Shape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:class nltl:mainFrame ;
            sh:property [
                sh:path nltl:hasStrengthenedPart ;
                sh:minCount 1 ;
                sh:node gen:TRF046MainStrengthenedPartShape
            ]
        ]
        [
            sh:class nltl:intermediateIceFrame ;
            sh:property [
                sh:path nltl:hasStrengthenedPart ;
                sh:minCount 1 ;
                sh:node gen:TRF046IntermediateStrengthenedPartShape
            ]
        ]
    ) .

gen:TRF046MainStrengthenedPartShape
    a sh:NodeShape ;
    sh:class nltl:frameStrengthenedPart ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF046MainUpperEndShape
    ] ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary non-ice-strengthened scantlings are permitted only when the upper end terminates above a deck or stringer at or above the ice-belt upper limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperEnd ?upperEnd .
                    ?upperEnd nltl:hasTerminationAboveSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
                }
            }
        """
    ] .

gen:TRF046IntermediateStrengthenedPartShape
    a sh:NodeShape ;
    sh:class nltl:frameStrengthenedPart ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF046IntermediateUpperEndShape
    ] ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary non-ice-strengthened scantlings are permitted only when the upper end terminates above a deck or stringer at or above the ice-belt upper limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true .
                FILTER NOT EXISTS {
                    $this nltl:hasUpperEnd ?upperEnd .
                    ?upperEnd nltl:hasTerminationAboveSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An intermediate frame horizontal connection member is permitted only when the upper end terminates above a deck or stringer at or above the ice-belt upper limit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasUpperEnd ?upperEnd .
                ?upperEnd nltl:hasHorizontalConnectionMember ?member .
                FILTER NOT EXISTS {
                    ?upperEnd nltl:hasTerminationAboveSupportingStructure ?supportingStructure .
                    ?supportingStructure nltl:supportingStructureAtOrAboveIceBeltUpperLimit true .
                }
            }
        """
    ] .

gen:TRF046MainUpperEndShape
    a sh:NodeShape ;
    sh:class nltl:frameEnd ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:iceStringer ]
            )
        ]
    ] ;
    sh:property [
        sh:path nltl:hasTerminationAboveSupportingStructure ;
        sh:node [
            sh:class nltl:supportingStructure ;
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gen:TRF046IntermediateUpperEndShape
    a sh:NodeShape ;
    sh:class nltl:frameEnd ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node [
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:iceStringer ]
            )
        ]
    ] ;
    sh:property [
        sh:path nltl:hasTerminationAboveSupportingStructure ;
        sh:node [
            sh:class nltl:supportingStructure ;
            sh:property [
                sh:path nltl:supportingStructureAtOrAboveIceBeltUpperLimit ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:node gen:TRF046HorizontalConnectionMemberShape
    ] .

gen:TRF046HorizontalConnectionMemberShape
    a sh:NodeShape ;
    sh:class nltl:horizontalConnectionMember ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:minCount 1 ;
        sh:class nltl:mainFrame
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .
<END_SHACL>
