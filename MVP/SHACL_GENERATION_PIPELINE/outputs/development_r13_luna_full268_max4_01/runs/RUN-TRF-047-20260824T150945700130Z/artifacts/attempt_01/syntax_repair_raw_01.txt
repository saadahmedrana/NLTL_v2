<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        gs:TRF047DirectAttachmentBranch
        gs:TRF047ConditionalConnectionBranch
    ) .

gs:TRF047DirectAttachmentBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each strengthened main or intermediate ice frame lower end must be attached to a deck structure, tank boundary plating, tank top, or ice stringer." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:hasStrengthenedPart ?part .
                ?part rdf:type ?partType .
                FILTER (?partType IN (
                    nltl:mainFrame,
                    nltl:intermediateIceFrame
                ))
                FILTER NOT EXISTS {
                    ?part nltl:hasLowerEnd ?lowerEnd .
                    ?lowerEnd nltl:hasAttachedSupportingStructure ?structure .
                    ?structure rdf:type ?structureType .
                    FILTER (?structureType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    ))
                }
            }
        """
    ] .

gs:TRF047ConditionalConnectionBranch
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A lower end without direct attachment must satisfy the permitted intermediate-frame horizontal connection alternative, including the ice-belt location, equal scantlings, adjacent main-frame connection, and ice strengthening of the main frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this WHERE {
                $this nltl:hasStrengthenedPart ?part .
                ?part rdf:type ?partType .
                FILTER (?partType IN (
                    nltl:mainFrame,
                    nltl:intermediateIceFrame
                ))
                FILTER NOT EXISTS {
                    ?part nltl:hasLowerEnd ?lowerEnd .
                    ?lowerEnd nltl:hasAttachedSupportingStructure ?structure .
                    ?structure rdf:type ?structureType .
                    FILTER (?structureType IN (
                        nltl:deckStructure,
                        nltl:tankBoundaryPlating,
                        nltl:tankTop,
                        nltl:iceStringer
                    ))
                }
                FILTER NOT EXISTS {
                    ?partTypeValue nltl:mainFrameBelowIceBeltStrengthened true .
                }
                FILTER (
                    ?partType = nltl:mainFrame ||
                    (
                        ?partType = nltl:intermediateIceFrame &&
                        EXISTS {
                            $this nltl:supportingStructureAtOrBelowIceBeltLowerLimit true ;
                                  nltl:sameScantlingsAsMainFrame true .
                            ?part nltl:hasLowerEnd ?alternativeLowerEnd .
                            ?alternativeLowerEnd nltl:hasTerminationBelowSupportingStructure ?terminationStructure .
                            ?terminationStructure rdf:type ?terminationType .
                            FILTER (?terminationType IN (
                                nltl:deckStructure,
                                nltl:tankBoundaryPlating,
                                nltl:tankTop,
                                nltl:iceStringer
                            ))
                            ?alternativeLowerEnd nltl:hasHorizontalConnectionMember ?member .
                            ?member nltl:connectsToAdjacentMainFrame ?adjacentMainFrame .
                            ?adjacentMainFrame nltl:mainFrameBelowIceBeltStrengthened true
                        }
                    )
                )
            }
        """
    ] .
<END_SHACL>
