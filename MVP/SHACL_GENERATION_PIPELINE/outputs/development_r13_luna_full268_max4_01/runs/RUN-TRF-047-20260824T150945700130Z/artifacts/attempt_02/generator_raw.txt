<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF047FrameShape a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:node gen:TRF047StrengthenedPartShape ;
        sh:message "Each main frame or intermediate ice frame strengthened part must satisfy the required lower-end attachment or permitted horizontal-connection condition."
    ] .

gen:TRF047StrengthenedPartShape a sh:NodeShape ;
    sh:or (
        gen:TRF047MainFramePartShape
        gen:TRF047IntermediateDirectAttachmentShape
        gen:TRF047IntermediateHorizontalConnectionShape
    ) .

gen:TRF047MainFramePartShape a sh:NodeShape ;
    sh:class nltl:mainFrame ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF047DirectAttachmentLowerEndShape ;
        sh:message "The lower end of a strengthened main frame must be attached to an allowed supporting structure."
    ] .

gen:TRF047IntermediateDirectAttachmentShape a sh:NodeShape ;
    sh:class nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF047DirectAttachmentLowerEndShape ;
        sh:message "The lower end of a strengthened intermediate ice frame must be attached to an allowed supporting structure unless the permitted horizontal-connection alternative applies."
    ] .

gen:TRF047IntermediateHorizontalConnectionShape a sh:NodeShape ;
    sh:class nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:minCount 1 ;
        sh:node gen:TRF047HorizontalConnectionLowerEndShape ;
        sh:message "A horizontal-connection alternative requires an eligible termination structure, an adjacent strengthened main frame, and matching scantlings."
    ] .

gen:TRF047DirectAttachmentLowerEndShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node gen:TRF047PermittedSupportingStructureShape ;
        sh:message "The lower end must be attached to a deck, tank boundary plating, tank top, or ice stringer."
    ] .

gen:TRF047PermittedSupportingStructureShape a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:deckStructure
        ]
        [
            sh:class nltl:tankBoundaryPlating
        ]
        [
            sh:class nltl:tankTop
        ]
        [
            sh:class nltl:iceStringer
        ]
    ) .

gen:TRF047HorizontalConnectionLowerEndShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasTerminationBelowSupportingStructure ;
        sh:minCount 1 ;
        sh:node gen:TRF047LowerLimitSupportingStructureShape ;
        sh:message "The intermediate frame must terminate below a supporting structure situated at or below the lower limit of the ice belt."
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:minCount 1 ;
        sh:node gen:TRF047HorizontalConnectionMemberShape ;
        sh:message "The lower end must have a horizontal connection member to an adjacent main frame."
    ] .

gen:TRF047LowerLimitSupportingStructureShape a sh:NodeShape ;
    sh:class nltl:supportingStructure ;
    sh:property [
        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The termination supporting structure must be at or below the lower limit of the ice belt."
    ] .

gen:TRF047HorizontalConnectionMemberShape a sh:NodeShape ;
    sh:class nltl:horizontalConnectionMember ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:minCount 1 ;
        sh:node gen:TRF047AdjacentMainFrameShape ;
        sh:message "The horizontal connection member must connect to an adjacent main frame."
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The horizontal connection member must have the same scantlings as the main frame."
    ] .

gen:TRF047AdjacentMainFrameShape a sh:NodeShape ;
    sh:class nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "An adjacent main frame connected by the horizontal member must be ice strengthened below the ice belt."
    ] .
<END_SHACL>
