<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:node gen:TRF047StrengthenedPartShape ;
        sh:message "Each strengthened main or intermediate ice frame part must satisfy the required lower-end attachment or permitted horizontal-connection condition."
    ] .

gen:TRF047StrengthenedPartShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:mainFrame ;
            sh:not [ sh:class nltl:intermediateIceFrame ] ;
            sh:property [
                sh:path nltl:hasLowerEnd ;
                sh:node gen:TRF047DirectLowerEndShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:class nltl:intermediateIceFrame ;
            sh:not [ sh:class nltl:mainFrame ] ;
            sh:property [
                sh:path nltl:hasLowerEnd ;
                sh:node gen:TRF047DirectLowerEndShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:class nltl:intermediateIceFrame ;
            sh:not [ sh:class nltl:mainFrame ] ;
            sh:property [
                sh:path nltl:hasLowerEnd ;
                sh:node gen:TRF047HorizontalLowerEndShape ;
                sh:minCount 1
            ]
        ]
    ] .

gen:TRF047DirectLowerEndShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:node [
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:tankTop ]
                [ sh:class nltl:iceStringer ]
            )
        ] ;
        sh:minCount 1 ;
        sh:message "The lower end must be attached to a deck, tank-boundary plating, tank top, or ice stringer."
    ] .

gen:TRF047HorizontalLowerEndShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasTerminationBelowSupportingStructure ;
        sh:node [
            sh:or (
                [
                    sh:class nltl:deckStructure ;
                    sh:property [
                        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:class nltl:tankBoundaryPlating ;
                    sh:property [
                        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:class nltl:tankTop ;
                    sh:property [
                        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:class nltl:iceStringer ;
                    sh:property [
                        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ]
                ]
            )
        ] ;
        sh:minCount 1 ;
        sh:message "A horizontal connection is permitted only when the terminating deck, tank structure, tank top, or ice stringer is at or below the lower limit of the ice belt."
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:node gen:TRF047HorizontalMemberShape ;
        sh:minCount 1 ;
        sh:message "An intermediate ice frame using the alternative connection must have a horizontal connection member."
    ] .

gen:TRF047HorizontalMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:node gen:TRF047AdjacentMainFrameShape ;
        sh:minCount 1 ;
        sh:message "The horizontal member must connect to an adjacent main frame."
    ] ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The horizontal member must have the same scantlings as the main frame."
    ] .

gen:TRF047AdjacentMainFrameShape
    a sh:NodeShape ;
    sh:class nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The adjacent main frame below the ice belt must be ice strengthened."
    ] .

gen:TRF047MainFrameStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A main frame below the ice belt must be ice strengthened."
    ] .
<END_SHACL>
