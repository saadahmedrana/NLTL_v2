nltlgen:TRF048LongitudinalFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;

    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:class nltl:frameBoundaryConditionTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A longitudinal frame must declare one frame boundary condition type."
    ] ;

    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A longitudinal frame must declare whether its boundary conditions are significantly different."
    ] ;

    sh:property [
        sh:path nltl:icePressure ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:spacing ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:span ;
        sh:node nltlgen:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node nltlgen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node nltlgen:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node nltlgen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node nltlgen:QuantityShapeSectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node nltlgen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:continuousBeamWithBracketsFrameCondition ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node nltlgen:ExactFrameMomentFactor133Shape ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:continuousBeamWithBracketsFrameCondition ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node nltlgen:UnitlessQuantityShape ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:in (
                    nltl:bulkCarrierTopWingTankFrameCondition
                    nltl:multiDeckOrStringerContinuousFrameCondition
                    nltl:singleDeckTankTopToMainDeckFrameCondition
                    nltl:twoDeckFrameCondition
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node nltlgen:UnitlessQuantityShape ;
                sh:minCount 1
            ]
        ]
    ) ;

    sh:sparql [
        sh:message "The reported frame calculation results must match the declared formulas within a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?p ;
                      nltl:designIceLoadHeight/qudt:numericValue ?h ;
                      nltl:spacing/qudt:numericValue ?s ;
                      nltl:span/qudt:numericValue ?l ;
                      nltl:frameMomentFactorM/qudt:numericValue ?m ;
                      nltl:yieldStrength/qudt:numericValue ?sigma ;
                      nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?reportedF4 ;
                      nltl:sectionModulus/qudt:numericValue ?reportedZ ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedA .
                BIND ((1.0 - (0.2 * ?h / ?s)) AS ?expectedF4)
                BIND (((?expectedF4 * ?p * ?h * ?l * ?l) / (4.0 * ?m * ?sigma)) * 1000000.0 AS ?expectedZ)
                BIND (((1.7320508075688772 * ?expectedF4 * 2.16 * ?p * ?h * ?l) / (2.0 * ?sigma)) * 10000.0 AS ?expectedA)
                FILTER (
                    ABS(?reportedF4 - ?expectedF4) > (ABS(?expectedF4) * 0.000001 + 0.000000001)
                    ||
                    ABS(?reportedZ - ?expectedZ) > (ABS(?expectedZ) * 0.000001 + 0.000000001)
                    ||
                    ABS(?reportedA - ?expectedA) > (ABS(?expectedA) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .
