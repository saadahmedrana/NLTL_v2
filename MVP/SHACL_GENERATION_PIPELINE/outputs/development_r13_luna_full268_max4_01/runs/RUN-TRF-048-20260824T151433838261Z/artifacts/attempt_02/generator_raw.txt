            ]

        [
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:node gen:MomentFactorLessThan133
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Frame calculation results must match the declared f4, section modulus, and required shear area formulas within a relative tolerance of 1e-6 plus a small absolute fallback." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:longitudinalFrameLoadDistributionFactorF4 ?f4Node ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?f4Node qudt:numericValue ?f4 .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND ((1 - (0.2 * ?h / ?s)) AS ?expectedF4)
                BIND (((1 - (0.2 * ?h / ?s)) * ?p * ?h * ?l * ?l / (4 * ?m * ?sigma) * 1000000) AS ?expectedZ)
                BIND ((1.7320508075688772 * (1 - (0.2 * ?h / ?s)) * 2.16 * ?p * ?h * ?l / (2 * ?sigma) * 10000) AS ?expectedA)

                FILTER (
                    (?f4 < (?expectedF4 - ((0.000001 * ?expectedF4) + 0.000000001))) ||
                    (?f4 > (?expectedF4 + ((0.000001 * ?expectedF4) + 0.000000001))) ||
                    (?z < (?expectedZ - ((0.000001 * ?expectedZ) + 0.000000001))) ||
                    (?z > (?expectedZ + ((0.000001 * ?expectedZ) + 0.000000001))) ||
                    (?a < (?expectedA - ((0.000001 * ?expectedA) + 0.000000001))) ||
                    (?a > (?expectedA + ((0.000001 * ?expectedA) + 0.000000001)))
                )
            }
        """
    ] .
