<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF049FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:frameAttachmentRecord
    ] ;
    sh:property [
        sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket ) ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:connectionBracket
    ] ;
    sh:property [
        sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket nltl:bracketEdgeStiffened ) ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket nltl:bucklingStiffening ) ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path ( nltl:hasFrameAttachment nltl:hasConnectionBracket nltl:bracketThickness ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Within the ice-strengthened area, each linked supporting structure must have an effective attachment through a supporting-structure attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                    ?attachment nltl:effectiveAttachmentConfirmed true .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A longitudinal frame must be attached by a bracket to every linked supporting web frame and bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this a nltl:longitudinalFrame ;
                      nltl:withinIceStrengthenedArea true .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure ;
                                nltl:effectiveAttachmentConfirmed true ;
                                nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When a transverse frame terminates at a deck or ice stringer, a bracket or similar attachment construction is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this a nltl:transverseFrame ;
                      nltl:withinIceStrengthenedArea true ;
                      nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A frame running through a supporting structure must have both web-plate connection sides connected." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount ?sideCount .
                    FILTER ( ?sideCount >= 2 )
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each installed bracket must be at least as thick as the frame web plate." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:frameWebThickness/qudt:numericValue ?frameThickness ;
                      nltl:hasFrameAttachment/nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness/qudt:numericValue ?bracketThickness .
                FILTER ( ?bracketThickness < ?frameThickness )
            }
        """
    ] .
<END_SHACL>
