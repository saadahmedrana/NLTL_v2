<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-049-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Within the ice-strengthened area, every frame attachment must be a confirmed attachment record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this nltl:hasFrameAttachment ?attachment .
                FILTER NOT EXISTS {
                    ?attachment rdf:type nltl:frameAttachmentRecord .
                    ?attachment nltl:effectiveAttachmentConfirmed true .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every supporting web frame or bulkhead must have a paired effective attachment from the target frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?supporting
            WHERE {
                $this nltl:withinIceStrengthenedArea true .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supporting .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment a nltl:frameAttachmentRecord ;
                        nltl:effectiveAttachmentConfirmed true ;
                        nltl:hasAttachedSupportingStructure ?supporting .
                    ?supporting nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A longitudinal frame must be attached by a bracket to every supporting web frame or bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?supporting
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                    rdf:type nltl:longitudinalFrame .
                $this (nltl:hasSupportingWebFrame|nltl:hasSupportingBulkhead) ?supporting .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment a nltl:frameAttachmentRecord ;
                        nltl:effectiveAttachmentConfirmed true ;
                        nltl:hasAttachedSupportingStructure ?supporting ;
                        nltl:hasConnectionBracket ?bracket .
                    ?supporting nltl:hasSupportingStructureAttachment ?attachment .
                    ?bracket a nltl:connectionBracket .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A transverse frame terminating at a deck or ice stringer must have a linked bracket or similar attachment construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                    rdf:type nltl:transverseFrame ;
                    nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment a nltl:frameAttachmentRecord .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                        ?bracket a nltl:connectionBracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A frame running through a supporting structure must have both web-plate connection sides connected." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                    nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount 2 .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An installed bracket must be at least as thick as the frame web and its edge must be stiffened against buckling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?bracket
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                    nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    ?bracket nltl:bracketEdgeStiffened true ;
                        nltl:bucklingStiffening true ;
                        nltl:bracketThickness ?bracketThickness .
                    ?bracketThickness qudt:numericValue ?bracketValue ;
                        qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    $this nltl:frameWebThickness ?webThickness .
                    ?webThickness qudt:numericValue ?webValue ;
                        qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER (DATATYPE(?bracketValue) = xsd:decimal)
                    FILTER (DATATYPE(?webValue) = xsd:decimal)
                    FILTER (?bracketValue >= ?webValue)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-049-FrameAttachmentRecordShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node <urn:nltl:generated-shape:TRF-049-ConnectionBracketShape>
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-049-ConnectionBracketShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

<urn:nltl:generated-shape:TRF-049-SupportingStructureShape>
    a sh:NodeShape ;
    sh:targetClass nltl:supportingStructure ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:class nltl:frameAttachmentRecord
    ] .
<END_SHACL>
