            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            WHERE {
                $this nltl:icePressure ?pressureNode ;
                      nltl:designIceLoadHeight ?heightNode ;
                      nltl:span ?spanNode ;
                      nltl:frameMomentFactorM ?momentNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:designLineLoad ?lineNode ;
                      nltl:sectionModulus ?sectionNode ;

                ?heightNode qudt:numericValue ?h .
                ?spanNode qudt:numericValue ?l .
                ?momentNode qudt:numericValue ?m .
                ?yieldNode qudt:numericValue ?sigma .
                ?lineNode qudt:numericValue ?lineReported .
                ?sectionNode qudt:numericValue ?zReported .
                ?areaNode qudt:numericValue ?aReported .

                BIND(?p * ?h AS ?rawLineMNPerM)
                BIND(IF(?rawLineMNPerM > 0.15, ?rawLineMNPerM, 0.15) AS ?lineExpectedMNPerM)
                BIND(?lineExpectedMNPerM * 1000 AS ?lineExpected)
                BIND(0.9 * 1.8 * ?lineExpectedMNPerM * ?l * ?l /
                     (?m * ?sigma) * 1000000 AS ?zExpected)
                BIND(1.7320508075688772 * 0.9 * 1.8 * 1.2 *
                     ?lineExpectedMNPerM * ?l / (2 * ?sigma) * 10000 AS ?aExpected)

                BIND(IF(?lineExpected > 0, 0.000001 * ?lineExpected, 0.000001) AS ?lineTolerance)
                BIND(IF(?zExpected > 0, 0.000001 * ?zExpected, 0.000001) AS ?zTolerance)
                BIND(IF(?aExpected > 0, 0.000001 * ?aExpected, 0.000001) AS ?aTolerance)

                FILTER(
                    ?lineReported < (?lineExpected - ?lineTolerance) ||
                    ?lineReported > (?lineExpected + ?lineTolerance) ||
                    ?zReported < (?zExpected - ?zTolerance) ||
                    ?zReported > (?zExpected + ?zTolerance) ||
                    ?aReported < (?aExpected - ?aTolerance) ||
                    ?aReported > (?aExpected + ?aTolerance)
                )
            }
        """
