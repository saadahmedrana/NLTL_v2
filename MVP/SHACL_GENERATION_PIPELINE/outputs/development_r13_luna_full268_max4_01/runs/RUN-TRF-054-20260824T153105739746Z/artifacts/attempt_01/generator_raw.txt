gen:TRF054IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:node gen:KiloNewtonPerMeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:CentiM3QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:CentiM2QuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "The calculated design line load, outside-ice-belt factor, section modulus, and required shear area must satisfy the specified formulas." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pNode ;
                      nltl:designIceLoadHeight ?hNode ;
                      nltl:span ?lNode ;
                      nltl:frameMomentFactorM ?mNode ;
                      nltl:yieldStrength ?sigmaNode ;
                      nltl:distanceToIceBelt ?hsNode ;
                      nltl:stringerSpan ?lsNode ;
                      nltl:designLineLoad ?dlNode ;
                      nltl:outsideIceBeltFactor ?fNode ;
                      nltl:sectionModulus ?zNode ;
                      nltl:requiredShearArea ?aNode .

                ?pNode qudt:numericValue ?p .
                ?hNode qudt:numericValue ?h .
                ?lNode qudt:numericValue ?l .
                ?mNode qudt:numericValue ?m .
                ?sigmaNode qudt:numericValue ?sigma .
                ?hsNode qudt:numericValue ?hs .
                ?lsNode qudt:numericValue ?ls .
                ?dlNode qudt:numericValue ?dl .
                ?fNode qudt:numericValue ?f .
                ?zNode qudt:numericValue ?z .
                ?aNode qudt:numericValue ?a .

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?lineLoadMNPerM)
                BIND((?lineLoadMNPerM * 1000) AS ?expectedDL)
                BIND((1 - (?hs / ?ls)) AS ?expectedF)
                BIND((0.80 * 1.8 * ?lineLoadMNPerM * ?l * ?l /
                      (?m * ?sigma) * ?expectedF * 1000000) AS ?expectedZ)
                BIND((1.7320508075688772 * 0.80 * 1.8 * 1.2 *
                      ?lineLoadMNPerM * ?l /
                      (2 * ?sigma) * ?expectedF * 10000) AS ?expectedA)

                FILTER(
                    ABS(?dl - ?expectedDL) >
                    (0.000001 * ABS(?expectedDL) + 0.000001)
                    ||
                    ABS(?f - ?expectedF) >
                    (0.000001 * ABS(?expectedF) + 0.000001)
                    ||
                    ABS(?z - ?expectedZ) >
                    (0.000001 * ABS(?expectedZ) + 0.000001)
                    ||
                    ABS(?a - ?expectedA) >
                    (0.000001 * ABS(?expectedA) + 0.000001)
                )
            }
        """
    ] .
