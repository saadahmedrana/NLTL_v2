<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:QuantityCentiM3 a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM3> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:QuantityCentiM2 a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM2> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NarrowDeckStripShape a sh:NodeShape ;
    sh:targetClass nltl:narrowDeckStrip ;
    sh:or (
        [
            sh:property [
                sh:path nltl:abreastOfHatch ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:servesAsIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:veryLongHatchOpening ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:actualSectionModulus ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:QuantityCentiM3
            ] ;
            sh:property [
                sh:path nltl:requiredSectionModulus ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:QuantityCentiM3
            ] ;
            sh:property [
                sh:path nltl:actualShearArea ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:QuantityCentiM2
            ] ;
            sh:property [
                sh:path nltl:requiredShearArea ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:QuantityCentiM2
            ]
        ]
        [
            sh:property [
                sh:path nltl:abreastOfHatch ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:servesAsIceStringer ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable narrow deck strip must meet the section modulus and shear area comparisons, or have an approved very-long-hatch reduced line load between 0.10 inclusive and 0.15 exclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:abreastOfHatch true ;
                      nltl:servesAsIceStringer true ;
                      nltl:veryLongHatchOpening ?veryLong .

                FILTER ( ?veryLong = true || ?veryLong = false )

                FILTER NOT EXISTS {
                    $this nltl:actualSectionModulus ?actualZ ;
                          nltl:requiredSectionModulus ?requiredZ ;
                          nltl:actualShearArea ?actualA ;
                          nltl:requiredShearArea ?requiredA .

                    ?actualZ qudt:numericValue ?actualZValue .
                    ?requiredZ qudt:numericValue ?requiredZValue .
                    ?actualA qudt:numericValue ?actualAValue .
                    ?requiredA qudt:numericValue ?requiredAValue .

                    FILTER (
                        ?actualZValue >= ?requiredZValue &&
                        ?actualAValue >= ?requiredAValue
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:veryLongHatchOpening true ;
                          nltl:permittedReducedLineLoad ?reducedLoad ;
                          nltl:reducedLineLoadApprovedByClassificationSociety true .

                    ?reducedLoad qudt:numericValue ?reducedLoadValue .

                    FILTER (
                        ?reducedLoadValue >= "0.10"^^xsd:decimal &&
                        ?reducedLoadValue < "0.15"^^xsd:decimal
                    )
                }
            }
        """
    ] .
<END_SHACL>
