<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF056ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide one ship breadth in metres." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasWeatherdeckHatch ;
        sh:node gen:TRF056WeatherdeckHatchShape
    ] ;
    sh:sparql [
        sh:message "For a weatherdeck hatch whose opening length exceeds half the ship breadth, ship-side deflection and both hatch-cover and hatch-fitting design evidence must be represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasWeatherdeckHatch ?hatch .
                $this nltl:shipBreadth ?breadthQuantity .
                ?breadthQuantity qudt:numericValue ?breadthValue ;
                    qudt:unit unit:M .
                ?hatch nltl:hatchOpeningLength ?openingQuantity .
                ?openingQuantity qudt:numericValue ?openingValue ;
                    qudt:unit unit:M .

                FILTER (
                    datatype(?breadthValue) = xsd:decimal &&
                    datatype(?openingValue) = xsd:decimal &&
                    ?openingValue > (?breadthValue / 2)
                )

                FILTER (
                    NOT EXISTS {
                        ?hatch nltl:shipSideDeflection ?deflectionQuantity .
                        ?deflectionQuantity a qudt:QuantityValue ;
                            qudt:numericValue ?deflectionValue ;
                            qudt:unit unit:M .
                        FILTER (datatype(?deflectionValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchCoverDesignEvidence ?coverEvidence .
                        ?coverEvidence a nltl:hatchCoverDesignEvidence
                    }
                    ||
                    NOT EXISTS {
                        ?hatch nltl:hasHatchFittingDesignEvidence ?fittingEvidence .
                        ?fittingEvidence a nltl:hatchFittingDesignEvidence
                    }
                )
            }
        """
    ] .

gen:TRF056WeatherdeckHatchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hatchOpeningLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The weatherdeck hatch must provide one hatch opening length in metres." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

<END_SHACL>
