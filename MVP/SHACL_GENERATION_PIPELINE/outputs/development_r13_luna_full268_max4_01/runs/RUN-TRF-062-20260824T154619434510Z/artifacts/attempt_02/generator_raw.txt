@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-062-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:minimumClearance ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/M> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "The ship must provide h0 as a single metre-valued minimum clearance quantity."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeTipHullClearance ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/M> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "The ship must provide a single metre-valued propeller blade tip to hull clearance quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The minimum propeller blade tip to hull clearance, including the stern-frame scope, must be greater than or equal to h0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:minimumClearance ?h0Quantity .
                ?h0Quantity qudt:numericValue ?h0Value .
                $this nltl:propellerBladeTipHullClearance ?clearanceQuantity .
                ?clearanceQuantity qudt:numericValue ?clearanceValue .
                FILTER (
                    datatype(?h0Value) = xsd:decimal &&
                    datatype(?clearanceValue) = xsd:decimal &&
                    ?clearanceValue < ?h0Value
                )
            }
        """
    ] .
<END_SHACL>
