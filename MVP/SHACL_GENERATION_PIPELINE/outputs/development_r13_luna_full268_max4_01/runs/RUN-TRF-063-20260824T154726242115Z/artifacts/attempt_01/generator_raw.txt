    ] ;

        sh:path nltl:strengtheningForwardExtent ;

        sh:minInclusive 1.5 ;

        sh:message "The forward strengthening extent must be at least 1.5 metres."

        sh:maxCount 1 ;
        sh:message "The aft strengthening extent must use metres as its unit."
    ] .

gen:TRF_063_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Propeller count is required to determine whether the twin- or triple-screw rule applies."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
                sh:node gen:TRF_063_EnvelopeShape ;
                sh:minCount 1 ;
                sh:message "A twin-screw ship must provide a compliant strengthening envelope for each linked side propeller envelope."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minInclusive 3 ;
                sh:maxInclusive 3
            ] ;
            sh:property [
                sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
                sh:node gen:TRF_063_EnvelopeShape ;
                sh:minCount 1 ;
                sh:message "A triple-screw ship must provide a compliant strengthening envelope for each linked side propeller envelope."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:maxInclusive 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerCount ;
                sh:minInclusive 4
            ]
        ]
    ) ;
    sh:message "For a twin- or triple-screw ship, shell and framing ice strengthening must extend to the tank top at least 1.5 metres forward and aft of each side propeller."
