<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF070ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:mainPropulsionThruster ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:thrusterType ;
                sh:in ( "azimuthing" "fixed" )
            ]
        ]
        [
            sh:property [
                sh:path nltl:mainPropulsionThruster ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:propellerIceInteractionLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thrusterBodyIceInteractionLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "An applicable azimuthing main-propulsion thruster must link its thruster-body ice-interaction load to the expected service-life load under a normal-operational-conditions design condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:mainPropulsionThruster true ;
                      nltl:thrusterType "azimuthing" .
                FILTER NOT EXISTS {
                    $this nltl:serviceLife ?serviceLife ;
                          nltl:thrusterBodyIceInteractionLoad ?bodyLoad ;
                          nltl:hasDesignCondition ?condition .
                    ?serviceLife qudt:numericValue ?serviceLifeValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/YR> .
                    ?bodyLoad qudt:numericValue ?bodyLoadValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    ?condition nltl:normalOperationalConditions true ;
                               nltl:designConditionExpectedLoad ?expectedLoad .
                    ?expectedLoad qudt:numericValue ?expectedValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER (
                        ABS(?bodyLoadValue - ?expectedValue) <=
                        (ABS(?expectedValue) * 0.000001 + 0.000001)
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For an azimuthing main-propulsion thruster body designed for extreme loads, local ice pressure and externally assessed local-strength capacity must be provided on the same design condition, using their declared units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:mainPropulsionThruster true ;
                      nltl:thrusterType "azimuthing" ;
                      nltl:extremeLoad ?extremeLoad .
                FILTER NOT EXISTS {
                    $this nltl:localIcePressure ?pressure ;
                          nltl:hasDesignCondition ?condition .
                    ?extremeLoad qudt:numericValue ?extremeValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    ?pressure qudt:numericValue ?pressureValue ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?condition nltl:normalOperationalConditions true ;
                               nltl:designConditionLocalStrengthCapacity ?capacity .
                    ?capacity qudt:numericValue ?capacityValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                }
            }
        """
    ] .
<END_SHACL>
