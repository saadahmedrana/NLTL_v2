<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF076Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller type is required to select the applicable load-case table branch."
    ] ;
    sh:or (
        gen:TRF076CPBranch
        gen:TRF076FPBranch
    ) ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:node gen:TRF076LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "Load case 1 must be covered for CP or FP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (?type = "CP" || ?type = "FP")
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER (?number = 1)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Load case 2 must be covered for CP or FP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (?type = "CP" || ?type = "FP")
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER (?number = 2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Load case 3 must be covered for CP or FP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (?type = "CP" || ?type = "FP")
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER (?number = 3)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Load case 4 must be covered for CP or FP propellers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type .
                FILTER (?type = "CP" || ?type = "FP")
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER (?number = 4)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Load case 5 must also be covered for a reversing FP propeller." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:propellerType ?type ;
                      nltl:propellerReversingCapability ?reversing .
                FILTER (?type = "FP" && ?reversing = true)
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?case .
                    ?case nltl:propellerBladeLoadCaseNumber ?number .
                    FILTER (?number = 5)
                }
            }
        """
    ] .

gen:TRF076CPBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:hasValue "CP"
    ] .

gen:TRF076FPBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:hasValue "FP"
    ] ;
    sh:property [
        sh:path nltl:propellerReversingCapability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Reversing capability is required to determine whether load case 5 applies to an FP propeller."
    ] .

gen:TRF076LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Each propeller blade load case must have an integer load-case number."
    ] .
<END_SHACL>
