<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF080Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "The ship must provide an ice load cycle count." ;
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "The ship must provide a propeller blade count." ;
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ]
        ] ;
        sh:message "The effective ice load cycle count must be provided as a unitless quantity." ;
    ] ;
    sh:sparql [
        sh:message "The effective ice load cycle count must equal the ice load cycle count multiplied by the propeller blade count, within a relative tolerance of 1e-6 and an absolute tolerance of 1e-6 near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?quantity .
                ?quantity qudt:numericValue ?actual .
                BIND(
                    xsd:decimal(?nice) * xsd:decimal(?blades)
                    AS ?expected
                )
                BIND(
                    IF(
                        ABS(?expected) * 0.000001 > 0.000001,
                        ABS(?expected) * 0.000001,
                        0.000001
                    )
                    AS ?tolerance
                )
                FILTER(
                    ?actual < (?expected - ?tolerance)
                    ||
                    ?actual > (?expected + ?tolerance)
                )
            }
        """
    ] .
<END_SHACL>
