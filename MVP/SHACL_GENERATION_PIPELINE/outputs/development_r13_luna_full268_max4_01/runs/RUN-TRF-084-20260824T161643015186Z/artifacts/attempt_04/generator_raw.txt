                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "For a CP propeller with known pitch, pitchValueKnown must be true."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;

        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP"^^xsd:string ;

            ] ;
            sh:property [
                sh:path nltl:pitchValueKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "The unknown-pitch CP branch requires pitchValueKnown to be false."
            ] ;
            sh:property [
                sh:path nltl:propellerPitchAtMcrBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;

                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "For a CP propeller with unknown pitch, bollard pitch must equal 0.7 times free-running pitch within the permitted tolerance." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerPitchAtMcrBollard ?bollardQuantity .
                        ?bollardQuantity qudt:numericValue ?bollard .
                        $this nltl:propellerPitchAtMcrFreeRunning ?freeQuantity .
                        ?freeQuantity qudt:numericValue ?free .
                        BIND ((0.7 * ?free) AS ?expected)
                        FILTER (
                            (?expected >= 0.001 &&
                                (
                                    ?bollard < (?expected - (?expected * 0.000001)) ||
                                    ?bollard > (?expected + (?expected * 0.000001))
                                )
                            ) ||
                            (?expected < 0.001 &&

                                    ?bollard < (?expected - 0.000000001) ||
                                    ?bollard > (?expected + 0.000000001)
                                )
                            )
                        )
                    }
                """

        ]
