<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:TRF085KnownBranch
        gen:TRF085TableBranch
    ) .

gen:TRF085KnownBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085RotationalSpeedQuantityShape
    ] .

gen:TRF085TableBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085DiameterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085RotationalSpeedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:node gen:TRF085LookupCaseShape
    ] ;
    sh:sparql [
        sh:message "When bollard propeller speed is unknown, the Table 6-9 input and lookup result must correspond to the ship values within a relative tolerance of 1e-6 plus an absolute fallback of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameter ;
                      nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed ;
                      nltl:hasTableLookupCase ?case .
                ?case nltl:lookupInputQuantity ?input ;
                      nltl:lookupResultQuantity ?result .
                ?diameter qudt:numericValue ?diameterValue .
                ?input qudt:numericValue ?inputValue .
                ?speed qudt:numericValue ?speedValue .
                ?result qudt:numericValue ?resultValue .
                FILTER (
                    ABS(?inputValue - ?diameterValue) >
                        (1e-6 * ABS(?diameterValue) + 0.000001)
                    ||
                    ABS(?resultValue - ?speedValue) >
                        (1e-6 * ABS(?resultValue) + 0.000001)
                )
            }
        """
    ] .

gen:TRF085DiameterQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:TRF085RotationalSpeedQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
    ] .

gen:TRF085LookupCaseShape a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085DiameterQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF085RotationalSpeedQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ]
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:traficomTable6Dash9
    ] .
<END_SHACL>
