<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether the bollard propeller speed is known."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
                    ]
                ] ;
                sh:message "When the bollard propeller speed is known, the known speed at MCR must be supplied in revolutions per minute."
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/M>
                    ]
                ] ;
                sh:message "When the bollard propeller speed is not known, the propeller diameter must be supplied in metres."
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
                    ]
                ] ;
                sh:message "The bollard propeller speed selected from Table 6-9 must be supplied in revolutions per minute."
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:tableLookupCase ;
                    sh:property [
                        sh:path nltl:tableLookupApplied ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:tableReference ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue nltl:traficomTable6Dash9
                    ] ;
                    sh:property [
                        sh:path nltl:lookupInputQuantity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:lookupResultQuantity ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:lookupSelectionEvidence ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:class nltl:evidenceArtifact
                        ]
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "The Table 6-9 lookup must use the propeller diameter as input and its selected result must equal the ship's bollard propeller speed at MCR." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  $this nltl:bollardPropellerSpeedKnown false .
  $this nltl:propellerDiameter ?diameterQuantity .
  ?diameterQuantity qudt:numericValue ?diameterValue .
  $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speedQuantity .
  ?speedQuantity qudt:numericValue ?speedValue .
  FILTER NOT EXISTS {
    $this nltl:hasTableLookupCase ?case .
    ?case nltl:lookupInputQuantity ?inputQuantity .
    ?inputQuantity qudt:numericValue ?inputValue .
    ?case nltl:lookupResultQuantity ?resultQuantity .
    ?resultQuantity qudt:numericValue ?resultValue .
    FILTER (?inputValue = ?diameterValue && ?resultValue = ?speedValue)
  }
}
"""
            ]
        ]
    ) .
