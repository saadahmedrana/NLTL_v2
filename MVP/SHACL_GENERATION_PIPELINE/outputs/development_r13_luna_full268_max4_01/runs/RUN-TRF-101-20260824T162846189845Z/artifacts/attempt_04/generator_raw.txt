@prefix nltl: <https://w3id.org/nltl/vocab#> .

@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF101Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must record blade failure status."
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PropellerDiameterQuantity ;
        sh:message "The propeller diameter is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:BladeTipOffsetQuantity ;
        sh:message "The blade tip offset is required in metres."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:BladeFailureUltimateLoadQuantity ;
        sh:message "The blade failure ultimate load is required in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node gen:LoadCaseShape ;
        sh:message "The ship must provide a load case for the blade failure load."
    ] ;
    sh:or (
        gen:FormulaBranch
        gen:ApprovedAnalysisBranch
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true exactly when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeFailureStatus ?status ;
                      nltl:bladeTipOffset ?offsetNode ;
                      nltl:propellerDiameter ?diameterNode .
                ?offsetNode qudt:numericValue ?offset .
                ?diameterNode qudt:numericValue ?diameter .
                FILTER (
                    (?status = true && ! (?offset > (0.10 * ?diameter))) ||
                    (?status = false && (?offset > (0.10 * ?diameter)))
                )
            }
        """
    ] .

gen:FormulaBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantity
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantity
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:LengthMetreQuantity
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureMegaPaQuantity
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureMegaPaQuantity
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PressureMegaPaQuantity
    ] ;
    sh:sparql [
        sh:message "The denominator of the blade failure load formula must be non-zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:propellerDiameter ?dNode .
                ?rNode qudt:numericValue ?r .
                ?dNode qudt:numericValue ?d .
                BIND ((0.8 * ?d) - (2 * ?r) AS ?denominator)
                FILTER (?denominator = 0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The reported reference strength and blade failure ultimate load must match the declared calculation within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeMinimumYieldStrength ?yieldNode ;
                      nltl:bladeReferenceStrength1 ?referenceNode ;
                      nltl:bladeUltimateTensileStrength ?ultimateNode ;
                      nltl:propellerDiameter ?dNode ;
                      nltl:bladeFailureUltimateLoad ?loadNode .
                ?cNode qudt:numericValue ?c .
                ?rNode qudt:numericValue ?r .
                ?tNode qudt:numericValue ?t .
                ?yieldNode qudt:numericValue ?yield .
                ?referenceNode qudt:numericValue ?reference .
                ?ultimateNode qudt:numericValue ?ultimate .
                ?dNode qudt:numericValue ?d .
                ?loadNode qudt:numericValue ?load .
                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expectedReference)
                BIND (((0.8 * ?d) - (2 * ?r)) AS ?denominator)
                FILTER (?denominator != 0)
                BIND ((300 * ?c * ?t * ?t * ?expectedReference) / ?denominator AS ?expectedLoad)
                BIND (IF(?expectedReference >= 0, ?expectedReference, 0 - ?expectedReference) AS ?referenceMagnitude)
                BIND (IF(?expectedLoad >= 0, ?expectedLoad, 0 - ?expectedLoad) AS ?loadMagnitude)
                BIND (IF((1e-6 * ?referenceMagnitude) > 1e-9, (1e-6 * ?referenceMagnitude), 1e-9) AS ?referenceTolerance)
                BIND (IF((1e-6 * ?loadMagnitude) > 1e-9, (1e-6 * ?loadMagnitude), 1e-9) AS ?loadTolerance)
                FILTER (
                    (?reference < (?expectedReference - ?referenceTolerance)) ||
                    (?reference > (?expectedReference + ?referenceTolerance)) ||
                    (?load < (?expectedLoad - ?loadTolerance)) ||
                    (?load > (?expectedLoad + ?loadTolerance))
                )
            }
        """
    ] .
