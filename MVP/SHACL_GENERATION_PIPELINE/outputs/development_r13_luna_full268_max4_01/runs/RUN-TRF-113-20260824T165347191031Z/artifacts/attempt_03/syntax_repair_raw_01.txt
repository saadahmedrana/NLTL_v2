<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF113ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:class nltl:steeringMechanism ;
        sh:message "The ship shall identify a steering mechanism subject to the blade plastic-bending design condition."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:class nltl:thrusterFitting ;
        sh:message "The ship shall identify a thruster fitting subject to the blade plastic-bending design condition."
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:minCount 1 ;
        sh:class nltl:thrusterBody ;
        sh:message "The ship shall identify a thruster body subject to the blade plastic-bending design condition."
    ] ;
    sh:property [
        sh:path nltl:bladePlasticBending ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship shall record the plastic-bending-without-damage design condition."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeOrientation ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "The ship shall identify the propeller blade orientation used for the maximum blade-loss load."
    ] ;
    sh:property [
        sh:path nltl:bladeLossLoad ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:KiloN
            ]
        ] ;
        sh:message "The ship shall record the blade-loss load in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:maximumComponentLoadBladeOrientationConfirmed ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ] ;
        sh:message "A load case shall confirm that the selected blade-loss orientation produces the maximum load on the component being studied."
    ] ;
    sh:property [
        sh:path nltl:thrusterBladeLossDesignEvidence ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:evidenceArtifact
        ] ;
        sh:message "Supporting design evidence shall be linked for the blade-loss maximum-load determination."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The plastic-bending design condition, affected thruster components, selected orientation, blade-loss load, confirming load case, and supporting evidence shall be represented together." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:bladePlasticBending ?bending ;
                          nltl:propellerBladeOrientation ?orientation ;
                          nltl:bladeLossLoad ?load ;
                          nltl:hasLoadCase ?case ;
                          nltl:thrusterBladeLossDesignEvidence ?evidence ;
                          nltl:hasComponent ?steering ;
                          nltl:hasComponent ?fitting ;
                          nltl:hasComponent ?body .

                    ?steering a nltl:steeringMechanism .
                    ?fitting a nltl:thrusterFitting .
                    ?body a nltl:thrusterBody .

                    ?load qudt:numericValue ?numeric ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .

                    ?case a nltl:loadCase ;
                          nltl:maximumComponentLoadBladeOrientationConfirmed ?confirmed .

                    ?evidence a nltl:evidenceArtifact .

                    FILTER (datatype(?bending) = xsd:string)
                    FILTER (datatype(?orientation) = xsd:string)
                    FILTER (datatype(?numeric) = xsd:decimal)
                    FILTER (datatype(?confirmed) = xsd:boolean && ?confirmed = true)
                }
            }
        """
    ] .
<END_SHACL>
