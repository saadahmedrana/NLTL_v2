<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF114ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide a typical ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:node gen:TRF114LoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide a thruster ice impact load case."
    ] ;
    sh:sparql [
        sh:message "The externally assessed thruster resistance capacity must satisfy the thruster ice impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:thrusterIceImpactDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                ?case nltl:thrusterImpactCaseAppliesToBody ?body .
                ?body nltl:thrusterResistanceCapacity ?capacity .
                ?capacity qudt:numericValue ?capacityValue .
                FILTER ( ?capacityValue < ?demandValue )
            }
        """
    ] .

gen:TRF114LoadCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadCaseIdentifier ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:node [
            sh:property [
                sh:path nltl:thrusterResistanceCapacity ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case must include evidence for the loaded area."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:class nltl:contactGeometryValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:hemisphericalContactGeometry
            ]
            [
                sh:hasValue nltl:nonHemisphericalContactGeometry
            ]
        ) ;
        sh:message "The impact case must classify its contact geometry as hemispherical or non-hemispherical."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path (
                    [
                        sh:inversePath nltl:hasThrusterIceImpactLoadCase
                    ]
                    nltl:equivalentImpactSphereRadius
                ) ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "A non-hemispherical contact geometry requires sphere-radius correspondence evidence."
            ]
        ]
    ) .
