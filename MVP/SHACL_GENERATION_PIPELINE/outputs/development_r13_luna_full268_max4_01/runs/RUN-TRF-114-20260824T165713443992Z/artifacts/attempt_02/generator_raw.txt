@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF_114_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide a typical ice operating speed in knots."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A non-hemispherical contact geometry requires an equivalent impact sphere radius in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radiusQuantity .
                    ?radiusQuantity qudt:numericValue ?radiusValue ;
                                    qudt:unit <http://qudt.org/vocab/unit/M> .
                    FILTER (datatype(?radiusValue) = xsd:decimal)
                }
            }
        """
    ] .

    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/KiloN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The impact case must provide the externally assessed thruster ice impact demand in kN."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case must include evidence for the loaded impact area."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:in (
            nltl:hemisphericalContactGeometry
            nltl:nonHemisphericalContactGeometry
        ) ;
        sh:minCount 1 ;
        sh:message "The impact case must classify the contact geometry as hemispherical or non-hemispherical."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:minCount 1

            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;

            ]
        ]

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The externally assessed thruster resistance capacity must be at least the impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:thrusterIceImpactDemand ?demandQuantity ;
                       nltl:thrusterImpactCaseAppliesToBody ?body .
                ?demandQuantity qudt:numericValue ?demandValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?body nltl:thrusterResistanceCapacity ?capacityQuantity .
                ?capacityQuantity qudt:numericValue ?capacityValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                FILTER (?capacityValue < ?demandValue)
            }
        """
    ] .
