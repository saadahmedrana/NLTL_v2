<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF_120_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql <urn:nltl:generated-shape:TRF_120_RequiredInputsConstraint> ;
    sh:sparql <urn:nltl:generated-shape:TRF_120_CalculationCaseConstraint> ;
    sh:property [
        sh:path ( nltl:hasComponent nltl:hasNaturalFrequencyCalculationCase ) ;
        sh:node [
            a sh:NodeShape ;
            sh:class nltl:calculationCase ;
            sh:property [
                sh:path nltl:vibrationDirection ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:longitudinalDirection
                    nltl:transverseDirection
                )
            ]
        ]
    ] .

<urn:nltl:generated-shape:TRF_120_RequiredInputsConstraint>
    a sh:SPARQLConstraint ;
    sh:message "A ship with a linked thruster body must provide damping, water added mass, and thruster natural frequency with their canonical quantity structures; each linked thruster body must provide ship attachment stiffness." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX qudt: <http://qudt.org/schema/qudt/>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
        PREFIX unit: <http://qudt.org/vocab/unit/>

        SELECT $this
        WHERE {
            $this nltl:hasComponent ?body .
            ?body a nltl:thrusterBody .

            FILTER NOT EXISTS {
                $this nltl:damping ?damping .
                FILTER (datatype(?damping) = xsd:string)
            }

            FILTER NOT EXISTS {
                $this nltl:waterAddedMass ?mass .
                ?mass a qudt:QuantityValue ;
                      qudt:numericValue ?massValue ;
                      qudt:unit unit:KiloGM .
                FILTER (datatype(?massValue) = xsd:decimal)
            }

            FILTER NOT EXISTS {
                $this nltl:thrusterNaturalFrequency ?frequency .
                ?frequency a qudt:QuantityValue ;
                           qudt:numericValue ?frequencyValue ;
                           qudt:unit unit:HZ .
                FILTER (datatype(?frequencyValue) = xsd:decimal)
            }

            FILTER NOT EXISTS {
                ?body nltl:shipAttachmentStiffness ?stiffness .
                ?stiffness a qudt:QuantityValue ;
                           qudt:numericValue ?stiffnessValue ;
                           qudt:unit unit:KiloN-PER-M .
                FILTER (datatype(?stiffnessValue) = xsd:decimal)
            }
        }
    """ .

<urn:nltl:generated-shape:TRF_120_CalculationCaseConstraint>
    a sh:SPARQLConstraint ;
    sh:message "Each linked thruster body must have a natural-frequency calculation case with a longitudinal or transverse vibration direction." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>

        SELECT $this
        WHERE {
            $this nltl:hasComponent ?body .
            ?body a nltl:thrusterBody .

            FILTER NOT EXISTS {
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                ?case a nltl:calculationCase ;
                      nltl:vibrationDirection ?direction .
                FILTER (
                    ?direction = nltl:longitudinalDirection ||
                    ?direction = nltl:transverseDirection
                )
            }
        }
    """ .
<END_SHACL>
