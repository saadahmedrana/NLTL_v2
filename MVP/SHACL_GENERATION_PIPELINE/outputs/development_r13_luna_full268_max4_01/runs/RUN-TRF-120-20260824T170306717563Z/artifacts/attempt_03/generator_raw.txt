<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF120_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:vibrationDirectionValue ;
            sh:in (
                nltl:longitudinalDirection
                nltl:transverseDirection
            )
        ]
    ] .

gen:TRF120_ThrusterBodyShape
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-PER-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gen:TRF120_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path (
            nltl:hasComponent
            nltl:hasNaturalFrequencyCalculationCase
        ) ;
        sh:minCount 1 ;
        sh:node gen:TRF120_CalculationCaseShape
    ] ;
    sh:sparql [
        sh:message "The ship must have a thruster body with paired longitudinal and transverse natural-frequency calculation cases, damping, water added mass, ship attachment stiffness, and thruster natural frequency." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasComponent ?body .
                    ?body a nltl:thrusterBody ;
                          nltl:hasNaturalFrequencyCalculationCase ?longCase ;
                          nltl:hasNaturalFrequencyCalculationCase ?transCase ;
                          nltl:shipAttachmentStiffness ?stiffness .
                    ?longCase nltl:vibrationDirection nltl:longitudinalDirection .
                    ?transCase nltl:vibrationDirection nltl:transverseDirection .
                    $this nltl:damping ?damping ;
                          nltl:waterAddedMass ?addedMass ;
                          nltl:thrusterNaturalFrequency ?frequency .
                }
            }
        """
    ] .
<END_SHACL>
