<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF120ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:damping ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must provide the damping used for thruster natural-frequency estimation."
    ] ;
    sh:property [
        sh:path nltl:waterAddedMass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloGM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide water added mass as a kilogram quantity."
    ] ;
    sh:property [
        sh:path nltl:thrusterNaturalFrequency ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HZ ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the thruster natural frequency as a hertz quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked thruster body must have paired longitudinal and transverse natural-frequency calculation cases, with damping, water added mass, natural frequency, and ship attachment stiffness represented for the estimation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body a nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?longCase .
                ?longCase nltl:vibrationDirection nltl:longitudinalDirection .
                ?body nltl:hasNaturalFrequencyCalculationCase ?transCase .
                ?transCase nltl:vibrationDirection nltl:transverseDirection .
                FILTER (?longCase != ?transCase)
                FILTER (
                    NOT EXISTS { $this nltl:damping ?damping } ||
                    NOT EXISTS { $this nltl:waterAddedMass ?waterMass } ||
                    NOT EXISTS { $this nltl:thrusterNaturalFrequency ?frequency } ||
                    NOT EXISTS { ?body nltl:shipAttachmentStiffness ?stiffness }
                )
            }
        """
    ] .

gen:TRF120ThrusterBodyShape
    a sh:NodeShape ;
    sh:targetClass nltl:thrusterBody ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:minCount 2 ;
        sh:node gen:TRF120CalculationCaseShape ;
        sh:message "The thruster body must have explicitly paired calculation cases for the longitudinal and transverse directions."
    ] ;
    sh:property [
        sh:path nltl:shipAttachmentStiffness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN-PER-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The thruster body must provide ship attachment stiffness as a kilonewton-per-metre quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The thruster body must have distinct longitudinal and transverse calculation cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasNaturalFrequencyCalculationCase ?longCase .
                    ?longCase nltl:vibrationDirection nltl:longitudinalDirection .
                    $this nltl:hasNaturalFrequencyCalculationCase ?transCase .
                    ?transCase nltl:vibrationDirection nltl:transverseDirection .
                    FILTER (?longCase != ?transCase)
                }
            }
        """
    ] .

gen:TRF120CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class nltl:vibrationDirectionValue ;
            sh:in (
                nltl:longitudinalDirection
                nltl:transverseDirection
            )
        ] ;
        sh:message "Each natural-frequency calculation case must identify a longitudinal or transverse vibration direction."
    ] .
<END_SHACL>
