<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DisplacementQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A displacement quantity must have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloTON_Metric ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A displacement quantity must use metric kilotonnes."
    ] .

gen:UpperIceWaterlineCaseShape
    a sh:NodeShape ;
    sh:class nltl:upperIceWaterlineCase ;
    sh:property [
        sh:path nltl:waterlineCaseDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DisplacementQuantityShape ;
        sh:message "Each upper ice waterline case must have exactly one displacement in metric kilotonnes."
    ] .

gen:UpperIceWaterlineDisplacementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DisplacementQuantityShape ;
        sh:message "The ship must have exactly one upper ice waterline displacement in metric kilotonnes."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node gen:UpperIceWaterlineCaseShape ;
        sh:message "The ship must have at least one upper ice waterline case."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline displacement must equal the greatest displacement among the ship's upper ice waterline cases." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:upperIceWaterlineDisplacement ?reportedQuantity .
    ?reportedQuantity
        qudt:numericValue ?reportedValue ;
        qudt:unit unit:KiloTON_Metric .
    $this nltl:hasUpperIceWaterlineCase ?waterlineCase .
    ?waterlineCase nltl:waterlineCaseDisplacement ?caseQuantity .
    ?caseQuantity
        qudt:numericValue ?caseValue ;
        qudt:unit unit:KiloTON_Metric .
}
GROUP BY $this ?reportedValue
HAVING (?reportedValue != MAX(?caseValue))
"""
    ] .
<END_SHACL>
