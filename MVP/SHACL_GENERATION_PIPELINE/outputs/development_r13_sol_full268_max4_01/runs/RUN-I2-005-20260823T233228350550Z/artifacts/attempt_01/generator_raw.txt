                sh:minCount 1 ;
                sh:maxCount 1 ;

                    a sh:NodeShape ;
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:datatype xsd:decimal ;
                        sh:minInclusive "7"^^xsd:decimal ;
                        sh:maxInclusive "7"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:DEG
                    ]
                ]
            ]
        ] ;
        sh:message "Each ship must identify a bottom/lower-region boundary point where the shell inclination is exactly 7 degrees from horizontal."
    ] .
