<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_006_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:asternIceOperationIntent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:aftSectionDesignBasis ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hullAreaRequirementSet ;
        sh:datatype xsd:string
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship intended to operate astern in ice must use the Bow and Bow Intermediate hull area requirements for its aft section." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:asternIceOperationIntent ?intent .
    FILTER(?intent = true)
    FILTER NOT EXISTS {
        $this nltl:aftSectionDesignBasis ?basis .
        $this nltl:hullAreaRequirementSet ?bow .
        $this nltl:hullAreaRequirementSet ?bowIntermediate .
        FILTER(
            DATATYPE(?basis) = xsd:string &&
            DATATYPE(?bow) = xsd:string &&
            DATATYPE(?bowIntermediate) = xsd:string &&
            STR(?basis) = "Bow and Bow Intermediate hull area requirements" &&
            STR(?bow) = "Bow" &&
            STR(?bowIntermediate) = "Bow Intermediate"
        )
    }
}
"""
    ] .
<END_SHACL>
