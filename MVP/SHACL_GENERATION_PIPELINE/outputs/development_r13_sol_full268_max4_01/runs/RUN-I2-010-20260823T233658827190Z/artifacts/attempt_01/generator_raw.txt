<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_010_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:message "The ship must have one recognized Polar Class."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:DEG
            ]
        ]
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                )
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:verticalSidedBowForm ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue false
                    ] ;
                    sh:property [
                        sh:path nltl:bowVerticalSideStatus ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:verticalSidedBowForm ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:bowVerticalSideStatus ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:bowDesignForceMethod ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue "I2.3.2.1 (iv)"
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a PC6 or PC7 ship, a normal frame angle from 0 to 10 degrees establishes a vertical-sided bow form." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:normalFrameAngle ?angle .
                ?angle qudt:numericValue ?beta ;
                       qudt:unit unit:DEG .
                FILTER(
                    ?polarClass IN (
                        nltl:polarClassPc6,
                        nltl:polarClassPc7
                    )
                )
                FILTER(
                    xsd:decimal(?beta) >= 0.0 &&
                    xsd:decimal(?beta) <= 10.0
                )
                FILTER NOT EXISTS {
                    $this nltl:verticalSidedBowForm true .
                }
            }
        """
    ] .
<END_SHACL>
