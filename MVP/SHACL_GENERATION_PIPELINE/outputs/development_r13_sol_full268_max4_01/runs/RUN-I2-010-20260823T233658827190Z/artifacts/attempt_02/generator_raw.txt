gen:I2_010_AngleDerivedVerticalBowShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this nltl:polarClass ?polarClass ;
                      nltl:normalFrameAngle ?suppliedAngle .
                FILTER(
                    ?polarClass IN (
                        nltl:polarClassPc6,
                        nltl:polarClassPc7
                    )
                )
                FILTER NOT EXISTS {
                    ?this nltl:normalFrameAngle ?consideredAngle .
                    FILTER NOT EXISTS {
                        ?consideredAngle
                            qudt:numericValue ?angleValue ;
                            qudt:unit unit:DEG .
                        FILTER(
                            datatype(?angleValue) = xsd:decimal &&
                            ?angleValue >= "0"^^xsd:decimal &&
                            ?angleValue <= "10"^^xsd:decimal
                        )
                    }
                }
                FILTER NOT EXISTS {
                    ?this nltl:normalFrameAngle ?consideredAngle .
                    ?consideredAngle qudt:numericValue ?angleValue .
                    FILTER(
                        datatype(?angleValue) != xsd:decimal ||
                        ?angleValue < "0"^^xsd:decimal ||
                        ?angleValue > "10"^^xsd:decimal
                    )
                }
                FILTER NOT EXISTS {
                    ?this nltl:normalFrameAngle ?consideredAngle .
                    ?consideredAngle qudt:unit ?angleUnit .
                    FILTER(?angleUnit != unit:DEG)
                }
            }
        """
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "When every supplied normal frame angle is between 0 and 10 DEG, the bow form must be identified as vertical-sided."
    ] .
