@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:ShipBowDataShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gs:Pc6Pc7VerticalSideSelectionShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:polarClass ?polarClass .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
            }
        """
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A PC6 or PC7 ship must explicitly record whether its bow has vertical sides."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bowVerticalSideStatus ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:bowVerticalSideStatus ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethod ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:in ( "I2.3.2.1 (iv)" ) ;
                sh:message "A PC6 or PC7 ship with a vertical-sided bow must use design ice forces calculated according to I2.3.2.1 (iv)."
            ]
        ]
    ) .

gs:Pc6Pc7VerticalBowInclusionShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:polarClass ?polarClass .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                {
                    ?this nltl:verticalSidedBowForm ?verticalForm .
                    FILTER (?verticalForm = true)
                }
                UNION
                {
                    ?this nltl:normalFrameAngle ?angle .
                    FILTER NOT EXISTS {
                        ?this nltl:normalFrameAngle ?consideredAngle .
                        FILTER NOT EXISTS {
                            ?consideredAngle
                                qudt:numericValue ?angleValue ;
                                qudt:unit <http://qudt.org/vocab/unit/DEG> .
                            FILTER (
                                datatype(?angleValue) = xsd:decimal &&
                                ?angleValue >= 0 &&
                                ?angleValue <= 10
                            )
                        }
                    }
                }
            }
        """
    ] ;
    sh:property [
        sh:path nltl:bowVerticalSideStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A PC6 or PC7 bow explicitly identified as vertical-sided, or having every supplied normal frame angle between 0 and 10 degrees inclusive, must be recorded as vertical-sided."
    ] ;
    sh:property [
        sh:path nltl:bowDesignForceMethod ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:in ( "I2.3.2.1 (iv)" ) ;
        sh:message "The applicable bow design ice forces must be calculated according to I2.3.2.1 (iv)."
    ] .
<END_SHACL>
