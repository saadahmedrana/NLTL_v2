                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                        $this nltl:bowDesignForce ?designQuantity ;

                        BIND(xsd:decimal(?methodIiiValue) * 1000.0 AS ?minimumValue)

                            IF(
                                ?relativeTolerance > 0.000000001,
                                ?relativeTolerance,
                                0.000000001
                            )
                            AS ?tolerance
                        )
                        FILTER(
                            ABS(xsd:decimal(?designValue) - ?expectedValue) > ?tolerance ||
                            xsd:decimal(?designValue) < ?minimumValue
                        )
                    }
                """

        ]
