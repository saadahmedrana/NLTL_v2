gen:I2_014_BowSubregionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "The ship must provide one bow length in metres."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The bow region must have four subregions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:node gen:BowSubregionCalculationCaseShape ;
        sh:message "The ship must have exactly four bow subregion calculation cases."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonQuantityShape ;
        sh:message "The ship must report one selected maximum bow force."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaNewtonPerMetreQuantityShape ;
        sh:message "The ship must report one selected maximum bow line load."
    ] ;
    sh:property [
        sh:path nltl:selectedMaximumBowPressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:message "The ship must report one selected maximum bow pressure."
    ] ;
    sh:sparql [
        sh:message "The four calculation cases must represent the midpoints of four equal-length bow subregions within a relative tolerance of 1e-6 and an absolute fallback of 1e-12 metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                {
                    SELECT $this
                           (SUM(IF(ABS(?position - ?expected1) <= ?tolerance1, 1, 0)) AS ?count1)
                           (SUM(IF(ABS(?position - ?expected2) <= ?tolerance2, 1, 0)) AS ?count2)
                           (SUM(IF(ABS(?position - ?expected3) <= ?tolerance3, 1, 0)) AS ?count3)
                           (SUM(IF(ABS(?position - ?expected4) <= ?tolerance4, 1, 0)) AS ?count4)
                    WHERE {
                        $this nltl:bowLength ?lengthQuantity ;
                              nltl:hasBowSubregionCalculationCase ?case .
                        ?lengthQuantity qudt:numericValue ?length .
                        ?case nltl:bowSubregionMidLengthPosition ?positionQuantity .
                        ?positionQuantity qudt:numericValue ?position .

                        BIND(?length / 8 AS ?expected1)
                        BIND((3 * ?length) / 8 AS ?expected2)
                        BIND((5 * ?length) / 8 AS ?expected3)
                        BIND((7 * ?length) / 8 AS ?expected4)

                        BIND(IF(
                            ABS(?expected1) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                            ABS(?expected1) * "0.000001"^^xsd:decimal,
                            "0.000000000001"^^xsd:decimal
                        ) AS ?tolerance1)
                        BIND(IF(
                            ABS(?expected2) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                            ABS(?expected2) * "0.000001"^^xsd:decimal,
                            "0.000000000001"^^xsd:decimal
                        ) AS ?tolerance2)
                        BIND(IF(
                            ABS(?expected3) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                            ABS(?expected3) * "0.000001"^^xsd:decimal,
                            "0.000000000001"^^xsd:decimal
                        ) AS ?tolerance3)
                        BIND(IF(
                            ABS(?expected4) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                            ABS(?expected4) * "0.000001"^^xsd:decimal,
                            "0.000000000001"^^xsd:decimal
                        ) AS ?tolerance4)
                    }
                    GROUP BY $this
                }
                FILTER(
                    ?count1 != 1 ||
                    ?count2 != 1 ||
                    ?count3 != 1 ||
                    ?count4 != 1
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected bow force, line load and pressure must equal their respective maxima across the four subregion cases within a relative tolerance of 1e-6 and an absolute fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:selectedMaximumBowForce ?selectedForceQuantity ;
                      nltl:selectedMaximumBowLineLoad ?selectedLineLoadQuantity ;
                      nltl:selectedMaximumBowPressure ?selectedPressureQuantity .
                ?selectedForceQuantity qudt:numericValue ?selectedForce .
                ?selectedLineLoadQuantity qudt:numericValue ?selectedLineLoad .
                ?selectedPressureQuantity qudt:numericValue ?selectedPressure .

                {
                    SELECT $this
                           (MAX(?force) AS ?maximumForce)
                           (MAX(?lineLoad) AS ?maximumLineLoad)
                           (MAX(?pressure) AS ?maximumPressure)
                    WHERE {
                        $this nltl:hasBowSubregionCalculationCase ?case .
                        ?case nltl:bowSubregionCalculatedForce ?forceQuantity ;
                              nltl:bowSubregionCalculatedLineLoad ?lineLoadQuantity ;
                              nltl:bowSubregionCalculatedPressure ?pressureQuantity .
                        ?forceQuantity qudt:numericValue ?force .
                        ?lineLoadQuantity qudt:numericValue ?lineLoad .
                        ?pressureQuantity qudt:numericValue ?pressure .
                    }
                    GROUP BY $this
                }

                BIND(IF(
                    ABS(?maximumForce) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                    ABS(?maximumForce) * "0.000001"^^xsd:decimal,
                    "0.000000000001"^^xsd:decimal
                ) AS ?forceTolerance)
                BIND(IF(
                    ABS(?maximumLineLoad) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                    ABS(?maximumLineLoad) * "0.000001"^^xsd:decimal,
                    "0.000000000001"^^xsd:decimal
                ) AS ?lineLoadTolerance)
                BIND(IF(
                    ABS(?maximumPressure) * "0.000001"^^xsd:decimal > "0.000000000001"^^xsd:decimal,
                    ABS(?maximumPressure) * "0.000001"^^xsd:decimal,
                    "0.000000000001"^^xsd:decimal
                ) AS ?pressureTolerance)

                FILTER(
                    ABS(?selectedForce - ?maximumForce) > ?forceTolerance ||
                    ABS(?selectedLineLoad - ?maximumLineLoad) > ?lineLoadTolerance ||
                    ABS(?selectedPressure - ?maximumPressure) > ?pressureTolerance
                )
            }
        """
    ] .
