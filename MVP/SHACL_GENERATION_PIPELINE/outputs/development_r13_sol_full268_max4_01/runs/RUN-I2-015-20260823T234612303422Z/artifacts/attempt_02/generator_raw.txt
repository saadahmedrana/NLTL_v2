    sh:or (

            sh:property [
                sh:path (

                sh:message "beta_i_prime is required for an applicable bow-subregion case."
            ]
        ]
        [
            sh:sparql [
                sh:message "Ship-level formula inputs are required when a linked case is classified as an I2.3.1(v) bow form." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
