gen:I2_018_ShipShape

    ] ;
    sh:property [
        sh:path nltl:hasIceLoadPatch ;
        sh:minCount 1 ;
        sh:class nltl:iceLoadPatch
    ] ;
    sh:property [
        sh:path nltl:hasPeakPressureFactorLookupCase ;
        sh:minCount 1 ;
        sh:class nltl:peakPressureFactorLookupCase ;
        sh:node gen:PeakPressureFactorLookupCaseShape ;
        sh:equals nltl:hasTableLookupCase
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:minCount 1 ;
        sh:class nltl:tableLookupCase ;
        sh:node gen:PeakPressureFactorLookupCaseShape ;
        sh:equals nltl:hasPeakPressureFactorLookupCase
    ] ;
    sh:sparql [
        sh:message "Average ice pressure must equal ice force divided by load-patch height and length, with kN converted to MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:averageIcePressure ?pressureQuantity ;
                      nltl:iceForce ?forceQuantity ;
                      nltl:loadPatchHeight ?heightQuantity ;
                      nltl:loadPatchLength ?lengthQuantity .
                ?pressureQuantity qudt:numericValue ?pressure .
                ?forceQuantity qudt:numericValue ?force .
                ?heightQuantity qudt:numericValue ?height .
                ?lengthQuantity qudt:numericValue ?length .
                BIND ((xsd:decimal("1000") * ?height * ?length) AS ?denominator)
                BIND (IF(
                    ?denominator = xsd:decimal("0"),
                    xsd:decimal("0"),
                    ?force / ?denominator
                ) AS ?expected)
                BIND (IF(
                    ABS(?expected) * xsd:decimal("0.000001") > xsd:decimal("0.000000000001"),
                    ABS(?expected) * xsd:decimal("0.000001"),
                    xsd:decimal("0.000000000001")
                ) AS ?tolerance)
                FILTER (
                    ?denominator = xsd:decimal("0") ||
                    ABS(?pressure - ?expected) > ?tolerance
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The reported peak pressure factor must match the Table 3 factor selected for the localized structural-member category." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:peakPressureFactor ?reportedQuantity ;
                      nltl:hasPeakPressureFactorLookupCase ?lookupCase .
                ?reportedQuantity qudt:numericValue ?reported .
                ?lookupCase nltl:selectedPeakPressureFactor ?selectedQuantity .
                ?selectedQuantity qudt:numericValue ?selected .
                BIND (IF(
                    ABS(?selected) * xsd:decimal("0.000001") > xsd:decimal("0.000000000001"),
                    ABS(?selected) * xsd:decimal("0.000001"),
                    xsd:decimal("0.000000000001")
                ) AS ?tolerance)
                FILTER (ABS(?reported - ?selected) > ?tolerance)
            }
        """
    ] .
