<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipHullAreaFactorShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:StructuralMemberHullAreaFactorShape
    ] .

gen:StructuralMemberHullAreaFactorShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:class nltl:hullAreaValue ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasSpannedHullArea ;
                sh:class nltl:hullAreaValue ;
                sh:node gen:SpannedHullAreaValueShape ;
                sh:minCount 2
            ] ;
            sh:property [
                sh:path nltl:selectedHullAreaFactor ;
                sh:node gen:UnitlessDecimalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For a member spanning at least two hull areas, the selected hull area factor must equal the maximum spanned-area factor within relative tolerance 1e-6 and absolute fallback 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasSpannedHullArea ?areaA, ?areaB, ?area ;
                      nltl:selectedHullAreaFactor ?selectedQuantity .
                FILTER (?areaA != ?areaB)

                ?area nltl:hullAreaFactor ?factorQuantity .
                ?factorQuantity qudt:numericValue ?factor ;
                                qudt:unit unit:UNITLESS .

                ?selectedQuantity qudt:numericValue ?reported ;
                                  qudt:unit unit:UNITLESS .
            }
            GROUP BY $this ?reported
            HAVING (
                ABS(?reported - MAX(?factor)) >
                IF(
                    ABS(MAX(?factor)) * "0.000001"^^xsd:decimal >
                        "0.000000000001"^^xsd:decimal,
                    ABS(MAX(?factor)) * "0.000001"^^xsd:decimal,
                    "0.000000000001"^^xsd:decimal
                )
            )
        """
    ] .

gen:SpannedHullAreaValueShape
    a sh:NodeShape ;
    sh:class nltl:hullAreaValue ;
    sh:property [
        sh:path nltl:hullAreaFactor ;
        sh:node gen:UnitlessDecimalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
