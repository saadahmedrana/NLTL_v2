<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipObliquePlatingInterpolationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:class nltl:obliquePlatingInterpolationCase ;
        sh:node gen:ObliquePlatingInterpolationCaseShape
    ] .

gen:ObliquePlatingInterpolationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DegreeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each oblique-plating interpolation case must state one framing angle."
    ] ;
    sh:property [
        sh:path nltl:longitudinalFramingNetPlateThickness ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:transverseFramingNetPlateThickness ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:node gen:InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolatedNetPlateThickness ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MillimetreQuantityShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:framingAngle ;
                sh:node gen:AngleAtOrBelowTwentyShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:framingAngle ;
                sh:node gen:AngleAtOrAboveSeventyShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:framingAngle ;
                sh:node gen:ApplicableObliqueAngleShape
            ] ;
            sh:property [
                sh:path nltl:longitudinalFramingNetPlateThickness ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:transverseFramingNetPlateThickness ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:interpolationLowerEndpoint ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:interpolationUpperEndpoint ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:interpolatedNetPlateThickness ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "Cases with framing angle greater than 20 degrees and less than 70 degrees require the declared interpolation inputs, endpoints, and computed result." .

gen:InterpolationPointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:class qudt:QuantityValue ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DecimalQuantityShape ;
        sh:minCount 1
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DecimalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AngleAtOrBelowTwentyShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:maxInclusive "20"^^xsd:decimal ;
        sh:minCount 1
    ] .

gen:AngleAtOrAboveSeventyShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "70"^^xsd:decimal ;
        sh:minCount 1
    ] .

gen:ApplicableObliqueAngleShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minExclusive "20"^^xsd:decimal ;
        sh:maxExclusive "70"^^xsd:decimal ;
        sh:minCount 1
    ] .
<END_SHACL>
