<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_034_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_034_StructuralMemberShape
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:class nltl:frameProfileTypeValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For each represented side-structure longitudinal-frame load case, combined shear-and-bending demand must not exceed the member plastic strength, and plastic strength must correspond to the midspan plastic-collapse load." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?member ?loadCase
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member nltl:longitudinalFrameOrientation true ;
            nltl:structuralLocation nltl:sideStructureLocation ;
            nltl:hasStructuralMemberLoadCase ?loadCase ;
            nltl:plasticStrength ?strengthQuantity ;
            nltl:midspanPlasticCollapseLoad ?collapseQuantity .
    ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
    ?strengthQuantity qudt:numericValue ?strength .
    ?collapseQuantity qudt:numericValue ?collapseLoad .
    ?demandQuantity qudt:numericValue ?demand .
    BIND(ABS(xsd:decimal(?collapseLoad)) * 0.000001 AS ?relativeTolerance)
    BIND(
        IF(
            ?relativeTolerance > 0.000000001,
            ?relativeTolerance,
            0.000000001
        ) AS ?tolerance
    )
    FILTER(
        xsd:decimal(?demand) > xsd:decimal(?strength) ||
        xsd:decimal(?demand) > xsd:decimal(?collapseLoad) ||
        ABS(xsd:decimal(?strength) - xsd:decimal(?collapseLoad)) > ?tolerance
    )
}
"""
    ] .

gen:I2_034_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each represented structural member must state whether it is a longitudinal frame."
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:sideStructureLocation
            nltl:bottomStructureLocation
        ) ;
        sh:message "Each represented structural member must identify its structural location."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:sideStructureLocation
            ] ;
            sh:property [
                sh:path nltl:plasticStrength ;
                sh:minCount 1 ;
                sh:node gen:I2_034_ForceQuantityShape ;
                sh:message "An applicable side-structure longitudinal frame must provide its plastic strength."
            ] ;
            sh:property [
                sh:path nltl:midspanPlasticCollapseLoad ;
                sh:minCount 1 ;
                sh:node gen:I2_034_ForceQuantityShape ;
                sh:message "An applicable side-structure longitudinal frame must provide its midspan plastic-collapse load."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMemberLoadCase ;
                sh:class nltl:loadCase ;
                sh:node gen:I2_034_LoadCaseShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:longitudinalFrameOrientation ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:structuralLocation ;
                sh:hasValue nltl:bottomStructureLocation
            ]
        ]
    ) .

gen:I2_034_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:node gen:I2_034_DemandQuantityShape ;
        sh:message "Each represented applicable load case must provide its combined shear-and-bending demand."
    ] .

gen:I2_034_ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloN>
    ] .

gen:I2_034_DemandQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
