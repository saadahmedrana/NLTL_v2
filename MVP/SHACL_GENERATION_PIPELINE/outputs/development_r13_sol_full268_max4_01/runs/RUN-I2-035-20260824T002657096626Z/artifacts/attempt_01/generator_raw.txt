        ]

                sh:path nltl:averageIcePressure ;

                    ] ;

                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:frameSpan ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [

                    ] ;

                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:peakPressureFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;

                ]
            ] ;
            sh:property [
                sh:path nltl:shearArea ;
                sh:minCount 1 ;

                    sh:class qudt:QuantityValue ;

                        sh:path qudt:numericValue ;

            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minExclusive "0"^^xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MegaPA ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each linked structural member must report the calculated required shear area, and the actual shear area must be at least that calculated requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?member
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:averageIcePressure/qudt:numericValue ?averagePressure ;
                      nltl:frameSpan/qudt:numericValue ?frameSpan ;
                      nltl:loadPatchHeight/qudt:numericValue ?patchHeight ;
                      nltl:peakPressureFactor/qudt:numericValue ?peakFactor ;
                      nltl:shearArea/qudt:numericValue ?actualShearArea ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength .
                ?member nltl:frameSpacing/qudt:numericValue ?frameSpacing ;
                        nltl:selectedHullAreaFactor/qudt:numericValue ?areaFactor ;
                        nltl:requiredLongitudinalFrameShearArea/qudt:numericValue ?reportedRequiredArea .
                BIND((?patchHeight / ?frameSpacing) AS ?bPrime)
                BIND(
                    IF(
                        ?bPrime < 2,
                        ?patchHeight * (1 - 0.25 * ?bPrime),
                        ?frameSpacing
                    )
                    AS ?b2
                )
                BIND(((1 - 0.3 / ?bPrime) * ?b2) AS ?b1)
                BIND(
                    (
                        10000 * (?areaFactor * ?peakFactor * ?averagePressure)
                        * 0.5 * ?b1 * ?frameSpan
                    ) / (0.577 * ?yieldStrength)
                    AS ?expectedRequiredArea
                )
                BIND(
                    IF(
                        ABS(?expectedRequiredArea) * 0.000001 > 0.000000001,
                        ABS(?expectedRequiredArea) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(
                    ABS(?reportedRequiredArea - ?expectedRequiredArea) > ?tolerance
                    ||
                    (?actualShearArea * 10000) < ?expectedRequiredArea
                )
            }
        """
    ] .
