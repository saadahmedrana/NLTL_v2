<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_037_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_037_StructuralMemberDispatchShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship with a web frame or load-carrying stringer must identify the applicable ice load patch." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    FILTER (
        EXISTS { ?member a nltl:webFrame . } ||
        EXISTS { ?member a nltl:loadCarryingStringer . }
    )
    FILTER NOT EXISTS {
        $this nltl:hasIceLoadPatch ?patch .
        ?patch a nltl:iceLoadPatch .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each applicable structural-member load case must be linked from the ship as the same ice-load-patch design case." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasStructuralMember ?member .
    FILTER (
        EXISTS { ?member a nltl:webFrame . } ||
        EXISTS { ?member a nltl:loadCarryingStringer . }
    )
    ?member nltl:hasStructuralMemberLoadCase ?case .
    FILTER NOT EXISTS {
        $this nltl:hasIceLoadPatchDesignCase ?case .
        ?case a nltl:iceLoadPatchDesignCase .
    }
}
"""
    ] .

gen:I2_037_StructuralMemberDispatchShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:not [
                sh:or (
                    [ sh:class nltl:webFrame ]
                    [ sh:class nltl:loadCarryingStringer ]
                )
            ]
        ]
        [
            sh:node gen:I2_037_ApplicableStructuralMemberShape
        ]
    ) .

gen:I2_037_ApplicableStructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:memberCapacity ;
        sh:minCount 1 ;
        sh:node gen:I2_037_MemberCapacityValueShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:minCount 1 ;
        sh:class nltl:loadCase ;
        sh:nodeKind sh:IRI ;
        sh:node gen:I2_037_IceLoadPatchDesignCaseShape
    ] .

gen:I2_037_MemberCapacityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_037_IceLoadPatchDesignCaseShape
    a sh:NodeShape ;
    sh:class nltl:iceLoadPatchDesignCase ;
    sh:property [
        sh:path nltl:combinedShearBendingEffect ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:loadPatchApplicationLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:memberCapacityMinimizationConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] .
<END_SHACL>
