<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SteelRenewalRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Gauged thickness must be provided as one decimal quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Net thickness must be provided as one decimal quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Steel renewal status must be provided as one boolean value."
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be required when gauged thickness is less than net thickness plus 0.5 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:gaugedThickness ?gaugedQuantity ;
                      nltl:netThickness ?netQuantity ;
                      nltl:steelRenewalRequired ?renewalRequired .
                ?gaugedQuantity qudt:numericValue ?gauged .
                ?netQuantity qudt:numericValue ?net .
                FILTER (
                    ?gauged < (?net + "0.0005"^^xsd:decimal) &&
                    ?renewalRequired != "true"^^xsd:boolean
                )
            }
        """
    ] .
<END_SHACL>
