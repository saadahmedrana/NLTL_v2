<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_047_LengthInMetresShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal numeric thickness value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Thickness must use metre as its unit."
    ] .

gen:I2_047_ShipShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:hasComponent ?structure .
                ?structure a nltl:hullStructure ;
                           nltl:iceStrengthenedStructureStatus "true"^^xsd:boolean .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:node gen:I2_047_LengthInMetresShape ;
        sh:minCount 1 ;
        sh:message "Gauged thickness is required for an ice-strengthened ship."
    ] ;
    sh:property [
        sh:path nltl:netThickness ;
        sh:node gen:I2_047_LengthInMetresShape ;
        sh:minCount 1 ;
        sh:message "Net thickness is required for an ice-strengthened ship."
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Steel renewal status is required for an ice-strengthened ship."
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be required when gauged thickness is less than net thickness plus 0.0005 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:gaugedThickness ?gaugedQuantity ;
                      nltl:netThickness ?netQuantity .
                ?gaugedQuantity qudt:numericValue ?gauged .
                ?netQuantity qudt:numericValue ?net .
                FILTER (?gauged < (?net + "0.0005"^^xsd:decimal))
                FILTER NOT EXISTS {
                    $this nltl:steelRenewalRequired "true"^^xsd:boolean .
                }
            }
        """
    ] .
<END_SHACL>
