<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_055_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designStress ;
        sh:minCount 1 ;
        sh:node gen:I2_055_StressValueShape ;
        sh:message "A design stress value in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:permissibleStress ;
        sh:minCount 1 ;
        sh:node gen:I2_055_StressValueShape ;
        sh:message "A permissible stress value in MPa is required."
    ] ;
    sh:property [
        sh:path nltl:strengthCriterionStatus ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:message "The strength criterion status must be true."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The design stress must not exceed the permissible stress." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:designStress ?designQuantity ;
          nltl:permissibleStress ?permissibleQuantity .
    ?designQuantity qudt:numericValue ?designStress .
    ?permissibleQuantity qudt:numericValue ?permissibleStress .
    FILTER (?designStress > ?permissibleStress)
}
"""
    ] .

gen:I2_055_StressValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The stress quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaPA ;
        sh:message "The stress quantity must use MPa."
    ] .
<END_SHACL>
