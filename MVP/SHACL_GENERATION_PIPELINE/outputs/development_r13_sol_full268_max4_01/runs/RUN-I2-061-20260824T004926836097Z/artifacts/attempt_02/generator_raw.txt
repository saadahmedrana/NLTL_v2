<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_061_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gen:I2_061_CalculationCaseShape
    ] .

gen:I2_061_CalculationCaseShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationCaseAssessedHullStructure ;
                sh:maxCount 0
            ]
        ]
        [
            sh:node gen:I2_061_AssessedCalculationCaseShape
        ]
    ) .

gen:I2_061_AssessedCalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationCaseAssessedHullStructure ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:hullStructure ;
        sh:node gen:I2_061_AssessedHullStructureShape ;
        sh:message "Each assessed hull structure must provide conforming thickness information."
    ] ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:hasValue nltl:prescribedAnalyticalProcedure ;
        sh:message "An assessed calculation case must use the prescribed analytical procedure."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A direct calculation method must not replace the prescribed analytical procedure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:calculationMethod nltl:directCalculationMethodValue .
            }
        """
    ] .

gen:I2_061_AssessedHullStructureShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:I2_061_ThicknessValueShape ;
        sh:message "An assessed hull structure must have thickness expressed as a decimal quantity in metres."
    ] .

gen:I2_061_ThicknessValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:maxCount 1
    ] .
<END_SHACL>
