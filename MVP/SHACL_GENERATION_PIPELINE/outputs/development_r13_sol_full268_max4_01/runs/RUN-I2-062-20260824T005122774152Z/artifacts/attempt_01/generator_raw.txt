<BEGIN_SHACL>
@prefix shape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:I2_062_DirectCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:property [
                sh:path nltl:partOfGrillageSystem ;
                sh:class nltl:grillageSystem
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Direct calculation use must be recorded as true when a linked load-carrying stringer or web frame forms part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType ;
                    nltl:partOfGrillageSystem ?grillage .
                FILTER (
                    ?memberType IN (
                        nltl:loadCarryingStringer,
                        nltl:webFrame
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:directCalculationUsed true .
                }
            }
        """
    ] .
<END_SHACL>
