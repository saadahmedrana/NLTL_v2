<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipCalculationCaseModelShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node gen:CalculationCaseSelectorShape ;
        sh:message "Each linked calculation case must identify its calculation method."
    ] .

gen:CalculationCaseSelectorShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:class nltl:calculationMethodValue ;
        sh:message "A calculation case must identify a calculation method value."
    ] .

gen:NonlinearCalculationShipReadinessShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:hasCalculationCase ?case .
                ?case nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:fractureMargin ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DimensionlessRatioValueShape ;
        sh:message "A nonlinear calculation requires a fracture margin."
    ] ;
    sh:property [
        sh:path nltl:majorBucklingMargin ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DimensionlessRatioValueShape ;
        sh:message "A nonlinear calculation requires a major buckling margin."
    ] ;
    sh:property [
        sh:path nltl:yieldingMargin ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:DimensionlessRatioValueShape ;
        sh:message "A nonlinear calculation requires a yielding margin."
    ] .

gen:NonlinearCalculationCaseReadinessShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?ship a nltl:ship ;
                    nltl:hasCalculationCase ?this .
                ?this nltl:calculationMethod nltl:nonlinearCalculationMethodValue .
            }
        """
    ] ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:permanentLateralDeformation ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:LengthValueShape ;
        sh:message "A nonlinear calculation case requires permanent lateral deformation."
    ] ;
    sh:property [
        sh:path nltl:permanentOutOfPlaneDeformation ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:LengthValueShape ;
        sh:message "A nonlinear calculation case requires permanent out-of-plane deformation."
    ] ;
    sh:property [
        sh:path nltl:relevantStructuralDimension ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node gen:LengthValueShape ;
        sh:message "A nonlinear calculation case requires the relevant structural dimension."
    ] ;
    sh:property [
        sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "A nonlinear calculation case requires acceptance evidence."
    ] ;
    sh:property [
        sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:message "The nonlinear calculation case must confirm that the minor permanent deformation assessment is satisfied."
    ] .

gen:DimensionlessRatioValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "A dimensionless margin requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS ;
        sh:message "A dimensionless margin must use the unit UNITLESS."
    ] .

gen:LengthValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "A deformation or structural dimension requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM ;
        sh:message "A deformation or structural dimension must use millimetres."
    ] .
<END_SHACL>
