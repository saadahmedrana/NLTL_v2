<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "The ship must declare at least one design ice condition."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:mediumFirstYearIceWithPossibleOldIceInclusions
            ] ;
            sh:property [
                sh:path nltl:shipCategory ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in ( nltl:polarShipCategoryA ) ;
                sh:message "A ship designed for medium first-year ice with possible old-ice inclusions must have only Polar Ship Category A values."
            ]
        ]
        [
            sh:not [
                sh:path nltl:designIceCondition ;
                sh:hasValue nltl:mediumFirstYearIceWithPossibleOldIceInclusions
            ]
        ]
    ) .
<END_SHACL>
