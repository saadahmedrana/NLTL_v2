<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO003ShipCategoryShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceConditionValue ;
        sh:message "A design ice condition is required to determine whether the Category C rule applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship designed for open water or ice conditions less severe than categories A and B shall have ship category C." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this
WHERE {
    $this nltl:designIceCondition ?condition .
    FILTER(
        ?condition IN (
            nltl:openWaterIceCondition,
            nltl:iceConditionLessSevereThanCategoryAAndB
        )
    )
    FILTER(
        NOT EXISTS {
            $this nltl:shipCategory nltl:polarShipCategoryC .
        }
        ||
        EXISTS {
            $this nltl:shipCategory ?category .
            FILTER(?category != nltl:polarShipCategoryC)
        }
    )
}
"""
    ] .
<END_SHACL>
