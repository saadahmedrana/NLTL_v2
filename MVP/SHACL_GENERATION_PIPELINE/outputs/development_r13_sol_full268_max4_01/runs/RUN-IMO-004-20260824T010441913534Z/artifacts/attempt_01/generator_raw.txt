<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FirstYearIceClassificationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Ice origin type is required to determine first-year ice classification."
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Winter growth count must be provided as an integer."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Ice thickness must be provided as a decimal quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Sea ice of not more than one winter growth and thickness from 0.3 m to 2.0 m inclusive must be classified as first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
          nltl:winterGrowthCount ?growthCount ;
          nltl:iceThickness ?thickness .
    ?thickness qudt:numericValue ?thicknessValue ;
               qudt:unit <http://qudt.org/vocab/unit/M> .
    FILTER (
        ?growthCount <= "1"^^xsd:integer &&
        ?thicknessValue >= "0.3"^^xsd:decimal &&
        ?thicknessValue <= "2.0"^^xsd:decimal
    )
    FILTER NOT EXISTS {
        $this nltl:iceTypeClassification nltl:firstYearIce .
    }
}
"""
    ] .
<END_SHACL>
