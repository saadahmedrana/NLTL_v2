<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IceFreeWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:totalIcePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ice condition is 'ice free waters', total ice present must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceCondition ?condition .
                FILTER (?condition = "ice free waters"^^xsd:string)
                FILTER (
                    NOT EXISTS {
                        $this nltl:totalIcePresent ?present .
                        FILTER (?present = "false"^^xsd:boolean)
                    }
                    ||
                    EXISTS {
                        $this nltl:totalIcePresent ?otherPresent .
                        FILTER (?otherPresent != "false"^^xsd:boolean)
                    }
                )
            }
        """
    ] .
<END_SHACL>
