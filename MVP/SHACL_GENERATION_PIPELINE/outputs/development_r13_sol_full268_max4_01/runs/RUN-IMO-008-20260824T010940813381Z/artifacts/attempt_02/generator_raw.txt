<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ThinFirstYearIceThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:message "Ice type classifications must be IRIs."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified as thin first-year ice must have ice thickness from 0.3 m through 0.7 m inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:thinFirstYearIce .
                {
                    FILTER NOT EXISTS {
                        $this nltl:iceThickness ?anyThickness .
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    FILTER (
                        NOT EXISTS {
                            ?thickness a qudt:QuantityValue .
                        }
                        ||
                        NOT EXISTS {
                            ?thickness qudt:unit unit:M .
                        }
                        ||
                        EXISTS {
                            ?thickness qudt:unit ?otherUnit .
                            FILTER (?otherUnit != unit:M)
                        }
                        ||
                        NOT EXISTS {
                            ?thickness qudt:numericValue ?value .
                        }
                        ||
                        EXISTS {
                            ?thickness qudt:numericValue ?value .
                            FILTER (
                                DATATYPE(?value) != xsd:decimal
                                ||
                                ?value < "0.3"^^xsd:decimal
                                ||
                                ?value > "0.7"^^xsd:decimal
                            )
                        }
                    )
                }
            }
        """
    ] .
<END_SHACL>
