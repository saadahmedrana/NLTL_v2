<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:BergyWatersShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:landOriginIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:totalIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For bergy waters, land-origin ice concentration must be less than 1 and total ice concentration must not exceed 1." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceCondition ?condition .
    FILTER(?condition = "bergy waters")
    FILTER(
        NOT EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
        }
        ||
        NOT EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
        }
        ||
        EXISTS {
            $this nltl:landOriginIceConcentration ?landQuantity .
            ?landQuantity qudt:numericValue ?landValue .
            FILTER(?landValue >= "1"^^xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:totalIceConcentration ?totalQuantity .
            ?totalQuantity qudt:numericValue ?totalValue .
            FILTER(?totalValue > "1"^^xsd:decimal)
        }
    )
}
"""
    ] .
<END_SHACL>
