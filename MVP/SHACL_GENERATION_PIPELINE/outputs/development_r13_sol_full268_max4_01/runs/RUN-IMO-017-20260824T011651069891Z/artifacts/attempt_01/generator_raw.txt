@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO017ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate validity dates are required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate survey dates are required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Polar Ship Certificate endorsement dates are required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasCertificateScheduleShape ;
        sh:minCount 1 ;
        sh:message "An applicable SOLAS certificate schedule is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The Polar Ship Certificate supplement must be recorded as present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017PolarCertificateSupplementShape ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate supplement with required equipment records is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate validity dates must match the applicable SOLAS certificate schedule." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasSolasCertificateSchedule ?schedule .
    {
        $this nltl:polarCertificateValidityDates ?date .
        FILTER NOT EXISTS {
            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
            ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                    nltl:certificateScheduleDate ?date .
        }
    }
    UNION
    {
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDateType nltl:validityDateCategory ;
                nltl:certificateScheduleDate ?date .
        FILTER NOT EXISTS {
            $this nltl:polarCertificateValidityDates ?date .
        }
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate survey dates must match the applicable SOLAS certificate schedule." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasSolasCertificateSchedule ?schedule .
    {
        $this nltl:surveyDates ?date .
        FILTER NOT EXISTS {
            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
            ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                    nltl:certificateScheduleDate ?date .
        }
    }
    UNION
    {
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDateType nltl:surveyDateCategory ;
                nltl:certificateScheduleDate ?date .
        FILTER NOT EXISTS {
            $this nltl:surveyDates ?date .
        }
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Polar Ship Certificate endorsement dates must match the applicable SOLAS certificate schedule." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasSolasCertificateSchedule ?schedule .
    {
        $this nltl:endorsementDates ?date .
        FILTER NOT EXISTS {
            ?schedule nltl:hasCertificateScheduleDateRecord ?record .
            ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                    nltl:certificateScheduleDate ?date .

    }
    UNION
    {
        ?schedule nltl:hasCertificateScheduleDateRecord ?record .
        ?record nltl:certificateScheduleDateType nltl:endorsementDateCategory ;
                nltl:certificateScheduleDate ?date .
        FILTER NOT EXISTS {
            $this nltl:endorsementDates ?date .
        }
    }
}
"""
    ] .

gen:IMO017SolasCertificateScheduleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017CertificateScheduleDateRecordShape ;
        sh:minCount 1 ;
        sh:message "The SOLAS certificate schedule must contain categorized date records."
    ] .

gen:IMO017CertificateScheduleDateRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A certificate schedule date record must contain a date."
    ] ;
    sh:property [
        sh:path nltl:certificateScheduleDateType ;
        sh:minCount 1 ;
        sh:in (
            nltl:validityDateCategory
            nltl:surveyDateCategory
            nltl:endorsementDateCategory
        ) ;
        sh:message "A certificate schedule date record must use a validity, survey, or endorsement category."
    ] .

gen:IMO017PolarCertificateSupplementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The certificate supplement must include required equipment records."
    ] .
