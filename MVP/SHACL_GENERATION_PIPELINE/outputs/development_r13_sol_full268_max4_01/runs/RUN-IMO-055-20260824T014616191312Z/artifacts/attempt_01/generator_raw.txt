<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO055ValveProtectionAccessibilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Valve exposure status is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed valve requires a valve type to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:valveExposed ?exposed .
                FILTER (?exposed = true)
                FILTER NOT EXISTS {
                    $this nltl:valveType ?type .
                    FILTER (datatype(?type) = xsd:string)
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed isolating, pressure, or vacuum valve must be protected from ice accretion and continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:valveExposed ?exposed ;
                      nltl:valveType ?type .
                FILTER (
                    ?exposed = true &&
                    ?type IN ("isolating", "pressure", "vacuum")
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceAccretionProtectionPresent ?protected .
                        FILTER (
                            datatype(?protected) = xsd:boolean &&
                            ?protected = true
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:continuouslyAccessible ?accessible .
                        FILTER (
                            datatype(?accessible) = xsd:boolean &&
                            ?accessible = true
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
