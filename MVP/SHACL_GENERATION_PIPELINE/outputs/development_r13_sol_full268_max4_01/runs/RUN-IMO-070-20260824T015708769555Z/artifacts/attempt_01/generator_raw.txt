<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PassengerShipSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:passengerShip ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Passenger-ship survival equipment shall be properly sized."
    ] ;
    sh:property [
        sh:path nltl:hasPersonOnBoardMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:personOnBoardMember ;
        sh:node gen:PersonSurvivalEquipmentAssignmentShape ;
        sh:message "Every represented person on board shall have a conforming survival-equipment assignment."
    ] .

gen:PersonSurvivalEquipmentAssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasPersonalSurvivalEquipmentAssignment ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:personalSurvivalEquipmentAssignment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SurvivalEquipmentAssignmentShape ;
        sh:message "Each person shall have one personal survival-equipment assignment."
    ] .

gen:SurvivalEquipmentAssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:assignedPersonalSurvivalEquipment ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:personalSurvivalEquipment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PermittedSurvivalEquipmentShape ;
        sh:message "The assignment shall identify one immersion suit or thermal protective aid."
    ] .

gen:PermittedSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:immersionSuitEquipment
        ]
        [
            sh:class nltl:thermalProtectiveAidEquipment
        ]
    ) ;
    sh:message "Assigned equipment shall be an immersion suit or a thermal protective aid." .
<END_SHACL>
