<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO082Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:solasRegulationV22Point1Point9Point4ComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The ship must comply with SOLAS regulation V/22.1.9.4."
    ] ;
    sh:property [
        sh:path nltl:bridgeConfigurationType ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:bridgeConfigurationValue ;
        sh:minCount 1 ;
        sh:message "A bridge configuration type is required to determine the astern-view obligation."
    ] ;
    sh:property [
        sh:path nltl:clearViewAsternStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship must have a clear view astern when required by its bridge configuration." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:bridgeConfigurationType nltl:asternViewRequiredBridgeConfiguration .
    FILTER NOT EXISTS {
        $this nltl:clearViewAsternStatus ?status .
        FILTER(
            DATATYPE(?status) = xsd:boolean &&
            ?status = "true"^^xsd:boolean
        )
    }
}
"""
    ] .
<END_SHACL>
