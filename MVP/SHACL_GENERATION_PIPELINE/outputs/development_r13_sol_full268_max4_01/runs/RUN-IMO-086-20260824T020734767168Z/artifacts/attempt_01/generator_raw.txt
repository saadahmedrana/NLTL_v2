<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO086ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:message "The ship shall have at least two non-magnetic heading means." ;
        sh:node [
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to main power."
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1 ;
                sh:message "Each non-magnetic heading means shall be connected to emergency power."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means shall identify another distinct means owned by the same ship as independent." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?means
WHERE {
    $this nltl:hasNonMagneticHeadingMeans ?means .
    FILTER NOT EXISTS {
        $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
        ?means nltl:independentFromHeadingMeans ?otherMeans .
        FILTER (?otherMeans != ?means)
    }
}
"""
    ] .
<END_SHACL>
