<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO087Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship proceeding above 80 degrees must have at least one GNSS compass or equivalent and connections to main and emergency power entities." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:plannedOrActualLatitude ?latitude .
    ?latitude qudt:numericValue ?latitudeValue ;
              qudt:unit unit:DEG .
    FILTER (
        datatype(?latitudeValue) = xsd:decimal &&
        ?latitudeValue > "80"^^xsd:decimal
    )
    FILTER (
        NOT EXISTS {
            $this nltl:gnssCompassOrEquivalentCount ?count .
            FILTER (
                datatype(?count) = xsd:integer &&
                ?count >= "1"^^xsd:integer
            )
        }
        ||
        NOT EXISTS {
            $this nltl:connectedToMainPower ?mainPower .
            ?mainPower rdf:type ?mainPowerType .
            ?mainPowerType rdfs:subClassOf* nltl:benchmarkEntity .
        }
        ||
        NOT EXISTS {
            $this nltl:connectedToEmergencyPower ?emergencyPower .
            ?emergencyPower rdf:type ?emergencyPowerType .
            ?emergencyPowerType rdfs:subClassOf* nltl:benchmarkEntity .
        }
    )
}
"""
    ] .
<END_SHACL>
