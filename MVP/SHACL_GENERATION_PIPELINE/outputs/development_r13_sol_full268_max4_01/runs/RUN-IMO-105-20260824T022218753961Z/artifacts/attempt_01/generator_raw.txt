<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO_105_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The operating area is required to determine whether the Arctic-waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "In Arctic waters, every recorded oil or oily mixture discharge-to-sea state must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:operatingArea ?area .
                FILTER(?area = "Arctic_waters"^^xsd:string)
                FILTER(
                    NOT EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?permittedState .
                        FILTER(isLiteral(?permittedState))
                        FILTER(datatype(?permittedState) = xsd:boolean)
                        FILTER(STR(?permittedState) IN ("false", "0"))
                    }
                    ||
                    EXISTS {
                        $this nltl:oilOrOilyMixtureDischargeToSea ?nonconformingState .
                        FILTER(
                            IF(
                                isLiteral(?nonconformingState),
                                datatype(?nonconformingState) != xsd:boolean ||
                                STR(?nonconformingState) NOT IN ("false", "0"),
                                true
                            )
                        )
                    }
                )
            }
        """
    ] .
<END_SHACL>
