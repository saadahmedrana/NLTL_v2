<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ArcticWatersOilDischargeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine whether the Arctic waters prohibition applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A recorded oil or oily mixture discharge to sea in Arctic waters must be boolean false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?discharge
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:oilOrOilyMixtureDischargeToSea ?discharge .
                FILTER (?area = "Arctic_waters"^^xsd:string)
                FILTER (
                    DATATYPE(?discharge) != xsd:boolean ||
                    ?discharge != false
                )
            }
        """
    ] .
<END_SHACL>
