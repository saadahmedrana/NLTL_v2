<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_114_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship category is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The Type 3 tank requirement status of the carried noxious liquid substance is required."
    ] ;
    sh:property [
        sh:path nltl:cargoTankType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A cargo tank type is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "An applicable ship must have Administration carriage approval and a certificate entry identifying polar-water operation." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:shipCategory ?category ;
          nltl:constructionDate ?constructionDate ;
          nltl:carriedNoxiousLiquidSubstanceRequiresType3Tank ?requiresType3 ;
          nltl:cargoTankType ?tankType .

    FILTER (
        ?category IN (
            nltl:polarShipCategoryA,
            nltl:polarShipCategoryB
        )
    )
    FILTER (DATATYPE(?constructionDate) = xsd:date)
    FILTER (
        YEAR(?constructionDate) > 2017 ||
        (
            YEAR(?constructionDate) = 2017 &&
            (
                MONTH(?constructionDate) > 1 ||
                (
                    MONTH(?constructionDate) = 1 &&
                    DAY(?constructionDate) >= 1
                )
            )
        )
    )
    FILTER (?requiresType3 = true)
    FILTER (?tankType = "3"^^xsd:string)

    FILTER (
        NOT EXISTS {
            $this nltl:administrationCarriageApprovalStatus
                  nltl:evidenceStateApproved .
        }
        ||
        NOT EXISTS {
            $this nltl:applicableCertificatePolarOperationEntryPresent
                  ?certificateEntry .
            FILTER (?certificateEntry = true)
        }
    )
}
"""
    ] .
<END_SHACL>
