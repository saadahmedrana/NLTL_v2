@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ApplicabilityInformationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area information is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:arcticWaters ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Arctic-waters information is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Food-waste discharge status is required to determine applicability."
    ] .

gen:ApplicableArcticFoodWasteDischargeShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT ?this
WHERE {
    ?this a nltl:ship ;
          nltl:operatingArea ?operatingArea ;
          nltl:arcticWaters ?arcticWaters ;
          nltl:foodWasteDischargeToSea ?discharge .
    FILTER(?operatingArea = ?arcticWaters && ?discharge = true)
}
"""
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestLandIceShelfOrFastIce ;
        sh:minCount 1 ;
        sh:message "At least one nearest-land, ice-shelf, or fast-ice distance is required for an applicable discharge." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minInclusive "12"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Every reported nearest distance must be at least 12 nautical miles."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:minCount 1 ;
        sh:message "Ice concentration is required for an applicable discharge." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:minCount 1 ;
        sh:message "Nearby ice concentration is required for an applicable discharge." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gen:IceConcentrationAboveOneTenthShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT ?this
WHERE {
    ?this a nltl:ship ;
          nltl:operatingArea ?operatingArea ;
          nltl:arcticWaters ?arcticWaters ;
          nltl:foodWasteDischargeToSea ?discharge ;
          nltl:iceConcentration ?iceConcentration .
    ?iceConcentration qudt:numericValue ?iceValue .
    FILTER(
        ?operatingArea = ?arcticWaters &&
        ?discharge = true &&
        ?iceValue > 1
    )
}
"""
    ] ;
    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:minCount 1 ;
        sh:message "Distance from an area with ice concentration above one tenth must be recorded." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
