<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MI_N ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AntarcticDischargeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The operating area is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Antarctic area value is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The MARPOL Annex V regulation 6.1 discharge status is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:distanceToNearestFastIce ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:node gen:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceConcentration ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeOntoIce ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a regulation 6.1 discharge in the Antarctic area, the nearest-fast-ice distance must be at least 12 nautical miles, the required concentration-area distance and ice concentrations must be recorded, and food waste discharge onto ice must be false." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:operatingArea ?operatingArea ;
          nltl:antarcticArea ?antarcticArea ;
          nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?dischargeOccurs .
    FILTER(
        ?operatingArea = ?antarcticArea &&
        ?dischargeOccurs = true
    )
    FILTER NOT EXISTS {
        $this nltl:distanceToNearestFastIce ?fastIceDistance ;
              nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?concentrationAreaDistance ;
              nltl:nearbyIceConcentration ?nearbyConcentration ;
              nltl:iceConcentration ?iceConcentration ;
              nltl:foodWasteDischargeOntoIce ?dischargeOntoIce .
        ?fastIceDistance qudt:numericValue ?fastIceDistanceValue .
        ?concentrationAreaDistance qudt:numericValue ?concentrationAreaDistanceValue .
        ?nearbyConcentration qudt:numericValue ?nearbyConcentrationValue .
        ?iceConcentration qudt:numericValue ?iceConcentrationValue .
        FILTER(
            ?fastIceDistanceValue >= "12"^^xsd:decimal &&
            ?nearbyConcentrationValue > "1"^^xsd:decimal &&
            ?iceConcentrationValue > "1"^^xsd:decimal &&
            ?dischargeOntoIce = false
        )
    }
}
"""
    ] .
<END_SHACL>
