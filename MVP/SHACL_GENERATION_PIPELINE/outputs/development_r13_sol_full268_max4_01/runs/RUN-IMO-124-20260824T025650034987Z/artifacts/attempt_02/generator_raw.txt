<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO124Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Operating area is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:antarcticArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "The Antarctic-area value is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The MARPOL Annex V regulation 6.1 discharge state is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a regulation 6.1 discharge in the Antarctic area: distance to nearest fast ice must be at least 12 nautical miles; the distance from an area with ice concentration above one tenth, nearby ice concentration, and qualifying ice concentration must be recorded; and food waste must not be discharged onto ice." ;
        sh:select """
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this
WHERE {
    $this nltl:operatingArea ?operatingArea ;
          nltl:antarcticArea ?antarcticArea ;
          nltl:marpolAnnexVRegulation6Point1DischargeOccurs true .

    FILTER(?operatingArea = ?antarcticArea)

    FILTER(
        NOT EXISTS {
            $this nltl:distanceToNearestFastIce ?fastIceDistance .
            ?fastIceDistance
                rdf:type qudt:QuantityValue ;
                qudt:numericValue ?fastIceValue ;
                qudt:unit unit:MI_N .
            FILTER(
                datatype(?fastIceValue) = xsd:decimal &&
                ?fastIceValue >= "12"^^xsd:decimal
            )
        }
        ||
        EXISTS {
            $this nltl:distanceToNearestFastIce ?fastIceDistance .
            FILTER NOT EXISTS {
                ?fastIceDistance
                    rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?fastIceValue ;
                    qudt:unit unit:MI_N .
                FILTER(
                    datatype(?fastIceValue) = xsd:decimal &&
                    ?fastIceValue >= "12"^^xsd:decimal
                )
            }
        }
        ||
        EXISTS {
            $this nltl:distanceToNearestFastIce ?fastIceDistance .
            ?fastIceDistance qudt:unit ?fastIceUnit .
            FILTER(?fastIceUnit != unit:MI_N)
        }
        ||
        NOT EXISTS {
            $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
            ?areaDistance
                rdf:type qudt:QuantityValue ;
                qudt:numericValue ?areaDistanceValue ;
                qudt:unit unit:MI_N .
            FILTER(datatype(?areaDistanceValue) = xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
            FILTER NOT EXISTS {
                ?areaDistance
                    rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?areaDistanceValue ;
                    qudt:unit unit:MI_N .
                FILTER(datatype(?areaDistanceValue) = xsd:decimal)
            }
        }
        ||
        EXISTS {
            $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
            ?areaDistance qudt:unit ?areaDistanceUnit .
            FILTER(?areaDistanceUnit != unit:MI_N)
        }
        ||
        NOT EXISTS {
            $this nltl:nearbyIceConcentration ?nearbyConcentration .
            ?nearbyConcentration
                rdf:type qudt:QuantityValue ;
                qudt:numericValue ?nearbyConcentrationValue ;
                qudt:unit unit:UNITLESS .
            FILTER(datatype(?nearbyConcentrationValue) = xsd:decimal)
        }
        ||
        EXISTS {
            $this nltl:nearbyIceConcentration ?nearbyConcentration .
            FILTER NOT EXISTS {
                ?nearbyConcentration
                    rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?nearbyConcentrationValue ;
                    qudt:unit unit:UNITLESS .
                FILTER(datatype(?nearbyConcentrationValue) = xsd:decimal)
            }
        }
        ||
        EXISTS {
            $this nltl:nearbyIceConcentration ?nearbyConcentration .
            ?nearbyConcentration qudt:unit ?nearbyConcentrationUnit .
            FILTER(?nearbyConcentrationUnit != unit:UNITLESS)
        }
        ||
        NOT EXISTS {
            $this nltl:iceConcentration ?iceConcentration .
            ?iceConcentration
                rdf:type qudt:QuantityValue ;
                qudt:numericValue ?iceConcentrationValue ;
                qudt:unit unit:UNITLESS .
            FILTER(
                datatype(?iceConcentrationValue) = xsd:decimal &&
                ?iceConcentrationValue > "1"^^xsd:decimal
            )
        }
        ||
        EXISTS {
            $this nltl:iceConcentration ?iceConcentration .
            FILTER NOT EXISTS {
                ?iceConcentration
                    rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?iceConcentrationValue ;
                    qudt:unit unit:UNITLESS .
                FILTER(
                    datatype(?iceConcentrationValue) = xsd:decimal &&
                    ?iceConcentrationValue > "1"^^xsd:decimal
                )
            }
        }
        ||
        EXISTS {
            $this nltl:iceConcentration ?iceConcentration .
            ?iceConcentration qudt:unit ?iceConcentrationUnit .
            FILTER(?iceConcentrationUnit != unit:UNITLESS)
        }
        ||
        NOT EXISTS {
            $this nltl:foodWasteDischargeOntoIce false .
        }
        ||
        EXISTS {
            $this nltl:foodWasteDischargeOntoIce ?foodWasteState .
            FILTER(
                datatype(?foodWasteState) != xsd:boolean ||
                !sameTerm(?foodWasteState, false)
            )
        }
    )
}
"""
    ] .
<END_SHACL>
