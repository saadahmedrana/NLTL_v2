<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:NonMagneticHeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to a main power source."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to an emergency power source."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must identify an independent non-magnetic heading means."
    ] .

shape:ShipNonMagneticHeadingMeansShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node shape:NonMagneticHeadingMeansShape ;
        sh:minCount 2 ;
        sh:message "The ship must have at least two distinct non-magnetic heading means."
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means must be independent from another distinct means owned by the same ship." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT $this ?means
WHERE {
    $this nltl:hasNonMagneticHeadingMeans ?means .
    FILTER NOT EXISTS {
        $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
        ?means nltl:independentFromHeadingMeans ?otherMeans .
        FILTER (?otherMeans != ?means)
    }
}
"""
    ] .
<END_SHACL>
