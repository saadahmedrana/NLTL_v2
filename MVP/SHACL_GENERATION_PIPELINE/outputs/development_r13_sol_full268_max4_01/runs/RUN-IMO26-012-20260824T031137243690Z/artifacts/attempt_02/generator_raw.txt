<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipHeadingMeansShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gen:ShipBelow500GrossTonnageShape
        gen:ShipAtLeast500GrossTonnageShape
    ) .

gen:ShipBelow500GrossTonnageShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:GrossTonnageBelow500ValueShape ;
        sh:message "Gross tonnage must be provided to determine applicability."
    ] .

gen:GrossTonnageBelow500ValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:maxExclusive "500"^^xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ShipAtLeast500GrossTonnageShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:grossTonnage ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node gen:GrossTonnageAtLeast500ValueShape ;
        sh:message "Gross tonnage must be provided to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:node gen:NonMagneticHeadingMeansShape ;
        sh:minCount 2 ;
        sh:message "A ship of at least 500 gross tonnage must have at least two non-magnetic heading means."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each non-magnetic heading means must be independent from another distinct means owned by the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?means
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?otherMeans .
                    $this nltl:hasNonMagneticHeadingMeans ?otherMeans .
                    FILTER (?otherMeans != ?means)
                }
            }
        """
    ] .

gen:GrossTonnageAtLeast500ValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "500"^^xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:NonMagneticHeadingMeansShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to main power."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:benchmarkEntity ;
        sh:minCount 1 ;
        sh:message "Each non-magnetic heading means must be connected to emergency power."
    ] ;
    sh:property [
        sh:path nltl:independentFromHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans
    ] .
<END_SHACL>
