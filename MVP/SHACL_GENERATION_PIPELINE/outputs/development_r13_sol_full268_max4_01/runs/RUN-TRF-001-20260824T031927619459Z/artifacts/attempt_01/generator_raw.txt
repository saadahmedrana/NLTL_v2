<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF001ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "Each stated construction contract date must be an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021 ;
        sh:message "The 2021 section 1.8 Ice Classes provision must apply irrespective of year of build."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021 ;
        sh:message "The 2021 chapter 2 Ice Class Draught provision must apply irrespective of year of build."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship contracted for construction on or after 5 July 2021 must identify the 2021 Ice Class Regulations as applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER (
                    YEAR(?date) > 2021 ||
                    (YEAR(?date) = 2021 && MONTH(?date) > 7) ||
                    (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) >= 5)
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2021 .
                }
            }
        """
    ] .
<END_SHACL>
