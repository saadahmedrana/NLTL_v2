<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:TRF003Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "A represented construction contract date must be an xsd:date."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship contracted from 1 January 2012 through 31 December 2018 must identify the 2010 Ice Class Regulations as applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate .
                FILTER (
                    YEAR(?contractDate) >= 2012 &&
                    YEAR(?contractDate) < 2019
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2010 .
                }
            }
        """
    ] .
<END_SHACL>
