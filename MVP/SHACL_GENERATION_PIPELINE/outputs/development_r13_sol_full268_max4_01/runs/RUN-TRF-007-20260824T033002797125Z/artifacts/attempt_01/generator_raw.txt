<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF007ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "The ship must identify its ice class so that applicability can be determined."
    ] ;
    sh:or (
        gen:TRF007OtherIceClassBranch
        gen:TRF007LaterConstructionBranch
        gen:TRF007CompliantMethodBranch
        gen:TRF007BeforeDeadlineBranch
    ) ;
    sh:message "A ship of ice class IA Super or IA constructed before 1 September 2003 must use an accepted 2017 compliance method once its compliance deadline has been reached." .

gen:TRF007OtherIceClassBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        )
    ] .

gen:TRF007LaterConstructionBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:qualifiedValueShape [
            sh:in (
                nltl:iceClassIaSuper
                nltl:iceClassIa
            )
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] .

gen:TRF007CompliantMethodBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:qualifiedValueShape [
            sh:in (
                nltl:iceClassIaSuper
                nltl:iceClassIa
            )
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:qualifiedValueShape [
            sh:datatype xsd:date ;
            sh:maxExclusive "2003-09-01"^^xsd:date
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:qualifiedValueShape [
            sh:in (
                nltl:traficom2017Section3Point2Point2Method
                nltl:traficom2017Section3Point2Point4Method
            )
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "Compliance must use the method in section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations."
    ] .

gen:TRF007BeforeDeadlineBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:qualifiedValueShape [
            sh:in (
                nltl:iceClassIaSuper
                nltl:iceClassIa
            )
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:qualifiedValueShape [
            sh:datatype xsd:date ;
            sh:maxExclusive "2003-09-01"^^xsd:date
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "The delivery date is required to determine the compliance deadline."
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "The assessment date is required to determine whether the compliance deadline has been reached."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The compliance deadline has been reached: an accepted 2017 compliance method is required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:assessmentDate ?assessmentDate ;
          nltl:deliveryDate ?deliveryDate .
    FILTER (YEAR(?assessmentDate) >= (YEAR(?deliveryDate) + 20))
}
"""
    ] .
<END_SHACL>
