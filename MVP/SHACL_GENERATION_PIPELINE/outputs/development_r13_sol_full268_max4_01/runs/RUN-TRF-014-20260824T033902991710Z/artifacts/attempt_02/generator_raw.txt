gen:CompleteMarkingEvidenceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node gen:LengthQuantityValueShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice-class draught mark amidships must indicate the maximum permissible ice-class draught amidships stated in the class certificate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER NOT EXISTS {
        $this nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity .
        ?markQuantity qudt:numericValue ?markValue .
        $this nltl:hasClassCertificate ?certificate .
        ?certificate nltl:maximumIceClassDraughtAmidships ?maximumQuantity .
        ?maximumQuantity qudt:numericValue ?maximumValue .
        FILTER(xsd:decimal(?markValue) = xsd:decimal(?maximumValue))
    }
}
"""
    ] .
