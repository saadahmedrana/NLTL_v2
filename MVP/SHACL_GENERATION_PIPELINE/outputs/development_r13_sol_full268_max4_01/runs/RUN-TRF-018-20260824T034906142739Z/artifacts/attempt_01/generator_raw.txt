<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF018RequiredEngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Applicable ships must provide exactly one calculated required power and one maximum continuous rating power, each as a decimal quantity in kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc,
                    nltl:iceClassIaSuper
                ))
                FILTER (
                    NOT EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        ?calculatedQuantity
                            qudt:numericValue ?calculatedValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        FILTER (DATATYPE(?calculatedValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity1,
                                                               ?calculatedQuantity2 .
                        FILTER (!sameTerm(?calculatedQuantity1, ?calculatedQuantity2))
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        FILTER NOT EXISTS {
                            ?calculatedQuantity qudt:numericValue ?calculatedValue .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        ?calculatedQuantity qudt:numericValue ?calculatedValue .
                        FILTER (DATATYPE(?calculatedValue) != xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        ?calculatedQuantity
                            qudt:numericValue ?calculatedValue1,
                                              ?calculatedValue2 .
                        FILTER (!sameTerm(?calculatedValue1, ?calculatedValue2))
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        FILTER NOT EXISTS {
                            ?calculatedQuantity qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:calculatedRequiredPower ?calculatedQuantity .
                        ?calculatedQuantity qudt:unit ?calculatedUnit .
                        FILTER (?calculatedUnit != <http://qudt.org/vocab/unit/KiloW>)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        ?ratingQuantity
                            qudt:numericValue ?ratingValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        FILTER (DATATYPE(?ratingValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity1,
                                                                    ?ratingQuantity2 .
                        FILTER (!sameTerm(?ratingQuantity1, ?ratingQuantity2))
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        FILTER NOT EXISTS {
                            ?ratingQuantity qudt:numericValue ?ratingValue .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        ?ratingQuantity qudt:numericValue ?ratingValue .
                        FILTER (DATATYPE(?ratingValue) != xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        ?ratingQuantity
                            qudt:numericValue ?ratingValue1,
                                              ?ratingValue2 .
                        FILTER (!sameTerm(?ratingValue1, ?ratingValue2))
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        FILTER NOT EXISTS {
                            ?ratingQuantity qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        }
                    }
                    ||
                    EXISTS {
                        $this nltl:maximumContinuousRatingPower ?ratingQuantity .
                        ?ratingQuantity qudt:unit ?ratingUnit .
                        FILTER (?ratingUnit != <http://qudt.org/vocab/unit/KiloW>)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be at least the calculated required power and the applicable minimum of 1000 kW or 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this
                    nltl:iceClass ?iceClass ;
                    nltl:calculatedRequiredPower ?calculatedQuantity ;
                    nltl:maximumContinuousRatingPower ?ratingQuantity .
                ?calculatedQuantity
                    qudt:numericValue ?calculatedValue ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?ratingQuantity
                    qudt:numericValue ?ratingValue ;
                    qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                FILTER (DATATYPE(?calculatedValue) = xsd:decimal)
                FILTER (DATATYPE(?ratingValue) = xsd:decimal)
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc,
                    nltl:iceClassIaSuper
                ))
                BIND (
                    IF (
                        ?iceClass = nltl:iceClassIaSuper,
                        "2800"^^xsd:decimal,
                        "1000"^^xsd:decimal
                    )
                    AS ?minimumPower
                )
                FILTER (
                    ?ratingValue < ?calculatedValue ||
                    ?ratingValue < ?minimumPower
                )
            }
        """
    ] .
<END_SHACL>
