<BEGIN_SHACL>
@prefix shape: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:TRF018RequiredEngineOutputShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this rdf:type nltl:ship ;
                      nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc,
                    nltl:iceClassIaSuper
                ))
            }
        """
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ships must have exactly one calculated required power in kW." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Applicable ships must have exactly one maximum continuous rating power in kW." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be at least the calculated required power and at least 1000 kW for IA, IB, or IC, or 2800 kW for IA Super." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:calculatedRequiredPower ?calculatedPower ;
                      nltl:maximumContinuousRatingPower ?mcrPower .
                ?calculatedPower qudt:numericValue ?calculatedValue ;
                                 qudt:unit unit:KiloW .
                ?mcrPower qudt:numericValue ?mcrValue ;
                          qudt:unit unit:KiloW .
                FILTER (
                    (?iceClass IN (
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    ) &&
                    (
                        ?mcrValue < ?calculatedValue ||
                        ?mcrValue < "1000"^^xsd:decimal
                    ))
                    ||
                    (?iceClass = nltl:iceClassIaSuper &&
                    (
                        ?mcrValue < ?calculatedValue ||
                        ?mcrValue < "2800"^^xsd:decimal
                    ))
                )
            }
        """
    ] .
<END_SHACL>
