<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DecimalQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KilowattQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MetreQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF020ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction-stage date is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A controlled ice-class value is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:maxExclusive "2003-09-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minInclusive "2003-09-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:minCount 1 ;
                sh:message "An applicable ship requires a lower ice waterline."
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:upperIceWaterlineBreadth ;
                        sh:minCount 1 ;
                        sh:node gen:MetreQuantityValueShape
                    ] ;
                    sh:property [
                        sh:path nltl:upperIceWaterlineLength ;
                        sh:minCount 1 ;
                        sh:node gen:MetreQuantityValueShape
                    ]
                ] ;
                sh:message "An applicable ship requires an upper ice waterline with its breadth and length."
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:minCount 1 ;
                sh:node gen:UnitlessQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:minCount 1 ;
                sh:node gen:DecimalQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:minCount 1 ;
                sh:node gen:DecimalQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:class nltl:propellerPitchControlTypeValue ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:class nltl:propulsionSystemTypeValue ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:minCount 1 ;
                sh:node gen:KilowattQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:minCount 1 ;
                sh:node gen:KilowattQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:minCount 1 ;
                sh:node gen:KilowattQuantityValueShape
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "Maximum continuous rating power must be at least both externally calculated minimum required powers." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:maximumContinuousRatingPower ?mcrQuantity ;
          nltl:minimumRequiredPowerAtLowerIceWaterline ?lowerQuantity ;
          nltl:minimumRequiredPowerAtUpperIceWaterline ?upperQuantity .
    ?mcrQuantity qudt:numericValue ?mcr .
    ?lowerQuantity qudt:numericValue ?lowerMinimum .
    ?upperQuantity qudt:numericValue ?upperMinimum .
    FILTER(
        xsd:decimal(?mcr) < xsd:decimal(?lowerMinimum) ||
        xsd:decimal(?mcr) < xsd:decimal(?upperMinimum)
    )
}
"""
            ]
        ]
    ) .

<END_SHACL>
