@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF024Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:nodeKind sh:IRI ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                )
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:constructionStageDate ;
                        sh:minCount 1 ;
                        sh:datatype xsd:date ;
                        sh:minInclusive "2003-09-01"^^xsd:date
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:constructionStageDate ;
                        sh:minCount 1 ;
                        sh:datatype xsd:date ;
                        sh:maxExclusive "2003-09-01"^^xsd:date
                    ] ;
                    sh:property [
                        sh:path nltl:deliveryDate ;
                        sh:minCount 1 ;
                        sh:datatype xsd:date
                    ] ;
                    sh:property [
                        sh:path nltl:assessmentDate ;
                        sh:minCount 1 ;
                        sh:datatype xsd:date
                    ]
                ]
            )
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "By the applicable deadline, an eligible IA or IA Super ship must use the section 3.2.2 compliance method." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate ;
          nltl:deliveryDate ?deliveryDate ;
          nltl:assessmentDate ?assessmentDate .

    FILTER(?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))

    FILTER(
        YEAR(?constructionDate) < 2003 ||
        (YEAR(?constructionDate) = 2003 && MONTH(?constructionDate) < 9)
    )

    BIND(YEAR(?deliveryDate) + 20 AS ?deadlineYear)

    FILTER(
        YEAR(?assessmentDate) > ?deadlineYear ||
        (
            YEAR(?assessmentDate) = ?deadlineYear &&
            (
                MONTH(?assessmentDate) > 1 ||
                (MONTH(?assessmentDate) = 1 && DAY(?assessmentDate) >= 1)
            )
        )
    )

    FILTER NOT EXISTS {
        $this nltl:engineOutputComplianceMethod
              nltl:traficom2017Section3Point2Point2Method .
    }
}
"""
    ] .
