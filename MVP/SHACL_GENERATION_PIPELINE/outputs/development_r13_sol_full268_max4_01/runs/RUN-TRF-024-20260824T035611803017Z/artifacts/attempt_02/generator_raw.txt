    }
}
"""

        a sh:SPARQLConstraint ;
        sh:message "The construction stage date of an IA or IA Super ship must be an xsd:date." ;

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this

    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate .
    FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))

}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable IA or IA Super ship must provide its delivery date." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:constructionStageDate ?constructionDate .
    FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
    FILTER (datatype(?constructionDate) = xsd:date)
    FILTER (
        YEAR(?constructionDate) < 2003 ||
        (YEAR(?constructionDate) = 2003 && MONTH(?constructionDate) < 9)
    )
    FILTER NOT EXISTS {
        $this nltl:deliveryDate ?deliveryDate .
    }
}
"""

SELECT $this

          nltl:deliveryDate ?deliveryDate .
    FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
    FILTER (datatype(?constructionDate) = xsd:date)
    FILTER (
        YEAR(?constructionDate) < 2003 ||
        (YEAR(?constructionDate) = 2003 && MONTH(?constructionDate) < 9)

"""

}

        a sh:SPARQLConstraint ;

    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "By 1 January of the twentieth year after the delivery year, the ship must use the section 3.2.2 engine-output compliance method." ;
        sh:select """

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
