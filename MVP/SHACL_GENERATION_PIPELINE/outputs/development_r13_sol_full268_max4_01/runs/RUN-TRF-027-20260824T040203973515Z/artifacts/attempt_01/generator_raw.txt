    ] ;

gen:TRF027ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassDraught ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipBreadth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:clampedHullGeometryTerm ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientC3ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientC4ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientC5 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientC5ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientF1ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientF2ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientF3ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientF4 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientF4ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientG1ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientG2ValueShape
    ] ;
    sh:property [
        sh:path nltl:coefficientG3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CoefficientG3ValueShape
    ] ;
    sh:sparql [
        sh:message "The clamped hull geometry term must equal ((L*T)/B^2)^3 clamped to the inclusive range from 5 to 20." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthQuantity ;
                      nltl:iceClassDraught ?draughtQuantity ;
                      nltl:shipBreadth ?breadthQuantity ;
                      nltl:clampedHullGeometryTerm ?resultQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                ?draughtQuantity qudt:numericValue ?draught .
                ?breadthQuantity qudt:numericValue ?breadth .
                ?resultQuantity qudt:numericValue ?reported .
                BIND((?breadth * ?breadth) AS ?breadthSquared)
                BIND(IF(?breadthSquared = 0, 0, (?length * ?draught) / ?breadthSquared) AS ?ratio)
                BIND((?ratio * ?ratio * ?ratio) AS ?raw)
                BIND(IF(?raw < 5, 5, IF(?raw > 20, 20, ?raw)) AS ?expected)
                BIND(ABS(?expected) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance > 0.000000000001,
                        ?relativeTolerance,
                        0.000000000001) AS ?tolerance)
                FILTER(
                    ?breadthSquared = 0 ||
                    ABS(?reported - ?expected) > ?tolerance
                )
            }
        """
    ] .
