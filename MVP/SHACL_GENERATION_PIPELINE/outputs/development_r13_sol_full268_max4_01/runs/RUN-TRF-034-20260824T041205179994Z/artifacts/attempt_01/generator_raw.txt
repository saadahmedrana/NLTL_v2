gs:EffectiveMemberCrossSectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:effectiveMemberCrossSection ;
    sh:nodeKind sh:IRI ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:node gs:SectionModulusQuantityShape ;
        sh:message "Each effective member cross section requires a section modulus input."
    ] ;
    sh:property [
        sh:path nltl:shearArea ;
        sh:minCount 1 ;
        sh:node gs:ShearAreaQuantityShape ;
        sh:message "Each effective member cross section requires a shear area input."
    ] ;
    sh:property [
        sh:path [ sh:inversePath nltl:hasEffectiveMemberCrossSection ] ;
        sh:class nltl:hullStructure ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:node gs:EffectiveCrossSectionOwnerShape ;
        sh:message "Each effective member cross section must be linked from a hull structure with member-orientation information."
    ] .

gs:SectionModulusQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/CentiM3> ;
        sh:maxCount 1 ;
        sh:message "Section modulus must use unit centimetre cubed."
    ] .

gs:ShearAreaQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M2> ;
        sh:maxCount 1 ;
        sh:message "Shear area must use unit square metre."
    ] .

gs:EffectiveCrossSectionOwnerShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:memberNormalToPlating ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:memberNormalToPlating ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasEffectiveMemberCrossSection ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:effectiveMemberCrossSection ;
                    sh:nodeKind sh:IRI ;
                    sh:property [
                        sh:path nltl:hasClassificationSocietySectionPropertyCalculationEvidence ;
                        sh:class nltl:evidenceArtifact ;
                        sh:nodeKind sh:IRI ;
                        sh:minCount 1 ;
                        sh:message "A non-normal member requires classification-society section-property calculation evidence."
                    ]
                ]
            ]
        ]
    ) .
