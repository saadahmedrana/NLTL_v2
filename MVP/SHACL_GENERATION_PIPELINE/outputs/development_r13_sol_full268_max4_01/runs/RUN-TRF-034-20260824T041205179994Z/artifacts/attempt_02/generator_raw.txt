            ]

        sh:message "The effective member cross section requires a section modulus in cubic centimetres."
    ] ;
    sh:property [

        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [

PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    ?hull nltl:hasEffectiveMemberCrossSection $this ;
          nltl:memberNormalToPlating ?normal .
    FILTER(?normal = "false"^^xsd:boolean)

        FILTER(isIRI(?evidence))
    }
}
"""
