<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF035ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF035LengthQuantityShape ;
        sh:message "The ship shall have exactly one rule length expressed in metres."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:TRF035LengthQuantityShape ;
        sh:message "The classification society rule length shall be provided exactly once in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length shall equal the rule length used by the classification society." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ruleLength ?ruleLengthQuantity ;
                      nltl:classificationSocietyRuleLength ?societyLengthQuantity .
                ?ruleLengthQuantity qudt:numericValue ?ruleLength .
                ?societyLengthQuantity qudt:numericValue ?societyLength .
                FILTER (?ruleLength != ?societyLength)
            }
        """
    ] .

gs:TRF035LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The length quantity shall have exactly one decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:message "The length quantity shall use metres."
    ] .
<END_SHACL>
