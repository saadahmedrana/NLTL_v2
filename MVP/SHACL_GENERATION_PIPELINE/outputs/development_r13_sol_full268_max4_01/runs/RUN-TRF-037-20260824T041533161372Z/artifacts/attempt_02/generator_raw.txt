<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF037ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientA ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Coefficient a is required as one unitless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientK ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Coefficient k is required as one unitless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientB ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Coefficient b is required as one unitless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:iceClassFactorCp ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Ice class factor cp is required as one unitless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:nominalIcePressureP0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037PressureQuantityShape ;
        sh:message "Nominal ice pressure p0 is required as one decimal quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputFactorCd ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Ship size engine output factor cd is required as one unitless decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037PressureQuantityShape ;
        sh:message "Design ice pressure is required as one decimal quantity in MPa."
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:minCount 1 ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:TRF037DirectAnalysisCaseShape ;
        sh:message "At least one direct analysis case with ice load area factor ca is required."
    ] ;
    sh:sparql [
        sh:message "The reported cd must equal min((a*k+b)/1000, 1), and each case pressure must equal reported cd*cp*ca*p0 within the declared numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:shipSizeEngineOutputCoefficientA/qudt:numericValue ?a ;
                      nltl:shipSizeEngineOutputCoefficientK/qudt:numericValue ?k ;
                      nltl:shipSizeEngineOutputCoefficientB/qudt:numericValue ?b ;
                      nltl:iceClassFactorCp/qudt:numericValue ?cp ;
                      nltl:nominalIcePressureP0/qudt:numericValue ?p0 ;
                      nltl:shipSizeEngineOutputFactorCd/qudt:numericValue ?reportedCd ;
                      nltl:icePressure/qudt:numericValue ?reportedPressure ;
                      nltl:hasDirectAnalysisCase ?case .
                ?case nltl:iceLoadAreaFactorCa/qudt:numericValue ?ca .
                BIND((?a * ?k + ?b) / 1000.0 AS ?rawCd)
                BIND(IF(?rawCd > 1.0, 1.0, ?rawCd) AS ?expectedCd)
                BIND(IF(ABS(?expectedCd) * 0.000001 > 0.000000001,
                        ABS(?expectedCd) * 0.000001,
                        0.000000001) AS ?cdTolerance)
                BIND(?reportedCd * ?cp * ?ca * ?p0 AS ?expectedPressure)
                BIND(IF(ABS(?expectedPressure) * 0.000001 > 0.000000001,
                        ABS(?expectedPressure) * 0.000001,
                        0.000000001) AS ?pressureTolerance)
                FILTER(
                    ABS(?reportedCd - ?expectedCd) > ?cdTolerance ||
                    ABS(?reportedPressure - ?expectedPressure) > ?pressureTolerance
                )
            }
        """
    ] .

gen:TRF037DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF037UnitlessQuantityShape ;
        sh:message "Each direct analysis case requires one unitless decimal ice load area factor ca."
    ] .

gen:TRF037UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF037PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
