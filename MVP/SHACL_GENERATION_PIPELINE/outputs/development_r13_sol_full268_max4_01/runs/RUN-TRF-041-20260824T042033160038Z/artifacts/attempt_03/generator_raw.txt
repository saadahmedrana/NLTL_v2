<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF041PlatingReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:platingFramingOrientation ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:transverseFramingOrientation
            nltl:longitudinalFramingOrientation
        ) ;
        sh:message "Exactly one supported plating framing orientation is required."
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:node gen:MetreQuantityShape ;
        sh:message "Plating spacing in metres is required."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gen:MegapascalQuantityShape ;
        sh:message "Plating yield strength in megapascals is required."
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "The corrosion and abrasion addition in millimetres is required."
    ] ;
    sh:property [
        sh:path nltl:requiredShellPlatingThickness ;
        sh:minCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "The externally determined required shell plating thickness in millimetres is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:platingFramingOrientation ;
                sh:hasValue nltl:transverseFramingOrientation
            ] ;
            sh:property [
                sh:path nltl:platingPressure ;
                sh:minCount 1 ;
                sh:node gen:MegapascalQuantityShape ;
                sh:message "Transversely framed plating requires plating pressure in megapascals."
            ] ;
            sh:property [
                sh:path nltl:transversePlatingFactorF1 ;
                sh:minCount 1 ;
                sh:node gen:UnitlessQuantityShape ;
                sh:message "Transversely framed plating requires transverse plating factor f1."
            ]
        ]
        [
            sh:property [
                sh:path nltl:platingFramingOrientation ;
                sh:hasValue nltl:longitudinalFramingOrientation
            ] ;
            sh:property [
                sh:path nltl:icePressure ;
                sh:minCount 1 ;
                sh:node gen:MegapascalQuantityShape ;
                sh:message "Longitudinally framed plating requires ice pressure in megapascals."
            ] ;
            sh:property [
                sh:path nltl:longitudinalPlatingFactorF2 ;
                sh:minCount 1 ;
                sh:node gen:UnitlessQuantityShape ;
                sh:message "Longitudinally framed plating requires longitudinal plating factor f2."
            ]
        ]
    ) .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .

gen:MegapascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaPA
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .
<END_SHACL>
