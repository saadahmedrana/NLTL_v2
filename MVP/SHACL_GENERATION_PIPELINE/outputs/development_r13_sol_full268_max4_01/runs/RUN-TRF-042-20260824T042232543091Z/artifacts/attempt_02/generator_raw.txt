    sh:sparql [
        sh:message "The reported f2 must satisfy the applicable piecewise formula for h/s within the defined range, using the required numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hQuantity ;
                      nltl:spacing ?sQuantity ;
                      nltl:longitudinalPlatingFactorF2 ?f2Quantity .
                ?hQuantity qudt:numericValue ?h .
                ?sQuantity qudt:numericValue ?s .
                ?f2Quantity qudt:numericValue ?reported .
                BIND(
                    IF(
                        ?s = "0"^^xsd:decimal,
                        "0"^^xsd:decimal,
                        ?h / ?s
                    ) AS ?ratio
                )
                BIND(
                    IF(
                        ?ratio <= "1"^^xsd:decimal,
                        ("0.4"^^xsd:decimal / ?ratio) + "0.6"^^xsd:decimal,
                        "1.4"^^xsd:decimal - ("0.4"^^xsd:decimal * ?ratio)
                    ) AS ?expected
                )
                BIND(
                    IF(
                        ("0.000001"^^xsd:decimal * ABS(?expected)) >
                        "0.000000000001"^^xsd:decimal,
                        "0.000001"^^xsd:decimal * ABS(?expected),
                        "0.000000000001"^^xsd:decimal
                    ) AS ?tolerance
                )
                FILTER(
                    ?s = "0"^^xsd:decimal ||
                    ?ratio > "1.8"^^xsd:decimal ||
                    ABS(?reported - ?expected) > ?tolerance
                )
            }
        """
    ] ;
