<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF043ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFramingIceStrengtheningTerminationCase ;
        sh:class nltl:framingIceStrengtheningTerminationCase ;
        sh:node gen:TRF043TerminationCaseShape
    ] ;
    sh:sparql [
        sh:message "Each linked frame must provide valid ice-class, hull-region, and applicable boolean branch evidence." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame .
    FILTER (
        NOT EXISTS {
            ?frame nltl:iceClass ?iceClass .
            FILTER (?iceClass IN (
                nltl:iceClassIaSuper,
                nltl:iceClassIa,
                nltl:iceClassIb,
                nltl:iceClassIc,
                nltl:iceClassIi,
                nltl:iceClassIii
            ))
        }
        ||
        EXISTS {
            ?frame nltl:iceClass ?iceClass .
            FILTER (?iceClass NOT IN (
                nltl:iceClassIaSuper,
                nltl:iceClassIa,
                nltl:iceClassIb,
                nltl:iceClassIc,
                nltl:iceClassIi,
                nltl:iceClassIii
            ))
        }
        ||
        NOT EXISTS {
            ?frame nltl:hullRegion ?region .
            FILTER (?region IN (
                nltl:bowRegion,
                nltl:midbodyRegion,
                nltl:sternRegion
            ))
        }
        ||
        EXISTS {
            ?frame nltl:hullRegion ?region .
            FILTER (?region NOT IN (
                nltl:bowRegion,
                nltl:midbodyRegion,
                nltl:sternRegion
            ))
        }
        ||
        NOT EXISTS {
            ?frame nltl:upperBowIceBeltRequired ?upperRequired .
            FILTER (DATATYPE(?upperRequired) = xsd:boolean)
        }
        ||
        EXISTS {
            ?frame nltl:upperBowIceBeltRequired ?upperRequired .
            FILTER (DATATYPE(?upperRequired) != xsd:boolean)
        }
        ||
        EXISTS {
            ?frame nltl:reachesUpperBowIceBeltTop ?reachesUpper .
            FILTER (DATATYPE(?reachesUpper) != xsd:boolean)
        }
        ||
        EXISTS {
            ?frame nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .
            FILTER (DATATYPE(?reachesTankTop) != xsd:boolean)
        }
        ||
        (
            EXISTS {
                ?frame nltl:upperBowIceBeltRequired ?upperRequired .
                FILTER (?upperRequired = "true"^^xsd:boolean)
            }
            &&
            NOT EXISTS {
                ?frame nltl:reachesUpperBowIceBeltTop ?reachesUpper .
                FILTER (?reachesUpper = "true"^^xsd:boolean)
            }
        )
        ||
        (
            EXISTS {
                ?frame nltl:iceClass nltl:iceClassIaSuper ;
                       nltl:hullRegion nltl:bowRegion .
            }
            &&
            NOT EXISTS {
                ?frame nltl:reachesTankTopOrBelowFloorTop ?reachesTankTop .
                FILTER (?reachesTankTop = "true"^^xsd:boolean)
            }
        )
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "Each represented framing-strengthening extent must use decimal metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame ;
           ?extentProperty ?quantity .
    FILTER (?extentProperty IN (
        nltl:framingIceStrengtheningAboveUpperIceWaterline,
        nltl:framingIceStrengtheningBelowLowerIceWaterline
    ))
    FILTER (
        NOT EXISTS {
            ?quantity qudt:unit unit:M .
        }
        ||
        EXISTS {
            ?quantity qudt:unit ?otherUnit .
            FILTER (?otherUnit != unit:M)
        }
        ||
        NOT EXISTS {
            ?quantity qudt:numericValue ?value .
            FILTER (DATATYPE(?value) = xsd:decimal)
        }
        ||
        EXISTS {
            ?quantity qudt:numericValue ?value .
            FILTER (DATATYPE(?value) != xsd:decimal)
        }
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "Frames of ice classes IA Super, IA, IB, and IC must provide an above-UIWL strengthening extent in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame ;
           nltl:iceClass ?iceClass .
    FILTER (?iceClass IN (
        nltl:iceClassIaSuper,
        nltl:iceClassIa,
        nltl:iceClassIb,
        nltl:iceClassIc
    ))
    FILTER NOT EXISTS {
        ?frame nltl:framingIceStrengtheningAboveUpperIceWaterline ?quantity .
        ?quantity qudt:numericValue ?value ;
                  qudt:unit unit:M .
        FILTER (DATATYPE(?value) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "The framing-strengthening extent above the upper ice waterline is below the applicable Table 4-6 minimum." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame ;
           nltl:iceClass ?iceClass ;
           nltl:framingIceStrengtheningAboveUpperIceWaterline ?quantity .
    ?quantity qudt:numericValue ?value ;
              qudt:unit unit:M .
    FILTER (DATATYPE(?value) = xsd:decimal)
    FILTER (
        (?iceClass = nltl:iceClassIaSuper && ?value < "1.2"^^xsd:decimal)
        ||
        (?iceClass IN (
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        ) && ?value < "1.0"^^xsd:decimal)
    )
}
"""
    ] ;
    sh:sparql [
        sh:message "The applicable Table 4-6 branch requires a below-LIWL strengthening extent in metres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame ;
           nltl:iceClass ?iceClass ;
           nltl:hullRegion ?region .
    FILTER (
        (?iceClass = nltl:iceClassIaSuper &&
         ?region IN (nltl:midbodyRegion, nltl:sternRegion))
        ||
        (?iceClass IN (
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        ) &&
         ?region IN (
            nltl:bowRegion,
            nltl:midbodyRegion,
            nltl:sternRegion
        ))
    )
    FILTER NOT EXISTS {
        ?frame nltl:framingIceStrengtheningBelowLowerIceWaterline ?quantity .
        ?quantity qudt:numericValue ?value ;
                  qudt:unit unit:M .
        FILTER (DATATYPE(?value) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "The framing-strengthening extent below the lower ice waterline is below the applicable Table 4-6 minimum." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasComponent ?frame .
    ?frame a nltl:frame ;
           nltl:iceClass ?iceClass ;
           nltl:hullRegion ?region ;
           nltl:framingIceStrengtheningBelowLowerIceWaterline ?quantity .
    ?quantity qudt:numericValue ?value ;
              qudt:unit unit:M .
    FILTER (DATATYPE(?value) = xsd:decimal)
    FILTER (
        (?iceClass = nltl:iceClassIaSuper &&
         ?region = nltl:midbodyRegion &&
         ?value < "2.0"^^xsd:decimal)
        ||
        (?iceClass = nltl:iceClassIaSuper &&
         ?region = nltl:sternRegion &&
         ?value < "1.6"^^xsd:decimal)
        ||
        (?iceClass IN (
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        ) &&
         ?region = nltl:bowRegion &&
         ?value < "1.6"^^xsd:decimal)
        ||
        (?iceClass IN (
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        ) &&
         ?region = nltl:midbodyRegion &&
         ?value < "1.3"^^xsd:decimal)
        ||
        (?iceClass IN (
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        ) &&
         ?region = nltl:sternRegion &&
         ?value < "1.0"^^xsd:decimal)
    )
}
"""
    ] .

gen:TRF043TerminationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:minCount 1 ;
        sh:node gen:TRF043MillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:terminationAdjacentBoundary ;
        sh:minCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
        )
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Termination at the adjacent boundary must be recorded as permitted when the extension beyond it is no more than 250 mm." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity .
    ?quantity qudt:numericValue ?extension ;
              qudt:unit unit:MilliM .
    FILTER (DATATYPE(?extension) = xsd:decimal)
    FILTER (?extension <= "250"^^xsd:decimal)
    FILTER NOT EXISTS {
        $this nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ?permitted .
        FILTER (?permitted = "true"^^xsd:boolean)
    }
}
"""
    ] .

gen:TRF043MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
