gen:TRF044TransverseFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameBoundaryConditionFactorM0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorMt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CubicCentimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareCentimetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "Frame moment factor, section modulus, and required shear area must satisfy the prescribed formulas within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:spacing ?spacingQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:span ?spanQuantity ;
                      nltl:frameBoundaryConditionFactorM0 ?m0Quantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:frameMomentFactorMt ?mtQuantity ;
                      nltl:sectionModulus ?sectionModulusQuantity ;
                      nltl:requiredShearArea ?shearAreaQuantity .

                ?pressureQuantity qudt:numericValue ?pressure .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?heightQuantity qudt:numericValue ?height .
                ?spanQuantity qudt:numericValue ?span .
                ?m0Quantity qudt:numericValue ?m0 .
                ?yieldQuantity qudt:numericValue ?yieldStrength .
                ?mtQuantity qudt:numericValue ?reportedMt .
                ?sectionModulusQuantity qudt:numericValue ?reportedSectionModulus .
                ?shearAreaQuantity qudt:numericValue ?reportedShearArea .

                BIND((7 * ?m0) / (7 - (5 * ?height / ?span)) AS ?expectedMt)
                BIND(
                    (?pressure * ?spacing * ?height * ?span * 1000000) /
                    (?expectedMt * ?yieldStrength)
                    AS ?expectedSectionModulus
                )
                BIND(
                    (1.7320508075688772935 * 1.2 * ?pressure * ?height * ?spacing * 10000) /
                    (2 * ?yieldStrength)
                    AS ?expectedShearArea
                )

                BIND(
                    IF(
                        ABS(?expectedMt) * 0.000001 > 0.000000001,
                        ABS(?expectedMt) * 0.000001,
                        0.000000001
                    )
                    AS ?mtTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedSectionModulus) * 0.000001 > 0.000000001,
                        ABS(?expectedSectionModulus) * 0.000001,
                        0.000000001
                    )
                    AS ?sectionModulusTolerance
                )
                BIND(
                    IF(
                        ABS(?expectedShearArea) * 0.000001 > 0.000000001,
                        ABS(?expectedShearArea) * 0.000001,
                        0.000000001
                    )
                    AS ?shearAreaTolerance
                )

                FILTER(
                    !BOUND(?expectedMt) ||
                    !BOUND(?expectedSectionModulus) ||
                    !BOUND(?expectedShearArea) ||
                    ABS(?reportedMt - ?expectedMt) > ?mtTolerance ||
                    ABS(?reportedSectionModulus - ?expectedSectionModulus) > ?sectionModulusTolerance ||
                    ABS(?reportedShearArea - ?expectedShearArea) > ?shearAreaTolerance
                )
            }
        """
    ] .
<END_SHACL>
