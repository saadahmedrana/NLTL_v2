<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix shape: <urn:nltl:generated-shape:> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shape:TRF045FrameScantlingsPermissionShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary frame scantlings may be used only when the frame span within the ice-strengthening zone is less than 15% of the total frame span, with both lengths expressed in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed ?used .
                FILTER(?used = true)
                FILTER NOT EXISTS {
                    $this nltl:span ?spanQuantity ;
                          nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
                    ?spanQuantity a qudt:QuantityValue ;
                                  qudt:numericValue ?spanValue ;
                                  qudt:unit unit:M .
                    ?zoneQuantity a qudt:QuantityValue ;
                                  qudt:numericValue ?zoneValue ;
                                  qudt:unit unit:M .
                    FILTER(
                        datatype(?spanValue) = xsd:decimal &&
                        datatype(?zoneValue) = xsd:decimal &&
                        ?zoneValue < (0.15 * ?spanValue)
                    )
                    FILTER NOT EXISTS {
                        ?spanQuantity qudt:unit ?otherSpanUnit .
                        FILTER(?otherSpanUnit != unit:M)
                    }
                    FILTER NOT EXISTS {
                        ?zoneQuantity qudt:unit ?otherZoneUnit .
                        FILTER(?otherZoneUnit != unit:M)
                    }
                }
            }
        """
    ] .
<END_SHACL>
