<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF046FrameUpperEndAttachmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The strengthened upper end of each main frame or intermediate ice frame must be attached to a deck structure, tank boundary plating, or ice stringer." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
SELECT DISTINCT $this
WHERE {
    $this a ?frameType .
    FILTER (?frameType IN (nltl:mainFrame, nltl:intermediateIceFrame))
    FILTER NOT EXISTS {
        $this nltl:hasStrengthenedPart ?strengthenedPart ;
              nltl:hasUpperEnd ?upperEnd .
        ?strengthenedPart a nltl:frameStrengthenedPart .
        ?upperEnd a nltl:frameEnd ;
                  nltl:hasAttachedSupportingStructure ?supportingStructure .
        ?supportingStructure a ?supportingStructureType .
        FILTER (?supportingStructureType IN (
            nltl:deckStructure,
            nltl:tankBoundaryPlating,
            nltl:iceStringer
        ))
    }
}
"""
    ] .
<END_SHACL>
