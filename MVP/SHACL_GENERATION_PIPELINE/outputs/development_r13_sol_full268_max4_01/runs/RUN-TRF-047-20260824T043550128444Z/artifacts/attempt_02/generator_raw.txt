<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:class nltl:frameStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A main frame must identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:class nltl:frameEnd ;
        sh:minCount 1 ;
        sh:node gen:AttachedLowerEndShape ;
        sh:message "Each lower end of a main frame must be attached to a permitted supporting structure."
    ] ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The main frame below the ice belt must be recorded as strengthened."
    ] .

gen:IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:class nltl:frameStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "An intermediate ice frame must identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasLowerEnd ;
        sh:class nltl:frameEnd ;
        sh:minCount 1 ;
        sh:node gen:IntermediateLowerEndShape ;
        sh:message "Each lower end of an intermediate ice frame must use a permitted attachment or horizontal-connection alternative."
    ] .

gen:IntermediateLowerEndShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:node gen:AttachedLowerEndShape
        ]
        [
            sh:node gen:ExceptionLowerEndShape
        ]
    ) .

gen:AttachedLowerEndShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:qualifiedValueShape gen:PermittedSupportingStructureShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "The lower end must be attached to a deck, tank boundary plating, tank top, or ice stringer."
    ] .

gen:PermittedSupportingStructureShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:class nltl:deckStructure
        ]
        [
            sh:class nltl:tankBoundaryPlating
        ]
        [
            sh:class nltl:tankTop
        ]
        [
            sh:class nltl:iceStringer
        ]
    ) .

gen:ExceptionLowerEndShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasTerminationBelowSupportingStructure ;
        sh:qualifiedValueShape gen:LowerLimitSupportingStructureShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "The horizontal-connection alternative requires termination below a permitted supporting structure at or below the ice-belt lower limit."
    ] ;
    sh:property [
        sh:path nltl:hasHorizontalConnectionMember ;
        sh:qualifiedValueShape gen:HorizontalConnectionMemberShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "The lower end must have a conforming horizontal connection member."
    ] .

gen:LowerLimitSupportingStructureShape
    a sh:NodeShape ;
    sh:node gen:PermittedSupportingStructureShape ;
    sh:property [
        sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

gen:HorizontalConnectionMemberShape
    a sh:NodeShape ;
    sh:class nltl:horizontalConnectionMember ;
    sh:property [
        sh:path nltl:sameScantlingsAsMainFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The horizontal connection member must have the same scantlings as the main frames."
    ] ;
    sh:property [
        sh:path nltl:connectsToAdjacentMainFrame ;
        sh:class nltl:mainFrame ;
        sh:minCount 2 ;
        sh:node gen:ConnectedMainFrameShape ;
        sh:message "The horizontal connection member must connect to at least two adjacent main frames."
    ] .

gen:ConnectedMainFrameShape
    a sh:NodeShape ;
    sh:class nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each connected main frame below the ice belt must be strengthened."
    ] .
<END_SHACL>
