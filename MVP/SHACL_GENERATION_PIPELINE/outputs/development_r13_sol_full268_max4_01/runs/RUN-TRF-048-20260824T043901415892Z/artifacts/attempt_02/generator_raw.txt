        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [

gen:ContinuousBeamMomentFactorBranchShape
    a sh:NodeShape ;
    sh:or (
        [
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path (
                    nltl:frameMomentFactorM
                    qudt:numericValue
                ) ;
                sh:minInclusive "13.3"^^xsd:decimal ;
                sh:maxInclusive "13.3"^^xsd:decimal
            ]
        ]
        [
            sh:property [
                sh:path nltl:significantlyDifferentBoundaryConditions ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path (
                    nltl:frameMomentFactorM
                    qudt:numericValue
                ) ;
                sh:maxExclusive "13.3"^^xsd:decimal
            ]
        ]
    ) .

gen:ContinuousBeamWithBracketsConditionShape
    a sh:NodeShape ;
    sh:targetNode nltl:continuousBeamWithBracketsFrameCondition ;
    sh:property [
        sh:path [
            sh:inversePath nltl:frameBoundaryConditionType
        ] ;
        sh:node gen:ContinuousBeamMomentFactorBranchShape ;
        sh:message "A continuous beam with brackets must use m = 13.3 unless significantly different boundary conditions require a smaller factor."
    ] .

gen:LongitudinalFrameCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;

    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:class nltl:frameBoundaryConditionTypeValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:significantlyDifferentBoundaryConditions ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:icePressure ;
        sh:node gen:DecimalStressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:node gen:DecimalLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gen:DecimalLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gen:DecimalLengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:node gen:DecimalUnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node gen:DecimalStressQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:longitudinalFrameLoadDistributionFactorF4 ;
        sh:node gen:DecimalUnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:node gen:DecimalSectionModulusQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:node gen:DecimalAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:grossFrameShearArea ;
        sh:node gen:DecimalAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bracketArea ;
        sh:node gen:DecimalAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:actualFrameShearArea ;
        sh:node gen:DecimalAreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;

    sh:sparql [
        sh:message "Spacing, frame moment factor, and yield strength must be non-zero for the prescribed calculations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:spacing/qudt:numericValue ?spacing ;
                      nltl:frameMomentFactorM/qudt:numericValue ?momentFactor ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength .
                FILTER (
                    ?spacing = "0"^^xsd:decimal ||
                    ?momentFactor = "0"^^xsd:decimal ||
                    ?yieldStrength = "0"^^xsd:decimal
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "The reported f4, section modulus, required shear area, or bracket-excluded actual frame shear area does not match the prescribed calculation within the expected-result tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?pressure ;
                      nltl:designIceLoadHeight/qudt:numericValue ?height ;
                      nltl:spacing/qudt:numericValue ?spacing ;
                      nltl:span/qudt:numericValue ?span ;
                      nltl:frameMomentFactorM/qudt:numericValue ?momentFactor ;
                      nltl:yieldStrength/qudt:numericValue ?yieldStrength ;
                      nltl:longitudinalFrameLoadDistributionFactorF4/qudt:numericValue ?reportedF4 ;
                      nltl:sectionModulus/qudt:numericValue ?reportedSectionModulus ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedRequiredArea ;
                      nltl:grossFrameShearArea/qudt:numericValue ?grossArea ;
                      nltl:bracketArea/qudt:numericValue ?bracketArea ;
                      nltl:actualFrameShearArea/qudt:numericValue ?reportedActualArea .

                FILTER (
                    ?spacing != "0"^^xsd:decimal &&
                    ?momentFactor != "0"^^xsd:decimal &&
                    ?yieldStrength != "0"^^xsd:decimal
                )

                BIND(
                    "1"^^xsd:decimal -
                    ("0.2"^^xsd:decimal * ?height / ?spacing)
                    AS ?expectedF4
                )
                BIND(
                    ?expectedF4 * ?pressure * ?height * ?span * ?span *
                    "1000000"^^xsd:decimal /
                    ("4"^^xsd:decimal * ?momentFactor * ?yieldStrength)
                    AS ?expectedSectionModulus
                )
                BIND(
                    "1.7320508075688772"^^xsd:decimal *
                    ?expectedF4 *
                    "2.16"^^xsd:decimal *
                    ?pressure * ?height * ?span *
                    "10000"^^xsd:decimal /
                    ("2"^^xsd:decimal * ?yieldStrength)
                    AS ?expectedRequiredArea
                )
                BIND(
                    ?grossArea - ?bracketArea
                    AS ?expectedActualArea
                )

                BIND(
                    IF(
                        ?expectedF4 < "0"^^xsd:decimal,
                        -"0.000001"^^xsd:decimal * ?expectedF4,
                        "0.000001"^^xsd:decimal * ?expectedF4
                    )
                    AS ?f4RelativeTolerance
                )
                BIND(
                    IF(
                        ?f4RelativeTolerance < "0.000000000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal,
                        ?f4RelativeTolerance
                    )
                    AS ?f4Tolerance
                )

                BIND(
                    IF(
                        ?expectedSectionModulus < "0"^^xsd:decimal,
                        -"0.000001"^^xsd:decimal * ?expectedSectionModulus,
                        "0.000001"^^xsd:decimal * ?expectedSectionModulus
                    )
                    AS ?sectionRelativeTolerance
                )
                BIND(
                    IF(
                        ?sectionRelativeTolerance < "0.000000000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal,
                        ?sectionRelativeTolerance
                    )
                    AS ?sectionTolerance
                )

                BIND(
                    IF(
                        ?expectedRequiredArea < "0"^^xsd:decimal,
                        -"0.000001"^^xsd:decimal * ?expectedRequiredArea,
                        "0.000001"^^xsd:decimal * ?expectedRequiredArea
                    )
                    AS ?requiredAreaRelativeTolerance
                )
                BIND(
                    IF(
                        ?requiredAreaRelativeTolerance < "0.000000000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal,
                        ?requiredAreaRelativeTolerance
                    )
                    AS ?requiredAreaTolerance
                )

                BIND(
                    IF(
                        ?expectedActualArea < "0"^^xsd:decimal,
                        -"0.000001"^^xsd:decimal * ?expectedActualArea,
                        "0.000001"^^xsd:decimal * ?expectedActualArea
                    )
                    AS ?actualAreaRelativeTolerance
                )
                BIND(
                    IF(
                        ?actualAreaRelativeTolerance < "0.000000000001"^^xsd:decimal,
                        "0.000000000001"^^xsd:decimal,
                        ?actualAreaRelativeTolerance
                    )
                    AS ?actualAreaTolerance
                )

                FILTER (
                    ?reportedF4 < (?expectedF4 - ?f4Tolerance) ||
                    ?reportedF4 > (?expectedF4 + ?f4Tolerance) ||
                    ?reportedSectionModulus < (?expectedSectionModulus - ?sectionTolerance) ||
                    ?reportedSectionModulus > (?expectedSectionModulus + ?sectionTolerance) ||
                    ?reportedRequiredArea < (?expectedRequiredArea - ?requiredAreaTolerance) ||
                    ?reportedRequiredArea > (?expectedRequiredArea + ?requiredAreaTolerance) ||
                    ?reportedActualArea < (?expectedActualArea - ?actualAreaTolerance) ||
                    ?reportedActualArea > (?expectedActualArea + ?actualAreaTolerance)
                )
            }
        """
    ] .
<END_SHACL>
