<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameAttachmentRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must state whether it is within the ice-strengthened area."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node gen:IceStrengthenedFrameShape
        ]
    ) .

gen:IceStrengthenedFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice-strengthened frame must state whether it passes through a supporting structure."
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:maxCount 1 ;
        sh:node gen:MillimetreLengthShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord
    ] ;
    sh:sparql [
        sh:message "Each linked supporting web frame must share an effective attachment with the frame." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:withinIceStrengthenedArea true ;
          nltl:hasSupportingWebFrame ?support .
    FILTER NOT EXISTS {
        ?support a nltl:supportingStructure ;
                 nltl:hasSupportingStructureAttachment ?attachment .
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:effectiveAttachmentConfirmed true .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Each linked supporting bulkhead must share an effective attachment with the frame." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:withinIceStrengthenedArea true ;
          nltl:hasSupportingBulkhead ?support .
    FILTER NOT EXISTS {
        ?support a nltl:supportingStructure ;
                 nltl:hasSupportingStructureAttachment ?attachment .
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:effectiveAttachmentConfirmed true .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A longitudinal frame must be bracketed to every linked supporting web frame." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this a nltl:longitudinalFrame ;
          nltl:withinIceStrengthenedArea true ;
          nltl:hasSupportingWebFrame ?support .
    FILTER NOT EXISTS {
        ?support a nltl:supportingStructure ;
                 nltl:hasSupportingStructureAttachment ?attachment .
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:hasConnectionBracket ?bracket .
        ?bracket a nltl:connectionBracket .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A longitudinal frame must be bracketed to every linked supporting bulkhead." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this a nltl:longitudinalFrame ;
          nltl:withinIceStrengthenedArea true ;
          nltl:hasSupportingBulkhead ?support .
    FILTER NOT EXISTS {
        ?support a nltl:supportingStructure ;
                 nltl:hasSupportingStructureAttachment ?attachment .
        $this nltl:hasFrameAttachment ?attachment .
        ?attachment nltl:hasAttachedSupportingStructure ?support ;
                    nltl:hasConnectionBracket ?bracket .
        ?bracket a nltl:connectionBracket .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "An ice-strengthened transverse frame must state whether it terminates at a deck or ice stringer." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this a nltl:transverseFrame ;
          nltl:withinIceStrengthenedArea true .
    FILTER NOT EXISTS {
        $this nltl:terminatesAtDeckOrIceStringer ?terminates .
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A terminating transverse frame must have a bracket or similar attachment construction." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this a nltl:transverseFrame ;
          nltl:withinIceStrengthenedArea true ;
          nltl:terminatesAtDeckOrIceStringer true .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        {
            ?attachment nltl:hasConnectionBracket ?bracket .
            ?bracket a nltl:connectionBracket .
        }
        UNION
        {
            ?attachment nltl:similarAttachmentConstructionPresent true .
        }
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A frame passing through a supporting structure must have both web-plate sides connected." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:withinIceStrengthenedArea true ;
          nltl:passesThroughSupportingStructure true .
    FILTER NOT EXISTS {
        $this nltl:webPlateConnectionSideCount ?sideCount .
        FILTER(?sideCount >= 2)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "A frame with an installed bracket must provide its web thickness in millimetres." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:withinIceStrengthenedArea true ;
          nltl:hasFrameAttachment ?attachment .
    ?attachment nltl:hasConnectionBracket ?bracket .
    FILTER NOT EXISTS {
        $this nltl:frameWebThickness ?frameThickness .
        ?frameThickness qudt:numericValue ?frameValue ;
                        qudt:unit unit:MilliM .
        FILTER(datatype(?frameValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:withinIceStrengthenedArea true ;
          nltl:frameWebThickness ?frameThickness ;
          nltl:hasFrameAttachment ?attachment .
    ?attachment nltl:hasConnectionBracket ?bracket .
    ?frameThickness qudt:numericValue ?frameValue ;
                    qudt:unit unit:MilliM .
    ?bracket nltl:bracketThickness ?bracketThickness .
    ?bracketThickness qudt:numericValue ?bracketValue ;
                      qudt:unit unit:MilliM .
    FILTER(datatype(?frameValue) = xsd:decimal)
    FILTER(datatype(?bracketValue) = xsd:decimal)
    FILTER(?bracketValue < ?frameValue)
}
"""
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasAttachedSupportingStructure ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasAttachedSupportingStructure ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:effectiveAttachmentConfirmed ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "An attachment to a supporting structure must be confirmed effective."
            ]
        ]
    ) .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MillimetreLengthShape ;
        sh:message "An installed bracket must provide one thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The bracket must be stiffened against buckling."
    ] .

gen:MillimetreLengthShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .
<END_SHACL>
