<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:FrameRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:node gen:ApplicableFrameShape
        ]
    ) .

gen:ApplicableFrameShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure ;
        sh:node gen:SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure ;
        sh:node gen:SupportingStructureShape
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:equals nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord ;
        sh:node gen:FrameAttachmentShape
    ] ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:class nltl:frameAttachmentRecord
    ] ;
    sh:property [
        sh:path (
            nltl:hasFrameAttachment
            nltl:hasConnectionBracket
        ) ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:passesThroughSupportingStructure ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:webPlateConnectionSideCount ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:minInclusive 2 ;
                sh:maxInclusive 2 ;
                sh:message "A frame passing through a supporting structure must have exactly two connected web-plate sides."
            ]
        ]
    ) ;
    sh:or (
        [
            sh:property [
                sh:path (
                    nltl:hasFrameAttachment
                    nltl:hasConnectionBracket
                ) ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path (
                    nltl:hasFrameAttachment
                    nltl:hasConnectionBracket
                ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:frameWebThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gen:MillimetreQuantityShape ;
                sh:message "Frame web thickness is required when a connection bracket is installed."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every linked supporting structure must have an effective reciprocal frame attachment." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?support
WHERE {
    $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
    ?support a nltl:supportingStructure .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        $this nltl:frameAttachment ?attachment .
        ?attachment a nltl:frameAttachmentRecord ;
            nltl:hasAttachedSupportingStructure ?support ;
            nltl:effectiveAttachmentConfirmed true .
        ?support nltl:hasSupportingStructureAttachment ?attachment .
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A connection bracket must be at least as thick as the frame web." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?bracket
WHERE {
    $this nltl:hasFrameAttachment ?attachment ;
        nltl:frameWebThickness ?frameThickness .
    ?attachment nltl:hasConnectionBracket ?bracket .
    ?bracket nltl:bracketThickness ?bracketThickness .
    ?frameThickness qudt:numericValue ?frameValue .
    ?bracketThickness qudt:numericValue ?bracketValue .
    FILTER (xsd:decimal(?bracketValue) < xsd:decimal(?frameValue))
}
"""
    ] .

gen:FrameAttachmentShape
    a sh:NodeShape ;
    sh:class nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket ;
        sh:node gen:ConnectionBracketShape
    ] .

gen:SupportingStructureShape
    a sh:NodeShape ;
    sh:class nltl:supportingStructure ;
    sh:property [
        sh:path nltl:hasSupportingStructureAttachment ;
        sh:class nltl:frameAttachmentRecord
    ] .

gen:ConnectionBracketShape
    a sh:NodeShape ;
    sh:class nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MillimetreQuantityShape ;
        sh:message "An installed connection bracket must have a bracket thickness."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The connection-bracket edge must be stiffened."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The connection bracket must be stiffened against buckling."
    ] .

gen:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .

gen:LongitudinalFrameRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "A longitudinal frame must be effectively attached by a bracket to every linked supporting web frame and bulkhead." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?support
WHERE {
    $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
    ?support a nltl:supportingStructure .
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        $this nltl:frameAttachment ?attachment .
        ?attachment a nltl:frameAttachmentRecord ;
            nltl:hasAttachedSupportingStructure ?support ;
            nltl:effectiveAttachmentConfirmed true ;
            nltl:hasConnectionBracket ?bracket .
        ?bracket a nltl:connectionBracket .
        ?support nltl:hasSupportingStructureAttachment ?attachment .
    }
}
"""
            ]
        ]
    ) .

gen:TransverseFrameRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue false
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:terminatesAtDeckOrIceStringer ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true
                    ] ;
                    sh:sparql [
                        a sh:SPARQLConstraint ;
                        sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar attachment construction." ;
                        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER NOT EXISTS {
        $this nltl:hasFrameAttachment ?attachment .
        $this nltl:frameAttachment ?attachment .
        ?attachment a nltl:frameAttachmentRecord .
        FILTER (
            EXISTS {
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket a nltl:connectionBracket .
            }
            ||
            EXISTS {
                ?attachment nltl:similarAttachmentConstructionPresent true .
            }
        )
    }
}
"""
                    ]
                ]
            )
        ]
    ) .
<END_SHACL>
