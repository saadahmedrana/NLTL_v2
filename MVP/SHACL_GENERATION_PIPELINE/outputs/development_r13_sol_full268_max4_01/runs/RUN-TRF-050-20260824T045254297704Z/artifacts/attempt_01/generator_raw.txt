@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:minCount 1 ;
        sh:hasValue nltl:doubleContinuousWeld ;
        sh:message "The frame attachment shall use a double continuous weld."
    ] ;
    sh:property [
        sh:path nltl:shellPlateButtCrossing ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Shell-plate-butt crossing status is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( true )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( false )
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( false ) ;
                sh:message "Scalloping is not allowed when the attachment does not cross a shell plate butt."
            ]
        ]
    ) .

gs:FrameAttachmentPathShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachmentRecord ;
        sh:node gs:FrameAttachmentRecordShape ;
        sh:message "Each frame shall have a conforming frame attachment record."
    ] .
