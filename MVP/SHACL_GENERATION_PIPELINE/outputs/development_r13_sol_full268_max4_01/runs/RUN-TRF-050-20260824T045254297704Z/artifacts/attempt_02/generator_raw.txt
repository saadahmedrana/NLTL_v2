<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:FrameAttachmentRecordShape
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachmentRecord ;
    sh:property [
        sh:path nltl:frameShellWeldType ;
        sh:minCount 1 ;
        sh:in ( nltl:doubleContinuousWeld ) ;
        sh:message "The frame-to-shell attachment shall use a double continuous weld."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in ( true )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shellPlateButtCrossing ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in ( false )
            ] ;
            sh:property [
                sh:path nltl:scallopingPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( false )
            ]
        ]
    ) ;
    sh:message "Scalloping shall be absent unless the attachment crosses a shell plate butt." .

gs:FrameAttachmentPathShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:frameAttachment ;
        sh:minCount 1 ;
        sh:class nltl:frameAttachmentRecord ;
        sh:node gs:FrameAttachmentRecordShape ;
        sh:message "Each frame shall have a conforming frame attachment record."
    ] .
<END_SHACL>
