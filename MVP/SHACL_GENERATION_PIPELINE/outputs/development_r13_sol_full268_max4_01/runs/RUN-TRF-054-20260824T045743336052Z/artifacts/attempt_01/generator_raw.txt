gen:TRF054IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MegaPascalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:class qudt:QuantityValue ;
        sh:node gen:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [

        sh:class qudt:QuantityValue ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;

        sh:maxCount 1
    ] ;
    sh:property [

        sh:minCount 1 ;
        sh:maxCount 1

    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:class qudt:QuantityValue ;

        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:class qudt:QuantityValue ;
        sh:node gen:SquareCentimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "TRF-054 calculated results must match the prescribed formulas within 1e-6 relative tolerance and 1e-9 absolute tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?pRaw ;
                      nltl:designIceLoadHeight/qudt:numericValue ?hRaw ;
                      nltl:span/qudt:numericValue ?lRaw ;
                      nltl:frameMomentFactorM/qudt:numericValue ?mRaw ;
                      nltl:yieldStrength/qudt:numericValue ?sigmaRaw ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hsRaw ;
                      nltl:stringerSpan/qudt:numericValue ?lsRaw ;
                      nltl:designLineLoad/qudt:numericValue ?lineLoadRaw ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?factorRaw ;
                      nltl:sectionModulus/qudt:numericValue ?sectionRaw ;
                      nltl:requiredShearArea/qudt:numericValue ?areaRaw .

                BIND(xsd:decimal(?pRaw) AS ?p)
                BIND(xsd:decimal(?hRaw) AS ?h)
                BIND(xsd:decimal(?lRaw) AS ?l)
                BIND(xsd:decimal(?mRaw) AS ?m)
                BIND(xsd:decimal(?sigmaRaw) AS ?sigma)
                BIND(xsd:decimal(?hsRaw) AS ?hs)
                BIND(xsd:decimal(?lsRaw) AS ?ls)
                BIND(xsd:decimal(?lineLoadRaw) AS ?lineLoad)
                BIND(xsd:decimal(?factorRaw) AS ?factor)
                BIND(xsd:decimal(?sectionRaw) AS ?section)
                BIND(xsd:decimal(?areaRaw) AS ?area)

                BIND(IF(?m = 0, 1, ?m) AS ?safeM)
                BIND(IF(?sigma = 0, 1, ?sigma) AS ?safeSigma)
                BIND(IF(?ls = 0, 1, ?ls) AS ?safeLs)

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?expectedLineLoadMNPerM)
                BIND((?expectedLineLoadMNPerM * 1000) AS ?expectedLineLoadKNPerM)
                BIND((1 - (?hs / ?safeLs)) AS ?expectedFactor)
                BIND(
                    (0.80 * 1.8 * ?expectedLineLoadMNPerM * ?l * ?l
                     / (?safeM * ?safeSigma) * ?expectedFactor * 1000000)
                    AS ?expectedSection
                )
                BIND(
                    (1.7320508075688772 * 0.80 * 1.8 * 1.2
                     * ?expectedLineLoadMNPerM * ?l
                     / (2 * ?safeSigma) * ?expectedFactor * 10000)
                    AS ?expectedArea
                )

                BIND(IF(ABS(?expectedLineLoadKNPerM) * 0.000001 > 0.000000001,
                        ABS(?expectedLineLoadKNPerM) * 0.000001,
                        0.000000001) AS ?lineTolerance)
                BIND(IF(ABS(?expectedFactor) * 0.000001 > 0.000000001,
                        ABS(?expectedFactor) * 0.000001,
                        0.000000001) AS ?factorTolerance)
                BIND(IF(ABS(?expectedSection) * 0.000001 > 0.000000001,
                        ABS(?expectedSection) * 0.000001,
                        0.000000001) AS ?sectionTolerance)
                BIND(IF(ABS(?expectedArea) * 0.000001 > 0.000000001,
                        ABS(?expectedArea) * 0.000001,
                        0.000000001) AS ?areaTolerance)

                FILTER(
                    ?m = 0 ||
                    ?sigma = 0 ||
                    ?ls = 0 ||
                    ABS(?lineLoad - ?expectedLineLoadKNPerM) > ?lineTolerance ||
                    ABS(?factor - ?expectedFactor) > ?factorTolerance ||
                    ABS(?section - ?expectedSection) > ?sectionTolerance ||
                    ABS(?area - ?expectedArea) > ?areaTolerance
                )
            }
        """
    ] .
