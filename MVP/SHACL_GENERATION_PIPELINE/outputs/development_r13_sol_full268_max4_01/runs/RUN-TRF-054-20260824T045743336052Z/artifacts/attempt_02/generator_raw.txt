gen:TRF054CalculationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPAQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stringerSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:frameMomentFactorM ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPAQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceToIceBelt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:distanceToAdjacentIceStringer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designLineLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:KiloNPerMetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:outsideIceBeltFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CubicCentimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:SquareCentimetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "Applicable outside-ice-belt stringer results must satisfy the design line load, outside factor, section modulus, and shear-area formulas within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure/qudt:numericValue ?pRaw ;
                      nltl:designIceLoadHeight/qudt:numericValue ?hRaw ;
                      nltl:stringerSpan/qudt:numericValue ?lRaw ;
                      nltl:frameMomentFactorM/qudt:numericValue ?mRaw ;
                      nltl:yieldStrength/qudt:numericValue ?sigmaRaw ;
                      nltl:distanceToIceBelt/qudt:numericValue ?hsRaw ;
                      nltl:distanceToAdjacentIceStringer/qudt:numericValue ?lsRaw ;
                      nltl:designLineLoad/qudt:numericValue ?reportedLineRaw ;
                      nltl:outsideIceBeltFactor/qudt:numericValue ?reportedFactorRaw ;
                      nltl:sectionModulus/qudt:numericValue ?reportedZRaw ;
                      nltl:requiredShearArea/qudt:numericValue ?reportedARaw .

                BIND(xsd:decimal(?pRaw) AS ?p)
                BIND(xsd:decimal(?hRaw) AS ?h)
                BIND(xsd:decimal(?lRaw) AS ?l)
                BIND(xsd:decimal(?mRaw) AS ?m)
                BIND(xsd:decimal(?sigmaRaw) AS ?sigma)
                BIND(xsd:decimal(?hsRaw) AS ?hs)
                BIND(xsd:decimal(?lsRaw) AS ?ls)
                BIND(xsd:decimal(?reportedLineRaw) AS ?reportedLine)
                BIND(xsd:decimal(?reportedFactorRaw) AS ?reportedFactor)
                BIND(xsd:decimal(?reportedZRaw) AS ?reportedZ)
                BIND(xsd:decimal(?reportedARaw) AS ?reportedA)

                BIND(IF((?p * ?h) < 0.15, 0.15, (?p * ?h)) AS ?expectedLineMNPerM)
                BIND((?expectedLineMNPerM * 1000.0) AS ?expectedLineKNPerM)
                BIND(IF(?ls != 0.0, (1.0 - (?hs / ?ls)), 0.0) AS ?expectedFactor)
                BIND(
                    IF(
                        (?m != 0.0 && ?sigma != 0.0 && ?ls != 0.0),
                        (0.80 * 1.8 * ?expectedLineMNPerM * ?l * ?l
                         / (?m * ?sigma) * ?expectedFactor * 1000000.0),
                        0.0
                    )
                    AS ?expectedZ
                )
                BIND(
                    IF(
                        (?sigma != 0.0 && ?ls != 0.0),
                        (1.7320508075688772 * 0.80 * 1.8 * 1.2
                         * ?expectedLineMNPerM * ?l
                         / (2.0 * ?sigma) * ?expectedFactor * 10000.0),
                        0.0
                    )
                    AS ?expectedA
                )

                BIND(IF(ABS(?expectedLineKNPerM) * 0.000001 > 0.000000001,
                        ABS(?expectedLineKNPerM) * 0.000001,
                        0.000000001) AS ?lineTolerance)
                BIND(IF(ABS(?expectedFactor) * 0.000001 > 0.000000001,
                        ABS(?expectedFactor) * 0.000001,
                        0.000000001) AS ?factorTolerance)
                BIND(IF(ABS(?expectedZ) * 0.000001 > 0.000000001,
                        ABS(?expectedZ) * 0.000001,
                        0.000000001) AS ?zTolerance)
                BIND(IF(ABS(?expectedA) * 0.000001 > 0.000000001,
                        ABS(?expectedA) * 0.000001,
                        0.000000001) AS ?aTolerance)

                FILTER(
                    ?ls = 0.0 ||
                    ?m = 0.0 ||
                    ?sigma = 0.0 ||
                    ABS(?reportedLine - ?expectedLineKNPerM) > ?lineTolerance ||
                    ABS(?reportedFactor - ?expectedFactor) > ?factorTolerance ||
                    ABS(?reportedZ - ?expectedZ) > ?zTolerance ||
                    ABS(?reportedA - ?expectedA) > ?aTolerance
                )
            }
        """
    ] .
