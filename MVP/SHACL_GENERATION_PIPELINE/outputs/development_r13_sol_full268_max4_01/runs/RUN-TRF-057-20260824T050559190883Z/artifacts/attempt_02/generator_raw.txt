<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:WebFrameIceLoadShape
    a sh:NodeShape ;
    sh:targetClass nltl:webFrame ;
    sh:property [
        sh:path nltl:iceLoadSourceType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceStringerLoadSource
            nltl:longitudinalFramingLoadSource
        ) ;
        sh:message "Ice load source type must be exactly one permitted source."
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameSpacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadLengthParameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:webFrameIceLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Web frame ice load must equal 1.8 times the greater of p*h and 0.15 MN/m, times spacing, within relative tolerance 1e-6 or absolute tolerance 1e-9 kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:iceLoadHeight ?heightQuantity ;
                      nltl:webFrameSpacing ?spacingQuantity ;
                      nltl:webFrameIceLoad ?loadQuantity .
                ?pressureQuantity qudt:numericValue ?p .
                ?heightQuantity qudt:numericValue ?h .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?loadQuantity qudt:numericValue ?reportedLoad .
                BIND(
                    IF(
                        (?p * ?h) < "0.15"^^xsd:decimal,
                        "0.15"^^xsd:decimal,
                        (?p * ?h)
                    )
                    AS ?effectiveLineLoad
                )
                BIND(
                    ("1800"^^xsd:decimal * ?effectiveLineLoad * ?spacing)
                    AS ?expectedLoad
                )
                BIND(
                    IF(
                        ABS(?expectedLoad) * "0.000001"^^xsd:decimal
                            > "0.000000001"^^xsd:decimal,
                        ABS(?expectedLoad) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?reportedLoad - ?expectedLoad) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Load length parameter must equal twice the web frame spacing within relative tolerance 1e-6 or absolute tolerance 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:webFrameSpacing ?spacingQuantity ;
                      nltl:loadLengthParameter ?loadLengthQuantity .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?loadLengthQuantity qudt:numericValue ?reportedLoadLength .
                BIND(
                    ("2"^^xsd:decimal * ?spacing)
                    AS ?expectedLoadLength
                )
                BIND(
                    IF(
                        ABS(?expectedLoadLength) * "0.000001"^^xsd:decimal
                            > "0.000000001"^^xsd:decimal,
                        ABS(?expectedLoadLength) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(
                    ABS(?reportedLoadLength - ?expectedLoadLength) > ?tolerance
                )
            }
        """
    ] .
<END_SHACL>
