<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:MetreQuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "A decimal numeric value is required."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:maxCount 1 ;
        sh:message "The quantity must have metre as its single unit."
    ] .

gs:PropellerBladeTipHullClearanceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerBladeTipHullClearance ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityValueShape ;
        sh:message "Propeller blade tip hull clearance is required."
    ] ;
    sh:property [
        sh:path nltl:minimumClearance ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityValueShape ;
        sh:message "The required minimum clearance is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Propeller blade tip hull clearance must be no less than the required minimum clearance." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:propellerBladeTipHullClearance ?actualQuantity ;
          nltl:minimumClearance ?minimumQuantity .
    ?actualQuantity qudt:numericValue ?actualValue .
    ?minimumQuantity qudt:numericValue ?minimumValue .
    FILTER (
        xsd:decimal(?actualValue) < xsd:decimal(?minimumValue)
    )
}
"""
    ] .
<END_SHACL>
