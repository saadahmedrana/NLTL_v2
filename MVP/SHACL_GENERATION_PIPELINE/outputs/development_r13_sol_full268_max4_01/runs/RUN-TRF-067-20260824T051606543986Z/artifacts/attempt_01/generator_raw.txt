@prefix gen: <urn:nltl:generated-shape:> .

@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF067ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship's ice class is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                )
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:rudderLoadAbsorbingArrangement ;
                sh:minCount 1 ;
                sh:message "Ships of ice class IA or IA Super require a rudder load absorbing arrangement."
            ]
        ]
    ) .
