<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF-067-ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship must have an IRI-valued ice class."
    ] ;
    sh:or (
        [
            sh:not [
                sh:or (
                    [
                        sh:property [
                            sh:path nltl:iceClass ;
                            sh:hasValue nltl:iceClassIa
                        ]
                    ]
                    [
                        sh:property [
                            sh:path nltl:iceClass ;
                            sh:hasValue nltl:iceClassIaSuper
                        ]
                    ]
                )
            ]
        ]
        [
            sh:or (
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:hasValue nltl:iceClassIa
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:iceClass ;
                        sh:hasValue nltl:iceClassIaSuper
                    ]
                ]
            ) ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:rudderLoadAbsorbingArrangement ;
                sh:minCount 1 ;
                sh:message "Ships of ice class IA or IA Super must have a rudder load absorbing arrangement."
            ]
        ]
    ) .
<END_SHACL>
