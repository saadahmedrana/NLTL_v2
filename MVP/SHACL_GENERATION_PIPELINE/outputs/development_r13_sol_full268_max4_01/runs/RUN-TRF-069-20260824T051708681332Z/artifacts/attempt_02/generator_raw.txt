<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:ApplicablePropulsionMachineryShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:iceClass ?iceClass ;
                    nltl:propellerType ?propellerType ;
                    nltl:pitchType ?pitchType .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    ) &&
                    ?propellerType IN (
                        "open"^^xsd:string,
                        "ducted"^^xsd:string
                    ) &&
                    ?pitchType IN (
                        "controllable"^^xsd:string,
                        "fixed"^^xsd:string
                    )
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:maximumIceTorque ;
        sh:minCount 1 ;
        sh:node g:MaximumIceTorqueValueShape ;
        sh:message "Maximum ice torque in kN·m is required for an applicable ship."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node g:PropellerIceLoadDesignConditionShape ;
        sh:message "An applicable ship must have a conforming propeller ice-load design condition."
    ] .

g:MaximumIceTorqueValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN-M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

g:PropellerIceLoadDesignConditionShape
    a sh:NodeShape ;
    sh:class nltl:designCondition ;
    sh:property [
        sh:path nltl:normalOperationalConditions ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:minCount 1 ;
        sh:message "The design condition must represent normal operational conditions."
    ] ;
    sh:property [
        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:minCount 1 ;
        sh:message "The design condition must cover normal service-life propeller ice loads."
    ] ;
    sh:property [
        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:minCount 1 ;
        sh:message "Stopped-propeller dragging must be recorded as outside the load-model scope."
    ] ;
    sh:property [
        sh:path nltl:radialIceEntryOutsideLoadModel ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:minCount 1 ;
        sh:message "Radial ice entry must be recorded as outside the load-model scope."
    ] .

g:ApplicableFixedPitchPropulsionMachineryShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:iceClass ?iceClass ;
                    nltl:propellerType ?propellerType ;
                    nltl:pitchType "fixed"^^xsd:string .
                FILTER (
                    ?iceClass IN (
                        nltl:iceClassIaSuper,
                        nltl:iceClassIa,
                        nltl:iceClassIb,
                        nltl:iceClassIc
                    ) &&
                    ?propellerType IN (
                        "open"^^xsd:string,
                        "ducted"^^xsd:string
                    )
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:hasValue "fixed"^^xsd:string ;
        sh:message "The fixed-pitch branch requires explicit fixed pitch-type evidence."
    ] ;
    sh:property [
        sh:path nltl:hasDesignCondition ;
        sh:minCount 1 ;
        sh:node g:FixedPitchReversalDesignConditionShape ;
        sh:message "A fixed-pitch ship must include reversal loads in its design condition."
    ] .

g:FixedPitchReversalDesignConditionShape
    a sh:NodeShape ;
    sh:class nltl:designCondition ;
    sh:property [
        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
        sh:datatype xsd:boolean ;
        sh:in ( true ) ;
        sh:minCount 1 ;
        sh:message "Fixed-pitch propeller reversal loads must be included."
    ] .
