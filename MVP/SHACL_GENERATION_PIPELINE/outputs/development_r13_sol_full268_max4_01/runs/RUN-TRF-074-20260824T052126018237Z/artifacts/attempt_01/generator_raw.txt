<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF074LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:message "The quantity must have a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The quantity must have exactly one unit, metre."
    ] .

gen:TRF074ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "This alternative applies only when the ship is not of ice class IB or IC." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:iceClass ?iceClass .
                        FILTER(?iceClass IN (nltl:iceClassIb, nltl:iceClassIc))
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in ( nltl:iceClassIb nltl:iceClassIc )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:node gen:TRF074LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "Ice-class IB or IC ships require the ballast-condition propeller-top depth."
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:node gen:TRF074LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "Ice-class IB or IC ships require the required propeller submersion depth hi."
            ] ;
            sh:sparql [
                sh:message "This alternative requires the ballast-condition propeller-top depth to be at least hi." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
                              nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
                        ?depthQuantity qudt:numericValue ?depth .
                        ?hiQuantity qudt:numericValue ?hi .
                        FILTER(xsd:decimal(?depth) < xsd:decimal(?hi))
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:in ( nltl:iceClassIb nltl:iceClassIc )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:node gen:TRF074LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "Ice-class IB or IC ships require the ballast-condition propeller-top depth."
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:node gen:TRF074LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "Ice-class IB or IC ships require the required propeller submersion depth hi."
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemDesignedForIceClass ;
                sh:nodeKind sh:IRI ;
                sh:hasValue nltl:iceClassIa ;
                sh:message "When the ballast-condition propeller-top depth is less than hi, the propulsion system must be designed according to ice class IA."
            ] ;
            sh:sparql [
                sh:message "This alternative applies only when the ballast-condition propeller-top depth is less than hi." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
                              nltl:requiredPropellerSubmersionDepthHi ?hiQuantity .
                        ?depthQuantity qudt:numericValue ?depth .
                        ?hiQuantity qudt:numericValue ?hi .
                        FILTER(xsd:decimal(?depth) >= xsd:decimal(?hi))
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
