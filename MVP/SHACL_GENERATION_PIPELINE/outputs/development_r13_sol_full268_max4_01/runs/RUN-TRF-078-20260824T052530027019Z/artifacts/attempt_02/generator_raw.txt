        sh:path nltl:tableReference ;

    ] ;
    sh:sparql [
        sh:message "Every Table 6-6 lookup case must have an ice-class selection key of class iceClassValue and a selected ice-class cycle count." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:tableReference nltl:traficomTable6Dash6 .
                FILTER(
                    NOT EXISTS {
                        $this nltl:iceClass ?iceClass
                    }
                    ||
                    EXISTS {
                        $this nltl:iceClass ?iceClass .
                        FILTER NOT EXISTS {
                            ?iceClass a nltl:iceClassValue
                        }
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:selectedIceClassCycleCount ?selectedCount
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every Table 6-7 lookup case must have a propeller-location selection key of class propellerLocationValue and a selected propeller-location factor." ;

                $this nltl:tableReference nltl:traficomTable6Dash7 .
