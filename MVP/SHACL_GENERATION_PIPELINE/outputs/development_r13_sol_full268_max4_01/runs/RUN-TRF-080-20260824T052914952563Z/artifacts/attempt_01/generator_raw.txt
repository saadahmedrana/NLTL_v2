<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF080Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one integer ice load cycle count is required."
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one integer propeller blade count is required."
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Exactly one unitless decimal effective ice load cycle count is required."
    ] ;
    sh:sparql [
        sh:message "The effective ice load cycle count must equal the ice load cycle count multiplied by the propeller blade count, within a relative tolerance of 1e-6 and an absolute fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCount ?cycles ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?effectiveQuantity .
                ?effectiveQuantity qudt:numericValue ?reported .
                BIND(
                    xsd:decimal(?cycles) * xsd:decimal(?blades)
                    AS ?expected
                )
                BIND(
                    IF(
                        ABS(?expected) * 0.000001 > 0.000000000001,
                        ABS(?expected) * 0.000001,
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(xsd:decimal(?reported) - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
