<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MeterQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KilonewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LoadApplicationRadiusRatioShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "0.8"^^xsd:decimal ;
        sh:maxInclusive "0.8"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:node gen:LoadApplicationRadiusRatioShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The blade failure load application radius ratio must be 0.8."
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The blade failure load must be applied in the weakest blade direction."
    ] .

gen:TRF101ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:node gen:MeterQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:node gen:KilonewtonQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:node gen:BladeFailureLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "At least one blade failure load case must record application at 0.8R in the weakest direction."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedNonlinearBladeStressAnalysis ;
                sh:class nltl:approvalRecord ;
                sh:minCount 1 ;
                sh:message "An approved nonlinear blade stress analysis is required when the prescribed formula branch is not used."
            ]
        ]
        [
            sh:property [
                sh:path nltl:bladeChordLengthAtWeakestSection ;
                sh:node gen:MeterQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
                sh:node gen:MeterQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeCylindricalRootRadius ;
                sh:node gen:MeterQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeMinimumYieldStrength ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeUltimateTensileStrength ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:bladeReferenceStrength1 ;
                sh:node gen:MegaPascalQuantityShape ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The reported reference strength and blade failure load must match the prescribed formulas within a relative tolerance of 1e-6, with an absolute fallback of 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:bladeChordLengthAtWeakestSection/qudt:numericValue ?c ;
                              nltl:bladeMaximumThicknessAtWeakestSection/qudt:numericValue ?t ;
                              nltl:bladeCylindricalRootRadius/qudt:numericValue ?r ;
                              nltl:propellerDiameter/qudt:numericValue ?diameter ;
                              nltl:bladeMinimumYieldStrength/qudt:numericValue ?yieldStrength ;
                              nltl:bladeUltimateTensileStrength/qudt:numericValue ?ultimateStrength ;
                              nltl:bladeReferenceStrength1/qudt:numericValue ?reportedReferenceStrength ;
                              nltl:bladeFailureUltimateLoad/qudt:numericValue ?reportedLoad .

                        BIND (
                            (0.6 * ?yieldStrength) + (0.4 * ?ultimateStrength)
                            AS ?expectedReferenceStrength
                        )
                        BIND (
                            IF(
                                ABS(?expectedReferenceStrength) * 0.000001 > 0.000000001,
                                ABS(?expectedReferenceStrength) * 0.000001,
                                0.000000001
                            )
                            AS ?referenceTolerance
                        )
                        BIND (
                            (0.8 * ?diameter) - (2 * ?r)
                            AS ?denominator
                        )
                        BIND (
                            IF(
                                ?denominator = 0,
                                0,
                                (300 * ?c * ?t * ?t * ?expectedReferenceStrength) / ?denominator
                            )
                            AS ?expectedLoad
                        )
                        BIND (
                            IF(
                                ABS(?expectedLoad) * 0.000001 > 0.000000001,
                                ABS(?expectedLoad) * 0.000001,
                                0.000000001
                            )
                            AS ?loadTolerance
                        )
                        FILTER (
                            ?denominator = 0 ||
                            ABS(?reportedReferenceStrength - ?expectedReferenceStrength) > ?referenceTolerance ||
                            ABS(?reportedLoad - ?expectedLoad) > ?loadTolerance
                        )
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset/qudt:numericValue ?tipOffset ;
                      nltl:propellerDiameter/qudt:numericValue ?diameter ;
                      nltl:bladeFailureStatus ?failureStatus .
                FILTER (
                    ?tipOffset > ("0.10"^^xsd:decimal * ?diameter) &&
                    ?failureStatus != true
                )
            }
        """
    ] .
<END_SHACL>
