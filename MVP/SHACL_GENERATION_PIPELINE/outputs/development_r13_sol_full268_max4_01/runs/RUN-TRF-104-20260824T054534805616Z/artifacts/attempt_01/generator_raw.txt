<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PropulsionLinePyramidStrengthShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The propeller blade-loss design case must be represented."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:minCount 1 ;
        sh:node gen:PropellerShaftLineComponentBladeLossShape ;
        sh:message "Each linked propeller shaft line component must satisfy the blade-loss damage constraint."
    ] .

gen:PropellerShaftLineComponentBladeLossShape
    a sh:NodeShape ;
    sh:class nltl:propellerShaftLineComponent ;
    sh:property [
        sh:path nltl:significantDamageFromBladeLoss ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( false ) ;
        sh:message "A propeller shaft line component must not sustain significant damage from propeller blade loss."
    ] .
<END_SHACL>
