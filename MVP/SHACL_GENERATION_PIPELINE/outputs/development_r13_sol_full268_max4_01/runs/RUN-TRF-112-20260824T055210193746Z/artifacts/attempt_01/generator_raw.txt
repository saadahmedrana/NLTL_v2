<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF112ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:class nltl:loadCase ;
        sh:node gs:TRF112LoadCaseSelectorShape ;
        sh:minCount 1 ;
        sh:message "A propulsion load case with an explicit shaft-line load-case type is required."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:qualifiedValueShape [
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:shaftLineLoadCaseType ;
                sh:hasValue nltl:bladeFailureLoadCase
            ]
        ] ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one propulsion load case must represent the blade-failure load case."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent ;
        sh:node gs:TRF112ShaftComponentResultShape ;
        sh:minCount 1 ;
        sh:message "At least one propeller shaft-line component with calculated yield safety factors is required."
    ] .

gs:TRF112LoadCaseSelectorShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:class nltl:shaftLineLoadCaseTypeValue ;
        sh:minCount 1 ;
        sh:message "Each propulsion load case must state its shaft-line load-case type."
    ] .

gs:TRF112BladeFailureLoadCaseShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?ship nltl:hasPropulsionLoadCase ?this .
                ?this nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .
            }
        """
    ] ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:class qudt:QuantityValue ;
        sh:node gs:TRF112KilonewtonQuantityShape ;
        sh:minCount 1 ;
        sh:message "A blade-failure load case requires the externally determined ultimate blade-failure load."
    ] ;
    sh:property [
        sh:path nltl:combinedAxialBendingTorsionLoad ;
        sh:class qudt:QuantityValue ;
        sh:node gs:TRF112KilonewtonQuantityShape ;
        sh:minCount 1 ;
        sh:message "A blade-failure load case requires the combined axial, bending, and torsion load input."
    ] .

gs:TRF112ShaftComponentResultShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bendingYieldSafetyFactor ;
        sh:class qudt:QuantityValue ;
        sh:node gs:TRF112MinimumYieldSafetyFactorShape ;
        sh:minCount 1 ;
        sh:message "Each propeller shaft-line component requires an externally calculated bending yield safety factor of at least 1.0."
    ] ;
    sh:property [
        sh:path nltl:torsionalYieldSafetyFactor ;
        sh:class qudt:QuantityValue ;
        sh:node gs:TRF112MinimumYieldSafetyFactorShape ;
        sh:minCount 1 ;
        sh:message "Each propeller shaft-line component requires an externally calculated torsional yield safety factor of at least 1.0."
    ] .

gs:TRF112KilonewtonQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF112MinimumYieldSafetyFactorShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.0 ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
