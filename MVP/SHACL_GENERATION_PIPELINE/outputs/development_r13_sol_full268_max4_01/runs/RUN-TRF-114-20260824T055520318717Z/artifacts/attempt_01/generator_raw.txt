<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SpeedQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ForceQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:ThrusterBodyReadinessShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:thrusterResistanceCapacity ;
        sh:class qudt:QuantityValue ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The thruster body must provide its externally assessed resistance capacity in kN."
    ] .

gen:ThrusterIceImpactLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:message "The impact case must reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:message "The impact case must reference TRAFICOM Table 6-16."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:class nltl:thrusterBody ;
        sh:node gen:ThrusterBodyReadinessShape ;
        sh:minCount 1 ;
        sh:message "The impact case must identify the applicable thruster body."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:class qudt:QuantityValue ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The impact case must provide the externally determined ice impact load in kN."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:class qudt:QuantityValue ;
        sh:node gen:ForceQuantityShape ;
        sh:minCount 1 ;
        sh:message "The impact case must provide the externally assessed ice impact demand in kN."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The impact case must provide loaded-area evidence."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry ;
                sh:in ( nltl:hemisphericalContactGeometry ) ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry ;
                sh:in ( nltl:nonHemisphericalContactGeometry ) ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:message "A non-hemispherical contact case must provide geometry-correspondence evidence."
            ]
        ]
    ) .

gen:TRF114Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOperatingSpeed ;
        sh:class qudt:QuantityValue ;
        sh:node gen:SpeedQuantityShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide its typical ice operating speed in knots."
    ] ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:node gen:ThrusterIceImpactLoadCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide at least one thruster ice impact load case."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasThrusterIceImpactLoadCase ;
                sh:node [
                    sh:property [
                        sh:path nltl:contactGeometryClassification ;
                        sh:hasValue nltl:hemisphericalContactGeometry ;
                        sh:in ( nltl:hemisphericalContactGeometry ) ;
                        sh:minCount 1
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:equivalentImpactSphereRadius ;
                sh:class qudt:QuantityValue ;
                sh:node gen:LengthQuantityShape ;
                sh:minCount 1 ;
                sh:message "A ship with a non-hemispherical contact case must provide an equivalent impact sphere radius in metres."
            ]
        ]
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The thruster resistance capacity must be at least the ice impact demand for each linked impact case." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?case ?body
WHERE {
    $this nltl:hasThrusterIceImpactLoadCase ?case .
    ?case nltl:thrusterIceImpactDemand ?demandQuantity ;
          nltl:thrusterImpactCaseAppliesToBody ?body .
    ?demandQuantity qudt:numericValue ?demand .
    ?body nltl:thrusterResistanceCapacity ?capacityQuantity .
    ?capacityQuantity qudt:numericValue ?capacity .
    FILTER(xsd:decimal(?capacity) < xsd:decimal(?demand))
}
"""
    ] .
<END_SHACL>
