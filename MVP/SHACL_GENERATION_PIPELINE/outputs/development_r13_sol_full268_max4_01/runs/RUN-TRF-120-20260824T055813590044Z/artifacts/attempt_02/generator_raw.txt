<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ThrusterNaturalFrequencyRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:node gen:ThrusterBodyNaturalFrequencyShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path (
                    nltl:hasComponent
                    nltl:hasNaturalFrequencyCalculationCase
                ) ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:damping ;
                sh:minCount 1 ;
                sh:datatype xsd:string ;
                sh:message "Damping information is required when a thruster natural-frequency calculation case is represented."
            ] ;
            sh:property [
                sh:path nltl:waterAddedMass ;
                sh:minCount 1 ;
                sh:node gen:WaterAddedMassValueShape ;
                sh:message "Water added mass is required when a thruster natural-frequency calculation case is represented."
            ] ;
            sh:property [
                sh:path nltl:thrusterNaturalFrequency ;
                sh:minCount 1 ;
                sh:node gen:ThrusterNaturalFrequencyValueShape ;
                sh:message "Thruster natural frequency is required when a natural-frequency calculation case is represented."
            ]
        ]
    ) .

gen:ThrusterBodyNaturalFrequencyShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasNaturalFrequencyCalculationCase ;
        sh:node gen:NaturalFrequencyCalculationCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasNaturalFrequencyCalculationCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipAttachmentStiffness ;
                sh:minCount 1 ;
                sh:node gen:ShipAttachmentStiffnessValueShape ;
                sh:message "Ship attachment stiffness is required for a thruster body with a natural-frequency calculation case."
            ]
        ]
    ) .

gen:NaturalFrequencyCalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:vibrationDirection ;
        sh:minCount 1 ;
        sh:in (
            nltl:longitudinalDirection
            nltl:transverseDirection
        ) ;
        sh:message "Each natural-frequency calculation case must identify a longitudinal or transverse vibration direction."
    ] .

gen:WaterAddedMassValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloGM
    ] .

gen:ThrusterNaturalFrequencyValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:HZ
    ] .

gen:ShipAttachmentStiffnessValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN-PER-M
    ] .
<END_SHACL>
