gs:TRF127AirReceiverCapacityShape
    a sh:NodeShape ;

                sh:datatype xsd:boolean ;

                sh:message "The ship must explicitly state whether the air receiver serves an additional purpose."
            ]
        ]
        [
            sh:property [
                sh:path nltl:airReceiverServesAdditionalPurpose ;
                sh:datatype xsd:boolean ;
                sh:in ( true ) ;
                sh:minCount 1 ;
                sh:message "The ship must explicitly state whether the air receiver serves an additional purpose."
            ] ;
            sh:property [
                sh:path nltl:airReceiverCapacity ;

                        sh:minCount 1

                        sh:path qudt:unit ;
                        sh:hasValue unit:M3 ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:message "Air receiver capacity is required when the receiver serves an additional purpose."
            ] ;

PREFIX qudt: <http://qudt.org/schema/qudt/>
SELECT $this
WHERE {
    $this nltl:airReceiverServesAdditionalPurpose true ;
          nltl:airReceiverCapacity ?capacityNode ;
          nltl:additionalPurposeRequiredAirCapacity ?requiredNode .
    ?capacityNode qudt:numericValue ?capacity .
    ?requiredNode qudt:numericValue ?requiredCapacity .
    FILTER (?capacity < ?requiredCapacity)
}
"""
