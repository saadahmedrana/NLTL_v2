<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF128Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one ice class to determine the applicable compressor charge-time limit."
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether propulsion-engine reversal is required for going astern."
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Compressor charge time must be provided once as a decimal quantity in hours."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "0.5"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "1"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                a sh:SPARQLConstraint ;
                sh:message "This branch applies only when the ship's ice class is not IA Super." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    FILTER(?iceClass = nltl:iceClassIaSuper)
}
"""
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "1"^^xsd:decimal
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "Compressor charge time must not exceed 0.5 hour for an IA Super ship requiring engine reversal for astern operation, and must not exceed 1 hour otherwise." .
<END_SHACL>
