<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF128Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have exactly one ice class identifying an iceClassValue."
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must state whether propulsion-engine reversal is required for going astern."
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxInclusive "1.0"^^xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Compressor charge time must not exceed one hour."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:HR ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Compressor charge time must be expressed in hours."
            ]
        ]
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For ice class IA Super ships requiring propulsion-engine reversal for astern, compressor charge time must not exceed 0.5 hour." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:propulsionEngineReversalRequiredForAstern ?reversal ;
                      nltl:compressorChargeTime ?chargeTime .
                ?chargeTime qudt:numericValue ?hours ;
                            qudt:unit unit:HR .
                FILTER (
                    ?reversal = "true"^^xsd:boolean &&
                    ?hours > "0.5"^^xsd:decimal
                )
            }
        """
    ] .
<END_SHACL>
