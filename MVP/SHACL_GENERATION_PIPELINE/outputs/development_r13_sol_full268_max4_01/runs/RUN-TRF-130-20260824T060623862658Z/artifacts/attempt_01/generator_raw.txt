@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:InletChestRequirements
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:node gen:VolumeQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:node gen:AreaQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:node gen:RatioQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The recommended chest volume shall approximate total relevant engine power divided by 750, and the strainer open area shall be at least four times the inlet pipe area." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    ?ship nltl:hasComponent $this ;
          nltl:totalRelevantEnginePower ?powerQuantity .
    ?powerQuantity qudt:numericValue ?power .
    $this nltl:recommendedChestVolume ?recommendedQuantity ;
          nltl:strainerOpenArea ?strainerQuantity ;
          nltl:inletPipeArea ?inletQuantity .
    ?recommendedQuantity qudt:numericValue ?reportedVolume .
    ?strainerQuantity qudt:numericValue ?strainerArea .
    ?inletQuantity qudt:numericValue ?inletArea .
    BIND((?power / 750) AS ?expectedVolume)
    BIND(ABS(?expectedVolume) * 0.000001 AS ?relativeTolerance)
    BIND(IF(?relativeTolerance > 0.000000000001,
            ?relativeTolerance,
            0.000000000001) AS ?tolerance)
    FILTER(
        ABS(?reportedVolume - ?expectedVolume) > ?tolerance ||
        ?strainerArea < (4 * ?inletArea)
    )
}
"""
    ] .

    ] .

    sh:node gen:InletChestRequirements .

gen:TRF130Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:node gen:PowerQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship shall state whether the cooling-water chest volume and height requirements can be met."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape [
                    sh:node gen:InletChestRequirements
                ] ;
                sh:qualifiedMinCount 1 ;
                sh:message "At least one cooling-water inlet chest shall satisfy the standard arrangement requirements."
            ]
        ]
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape [
                    sh:node gen:AlternatingInletChestRequirements
                ] ;
                sh:qualifiedMinCount 2 ;
                sh:message "The alternative arrangement shall provide at least two inlet chests for alternating cooling-water intake and discharge."
            ]
        ]
    ) .
