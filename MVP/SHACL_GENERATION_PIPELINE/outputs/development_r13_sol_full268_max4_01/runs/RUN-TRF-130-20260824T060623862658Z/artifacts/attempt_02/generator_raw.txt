@prefix gen: <urn:nltl:generated-shape:> .

@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF130ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerQuantityShape ;
        sh:message "Maximum continuous rating power is required in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PowerQuantityShape ;
        sh:message "Total relevant engine power is required in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The volume-and-height applicability selector is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape gen:TRF130StandardInletChestShape ;
                sh:qualifiedMinCount 1 ;
                sh:message "At least one linked inlet chest must satisfy the standard arrangement."
            ]
        ]
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:message "The alternative cooling-water-chest arrangement must be recorded as used."
            ] ;
            sh:property [
                sh:path nltl:hasComponent ;
                sh:qualifiedValueShape gen:TRF130AlternatingInletChestShape ;
                sh:qualifiedMinCount 2 ;
                sh:message "At least two linked inlet chests must satisfy the alternating arrangement."
            ]
        ]
    ) .

gen:TRF130StandardInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:node gen:TRF130CommonInletChestShape ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape ;
        sh:message "Recommended chest volume is required in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:VolumeQuantityShape ;
        sh:message "Sea-chest volume is required in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The inlet chest must allow ice accumulation above the inlet pipe."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Recommended chest volume must equal total relevant engine power divided by 750 within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    ?ship nltl:hasComponent $this ;
          nltl:totalRelevantEnginePower ?powerQuantity .
    ?powerQuantity qudt:numericValue ?powerValue .
    $this nltl:recommendedChestVolume ?volumeQuantity .
    ?volumeQuantity qudt:numericValue ?volumeValue .
    BIND(?powerValue / xsd:decimal(750) AS ?expectedVolume)
    BIND(ABS(?expectedVolume) * xsd:decimal(0.000001) AS ?relativeTolerance)
    BIND(
        IF(
            ?relativeTolerance > xsd:decimal(0.000000001),
            ?relativeTolerance,
            xsd:decimal(0.000000001)
        )
        AS ?tolerance
    )
    FILTER(ABS(?volumeValue - ?expectedVolume) > ?tolerance)
}
"""
    ] .

gen:TRF130AlternatingInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:node gen:TRF130CommonInletChestShape ;
    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "Each alternative inlet chest must support alternating intake and discharge."
    ] .

gen:TRF130CommonInletChestShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Sea inlet location must be recorded."
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "The sea inlet must be near the centreline and well aft if possible."
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:message "A full-capacity discharge pipe must be connected."
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape ;
        sh:message "Strainer open area is required in square metres."
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape ;
        sh:message "Inlet pipe area is required in square metres."
    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:RatioQuantityShape ;
        sh:message "Strainer open-area ratio is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Strainer open area must be at least four times inlet pipe area, and the reported ratio must match their quotient within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:strainerOpenArea ?strainerQuantity ;
          nltl:inletPipeArea ?inletQuantity ;
          nltl:strainerOpenAreaRatio ?ratioQuantity .
    ?strainerQuantity qudt:numericValue ?strainerValue .
    ?inletQuantity qudt:numericValue ?inletValue .
    ?ratioQuantity qudt:numericValue ?reportedRatio .
    FILTER(
        ?inletValue = xsd:decimal(0)
        ||
        ?strainerValue < (xsd:decimal(4) * ?inletValue)
        ||
        IF(
            ?inletValue = xsd:decimal(0),
            false,
            ABS(?reportedRatio - (?strainerValue / ?inletValue))
            >
            IF(
                ABS(?strainerValue / ?inletValue) * xsd:decimal(0.000001)
                    > xsd:decimal(0.000000001),
                ABS(?strainerValue / ?inletValue) * xsd:decimal(0.000001),
                xsd:decimal(0.000000001)
            )
        )
    )
}
"""
    ] .
