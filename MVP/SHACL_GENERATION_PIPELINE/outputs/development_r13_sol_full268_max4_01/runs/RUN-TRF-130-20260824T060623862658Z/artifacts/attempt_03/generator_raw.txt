gen:StandardChestBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasComponent ;
        sh:qualifiedValueShape gen:StandardInletChestShape ;
        sh:qualifiedMinCount 1 ;
        sh:message "At least one linked inlet chest must satisfy the standard cooling-water chest arrangement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "No linked inlet chest satisfies the required volume and strainer calculations." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    FILTER NOT EXISTS {
        $this nltl:totalRelevantEnginePower ?powerQuantity ;
              nltl:hasComponent ?chest .
        ?powerQuantity a qudt:QuantityValue ;
                       qudt:numericValue ?power ;
                       qudt:unit unit:KiloW .
        ?chest a nltl:inletChest ;
               nltl:seaInletLocation ?location ;
               nltl:nearCentrelineAndWellAftIfPossible true ;
               nltl:iceAccumulationSpaceAboveInlet true ;
               nltl:fullCapacityDischargePipeConnected true ;
               nltl:recommendedChestVolume ?recommendedQuantity ;
               nltl:seaChestVolume ?seaVolumeQuantity ;
               nltl:strainerOpenArea ?openAreaQuantity ;
               nltl:inletPipeArea ?inletAreaQuantity ;
               nltl:strainerOpenAreaRatio ?ratioQuantity .
        ?recommendedQuantity a qudt:QuantityValue ;
                             qudt:numericValue ?recommended ;
                             qudt:unit unit:M3 .
        ?seaVolumeQuantity a qudt:QuantityValue ;
                           qudt:numericValue ?seaVolume ;
                           qudt:unit unit:M3 .
        ?openAreaQuantity a qudt:QuantityValue ;
                          qudt:numericValue ?openArea ;
                          qudt:unit unit:M2 .
        ?inletAreaQuantity a qudt:QuantityValue ;
                           qudt:numericValue ?inletArea ;
                           qudt:unit unit:M2 .
        ?ratioQuantity a qudt:QuantityValue ;
                       qudt:numericValue ?ratio ;
                       qudt:unit unit:UNITLESS .
        BIND(?power / "750"^^xsd:decimal AS ?expectedVolume)
        BIND(
            IF(
                ABS(?expectedVolume) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                ABS(?expectedVolume) * "0.000001"^^xsd:decimal,
                "0.000000001"^^xsd:decimal
            ) AS ?volumeTolerance
        )
        BIND(
            IF(
                ?inletArea != "0"^^xsd:decimal,
                ?openArea / ?inletArea,
                "0"^^xsd:decimal
            ) AS ?expectedRatio
        )
        BIND(
            IF(
                ABS(?expectedRatio) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
                ABS(?expectedRatio) * "0.000001"^^xsd:decimal,
                "0.000000001"^^xsd:decimal
            ) AS ?ratioTolerance
        )
        FILTER(
            DATATYPE(?location) = xsd:string &&
            ?inletArea != "0"^^xsd:decimal &&
            ?openArea >= "4"^^xsd:decimal * ?inletArea &&
            ?ratio >= "4"^^xsd:decimal &&
            ABS(?recommended - ?expectedVolume) <= ?volumeTolerance &&
            ABS(?ratio - ?expectedRatio) <= ?ratioTolerance
        )
    }
}
"""
    ] .

gen:AlternativeInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;

        sh:maxCount 1 ;

    ] ;
    sh:property [
        sh:path nltl:strainerOpenAreaRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:StrainerRatioQuantityShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The strainer open area must be at least four times the inlet pipe area, and the reported ratio must correspond to those areas." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:strainerOpenArea ?openAreaQuantity ;
          nltl:inletPipeArea ?inletAreaQuantity ;
          nltl:strainerOpenAreaRatio ?ratioQuantity .
    ?openAreaQuantity qudt:numericValue ?openArea .
    ?inletAreaQuantity qudt:numericValue ?inletArea .
    ?ratioQuantity qudt:numericValue ?ratio .
    BIND(
        IF(
            ?inletArea != "0"^^xsd:decimal,
            ?openArea / ?inletArea,
            "0"^^xsd:decimal
        ) AS ?expectedRatio
    )
    BIND(
        IF(
            ABS(?expectedRatio) * "0.000001"^^xsd:decimal > "0.000000001"^^xsd:decimal,
            ABS(?expectedRatio) * "0.000001"^^xsd:decimal,
            "0.000000001"^^xsd:decimal
        ) AS ?ratioTolerance
    )
    FILTER(
        ?inletArea = "0"^^xsd:decimal ||
        ?openArea < "4"^^xsd:decimal * ?inletArea ||
        ?ratio < "4"^^xsd:decimal ||
        ABS(?ratio - ?expectedRatio) > ?ratioTolerance
    )
}
"""
    ] .
