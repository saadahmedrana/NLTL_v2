gen:StandardInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:node gen:CommonInletChestShape ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CubicMetreQuantityShape ;
        sh:message "Recommended chest volume must be provided in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:seaChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CubicMetreQuantityShape ;
        sh:message "Sea chest volume must be provided in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The inlet chest must provide ice-accumulation space above the inlet."
    ] ;
    sh:sparql [
        sh:message "Recommended chest volume must match total relevant engine power divided by 750 within tolerance, and the strainer area and ratio must each be at least four times the inlet basis." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    ?ship nltl:hasComponent $this ;
          nltl:totalRelevantEnginePower ?powerQuantity .
    ?powerQuantity qudt:numericValue ?powerValue .

    $this nltl:recommendedChestVolume ?recommendedQuantity ;
          nltl:strainerOpenArea ?openAreaQuantity ;
          nltl:inletPipeArea ?inletAreaQuantity ;
          nltl:strainerOpenAreaRatio ?ratioQuantity .
    ?recommendedQuantity qudt:numericValue ?recommendedValue .
    ?openAreaQuantity qudt:numericValue ?openAreaValue .
    ?inletAreaQuantity qudt:numericValue ?inletAreaValue .
    ?ratioQuantity qudt:numericValue ?ratioValue .

    BIND(xsd:decimal(?powerValue) / xsd:decimal("750") AS ?expectedVolume)
    BIND(
        IF(
            ABS(?expectedVolume) * xsd:decimal("0.000001") > xsd:decimal("0.000000001"),
            ABS(?expectedVolume) * xsd:decimal("0.000001"),
            xsd:decimal("0.000000001")
        )
        AS ?volumeTolerance
    )

    FILTER(
        ABS(xsd:decimal(?recommendedValue) - ?expectedVolume) > ?volumeTolerance ||
        xsd:decimal(?openAreaValue) < xsd:decimal("4") * xsd:decimal(?inletAreaValue) ||
        xsd:decimal(?ratioValue) < xsd:decimal("4")
    )
}
"""
    ] .

gen:AlternativeInletChestShape
    a sh:NodeShape ;
    sh:class nltl:inletChest ;
    sh:node gen:CommonInletChestShape ;
    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each alternative inlet chest must support alternating intake and discharge."
    ] ;
    sh:sparql [
        sh:message "For each alternative inlet chest, the strainer area and reported strainer ratio must satisfy the factor-of-four minimum." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:strainerOpenArea ?openAreaQuantity ;
          nltl:inletPipeArea ?inletAreaQuantity ;
          nltl:strainerOpenAreaRatio ?ratioQuantity .
    ?openAreaQuantity qudt:numericValue ?openAreaValue .
    ?inletAreaQuantity qudt:numericValue ?inletAreaValue .
    ?ratioQuantity qudt:numericValue ?ratioValue .

    FILTER(
        xsd:decimal(?openAreaValue) < xsd:decimal("4") * xsd:decimal(?inletAreaValue) ||
        xsd:decimal(?ratioValue) < xsd:decimal("4")
    )
}
"""
    ] .
