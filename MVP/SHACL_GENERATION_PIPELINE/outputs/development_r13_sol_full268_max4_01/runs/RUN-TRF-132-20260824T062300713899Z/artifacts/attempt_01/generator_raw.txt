<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KilowattQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF132EngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice class is required to determine the applicable minimum engine output."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Maximum continuous rating power must be supplied once in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:node gen:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula base power P0 must be supplied once in kilowatts."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:node gen:DecimalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula displacement must have one decimal numeric value."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f1 must be supplied once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f2 must be supplied once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:node gen:UnitlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f3 must be supplied once as a unitless quantity."
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:node gen:DecimalQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Formula factor f4 must have one decimal numeric value."
    ] ;
    sh:sparql [
        sh:message "Engine output must be at least f1*f2*f3*(f4*Delta + P0), and at least 740 kW for ice classes IB and IC." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:maximumContinuousRatingPower ?mcrQuantity ;
                      nltl:engineOutputFormulaBasePowerP0 ?p0Quantity ;
                      nltl:engineOutputFormulaDisplacement ?displacementQuantity ;
                      nltl:engineOutputFormulaFactorF1 ?f1Quantity ;
                      nltl:engineOutputFormulaFactorF2 ?f2Quantity ;
                      nltl:engineOutputFormulaFactorF3 ?f3Quantity ;
                      nltl:engineOutputFormulaFactorF4 ?f4Quantity .

                ?mcrQuantity qudt:numericValue ?mcr .
                ?p0Quantity qudt:numericValue ?p0 .
                ?displacementQuantity qudt:numericValue ?displacement .
                ?f1Quantity qudt:numericValue ?f1 .
                ?f2Quantity qudt:numericValue ?f2 .
                ?f3Quantity qudt:numericValue ?f3 .
                ?f4Quantity qudt:numericValue ?f4 .

                FILTER (
                    ?mcr < (?f1 * ?f2 * ?f3 * ((?f4 * ?displacement) + ?p0))
                    ||
                    (
                        ?iceClass IN (nltl:iceClassIb, nltl:iceClassIc)
                        &&
                        ?mcr < "740"^^xsd:decimal
                    )
                )
            }
        """
    ] .
<END_SHACL>
