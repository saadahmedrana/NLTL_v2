<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF133ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:message "The ship must state its ice class."
    ] ;
    sh:property [
        sh:path nltl:timberLoadLineMarkApplicable ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Timber load line applicability must be stated."
    ] ;
    sh:property [
        sh:path nltl:hasWarningTriangleMarking ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:warningTriangleMarking ;
        sh:node gen:TRF133WarningTriangleMarkingShape ;
        sh:message "The ship must have a conforming warning triangle marking."
    ] ;
    sh:property [
        sh:path nltl:hasIceClassDraughtMarking ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassDraughtMarking ;
        sh:node gen:TRF133IceClassDraughtMarkingShape ;
        sh:message "The ship must have a conforming ice-class draught marking."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( true )
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:minCount 1 ;
                sh:node gen:TRF133TimberReferenceShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:timberLoadLineMarkApplicable ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( false )
            ] ;
            sh:property [
                sh:path nltl:hasIceClassDraughtMarking ;
                sh:minCount 1 ;
                sh:node gen:TRF133LoadLineRingReferenceShape
            ]
        ]
    ) .

gen:TRF133WarningTriangleMarkingShape
    a sh:NodeShape ;
    sh:class nltl:warningTriangleMarking ;
    sh:node gen:TRF133AggregateMarkingShape ;
    sh:property [
        sh:path nltl:warningTriangleUpperEdgeVerticallyAboveIceMark ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:markingAtOrBelowDeckLine ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:verticalOffsetAboveSummerFreshWaterLoadLine ;
        sh:minCount 1 ;
        sh:node gen:TRF133Exact1000MillimetresShape
    ] ;
    sh:property [
        sh:path nltl:warningTriangleSideLength ;
        sh:minCount 1 ;
        sh:node gen:TRF133Exact300MillimetresShape
    ] .

gen:TRF133IceClassDraughtMarkingShape
    a sh:NodeShape ;
    sh:class nltl:iceClassDraughtMarking ;
    sh:node gen:TRF133AggregateMarkingShape ;
    sh:property [
        sh:path nltl:draughtMarkAftOffset ;
        sh:minCount 1 ;
        sh:node gen:TRF133Exact540MillimetresShape
    ] ;
    sh:property [
        sh:path nltl:letterDimensionsEqualLoadLineMark ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] .

gen:TRF133AggregateMarkingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:markingPlateThickness ;
        sh:minCount 1 ;
        sh:node gen:TRF133PlateThicknessShape
    ] ;
    sh:property [
        sh:path nltl:markingWeldedToShipSide ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:reflectingMarkingColour ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:redReflectingMarkingColour
            nltl:yellowReflectingMarkingColour
        )
    ] ;
    sh:property [
        sh:path nltl:markingPlainlyVisibleInIceConditions ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] .

gen:TRF133TimberReferenceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:draughtMarkAftReferencePoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:timberLoadLineVerticalReference )
    ] .

gen:TRF133LoadLineRingReferenceShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:draughtMarkAftReferencePoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:loadLineRingCentreReference )
    ] .

gen:TRF133Exact1000MillimetresShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "1000"^^xsd:decimal ;
        sh:maxInclusive "1000"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in ( unit:MilliM )
    ] .
