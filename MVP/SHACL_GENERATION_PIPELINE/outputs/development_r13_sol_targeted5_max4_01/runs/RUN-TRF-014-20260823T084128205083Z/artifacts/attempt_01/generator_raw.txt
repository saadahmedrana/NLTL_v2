gen:TRF014MarkingShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:warningTrianglePresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A warning triangle is required when the draught-marking condition applies."
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An ice-class draught mark is required when the marking condition applies."
    ] ;
    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF014LengthQuantityShape ;
        sh:message "The ice-class draught mark must state the draught amidships."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ice-class draught mark amidships must equal the maximum ice-class draught amidships stated in the class certificate." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceClassDraughtMarkDraughtAmidships ?markQuantity .
    ?markQuantity qudt:numericValue ?markValue .
    $this nltl:hasClassCertificate ?certificate .
    ?certificate nltl:maximumIceClassDraughtAmidships ?maximumQuantity .
    ?maximumQuantity qudt:numericValue ?maximumValue .
    FILTER (xsd:decimal(?markValue) != xsd:decimal(?maximumValue))
}
"""
    ] .

gen:TRF014SummerLoadLineNotAboveUiwlShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "This unmarked alternative is permitted only when the summer load line in fresh water is not above the upper ice waterline." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:summerLoadLineFreshWaterDraught ?summerQuantity .
    ?summerQuantity qudt:numericValue ?summerValue .
    $this nltl:upperIceWaterlineDraught ?uiwlQuantity .
    ?uiwlQuantity qudt:numericValue ?uiwlValue .
    FILTER (xsd:decimal(?summerValue) > xsd:decimal(?uiwlValue))
}
"""
    ] .

gen:TRF014DryDockingAfterAssessmentShape
    a sh:NodeShape ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "If the first scheduled dry docking is on or before the assessment date, complete warning-triangle and ice-class-draught-mark evidence is required." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
    $this nltl:assessmentDate ?assessmentDate .
    FILTER (
        YEAR(?dryDockingDate) < YEAR(?assessmentDate) ||
        (
            YEAR(?dryDockingDate) = YEAR(?assessmentDate) &&
            MONTH(?dryDockingDate) < MONTH(?assessmentDate)
        ) ||
        (
            YEAR(?dryDockingDate) = YEAR(?assessmentDate) &&
            MONTH(?dryDockingDate) = MONTH(?assessmentDate) &&
            DAY(?dryDockingDate) <= DAY(?assessmentDate)
        )
    )
}
"""
    ] .
