<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:I2_026ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_026StructuralMemberShape
    ] .

gs:I2_026StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        )
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:minCount 1 ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:continuousThroughSupport ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:hasMemberSupport ;
                        sh:minCount 1 ;
                        sh:qualifiedValueShape [
                            sh:property [
                                sh:path nltl:connectionBracketPresent ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:hasValue true
                            ]
                        ] ;
                        sh:qualifiedMinCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:continuousThroughSupport ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue false
                    ] ;
                    sh:property [
                        sh:path nltl:hasMemberSupport ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path nltl:connectionBracketPresent ;
                                sh:datatype xsd:boolean ;
                                sh:minCount 1 ;
                                sh:hasValue false
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:significantRotationalRestraintEvidence ;
                        sh:minCount 1 ;
                        sh:class nltl:evidenceArtifact
                    ]
                ]
            )
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:minCount 1 ;
                sh:hasValue nltl:simpleFrameSupport
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:connectionBracketPresent ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:hasValue false
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:maxCount 0
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A framing member terminating within an ice-strengthened area requires fixed frame support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:terminatesWithinIceStrengthenedArea true .
                FILTER NOT EXISTS {
                    $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                }
            }
        """
    ] .
<END_SHACL>
