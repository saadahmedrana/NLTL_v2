<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_052ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gs:I2_052CalculationCaseShape
    ] .

gs:I2_052CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gs:I2_052MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node gs:I2_052UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_052InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node gs:I2_052InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:node gs:I2_052InterpolationPointShape
    ] ;
    sh:sparql [
        sh:message "Interpolation endpoints must have distinct coordinates." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:interpolationLowerEndpoint ?lower ;
                      nltl:interpolationUpperEndpoint ?upper .
                ?lower nltl:interpolationCoordinate ?lowerCoordinateQuantity .
                ?upper nltl:interpolationCoordinate ?upperCoordinateQuantity .
                ?lowerCoordinateQuantity qudt:numericValue ?xLow .
                ?upperCoordinateQuantity qudt:numericValue ?xHigh .
                FILTER (?xLow = ?xHigh)
            }
        """
    ] ;
    sh:sparql [
        sh:message "The interpolated longitudinal distribution factor does not equal the linear interpolation result within relative tolerance 1e-6 and absolute tolerance 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:interpolatedLongitudinalDistributionFactor ?factorQuantity ;
                      nltl:interpolationLowerEndpoint ?lower ;
                      nltl:interpolationUpperEndpoint ?upper .
                ?positionQuantity qudt:numericValue ?x .
                ?factorQuantity qudt:numericValue ?reported .
                ?lower nltl:interpolationCoordinate ?lowerCoordinateQuantity ;
                       nltl:interpolationPointValue ?lowerValueQuantity .
                ?upper nltl:interpolationCoordinate ?upperCoordinateQuantity ;
                       nltl:interpolationPointValue ?upperValueQuantity .
                ?lowerCoordinateQuantity qudt:numericValue ?xLow .
                ?upperCoordinateQuantity qudt:numericValue ?xHigh .
                ?lowerValueQuantity qudt:numericValue ?yLow .
                ?upperValueQuantity qudt:numericValue ?yHigh .
                FILTER (?xHigh != ?xLow)
                BIND((?yLow + ((?x - ?xLow) / (?xHigh - ?xLow)) * (?yHigh - ?yLow)) AS ?expected)
                FILTER (ABS(?reported - ?expected) > ((ABS(?expected) * 0.000001) + 0.000000001))
            }
        """
    ] .

gs:I2_052InterpolationPointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node gs:I2_052DecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node gs:I2_052DecimalQuantityShape
    ] .

gs:I2_052DecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] .

gs:I2_052MetreQuantityShape
    a sh:NodeShape ;
    sh:node gs:I2_052DecimalQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .

gs:I2_052UnitlessQuantityShape
    a sh:NodeShape ;
    sh:node gs:I2_052DecimalQuantityShape ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .
<END_SHACL>
