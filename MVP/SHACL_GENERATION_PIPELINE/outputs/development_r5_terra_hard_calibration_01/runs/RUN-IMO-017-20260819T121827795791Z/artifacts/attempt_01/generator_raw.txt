<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO017CertificateScheduleDateRecordShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateScheduleDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Each certificate schedule date record must provide a certificate schedule date."
    ] .

gen:IMO017SolasCertificateScheduleShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasCertificateScheduleDateRecord ;
        sh:class nltl:certificateScheduleDateRecord ;
        sh:node gen:IMO017CertificateScheduleDateRecordShape ;
        sh:minCount 1 ;
        sh:message "The SOLAS certificate schedule must include certificate schedule date records."
    ] .

gen:IMO017PolarCertificateSupplementShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasRequiredEquipmentRecords ;
        sh:class nltl:requiredEquipmentRecords ;
        sh:minCount 1 ;
        sh:message "The Polar Ship Certificate supplement must record the equipment required by the Code."
    ] .

gen:IMO017PolarCertificateConsistencyShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCertificateValidityDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate validity date is required."
    ] ;
    sh:property [
        sh:path nltl:surveyDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A survey date is required."
    ] ;
    sh:property [
        sh:path nltl:endorsementDates ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "An endorsement date is required."
    ] ;
    sh:property [
        sh:path nltl:polarCertificateSupplementPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The Polar Ship Certificate supplement must be present."
    ] ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:class nltl:polarShipCertificateForm ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate record is required."
    ] ;
    sh:property [
        sh:path nltl:hasSolasCertificateSchedule ;
        sh:class nltl:solasCertificateSchedule ;
        sh:node gen:IMO017SolasCertificateScheduleShape ;
        sh:minCount 1 ;
        sh:message "The applicable SOLAS certificate schedule is required."
    ] ;
    sh:property [
        sh:path nltl:hasPolarCertificateSupplement ;
        sh:class nltl:polarCertificateSupplement ;
        sh:node gen:IMO017PolarCertificateSupplementShape ;
        sh:minCount 1 ;
        sh:message "A Polar Ship Certificate supplement with required equipment records is required."
    ] ;
    sh:sparql [
        sh:message "Each Polar certificate validity, survey, and endorsement date must match a date in the linked SOLAS certificate schedule." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?date
            WHERE {
                $this (nltl:polarCertificateValidityDates|nltl:surveyDates|nltl:endorsementDates) ?date .
                FILTER NOT EXISTS {
                    $this nltl:hasSolasCertificateSchedule ?schedule .
                    ?schedule nltl:hasCertificateScheduleDateRecord ?record .
                    ?record nltl:certificateScheduleDate ?date .
                }
            }
        """
    ] .
<END_SHACL>
