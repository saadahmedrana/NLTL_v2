<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:IMO102PolarTrainingShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        )
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        )
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gs:IMO102CrewRoleShape
    ] ;
    sh:sparql [
        sh:message "Applicable crew member lacks a matching valid STCW polar-training record for the required matrix level." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?crew
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role .

                FILTER(
                    (?ice = nltl:otherWatersIceCondition) ||
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType IN (
                            nltl:tankerPolarTrainingShipType,
                            nltl:passengerShipPolarTrainingShipType
                        )
                    )
                )

                BIND(
                    IF(
                        ?ice = nltl:otherWatersIceCondition &&
                        ?role IN (
                            nltl:masterPolarTrainingRole,
                            nltl:chiefMatePolarTrainingRole
                        ),
                        nltl:advancedPolarTrainingLevel,
                        nltl:basicPolarTrainingLevel
                    )
                    AS ?expectedLevel
                )

                FILTER NOT EXISTS {
                    ?crew nltl:hasCrewTrainingRecord ?record ;
                          nltl:hasPolarTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?expectedLevel ;
                            nltl:trainingRecordLevel ?expectedLevel ;
                            nltl:stcwQualificationValid true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A polar-training record in a not-applicable matrix cell must be linked through both declared record relationships and record the not-applicable level." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?crew ?record
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role ;
                      nltl:hasPolarTrainingRecord ?record .

                FILTER(
                    ?ice = nltl:iceFreeIceCondition ||
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType = nltl:otherShipPolarTrainingType
                    )
                )

                FILTER NOT EXISTS {
                    ?crew nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel nltl:polarTrainingNotApplicable ;
                            nltl:trainingRecordLevel nltl:polarTrainingNotApplicable .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A crew-training record in a not-applicable matrix cell must also be the declared polar-training record and record the not-applicable level." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?crew ?record
            WHERE {
                $this nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType ;
                      nltl:hasCrewMemberInventory ?crew .
                ?crew nltl:polarTrainingCrewRoleClassification ?role ;
                      nltl:hasCrewTrainingRecord ?record .

                FILTER(
                    ?ice = nltl:iceFreeIceCondition ||
                    (
                        ?ice = nltl:openWaterIceCondition &&
                        ?shipType = nltl:otherShipPolarTrainingType
                    )
                )

                FILTER NOT EXISTS {
                    ?crew nltl:hasPolarTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel nltl:polarTrainingNotApplicable ;
                            nltl:trainingRecordLevel nltl:polarTrainingNotApplicable .
                }
            }
        """
    ] .

gs:IMO102CrewRoleShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        )
    ] .
<END_SHACL>
