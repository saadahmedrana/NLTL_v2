@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-069-PropulsionMachineryScopeShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For the applicable ice class, propeller type, and pitch type, a linked design condition must record the required normal-service-life load scope and stated exclusions." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER(?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER(?propellerType IN (
                    "open"^^xsd:string,
                    "ducted"^^xsd:string
                ))
                FILTER(?pitchType IN (
                    "controllable"^^xsd:string,
                    "fixed"^^xsd:string
                ))

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?condition .
                    ?condition nltl:normalOperationalConditions true ;
                               nltl:normalServiceLifePropellerIceLoadCondition true ;
                               nltl:fixedPitchPropellerReversalLoadIncluded true ;
                               nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                               nltl:radialIceEntryOutsideLoadModel true .
                }
            }
        """
    ] .
