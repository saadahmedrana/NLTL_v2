<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF069PropulsionMachineryScopeShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "For an applicable propulsion-machinery configuration, record maximum ice torque in kN-m and a linked design condition covering the required normal-service-life load scope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER(?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER(?propellerType IN ("open"^^xsd:string, "ducted"^^xsd:string))
                FILTER(?pitchType IN ("controllable"^^xsd:string, "fixed"^^xsd:string))

                FILTER NOT EXISTS {
                    $this nltl:maximumIceTorque ?maximumTorque ;
                          nltl:hasDesignCondition ?condition .

                    ?maximumTorque a qudt:QuantityValue ;
                        qudt:numericValue ?torqueValue ;
                        qudt:unit unit:KiloN-M .

                    FILTER(datatype(?torqueValue) = xsd:decimal)

                    ?condition nltl:normalOperationalConditions true ;
                        nltl:normalServiceLifePropellerIceLoadCondition true ;
                        nltl:fixedPitchPropellerReversalLoadIncluded true ;
                        nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                        nltl:radialIceEntryOutsideLoadModel true .
                }
            }
        """
    ] .
<END_SHACL>
