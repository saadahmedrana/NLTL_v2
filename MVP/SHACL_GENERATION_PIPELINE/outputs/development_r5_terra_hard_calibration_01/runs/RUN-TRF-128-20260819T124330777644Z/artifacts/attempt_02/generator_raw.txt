<urn:nltl:generated-shape:TRF-128-AirCompressorChargingShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice class is required to determine the applicable compressor charging-time limit."
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "The propulsion-engine reversal status is required to determine the applicable compressor charging-time limit."
    ] ;
    sh:property [
        sh:path nltl:airReceiverCapacity ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M3
            ]
        ] ;
        sh:message "Air receiver capacity must be represented as a quantity in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:HR
            ]
        ] ;
        sh:message "Compressor charge time must be represented as a quantity in hours."
    ] ;
    sh:sparql [
        sh:message "Compressor charge time for the stated air receiver capacity must not exceed 0.5 hours for ice class IA Super with propulsion-engine reversal required for astern, and must not exceed 1 hour otherwise." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:airReceiverCapacity ?airReceiverCapacity ;
                      nltl:compressorChargeTime ?compressorChargeTime ;
                      nltl:iceClass ?iceClass ;
                      nltl:propulsionEngineReversalRequiredForAstern ?reversalRequired .
                ?airReceiverCapacity qudt:numericValue ?capacityValue ;
                                     qudt:unit unit:M3 .
                ?compressorChargeTime qudt:numericValue ?chargeTime ;
                                      qudt:unit unit:HR .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper &&
                     ?reversalRequired = true &&
                     ?chargeTime > "0.5"^^xsd:decimal)
                    ||
                    ((?iceClass != nltl:iceClassIaSuper ||
                      ?reversalRequired != true) &&
                     ?chargeTime > "1.0"^^xsd:decimal)
                )
            }
        """
    ] .
