@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .

@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_031ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:minCount 1 ;
        sh:class nltl:frameProfileTypeValue ;
        sh:message "A frame profile type is required on the ship."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_031MemberApplicabilityShape
    ] ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:node gs:I2_031ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "A patch load is required when a linked structural member is in the bottom structure." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation nltl:bottomStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:patchLoad ?patchLoad .
                }
            }
        """
    ] .

gs:I2_031MemberApplicabilityShape
    a sh:NodeShape ;
    sh:or (
        gs:I2_031BottomMemberShape
        gs:I2_031TransverseSideMemberShape
        gs:I2_031NonTransverseSideMemberShape
    ) ;
    sh:message "A structural member must have a supported structural location and, for side structure, a recorded transverse-frame orientation." .

gs:I2_031ForceQuantityShape

    ] ;

gs:I2_031DemandQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] .

gs:I2_031PlasticStrengthDefinitionConstraint
    sh:message "Plastic strength must equal the midspan plastic collapse load within a relative tolerance of 1e-6 and an absolute tolerance of 1e-6 kN." ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl/vocab#>
        PREFIX qudt: <http://qudt.org/schema/qudt/>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
        SELECT $this
        WHERE {
            $this nltl:plasticStrength ?strengthQuantity .
            ?strengthQuantity qudt:numericValue ?strengthValue .
            $this nltl:midspanPlasticCollapseLoad ?collapseQuantity .
            ?collapseQuantity qudt:numericValue ?collapseValue .
            FILTER (
                ABS(xsd:decimal(?strengthValue) - xsd:decimal(?collapseValue))
                > ((0.000001 * ABS(xsd:decimal(?collapseValue))) + 0.000001)
            )
        }
    """ .

gs:I2_031ApplicableBottomMemberShape
    sh:sparql [
        sh:message "Combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strengthValue .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demandValue .
                FILTER (xsd:decimal(?demandValue) > xsd:decimal(?strengthValue))
            }
        """
    ] .
