<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:I2_026ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gen:I2_026StructuralMemberShape
    ] .

gen:I2_026StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Record whether the framing member is continuous through its support."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Record whether the framing member terminates within an ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gen:I2_026MemberSupportShape
    ] ;
    sh:sparql [
        sh:message "Significant rotational restraint must be supported by evidence when it is demonstrated at a support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The frame support condition must be fixed where fixity is established; otherwise it must be simple support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:continuousThroughSupport ?continuous ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates ;
                      nltl:hasMemberSupport ?support .
                ?support nltl:connectionBracketPresent ?bracket ;
                         nltl:demonstratedRotationalRestraint ?demonstrated ;
                         nltl:frameSupportCondition ?condition .
                FILTER (
                    (
                        (
                            ?continuous = true ||
                            ?terminates = true ||
                            ?bracket = true ||
                            (
                                ?demonstrated = true &&
                                EXISTS {
                                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                                }
                            )
                        ) &&
                        ?condition != nltl:fixedFrameSupport
                    )
                    ||
                    (
                        ?continuous = false &&
                        ?terminates = false &&
                        ?bracket = false &&
                        ?demonstrated = false &&
                        ?condition != nltl:simpleFrameSupport
                    )
                )
            }
        """
    ] .

gen:I2_026MemberSupportShape
    a sh:NodeShape ;
    sh:class nltl:memberSupport ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Record whether the support connection has a connection bracket."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Record whether significant rotational restraint is demonstrated at the support."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:message "Frame support condition must be fixed frame support or simple frame support."
    ] .
<END_SHACL>
