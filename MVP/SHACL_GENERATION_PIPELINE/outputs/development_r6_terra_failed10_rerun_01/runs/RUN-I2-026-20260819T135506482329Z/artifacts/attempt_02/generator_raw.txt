<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026StructuralMemberShape
    ] .

gen:I2_026StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:frameSupportConditionValue ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:message "A structural member must record its frame support condition as fixed or simple support."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Whether the structural member is continuous through its support must be recorded."
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Whether the structural member terminates within an ice-strengthened area must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026MemberSupportShape ;
        sh:message "A structural member must identify its member support to establish the applicable support condition."
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        sh:message "The frame support condition must be fixed where continuity, a connection bracket, significant demonstrated rotational restraint, or termination in an ice-strengthened area establishes fixity; otherwise it must be simple support." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition ?condition .
                FILTER (
                    (
                        ?condition = nltl:fixedFrameSupport &&
                        !EXISTS { $this nltl:continuousThroughSupport true } &&
                        !EXISTS { $this nltl:terminatesWithinIceStrengthenedArea true } &&
                        !EXISTS {
                            $this nltl:hasMemberSupport ?support .
                            ?support nltl:connectionBracketPresent true .
                        } &&
                        !EXISTS {
                            $this nltl:hasMemberSupport ?support .
                            ?support nltl:demonstratedRotationalRestraint true .
                        }
                    ) ||
                    (
                        ?condition = nltl:simpleFrameSupport &&
                        (
                            EXISTS { $this nltl:continuousThroughSupport true } ||
                            EXISTS { $this nltl:terminatesWithinIceStrengthenedArea true } ||
                            EXISTS {
                                $this nltl:hasMemberSupport ?support .
                                ?support nltl:connectionBracketPresent true .
                            } ||
                            EXISTS {
                                $this nltl:hasMemberSupport ?support .
                                ?support nltl:demonstratedRotationalRestraint true .
                            }
                        )
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Significant rotational restraint must be supported by evidence."
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """
    ] .

gen:I2_026MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Whether the member support has a connection bracket must be recorded."
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:message "Whether the member support provides demonstrated rotational restraint must be recorded."
    ] .
<END_SHACL>
