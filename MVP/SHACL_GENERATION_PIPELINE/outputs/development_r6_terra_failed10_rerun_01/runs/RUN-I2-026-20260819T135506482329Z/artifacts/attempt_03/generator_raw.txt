<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_026ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:structuralMember ;
        sh:node gen:I2_026StructuralMemberShape
    ] .

gen:I2_026StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:fixedFrameSupport
            nltl:simpleFrameSupport
        )
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:memberSupport ;
        sh:node gen:I2_026MemberSupportShape
    ] ;
    sh:property [
        sh:path nltl:significantRotationalRestraintEvidence ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact
    ] ;
    sh:sparql [
        sh:message "The frame support condition must be fixed where continuity, a connection bracket, demonstrated rotational restraint, or termination within an ice-strengthened area requires fixity; otherwise it must be simple." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition ?condition ;
                      nltl:continuousThroughSupport ?continuous ;
                      nltl:terminatesWithinIceStrengthenedArea ?terminates .
                OPTIONAL {
                    $this nltl:hasMemberSupport ?support .
                    ?support nltl:connectionBracketPresent ?bracket ;
                             nltl:demonstratedRotationalRestraint ?demonstrated .
                }
                FILTER (
                    (
                        (
                            ?terminates = true ||
                            ?continuous = true ||
                            (BOUND(?support) && ?bracket = true) ||
                            (BOUND(?support) && ?bracket = false && ?demonstrated = true)
                        ) &&
                        ?condition != nltl:fixedFrameSupport
                    ) ||
                    (
                        BOUND(?support) &&
                        ?bracket = false &&
                        ?demonstrated = false &&
                        ?condition != nltl:simpleFrameSupport
                    )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Demonstrated significant rotational restraint requires significant rotational restraint evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """
    ] .

gen:I2_026MemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] .
<END_SHACL>
