        sh:class nltl:structuralMember ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCasePlating ;
        sh:class nltl:plating ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:PlatingShape>
    ] ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node <urn:nltl:generated-shape:QuantityMillimetreShape>

    sh:sparql [
        sh:message "The local-frame section case must be linked from the same structural member used by sectionCalculationCaseStructuralMember." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    ?member nltl:hasLocalFrameSectionCalculationCase $this .
                    $this nltl:sectionCalculationCaseStructuralMember ?member .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Local-frame flange area, attached shell plate thickness, and frame spacing are required to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                FILTER (
                    NOT EXISTS {
                        ?member nltl:netLocalFrameFlangeArea ?a .
                        ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                    } ||
                    NOT EXISTS {
                        ?member nltl:netAttachedShellPlateThickness ?tp .
                        ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                    } ||
                    NOT EXISTS {
                        ?member nltl:frameSpacing ?s .
                        ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "framingAngleOmega is required to select the direct or interpolation section-modulus branch." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating .
                ?member nltl:netLocalFrameFlangeArea ?a ;
                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv)
                FILTER NOT EXISTS {
                    ?plating nltl:framingAngleOmega ?omega .
                    ?omega qudt:numericValue ?ov ; qudt:unit unit:DEG .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "plasticNeutralAxisHeight is required when the local-frame area exceeds the attached-plate-flange area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?a ;
                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv)
                FILTER NOT EXISTS {

    sh:sparql [
        sh:message "All operands and netEffectivePlasticSectionModulus are required for the applicable local-frame section calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating .
                ?member nltl:netLocalFrameFlangeArea ?a ;
                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv)
                FILTER (
                    NOT EXISTS { ?member nltl:webHeight ?v . ?v qudt:numericValue ?n ; qudt:unit unit:MilliM . } ||
                    NOT EXISTS { ?member nltl:netWebThickness ?v . ?v qudt:numericValue ?n ; qudt:unit unit:MilliM . } ||
                    NOT EXISTS { ?member nltl:localFrameFlangeCentreHeight ?v . ?v qudt:numericValue ?n ; qudt:unit unit:MilliM . } ||
                    NOT EXISTS { ?member nltl:webAngleToShellPlate ?v . ?v qudt:numericValue ?n ; qudt:unit unit:DEG . } ||
                    NOT EXISTS { ?member nltl:webToFlangeCentreDistance ?v . ?v qudt:numericValue ?n ; qudt:unit unit:MilliM . } ||
                    NOT EXISTS { ?member nltl:netEffectivePlasticSectionModulus ?v . ?v qudt:numericValue ?n ; qudt:unit unit:CentiM3 . }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Interpolation endpoints with coordinates and results are required for oblique framing." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;

                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                ?plating nltl:framingAngleOmega ?omega .
                ?omega qudt:numericValue ?ov ; qudt:unit unit:DEG .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv && ?ov > 20 && ?ov < 70)
                FILTER (
                    NOT EXISTS {
                        $this nltl:interpolationLowerEndpoint ?lower .
                        ?lower nltl:interpolationPointCoordinate ?lc ;
                               nltl:interpolationPointResult ?lr .
                        ?lc qudt:numericValue ?lcv ; qudt:unit unit:UNITLESS .
                        ?lr qudt:numericValue ?lrv .
                    } ||
                    NOT EXISTS {
                        $this nltl:interpolationUpperEndpoint ?upper .
                        ?upper nltl:interpolationPointCoordinate ?uc ;
                               nltl:interpolationPointResult ?ur .
                        ?uc qudt:numericValue ?ucv ; qudt:unit unit:UNITLESS .
                        ?ur qudt:numericValue ?urv .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "plasticNeutralAxisHeight must equal the prescribed local-frame formula within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:plasticNeutralAxisHeight ?zna .
                ?member nltl:netLocalFrameFlangeArea ?a ;
                        nltl:webHeight ?hw ;
                        nltl:netWebThickness ?tw ;
                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?hw qudt:numericValue ?hwv ; qudt:unit unit:MilliM .
                ?tw qudt:numericValue ?twv ; qudt:unit unit:MilliM .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                ?zna qudt:numericValue ?znav ; qudt:unit unit:MilliM .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv)
                BIND ((100 * ?av + ?hwv * ?twv - 1000 * ?tpv * ?sv) / (2 * ?twv) AS ?expected)
                FILTER (ABS(?znav - ?expected) > (0.000001 * ABS(?expected) + 0.000000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "netEffectivePlasticSectionModulus must equal the prescribed direct formula outside the oblique-framing interpolation range." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating ;
                      nltl:plasticNeutralAxisHeight ?zna .
                ?member nltl:netLocalFrameFlangeArea ?a ;
                        nltl:webHeight ?hw ;
                        nltl:netWebThickness ?tw ;
                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s ;
                        nltl:localFrameFlangeCentreHeight ?hfc ;
                        nltl:webAngleToShellPlate ?phi ;
                        nltl:webToFlangeCentreDistance ?bw ;
                        nltl:netEffectivePlasticSectionModulus ?zp .
                ?plating nltl:framingAngleOmega ?omega .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?hw qudt:numericValue ?hwv ; qudt:unit unit:MilliM .
                ?tw qudt:numericValue ?twv ; qudt:unit unit:MilliM .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                ?hfc qudt:numericValue ?hfcv ; qudt:unit unit:MilliM .
                ?phi qudt:numericValue ?phiv ; qudt:unit unit:DEG .
                ?bw qudt:numericValue ?bwv ; qudt:unit unit:MilliM .
                ?zna qudt:numericValue ?znav ; qudt:unit unit:MilliM .
                ?zp qudt:numericValue ?zpv ; qudt:unit unit:CentiM3 .
                ?omega qudt:numericValue ?ov ; qudt:unit unit:DEG .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv)
                FILTER (!(?ov > 20 && ?ov < 70))
                BIND (?phiv * 3.141592653589793 / 180 AS ?radians)
                BIND (
                    ?tpv * ?sv * (?znav + ?tpv / 2) * math:sin(?radians) +
                    (((?hwv - ?znav) * (?hwv - ?znav) + ?znav * ?znav) * ?twv * math:sin(?radians)) / 2000 +
                    ?av * ((?hfcv - ?znav) * math:sin(?radians) - ?bwv * math:cos(?radians)) / 10
                    AS ?expected
                )
                FILTER (ABS(?zpv - ?expected) > (0.000001 * ABS(?expected) + 0.000000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "netEffectivePlasticSectionModulus must equal the linear interpolation result for oblique framing." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;

                        nltl:netAttachedShellPlateThickness ?tp ;
                        nltl:frameSpacing ?s ;

                ?upper nltl:interpolationPointCoordinate ?uc ;
                       nltl:interpolationPointResult ?ur .
                ?a qudt:numericValue ?av ; qudt:unit unit:CentiM2 .
                ?tp qudt:numericValue ?tpv ; qudt:unit unit:MilliM .
                ?s qudt:numericValue ?sv ; qudt:unit unit:M .
                ?zp qudt:numericValue ?zpv ; qudt:unit unit:CentiM3 .
                ?omega qudt:numericValue ?ov ; qudt:unit unit:DEG .
                ?lc qudt:numericValue ?lcv ; qudt:unit unit:UNITLESS .
                ?uc qudt:numericValue ?ucv ; qudt:unit unit:UNITLESS .
                ?lr qudt:numericValue ?lrv .
                ?ur qudt:numericValue ?urv .
                FILTER (100 * ?av > 1000 * ?tpv * ?sv && ?ov > 20 && ?ov < 70)
                FILTER (?ucv != ?lcv)
                BIND (?lrv + ((?ov - ?lcv) * (?urv - ?lrv) / (?ucv - ?lcv)) AS ?expected)
                FILTER (ABS(?zpv - ?expected) > (0.000001 * ABS(?expected) + 0.000000001))
            }
        """
    ] .
