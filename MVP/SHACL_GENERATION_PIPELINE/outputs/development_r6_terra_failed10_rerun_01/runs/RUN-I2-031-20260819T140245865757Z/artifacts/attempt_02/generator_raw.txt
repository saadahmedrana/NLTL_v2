<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_031Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_031StructuralMemberSelectorShape
    ] ;
    sh:sparql [
        sh:message "An applicable member must have a plastic strength quantity in kN with a decimal value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                FILTER NOT EXISTS {
                    ?member nltl:plasticStrength ?strength .
                    ?strength qudt:numericValue ?value ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable member must have a midspan plastic-collapse load quantity in kN with a decimal value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                FILTER NOT EXISTS {
                    ?member nltl:midspanPlasticCollapseLoad ?load .
                    ?load qudt:numericValue ?value ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each load case of an applicable member must provide a decimal combined shear and bending demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                FILTER NOT EXISTS {
                    ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                    ?demand qudt:numericValue ?value .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Bottom-structure load cases must apply the patch-load height parallel to the frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation nltl:bottomStructureLocation ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    ?loadCase nltl:loadPatchOrientation
                        nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A bottom-structure case requires a patch-load quantity in kN with a decimal value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation nltl:bottomStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:patchLoad ?load .
                    ?load qudt:numericValue ?value ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable member requires a recorded plastic collapse mechanism." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                FILTER NOT EXISTS {
                    $this nltl:plasticCollapseMechanism ?mechanism .
                    FILTER(datatype(?mechanism) = xsd:string)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Combined shear and bending demand must not exceed the member plastic strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location ;
                        nltl:plasticStrength ?strength ;
                        nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                ?strength qudt:numericValue ?strengthValue ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue .
                FILTER(
                    datatype(?strengthValue) = xsd:decimal &&
                    datatype(?demandValue) = xsd:decimal &&
                    ?demandValue > ?strengthValue
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Plastic strength must equal the midspan plastic-collapse load within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:structuralLocation ?location ;
                        nltl:plasticStrength ?strength ;
                        nltl:midspanPlasticCollapseLoad ?collapseLoad .
                FILTER(
                    ?location = nltl:bottomStructureLocation ||
                    (?location = nltl:sideStructureLocation &&
                     EXISTS { ?member nltl:transverseFrameOrientation true })
                )
                ?strength qudt:numericValue ?strengthValue ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?collapseLoad qudt:numericValue ?collapseValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                FILTER(
                    datatype(?strengthValue) = xsd:decimal &&
                    datatype(?collapseValue) = xsd:decimal &&
                    ABS(?strengthValue - ?collapseValue) >
                    ((0.000001 * ABS(?collapseValue)) + 0.000000001)
                )
            }
        """
    ] .

gs:I2_031StructuralMemberSelectorShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "A side-structure member requires a boolean transverse-frame orientation to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:structuralLocation nltl:sideStructureLocation .
                FILTER NOT EXISTS {
                    $this nltl:transverseFrameOrientation ?orientation .
                    FILTER(datatype(?orientation) = xsd:boolean)
                }
            }
        """
    ] .
<END_SHACL>
