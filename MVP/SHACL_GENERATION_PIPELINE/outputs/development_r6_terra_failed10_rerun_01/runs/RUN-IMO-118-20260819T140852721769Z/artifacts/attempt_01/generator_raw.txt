<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO118SewageDischargeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "Construction date is required to determine sewage-discharge applicability."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:minCount 1
            ]
        ]
        [
            sh:class nltl:passengerShip
        ]
    ) ;
    sh:message "Ship category or authoritative passenger-ship classification is required to determine sewage-discharge applicability." ;
    sh:property [
        sh:path nltl:sewageDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Applicable ships shall not discharge sewage to sea unless paragraph 4.2.1.3 compliance is recorded as true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date .

                FILTER (
                    YEAR(?date) > 2017 ||
                    (
                        YEAR(?date) = 2017 &&
                        (
                            MONTH(?date) > 1 ||
                            (MONTH(?date) = 1 && DAY(?date) >= 1)
                        )
                    )
                )

                FILTER (
                    EXISTS {
                        $this nltl:shipCategory ?category .
                        FILTER (
                            ?category IN (
                                nltl:polarShipCategoryA,
                                nltl:polarShipCategoryB
                            )
                        )
                    }
                    ||
                    EXISTS {
                        $this a nltl:passengerShip .
                    }
                )

                FILTER NOT EXISTS {
                    $this nltl:polarCodeSewageParagraph4Point2Point1Point3ComplianceStatus true .
                }

                FILTER (
                    ! EXISTS {
                        $this nltl:sewageDischargeToSea false .
                    }
                    ||
                    EXISTS {
                        $this nltl:sewageDischargeToSea ?reportedDischarge .
                        FILTER (?reportedDischarge != false)
                    }
                )
            }
        """
    ] .
<END_SHACL>
