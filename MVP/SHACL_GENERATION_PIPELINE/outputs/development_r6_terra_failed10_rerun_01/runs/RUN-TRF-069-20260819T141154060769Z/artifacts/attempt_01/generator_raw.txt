<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-069ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "An applicable ship must link to a design condition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER(?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER(?propellerType IN (
                    "open"^^xsd:string,
                    "ducted"^^xsd:string
                ))
                FILTER(?pitchType IN (
                    "controllable"^^xsd:string,
                    "fixed"^^xsd:string
                ))

                FILTER NOT EXISTS {
                    $this nltl:hasDesignCondition ?designCondition .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ship must provide maximum ice torque as a decimal quantity in kN*m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER(?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER(?propellerType IN (
                    "open"^^xsd:string,
                    "ducted"^^xsd:string
                ))
                FILTER(?pitchType IN (
                    "controllable"^^xsd:string,
                    "fixed"^^xsd:string
                ))

                FILTER NOT EXISTS {
                    $this nltl:maximumIceTorque ?torque .
                    ?torque a qudt:QuantityValue ;
                            qudt:numericValue ?value ;
                            qudt:unit unit:KiloN-M .
                    FILTER(DATATYPE(?value) = xsd:decimal)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-069DesignConditionShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasDesignCondition ;
    sh:sparql [
        sh:message "The applicable design condition must cover normal service-life propeller/ice loads, include fixed-pitch reversal loads, and record both excluded load-model cases." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasDesignCondition $this ;
                      nltl:iceClass ?iceClass ;
                      nltl:propellerType ?propellerType ;
                      nltl:pitchType ?pitchType .

                FILTER(?iceClass IN (
                    nltl:iceClassIaSuper,
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER(?propellerType IN (
                    "open"^^xsd:string,
                    "ducted"^^xsd:string
                ))
                FILTER(?pitchType IN (
                    "controllable"^^xsd:string,
                    "fixed"^^xsd:string
                ))

                FILTER NOT EXISTS {
                    $this nltl:normalOperationalConditions true ;
                          nltl:normalServiceLifePropellerIceLoadCondition true ;
                          nltl:fixedPitchPropellerReversalLoadIncluded true ;
                          nltl:stoppedPropellerDraggingOutsideLoadModel true ;
                          nltl:radialIceEntryOutsideLoadModel true .
                }
            }
        """
    ] .
<END_SHACL>
