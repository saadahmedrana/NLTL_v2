<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-001Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Ships contracted on or after 1 January 2021 must record UR I2 Rev.4 as applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate .
                FILTER(?contractDate >= "2021-01-01"^^xsd:date)
                FILTER NOT EXISTS {
                    $this nltl:urI2RevisionApplicability true .
                }
            }
        """
    ] .
<END_SHACL>
