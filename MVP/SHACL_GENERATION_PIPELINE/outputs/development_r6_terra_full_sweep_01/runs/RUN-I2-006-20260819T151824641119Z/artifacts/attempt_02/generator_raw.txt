<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-006Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:asternIceOperationIntent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Astern ice operation intent is required to determine applicability."
    ] ;
    sh:sparql [
        sh:message "A ship intended to operate astern in ice regions must state that its aft section design basis uses the Bow and Bow Intermediate hull area requirements." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:asternIceOperationIntent true .
                FILTER NOT EXISTS {
                    $this nltl:aftSectionDesignBasis ?basis .
                    FILTER(
                        DATATYPE(?basis) = xsd:string &&
                        STR(?basis) = "Bow and Bow Intermediate hull area requirements"
                    )
                }
            }
        """
    ] .
<END_SHACL>
