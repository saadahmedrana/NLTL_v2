<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-009QuantityMPaShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-009QuantityMetreShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-009QuantityDegreeShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-009QuantityUnitlessShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:I2-009ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityMPaShape>
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityMetreShape>
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityMetreShape>
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityUnitlessShape>
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityDegreeShape>
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node <urn:nltl:generated-shape:I2-009QuantityDegreeShape>
    ] ;
    sh:sparql [
        sh:message "Other ice-strengthened areas require Pavg, bNonBow, wNonBow, and load patch aspect ratio AR = 3.6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasHullAreaAssignment ?assignment .
                ?assignment nltl:assignedHullArea nltl:otherIceStrengthenedArea .
                FILTER NOT EXISTS {
                    $this nltl:averageIcePressure ?pressure ;
                          nltl:nonBowLoadPatchHeight ?height ;
                          nltl:nonBowLoadPatchLength ?length ;
                          nltl:loadPatchAspectRatio ?ratio .
                    ?pressure qudt:numericValue ?pressureValue ;
                              qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?height qudt:numericValue ?heightValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?length qudt:numericValue ?lengthValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?ratio qudt:numericValue ?ratioValue ;
                           qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (
                        datatype(?pressureValue) = xsd:decimal &&
                        datatype(?heightValue) = xsd:decimal &&
                        datatype(?lengthValue) = xsd:decimal &&
                        datatype(?ratioValue) = xsd:decimal &&
                        ?ratioValue >= 3.6 &&
                        ?ratioValue <= 3.6
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A bow-form applicability assertion requires 0 deg < buttock angle < 80 deg and normal frame angle > 10 deg." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowFormApplicability true .
                FILTER NOT EXISTS {
                    $this nltl:buttockAngle ?buttock ;
                          nltl:normalFrameAngle ?frame .
                    ?buttock qudt:numericValue ?buttockValue ;
                             qudt:unit <http://qudt.org/vocab/unit/DEG> .
                    ?frame qudt:numericValue ?frameValue ;
                           qudt:unit <http://qudt.org/vocab/unit/DEG> .
                    FILTER (
                        datatype(?buttockValue) = xsd:decimal &&
                        datatype(?frameValue) = xsd:decimal &&
                        ?buttockValue > 0 &&
                        ?buttockValue < 80 &&
                        ?frameValue > 10
                    )
                }
            }
        """
    ] .
<END_SHACL>
