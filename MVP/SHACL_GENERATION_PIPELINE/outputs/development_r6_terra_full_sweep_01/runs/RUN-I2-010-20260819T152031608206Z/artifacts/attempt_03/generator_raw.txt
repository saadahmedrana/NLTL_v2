<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-010Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:verticalSidedBowForm ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "For PC6 or PC7 ships with a vertical-sided bow, each normalFrameAngle must have a decimal numeric value between 0 and 10 degrees inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:verticalSidedBowForm true ;
                      nltl:normalFrameAngle ?angle .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                ?angle qudt:numericValue ?value .
                FILTER (
                    datatype(?value) != xsd:decimal ||
                    ?value < "0.0"^^xsd:decimal ||
                    ?value > "10.0"^^xsd:decimal
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For PC6 or PC7 ships with a vertical-sided bow, each normalFrameAngle must provide a decimal numeric value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:verticalSidedBowForm true ;
                      nltl:normalFrameAngle ?angle .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                FILTER NOT EXISTS {
                    ?angle qudt:numericValue ?value .
                    FILTER (datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For PC6 or PC7 ships with a vertical-sided bow, each normalFrameAngle must use DEG." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:verticalSidedBowForm true ;
                      nltl:normalFrameAngle ?angle .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                FILTER NOT EXISTS {
                    ?angle qudt:unit <http://qudt.org/vocab/unit/DEG> .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "For PC6 or PC7 ships with a vertical-sided bow, normalFrameAngle must not use a unit other than DEG." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass ;
                      nltl:verticalSidedBowForm true ;
                      nltl:normalFrameAngle ?angle .
                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                ?angle qudt:unit ?unit .
                FILTER (?unit != <http://qudt.org/vocab/unit/DEG>)
            }
        """
    ] .
<END_SHACL>
