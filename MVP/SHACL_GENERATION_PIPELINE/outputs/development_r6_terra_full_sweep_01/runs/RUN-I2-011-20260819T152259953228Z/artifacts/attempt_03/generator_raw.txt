<urn:nltl:generated-shape:I2-011BowDesignForceShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc1
                    nltl:polarClassPc2
                    nltl:polarClassPc3
                    nltl:polarClassPc4
                    nltl:polarClassPc5
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:polarClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:polarClassPc6
                    nltl:polarClassPc7
                )
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:MegaN
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:MegaN
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For a PC6 or PC7 ship with a bulbous bow, the I2.3.2.1(iv) bow design force must not be less than the I2.3.2.1(iii) bow design force." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this a nltl:ship ;
                    nltl:bulbousBowPresent true ;
                    nltl:polarClass ?polarClass ;
                    nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?forceIii ;
                    nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?forceIv .

                ?forceIii qudt:numericValue ?valueIii .
                ?forceIv qudt:numericValue ?valueIv .

                FILTER (?polarClass IN (nltl:polarClassPc6, nltl:polarClassPc7))
                FILTER (DATATYPE(?valueIii) = xsd:decimal)
                FILTER (DATATYPE(?valueIv) = xsd:decimal)
                FILTER (?valueIv < ?valueIii)
            }
        """
    ] .
