        sh:message "The four bow subregion cases must have distinct midpoint positions at LBOW/8, 3*LBOW/8, 5*LBOW/8, and 7*LBOW/8 within a relative tolerance of 1e-6 and an absolute fallback of 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?lengthQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                FILTER NOT EXISTS {
                    $this nltl:hasBowSubregionCalculationCase ?case1 ;
                          nltl:hasBowSubregionCalculationCase ?case2 ;
                          nltl:hasBowSubregionCalculationCase ?case3 ;
                          nltl:hasBowSubregionCalculationCase ?case4 .
                    FILTER (
                        ?case1 != ?case2 &&
                        ?case1 != ?case3 &&
                        ?case1 != ?case4 &&
                        ?case2 != ?case3 &&
                        ?case2 != ?case4 &&
                        ?case3 != ?case4

                    ?case2 nltl:bowSubregionMidLengthPosition ?positionQuantity2 .
                    ?positionQuantity2 qudt:numericValue ?position2 .

                    ?positionQuantity3 qudt:numericValue ?position3 .
                    ?case4 nltl:bowSubregionMidLengthPosition ?positionQuantity4 .
                    ?positionQuantity4 qudt:numericValue ?position4 .
                    BIND((?length / 8) AS ?expected1)
                    BIND((3 * ?length / 8) AS ?expected2)
                    BIND((5 * ?length / 8) AS ?expected3)
                    BIND((7 * ?length / 8) AS ?expected4)
