            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:peakPressureFactor ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:class nltl:plating ;
            sh:property [
                sh:path nltl:framingAngleOmega ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:platingSupportedByStructuralMember ;
                sh:node [
                    sh:class nltl:structuralMember ;
                    sh:property [
                        sh:path nltl:frameSpacing ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:selectedHullAreaFactor ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each linked plating needs framing-angle and hull-area classification information to determine I2.4.2 applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this ?plating
            WHERE {
                $this nltl:hasPlating ?plating .
                FILTER NOT EXISTS {
                    ?plating nltl:framingAngleOmega ?omega ;
                             nltl:platingHullAreaClassification ?area .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Applicable plating requires a linked supporting structural member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this ?plating
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:platingHullAreaClassification ?area ;
                         nltl:framingAngleOmega ?omegaQuantity .
                ?omegaQuantity qudt:numericValue ?omega ;
                               qudt:unit unit:DEG .
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
                FILTER (?omega >= 70)
                FILTER NOT EXISTS {
                    ?plating nltl:platingSupportedByStructuralMember ?member .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Applicable plating and its supporting structural member require all I2.4.2 formula inputs and the reported net thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this ?plating ?member
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:platingHullAreaClassification ?area ;
                         nltl:framingAngleOmega ?omegaQuantity ;
                         nltl:platingSupportedByStructuralMember ?member .
                ?omegaQuantity qudt:numericValue ?omega ;
                               qudt:unit unit:DEG .
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
                FILTER (?omega >= 70)
                FILTER NOT EXISTS {
                    $this nltl:averageIcePressure ?pressureQuantity ;
                          nltl:peakPressureFactor ?peakQuantity ;
                          nltl:yieldStrength ?yieldQuantity ;
                          nltl:loadPatchHeight ?heightQuantity .
                    ?pressureQuantity qudt:numericValue ?pressure ;
                                      qudt:unit unit:MegaPA .
                    ?peakQuantity qudt:numericValue ?peak ;
                                  qudt:unit unit:UNITLESS .
                    ?yieldQuantity qudt:numericValue ?yield ;
                                   qudt:unit unit:MegaPA .
                    ?heightQuantity qudt:numericValue ?height ;
                                    qudt:unit unit:M .
                    ?member nltl:frameSpacing ?spacingQuantity ;
                            nltl:selectedHullAreaFactor ?areaFactorQuantity .
                    ?spacingQuantity qudt:numericValue ?spacing ;
                                     qudt:unit unit:M .
                    ?areaFactorQuantity qudt:numericValue ?areaFactor ;
                                        qudt:unit unit:UNITLESS .
                    ?plating nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
                    ?thicknessQuantity qudt:numericValue ?thickness ;
                                       qudt:unit unit:MilliM .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ice-load required net plate thickness must match the I2.4.2 formula within relative tolerance 1e-6 and absolute tolerance 1e-9 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this ?plating ?member
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:averageIcePressure ?pressureQuantity ;
                      nltl:peakPressureFactor ?peakQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:loadPatchHeight ?heightQuantity .
                ?plating nltl:platingHullAreaClassification ?area ;
                         nltl:framingAngleOmega ?omegaQuantity ;
                         nltl:platingSupportedByStructuralMember ?member ;
                         nltl:iceLoadRequiredNetPlateThickness ?thicknessQuantity .
                ?omegaQuantity qudt:numericValue ?omega ;
                               qudt:unit unit:DEG .
                ?pressureQuantity qudt:numericValue ?pressure ;
                                  qudt:unit unit:MegaPA .
                ?peakQuantity qudt:numericValue ?peak ;
                              qudt:unit unit:UNITLESS .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit unit:MegaPA .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit unit:M .
                ?member nltl:frameSpacing ?spacingQuantity ;
                        nltl:selectedHullAreaFactor ?areaFactorQuantity .
                ?spacingQuantity qudt:numericValue ?spacing ;
                                 qudt:unit unit:M .
                ?areaFactorQuantity qudt:numericValue ?areaFactor ;
                                    qudt:unit unit:UNITLESS .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                                   qudt:unit unit:MilliM .
                FILTER (?area IN (
                    nltl:bowIntermediateBottomHullArea,
                    nltl:midbodyBottomHullArea,
                    nltl:sternBottomHullArea
                ))
                FILTER (?omega >= 70)
                BIND(
                    (500 * ?spacing * SQRT((?areaFactor * ?peak * ?pressure) / ?yield))
                    / (1 + (?spacing / (2 * ?height)))
                    AS ?expected
                )
                FILTER (
                    ABS(?thickness - ?expected) >
                    (0.000000001 + (0.000001 * ABS(?expected)))
                )
            }
        """
    ] .
