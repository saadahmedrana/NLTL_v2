<urn:nltl:generated-shape:I2_023ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:or (
                <urn:nltl:generated-shape:NonLongitudinalPlating>
                <urn:nltl:generated-shape:LongitudinalPlating>
            )
        ]
    ] ;
    sh:sparql [
        sh:message "For longitudinally-framed plating, reported net thickness must match the I2.4.2 formula within relative tolerance 1e-6 with a small absolute fallback." ;
        sh:select """
            SELECT $this
            WHERE {
                $this <https://w3id.org/nltl/vocab#hasPlating> ?plating ;
                      <https://w3id.org/nltl/vocab#averageIcePressure> ?pAvgQ ;
                      <https://w3id.org/nltl/vocab#frameSpan> ?spanQ ;
                      <https://w3id.org/nltl/vocab#loadPatchHeight> ?bQ ;
                      <https://w3id.org/nltl/vocab#peakPressureFactor> ?ppfQ ;
                      <https://w3id.org/nltl/vocab#yieldStrength> ?yieldQ .
                ?plating <https://w3id.org/nltl/vocab#framingAngleOmega> ?omegaQ ;
                         <https://w3id.org/nltl/vocab#iceLoadRequiredNetPlateThickness> ?tQ ;
                         <https://w3id.org/nltl/vocab#platingSupportedByStructuralMember> ?member .
                ?member <https://w3id.org/nltl/vocab#frameSpacing> ?sQ ;
                        <https://w3id.org/nltl/vocab#selectedHullAreaFactor> ?afQ .
                ?omegaQ <http://qudt.org/schema/qudt/numericValue> ?omega ;
                        <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/DEG> .
                ?tQ <http://qudt.org/schema/qudt/numericValue> ?t ;
                    <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/MilliM> .
                ?sQ <http://qudt.org/schema/qudt/numericValue> ?s ;
                    <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?afQ <http://qudt.org/schema/qudt/numericValue> ?af ;
                     <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/UNITLESS> .
                ?pAvgQ <http://qudt.org/schema/qudt/numericValue> ?pAvg ;
                       <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/MegaPA> .
                ?spanQ <http://qudt.org/schema/qudt/numericValue> ?l ;
                       <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?bQ <http://qudt.org/schema/qudt/numericValue> ?b ;
                    <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/M> .
                ?ppfQ <http://qudt.org/schema/qudt/numericValue> ?ppf ;
                      <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/UNITLESS> .
                ?yieldQ <http://qudt.org/schema/qudt/numericValue> ?yield ;
                        <http://qudt.org/schema/qudt/unit> <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (?omega <= 20)
                FILTER (?s != 0 && ?l != 0 && ?yield != 0)
                BIND ((1 + (?s / (2 * ?l))) AS ?denominator)
                FILTER (?denominator != 0)
                BIND ((?af * ?ppf * ?pAvg) / ?yield AS ?base)
                FILTER (?base >= 0)
                BIND (IF(?b >= ?s, 1.0, ((2 * ?b / ?s) - ((?b / ?s) * (?b / ?s)))) AS ?geometry)
                FILTER (?geometry >= 0)
                BIND ((250000 * ?s * ?s * ?base * ?geometry) / (?denominator * ?denominator) AS ?expectedSquared)
                BIND (?t * ?t AS ?reportedSquared)
                FILTER (
                    ?t < 0 ||
                    (?reportedSquared - ?expectedSquared) > ((0.000002 * ?expectedSquared) + 0.000000000001) ||
                    (?expectedSquared - ?reportedSquared) > ((0.000002 * ?expectedSquared) + 0.000000000001)
                )
            }
        """
    ] .
