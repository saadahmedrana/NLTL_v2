<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_023_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:node [
            sh:property [
                sh:path nltl:framingAngleOmega ;
                sh:minCount 1 ;
                sh:message "Each linked plating requires a framing angle to establish applicability."
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Longitudinally-framed plating requires complete, valid formula inputs and relationships before thickness conformance can be established." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (datatype(?omega) = xsd:decimal && ?omega <= 20)
                FILTER NOT EXISTS {
                    $this nltl:averageIcePressure ?pressureQ ;
                          nltl:hullAreaFactor ?hullAfQ ;
                          nltl:loadPatchHeight ?bQ ;
                          nltl:peakPressureFactor ?ppfQ ;
                          nltl:frameSpan ?lQ ;
                          nltl:yieldStrength ?yieldQ .
                    ?plating nltl:iceLoadRequiredNetPlateThickness ?thicknessQ ;
                             nltl:platingSupportedByStructuralMember ?member .
                    ?member nltl:frameSpacing ?sQ ;
                            nltl:selectedHullAreaFactor ?selectedAfQ .
                    ?pressureQ qudt:numericValue ?pressure ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?hullAfQ qudt:numericValue ?hullAf ;
                             qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?bQ qudt:numericValue ?b ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?ppfQ qudt:numericValue ?ppf ;
                          qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?lQ qudt:numericValue ?l ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?yieldQ qudt:numericValue ?yield ;
                            qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                    ?thicknessQ qudt:numericValue ?thickness ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?sQ qudt:numericValue ?s ;
                        qudt:unit <http://qudt.org/vocab/unit/M> .
                    ?selectedAfQ qudt:numericValue ?selectedAf ;
                                 qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    FILTER (
                        datatype(?pressure) = xsd:decimal &&
                        datatype(?hullAf) = xsd:decimal &&
                        datatype(?b) = xsd:decimal &&
                        datatype(?ppf) = xsd:decimal &&
                        datatype(?l) = xsd:decimal &&
                        datatype(?yield) = xsd:decimal &&
                        datatype(?thickness) = xsd:decimal &&
                        datatype(?s) = xsd:decimal &&
                        datatype(?selectedAf) = xsd:decimal &&
                        ?pressure >= 0 &&
                        ?hullAf >= 0 &&
                        ?b >= 0 &&
                        ?ppf >= 0 &&
                        ?l > 0 &&
                        ?yield > 0 &&
                        ?s > 0 &&
                        ?selectedAf >= 0
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The selected hull area factor of the supporting structural member must correspond to the ship hull area factor used for this plating calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:hullAreaFactor ?hullAfQ .
                ?plating nltl:framingAngleOmega ?omegaQ ;
                         nltl:platingSupportedByStructuralMember ?member .
                ?member nltl:selectedHullAreaFactor ?selectedAfQ .
                ?omegaQ qudt:numericValue ?omega .
                ?hullAfQ qudt:numericValue ?hullAf .
                ?selectedAfQ qudt:numericValue ?selectedAf .
                FILTER (
                    datatype(?omega) = xsd:decimal &&
                    datatype(?hullAf) = xsd:decimal &&
                    datatype(?selectedAf) = xsd:decimal &&
                    ?omega <= 20 &&
                    ?selectedAf != ?hullAf
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ice load required net plate thickness must match the applicable I2.4.2 formula within a relative tolerance of 1e-6 plus 0.000001 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:averageIcePressure ?pressureQ ;
                      nltl:loadPatchHeight ?bQ ;
                      nltl:peakPressureFactor ?ppfQ ;
                      nltl:frameSpan ?lQ ;
                      nltl:yieldStrength ?yieldQ .
                ?plating nltl:framingAngleOmega ?omegaQ ;
                         nltl:iceLoadRequiredNetPlateThickness ?thicknessQ ;
                         nltl:platingSupportedByStructuralMember ?member .
                ?member nltl:frameSpacing ?sQ ;
                        nltl:selectedHullAreaFactor ?selectedAfQ .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                ?pressureQ qudt:numericValue ?pressure ;
                           qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?bQ qudt:numericValue ?b ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                ?ppfQ qudt:numericValue ?ppf ;
                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?lQ qudt:numericValue ?l ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                ?yieldQ qudt:numericValue ?yield ;
                        qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?thicknessQ qudt:numericValue ?thickness ;
                            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?sQ qudt:numericValue ?s ;
                    qudt:unit <http://qudt.org/vocab/unit/M> .
                ?selectedAfQ qudt:numericValue ?selectedAf ;
                             qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                FILTER (
                    datatype(?omega) = xsd:decimal &&
                    datatype(?pressure) = xsd:decimal &&
                    datatype(?b) = xsd:decimal &&
                    datatype(?ppf) = xsd:decimal &&
                    datatype(?l) = xsd:decimal &&
                    datatype(?yield) = xsd:decimal &&
                    datatype(?thickness) = xsd:decimal &&
                    datatype(?s) = xsd:decimal &&
                    datatype(?selectedAf) = xsd:decimal &&
                    ?omega <= 20 &&
                    ?pressure >= 0 &&
                    ?b >= 0 &&
                    ?ppf >= 0 &&
                    ?l > 0 &&
                    ?yield > 0 &&
                    ?s > 0 &&
                    ?selectedAf >= 0
                )
                BIND((?b / ?s) AS ?bOverS)
                BIND(
                    IF(
                        ?b >= ?s,
                        1.0,
                        SQRT((2.0 * ?bOverS) - (?bOverS * ?bOverS))
                    )
                    AS ?geometryFactor
                )
                BIND(
                    (
                        500.0 * ?s *
                        SQRT((?selectedAf * ?ppf * ?pressure) / ?yield) *
                        ?geometryFactor /
                        (1.0 + (?s / (2.0 * ?l)))
                    )
                    AS ?expectedThickness
                )
                FILTER (
                    ABS(?thickness - ?expectedThickness) >
                    ((0.000001 * ABS(?expectedThickness)) + 0.000001)
                )
            }
        """
    ] .
