@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_026ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_026StructuralMemberShape
    ] .

gs:I2_026StructuralMemberShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:minCount 1 ;
        sh:in ( nltl:fixedFrameSupport nltl:simpleFrameSupport ) ;
        sh:message "A framing member must record its frame support condition as fixed or simple support."
    ] ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesWithinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:node gs:I2_026MemberSupportShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:property [
                sh:path nltl:continuousThroughSupport ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:qualifiedMinCount 1 ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:connectionBracketPresent ;
                        sh:hasValue true
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:property [
                sh:path nltl:hasMemberSupport ;
                sh:qualifiedMinCount 1 ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:demonstratedRotationalRestraint ;
                        sh:hasValue true
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:significantRotationalRestraintEvidence ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:fixedFrameSupport
            ] ;
            sh:property [
                sh:path nltl:terminatesWithinIceStrengthenedArea ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameSupportCondition ;
                sh:hasValue nltl:simpleFrameSupport
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Fixed frame support requires continuity, a connection bracket, demonstrated rotational restraint with evidence, or termination within an ice-strengthened area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition nltl:fixedFrameSupport .
                FILTER (
                    !EXISTS {
                        $this nltl:continuousThroughSupport true .
                    }
                    && !EXISTS {
                        $this nltl:terminatesWithinIceStrengthenedArea true .
                    }
                    && !EXISTS {
                        $this nltl:hasMemberSupport ?support .
                        ?support nltl:connectionBracketPresent true .
                    }
                    && !EXISTS {
                        $this nltl:hasMemberSupport ?support .
                        ?support nltl:demonstratedRotationalRestraint true .
                        $this nltl:significantRotationalRestraintEvidence ?evidence .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Simple frame support is not applicable where a fixed-support basis is established." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:frameSupportCondition nltl:simpleFrameSupport .
                FILTER (
                    EXISTS {
                        $this nltl:continuousThroughSupport true .
                    }
                    || EXISTS {
                        $this nltl:terminatesWithinIceStrengthenedArea true .
                    }
                    || EXISTS {
                        $this nltl:hasMemberSupport ?support .
                        ?support nltl:connectionBracketPresent true .
                    }
                    || EXISTS {
                        $this nltl:hasMemberSupport ?support .
                        ?support nltl:demonstratedRotationalRestraint true .
                        $this nltl:significantRotationalRestraintEvidence ?evidence .
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Demonstrated rotational restraint requires significant rotational restraint evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """
    ] .

gs:I2_026MemberSupportShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
