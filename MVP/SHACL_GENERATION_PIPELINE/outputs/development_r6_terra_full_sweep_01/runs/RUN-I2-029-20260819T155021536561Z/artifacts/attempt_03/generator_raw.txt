gs:I2_029Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasCalculationCase ;
                sh:minCount 1 ;
                sh:node gs:CalculationCaseShape
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:minCount 1 ;
                sh:node gs:StructuralMemberShape
            ] ;
            sh:property [
                sh:path nltl:sectionModulus ;
                sh:minCount 1 ;
                sh:node gs:CubicCentimetreQuantityShape ;
                sh:message "A section modulus is required for the calculation."
            ] ;
            sh:property [
                sh:path nltl:shearArea ;
                sh:minCount 1 ;
                sh:node gs:SquareMetreQuantityShape ;
                sh:message "A shear area is required for the calculation."
            ] ;
            sh:property [
                sh:path nltl:thickness ;
                sh:minCount 1 ;
                sh:node gs:MetreQuantityShape ;
                sh:message "A thickness operand is required for the calculation."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Where a linked structural member has a fitted flange, each calculation case must explicitly select whether flange material is included in the shear area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this ?case
            WHERE {
                $this nltl:hasCalculationCase ?case .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:flangeFitted true .
                FILTER NOT EXISTS {
                    ?case nltl:flangeMaterialIncludedInShearArea ?selection .
                    FILTER(?selection IN (true, false))
                }
            }
        """
    ] .
