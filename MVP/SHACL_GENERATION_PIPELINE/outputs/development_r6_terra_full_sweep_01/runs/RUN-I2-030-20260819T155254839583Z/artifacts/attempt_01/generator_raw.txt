    sh:sparql [
        sh:message "Local-frame area, attached-plate thickness, and frame spacing are required to determine applicability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:sectionCalculationCaseStructuralMember ?member .
                    ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                            nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                            nltl:frameSpacing ?spacingQ .
                    ?areaQ qudt:numericValue ?area ;
                           qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                    ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                     qudt:unit <http://qudt.org/vocab/unit/MilliM> .

    sh:sparql [
        sh:message "Applicable local-frame section calculations require all formula operands and both reported results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ .
                ?areaQ qudt:numericValue ?area .
                ?plateThicknessQ qudt:numericValue ?plateThickness .
                ?spacingQ qudt:numericValue ?spacing .
                FILTER (100 * ?area > 1000 * ?plateThickness * ?spacing)
                FILTER NOT EXISTS {
                    $this nltl:plasticNeutralAxisHeight ?znaQ .
                    ?znaQ qudt:numericValue ?zna ;
                          qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?member nltl:webHeight ?webHeightQ ;
                            nltl:netWebThickness ?webThicknessQ ;
                            nltl:localFrameFlangeCentreHeight ?flangeHeightQ ;
                            nltl:webAngleToShellPlate ?webAngleQ ;
                            nltl:webToFlangeCentreDistance ?distanceQ ;
                            nltl:netEffectivePlasticSectionModulus ?modulusQ .
                    ?webHeightQ qudt:numericValue ?webHeight ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?webThicknessQ qudt:numericValue ?webThickness ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?flangeHeightQ qudt:numericValue ?flangeHeight ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?webAngleQ qudt:numericValue ?webAngle ;
                               qudt:unit <http://qudt.org/vocab/unit/DEG> .
                    ?distanceQ qudt:numericValue ?distance ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?modulusQ qudt:numericValue ?modulus ;
                              qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Oblique framing requires two interpolation endpoints with coordinates and results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (?omega > 20 && ?omega < 70)
                FILTER NOT EXISTS {
                    $this nltl:interpolationLowerEndpoint ?lower ;
                          nltl:interpolationUpperEndpoint ?upper .
                    ?lower nltl:interpolationPointCoordinate ?lowerCoordinateQ ;
                           nltl:interpolationPointResult ?lowerResultQ .
                    ?upper nltl:interpolationPointCoordinate ?upperCoordinateQ ;
                           nltl:interpolationPointResult ?upperResultQ .
                    ?lowerCoordinateQ qudt:numericValue ?lowerCoordinate ;
                                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?upperCoordinateQ qudt:numericValue ?upperCoordinate ;
                                      qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                    ?lowerResultQ qudt:numericValue ?lowerResult .
                    ?upperResultQ qudt:numericValue ?upperResult .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Plastic neutral axis height must satisfy the applicable local-frame equation within a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:plasticNeutralAxisHeight ?znaQ .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:webHeight ?webHeightQ ;
                        nltl:netWebThickness ?webThicknessQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                ?webHeightQ qudt:numericValue ?webHeight ;
                            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?webThicknessQ qudt:numericValue ?webThickness ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                ?znaQ qudt:numericValue ?zna ;
                      qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (100 * ?area > 1000 * ?plateThickness * ?spacing)
                BIND ((100 * ?area + ?webHeight * ?webThickness - 1000 * ?plateThickness * ?spacing) / (2 * ?webThickness) AS ?expected)
                FILTER (ABS(?zna - ?expected) > (ABS(?expected) * 0.000001 + 0.000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "For non-oblique framing, net effective plastic section modulus must satisfy the local-frame equation within a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating ;
                      nltl:plasticNeutralAxisHeight ?znaQ .
                ?member nltl:netLocalFrameFlangeArea ?areaQ ;
                        nltl:webHeight ?webHeightQ ;
                        nltl:netWebThickness ?webThicknessQ ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQ ;
                        nltl:frameSpacing ?spacingQ ;
                        nltl:localFrameFlangeCentreHeight ?flangeHeightQ ;
                        nltl:webAngleToShellPlate ?webAngleQ ;
                        nltl:webToFlangeCentreDistance ?distanceQ ;
                        nltl:netEffectivePlasticSectionModulus ?modulusQ .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?areaQ qudt:numericValue ?area ;
                       qudt:unit <http://qudt.org/vocab/unit/CentiM2> .
                ?webHeightQ qudt:numericValue ?webHeight ;
                            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?webThicknessQ qudt:numericValue ?webThickness ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?plateThicknessQ qudt:numericValue ?plateThickness ;
                                 qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?spacingQ qudt:numericValue ?spacing ;
                          qudt:unit <http://qudt.org/vocab/unit/M> .
                ?flangeHeightQ qudt:numericValue ?flangeHeight ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?webAngleQ qudt:numericValue ?webAngle ;
                           qudt:unit <http://qudt.org/vocab/unit/DEG> .
                ?distanceQ qudt:numericValue ?distance ;
                           qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?znaQ qudt:numericValue ?zna ;
                      qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?modulusQ qudt:numericValue ?modulus ;
                          qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                FILTER (100 * ?area > 1000 * ?plateThickness * ?spacing)
                FILTER (!(?omega > 20 && ?omega < 70))
                BIND (?webAngle * 3.141592653589793 / 180 AS ?webAngleRadians)
                BIND (
                    ?plateThickness * ?spacing * (?zna + ?plateThickness / 2) * math:sin(?webAngleRadians) +
                    (((?webHeight - ?zna) * (?webHeight - ?zna) + ?zna * ?zna) * ?webThickness * math:sin(?webAngleRadians)) / 2000 +
                    ?area * ((?flangeHeight - ?zna) * math:sin(?webAngleRadians) - ?distance * math:cos(?webAngleRadians)) / 10
                    AS ?expected
                )
                FILTER (ABS(?modulus - ?expected) > (ABS(?expected) * 0.000001 + 0.000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "For oblique framing, net effective plastic section modulus must be linearly interpolated between the declared endpoints within a relative tolerance of 1e-6." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member ;
                      nltl:sectionCalculationCasePlating ?plating ;
                      nltl:interpolationLowerEndpoint ?lower ;
                      nltl:interpolationUpperEndpoint ?upper .
                ?member nltl:netEffectivePlasticSectionModulus ?modulusQ .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?lower nltl:interpolationPointCoordinate ?lowerCoordinateQ ;
                       nltl:interpolationPointResult ?lowerResultQ .
                ?upper nltl:interpolationPointCoordinate ?upperCoordinateQ ;
                       nltl:interpolationPointResult ?upperResultQ .
                ?modulusQ qudt:numericValue ?modulus ;
                          qudt:unit <http://qudt.org/vocab/unit/CentiM3> .
                ?omegaQ qudt:numericValue ?omega ;
                        qudt:unit <http://qudt.org/vocab/unit/DEG> .
                ?lowerCoordinateQ qudt:numericValue ?lowerCoordinate ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?upperCoordinateQ qudt:numericValue ?upperCoordinate ;
                                  qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                ?lowerResultQ qudt:numericValue ?lowerResult .
                ?upperResultQ qudt:numericValue ?upperResult .
                FILTER (?omega > 20 && ?omega < 70)
                FILTER (?upperCoordinate != ?lowerCoordinate)
                BIND (?lowerResult + ((?omega - ?lowerCoordinate) * (?upperResult - ?lowerResult) / (?upperCoordinate - ?lowerCoordinate)) AS ?expected)
                FILTER (ABS(?modulus - ?expected) > (ABS(?expected) * 0.000001 + 0.000001))
            }
        """
    ] .
