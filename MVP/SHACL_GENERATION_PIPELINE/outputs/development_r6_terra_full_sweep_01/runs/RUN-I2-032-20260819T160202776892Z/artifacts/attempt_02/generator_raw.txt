gs:I2_032_ShearAreaShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:StructuralMemberI2032Shape
    ] ;
    sh:sparql [
        sh:message "requiredTransverseFrameShearArea must equal the I2.6.2 formula result within relative tolerance 1e-6 and absolute tolerance 1e-6 cm2." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:averageIcePressure ?pAvgQ ;
                      nltl:loadPatchHeight ?loadHeightQ ;
                      nltl:peakPressureFactor ?ppfQ ;
                      nltl:span ?spanQ ;
                      nltl:yieldStrength ?yieldQ .
                ?member nltl:frameSpacing ?spacingQ ;
                        nltl:selectedHullAreaFactor ?afQ ;
                        nltl:requiredTransverseFrameShearArea ?atQ .
                ?pAvgQ qudt:numericValue ?pAvg .
                ?loadHeightQ qudt:numericValue ?loadHeight .
                ?ppfQ qudt:numericValue ?ppf .
                ?spanQ qudt:numericValue ?span .
                ?yieldQ qudt:numericValue ?yield .
                ?spacingQ qudt:numericValue ?spacing .
                ?afQ qudt:numericValue ?af .
                ?atQ qudt:numericValue ?at .
                BIND(
                    5000 * IF(?span < ?loadHeight, ?span, ?loadHeight) *
                    ?spacing * (?af * ?ppf * ?pAvg) / (0.577 * ?yield)
                    AS ?expectedAt
                )
                FILTER(ABS(?at - ?expectedAt) >
                    (ABS(?expectedAt) * 0.000001 + 0.000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Actual shearArea, converted from m2 to cm2, must be at least requiredTransverseFrameShearArea." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:shearArea ?awQ .
                ?member nltl:requiredTransverseFrameShearArea ?atQ .
                ?awQ qudt:numericValue ?aw .
                ?atQ qudt:numericValue ?at .
                FILTER((?aw * 10000) < ?at)
            }
        """
    ] .
