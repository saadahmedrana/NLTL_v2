<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_034ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_034StructuralMemberSelectorShape
    ] .

gs:I2_034StructuralMemberSelectorShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:longitudinalFrameOrientation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1
    ] .

gs:I2_034ApplicableFrameShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?ship nltl:hasStructuralMember ?this .
                ?this nltl:longitudinalFrameOrientation true ;
                      nltl:structuralLocation nltl:sideStructureLocation .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:node gs:I2_034StrengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:node gs:I2_034CollapseLoadQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node gs:I2_034LoadCaseShape
    ] ;
    sh:sparql [
        sh:message "Each represented load-case combined shear-and-bending demand shall not exceed the member plastic strength or midspan plastic-collapse load in the same unit." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strength ;
                      nltl:midspanPlasticCollapseLoad ?collapseLoad ;
                      nltl:hasStructuralMemberLoadCase ?loadCase .

                ?strength qudt:numericValue ?strengthValue ;
                          qudt:unit ?strengthUnit .

                ?collapseLoad qudt:numericValue ?collapseValue ;
                              qudt:unit ?collapseUnit .

                ?loadCase nltl:combinedShearAndBendingDemand ?demand .
                ?demand qudt:numericValue ?demandValue ;
                        qudt:unit ?demandUnit .

                FILTER (
                    ?demandUnit != ?strengthUnit ||
                    ?demandUnit != ?collapseUnit ||
                    ?demandValue > ?strengthValue ||
                    ?demandValue > ?collapseValue
                )
            }
        """
    ] .

gs:I2_034LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:node gs:I2_034DemandQuantityShape
    ] .

gs:I2_034DemandQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:I2_034StrengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

gs:I2_034CollapseLoadQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .
<END_SHACL>
