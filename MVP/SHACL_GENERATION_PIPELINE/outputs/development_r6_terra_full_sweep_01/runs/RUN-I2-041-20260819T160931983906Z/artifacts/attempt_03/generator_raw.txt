<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-041Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:netAttachedShellPlateThickness ;
                sh:minCount 1 ;
                sh:message "Net attached shell plate thickness is required." ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:message "Yield strength is required." ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MegaPA ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:netWebThickness ;
                sh:minCount 1 ;
                sh:message "Net web thickness is required." ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Net web thickness must be at least 0.35 times net attached shell plate thickness times the square root of yield strength divided by 235." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:netWebThickness ?webQuantity ;
                              nltl:netAttachedShellPlateThickness ?shellQuantity ;
                              nltl:yieldStrength ?yieldQuantity .

                        ?webQuantity qudt:numericValue ?web ;
                                     qudt:unit unit:MilliM .
                        ?shellQuantity qudt:numericValue ?shell ;
                                       qudt:unit unit:MilliM .
                        ?yieldQuantity qudt:numericValue ?yield ;
                                       qudt:unit unit:MegaPA .

                        FILTER (?web < (0.35 * ?shell * SQRT(?yield / 235)))
                    }
                """
            ]
        ]
    ] .
<END_SHACL>
