<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2_047ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:gaugedThickness ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:message "Gauged thickness is required as a quantity value." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:steelRenewalRequired ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Steel renewal must be recorded as required when gauged thickness is less than the ice-load required net plate thickness plus 0.5 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPlating ?plating ;
                      nltl:gaugedThickness ?gaugedQuantity .
                ?gaugedQuantity qudt:numericValue ?gaugedValue ;
                                qudt:unit <http://qudt.org/vocab/unit/M> .
                ?plating nltl:iceLoadRequiredNetPlateThickness ?tnetQuantity .
                ?tnetQuantity qudt:numericValue ?tnetValue ;
                              qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (
                    xsd:decimal(?gaugedValue) <
                    ((xsd:decimal(?tnetValue) * 0.001) + 0.0005)
                )
                FILTER NOT EXISTS {
                    $this nltl:steelRenewalRequired true .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:I2_047PlatingShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasPlating ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:message "Ice-load required net plate thickness is required as a quantity value." ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .
<END_SHACL>
