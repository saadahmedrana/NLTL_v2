<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-048ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:equals nltl:hasTableLookupCase ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node <urn:nltl:generated-shape:I2-048SteelGradeRequirementCaseShape>
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:class nltl:tableLookupCase ;
        sh:node <urn:nltl:generated-shape:I2-048TableLookupCaseShape>
    ] .

<urn:nltl:generated-shape:I2-048SteelGradeRequirementCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasSteelGradeRequirementCase
        ] ;
        sh:node <urn:nltl:generated-shape:I2-048ShipPolarClassShape>
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:I2-048PlateThicknessShape>
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        )
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        )
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:node <urn:nltl:generated-shape:I2-048SteelGradeValueShape>
    ] ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:node <urn:nltl:generated-shape:I2-048SteelGradeValueShape>
    ] ;
    sh:node <urn:nltl:generated-shape:I2-048TableLookupCaseShape> ;
    sh:sparql [
        sh:message "Actual hull structural steel grade must not be less than the Table 8 required grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:actualHullStructuralSteelGrade ?actualGrade ;
                      nltl:requiredHullStructuralSteelGrade ?requiredGrade .
                ?actualGrade nltl:steelGradeOrderRank ?actualRank .
                ?requiredGrade nltl:steelGradeOrderRank ?requiredRank .
                FILTER (?actualRank < ?requiredRank)
            }
        """
    ] .

            nltl:polarClassPc7
        )
    ] .

<urn:nltl:generated-shape:I2-048PlateThicknessShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
