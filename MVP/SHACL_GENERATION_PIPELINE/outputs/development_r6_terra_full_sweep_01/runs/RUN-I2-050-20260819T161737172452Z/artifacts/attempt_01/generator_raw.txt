            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this

                $this nltl:hasTableLookupCase ?case ;
                      nltl:iceStrengthCoefficientKI ?kiQ ;
                      nltl:stemAngle ?gammaQ ;
                      nltl:upperIceWaterlineDraughtDUI ?duiQ ;
                      nltl:hullFormCoefficientKh ?khQ ;

                ?duiQ qudt:numericValue ?dui .

                ?fib2Q qudt:numericValue ?fib2 .
                ?fibQ qudt:numericValue ?fib .

                BIND(math:sin(?gamma * 3.141592653589793 / 180.0) AS ?sinGamma)
                BIND(
                    0.534 *
                    math:pow(?ki, 0.15) *
                    math:pow(?sinGamma, 0.2) *
                    math:pow((?dui * ?kh), 0.5) *
                    ?cfl

                BIND(1.20 * ?cff AS ?expectedFib2)
                BIND(

                    AS ?expectedFib

                BIND(0.000000001 + (0.000001 * ABS(?expectedFib1)) AS ?fib1Tolerance)
                BIND(0.000000001 + (0.000001 * ABS(?expectedFib2)) AS ?fib2Tolerance)
                BIND(0.000000001 + (0.000001 * ABS(?expectedFib)) AS ?fibTolerance)

                FILTER(
                    ABS(?fib1 - ?expectedFib1) > ?fib1Tolerance ||
                    ABS(?fib2 - ?expectedFib2) > ?fib2Tolerance ||
                    ABS(?fib - ?expectedFib) > ?fibTolerance
                )
            }
        """
    ] .
