<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_052ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:calculationCase ;
        sh:node gs:I2_052CalculationCaseShape
    ] .

gs:I2_052CalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gs:I2_052LengthQuantityShape ;
        sh:message "A calculation case requires the longitudinal position from aft."
    ] ;
    sh:property [
        sh:path nltl:interpolatedLongitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:node gs:I2_052DimensionlessQuantityShape ;
        sh:message "A calculation case requires the interpolated longitudinal distribution factor."
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_052InterpolationPointShape
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_052InterpolationPointShape ;
        sh:message "A calculation case requires a lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:interpolationPoint ;
        sh:node gs:I2_052InterpolationPointShape ;
        sh:message "A calculation case requires an upper interpolation endpoint."
    ] ;
    sh:sparql [
        sh:message "The interpolation endpoints must bracket the calculation-case longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:interpolationLowerEndpoint ?lowerPoint ;
                      nltl:interpolationUpperEndpoint ?upperPoint .
                ?positionQuantity qudt:numericValue ?position .
                ?lowerPoint nltl:interpolationPointCoordinate ?lowerCoordinateQuantity .
                ?upperPoint nltl:interpolationPointCoordinate ?upperCoordinateQuantity .
                ?lowerCoordinateQuantity qudt:numericValue ?lowerCoordinate .
                ?upperCoordinateQuantity qudt:numericValue ?upperCoordinate .
                FILTER (!(?lowerCoordinate < ?position && ?position < ?upperCoordinate))
            }
        """
    ] ;
    sh:sparql [
        sh:message "The interpolated longitudinal distribution factor must equal the linear interpolation of the endpoint values within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:interpolatedLongitudinalDistributionFactor ?resultQuantity ;
                      nltl:interpolationLowerEndpoint ?lowerPoint ;
                      nltl:interpolationUpperEndpoint ?upperPoint .
                ?positionQuantity qudt:numericValue ?position .
                ?resultQuantity qudt:numericValue ?reportedResult .
                ?lowerPoint nltl:interpolationPointCoordinate ?lowerCoordinateQuantity ;
                            nltl:interpolationPointValue ?lowerValueQuantity .
                ?upperPoint nltl:interpolationPointCoordinate ?upperCoordinateQuantity ;
                            nltl:interpolationPointValue ?upperValueQuantity .
                ?lowerCoordinateQuantity qudt:numericValue ?lowerCoordinate .
                ?upperCoordinateQuantity qudt:numericValue ?upperCoordinate .
                ?lowerValueQuantity qudt:numericValue ?lowerValue .
                ?upperValueQuantity qudt:numericValue ?upperValue .
                FILTER (?lowerCoordinate < ?position && ?position < ?upperCoordinate)
                BIND(
                    ?lowerValue +
                    ((?position - ?lowerCoordinate) *
                     (?upperValue - ?lowerValue) /
                     (?upperCoordinate - ?lowerCoordinate))
                    AS ?expectedResult
                )
                FILTER (
                    ABS(?reportedResult - ?expectedResult) >
                    ((0.000001 * ABS(?expectedResult)) + 0.000000001)
                )
            }
        """
    ] .

gs:I2_052InterpolationPointShape
    a sh:NodeShape ;
    sh:class nltl:interpolationPoint ;
    sh:property [
        sh:path nltl:interpolationCoordinate ;
        sh:minCount 1 ;
        sh:node gs:I2_052DimensionlessQuantityShape ;
        sh:message "An interpolation point requires an interpolation coordinate."
    ] ;
    sh:property [
        sh:path nltl:interpolationPointValue ;
        sh:minCount 1 ;
        sh:node gs:I2_052NumericQuantityShape ;
        sh:message "An interpolation point requires a value."
    ] .

gs:I2_052LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gs:I2_052DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .

gs:I2_052NumericQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] .
<END_SHACL>
