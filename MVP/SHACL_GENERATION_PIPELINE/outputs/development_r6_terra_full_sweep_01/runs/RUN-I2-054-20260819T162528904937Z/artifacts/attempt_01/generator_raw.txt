gs:I2_054HullGirderCaseShape
    a sh:NodeShape ;
    sh:class nltl:hullGirderBendingMomentCalculationCase ;
    sh:property [
        sh:path nltl:caseHullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:minInclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:caseBendingMomentDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:caseDesignVerticalIceBendingMoment ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN-M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasHullGirderBendingMomentCalculationCase
        ] ;
        sh:node gs:I2_054HullGirderCaseShipInputsShape
    ] ;
    sh:sparql [
        sh:message "Hull-girder longitudinal position must not exceed L_UI from aft." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:caseHullGirderLongitudinalPositionFromAft ?positionQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?ship nltl:hasHullGirderBendingMomentCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?lengthQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                FILTER (?position > ?length)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Case bendingMomentDistributionFactor must follow the specified C_m distribution within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:caseHullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:caseBendingMomentDistributionFactor ?factorQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?factorQuantity qudt:numericValue ?factor .
                ?ship nltl:hasHullGirderBendingMomentCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?lengthQuantity .
                ?lengthQuantity qudt:numericValue ?length .
                FILTER (?length > 0)
                BIND(xsd:decimal(?position) / xsd:decimal(?length) AS ?coordinate)
                FILTER (?coordinate >= 0 && ?coordinate <= 1)
                BIND(
                    IF(
                        ?coordinate <= 0.5,
                        2 * ?coordinate,
                        IF(
                            ?coordinate <= 0.7,
                            1,
                            IF(
                                ?coordinate <= 0.95,
                                1 - 2.8 * (?coordinate - 0.7),
                                6 * (1 - ?coordinate)
                            )
                        )
                    )
                    AS ?expectedFactor
                )
                FILTER (
                    ABS(xsd:decimal(?factor) - ?expectedFactor) >
                    (0.000000001 + 0.000001 * ABS(?expectedFactor))
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Stem angle must provide a positive sine for the bending-moment calculation." ;
        sh:select """
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasHullGirderBendingMomentCalculationCase $this ;
                      nltl:stemAngle ?angleQuantity .
                ?angleQuantity qudt:numericValue ?angle .
                BIND(
                    math:sin(xsd:decimal(?angle) * 3.141592653589793 / 180)
                    AS ?sine
                )
                FILTER (?sine <= 0)
            }
        """
    ] .
