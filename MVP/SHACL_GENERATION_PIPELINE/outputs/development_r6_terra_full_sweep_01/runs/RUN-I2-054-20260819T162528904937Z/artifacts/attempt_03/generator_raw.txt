            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:node gs:InterpolationResultQuantityShape
    ] ;
    sh:sparql [
        sh:message "Interpolation-point result must follow the specified C_m linear distribution." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:interpolationPointCoordinate ?coordinate ;
                      nltl:interpolationPointResult ?result .
                ?coordinate qudt:numericValue ?x .
                ?result qudt:numericValue ?reported .
                BIND(
                    IF(?x <= 0.5,
                       2.0 * ?x,
                       IF(?x <= 0.7,
                          1.0,
                          IF(?x <= 0.95,
                             2.96 - (2.8 * ?x),
                             6.0 - (6.0 * ?x)
                          )
                       )
                    ) AS ?expected
                )
                FILTER(
                    ABS(?reported - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] .

gs:HullGirderBendingMomentCalculationCaseShape a sh:NodeShape ;
    sh:class nltl:hullGirderBendingMomentCalculationCase ;
    sh:property [
        sh:path nltl:caseHullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:caseBendingMomentDistributionFactor ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:minInclusive "0.0"^^xsd:decimal ;
                sh:maxInclusive "1.0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:caseDesignVerticalIceBendingMoment ;
        sh:minCount 1 ;
        sh:node gs:TorqueQuantityShape
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasHullGirderBendingMomentCalculationCase ]
            nltl:upperIceWaterlineLengthLUI
        ) ;
        sh:minCount 1 ;
        sh:node gs:LengthQuantityShape
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasHullGirderBendingMomentCalculationCase ]
            nltl:stemAngle
        ) ;
        sh:minCount 1 ;
        sh:node gs:AngleQuantityShape
    ] ;
    sh:property [
        sh:path (
            [ sh:inversePath nltl:hasHullGirderBendingMomentCalculationCase ]
            nltl:designVerticalIceForceAtBow
        ) ;
        sh:minCount 1 ;
        sh:node gs:ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "Case bending-moment distribution factor must follow the specified C_m distribution at the case position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                ?ship nltl:hasHullGirderBendingMomentCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?length .
                $this nltl:caseHullGirderLongitudinalPositionFromAft ?position ;
                      nltl:caseBendingMomentDistributionFactor ?factor .
                ?length qudt:numericValue ?lui .
                ?position qudt:numericValue ?positionValue .
                ?factor qudt:numericValue ?reported .
                FILTER(?lui > 0.0)
                BIND(?positionValue / ?lui AS ?x)
                BIND(
                    IF(?x <= 0.5,
                       2.0 * ?x,
                       IF(?x <= 0.7,
                          1.0,
                          IF(?x <= 0.95,
                             2.96 - (2.8 * ?x),
                             6.0 - (6.0 * ?x)
                          )
                       )
                    ) AS ?expected
                )
                FILTER(
                    ?x < 0.0 || ?x > 1.0 ||
                    ABS(?reported - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Case design vertical ice bending moment must conform to the specified M_I formula." ;
        sh:select """
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                ?ship nltl:hasHullGirderBendingMomentCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?length ;
                      nltl:stemAngle ?angle ;
                      nltl:designVerticalIceForceAtBow ?force .
                $this nltl:caseBendingMomentDistributionFactor ?factor ;
                      nltl:caseDesignVerticalIceBendingMoment ?moment .
                ?length qudt:numericValue ?lui .
                ?angle qudt:numericValue ?angleDegrees .
                ?force qudt:numericValue ?fib .
                ?factor qudt:numericValue ?cm .
                ?moment qudt:numericValue ?mi .
                BIND(math:sin(?angleDegrees * 0.017453292519943295) AS ?sine)
                BIND(0.1 * ?cm * ?lui * ?fib AS ?base)
                BIND(?mi * ?mi * ?mi * ?mi * ?mi * ?sine AS ?reportedFifthPower)
                BIND(?base * ?base * ?base * ?base * ?base AS ?expectedFifthPower)
                FILTER(
                    ?sine <= 0.0 ||
                    (?cm = 0.0 && ABS(?mi) > 0.000000001) ||
                    (?cm > 0.0 &&
                     ABS(?reportedFifthPower - ?expectedFifthPower) >
                     ((ABS(?expectedFifthPower) * 0.000001) + 0.000000001))
                )
            }
        """
    ] .
