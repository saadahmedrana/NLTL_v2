@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:I2_060_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:memberInstability ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:structuralStiffening ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "A member in way of a cut-out must have no member instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout ;
                        nltl:memberInWayOfCutout "true"^^xsd:boolean .
                FILTER NOT EXISTS {
                    $this nltl:memberInstability "false"^^xsd:boolean .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A member in way of a cut-out must not be recorded as unstable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:memberInstability ?instability .
                ?member nltl:hasCutout ?cutout ;
                        nltl:memberInWayOfCutout "true"^^xsd:boolean .
                FILTER ( ?instability = "true"^^xsd:boolean )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Structural stiffening must be recorded where it is necessary for a member in way of a cut-out." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasCutout ?cutout ;
                        nltl:stiffeningNecessary "true"^^xsd:boolean .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                }
            }
        """
    ] .
<END_SHACL>
