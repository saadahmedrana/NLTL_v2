<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-062DirectCalculationShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:directCalculationUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Direct calculations must be recorded as used when a load carrying stringer or web frame is part of a grillage system." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member a ?memberType ;
                    nltl:partOfGrillageSystem ?grillageSystem .
                FILTER (?memberType IN (nltl:loadCarryingStringer, nltl:webFrame))
                FILTER NOT EXISTS {
                    $this nltl:directCalculationUsed true .
                }
            }
        """
    ] .
<END_SHACL>
