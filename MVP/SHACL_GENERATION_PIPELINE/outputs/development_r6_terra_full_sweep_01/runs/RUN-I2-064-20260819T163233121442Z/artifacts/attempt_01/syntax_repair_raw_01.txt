<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_064_LinearStructuralCalculationShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasStructuralMember nltl:hasStructuralMemberWeb ) ;
                sh:maxCount 0
            ] ;
            sh:property [
                sh:path ( nltl:hasStructuralMember nltl:hasStructuralMemberFlange ) ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:MegaPA
                    ]
                ] ;
                sh:message "Yield strength in MPa is required for represented structural member web or flange stresses."
            ] ;
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:node [
                    sh:property [
                        sh:path nltl:hasStructuralMemberWeb ;
                        sh:node [
                            sh:property [
                                sh:path nltl:nominalShearStress ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minCount 1 ;
                                        sh:datatype xsd:decimal
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1 ;
                                        sh:hasValue unit:MegaPA
                                    ]
                                ] ;
                                sh:message "A represented member web requires nominal shear stress in MPa."
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:hasStructuralMemberFlange ;
                        sh:node [
                            sh:property [
                                sh:path nltl:nominalFlangeVonMisesStress ;
                                sh:minCount 1 ;
                                sh:node [
                                    sh:property [
                                        sh:path qudt:numericValue ;
                                        sh:minCount 1 ;
                                        sh:datatype xsd:decimal
                                    ] ;
                                    sh:property [
                                        sh:path qudt:unit ;
                                        sh:minCount 1 ;
                                        sh:maxCount 1 ;
                                        sh:hasValue unit:MegaPA
                                    ]
                                ] ;
                                sh:message "A represented member flange requires nominal von Mises stress in MPa."
                            ]
                        ]
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Nominal shear stress in a member web must be less than yield strength divided by sqrt(3)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?member nltl:hasStructuralMemberWeb ?web .
                ?web nltl:nominalShearStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (
                    ( ( ?stress >= 0 ) && ( ( ?yield <= 0 ) || ( ( 3 * ?stress * ?stress ) >= ( ?yield * ?yield ) ) ) )
                    ||
                    ( ( ?stress < 0 ) && ( ?yield < 0 ) && ( ( 3 * ?stress * ?stress ) <= ( ?yield * ?yield ) ) )
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Nominal von Mises stress in a member flange must be less than 1.15 times yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?member nltl:hasStructuralMemberFlange ?flange .
                ?flange nltl:nominalFlangeVonMisesStress ?stressQuantity .
                ?stressQuantity qudt:numericValue ?stress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER ( ?stress >= ( 1.15 * ?yield ) )
            }
        """
    ] .
<END_SHACL>
