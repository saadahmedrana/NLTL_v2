<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO002ShipCategoryShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:minCount 1 ;
        sh:datatype xsd:string ;
        sh:message "A design ice condition is required to determine the applicable Polar Ship Category definition."
    ] ;
    sh:sparql [
        sh:message "A ship designed for at least thin first-year ice, including any stated old-ice inclusions, and not classified as Category A must be classified as Category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?iceCondition .
                FILTER(datatype(?iceCondition) = xsd:string)
                FILTER(
                    REGEX(
                        ?iceCondition,
                        "^(thin|medium|thick) first-year ice( (with|including|which may include) (possible )?old[- ]ice inclusions)?$",
                        "i"
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryA .
                }
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryB .
                }
            }
        """
    ] .
<END_SHACL>
