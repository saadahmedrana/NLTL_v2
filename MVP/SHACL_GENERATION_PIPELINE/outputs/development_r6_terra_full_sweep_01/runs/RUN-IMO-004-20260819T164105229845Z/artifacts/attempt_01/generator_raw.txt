<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-004-IceTypeDefinitionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Sea ice with no more than one winter growth and thickness from 0.3 m to 2.0 m must be classified as first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                      nltl:winterGrowthCount ?growth ;
                      nltl:iceThickness ?thickness .
                ?thickness qudt:numericValue ?value ;
                           qudt:unit unit:M .
                FILTER(?growth <= 1)
                FILTER(?value >= 0.3 && ?value <= 2.0)
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ice type classification must be first-year ice for sea ice with no more than one winter growth and thickness from 0.3 m to 2.0 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                      nltl:winterGrowthCount ?growth ;
                      nltl:iceThickness ?thickness ;
                      nltl:iceTypeClassification ?reportedType .
                ?thickness qudt:numericValue ?value ;
                           qudt:unit unit:M .
                FILTER(?growth <= 1)
                FILTER(?value >= 0.3 && ?value <= 2.0)
                FILTER(?reportedType != nltl:firstYearIce)
            }
        """
    ] .
<END_SHACL>
