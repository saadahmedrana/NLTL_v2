<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-007-OpenWaterDefinitionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:sparql [
        sh:message "Open-water classification requires a sea-ice concentration value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:seaIceConcentration ?concentration .
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each open-water sea-ice concentration requires a numeric value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?concentration .
                FILTER NOT EXISTS {
                    ?concentration qudt:numericValue ?value .
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each open-water sea-ice concentration numeric value must be an xsd:decimal." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?concentration .
                ?concentration qudt:numericValue ?value .
                FILTER (datatype(?value) != xsd:decimal)
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each open-water sea-ice concentration must be less than 1." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?concentration .
                ?concentration qudt:numericValue ?value .
                FILTER (
                    datatype(?value) = xsd:decimal &&
                    ?value >= "1"^^xsd:decimal
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each open-water sea-ice concentration requires the unit UNITLESS." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?concentration .
                FILTER NOT EXISTS {
                    ?concentration qudt:unit ?unit .
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each open-water sea-ice concentration must use the unit UNITLESS." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:seaIceConcentration ?concentration .
                ?concentration qudt:unit ?unit .
                FILTER (?unit != <http://qudt.org/vocab/unit/UNITLESS>)
            }
        """
    ] ;

    sh:sparql [
        sh:message "Open-water classification requires ice-of-land-origin presence to be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER NOT EXISTS {
                    $this nltl:iceOfLandOriginPresent ?value .
                    FILTER (datatype(?value) = xsd:boolean && ?value = false)
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "Ice-of-land-origin presence for open water must be an xsd:boolean." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:iceOfLandOriginPresent ?value .
                FILTER (datatype(?value) != xsd:boolean)
            }
        """
    ] ;

    sh:sparql [
        sh:message "Ice of land origin must not be present for open water." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition ;
                      nltl:iceOfLandOriginPresent ?value .
                FILTER (datatype(?value) = xsd:boolean && ?value != false)
            }
        """
    ] .
<END_SHACL>
