<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-008-ThinFirstYearIceThicknessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Thin first-year ice requires every asserted ice thickness quantity to be a metre-valued decimal from 0.3 to 0.7 inclusive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceTypeClassification nltl:thinFirstYearIce .

                {
                    FILTER NOT EXISTS {
                        $this nltl:iceThickness ?thickness .
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    FILTER NOT EXISTS {
                        ?thickness a qudt:QuantityValue .
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    FILTER NOT EXISTS {
                        ?thickness qudt:unit unit:M .
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness ;
                               qudt:unit ?unitValue .
                    FILTER(?unitValue != unit:M)
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness .
                    FILTER NOT EXISTS {
                        ?thickness qudt:numericValue ?decimalValue .
                        FILTER(datatype(?decimalValue) = xsd:decimal)
                    }
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness ;
                               qudt:numericValue ?value .
                    FILTER(datatype(?value) != xsd:decimal)
                }
                UNION
                {
                    $this nltl:iceThickness ?thickness ;
                               qudt:numericValue ?value .
                    FILTER(
                        datatype(?value) = xsd:decimal &&
                        (?value < "0.3"^^xsd:decimal || ?value > "0.7"^^xsd:decimal)
                    )
                }
            }
        """
    ] .
<END_SHACL>
