<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO041Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Opening actuation type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:hydraulicFluidFreezingOrViscosityControlPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Hydraulically operated hatches or doors require freezing or excessive-viscosity control for hydraulic fluid." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:openingActuationType "hydraulic"^^xsd:string .
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent true .
                }
            }
        """
    ] .
<END_SHACL>
