<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO047Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:minCount 1 ;
                sh:class nltl:essentialEngine
            ] ;
            sh:property [
                sh:path nltl:combustionAirTemperatureControlPresent ;
                sh:minCount 1 ;
                sh:datatype xsd:boolean ;
                sh:in ( true )
            ] ;
            sh:property [
                sh:path nltl:combustionAirTemperature ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:DEG_C
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:manufacturerMinAirTemperature ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:DEG_C
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:manufacturerMaxAirTemperature ;
                sh:minCount 1 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minCount 1 ;
                        sh:datatype xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue unit:DEG_C
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Combustion air temperature must be within the manufacturer minimum and maximum air-temperature criteria." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>

                    SELECT $this
                    WHERE {
                        $this nltl:hasEssentialEngine ?engine ;
                              nltl:combustionAirTemperature ?temperatureValue ;
                              nltl:manufacturerMinAirTemperature ?minimumValue ;
                              nltl:manufacturerMaxAirTemperature ?maximumValue .
                        ?engine a nltl:essentialEngine .
                        ?temperatureValue qudt:numericValue ?temperature ;
                                          qudt:unit unit:DEG_C .
                        ?minimumValue qudt:numericValue ?minimum ;
                                      qudt:unit unit:DEG_C .
                        ?maximumValue qudt:numericValue ?maximum ;
                                      qudt:unit unit:DEG_C .
                        FILTER (?temperature < ?minimum || ?temperature > ?maximum)
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:maxCount 0
            ]
        ]
    ) .
<END_SHACL>
