<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-050-FireSafetyProtectionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fireSafetyComponentExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Fire-safety component exposure status must be recorded."
    ] ;
    sh:property [
        sh:path nltl:iceAndSnowProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Ice and snow protection must be present when a fire-safety component is exposed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:fireSafetyComponentExposed true .
                FILTER NOT EXISTS {
                    $this nltl:iceAndSnowProtectionPresent true .
                }
            }
        """
    ] .
<END_SHACL>
