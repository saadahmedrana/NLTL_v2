<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO_069_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:addedLifeSavingDeviceRequiresPower ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:devicePowerSourceIndependentOfMainPower ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "A power source for an added life-saving device requiring power must operate independently of the ship's main power source." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:addedLifeSavingDeviceRequiresPower true .
                FILTER NOT EXISTS {
                    $this nltl:devicePowerSourceIndependentOfMainPower true .
                }
            }
        """
    ] .
<END_SHACL>
