<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO072ExtendedDarknessSearchlightShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extendedDarknessOperation ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Extended darkness operation status is required to determine searchlight applicability."
    ] ;
    sh:sparql [
        sh:message "Each lifeboat must have an assigned searchlight approved for continuous use when extended darkness operation is true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:extendedDarknessOperation true ;
                      nltl:hasLifeboat ?lifeboat .
                FILTER NOT EXISTS {
                    ?lifeboat nltl:hasAssignedSearchlight ?searchlight .
                    ?searchlight nltl:searchlightContinuousUseSuitabilityStatus true .
                }
            }
        """
    ] .
<END_SHACL>
