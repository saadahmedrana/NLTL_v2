<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO077Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:survivalCraftAvailableCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:personsCapacityRequirement ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:additionalEquipmentCapacityRequirement ;
                sh:datatype xsd:integer ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:launchingApplianceCapacity ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloGM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:loadedSurvivalCraftRequirement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloGM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Survival-craft available capacity or launching-appliance capacity is insufficient for the required additional equipment." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:survivalCraftAvailableCapacity ?available ;
                              nltl:personsCapacityRequirement ?persons ;
                              nltl:additionalEquipmentCapacityRequirement ?additional ;
                              nltl:launchingApplianceCapacity ?launchingQuantity ;
                              nltl:loadedSurvivalCraftRequirement ?loadedQuantity .

                        ?launchingQuantity qudt:numericValue ?launching ;
                                          qudt:unit unit:KiloGM .
                        ?loadedQuantity qudt:numericValue ?loaded ;
                                        qudt:unit unit:KiloGM .

                        FILTER (
                            xsd:decimal(?available) <
                            (xsd:decimal(?persons) + xsd:decimal(?additional))
                            ||
                            xsd:decimal(?launching) < xsd:decimal(?loaded)
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:additionalEquipmentCarriedInSurvivalCraft ;
                sh:hasValue false
            ]
        ]
    ) .
<END_SHACL>
