<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO_081Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "A ship constructed on or after 2017-01-01 and ice strengthened must have either at least two independent echo-sounding devices, or at least one echo-sounding device with at least two independent transducers." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionDate ?date ;
                      nltl:shipIceStrengthened ?iceStrengthened .

                FILTER(
                    ?iceStrengthened = true &&
                    (
                        YEAR(?date) > 2017 ||
                        (YEAR(?date) = 2017 && MONTH(?date) > 1) ||
                        (YEAR(?date) = 2017 && MONTH(?date) = 1 && DAY(?date) >= 1)
                    )
                )

                FILTER NOT EXISTS {
                    {
                        $this nltl:independentEchoSoundingDeviceCount ?independentDeviceCount .
                        FILTER(xsd:integer(?independentDeviceCount) >= 2)
                    }
                    UNION
                    {
                        $this nltl:echoSoundingDeviceCount ?deviceCount ;
                              nltl:independentTransducerCount ?transducerCount .
                        FILTER(
                            xsd:integer(?deviceCount) >= 1 &&
                            xsd:integer(?transducerCount) >= 2
                        )
                    }
                }
            }
        """
    ] .
<END_SHACL>
