<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-085-BridgeWingProtectionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:bridgeWingsEnclosed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:bridgeWingProtectionDesignStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Category A or B ships constructed on or after 2017-01-01 must have enclosed bridge wings or an approved bridge-wing protection design." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category ;
                      nltl:constructionDate ?date .
                FILTER(?category IN (
                    nltl:polarShipCategoryA,
                    nltl:polarShipCategoryB
                ))
                FILTER(?date >= "2017-01-01"^^xsd:date)
                FILTER NOT EXISTS {
                    $this nltl:bridgeWingsEnclosed true .
                }
                FILTER NOT EXISTS {
                    $this nltl:bridgeWingProtectionDesignStatus true .
                }
            }
        """
    ] .
<END_SHACL>
