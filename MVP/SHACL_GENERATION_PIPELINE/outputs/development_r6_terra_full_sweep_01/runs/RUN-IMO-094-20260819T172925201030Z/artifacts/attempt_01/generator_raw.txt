<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-094-ShipCommunicationShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipProvidesIcebreakingEscort ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Icebreaking escort status must be recorded."
    ] ;
    sh:property [
        sh:path nltl:asternFacingSoundSignalingSystemPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:internationalCodeOfSignalsComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Ships providing icebreaking escort must have an astern-facing sound signaling system and comply with the International Code of Signals." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:shipProvidesIcebreakingEscort true .
                FILTER NOT EXISTS {
                    $this nltl:asternFacingSoundSignalingSystemPresent true ;
                          nltl:internationalCodeOfSignalsComplianceStatus true .
                }
            }
        """
    ] .
<END_SHACL>
