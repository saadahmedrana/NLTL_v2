<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:IMO094Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipProvidesIcebreakingEscort ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:asternFacingSoundSignalingSystemPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:internationalCodeOfSignalsComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When the ship provides icebreaking escort, the astern-facing sound signaling system and International Code of Signals compliance status must both be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipProvidesIcebreakingEscort true .
                {
                    FILTER NOT EXISTS {
                        $this nltl:asternFacingSoundSignalingSystemPresent true .
                    }
                }
                UNION
                {
                    $this nltl:asternFacingSoundSignalingSystemPresent ?asternValue .
                    FILTER(?asternValue != true)
                }
                UNION
                {
                    FILTER NOT EXISTS {
                        $this nltl:internationalCodeOfSignalsComplianceStatus true .
                    }
                }
                UNION
                {
                    $this nltl:internationalCodeOfSignalsComplianceStatus ?complianceValue .
                    FILTER(?complianceValue != true)
                }
            }
        """
    ] .
<END_SHACL>
