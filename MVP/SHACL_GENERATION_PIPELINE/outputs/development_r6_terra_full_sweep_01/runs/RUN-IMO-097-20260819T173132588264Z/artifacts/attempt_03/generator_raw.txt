    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasDistressAlertDevice ;
        sh:class nltl:equipmentItem
    ] .

<urn:nltl:generated-shape:IMO097SurvivalCraftRelationshipShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:survivalCraftHasShipToShoreDistressAlertDevice ;
        sh:class nltl:shipToShoreDistressAlertDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasLocationSignalDevice ;
        sh:class nltl:locationSignalDevice
    ] ;
    sh:property [
        sh:path nltl:survivalCraftHasOnSceneCommunicationDevice ;
        sh:class nltl:onSceneCommunicationDevice
    ] .

<urn:nltl:generated-shape:IMO097Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:class nltl:lifeboat ;
        sh:node <urn:nltl:generated-shape:IMO097LifeboatRelationshipShape>
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:class nltl:otherSurvivalCraft
    ] ;
    sh:property [
        sh:path nltl:hasRequiredRescueBoatOrLifeboat ;
        sh:class nltl:lifeboat
    ] ;
    sh:property [
        sh:path nltl:hasSurvivalCraft ;
        sh:class nltl:survivalCraft ;
        sh:node <urn:nltl:generated-shape:IMO097SurvivalCraftRelationshipShape>
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipOperatesInLowAirTemperature ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( false )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipOperatesInLowAirTemperature ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( true )
            ] ;
            sh:property [
                sh:path nltl:releasedRescueBoatOrLifeboat ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( false )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipOperatesInLowAirTemperature ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( true )
            ] ;
            sh:property [
                sh:path nltl:releasedRescueBoatOrLifeboat ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( true )
            ] ;
            sh:sparql [
                sh:message "Each required released rescue boat or lifeboat must be represented as a linked survival craft and carry all three required communication devices." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    SELECT $this
                    WHERE {
                        $this nltl:hasRequiredRescueBoatOrLifeboat ?craft .
                        FILTER NOT EXISTS {
                            $this nltl:hasSurvivalCraft ?craft .
                            ?craft nltl:survivalCraftHasShipToShoreDistressAlertDevice ?alertDevice .
                            ?craft nltl:survivalCraftHasLocationSignalDevice ?locationDevice .
                            ?craft nltl:survivalCraftHasOnSceneCommunicationDevice ?communicationDevice .
                        }
                    }
                """
            ]
        ]
    ) .
