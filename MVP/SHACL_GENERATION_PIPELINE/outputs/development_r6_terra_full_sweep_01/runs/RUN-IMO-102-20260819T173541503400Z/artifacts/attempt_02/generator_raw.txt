<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO102PolarTrainingShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceFreeIceCondition
            nltl:openWaterIceCondition
            nltl:otherWatersIceCondition
        )
    ] ;
    sh:property [
        sh:path nltl:polarTrainingShipTypeClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:tankerPolarTrainingShipType
            nltl:passengerShipPolarTrainingShipType
            nltl:otherShipPolarTrainingType
        )
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node gen:IMO102CrewMemberShape
    ] .

gen:IMO102CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarTrainingCrewRoleClassification ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:masterPolarTrainingRole
            nltl:chiefMatePolarTrainingRole
            nltl:officerInChargeOfNavigationalWatchPolarTrainingRole
        )
    ] ;
    sh:property [
        sh:path nltl:hasCrewTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] ;
    sh:property [
        sh:path nltl:hasPolarTrainingRecord ;
        sh:node gen:IMO102TrainingRecordShape
    ] ;
    sh:sparql [
        sh:message "An applicable crew member requires a matching valid STCW polar-training record." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                      nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role .

                BIND(
                    IF(
                        ?ice = nltl:iceFreeIceCondition,
                        nltl:polarTrainingNotApplicable,
                        IF(
                            ?ice = nltl:openWaterIceCondition && ?shipType = nltl:otherShipPolarTrainingType,
                            nltl:polarTrainingNotApplicable,
                            IF(
                                ?ice = nltl:openWaterIceCondition,
                                nltl:basicPolarTrainingLevel,
                                IF(
                                    ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole,
                                    nltl:basicPolarTrainingLevel,
                                    nltl:advancedPolarTrainingLevel
                                )
                            )
                        )
                    )
                    AS ?expected
                )

                FILTER(?expected != nltl:polarTrainingNotApplicable)

                FILTER NOT EXISTS {
                    $this nltl:hasCrewTrainingRecord ?record .
                    ?record nltl:requiredTrainingLevel ?expected ;
                            nltl:trainingRecordLevel ?expected ;
                            nltl:stcwQualificationValid true .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A linked polar-training record does not match the training level required by the 12.3.1 matrix." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this ?record
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                      nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role ;
                      (nltl:hasCrewTrainingRecord|nltl:hasPolarTrainingRecord) ?record .
                ?record nltl:requiredTrainingLevel ?required ;
                        nltl:trainingRecordLevel ?recorded .

                BIND(
                    IF(
                        ?ice = nltl:iceFreeIceCondition,
                        nltl:polarTrainingNotApplicable,
                        IF(
                            ?ice = nltl:openWaterIceCondition && ?shipType = nltl:otherShipPolarTrainingType,
                            nltl:polarTrainingNotApplicable,
                            IF(
                                ?ice = nltl:openWaterIceCondition,
                                nltl:basicPolarTrainingLevel,
                                IF(
                                    ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole,
                                    nltl:basicPolarTrainingLevel,
                                    nltl:advancedPolarTrainingLevel
                                )
                            )
                        )
                    )
                    AS ?expected
                )

                FILTER(?required != ?expected || ?recorded != ?expected)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A linked applicable polar-training record must confirm a valid STCW qualification." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this ?record
            WHERE {
                ?ship nltl:hasCrewMemberInventory $this ;
                      nltl:iceConditionClassification ?ice ;
                      nltl:polarTrainingShipTypeClassification ?shipType .
                $this nltl:polarTrainingCrewRoleClassification ?role ;
                      (nltl:hasCrewTrainingRecord|nltl:hasPolarTrainingRecord) ?record .

                BIND(
                    IF(
                        ?ice = nltl:iceFreeIceCondition,
                        nltl:polarTrainingNotApplicable,
                        IF(
                            ?ice = nltl:openWaterIceCondition && ?shipType = nltl:otherShipPolarTrainingType,
                            nltl:polarTrainingNotApplicable,
                            IF(
                                ?ice = nltl:openWaterIceCondition,
                                nltl:basicPolarTrainingLevel,
                                IF(
                                    ?role = nltl:officerInChargeOfNavigationalWatchPolarTrainingRole,
                                    nltl:basicPolarTrainingLevel,
                                    nltl:advancedPolarTrainingLevel
                                )
                            )
                        )
                    )
                    AS ?expected
                )

                FILTER(?expected != nltl:polarTrainingNotApplicable)

                FILTER NOT EXISTS {
                    ?record nltl:stcwQualificationValid true .
                }
            }
        """
    ] .

gen:IMO102TrainingRecordShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:requiredTrainingLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        )
    ] ;
    sh:property [
        sh:path nltl:trainingRecordLevel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:basicPolarTrainingLevel
            nltl:advancedPolarTrainingLevel
            nltl:polarTrainingNotApplicable
        )
    ] ;
    sh:property [
        sh:path nltl:stcwQualificationValid ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] .
<END_SHACL>
