<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO-105-ConditionalProhibitionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "In Arctic_waters, oil or oily mixture discharge to sea must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea "Arctic_waters" .
                FILTER NOT EXISTS {
                    $this nltl:oilOrOilyMixtureDischargeToSea false .
                }
            }
        """
    ] .
<END_SHACL>
