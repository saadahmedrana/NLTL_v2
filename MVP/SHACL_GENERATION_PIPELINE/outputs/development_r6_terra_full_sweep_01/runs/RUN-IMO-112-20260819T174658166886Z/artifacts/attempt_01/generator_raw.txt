<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-112Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "In Arctic_waters, discharge to sea of noxious liquid substances or mixtures containing them must be recorded as false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea "Arctic_waters" .
                FILTER NOT EXISTS {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea false .
                }
            }
        """
    ] .
<END_SHACL>
