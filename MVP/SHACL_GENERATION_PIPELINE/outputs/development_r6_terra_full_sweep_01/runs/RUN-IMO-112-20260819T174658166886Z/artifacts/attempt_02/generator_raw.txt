<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO112ConditionalProhibitionShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Ships operating in Arctic_waters must record at least one NLS discharge value of false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea "Arctic_waters" .
                FILTER NOT EXISTS {
                    $this nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea false .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "NLS discharge values for ships operating in Arctic_waters must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:operatingArea "Arctic_waters" ;
                    nltl:noxiousLiquidSubstanceOrNoxiousLiquidSubstanceMixtureDischargeToSea ?discharge .
                FILTER (
                    !isLiteral(?discharge) ||
                    datatype(?discharge) != xsd:boolean ||
                    ?discharge != false
                )
            }
        """
    ] .
<END_SHACL>
