<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO-115-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:sewageDischargeType ;
        sh:datatype xsd:string
    ] ;

    sh:property [
        sh:path nltl:comminutedAndDisinfected ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus ;
        sh:datatype xsd:boolean
    ] ;

    sh:property [
        sh:path nltl:distanceToNearestIceShelfOrFastIce ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:nearbyIceConcentration ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:sparql [
        sh:message "A comminuted and disinfected sewage discharge requires MARPOL Annex IV regulation 11.1.1 compliance and a distance greater than 3 nautical miles from ice-shelf or fast ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType "comminuted_and_disinfected"^^xsd:string ;
                      nltl:comminutedAndDisinfected true .

                FILTER NOT EXISTS {
                    $this nltl:marpolAnnexIvRegulation11Point1Point1ComplianceStatus true .
                }

                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestIceShelfOrFastIce ?distance .
                    ?distance a qudt:QuantityValue ;
                              qudt:numericValue ?value ;
                              qudt:unit unit:MI_N .
                    FILTER ( datatype(?value) = xsd:decimal &&
                             ?value > "3"^^xsd:decimal )
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "Record the distance from the area with ice concentration above one tenth when applicable discharge conditions and nearby ice concentration above one are recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType "comminuted_and_disinfected"^^xsd:string ;
                      nltl:comminutedAndDisinfected true ;
                      nltl:nearbyIceConcentration ?concentration .

                ?concentration a qudt:QuantityValue ;
                               qudt:numericValue ?concentrationValue ;
                               qudt:unit unit:UNITLESS .

                FILTER ( datatype(?concentrationValue) = xsd:decimal &&
                         ?concentrationValue > "1"^^xsd:decimal )

                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?distance .
                    ?distance a qudt:QuantityValue ;
                              qudt:numericValue ?distanceValue ;
                              qudt:unit unit:MI_N .
                    FILTER ( datatype(?distanceValue) = xsd:decimal )
                }
            }
        """
    ] .
<END_SHACL>
