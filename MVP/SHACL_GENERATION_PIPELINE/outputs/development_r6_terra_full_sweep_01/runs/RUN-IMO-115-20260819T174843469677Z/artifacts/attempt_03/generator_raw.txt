    sh:sparql [
        sh:message "A comminuted-and-disinfected sewage discharge requires a distance to the nearest ice shelf or fast ice greater than 3 nautical miles." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:sewageDischargeType ?dischargeType ;
                      nltl:comminutedAndDisinfected true .
                FILTER(?dischargeType = "comminuted_and_disinfected"^^xsd:string)
                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestIceShelfOrFastIce ?distance .
                    ?distance a qudt:QuantityValue ;
                              qudt:numericValue ?distanceValue ;
                              qudt:unit unit:MI_N .
                    FILTER(
                        datatype(?distanceValue) = xsd:decimal &&
                        ?distanceValue > "3"^^xsd:decimal
                    )
                }
            }
        """
    ] ;

            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>

            SELECT $this

                      nltl:comminutedAndDisinfected true ;
                      nltl:nearbyIceConcentration ?iceConcentration .

                ?iceConcentration a qudt:QuantityValue ;

                                  qudt:unit unit:UNITLESS .
                FILTER(
                    datatype(?concentrationValue) = xsd:decimal &&
                    ?concentrationValue > "1"^^xsd:decimal
                )
                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?distance .
                    ?distance a qudt:QuantityValue ;

                              qudt:unit unit:MI_N .

                }
            }
        """
