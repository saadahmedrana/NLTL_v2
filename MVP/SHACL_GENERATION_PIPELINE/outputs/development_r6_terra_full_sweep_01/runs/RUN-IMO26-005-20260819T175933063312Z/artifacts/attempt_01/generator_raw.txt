<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO26_005Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:headingOrPositionSystem ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:intendedArea ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:intendedAreaSuitabilityApprovalStatus ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceState
    ] ;
    sh:sparql [
        sh:message "A heading-reference or position-fixing system requires an approved intended-area suitability status." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:headingOrPositionSystem ?system .
                FILTER NOT EXISTS {
                    $this nltl:intendedAreaSuitabilityApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Intended-area suitability status for a heading-reference or position-fixing system must be approved." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:headingOrPositionSystem ?system ;
                      nltl:intendedAreaSuitabilityApprovalStatus ?status .
                FILTER(?status != nltl:evidenceStateApproved)
            }
        """
    ] .
<END_SHACL>
