<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:IMO26-010Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:requiredSensorProjectsBelowHull ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:sensorIceProtectionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Sensors projecting below the hull require ice protection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:requiredSensorProjectsBelowHull true .
                FILTER NOT EXISTS {
                    $this nltl:sensorIceProtectionPresent true .
                }
            }
        """
    ] .
<END_SHACL>
