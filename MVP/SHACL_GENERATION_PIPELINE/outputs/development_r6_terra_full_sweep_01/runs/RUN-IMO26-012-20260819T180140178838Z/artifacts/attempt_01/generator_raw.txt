<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:IMO26-012Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasNonMagneticHeadingMeans ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:nonMagneticHeadingMeans ;
        sh:minCount 2 ;
        sh:node [
            sh:property [
                sh:path nltl:connectedToMainPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:connectedToEmergencyPower ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:benchmarkEntity ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:independentFromHeadingMeans ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:nonMagneticHeadingMeans ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each non-magnetic heading means must be independent from another non-magnetic heading means owned by the same ship." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasNonMagneticHeadingMeans ?means .
                FILTER NOT EXISTS {
                    ?means nltl:independentFromHeadingMeans ?other .
                    $this nltl:hasNonMagneticHeadingMeans ?other .
                    FILTER (?other != ?means)
                }
            }
        """
    ] .
<END_SHACL>
