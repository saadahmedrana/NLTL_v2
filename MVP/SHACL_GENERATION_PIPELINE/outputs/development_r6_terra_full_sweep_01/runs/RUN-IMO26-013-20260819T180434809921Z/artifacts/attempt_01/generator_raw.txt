@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:IMO26-013LatitudeReadinessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:plannedOrActualLatitude ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:DEG ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gs:IMO26-013ApplicableShipShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:plannedOrActualLatitude ?latitudeValue .
                ?latitudeValue a qudt:QuantityValue ;
                    qudt:numericValue ?latitude ;
                    qudt:unit unit:DEG .
                FILTER (
                    datatype(?latitude) = xsd:decimal &&
                    ?latitude > "80"^^xsd:decimal
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:gnssCompassOrEquivalentCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive "1"^^xsd:integer ;
        sh:message "A ship proceeding to a latitude over 80 degrees must have at least one GNSS compass or equivalent."
    ] ;
    sh:property [
        sh:path nltl:connectedToMainPower ;
        sh:minCount 1 ;
        sh:class nltl:benchmarkEntity ;
        sh:message "A ship proceeding to a latitude over 80 degrees must have its GNSS compass or equivalent connected to the main power source."
    ] ;
    sh:property [
        sh:path nltl:connectedToEmergencyPower ;
        sh:minCount 1 ;
        sh:class nltl:benchmarkEntity ;
        sh:message "A ship proceeding to a latitude over 80 degrees must have its GNSS compass or equivalent connected to the emergency power source."
    ] .
