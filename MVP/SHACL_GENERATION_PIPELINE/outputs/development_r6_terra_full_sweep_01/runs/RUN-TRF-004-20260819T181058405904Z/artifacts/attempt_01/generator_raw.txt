<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-004-ApplicationOf2008IceClassRegulations>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date
    ] ;
    sh:sparql [
        sh:message "Ships contracted from 1 January 2010 through 31 December 2011 must identify the 2008 Ice Class Regulations as applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate .
                FILTER(
                    YEAR(?contractDate) >= 2010 &&
                    YEAR(?contractDate) < 2012
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2008 .
                }
            }
        """
    ] .
<END_SHACL>
