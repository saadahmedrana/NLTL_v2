<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-006-ApplicationOf1985IceClassRulesShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:sparql [
        sh:message "Ships constructed from 1 November 1986 through 31 August 2003 shall use the 1985 Ice Class Rules edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date .
                FILTER (
                    (YEAR(?date) > 1986 ||
                     (YEAR(?date) = 1986 &&
                      (MONTH(?date) > 11 ||
                       (MONTH(?date) = 11 && DAY(?date) >= 1))))
                    &&
                    (YEAR(?date) < 2003 ||
                     (YEAR(?date) = 2003 &&
                      (MONTH(?date) < 9 ||
                       (MONTH(?date) = 9 && DAY(?date) < 1))))
                )
                OPTIONAL {
                    $this nltl:applicableIceClassRegulationEdition ?edition .
                }
                FILTER (
                    !BOUND(?edition) ||
                    ?edition != nltl:iceClassRuleEdition1985
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For ships constructed from 1 November 1986 through 31 August 2003, the engine-output regulation edition may be the 1985 Rules edition or the 2008 Ice Class Regulations edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?date .
                FILTER (
                    (YEAR(?date) > 1986 ||
                     (YEAR(?date) = 1986 &&
                      (MONTH(?date) > 11 ||
                       (MONTH(?date) = 11 && DAY(?date) >= 1))))
                    &&
                    (YEAR(?date) < 2003 ||
                     (YEAR(?date) = 2003 &&
                      (MONTH(?date) < 9 ||
                       (MONTH(?date) = 9 && DAY(?date) < 1))))
                )
                $this nltl:engineOutputRegulationEdition ?engineEdition .
                FILTER (
                    ?engineEdition NOT IN (
                        nltl:iceClassRuleEdition1985,
                        nltl:iceClassRegulationEdition2008
                    )
                )
            }
        """
    ] .
<END_SHACL>
