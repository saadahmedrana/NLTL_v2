<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

g:TRF007ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "An IA Super or IA ship requires a valid construction stage date to determine applicability of TRF-007." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER(?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionDate .
                    FILTER(datatype(?constructionDate) = xsd:date)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ship within the TRF-007 applicability condition must use the section 3.2.2 or section 3.2.4 compliance method." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionDate .
                FILTER(?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER(datatype(?constructionDate) = xsd:date)
                FILTER(
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        (
                            MONTH(?constructionDate) < 9 ||
                            (MONTH(?constructionDate) = 9 && DAY(?constructionDate) < 1)
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:engineOutputComplianceMethod ?method .
                    FILTER(?method IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    ))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ship within the TRF-007 applicability condition must have an assessment date no later than 1 January of the delivery year plus twenty years." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionDate .
                FILTER(?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER(datatype(?constructionDate) = xsd:date)
                FILTER(
                    YEAR(?constructionDate) < 2003 ||
                    (
                        YEAR(?constructionDate) = 2003 &&
                        (
                            MONTH(?constructionDate) < 9 ||
                            (MONTH(?constructionDate) = 9 && DAY(?constructionDate) < 1)
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate ;
                        nltl:assessmentDate ?assessmentDate .
                    FILTER(datatype(?deliveryDate) = xsd:date)
                    FILTER(datatype(?assessmentDate) = xsd:date)
                    BIND((YEAR(?deliveryDate) + 20) AS ?deadlineYear)
                    FILTER(
                        YEAR(?assessmentDate) < ?deadlineYear ||
                        (
                            YEAR(?assessmentDate) = ?deadlineYear &&
                            MONTH(?assessmentDate) = 1 &&
                            DAY(?assessmentDate) <= 1
                        )
                    )
                }
            }
        """
    ] .
<END_SHACL>
