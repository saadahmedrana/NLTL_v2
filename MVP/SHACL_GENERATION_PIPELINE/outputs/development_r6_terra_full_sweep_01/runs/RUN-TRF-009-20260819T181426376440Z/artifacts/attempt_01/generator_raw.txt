<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:traficom2017Section3Point2Point2Method
            nltl:traficom2017Section3Point2Point4Method
        )
    ] ;
    sh:sparql [
        sh:message "Applicable IA Super or IA ships constructed before 1 September 2003 require a permitted 2017 Ice Class Regulations compliance method and an assessment no later than 1 January of the delivery year plus twenty years." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionDate .

                FILTER(
                    ?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa) &&
                    DATATYPE(?constructionDate) = xsd:date &&
                    (
                        YEAR(?constructionDate) < 2003 ||
                        (
                            YEAR(?constructionDate) = 2003 &&
                            (
                                MONTH(?constructionDate) < 9 ||
                                (
                                    MONTH(?constructionDate) = 9 &&
                                    DAY(?constructionDate) < 1
                                )
                            )
                        )
                    )
                )

                OPTIONAL {
                    $this nltl:engineOutputComplianceMethod ?complianceMethod .
                    FILTER(
                        ?complianceMethod IN (
                            nltl:traficom2017Section3Point2Point2Method,
                            nltl:traficom2017Section3Point2Point4Method
                        )
                    )
                }

                OPTIONAL {
                    $this nltl:deliveryDate ?deliveryDate .
                    FILTER(DATATYPE(?deliveryDate) = xsd:date)
                }

                OPTIONAL {
                    $this nltl:assessmentDate ?assessmentDate .
                    FILTER(DATATYPE(?assessmentDate) = xsd:date)
                }

                FILTER(
                    !BOUND(?complianceMethod) ||
                    !BOUND(?deliveryDate) ||
                    !BOUND(?assessmentDate) ||
                    !(
                        YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20 ||
                        (
                            YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                            MONTH(?assessmentDate) = 1 &&
                            DAY(?assessmentDate) = 1
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
