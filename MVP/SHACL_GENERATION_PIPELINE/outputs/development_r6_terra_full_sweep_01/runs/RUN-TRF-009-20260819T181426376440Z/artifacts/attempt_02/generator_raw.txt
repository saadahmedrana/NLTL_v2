        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:engineOutputComplianceMethodValue
    ] ;
    sh:sparql [
        sh:message "An IA Super or IA ship requires a construction stage date to establish applicability of this requirement." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionStageDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ship requires an engine output compliance method under section 3.2.2 or section 3.2.4 of the 2017 Ice Class Regulations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        (
                            MONTH(?constructionStageDate) < 9 ||
                            (
                                MONTH(?constructionStageDate) = 9 &&
                                DAY(?constructionStageDate) < 1
                            )
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:engineOutputComplianceMethod ?method .
                    FILTER (?method IN (
                        nltl:traficom2017Section3Point2Point2Method,
                        nltl:traficom2017Section3Point2Point4Method
                    ))
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ship requires a delivery date to determine the compliance deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        (
                            MONTH(?constructionStageDate) < 9 ||
                            (
                                MONTH(?constructionStageDate) = 9 &&
                                DAY(?constructionStageDate) < 1
                            )
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An applicable ship requires an assessment date to establish compliance by its deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        (
                            MONTH(?constructionStageDate) < 9 ||
                            (
                                MONTH(?constructionStageDate) = 9 &&
                                DAY(?constructionStageDate) < 1
                            )
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:assessmentDate ?assessmentDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The applicable ship must be assessed no later than 1 January of the year twenty years after its delivery year." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:constructionStageDate ?constructionStageDate ;
                      nltl:deliveryDate ?deliveryDate ;
                      nltl:assessmentDate ?assessmentDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (
                        YEAR(?constructionStageDate) = 2003 &&
                        (
                            MONTH(?constructionStageDate) < 9 ||
                            (
                                MONTH(?constructionStageDate) = 9 &&
                                DAY(?constructionStageDate) < 1
                            )
                        )
                    )
                )
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        (
                            MONTH(?assessmentDate) > 1 ||
                            (
                                MONTH(?assessmentDate) = 1 &&
                                DAY(?assessmentDate) > 1
                            )
                        )
                    )
                )
            }
        """
    ] .
