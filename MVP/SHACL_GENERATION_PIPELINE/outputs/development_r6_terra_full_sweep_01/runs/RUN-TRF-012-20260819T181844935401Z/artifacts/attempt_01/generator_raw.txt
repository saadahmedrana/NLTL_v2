<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:TRF012ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF012IceWaterlineShape ;
        sh:message "A lower ice waterline is required."
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceWaterline ;
        sh:minCount 1 ;
        sh:node gen:TRF012IceWaterlineShape ;
        sh:message "Intended ice-operating waterline data is required to establish the lower ice waterline envelope."
    ] ;
    sh:sparql [
        sh:message "Each lower ice-waterline profile point must equal the lowest intended ice-operating waterline point at the same longitudinal position."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this ?lowerPoint
            WHERE {
                $this nltl:hasLowerIceWaterline ?lowerLine .
                ?lowerLine nltl:hasWaterlineProfilePoint ?lowerPoint .
                ?lowerPoint nltl:longitudinalPosition ?lowerXQuantity ;
                            nltl:verticalCoordinate ?lowerZQuantity .
                ?lowerXQuantity qudt:numericValue ?lowerX .
                ?lowerZQuantity qudt:numericValue ?lowerZ .

                FILTER NOT EXISTS {
                    $this nltl:hasIntendedIceOperatingWaterline ?intendedLine .
                    ?intendedLine nltl:hasWaterlineProfilePoint ?intendedPoint .
                    ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                                   nltl:verticalCoordinate ?intendedZQuantity .
                    ?intendedXQuantity qudt:numericValue ?intendedX .
                    ?intendedZQuantity qudt:numericValue ?intendedZ .

                    FILTER (
                        ABS(?lowerX - ?intendedX) <=
                        (0.000000001 + 0.000001 * ABS(?intendedX))
                    )
                    FILTER (
                        ABS(?lowerZ - ?intendedZ) <=
                        (0.000000001 + 0.000001 * ABS(?intendedZ))
                    )

                    FILTER NOT EXISTS {
                        $this nltl:hasIntendedIceOperatingWaterline ?otherLine .
                        ?otherLine nltl:hasWaterlineProfilePoint ?otherPoint .
                        ?otherPoint nltl:longitudinalPosition ?otherXQuantity ;
                                    nltl:verticalCoordinate ?otherZQuantity .
                        ?otherXQuantity qudt:numericValue ?otherX .
                        ?otherZQuantity qudt:numericValue ?otherZ .

                        FILTER (
                            ABS(?otherX - ?intendedX) <=
                            (0.000000001 + 0.000001 * ABS(?intendedX))
                        )
                        FILTER (
                            ?otherZ <
                            ?intendedZ -
                            (0.000000001 + 0.000001 * ABS(?intendedZ))
                        )
                    }
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Every lowest intended ice-operating waterline point must be represented by a corresponding lower ice-waterline profile point."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this ?intendedPoint
            WHERE {
                $this nltl:hasIntendedIceOperatingWaterline ?intendedLine .
                ?intendedLine nltl:hasWaterlineProfilePoint ?intendedPoint .
                ?intendedPoint nltl:longitudinalPosition ?intendedXQuantity ;
                               nltl:verticalCoordinate ?intendedZQuantity .
                ?intendedXQuantity qudt:numericValue ?intendedX .
                ?intendedZQuantity qudt:numericValue ?intendedZ .

                FILTER NOT EXISTS {
                    $this nltl:hasIntendedIceOperatingWaterline ?otherLine .
                    ?otherLine nltl:hasWaterlineProfilePoint ?otherPoint .
                    ?otherPoint nltl:longitudinalPosition ?otherXQuantity ;
                                nltl:verticalCoordinate ?otherZQuantity .
                    ?otherXQuantity qudt:numericValue ?otherX .
                    ?otherZQuantity qudt:numericValue ?otherZ .

                    FILTER (
                        ABS(?otherX - ?intendedX) <=
                        (0.000000001 + 0.000001 * ABS(?intendedX))
                    )
                    FILTER (
                        ?otherZ <
                        ?intendedZ -
                        (0.000000001 + 0.000001 * ABS(?intendedZ))
                    )
                }

                FILTER NOT EXISTS {
                    $this nltl:hasLowerIceWaterline ?lowerLine .
                    ?lowerLine nltl:hasWaterlineProfilePoint ?lowerPoint .
                    ?lowerPoint nltl:longitudinalPosition ?lowerXQuantity ;
                                nltl:verticalCoordinate ?lowerZQuantity .
                    ?lowerXQuantity qudt:numericValue ?lowerX .
                    ?lowerZQuantity qudt:numericValue ?lowerZ .

                    FILTER (
                        ABS(?lowerX - ?intendedX) <=
                        (0.000000001 + 0.000001 * ABS(?intendedX))
                    )
                    FILTER (
                        ABS(?lowerZ - ?intendedZ) <=
                        (0.000000001 + 0.000001 * ABS(?intendedZ))
                    )
                }
            }
        """
    ] .

gen:TRF012IceWaterlineShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:waterlineProfilePoint ;
        sh:minCount 1 ;
        sh:node gen:TRF012WaterlineProfilePointShape ;
        sh:message "An ice waterline requires at least one profile point."
    ] .

gen:TRF012WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF012MetreQuantityShape ;
        sh:message "A waterline profile point requires one longitudinal position in metres."
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF012MetreQuantityShape ;
        sh:message "A waterline profile point requires one vertical coordinate in metres."
    ] .

gen:TRF012MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
