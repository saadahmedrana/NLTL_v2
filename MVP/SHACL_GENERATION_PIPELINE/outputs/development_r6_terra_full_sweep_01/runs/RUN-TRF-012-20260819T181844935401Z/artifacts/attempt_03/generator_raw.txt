<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF012ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:nodeKind sh:IRI ;
        sh:node gs:TRF012IceWaterlineShape
    ] ;
    sh:property [
        sh:path nltl:hasIntendedIceOperatingWaterline ;
        sh:nodeKind sh:IRI ;
        sh:node gs:TRF012IceWaterlineShape
    ] ;
    sh:sparql [
        sh:message "Each represented LIWL profile point must equal the minimum intended ice-operating-waterline vertical coordinate at its longitudinal position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this ?liwlPoint
            WHERE {
                {
                    SELECT ?ship ?liwlPoint (MIN(?intendedVerticalValue) AS ?minimumVerticalValue)
                    WHERE {
                        ?ship nltl:hasLowerIceWaterline ?liwl ;
                              nltl:hasIntendedIceOperatingWaterline ?intendedWaterline .

                        ?liwl nltl:hasWaterlineProfilePoint ?liwlPoint .
                        ?liwlPoint nltl:longitudinalPosition ?liwlLongitudinalQuantity .

                        ?liwlLongitudinalQuantity <http://qudt.org/schema/qudt/numericValue> ?longitudinalValue .

                        ?intendedWaterline nltl:hasWaterlineProfilePoint ?intendedPoint .
                        ?intendedPoint nltl:longitudinalPosition ?intendedLongitudinalQuantity ;
                                       nltl:verticalCoordinate ?intendedVerticalQuantity .

                        ?intendedLongitudinalQuantity <http://qudt.org/schema/qudt/numericValue> ?intendedLongitudinalValue .
                        ?intendedVerticalQuantity <http://qudt.org/schema/qudt/numericValue> ?intendedVerticalValue .

                        FILTER(?longitudinalValue = ?intendedLongitudinalValue)
                    }
                    GROUP BY ?ship ?liwlPoint
                }

                FILTER(?ship = $this)

                ?liwlPoint nltl:verticalCoordinate ?liwlVerticalQuantity .
                ?liwlVerticalQuantity <http://qudt.org/schema/qudt/numericValue> ?liwlVerticalValue .

                FILTER(?liwlVerticalValue != ?minimumVerticalValue)
            }
        """
    ] .

gs:TRF012IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path nltl:hasWaterlineProfilePoint ;
        sh:nodeKind sh:IRI ;
        sh:node gs:TRF012WaterlineProfilePointShape
    ] .

gs:TRF012WaterlineProfilePointShape
    a sh:NodeShape ;
    sh:class nltl:waterlineProfilePoint ;
    sh:property [
        sh:path nltl:longitudinalPosition ;
        sh:minCount 1 ;
        sh:node gs:TRF012MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalCoordinate ;
        sh:minCount 1 ;
        sh:node gs:TRF012MetreQuantityShape
    ] .

gs:TRF012MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
