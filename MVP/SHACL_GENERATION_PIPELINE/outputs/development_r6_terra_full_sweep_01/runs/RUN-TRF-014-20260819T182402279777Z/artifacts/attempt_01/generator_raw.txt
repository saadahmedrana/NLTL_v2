    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:summerLoadLineFreshWaterDraught ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;

    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;

    sh:property [
        sh:path nltl:iceClassDraughtMarkDraughtAmidships ;
        sh:node gs:MetreQuantityShape
    ] ;

    sh:sparql [
        sh:message "A ship built on or after 1 July 2007 with summer load-line draught numerically below UIWL must have a warning triangle and an ice-class draught mark at the maximum permissible amidships draught." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summerDraught ;
                      nltl:upperIceWaterlineDraught ?uiwlDraught .
                ?summerDraught qudt:numericValue ?summerValue ;
                               qudt:unit unit:M .
                ?uiwlDraught qudt:numericValue ?uiwlValue ;
                             qudt:unit unit:M .
                FILTER(
                    YEAR(?constructionDate) > 2007 ||
                    (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) > 7) ||
                    (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) = 7 && DAY(?constructionDate) >= 1)
                )
                FILTER(?summerValue < ?uiwlValue)
                FILTER NOT EXISTS {
                    $this nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?markDraught ;
                          nltl:maximumIceClassDraughtAmidships ?maximumDraught .
                    ?markDraught qudt:numericValue ?markValue ;
                                 qudt:unit unit:M .
                    ?maximumDraught qudt:numericValue ?maximumValue ;
                                    qudt:unit unit:M .
                    FILTER(ABS(?markValue - ?maximumValue) <= ((ABS(?maximumValue) * 0.000001) + 0.000000001))
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "For a pre-1 July 2007 ship with summer load-line draught numerically below UIWL, incomplete warning-triangle or draught-mark evidence requires a first scheduled dry-docking date after 1 July 2007 to assess the deadline." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                $this nltl:constructionDate ?constructionDate ;
                      nltl:summerLoadLineFreshWaterDraught ?summerDraught ;
                      nltl:upperIceWaterlineDraught ?uiwlDraught .
                ?summerDraught qudt:numericValue ?summerValue ;
                               qudt:unit unit:M .
                ?uiwlDraught qudt:numericValue ?uiwlValue ;
                             qudt:unit unit:M .
                FILTER(
                    YEAR(?constructionDate) < 2007 ||
                    (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) < 7) ||
                    (YEAR(?constructionDate) = 2007 && MONTH(?constructionDate) = 7 && DAY(?constructionDate) < 1)
                )
                FILTER(?summerValue < ?uiwlValue)
                FILTER NOT EXISTS {
                    $this nltl:warningTrianglePresent true ;
                          nltl:iceClassDraughtMarkPresent true ;
                          nltl:iceClassDraughtMarkDraughtAmidships ?markDraught ;
                          nltl:maximumIceClassDraughtAmidships ?maximumDraught .
                    ?markDraught qudt:numericValue ?markValue ;
                                 qudt:unit unit:M .
                    ?maximumDraught qudt:numericValue ?maximumValue ;
                                    qudt:unit unit:M .
                    FILTER(ABS(?markValue - ?maximumValue) <= ((ABS(?maximumValue) * 0.000001) + 0.000000001))
                }
                FILTER NOT EXISTS {
                    $this nltl:firstScheduledDryDockingDate ?dryDockingDate .
                    FILTER(
                        YEAR(?dryDockingDate) > 2007 ||
                        (YEAR(?dryDockingDate) = 2007 && MONTH(?dryDockingDate) > 7) ||
                        (YEAR(?dryDockingDate) = 2007 && MONTH(?dryDockingDate) = 7 && DAY(?dryDockingDate) > 1)
                    )
                }
            }
        """
    ] .
