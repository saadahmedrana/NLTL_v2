<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF015Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:intendedRouteSeaWaterSalinity ;
        sh:minCount 1 ;
        sh:node gs:SalinityQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadingCalculationSeaWaterSalinity ;
        sh:minCount 1 ;
        sh:node gs:SalinityQuantityShape
    ] ;
    sh:sparql [
        sh:message "Loading calculation sea water salinity must account for intended-route sea water salinity." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:intendedRouteSeaWaterSalinity ?routeSalinity .
                ?routeSalinity qudt:numericValue ?routeValue .
                $this nltl:loadingCalculationSeaWaterSalinity ?loadingSalinity .
                ?loadingSalinity qudt:numericValue ?loadingValue .
                FILTER (?routeValue != ?loadingValue)
            }
        """
    ] ;
    sh:or (
        gs:TRF015NotNavigatingInIceShape
        gs:TRF015NavigatingInIceShape
    ) .

gs:SalinityQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue nltl:gramPerKilogramSalinityUnit ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:MetreQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF015NotNavigatingInIceShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:hasValue false
    ] .

gs:TRF015NavigatingInIceShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:operatingDraught ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraught ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:operatingTrim ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumPermittedIceTrim ;
        sh:minCount 1 ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:sparql [
        sh:message "Operating draught must not exceed the upper ice waterline draught when navigating in ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:operatingDraught ?operatingDraught .
                ?operatingDraught qudt:numericValue ?operatingDraughtValue .
                $this nltl:upperIceWaterlineDraught ?uiwlDraught .
                ?uiwlDraught qudt:numericValue ?uiwlDraughtValue .
                FILTER (?operatingDraughtValue > ?uiwlDraughtValue)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Operating trim must not exceed the maximum permitted ice trim when navigating in ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:operatingTrim ?operatingTrim .
                ?operatingTrim qudt:numericValue ?operatingTrimValue .
                $this nltl:maximumPermittedIceTrim ?maximumTrim .
                ?maximumTrim qudt:numericValue ?maximumTrimValue .
                FILTER (?operatingTrimValue > ?maximumTrimValue)
            }
        """
    ] .
<END_SHACL>
