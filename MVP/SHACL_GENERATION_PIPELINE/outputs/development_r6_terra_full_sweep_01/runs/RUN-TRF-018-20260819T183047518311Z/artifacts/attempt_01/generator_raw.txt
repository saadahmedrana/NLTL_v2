<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF018RequiredEngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:sparql [
        sh:message "For ice classes IA, IB, and IC, maximum continuous rating power must be at least the calculated required power and 1000 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (
                    nltl:iceClassIa,
                    nltl:iceClassIb,
                    nltl:iceClassIc
                ))
                FILTER NOT EXISTS {
                    $this nltl:calculatedRequiredPower ?calculatedQuantity ;
                          nltl:maximumContinuousRatingPower ?mcrQuantity .
                    ?calculatedQuantity
                        qudt:numericValue ?calculatedPower ;
                        qudt:unit unit:KiloW .
                    ?mcrQuantity
                        qudt:numericValue ?mcrPower ;
                        qudt:unit unit:KiloW .
                    FILTER (
                        datatype(?calculatedPower) = xsd:decimal &&
                        datatype(?mcrPower) = xsd:decimal &&
                        ?mcrPower >= ?calculatedPower &&
                        ?mcrPower >= 1000
                    )
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "For ice class IA Super, maximum continuous rating power must be at least the calculated required power and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper .
                FILTER NOT EXISTS {
                    $this nltl:calculatedRequiredPower ?calculatedQuantity ;
                          nltl:maximumContinuousRatingPower ?mcrQuantity .
                    ?calculatedQuantity
                        qudt:numericValue ?calculatedPower ;
                        qudt:unit unit:KiloW .
                    ?mcrQuantity
                        qudt:numericValue ?mcrPower ;
                        qudt:unit unit:KiloW .
                    FILTER (
                        datatype(?calculatedPower) = xsd:decimal &&
                        datatype(?mcrPower) = xsd:decimal &&
                        ?mcrPower >= ?calculatedPower &&
                        ?mcrPower >= 2800
                    )
                }
            }
        """
    ] .
<END_SHACL>
