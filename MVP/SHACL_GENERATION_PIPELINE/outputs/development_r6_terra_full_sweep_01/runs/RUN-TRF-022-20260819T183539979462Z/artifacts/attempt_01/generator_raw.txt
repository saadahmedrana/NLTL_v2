                      nltl:clampedHullGeometryTerm/qudt:numericValue ?reported .
                FILTER (?breadth != 0)

                FILTER (
                    ABS(?reported - ?expected) >
                    ((0.000001 * ABS(?expected)) + 0.000000001)
                )
            }
        """
    ] .
