<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF024Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:deliveryDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:assessmentDate ;
        sh:datatype xsd:date
    ] ;
    sh:property [
        sh:path nltl:engineOutputComplianceMethod ;
        sh:nodeKind sh:IRI
    ] ;
    sh:sparql [
        sh:message "Construction stage date is needed to determine applicability for a ship retaining ice class IA Super or IA." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER NOT EXISTS {
                    $this nltl:constructionStageDate ?constructionStageDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Delivery date is required where the ship has ice class IA Super or IA and construction began before 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (datatype(?constructionStageDate) = xsd:date)
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) < 9) ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) = 9 && DAY(?constructionStageDate) < 1)
                )
                FILTER NOT EXISTS {
                    $this nltl:deliveryDate ?deliveryDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Assessment date is required where the ship has ice class IA Super or IA and construction began before 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (datatype(?constructionStageDate) = xsd:date)
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) < 9) ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) = 9 && DAY(?constructionStageDate) < 1)
                )
                FILTER NOT EXISTS {
                    $this nltl:assessmentDate ?assessmentDate .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The section 3.2.2 compliance method is required where the ship has ice class IA Super or IA and construction began before 1 September 2003." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionStageDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (datatype(?constructionStageDate) = xsd:date)
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) < 9) ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) = 9 && DAY(?constructionStageDate) < 1)
                )
                FILTER NOT EXISTS {
                    $this nltl:engineOutputComplianceMethod ?method .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A ship within the applicability condition must use the TRAFICOM 2017 section 3.2.2 compliance method." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionStageDate ;
                    nltl:engineOutputComplianceMethod ?method .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (datatype(?constructionStageDate) = xsd:date)
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) < 9) ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) = 9 && DAY(?constructionStageDate) < 1)
                )
                FILTER (?method != nltl:traficom2017Section3Point2Point2Method)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Assessment must be no later than 1 January of the year twenty years after delivery." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                    nltl:constructionStageDate ?constructionStageDate ;
                    nltl:deliveryDate ?deliveryDate ;
                    nltl:assessmentDate ?assessmentDate .
                FILTER (?iceClass IN (nltl:iceClassIaSuper, nltl:iceClassIa))
                FILTER (
                    datatype(?constructionStageDate) = xsd:date &&
                    datatype(?deliveryDate) = xsd:date &&
                    datatype(?assessmentDate) = xsd:date
                )
                FILTER (
                    YEAR(?constructionStageDate) < 2003 ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) < 9) ||
                    (YEAR(?constructionStageDate) = 2003 && MONTH(?constructionStageDate) = 9 && DAY(?constructionStageDate) < 1)
                )
                FILTER (
                    YEAR(?assessmentDate) > YEAR(?deliveryDate) + 20 ||
                    (
                        YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20 &&
                        (
                            MONTH(?assessmentDate) > 1 ||
                            (MONTH(?assessmentDate) = 1 && DAY(?assessmentDate) > 1)
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
