    sh:property [
        sh:path nltl:shipLength ;
        sh:minCount 1 ;

    sh:sparql [
        sh:message "For an IA Super ship with a bulb, C1 and C2 must match the prescribed formulae within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:brashIceResistanceCoefficientC1 ?c1q ;
          nltl:brashIceResistanceCoefficientC2 ?c2q ;
          nltl:coefficientF1 ?f1q ;
          nltl:coefficientF2 ?f2q ;
          nltl:coefficientF3 ?f3q ;
          nltl:coefficientF4 ?f4q ;
          nltl:coefficientG1 ?g1q ;
          nltl:coefficientG2 ?g2q ;
          nltl:coefficientG3 ?g3q ;
          nltl:iceClassDraught ?tq ;
          nltl:shipBreadth ?bq ;
          nltl:shipLength ?lq .

    ?c1q qudt:numericValue ?c1Value .
    ?c2q qudt:numericValue ?c2Value .
    ?f1q qudt:numericValue ?f1Value .
    ?f2q qudt:numericValue ?f2Value .
    ?f3q qudt:numericValue ?f3Value .
    ?f4q qudt:numericValue ?f4Value .
    ?g1q qudt:numericValue ?g1Value .
    ?g2q qudt:numericValue ?g2Value .
    ?g3q qudt:numericValue ?g3Value .
    ?tq qudt:numericValue ?tValue .
    ?bq qudt:numericValue ?bValue .
    ?lq qudt:numericValue ?lValue .

    BIND(xsd:decimal(?c1Value) AS ?c1)
    BIND(xsd:decimal(?c2Value) AS ?c2)
    BIND(xsd:decimal(?f1Value) AS ?f1)
    BIND(xsd:decimal(?f2Value) AS ?f2)
    BIND(xsd:decimal(?f3Value) AS ?f3)
    BIND(xsd:decimal(?f4Value) AS ?f4)
    BIND(xsd:decimal(?g1Value) AS ?g1)
    BIND(xsd:decimal(?g2Value) AS ?g2)
    BIND(xsd:decimal(?g3Value) AS ?g3)
    BIND(xsd:decimal(?tValue) AS ?t)
    BIND(xsd:decimal(?bValue) AS ?b)
    BIND(xsd:decimal(?lValue) AS ?l)

    BIND(
        (?f1 * (?b * ?l) / ((2 * ?t / ?b) + 1))
        + (2.89 * ((?f2 * ?b) + (?f3 * ?l) + (?f4 * ?b * ?l)))
        AS ?expectedC1
    ) .
    BIND(
        (6.67 * (?g1 + (?g2 * ?b)))
        + (?g3 * (1 + (1.2 * ?t / ?b)) * ?b * ?b / SQRT(?l))
        AS ?expectedC2
    )

    FILTER (
        ABS(?c1 - ?expectedC1) > (0.000000001 + (0.000001 * ABS(?expectedC1)))
        ||
        ABS(?c2 - ?expectedC2) > (0.000000001 + (0.000001 * ABS(?expectedC2)))
    )
}
"""
    ] .
