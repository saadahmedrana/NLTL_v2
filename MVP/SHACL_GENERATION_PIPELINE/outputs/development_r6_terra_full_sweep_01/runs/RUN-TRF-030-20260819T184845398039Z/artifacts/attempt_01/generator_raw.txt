g:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:node g:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:minCount 1 ;
        sh:node g:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:node g:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:minCount 4 ;
        sh:node g:TRF030DirectAnalysisCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ]
        ] ;
        sh:qualifiedMinCount 2
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:midSpanOrSpacingCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:loadPatchLength ;
                        sh:minCount 1 ;
                        sh:node g:LengthQuantityShape
                    ] ;
                    sh:property [
                        sh:path nltl:iceLoadAreaFactorCa ;
                        sh:minCount 1 ;
                        sh:node g:DimensionlessQuantityShape
                    ]
                ] ;
                sh:qualifiedMinCount 2 ;
                sh:message "When load length is not determined from the arrangement, at least two analysis cases must pair load-patch length with ice-load area factor Ca."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times ice pressure within the stated numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:appliedIcePressure ?appliedQuantity .
                ?pressureQuantity qudt:numericValue ?pressure .
                ?appliedQuantity qudt:numericValue ?applied .
                BIND((1.8 * ?pressure) AS ?expected)
                FILTER (ABS(?applied - ?expected) >
                        ((0.000001 * ABS(?expected)) + 0.000000001))
            }
        """
    ] .

g:TRF030DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        )
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineReferencePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterlineReferencePosition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node g:LengthQuantityShape
    ] ;
    sh:sparql [
        sh:message "A case centred at the upper ice waterline must use the upper ice waterline reference position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:verticalLoadPositionType nltl:upperIceWaterlineCenteredPosition ;
                      nltl:verticalLoadPosition ?positionQuantity ;
                      nltl:upperIceWaterlineReferencePosition ?referenceQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?referenceQuantity qudt:numericValue ?reference .
                FILTER (ABS(?position - ?reference) >
                        ((0.000001 * ABS(?reference)) + 0.000000001))
            }
        """
    ] ;
    sh:sparql [
        sh:message "A case half a load-patch height below the lower ice waterline must use the corresponding vertical position." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:verticalLoadPositionType nltl:halfLoadHeightBelowLowerIceWaterlinePosition ;
                      nltl:verticalLoadPosition ?positionQuantity ;
                      nltl:lowerIceWaterlineReferencePosition ?lowerQuantity .
                ?ship nltl:hasDirectAnalysisCase $this ;
                      nltl:loadPatchHeight ?heightQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?lowerQuantity qudt:numericValue ?lower .
                ?heightQuantity qudt:numericValue ?height .
                BIND((?lower - (0.5 * ?height)) AS ?expected)
                FILTER (ABS(?position - ?expected) >
                        ((0.000001 * ABS(?expected)) + 0.000000001))
            }
        """
    ] .
