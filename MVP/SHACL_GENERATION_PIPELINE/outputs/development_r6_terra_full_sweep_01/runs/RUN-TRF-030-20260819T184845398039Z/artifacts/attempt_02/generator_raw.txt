gs:ShipDirectAnalysisShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:class nltl:directAnalysisCase ;
        sh:node gs:DirectAnalysisCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:upperIceWaterlineCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:verticalLoadPositionType ;
                sh:hasValue nltl:intermediateVerticalPosition
            ]
        ] ;
        sh:qualifiedMinCount 2
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape [
            sh:property [
                sh:path nltl:horizontalLoadPositionType ;
                sh:hasValue nltl:midSpanOrSpacingCenteredPosition
            ]
        ] ;
        sh:qualifiedMinCount 1
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:loadLengthDeterminedFromArrangement ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasDirectAnalysisCase ;
                sh:qualifiedValueShape [
                    sh:property [
                        sh:path nltl:iceLoadAreaFactorCa ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node gs:DimensionlessQuantityShape
                    ]
                ] ;
                sh:qualifiedMinCount 2
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Applied ice pressure must equal 1.8 times ice pressure within relative tolerance 1e-6 and absolute tolerance 1e-9 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:appliedIcePressure ?appliedQuantity .
                ?pressureQuantity qudt:numericValue ?p .
                ?appliedQuantity qudt:numericValue ?applied .
                BIND((1.8 * ?p) AS ?expected)
                BIND(IF(ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(?applied - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A UIWL-centred case must have its vertical load position equal to its upper ice waterline reference position within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasDirectAnalysisCase ?case .
                ?case nltl:verticalLoadPositionType nltl:upperIceWaterlineCenteredPosition ;
                      nltl:verticalLoadPosition ?positionQuantity ;
                      nltl:upperIceWaterlineReferencePosition ?upperQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?upperQuantity qudt:numericValue ?upper .
                BIND(IF(ABS(?upper) * 0.000001 > 0.000000001,
                        ABS(?upper) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(?position - ?upper) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A half-load-height-below-LIWL case must have vertical load position equal to LIWL minus 0.5 times load patch height within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:loadPatchHeight ?heightQuantity ;
                      nltl:hasDirectAnalysisCase ?case .
                ?heightQuantity qudt:numericValue ?height .
                ?case nltl:verticalLoadPositionType nltl:halfLoadHeightBelowLowerIceWaterlinePosition ;
                      nltl:verticalLoadPosition ?positionQuantity ;
                      nltl:lowerIceWaterlineReferencePosition ?lowerQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?lowerQuantity qudt:numericValue ?lower .
                BIND((?lower - (0.5 * ?height)) AS ?expected)
                BIND(IF(ABS(?expected) * 0.000001 > 0.000000001,
                        ABS(?expected) * 0.000001,
                        0.000000001) AS ?tolerance)
                FILTER(ABS(?position - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two intermediate vertical load locations with distinct vertical positions must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1, ?case2 .
                    ?case1 nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionQuantity1 .
                    ?case2 nltl:verticalLoadPositionType nltl:intermediateVerticalPosition ;
                           nltl:verticalLoadPosition ?positionQuantity2 .
                    ?positionQuantity1 qudt:numericValue ?position1 .
                    ?positionQuantity2 qudt:numericValue ?position2 .
                    FILTER(?case1 != ?case2 && ?position1 != ?position2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "At least two distinct horizontal load locations must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1, ?case2 .
                    ?case1 nltl:horizontalLoadPosition ?positionQuantity1 .
                    ?case2 nltl:horizontalLoadPosition ?positionQuantity2 .
                    ?positionQuantity1 qudt:numericValue ?position1 .
                    ?positionQuantity2 qudt:numericValue ?position2 .
                    FILTER(?case1 != ?case2 && ?position1 != ?position2)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "When load length is not determined from arrangement, at least two distinct load-patch lengths with corresponding ca values must be checked." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:loadLengthDeterminedFromArrangement false .
                FILTER NOT EXISTS {
                    $this nltl:hasDirectAnalysisCase ?case1, ?case2 .
                    ?case1 nltl:loadPatchLength ?lengthQuantity1 ;
                           nltl:iceLoadAreaFactorCa ?caQuantity1 .
                    ?case2 nltl:loadPatchLength ?lengthQuantity2 ;
                           nltl:iceLoadAreaFactorCa ?caQuantity2 .
                    ?lengthQuantity1 qudt:numericValue ?length1 .
                    ?lengthQuantity2 qudt:numericValue ?length2 .
                    FILTER(?case1 != ?case2 && ?length1 != ?length2)
                }
            }
        """
    ] .
