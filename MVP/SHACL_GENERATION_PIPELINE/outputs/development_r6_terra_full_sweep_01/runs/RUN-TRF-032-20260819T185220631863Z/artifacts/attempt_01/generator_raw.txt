<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF032ScantlingSelectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:regulationRequiredScantling ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:classificationSocietyRequiredScantling ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ]
                ]
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When the regulation-required scantling is less than the classification-society-required scantling for a ship not ice strengthened, the selected design scantling shall use the classification-society-required value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:regulationRequiredScantling ?regulationQuantity ;
                      nltl:classificationSocietyRequiredScantling ?classificationQuantity .
                ?regulationQuantity qudt:numericValue ?regulationValue .
                ?classificationQuantity qudt:numericValue ?classificationValue .
                FILTER(?regulationValue < ?classificationValue)
                FILTER NOT EXISTS {
                    $this nltl:selectedDesignScantling ?selectedQuantity .
                    ?selectedQuantity qudt:numericValue ?selectedValue .
                    FILTER(
                        ABS(?selectedValue - ?classificationValue)
                        <= ((ABS(?classificationValue) * 0.000001) + 0.000000001)
                    )
                }
            }
        """
    ] .
<END_SHACL>
