<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-032Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceStrengtheningStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "When regulation-required scantling is less than classification-society-required scantling for a ship without ice strengthening, selected design scantling must be a QuantityValue matching the classification-society value within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:iceStrengtheningStatus false ;
                      nltl:regulationRequiredScantling ?regulationQuantity ;
                      nltl:classificationSocietyRequiredScantling ?classificationQuantity .

                ?regulationQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?regulationValue .

                ?classificationQuantity a qudt:QuantityValue ;
                    qudt:numericValue ?classificationValue .

                FILTER (
                    isNumeric(?regulationValue) &&
                    isNumeric(?classificationValue) &&
                    ?regulationValue < ?classificationValue
                )

                FILTER NOT EXISTS {
                    $this nltl:selectedDesignScantling ?selectedQuantity .
                    ?selectedQuantity a qudt:QuantityValue ;
                        qudt:numericValue ?selectedValue .

                    FILTER (isNumeric(?selectedValue))
                    FILTER (
                        ABS(?selectedValue - ?classificationValue) <=
                        ((ABS(?classificationValue) * 0.000001) + 0.000000001)
                    )
                }
            }
        """
    ] .
<END_SHACL>
