gen:TRF042PlatingShape a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042DesignIceLoadHeightQuantityShape
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042SpacingQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042F2QuantityShape
    ] ;
    sh:property [
        sh:path nltl:hullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:steelGradeValue
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042YieldStrengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:TRF042CorrosionAdditionQuantityShape
    ] ;
    sh:xone (
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:TRF042CorrosionAdditionTwoMmShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:corrosionAbrasionAddition ;
                sh:node gen:TRF042CorrosionAdditionLowerThanTwoMmShape
            ] ;
            sh:property [
                sh:path nltl:specialSurfaceCoatingMaintained ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:materialApprovalStatus ;
                sh:hasValue nltl:evidenceStateApproved ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "longitudinalPlatingFactorF2 must equal the applicable f2 formula within max(1e-6 times the expected value magnitude, 1e-12)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:spacing ?spacingQuantity ;
                      nltl:longitudinalPlatingFactorF2 ?f2Quantity .
                ?heightQuantity qudt:numericValue ?height .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?f2Quantity qudt:numericValue ?f2 .
                BIND(?height / ?spacing AS ?ratio)
                FILTER(?ratio > 0 && ?ratio <= 1.8)
                BIND(
                    IF(?ratio <= 1,
                       (0.4 / ?ratio) + 0.6,
                       1.4 - (0.4 * ?ratio)
                    ) AS ?expected
                )
                BIND(
                    IF((0.000001 * ABS(?expected)) > 0.000000000001,
                       0.000001 * ABS(?expected),
                       0.000000000001
                    ) AS ?tolerance
                )
                FILTER(ABS(?f2 - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Normal-strength hull structural steel must have yield strength exactly 235 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade nltl:normalStrengthHullStructuralSteel ;
                      nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER(?yield < 235 || ?yield > 235)
            }
        """
    ] ;
    sh:sparql [
        sh:message "High-strength hull structural steel must have yield strength at least 315 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade nltl:highStrengthHullStructuralSteel ;
                      nltl:yieldStrength ?yieldQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER(?yield < 315)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A steel grade other than the normal-strength or high-strength canonical grades requires approved material status for its accepted actual yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade ?grade .
                FILTER(?grade NOT IN (
                    nltl:normalStrengthHullStructuralSteel,
                    nltl:highStrengthHullStructuralSteel
                ))
                FILTER NOT EXISTS {
                    $this nltl:materialApprovalStatus nltl:evidenceStateApproved .
                }
            }
        """
    ] .
