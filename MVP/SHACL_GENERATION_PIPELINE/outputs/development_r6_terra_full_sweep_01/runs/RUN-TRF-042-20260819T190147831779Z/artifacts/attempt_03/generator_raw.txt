                        1.4 - (0.4 * ?ratio)

                BIND(

                        (0.000001 * ABS(?expected)) > 0.000000000001,
                        (0.000001 * ABS(?expected)),

                    AS ?tolerance
                )
                FILTER (ABS(?actual - ?expected) > ?tolerance)
            }
        """
