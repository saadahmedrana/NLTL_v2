<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:MeterQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MegaPAQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:CentiM2QuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:CentiM3QuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:CentiM3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF044TransverseFrameShape a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;

    sh:property [
        sh:path nltl:frameBoundaryConditionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bulkCarrierTopWingTankFrameCondition
            nltl:singleDeckTankTopToMainDeckFrameCondition
            nltl:multiDeckOrStringerContinuousFrameCondition
            nltl:twoDeckFrameCondition
        )
    ] ;

    sh:property [
        sh:path nltl:frameBoundaryConditionFactorM0 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;

    sh:property [
        sh:path nltl:frameMomentFactorMt ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;

    sh:property [
        sh:path nltl:frameShearDistributionFactorF3 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:and (
                gen:UnitlessQuantityShape
                [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "1.2"^^xsd:decimal ;
                        sh:maxInclusive "1.2"^^xsd:decimal
                    ]
                ]
            )
        ]
    ] ;

    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPAQuantityShape
    ] ;

    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;

    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;

    sh:property [
        sh:path nltl:effectiveFrameSpan ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeterQuantityShape
    ] ;

    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MegaPAQuantityShape
    ] ;

    sh:property [
        sh:path nltl:sectionModulus ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CentiM3QuantityShape
    ] ;

    sh:property [
        sh:path nltl:requiredShearArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:CentiM2QuantityShape
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:bulkCarrierTopWingTankFrameCondition
            ] ;
            sh:property [
                sh:path nltl:frameBoundaryConditionFactorM0 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "7"^^xsd:decimal ;
                        sh:maxInclusive "7"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:singleDeckTankTopToMainDeckFrameCondition
            ] ;
            sh:property [
                sh:path nltl:frameBoundaryConditionFactorM0 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "6"^^xsd:decimal ;
                        sh:maxInclusive "6"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:multiDeckOrStringerContinuousFrameCondition
            ] ;
            sh:property [
                sh:path nltl:frameBoundaryConditionFactorM0 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "5.7"^^xsd:decimal ;
                        sh:maxInclusive "5.7"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:frameBoundaryConditionType ;
                sh:hasValue nltl:twoDeckFrameCondition
            ] ;
            sh:property [
                sh:path nltl:frameBoundaryConditionFactorM0 ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "5"^^xsd:decimal ;
                        sh:maxInclusive "5"^^xsd:decimal
                    ]
                ]
            ]
        ]
    ) ;

    sh:sparql [
        sh:message "Frame moment factor cannot be calculated when 7 - 5*h/l equals zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:effectiveFrameSpan ?spanQuantity .
                ?heightQuantity qudt:numericValue ?height .
                ?spanQuantity qudt:numericValue ?span .
                FILTER ((7 - (5 * ?height / ?span)) = 0)
            }
        """
    ] ;

    sh:sparql [
        sh:message "Frame moment factor, section modulus, or required shear area does not match the prescribed calculation within relative tolerance 1e-6 and absolute fallback 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameBoundaryConditionFactorM0 ?m0Quantity ;
                      nltl:frameMomentFactorMt ?mtQuantity ;
                      nltl:frameShearDistributionFactorF3 ?f3Quantity ;
                      nltl:icePressure ?pressureQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:spacing ?spacingQuantity ;
                      nltl:effectiveFrameSpan ?spanQuantity ;
                      nltl:yieldStrength ?yieldQuantity ;
                      nltl:sectionModulus ?sectionQuantity ;
                      nltl:requiredShearArea ?areaQuantity .
                ?m0Quantity qudt:numericValue ?m0 .
                ?mtQuantity qudt:numericValue ?mt .
                ?f3Quantity qudt:numericValue ?f3 .
                ?pressureQuantity qudt:numericValue ?pressure .
                ?heightQuantity qudt:numericValue ?height .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?spanQuantity qudt:numericValue ?span .
                ?yieldQuantity qudt:numericValue ?yield .
                ?sectionQuantity qudt:numericValue ?section .
                ?areaQuantity qudt:numericValue ?area .
                FILTER ((7 - (5 * ?height / ?span)) != 0)
                BIND((7 * ?m0) / (7 - (5 * ?height / ?span)) AS ?expectedMt)
                BIND((?pressure * ?spacing * ?height * ?span / (?expectedMt * ?yield)) * 1000000 AS ?expectedSection)
                BIND((1.7320508075688772 * ?f3 * ?pressure * ?height * ?spacing / (2 * ?yield)) * 10000 AS ?expectedArea)
                FILTER (
                    ABS(?mt - ?expectedMt) > ((ABS(?expectedMt) * 0.000001) + 0.000000001) ||
                    ABS(?section - ?expectedSection) > ((ABS(?expectedSection) * 0.000001) + 0.000000001) ||
                    ABS(?area - ?expectedArea) > ((ABS(?expectedArea) * 0.000001) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
