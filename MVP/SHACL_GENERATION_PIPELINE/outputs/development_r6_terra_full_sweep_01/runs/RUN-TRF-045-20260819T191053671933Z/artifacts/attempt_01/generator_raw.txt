<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:TRF045FrameScantlingsShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:frameSpanWithinIceStrengtheningZone ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Ordinary frame scantlings require decimal metre values for the frame span and the span within the ice-strengthening zone." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true .
                FILTER NOT EXISTS {
                    $this nltl:span ?spanQuantity ;
                          nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
                    ?spanQuantity qudt:numericValue ?spanValue ;
                                  qudt:unit unit:M .
                    ?zoneQuantity qudt:numericValue ?zoneValue ;
                                  qudt:unit unit:M .
                    FILTER (
                        datatype(?spanValue) = xsd:decimal &&
                        datatype(?zoneValue) = xsd:decimal
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Ordinary frame scantlings may be used only when the frame span within the ice-strengthening zone is less than 15 percent of the frame span." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true ;
                      nltl:span ?spanQuantity ;
                      nltl:frameSpanWithinIceStrengtheningZone ?zoneQuantity .
                ?spanQuantity qudt:numericValue ?spanValue ;
                              qudt:unit unit:M .
                ?zoneQuantity qudt:numericValue ?zoneValue ;
                              qudt:unit unit:M .
                FILTER (
                    datatype(?spanValue) = xsd:decimal &&
                    datatype(?zoneValue) = xsd:decimal
                )
                FILTER (?zoneValue >= (?spanValue * "0.15"^^xsd:decimal))
            }
        """
    ] .
<END_SHACL>
