<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix qk: <http://qudt.org/vocab/quantitykind/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:LengthMillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] ;
    sh:property [
        sh:path qudt:hasQuantityKind ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qk:Length
    ] .

gs:FrameAttachmentShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasFrameAttachment ;
    sh:targetObjectsOf nltl:hasSupportingStructureAttachment ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket
    ] .

gs:InstalledConnectionBracketShape
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasConnectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:LengthMillimetreQuantityShape ;
        sh:message "An installed connection bracket must have a millimetre length thickness value."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "An installed connection bracket must have an edge stiffened against buckling."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "An installed connection bracket must be stiffened against buckling."
    ] .

gs:FrameAttachmentComplianceShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:maxCount 1 ;
        sh:node gs:LengthMillimetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:class nltl:benchmarkEntity
    ] ;
    sh:property [
        sh:path nltl:hasSupportingWebFrame ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasSupportingBulkhead ;
        sh:class nltl:supportingStructure
    ] ;
    sh:sparql [
        sh:message "Within the ice-strengthened area, every linked supporting structure must have an effective shared frame attachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      ?supportLink ?support .
                FILTER (?supportLink IN (
                    nltl:hasSupportingWebFrame,
                    nltl:hasSupportingBulkhead
                ))
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:effectiveAttachmentConfirmed true .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A longitudinal frame must use a connection bracket for every linked supporting web frame and bulkhead." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:longitudinalFrame ;
                      nltl:withinIceStrengthenedArea true ;
                      ?supportLink ?support .
                FILTER (?supportLink IN (
                    nltl:hasSupportingWebFrame,
                    nltl:hasSupportingBulkhead
                ))
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar attachment construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this a nltl:transverseFrame ;
                      nltl:withinIceStrengthenedArea true ;
                      nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:similarAttachmentConstructionPresent true .
                }
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame passing through a supporting structure must have both sides of its web plate connected." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount 2 .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An installed bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      nltl:hasFrameAttachment ?attachment ;
                      nltl:frameWebThickness ?webThickness .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?webThicknessValue .
                    ?webThicknessValue nltl:hasQuantityKind <http://qudt.org/vocab/quantitykind/Length> ;
                                       nltl:unit <http://qudt.org/vocab/unit/MilliM> ;
                                       nltl:numericValue ?webValue .
                    ?bracket nltl:bracketThickness ?bracketThicknessValue .
                    ?bracketThicknessValue nltl:hasQuantityKind <http://qudt.org/vocab/quantitykind/Length> ;
                                           nltl:unit <http://qudt.org/vocab/unit/MilliM> ;
                                           nltl:numericValue ?bracketValue .
                    FILTER (?bracketValue >= ?webValue)
                }
            }
        """
    ] .
<END_SHACL>
