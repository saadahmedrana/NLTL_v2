<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF049QuantityValueShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM>
    ] .

<urn:nltl:generated-shape:TRF049ConnectionBracketShape>
    a sh:NodeShape ;
    sh:targetClass nltl:connectionBracket ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF049QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] .

<urn:nltl:generated-shape:TRF049FrameAttachmentShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frameAttachment ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:similarAttachmentConstructionPresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:class nltl:supportingStructure
    ] ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:class nltl:connectionBracket
    ] .

<urn:nltl:generated-shape:TRF049FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:targetClass nltl:transverseFrame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:passesThroughSupportingStructure ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:webPlateConnectionSideCount ;
        sh:datatype xsd:integer ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:class qudt:QuantityValue ;
        sh:node <urn:nltl:generated-shape:TRF049QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:class nltl:frameAttachment
    ] ;
    sh:sparql [
        sh:message "Each supporting structure of a frame within the ice-strengthened area must share an effectively confirmed attachment with that frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:effectiveAttachmentConfirmed true .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame with an installed connection bracket must provide its web thickness in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?frameQuantity .
                    ?frameQuantity qudt:numericValue ?frameValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameQuantity ;
                      nltl:hasFrameAttachment ?attachment .
                ?frameQuantity qudt:numericValue ?frameValue ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketQuantity .
                ?bracketQuantity qudt:numericValue ?bracketValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] ;
    sh:sparql [
        sh:message "A frame running through a supporting structure must record connection of both web-plate sides." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:passesThroughSupportingStructure true .
                FILTER NOT EXISTS {
                    $this nltl:webPlateConnectionSideCount 2 .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF049LongitudinalFrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:longitudinalFrame ;
    sh:sparql [
        sh:message "Each supporting web frame or bulkhead of a longitudinal frame must share an attachment having a connection bracket." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:hasConnectionBracket ?bracket .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF049TransverseFrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:transverseFrame ;
    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have an attachment with a bracket or similar construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    {
                        ?attachment nltl:hasConnectionBracket ?bracket .
                    }
                    UNION
                    {
                        ?attachment nltl:similarAttachmentConstructionPresent true .
                    }
                }
            }
        """
    ] .
<END_SHACL>
