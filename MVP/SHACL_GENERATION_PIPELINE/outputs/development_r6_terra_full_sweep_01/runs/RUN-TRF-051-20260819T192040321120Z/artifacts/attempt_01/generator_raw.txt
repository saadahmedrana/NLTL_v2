<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:MillimetreQuantity a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MilliM
    ] .

gs:YieldStrengthQuantity a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaPA
    ] .

gs:TRF051HullStructureShape a sh:NodeShape ;
    sh:targetClass nltl:hullStructure ;
    sh:property [
        sh:path nltl:inLieuOfFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantity ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive 9
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:adjacentFrameHeight ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantity ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive 0
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:node gs:YieldStrengthQuantity
    ] ;
    sh:property [
        sh:path nltl:netShellPlatingThickness ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantity
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:node gs:MillimetreQuantity
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:inLieuOfFrame ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:inLieuOfFrame ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameProfileType ;
                sh:hasValue nltl:profileSection
            ]
        ]
        [
            sh:property [
                sh:path nltl:inLieuOfFrame ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameProfileType ;
                sh:hasValue nltl:flatBarSection
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Frame web thickness is below the h_w*sqrt(sigma_y)/805 minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:inLieuOfFrame true ;
                      nltl:frameWebThickness ?webQuantity ;
                      nltl:adjacentFrameHeight ?heightQuantity ;
                      nltl:yieldStrength ?yieldQuantity .
                ?webQuantity qudt:numericValue ?web .
                ?heightQuantity qudt:numericValue ?height .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?web * ?web * 648025 < ?height * ?height * ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Profile frame web thickness is below the h_w*sqrt(sigma_y)/805 minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:inLieuOfFrame false ;
                      nltl:frameProfileType nltl:profileSection ;
                      nltl:frameWebThickness ?webQuantity ;
                      nltl:adjacentFrameHeight ?heightQuantity ;
                      nltl:yieldStrength ?yieldQuantity .
                ?webQuantity qudt:numericValue ?web .
                ?heightQuantity qudt:numericValue ?height .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?web * ?web * 648025 < ?height * ?height * ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Flat-bar frame web thickness is below the h_w*sqrt(sigma_y)/282 minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:inLieuOfFrame false ;
                      nltl:frameProfileType nltl:flatBarSection ;
                      nltl:frameWebThickness ?webQuantity ;
                      nltl:adjacentFrameHeight ?heightQuantity ;
                      nltl:yieldStrength ?yieldQuantity .
                ?webQuantity qudt:numericValue ?web .
                ?heightQuantity qudt:numericValue ?height .
                ?yieldQuantity qudt:numericValue ?yield .
                FILTER (?web * ?web * 79524 < ?height * ?height * ?yield)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Frame web thickness is below half of net shell-plating thickness minus corrosion abrasion addition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?webQuantity ;
                      nltl:netShellPlatingThickness ?netQuantity ;
                      nltl:corrosionAbrasionAddition ?additionQuantity .
                ?webQuantity qudt:numericValue ?web .
                ?netQuantity qudt:numericValue ?net .
                ?additionQuantity qudt:numericValue ?addition .
                FILTER (?web < ((?net - ?addition) / 2))
            }
        """
    ] .
<END_SHACL>
