gs:TRF053IceStringerShape a sh:NodeShape ;
    sh:targetClass nltl:iceStringer ;
    sh:property [
        sh:path nltl:locatedWithinIceBelt ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "locatedWithinIceBelt is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:locatedWithinIceBelt ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:locatedWithinIceBelt ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:icePressure ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:PressureValueShape
            ] ;
            sh:property [
                sh:path nltl:designIceLoadHeight ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:LengthMValueShape
            ] ;
            sh:property [
                sh:path nltl:designLineLoad ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:LineLoadValueShape
            ] ;
            sh:property [
                sh:path nltl:span ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:LengthMValueShape
            ] ;
            sh:property [
                sh:path nltl:frameMomentFactorM ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:UnitlessValueShape
            ] ;
            sh:property [
                sh:path nltl:yieldStrength ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:PressureValueShape
            ] ;
            sh:property [
                sh:path nltl:iceStringerDistributionFactorF6 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:F6ValueShape
            ] ;
            sh:property [
                sh:path nltl:iceStringerSafetyFactorF7 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:F7ValueShape
            ] ;
            sh:property [
                sh:path nltl:iceStringerShearFactorF8 ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:F8ValueShape
            ] ;
            sh:property [
                sh:path nltl:sectionModulus ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:SectionModulusValueShape
            ] ;
            sh:property [
                sh:path nltl:requiredShearArea ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node gs:ShearAreaValueShape
            ] ;
            sh:sparql [
                sh:message "frameMomentFactorM and yieldStrength must be non-zero to calculate required section modulus and shear area." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:locatedWithinIceBelt true ;
                              nltl:frameMomentFactorM ?mNode ;
                              nltl:yieldStrength ?sigmaNode .
                        ?mNode qudt:numericValue ?m .
                        ?sigmaNode qudt:numericValue ?sigma .
                        FILTER (?m = 0 || ?sigma = 0)
                    }
                """
            ] ;
            sh:sparql [
                sh:message "Design line load, section modulus, and required shear area must conform to the TRF-053 formulae within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:locatedWithinIceBelt true ;
                              nltl:icePressure ?pressureNode ;
                              nltl:designIceLoadHeight ?heightNode ;
                              nltl:designLineLoad ?lineLoadNode ;
                              nltl:span ?spanNode ;
                              nltl:frameMomentFactorM ?mNode ;
                              nltl:yieldStrength ?sigmaNode ;
                              nltl:iceStringerDistributionFactorF6 ?f6Node ;
                              nltl:iceStringerSafetyFactorF7 ?f7Node ;
                              nltl:iceStringerShearFactorF8 ?f8Node ;
                              nltl:sectionModulus ?sectionNode ;
                              nltl:requiredShearArea ?areaNode .
                        ?pressureNode qudt:numericValue ?pressure .
                        ?heightNode qudt:numericValue ?height .
                        ?lineLoadNode qudt:numericValue ?lineLoad .
                        ?spanNode qudt:numericValue ?span .
                        ?mNode qudt:numericValue ?m .
                        ?sigmaNode qudt:numericValue ?sigma .
                        ?f6Node qudt:numericValue ?f6 .
                        ?f7Node qudt:numericValue ?f7 .
                        ?f8Node qudt:numericValue ?f8 .
                        ?sectionNode qudt:numericValue ?section .
                        ?areaNode qudt:numericValue ?area .
                        FILTER (?m != 0 && ?sigma != 0)
                        BIND(1000 * IF((?pressure * ?height) < 0.15, 0.15, (?pressure * ?height)) AS ?expectedLineLoad)
                        BIND((?f6 * ?f7 * (?lineLoad / 1000) * ?span * ?span / (?m * ?sigma) * 1000000) AS ?expectedSection)
                        BIND((1.7320508075688772 * ?f6 * ?f7 * ?f8 * (?lineLoad / 1000) * ?span / (2 * ?sigma) * 10000) AS ?expectedArea)
                        FILTER (
                            ABS(?lineLoad - ?expectedLineLoad) > ((ABS(?expectedLineLoad) * 0.000001) + 0.000000001) ||
                            ABS(?section - ?expectedSection) > ((ABS(?expectedSection) * 0.000001) + 0.000000001) ||
                            ABS(?area - ?expectedArea) > ((ABS(?expectedArea) * 0.000001) + 0.000000001)
                        )
                    }
                """
            ]
        ]
    ) .
