            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:locatedOutsideIceBelt true ;
                      nltl:supportsIceStrengthenedFrames true .

                OPTIONAL {
                    $this nltl:icePressure ?pQ .

                        qudt:unit unit:MegaPA .
                    FILTER (datatype(?p) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:designIceLoadHeight ?hQ .
                    ?hQ qudt:numericValue ?h ;
                        qudt:unit unit:M .
                    FILTER (datatype(?h) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:designLineLoad ?loadQ .
                    ?loadQ qudt:numericValue ?load ;
                        qudt:unit unit:KiloN-PER-M .
                    FILTER (datatype(?load) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:distanceToIceBelt ?hsQ .
                    ?hsQ qudt:numericValue ?hs ;
                        qudt:unit unit:M .
                    FILTER (datatype(?hs) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:distanceToAdjacentIceStringer ?lsQ .
                    ?lsQ qudt:numericValue ?ls ;
                        qudt:unit unit:M .
                    FILTER (datatype(?ls) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:stringerSpan ?lQ .
                    ?lQ qudt:numericValue ?l ;
                        qudt:unit unit:M .
                    FILTER (datatype(?l) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:frameMomentFactorM ?mQ .
                    ?mQ qudt:numericValue ?m ;
                        qudt:unit unit:UNITLESS .
                    FILTER (datatype(?m) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:yieldStrength ?sigmaQ .
                    ?sigmaQ qudt:numericValue ?sigma ;
                        qudt:unit unit:MegaPA .
                    FILTER (datatype(?sigma) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:outsideIceBeltDistributionFactorF9 ?f9Q .
                    ?f9Q qudt:numericValue ?f9 ;
                        qudt:unit unit:UNITLESS .
                    FILTER (datatype(?f9) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:outsideIceBeltSafetyFactorF10 ?f10Q .
                    ?f10Q qudt:numericValue ?f10 ;
                        qudt:unit unit:UNITLESS .
                    FILTER (datatype(?f10) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:outsideIceBeltShearFactorF11 ?f11Q .
                    ?f11Q qudt:numericValue ?f11 ;
                        qudt:unit unit:UNITLESS .
                    FILTER (datatype(?f11) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:outsideIceBeltFactor ?factorQ .
                    ?factorQ qudt:numericValue ?factor ;
                        qudt:unit unit:UNITLESS .
                    FILTER (datatype(?factor) = xsd:decimal)
                }
                OPTIONAL {
                    $this nltl:sectionModulus ?zQ .
                    ?zQ qudt:numericValue ?z ;

                    $this nltl:requiredShearArea ?aQ .
                    ?aQ qudt:numericValue ?a ;
                        qudt:unit unit:CentiM2 .
                    FILTER (datatype(?a) = xsd:decimal)

                BIND(IF((?p * ?h) < 0.15, 150.0, (?p * ?h * 1000.0)) AS ?expectedLoad)

                    !BOUND(?hs) || !BOUND(?ls) || !BOUND(?l) || !BOUND(?m) || !BOUND(?sigma) || !BOUND(?f9) ||

                    (?f9 != 0.80) || (?f10 != 1.8) || (?f11 != 1.2) ||
                    (ABS(?load - ?expectedLoad) > ((ABS(?expectedLoad) * 0.000001) + 0.000000001)) ||
                    (ABS(?factor - ?expectedFactor) > ((ABS(?expectedFactor) * 0.000001) + 0.000000001)) ||
