            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:icePressure ?pressureQuantity ;
                      nltl:iceLoadHeight ?heightQuantity ;
                      nltl:webFrameSpacing ?spacingQuantity ;
                      nltl:loadLengthParameter ?lengthQuantity ;
                      nltl:minimumLineLoad ?lineQuantity ;
                      nltl:webFrameSafetyFactor ?factorQuantity ;
                      nltl:webFrameIceLoad ?loadQuantity .

                ?pressureQuantity qudt:numericValue ?pressure .
                ?heightQuantity qudt:numericValue ?height .
                ?spacingQuantity qudt:numericValue ?spacing .
                ?lengthQuantity qudt:numericValue ?length .
                ?lineQuantity qudt:numericValue ?line .
                ?factorQuantity qudt:numericValue ?factor .
                ?loadQuantity qudt:numericValue ?load .

                BIND(?pressure * ?height AS ?pressureHeight)
                BIND(IF(?pressureHeight < 0.15, 0.15, ?pressureHeight) AS ?expectedLine)
                BIND(2 * ?spacing AS ?expectedLength)
                BIND(?factor * ?expectedLine * ?spacing * 1000 AS ?expectedLoad)

                FILTER(
                    ABS(?length - ?expectedLength) > (0.000001 * ABS(?expectedLength) + 0.000000001) ||
                    ABS(?line - ?expectedLine) > (0.000001 * ABS(?expectedLine) + 0.000000001) ||
                    ABS(?load - ?expectedLoad) > (0.000001 * ABS(?expectedLoad) + 0.000000001)
                )
            }
        """
    ] .
