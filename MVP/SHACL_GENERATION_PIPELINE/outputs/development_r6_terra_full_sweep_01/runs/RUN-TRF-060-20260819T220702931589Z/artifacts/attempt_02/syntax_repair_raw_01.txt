        sh:message "The section-modulus formula has an undefined domain: actual area and yield strength must be non-zero, and the square-root denominator must be positive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:webFrameFreeFlangeArea/qudt:numericValue ?af ;
                      nltl:webFrameWebArea/qudt:numericValue ?aw ;
                      nltl:yieldStrength/qudt:numericValue ?yield ;
                      nltl:hasTableLookupCase/nltl:webFrameTable4Dash8GammaFactor/qudt:numericValue ?gamma ;
                      nltl:requiredShearArea/qudt:numericValue ?requiredArea .

                BIND((?af + ?aw) AS ?expectedActualArea)
                BIND(
                    IF(
                        ?expectedActualArea = 0,
                        -1,
                        1 - ((?gamma * ?requiredArea / ?expectedActualArea) *
                             (?gamma * ?requiredArea / ?expectedActualArea))

        sh:message "Web-frame bending moment, actual area, and section modulus must match the prescribed calculation within a relative tolerance of 1e-6 plus an absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:webFrameIceLoad/qudt:numericValue ?force ;
                      nltl:webFrameSpan/qudt:numericValue ?span ;
                      nltl:webFrameMaximumIceLoadBendingMoment/qudt:numericValue ?reportedMoment ;
                      nltl:webFrameFreeFlangeArea/qudt:numericValue ?freeFlangeArea ;
                      nltl:webFrameWebArea/qudt:numericValue ?webArea ;
                      nltl:actualWebFrameCrossSectionalArea/qudt:numericValue ?reportedActualArea ;
                      nltl:requiredShearArea/qudt:numericValue ?requiredArea ;
                      nltl:yieldStrength/qudt:numericValue ?yield ;
                      nltl:sectionModulus/qudt:numericValue ?reportedSectionModulus ;
                      nltl:hasTableLookupCase/nltl:webFrameTable4Dash8GammaFactor/qudt:numericValue ?gamma .

                BIND((0.000193 * ?force * ?span) AS ?expectedMoment)
                BIND((?freeFlangeArea + ?webArea) AS ?expectedActualArea)
                FILTER(?expectedActualArea != 0 && ?yield != 0)

                BIND(
                    (1 - ((?gamma * ?requiredArea / ?expectedActualArea) *
                         (?gamma * ?requiredArea / ?expectedActualArea)))
                    AS ?denominator
                )
                FILTER(?denominator > 0)

                BIND(
                    ((?expectedMoment / ?yield) * SQRT(1 / ?denominator) * 1000000)
                    AS ?expectedSectionModulus
                )

                BIND(
                    (0.000001 * IF(?expectedMoment < 0, -?expectedMoment, ?expectedMoment) + 0.000000001)
                    AS ?momentTolerance
                )
                BIND(
                    (0.000001 * IF(?expectedActualArea < 0, -?expectedActualArea, ?expectedActualArea) + 0.000000001)
                    AS ?areaTolerance
                )
                BIND(
                    (0.000001 * IF(?expectedSectionModulus < 0, -?expectedSectionModulus, ?expectedSectionModulus) + 0.000000001)
                    AS ?sectionModulusTolerance
                )

                FILTER(
                    ?reportedMoment < (?expectedMoment - ?momentTolerance) ||
                    ?reportedMoment > (?expectedMoment + ?momentTolerance) ||
                    ?reportedActualArea < (?expectedActualArea - ?areaTolerance) ||
                    ?reportedActualArea > (?expectedActualArea + ?areaTolerance) ||
                    ?reportedSectionModulus < (?expectedSectionModulus - ?sectionModulusTolerance) ||
                    ?reportedSectionModulus > (?expectedSectionModulus + ?sectionModulusTolerance)
                )
            }
        """
    ] .
