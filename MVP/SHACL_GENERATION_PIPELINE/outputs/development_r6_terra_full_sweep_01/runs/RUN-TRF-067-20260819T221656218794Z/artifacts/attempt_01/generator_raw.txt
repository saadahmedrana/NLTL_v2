<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:TRF067RudderProtectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "An ice class is required to determine applicability of the rudder protection requirement."
    ] ;
    sh:sparql [
        sh:message "Ships of ice class IA or IA Super require a rudder protection arrangement to absorb rudder loads when sailing astern in ice or ice ridges." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass .
                FILTER (?iceClass IN (nltl:iceClassIa, nltl:iceClassIaSuper))
                FILTER NOT EXISTS {
                    $this nltl:hasRudderProtectionArrangement ?arrangement .
                    ?arrangement a nltl:rudderProtectionArrangement .
                }
            }
        """
    ] .
<END_SHACL>
