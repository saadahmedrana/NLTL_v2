<urn:nltl:generated-shape:TRF-070-LoadAdequacyShape>
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:mainPropulsionThruster true ;
                    nltl:thrusterType ?thrusterType .
                FILTER (?thrusterType IN ("azimuthing", "fixed"))
            }
        """
    ] ;
    sh:sparql [
        sh:message "Expected load shall cover the propeller/ice and thruster-body/ice interaction loads, and local strength capacity shall be at least the expected load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerIceInteractionLoad ?propellerLoad ;
                      nltl:thrusterBodyIceInteractionLoad ?bodyLoad ;
                      nltl:hasDesignCondition ?condition .
                ?condition nltl:normalOperationalConditions true ;
                           nltl:designConditionExpectedLoad ?expectedLoad ;
                           nltl:designConditionLocalStrengthCapacity ?capacity .
                ?propellerLoad qudt:numericValue ?propellerValue ;
                               qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?bodyLoad qudt:numericValue ?bodyValue ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?expectedLoad qudt:numericValue ?expectedValue ;
                              qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                ?capacity qudt:numericValue ?capacityValue ;
                          qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                FILTER (
                    ?expectedValue < ?propellerValue ||
                    ?expectedValue < ?bodyValue ||
                    ?capacityValue < ?expectedValue
                )
            }
        """
    ] .
