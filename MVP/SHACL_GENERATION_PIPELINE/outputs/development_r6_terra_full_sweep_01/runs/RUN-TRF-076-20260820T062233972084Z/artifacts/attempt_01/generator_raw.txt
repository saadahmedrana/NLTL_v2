<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-076ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A propeller type is required to determine the applicable blade load cases."
    ] ;
    sh:property [
        sh:path nltl:propellerReversingCapability ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase
    ] ;
    sh:sparql [
        sh:message "CP and FP propellers must cover blade load cases 1 through 4." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:propellerType ?propellerType .
                FILTER(?propellerType IN ("CP", "FP"))
                OPTIONAL {
                    $this nltl:hasPropellerBladeLoadCase ?loadCase .
                    ?loadCase nltl:propellerBladeLoadCaseNumber ?caseNumber .
                    FILTER(?caseNumber IN (1, 2, 3, 4))
                }
            }
            GROUP BY $this
            HAVING (COUNT(DISTINCT ?caseNumber) < 4)
        """
    ] ;
    sh:sparql [
        sh:message "An FP propeller requires a reversing-capability value to determine whether load case 5 is required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:propellerType "FP" .
                FILTER NOT EXISTS {
                    $this nltl:propellerReversingCapability ?reversingCapability .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A reversing FP propeller must cover blade load case 5." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:propellerType "FP" ;
                      nltl:propellerReversingCapability true .
                FILTER NOT EXISTS {
                    $this nltl:hasPropellerBladeLoadCase ?loadCase .
                    ?loadCase nltl:propellerBladeLoadCaseNumber 5 .
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-076PropellerBladeLoadCaseShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasPropellerBladeLoadCase ;
    sh:class nltl:propellerBladeLoadCase ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:message "Each linked propeller blade load case requires a load case number."
    ] .
<END_SHACL>
