<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF077PropellerTypeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string
    ] .

gen:TRF077ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:TRF077OpenPropellerSpectrumShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:propellerType "open propeller" .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:loadSpectrumShapeParameter ;
        sh:minCount 1 ;
        sh:message "An open propeller requires a load spectrum shape parameter of 0.75." ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:minInclusive "0.75"^^xsd:decimal ;
                sh:maxInclusive "0.75"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] .

gen:TRF077BladeIceLoadEnvelopeShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:propellerType ?type .
                FILTER (?type IN ("open propeller", "ducted propeller blade"))
            }
        """
    ] ;
    sh:property [
        sh:path nltl:bladeIceLoad ;
        sh:minCount 1 ;
        sh:message "Blade ice load is required to establish the ice-load envelope." ;
        sh:node gen:TRF077ForceQuantityShape
    ] ;
    sh:property [
        sh:path nltl:maximumBladeIceLoad ;
        sh:minCount 1 ;
        sh:message "Maximum blade ice load is required to establish the ice-load envelope." ;
        sh:node gen:TRF077ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "Blade ice load must be greater than or equal to zero and must not exceed maximum blade ice load." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:bladeIceLoad ?load ;
                      nltl:maximumBladeIceLoad ?maximum .
                ?load qudt:numericValue ?loadValue .
                ?maximum qudt:numericValue ?maximumValue .
                FILTER (?loadValue < 0 || ?loadValue > ?maximumValue)
            }
        """
    ] .
<END_SHACL>
