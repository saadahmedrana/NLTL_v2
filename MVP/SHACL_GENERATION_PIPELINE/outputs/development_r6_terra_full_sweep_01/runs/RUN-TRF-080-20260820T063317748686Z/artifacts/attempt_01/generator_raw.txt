<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-080IceLoadCycleShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceLoadCycleCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:propellerBladeCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:effectiveIceLoadCycleCount ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Propeller/ice interaction components require valid ice load cycle, blade count, and effective ice load cycle inputs." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component .
                ?component a nltl:propellerIceInteractionComponent .
                FILTER NOT EXISTS {
                    $this nltl:iceLoadCycleCount ?nice ;
                          nltl:propellerBladeCount ?blades ;
                          nltl:effectiveIceLoadCycleCount ?effective .
                    ?effective a qudt:QuantityValue ;
                               qudt:numericValue ?reported ;
                               qudt:unit unit:UNITLESS .
                    FILTER(datatype(?nice) = xsd:integer)
                    FILTER(datatype(?blades) = xsd:integer)
                    FILTER(datatype(?reported) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Effective ice load cycle count must equal ice load cycle count multiplied by propeller blade count." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?component ;
                      nltl:iceLoadCycleCount ?nice ;
                      nltl:propellerBladeCount ?blades ;
                      nltl:effectiveIceLoadCycleCount ?effective .
                ?component a nltl:propellerIceInteractionComponent .
                ?effective a qudt:QuantityValue ;
                           qudt:numericValue ?reported ;
                           qudt:unit unit:UNITLESS .
                FILTER(datatype(?nice) = xsd:integer)
                FILTER(datatype(?blades) = xsd:integer)
                FILTER(datatype(?reported) = xsd:decimal)
                BIND(xsd:decimal(?nice) * xsd:decimal(?blades) AS ?expected)
                BIND(ABS(?expected) * 0.000001 AS ?relativeTolerance)
                BIND(IF(?relativeTolerance < 0.000001, 0.000001, ?relativeTolerance) AS ?tolerance)
                FILTER(ABS(xsd:decimal(?reported) - ?expected) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
