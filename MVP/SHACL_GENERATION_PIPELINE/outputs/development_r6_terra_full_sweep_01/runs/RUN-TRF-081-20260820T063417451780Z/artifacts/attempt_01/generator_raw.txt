<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:KiloNewtonQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF081PropellerShaftDesignThrustShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:forwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardBladeForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hydrodynamicBollardThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardIceThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:forwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:backwardPropellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerShaftDesignThrust ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:KiloNewtonQuantityShape
    ] ;
    sh:sparql [
        sh:message "Ice thrust and propeller shaft design thrust values must conform to the required formulae." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:forwardBladeForce ?ff ;
                      nltl:backwardBladeForce ?fb ;
                      nltl:hydrodynamicBollardThrust ?t ;
                      nltl:forwardIceThrust ?tf ;
                      nltl:backwardIceThrust ?tb ;
                      nltl:forwardPropellerShaftDesignThrust ?forwardDesign ;
                      nltl:backwardPropellerShaftDesignThrust ?backwardDesign ;
                      nltl:propellerShaftDesignThrust ?design .

                ?ff qudt:numericValue ?ffValue .
                ?fb qudt:numericValue ?fbValue .
                ?t qudt:numericValue ?tValue .
                ?tf qudt:numericValue ?tfValue .
                ?tb qudt:numericValue ?tbValue .
                ?forwardDesign qudt:numericValue ?forwardDesignValue .
                ?backwardDesign qudt:numericValue ?backwardDesignValue .
                ?design qudt:numericValue ?designValue .

                BIND((1.1 * ?ffValue) AS ?expectedTf)
                BIND((1.1 * ?fbValue) AS ?expectedTb)
                BIND((?tValue + (2.2 * ?tfValue)) AS ?expectedForwardDesign)
                BIND((1.5 * ?tbValue) AS ?expectedBackwardDesign)
                BIND(IF(?forwardDesignValue >= ?backwardDesignValue,
                        ?forwardDesignValue,
                        ?backwardDesignValue) AS ?expectedDesign)

                FILTER (
                    ABS(?tfValue - ?expectedTf) >
                        ((0.000001 * ABS(?expectedTf)) + 0.000000001) ||
                    ABS(?tbValue - ?expectedTb) >
                        ((0.000001 * ABS(?expectedTb)) + 0.000000001) ||
                    ABS(?forwardDesignValue - ?expectedForwardDesign) >
                        ((0.000001 * ABS(?expectedForwardDesign)) + 0.000000001) ||
                    ABS(?backwardDesignValue - ?expectedBackwardDesign) >
                        ((0.000001 * ABS(?expectedBackwardDesign)) + 0.000000001) ||
                    ABS(?designValue - ?expectedDesign) >
                        ((0.000001 * ABS(?expectedDesign)) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
