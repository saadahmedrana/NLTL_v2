<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MetreQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/M>
    ] .

gen:RpmQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN>
    ] .

gen:LookupQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] .

gen:TableLookupCaseShape a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash9
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:node gen:LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:minCount 1 ;
        sh:node gen:LookupQuantityShape
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact
    ] .

gen:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node gen:TableLookupCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node gen:RpmQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:node gen:MetreQuantityShape
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node gen:RpmQuantityShape
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For an unknown bollard propeller speed, the Table 6-9 lookup input and result shall correspond to the propeller diameter and reported rotational speed." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:bollardPropellerSpeedKnown false ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speedQuantity ;
                      nltl:hasTableLookupCase ?lookupCase .
                ?diameterQuantity qudt:numericValue ?diameterValue .
                ?speedQuantity qudt:numericValue ?speedValue .
                ?lookupCase nltl:lookupInputQuantity ?inputQuantity ;
                            nltl:lookupResultQuantity ?resultQuantity .
                ?inputQuantity qudt:numericValue ?inputValue .
                ?resultQuantity qudt:numericValue ?resultValue .
                FILTER (?inputValue != ?diameterValue || ?resultValue != ?speedValue)
            }
        """
    ] .
<END_SHACL>
