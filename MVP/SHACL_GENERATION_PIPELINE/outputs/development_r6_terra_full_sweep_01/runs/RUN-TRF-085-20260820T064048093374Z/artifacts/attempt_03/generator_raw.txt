<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF085Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node gen:TRF085BollardSpeedQuantityShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:minCount 1 ;
                sh:node gen:TRF085PropellerDiameterQuantityShape
            ] ;
            sh:property [
                sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
                sh:minCount 1 ;
                sh:node gen:TRF085BollardSpeedQuantityShape
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:class nltl:tableLookupCase ;
                sh:node gen:TRF085TableLookupCaseShape
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each Table 6-9 lookup case must use the ship propeller diameter as its input and the ship bollard MCR propeller speed as its result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:bollardPropellerSpeedKnown false ;
                      nltl:hasTableLookupCase ?case .

                FILTER (
                    NOT EXISTS {
                        ?case nltl:lookupInputQuantity ?input .
                        $this nltl:propellerDiameter ?input .
                    }
                    ||
                    NOT EXISTS {
                        ?case nltl:lookupResultQuantity ?result .
                        $this nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?result .
                    }
                )
            }
        """
    ] .

gen:TRF085TableLookupCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:minCount 1 ;
        sh:hasValue nltl:traficomTable6Dash9
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:minCount 1 ;
        sh:class qudt:QuantityValue
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact
    ] .
