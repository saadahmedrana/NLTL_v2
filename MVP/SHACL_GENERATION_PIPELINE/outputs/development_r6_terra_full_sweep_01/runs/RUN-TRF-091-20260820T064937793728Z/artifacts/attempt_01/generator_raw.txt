<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:generated-shape:TRF-091Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasIceMillingSequenceCase ;
        sh:node [
            sh:class nltl:iceMillingSequenceCase ;
            sh:class nltl:loadCase ;
            sh:property [
                sh:path nltl:loadCasePurpose ;
                sh:nodeKind sh:IRI ;
                sh:class nltl:loadCasePurposeValue ;
                sh:hasValue nltl:propulsionLineStrengthEvaluationPurpose ;
                sh:message "An ice-milling load case shall be used for propulsion-line strength evaluation."
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each ice-milling sequence case shall also be linked as a propulsion load case."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasIceMillingSequenceCase ?case .
                FILTER NOT EXISTS {
                    $this nltl:hasPropulsionLoadCase ?case .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "An ice-milling load sequence shall not be assigned a stalling-analysis purpose."
        ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasIceMillingSequenceCase ?case .
                ?case nltl:loadCasePurpose nltl:stallingAnalysisPurposeValue .
            }
        """
    ] .
<END_SHACL>
