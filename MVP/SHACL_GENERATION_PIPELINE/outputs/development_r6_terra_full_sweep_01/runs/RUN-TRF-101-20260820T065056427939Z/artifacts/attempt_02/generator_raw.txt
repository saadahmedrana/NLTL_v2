    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:minCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:minInclusive "0.8"^^xsd:decimal ;
                sh:maxInclusive "0.8"^^xsd:decimal

                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;

                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;

    ] .

gs:TRF101FormulaShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:minCount 1 ;
        sh:node gs:TRF101PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:minCount 1 ;
        sh:node gs:TRF101PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:minCount 1 ;
        sh:node gs:TRF101PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF101ForceQuantityShape
    ] ;
    sh:sparql [
        sh:message "bladeReferenceStrength1 must equal 0.6 times bladeMinimumYieldStrength plus 0.4 times bladeUltimateTensileStrength within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeMinimumYieldStrength ?yieldQuantity ;
                      nltl:bladeUltimateTensileStrength ?ultimateQuantity ;
                      nltl:bladeReferenceStrength1 ?referenceQuantity .
                ?yieldQuantity qudt:numericValue ?yield .
                ?ultimateQuantity qudt:numericValue ?ultimate .
                ?referenceQuantity qudt:numericValue ?reference .
                BIND ((0.6 * ?yield) + (0.4 * ?ultimate) AS ?expected)
                FILTER (
                    ABS(?reference - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "The blade-failure load formula denominator 0.8 times propellerDiameter minus 2 times bladeCylindricalRootRadius must not be zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:propellerDiameter ?diameterQuantity ;
                      nltl:bladeCylindricalRootRadius ?rootQuantity .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?rootQuantity qudt:numericValue ?rootRadius .
                FILTER ((0.8 * ?diameter) - (2 * ?rootRadius) = 0)
            }
        """
    ] ;
    sh:sparql [
        sh:message "bladeFailureUltimateLoad must equal 300 times chord times thickness squared times reference strength divided by (0.8 times diameter minus 2 times root radius), within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?chordQuantity ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?thicknessQuantity ;
                      nltl:bladeReferenceStrength1 ?referenceQuantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:bladeCylindricalRootRadius ?rootQuantity ;
                      nltl:bladeFailureUltimateLoad ?loadQuantity .
                ?chordQuantity qudt:numericValue ?chord .
                ?thicknessQuantity qudt:numericValue ?thickness .
                ?referenceQuantity qudt:numericValue ?reference .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?rootQuantity qudt:numericValue ?rootRadius .
                ?loadQuantity qudt:numericValue ?load .
                BIND ((0.8 * ?diameter) - (2 * ?rootRadius) AS ?denominator)
                FILTER (?denominator != 0)
                BIND ((300 * ?chord * ?thickness * ?thickness * ?reference) / ?denominator AS ?expected)
                FILTER (
                    ABS(?load - ?expected) >
                    ((ABS(?expected) * 0.000001) + 0.000000001)
                )
            }
        """
    ] .

gs:TRF101Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:node gs:TRF101LengthQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node gs:TRF101LoadCaseShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedNonlinearBladeStressAnalysis ;
                sh:minCount 1 ;
                sh:class nltl:approvalRecord
            ]
        ]
        gs:TRF101FormulaShape
    ) ;
    sh:sparql [
        sh:message "bladeFailureStatus must be true when bladeTipOffset exceeds 10 percent of propellerDiameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset ?offsetQuantity ;
                      nltl:propellerDiameter ?diameterQuantity .
                ?offsetQuantity qudt:numericValue ?offset .
                ?diameterQuantity qudt:numericValue ?diameter .
                FILTER (?offset > (0.10 * ?diameter))
                FILTER NOT EXISTS {
                    $this nltl:bladeFailureStatus true .
                }
            }
        """
    ] .
