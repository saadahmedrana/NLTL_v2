<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-111-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each propeller shaft line component requires a fatigue safety factor of at least 1.5 when a fatigue load case is represented." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase ;
                      nltl:hasPropellerShaftLineComponent ?component .
                ?loadCase nltl:shaftLineLoadCaseType nltl:fatigueLoadCase .
                FILTER NOT EXISTS {
                    ?component nltl:fatigueSafetyFactor ?factor .
                    ?factor a qudt:QuantityValue ;
                            qudt:numericValue ?value ;
                            qudt:unit unit:UNITLESS .
                    FILTER (datatype(?value) = xsd:decimal && ?value >= "1.5"^^xsd:decimal)
                }
            }
        """
    ] .

<urn:nltl:generated-shape:TRF-111-ComponentYieldSafetyFactorShape>
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:componentYieldSafetyFactor ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-111-FatigueSafetyFactorShape>
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:fatigueSafetyFactor ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:generated-shape:TRF-111-LoadCaseShape>
    a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasPropulsionLoadCase ;
    sh:sparql [
        sh:message "An extreme operational or blade failure load case requires a component yield safety factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:shaftLineLoadCaseType ?caseType .
                FILTER (?caseType IN (nltl:extremeOperationalLoadCase, nltl:bladeFailureLoadCase))
                FILTER NOT EXISTS {
                    $this nltl:componentYieldSafetyFactor ?factor .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Component yield safety factor must be at least 1.3 for extreme operational loads and at least 1.0 for blade failure loads." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shaftLineLoadCaseType ?caseType ;
                      nltl:componentYieldSafetyFactor ?factor .
                ?factor qudt:numericValue ?value .
                FILTER (
                    (?caseType = nltl:extremeOperationalLoadCase &&
                     datatype(?value) = xsd:decimal &&
                     ?value < "1.3"^^xsd:decimal)
                    ||
                    (?caseType = nltl:bladeFailureLoadCase &&
                     datatype(?value) = xsd:decimal &&
                     ?value < "1.0"^^xsd:decimal)
                )
            }
        """
    ] .
<END_SHACL>
