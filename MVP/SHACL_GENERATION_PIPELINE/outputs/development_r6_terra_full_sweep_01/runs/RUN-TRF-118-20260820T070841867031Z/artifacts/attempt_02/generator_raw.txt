<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF118ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thrusterExtremeLoad ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:KiloNQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:yieldSafetyFactor ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:SafetyFactorShape>
    ] ;
    sh:property [
        sh:path nltl:nominalVonMisesStress ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:MegaPAQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:localStressConcentration ;
        sh:node <urn:nltl:generated-shape:UnitlessQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hasFactoredExtremeLoadOperabilityAssessment ;
        sh:minCount 1 ;
        sh:class nltl:factoredExtremeLoadOperabilityAssessment ;
        sh:node <urn:nltl:generated-shape:FactoredExtremeLoadOperabilityAssessmentShape>
    ] ;
    sh:sparql [
        sh:message "Thruster stress margins and the assessed factored extreme load must conform to the required calculation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thrusterExtremeLoad ?extremeLoad ;
                      nltl:yieldSafetyFactor ?safetyFactor ;
                      nltl:nominalVonMisesStress ?nominalStress ;
                      nltl:hasFactoredExtremeLoadOperabilityAssessment ?assessment .
                ?extremeLoad qudt:numericValue ?extreme .
                ?safetyFactor qudt:numericValue ?factor .
                ?nominalStress qudt:numericValue ?nominal .
                ?assessment nltl:assessedFactoredExtremeLoad ?factoredLoad ;
                            nltl:assessedThrusterComponent ?component .
                ?factoredLoad qudt:numericValue ?factored .
                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldStrength .
                ?yieldStrength qudt:numericValue ?yield .
                OPTIONAL {
                    $this nltl:localStressConcentration ?localConcentration .
                    ?localConcentration qudt:numericValue ?local .
                }
                FILTER (
                    (
                        (!BOUND(?local) && (?nominal * 1.3 > ?yield)) ||
                        (BOUND(?local) && (?nominal * ?local > ?yield))
                    ) ||
                    (
                        ABS(?factored - (?extreme * ?factor)) >
                        ((ABS(?extreme * ?factor) * 0.000001) + 0.000000001)
                    )
                )
            }
        """
    ] .

<urn:nltl:generated-shape:FactoredExtremeLoadOperabilityAssessmentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:assessedThrusterComponent ;
        sh:minCount 1 ;
        sh:class nltl:shipComponent ;
        sh:node <urn:nltl:generated-shape:AssessedThrusterComponentShape>
    ] ;
    sh:property [
        sh:path nltl:assessedFactoredExtremeLoad ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:KiloNQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:assessmentOperabilityMaintained ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:assessmentRepairRequired ;
        sh:minCount 1 ;
        sh:datatype xsd:boolean ;
        sh:in ( false )
    ] .

<urn:nltl:generated-shape:AssessedThrusterComponentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:class nltl:materialProperties ;
        sh:node <urn:nltl:generated-shape:MaterialPropertiesShape>
    ] .

<urn:nltl:generated-shape:MaterialPropertiesShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:MegaPAQuantityShape>
    ] .

<urn:nltl:generated-shape:KiloNQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

<urn:nltl:generated-shape:MegaPAQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:MegaPA
    ] .

<urn:nltl:generated-shape:UnitlessQuantityShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .

<urn:nltl:generated-shape:SafetyFactorShape>
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive 1.3 ;
        sh:maxInclusive 1.3
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .
<END_SHACL>
