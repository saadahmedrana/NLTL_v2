<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:MPaQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:KiloNQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UnitlessQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:SafetyFactorQuantityShape a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 1.3 ;
        sh:maxInclusive 1.3
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MaterialPropertiesShape a sh:NodeShape ;
    sh:class nltl:materialProperties ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:node gen:MPaQuantityShape
    ] .

gen:AssessedThrusterComponentShape a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node gen:MaterialPropertiesShape
    ] .

gen:FactoredExtremeLoadOperabilityAssessmentShape a sh:NodeShape ;
    sh:class nltl:factoredExtremeLoadOperabilityAssessment ;
    sh:property [
        sh:path nltl:assessedThrusterComponent ;
        sh:minCount 1 ;
        sh:node gen:AssessedThrusterComponentShape
    ] ;
    sh:property [
        sh:path nltl:assessedFactoredExtremeLoad ;
        sh:minCount 1 ;
        sh:node gen:KiloNQuantityShape
    ] ;
    sh:property [
        sh:path nltl:assessmentOperabilityMaintained ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true )
    ] ;
    sh:property [
        sh:path nltl:assessmentRepairRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( false )
    ] .

gen:TRF118Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:nominalVonMisesStress ;
        sh:minCount 1 ;
        sh:node gen:MPaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:localStressConcentration ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:thrusterExtremeLoad ;
        sh:minCount 1 ;
        sh:node gen:KiloNQuantityShape
    ] ;
    sh:property [
        sh:path nltl:yieldSafetyFactor ;
        sh:minCount 1 ;
        sh:node gen:SafetyFactorQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasFactoredExtremeLoadOperabilityAssessment ;
        sh:node gen:FactoredExtremeLoadOperabilityAssessmentShape
    ] ;
    sh:sparql [
        sh:message "Factored load, nominal stress safety margin, or local stress concentration criterion is not satisfied." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:thrusterExtremeLoad ?extremeLoadNode ;
                      nltl:yieldSafetyFactor ?factorNode ;
                      nltl:nominalVonMisesStress ?nominalNode ;
                      nltl:hasFactoredExtremeLoadOperabilityAssessment ?assessment .

                ?extremeLoadNode qudt:numericValue ?extreme .
                ?factorNode qudt:numericValue ?factor .
                ?nominalNode qudt:numericValue ?nominal .

                ?assessment nltl:assessedFactoredExtremeLoad ?assessedLoadNode ;
                            nltl:assessedThrusterComponent ?component .

                ?assessedLoadNode qudt:numericValue ?assessed .

                ?component nltl:hasMaterialProperties ?materialProperties .
                ?materialProperties nltl:componentMaterialYieldStrength ?yieldNode .
                ?yieldNode qudt:numericValue ?yield .

                OPTIONAL {
                    $this nltl:localStressConcentration ?localNode .
                    ?localNode qudt:numericValue ?local .
                }

                FILTER (
                    ABS(?assessed - (?extreme * ?factor)) >
                    ((0.000001 * ABS(?extreme * ?factor)) + 0.000000001)
                    ||
                    (?nominal * ?factor > ?yield)
                    ||
                    (BOUND(?local) && (?nominal * ?local > ?yield))
                )
            }
        """
    ] .
<END_SHACL>
