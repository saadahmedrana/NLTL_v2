<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:PressureQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:AssessedComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:minCount 1 ;
        sh:node shape:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:materialYieldStrength ;
        sh:minCount 1 ;
        sh:node shape:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:materialProperties ;
            sh:property [
                sh:path nltl:componentMaterialYieldStrength ;
                sh:minCount 1 ;
                sh:node shape:PressureQuantityShape
            ]
        ]
    ] .

shape:OccasionalForceLoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:propellerBladeExcludedFromOccasionalForceScope ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:occasionalForceCaseAssessedComponent ;
                sh:minCount 1 ;
                sh:node shape:AssessedComponentShape
            ] ;
            sh:property [
                sh:path nltl:componentYieldSafetyFactor ;
                sh:minCount 1 ;
                sh:node shape:DimensionlessQuantityShape
            ] ;
            sh:sparql [
                sh:message "Occasional-force component stress must not exceed the component material yield strength." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:occasionalForceCaseAssessedComponent ?component .
                        ?component nltl:occasionalForceComponentStress ?stressQuantity ;
                                   nltl:hasMaterialProperties ?materialProperties .
                        ?stressQuantity qudt:numericValue ?stress .
                        ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .
                        ?yieldQuantity qudt:numericValue ?yield .
                        FILTER (?stress > ?yield)
                    }
                """
            ]
        ]
    ) .

shape:TRF123Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:class nltl:occasionalForceLoadCase ;
        sh:node shape:OccasionalForceLoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:class nltl:propellerShaftLineComponent
    ] .
<END_SHACL>
