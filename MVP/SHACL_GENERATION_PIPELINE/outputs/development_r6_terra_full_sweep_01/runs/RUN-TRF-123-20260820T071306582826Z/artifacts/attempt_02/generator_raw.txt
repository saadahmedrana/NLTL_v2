<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF123ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasOccasionalForceLoadCase ;
        sh:node <urn:nltl:generated-shape:TRF123OccasionalForceLoadCaseShape>
    ] .

<urn:nltl:generated-shape:TRF123OccasionalForceLoadCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:occasionalForceCaseAssessedComponent ;
        sh:node <urn:nltl:generated-shape:TRF123AssessedComponentShape>
    ] ;
    sh:property [
        sh:path nltl:componentYieldSafetyFactor ;
        sh:node <urn:nltl:generated-shape:TRF123SafetyFactorQuantityShape>
    ] ;
    sh:sparql [
        sh:message "An assessed component requires a component yield safety factor." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:occasionalForceCaseAssessedComponent ?component .
                FILTER NOT EXISTS {
                    $this nltl:componentYieldSafetyFactor ?factorQuantity .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Assessed component stress multiplied by the component yield safety factor must not exceed the linked material yield strength." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?component
            WHERE {
                $this nltl:occasionalForceCaseAssessedComponent ?component ;
                      nltl:componentYieldSafetyFactor ?factorQuantity .

                ?factorQuantity qudt:numericValue ?factor ;
                                qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .

                ?component nltl:occasionalForceComponentStress ?stressQuantity ;
                           nltl:hasMaterialProperties ?materialProperties .

                ?stressQuantity qudt:numericValue ?stress ;
                                qudt:unit <http://qudt.org/vocab/unit/MegaPA> .

                ?materialProperties nltl:componentMaterialYieldStrength ?yieldQuantity .

                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .

                FILTER (
                    (xsd:decimal(?stress) * xsd:decimal(?factor))
                    > xsd:decimal(?yield)
                )
            }
        """
    ] .

<urn:nltl:generated-shape:TRF123AssessedComponentShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:occasionalForceComponentStress ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF123StressQuantityShape>
    ] ;
    sh:property [
        sh:path nltl:hasMaterialProperties ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF123MaterialPropertiesShape>
    ] .

<urn:nltl:generated-shape:TRF123MaterialPropertiesShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:componentMaterialYieldStrength ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:TRF123YieldStrengthQuantityShape>
    ] .

<urn:nltl:generated-shape:TRF123StressQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA>
    ] .

<urn:nltl:generated-shape:TRF123YieldStrengthQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA>
    ] .

<urn:nltl:generated-shape:TRF123SafetyFactorQuantityShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS>
    ] .
<END_SHACL>
