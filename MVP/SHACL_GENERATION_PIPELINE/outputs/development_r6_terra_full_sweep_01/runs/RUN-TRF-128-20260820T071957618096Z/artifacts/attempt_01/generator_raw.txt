<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:TRF128OneHourChargeQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive 1.0
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:HR
    ] .

shape:TRF128HalfHourChargeQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:maxInclusive 0.5
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:HR
    ] .

shape:TRF128AirCompressorChargeTimeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propulsionEngineReversalRequiredForAstern ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node shape:TRF128OneHourChargeQuantity ;
                sh:message "Compressor charge time must not exceed one hour."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node shape:TRF128OneHourChargeQuantity ;
                sh:message "Compressor charge time must not exceed one hour."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node shape:TRF128HalfHourChargeQuantity ;
                sh:message "Compressor charge time must not exceed half an hour for IA Super ships requiring propulsion-engine reversal for astern movement."
            ]
        ]
    ) .
<END_SHACL>
