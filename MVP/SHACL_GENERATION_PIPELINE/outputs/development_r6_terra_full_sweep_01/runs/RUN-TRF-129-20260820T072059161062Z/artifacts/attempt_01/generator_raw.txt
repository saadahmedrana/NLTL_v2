<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:TRF129CoolingWaterSupplyShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:navigatingInIce ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:iceNavigationCondition ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:coolingWaterSupplySecured ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "Cooling water supply must be secured when navigating in ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:navigatingInIce true .
                FILTER NOT EXISTS {
                    $this nltl:coolingWaterSupplySecured true .
                }
            }
        """
    ] .
<END_SHACL>
