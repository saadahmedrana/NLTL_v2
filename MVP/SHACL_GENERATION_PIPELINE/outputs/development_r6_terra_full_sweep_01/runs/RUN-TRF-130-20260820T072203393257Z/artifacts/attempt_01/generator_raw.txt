<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:TRF130CoolingWaterInletChestShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Record whether the cooling-water chest volume and height requirements cannot be met."
    ] ;
    sh:or (
        shape:TRF130StandardChestArrangementShape
        shape:TRF130AlternativeChestArrangementShape
    ) .

shape:TRF130StandardChestArrangementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:node shape:TRF130VolumeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:minCount 1 ;
        sh:node shape:TRF130PowerQuantityShape
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:node shape:TRF130AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:node shape:TRF130AreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "Chest volume shall be approximately total relevant engine power divided by 750, and strainer open area shall be at least four times inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:recommendedChestVolume ?volumeQuantity ;
                        nltl:totalRelevantEnginePower ?powerQuantity ;
                        nltl:strainerOpenArea ?strainerQuantity ;
                        nltl:inletPipeArea ?inletQuantity .
                    ?volumeQuantity qudt:numericValue ?volume ;
                        qudt:unit unit:M3 .
                    ?powerQuantity qudt:numericValue ?power ;
                        qudt:unit unit:KiloW .
                    ?strainerQuantity qudt:numericValue ?strainerArea ;
                        qudt:unit unit:M2 .
                    ?inletQuantity qudt:numericValue ?inletArea ;
                        qudt:unit unit:M2 .
                    FILTER (
                        ABS(?volume - (?power / 750)) <=
                        ((ABS(?power / 750) * 0.000001) + 0.000000001) &&
                        ?strainerArea >= (4 * ?inletArea)
                    )
                }
            }
        """
    ] .

shape:TRF130AlternativeChestArrangementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:coolingWaterChestAlternativeArrangementUsed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:node shape:TRF130AreaQuantityShape
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:node shape:TRF130AreaQuantityShape
    ] ;
    sh:sparql [
        sh:message "Strainer open area shall be at least four times inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:strainerOpenArea ?strainerQuantity ;
                        nltl:inletPipeArea ?inletQuantity .
                    ?strainerQuantity qudt:numericValue ?strainerArea ;
                        qudt:unit unit:M2 .
                    ?inletQuantity qudt:numericValue ?inletArea ;
                        qudt:unit unit:M2 .
                    FILTER (?strainerArea >= (4 * ?inletArea))
                }
            }
        """
    ] .

shape:TRF130VolumeQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M3 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:TRF130PowerQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

shape:TRF130AreaQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M2 ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
