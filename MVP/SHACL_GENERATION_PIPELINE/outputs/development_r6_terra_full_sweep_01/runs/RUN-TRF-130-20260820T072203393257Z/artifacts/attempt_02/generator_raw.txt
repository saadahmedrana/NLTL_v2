    ] ;

    sh:property [
        sh:path nltl:alternatingCoolingWaterIntakeAndDischarge ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;

    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1
    ] ;

    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;

            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;

    sh:sparql [
        sh:message "The strainer open area must be at least four times the inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:strainerOpenArea ?strainerQuantity ;
                      nltl:inletPipeArea ?inletQuantity .
                ?strainerQuantity qudt:numericValue ?strainerArea ;
                                  qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?inletQuantity qudt:numericValue ?inletArea ;
                                qudt:unit <http://qudt.org/vocab/unit/M2> .
                FILTER ( xsd:decimal(?strainerArea) < (4 * xsd:decimal(?inletArea)) )
            }
        """
    ] ;

    sh:or (
        [
            sh:property [
                sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;

            ] ;
            sh:property [

                sh:hasValue false
            ] ;

                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1

                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:totalRelevantEnginePower ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloW ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "The recommended chest volume must be approximately total relevant engine power divided by 750." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                    SELECT $this
                    WHERE {
                        $this nltl:recommendedChestVolume ?volumeQuantity ;
                              nltl:totalRelevantEnginePower ?powerQuantity .
                        ?volumeQuantity qudt:numericValue ?volume ;
                                        qudt:unit <http://qudt.org/vocab/unit/M3> .
                        ?powerQuantity qudt:numericValue ?power ;
                                       qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                        BIND (xsd:decimal(?power) / 750 AS ?expected)
                        FILTER (
                            ABS(xsd:decimal(?volume) - ?expected) >
                            ((0.000001 * ABS(?expected)) + 0.000001)
                        )
                    }
                """
            ]
