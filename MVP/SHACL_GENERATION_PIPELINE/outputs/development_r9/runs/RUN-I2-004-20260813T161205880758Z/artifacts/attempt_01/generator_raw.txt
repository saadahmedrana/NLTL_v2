<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:MassQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloTON_Metric ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:UpperIceWaterlineCaseShape
    a sh:NodeShape ;
    sh:class nltl:upperIceWaterlineCase ;
    sh:property [
        sh:path nltl:waterlineCaseDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MassQuantityShape ;
        sh:message "Each upper ice waterline case must have one displacement expressed in metric kilotonnes."
    ] .

gen:UpperIceWaterlineDisplacementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:upperIceWaterlineDisplacement ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MassQuantityShape ;
        sh:message "The ship must have one upper ice waterline displacement expressed in metric kilotonnes."
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterlineCase ;
        sh:minCount 1 ;
        sh:node gen:UpperIceWaterlineCaseShape ;
        sh:message "At least one upper ice waterline case with displacement data is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The upper ice waterline displacement must equal the greatest case displacement within relative tolerance 1e-6 and absolute fallback 1e-9 kt." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:upperIceWaterlineDisplacement ?reportedQuantity .
    ?reportedQuantity qudt:numericValue ?reportedValue ;
                      qudt:unit unit:KiloTON_Metric .
    {
        SELECT $this (MAX(?caseValue) AS ?expectedValue)
        WHERE {
            $this nltl:hasUpperIceWaterlineCase ?case .
            ?case nltl:waterlineCaseDisplacement ?caseQuantity .
            ?caseQuantity qudt:numericValue ?caseValue ;
                          qudt:unit unit:KiloTON_Metric .
        }
        GROUP BY $this
    }
    BIND(
        IF(
            ABS(xsd:decimal(?expectedValue)) * 0.000001 > 0.000000001,
            ABS(xsd:decimal(?expectedValue)) * 0.000001,
            0.000000001
        )
        AS ?tolerance
    )
    FILTER(
        ABS(
            xsd:decimal(?reportedValue) -
            xsd:decimal(?expectedValue)
        ) > ?tolerance
    )
}
"""
    ] .
<END_SHACL>
