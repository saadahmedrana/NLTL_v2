<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG
    ] .

gen:I2_009_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:minCount 1 ;
        sh:node gen:DegreeQuantityShape ;
        sh:message "Buttock angle is required in degrees to determine bow-form applicability."
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:minCount 1 ;
        sh:node gen:DegreeQuantityShape ;
        sh:message "Normal frame angle is required in degrees to determine bow-form applicability."
    ] ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Bow-form applicability must be recorded as a boolean."
    ] ;
    sh:sparql [
        sh:message "Bow-form applicability must be true when the buttock angle is greater than 0 and less than 80 degrees and the normal frame angle is greater than 10 degrees." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:buttockAngle ?buttockQuantity ;
                      nltl:normalFrameAngle ?frameQuantity ;
                      nltl:bowFormApplicability ?applicable .
                ?buttockQuantity qudt:numericValue ?buttockAngle .
                ?frameQuantity qudt:numericValue ?frameAngle .
                FILTER (
                    ?buttockAngle > 0.0 &&
                    ?buttockAngle < 80.0 &&
                    ?frameAngle > 10.0 &&
                    ?applicable != "true"^^xsd:boolean
                )
            }
        """
    ] .
<END_SHACL>
