gs:I2_014Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowLength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:MetreQuantityShape ;
        sh:message "Bow length must be provided in metres for the bow subregion calculation."
    ] ;
    sh:property [
        sh:path nltl:bowSubregionCount ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "4"^^xsd:integer ;
        sh:maxInclusive "4"^^xsd:integer ;
        sh:message "The bow region must be divided into four subregions."
    ] ;
    sh:property [
        sh:path nltl:hasBowSubregionCalculationCase ;
        sh:minCount 4 ;
        sh:maxCount 4 ;
        sh:class nltl:bowSubregionCalculationCase ;
        sh:node gs:BowSubregionCalculationCaseShape ;
        sh:message "Exactly four bow subregion calculation cases are required."
    ] ;
    sh:property [
        sh:path nltl:totalGlancingImpactForce ;
        sh:minCount 1 ;
        sh:node gs:GlancingImpactForceQuantityShape ;
        sh:message "A total glancing impact force result must be provided in kN."
    ] ;
    sh:sparql [
        sh:message "The four subregion mid-length positions must be LBOW/8, 3*LBOW/8, 5*LBOW/8, and 7*LBOW/8, within relative tolerance 1e-6 and absolute tolerance floor 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bowLength ?bowLengthQuantity ;
                      nltl:hasBowSubregionCalculationCase ?case .
                ?bowLengthQuantity qudt:numericValue ?bowLength ;
                                   qudt:unit unit:M .
                ?case nltl:bowSubregionMidLengthPosition ?midQuantity .
                ?midQuantity qudt:numericValue ?midPosition ;
                             qudt:unit unit:M .
                BIND(xsd:decimal(?bowLength) AS ?length)
                BIND(xsd:decimal(?midPosition) AS ?mid)
                BIND(?length / 8 AS ?expected1)
                BIND((3 * ?length) / 8 AS ?expected2)
                BIND((5 * ?length) / 8 AS ?expected3)
                BIND((7 * ?length) / 8 AS ?expected4)
                BIND(IF(ABS(?expected1) * 0.000001 > 0.000000001, ABS(?expected1) * 0.000001, 0.000000001) AS ?tolerance1)
                BIND(IF(ABS(?expected2) * 0.000001 > 0.000000001, ABS(?expected2) * 0.000001, 0.000000001) AS ?tolerance2)
                BIND(IF(ABS(?expected3) * 0.000001 > 0.000000001, ABS(?expected3) * 0.000001, 0.000000001) AS ?tolerance3)
                BIND(IF(ABS(?expected4) * 0.000001 > 0.000000001, ABS(?expected4) * 0.000001, 0.000000001) AS ?tolerance4)
                BIND(
                    IF(ABS(?mid - ?expected1) <= ?tolerance1, 1,
                        IF(ABS(?mid - ?expected2) <= ?tolerance2, 2,
                            IF(ABS(?mid - ?expected3) <= ?tolerance3, 3,
                                IF(ABS(?mid - ?expected4) <= ?tolerance4, 4, 0)
                            )
                        )
                    ) AS ?slot
                )
            }
            GROUP BY $this
            HAVING ((MIN(?slot) = 0) || (COUNT(DISTINCT ?slot) != 4))
        """
    ] .
