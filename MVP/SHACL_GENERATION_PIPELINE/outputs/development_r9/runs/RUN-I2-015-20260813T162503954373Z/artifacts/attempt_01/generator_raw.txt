<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipBowShapeCoefficientShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasBowSubregionCalculationCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasBowSubregionCalculationCase ;
                sh:minCount 1 ;
                sh:class nltl:bowSubregionCalculationCase ;
                sh:node gen:BowSubregionCalculationCaseShape
            ] ;
            sh:property [
                sh:path nltl:betaIPrime ;
                sh:minCount 1 ;
                sh:node gen:UnitlessDecimalQuantityShape
            ] ;
            sh:property [
                sh:path nltl:upperIceWaterlineLengthLUI ;
                sh:minCount 1 ;
                sh:node gen:MetreDecimalQuantityShape
            ]
        ]
    ) .

gen:BowSubregionCalculationCaseShape
    a sh:NodeShape ;
    sh:class nltl:bowSubregionCalculationCase ;
    sh:property [
        sh:path nltl:bowSubregionMidLengthPosition ;
        sh:minCount 1 ;
        sh:node gen:MetreDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:bowShapeCoefficient ;
        sh:minCount 1 ;
        sh:node gen:BowShapeCoefficientValueShape
    ] .

gen:TableLookupFailureFactorsShape
    a sh:NodeShape ;
    sh:targetSubjectsOf nltl:crushingFailureClassFactor ;
    sh:targetSubjectsOf nltl:flexuralFailureClassFactor ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessDecimalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:flexuralFailureClassFactor ;
        sh:minCount 1 ;
        sh:node gen:UnitlessDecimalQuantityShape
    ] .

gen:UnitlessDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .

gen:MetreDecimalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:M
    ] .

gen:BowShapeCoefficientValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:maxInclusive "0.60"^^xsd:decimal ;
        sh:message "The bow shape coefficient must not exceed 0.60."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:UNITLESS
    ] .
<END_SHACL>
