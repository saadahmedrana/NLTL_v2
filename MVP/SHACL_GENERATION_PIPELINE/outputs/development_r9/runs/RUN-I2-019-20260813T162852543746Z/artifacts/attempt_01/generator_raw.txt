gs:I2_019_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node gs:I2_019_StructuralMemberReadinessShape ;
        sh:message "Each linked structural member must identify at least one hull area that it spans."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:node gs:I2_019_SingleHullAreaMemberShape
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasStructuralMember ;
                sh:qualifiedValueShape gs:I2_019_BoundarySpanningMemberShape ;
                sh:qualifiedMinCount 1
            ] ;
            sh:property [
                sh:path nltl:hullAreaFactor ;
                sh:minCount 1 ;
                sh:node gs:I2_019_HullAreaFactorShape ;
                sh:message "A unitless hull area factor is required when a structural member spans more than one hull area."
            ]
        ]
    ) .

gs:I2_019_StructuralMemberReadinessShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:spansHullArea ;
        sh:minCount 1 ;
        sh:class nltl:hullRegionValue
    ] .

gs:I2_019_SingleHullAreaMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:spansHullArea ;
        sh:maxCount 1
    ] .

gs:I2_019_BoundarySpanningMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:spansHullArea ;
        sh:minCount 2 ;
        sh:class nltl:hullRegionValue
    ] .

gs:I2_019_HullAreaFactorShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1
    ] .
