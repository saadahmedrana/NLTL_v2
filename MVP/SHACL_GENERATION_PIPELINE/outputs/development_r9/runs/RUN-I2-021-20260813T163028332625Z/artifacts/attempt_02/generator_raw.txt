<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2-021Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadRequiredNetPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAllowance ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Thickness must equal ice-load required net plate thickness plus corrosion and abrasion allowance, converted from millimetres to metres, within relative tolerance 1e-6 or absolute fallback 1e-9 m." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:thickness ?thicknessQuantity ;
                      nltl:iceLoadRequiredNetPlateThickness ?netQuantity ;
                      nltl:corrosionAbrasionAllowance ?allowanceQuantity .
                ?thicknessQuantity qudt:numericValue ?reportedThickness .
                ?netQuantity qudt:numericValue ?netThickness .
                ?allowanceQuantity qudt:numericValue ?allowance .
                BIND((?netThickness + ?allowance) / 1000.0 AS ?expectedThickness)
                BIND(
                    IF(
                        ABS(?expectedThickness) * 0.000001 > 0.000000001,
                        ABS(?expectedThickness) * 0.000001,
                        0.000000001
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?reportedThickness - ?expectedThickness) > ?tolerance)
            }
        """
    ] .
<END_SHACL>
