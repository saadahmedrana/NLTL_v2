<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_034_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each linked longitudinal local frame must have one unambiguous boolean side-structure applicability state." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame .
    FILTER (
        NOT EXISTS {
            ?member nltl:sideStructureApplicability ?applicable .
            FILTER(datatype(?applicable) = xsd:boolean)
        }
        ||
        EXISTS {
            ?member nltl:sideStructureApplicability ?invalidApplicability .
            FILTER(datatype(?invalidApplicability) != xsd:boolean)
        }
        ||
        (
            EXISTS {
                ?member nltl:sideStructureApplicability true .
            }
            &&
            EXISTS {
                ?member nltl:sideStructureApplicability false .
            }
        )
    )
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable longitudinal local frame must have plastic strength expressed as a decimal quantity in kilonewtons." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability true .
    FILTER NOT EXISTS {
        ?member nltl:plasticStrength ?strength .
        ?strength qudt:numericValue ?strengthValue ;
            qudt:unit unit:KiloN .
        FILTER(datatype(?strengthValue) = xsd:decimal)
    }
}
"""
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable longitudinal local frame must identify its plastic collapse mechanism." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this ?member
WHERE {
    $this nltl:hasStructuralMember ?member .
    ?member a nltl:longitudinalLocalFrame ;
        nltl:sideStructureApplicability true .
    FILTER NOT EXISTS {
        ?member nltl:plasticCollapseMechanism ?mechanism .
        FILTER(datatype(?mechanism) = xsd:string)
    }
}
"""
    ] .
<END_SHACL>
