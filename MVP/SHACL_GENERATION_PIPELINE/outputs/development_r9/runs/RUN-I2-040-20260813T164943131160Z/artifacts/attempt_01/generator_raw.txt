<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_040_CommonWebBucklingEnvelopeShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each linked structural member requires positive decimal web height, net web thickness, and ship yield strength values in the specified units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember .
                FILTER NOT EXISTS {
                    ?member nltl:webHeight ?heightQuantity ;
                            nltl:netWebThickness ?thicknessQuantity .
                    $this nltl:yieldStrength ?strengthQuantity .
                    ?heightQuantity a qudt:QuantityValue ;
                                    qudt:numericValue ?height ;
                                    qudt:unit unit:MilliM .
                    ?thicknessQuantity a qudt:QuantityValue ;
                                       qudt:numericValue ?thickness ;
                                       qudt:unit unit:MilliM .
                    ?strengthQuantity a qudt:QuantityValue ;
                                      qudt:numericValue ?strength ;
                                      qudt:unit unit:MegaPA .
                    FILTER (
                        datatype(?height) = xsd:decimal &&
                        datatype(?thickness) = xsd:decimal &&
                        datatype(?strength) = xsd:decimal &&
                        ?height >= 0 &&
                        ?thickness > 0 &&
                        ?strength > 0
                    )
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The structural member exceeds the common h_w/t_wn <= 805/sqrt(sigma_y) web-buckling envelope." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasComponent ?member ;
                      nltl:yieldStrength ?strengthQuantity .
                ?member a nltl:structuralMember ;
                        nltl:webHeight ?heightQuantity ;
                        nltl:netWebThickness ?thicknessQuantity .
                ?heightQuantity a qudt:QuantityValue ;
                                qudt:numericValue ?height ;
                                qudt:unit unit:MilliM .
                ?thicknessQuantity a qudt:QuantityValue ;
                                   qudt:numericValue ?thickness ;
                                   qudt:unit unit:MilliM .
                ?strengthQuantity a qudt:QuantityValue ;
                                  qudt:numericValue ?strength ;
                                  qudt:unit unit:MegaPA .
                FILTER (
                    datatype(?height) = xsd:decimal &&
                    datatype(?thickness) = xsd:decimal &&
                    datatype(?strength) = xsd:decimal &&
                    ?height >= 0 &&
                    ?thickness > 0 &&
                    ?strength > 0 &&
                    (?height * ?height * ?strength) >
                    (648025 * ?thickness * ?thickness)
                )
            }
        """
    ] .
<END_SHACL>
