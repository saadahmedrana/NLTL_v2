<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2-041Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each linked structural member requires valid net web thickness, net attached shell-plate thickness, and yield strength inputs in the prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?member
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember .
                FILTER (
                    NOT EXISTS {
                        ?member a nltl:shipComponent ;
                                a nltl:benchmarkEntity .
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:netWebThickness ?webQuantity .
                        ?webQuantity a qudt:QuantityValue ;
                                     qudt:numericValue ?web ;
                                     qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?web) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:netAttachedShellPlateThickness ?plateQuantity .
                        ?plateQuantity a qudt:QuantityValue ;
                                       qudt:numericValue ?plate ;
                                       qudt:unit unit:MilliM .
                        FILTER (DATATYPE(?plate) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        ?member nltl:yieldStrength ?yieldQuantity .
                        ?yieldQuantity a qudt:QuantityValue ;
                                       qudt:numericValue ?yield ;
                                       qudt:unit unit:MegaPA .
                        FILTER (DATATYPE(?yield) = xsd:decimal)
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Net web thickness must be at least 0.35 times the net attached shell-plate thickness times the square root of yield strength divided by 235 MPa." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this ?member
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember ;
                        a nltl:benchmarkEntity ;
                        nltl:netWebThickness ?webQuantity ;
                        nltl:netAttachedShellPlateThickness ?plateQuantity ;
                        nltl:yieldStrength ?yieldQuantity .
                ?webQuantity a qudt:QuantityValue ;
                             qudt:numericValue ?web ;
                             qudt:unit unit:MilliM .
                ?plateQuantity a qudt:QuantityValue ;
                               qudt:numericValue ?plate ;
                               qudt:unit unit:MilliM .
                ?yieldQuantity a qudt:QuantityValue ;
                               qudt:numericValue ?yield ;
                               qudt:unit unit:MegaPA .
                FILTER (
                    DATATYPE(?web) = xsd:decimal &&
                    DATATYPE(?plate) = xsd:decimal &&
                    DATATYPE(?yield) = xsd:decimal
                )
                FILTER (
                    ?yield < "0"^^xsd:decimal
                    ||
                    (
                        ?plate >= "0"^^xsd:decimal
                        &&
                        (
                            ?web < "0"^^xsd:decimal
                            ||
                            (
                                "235"^^xsd:decimal * ?web * ?web
                                <
                                "0.1225"^^xsd:decimal * ?plate * ?plate * ?yield
                            )
                        )
                    )
                    ||
                    (
                        ?plate < "0"^^xsd:decimal
                        &&
                        ?web < "0"^^xsd:decimal
                        &&
                        (
                            "235"^^xsd:decimal * ?web * ?web
                            >
                            "0.1225"^^xsd:decimal * ?plate * ?plate * ?yield
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
