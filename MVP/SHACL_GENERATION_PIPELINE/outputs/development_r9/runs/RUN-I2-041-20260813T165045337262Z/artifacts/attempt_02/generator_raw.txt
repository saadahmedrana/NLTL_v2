<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_041_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_041_MillimetreQuantityShape ;
        sh:message "Net web thickness must be provided as one nonnegative decimal quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:netAttachedShellPlateThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_041_MillimetreQuantityShape ;
        sh:message "Net attached shell plate thickness must be provided as one nonnegative decimal quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:I2_041_MegapascalQuantityShape ;
        sh:message "Yield strength must be provided as one nonnegative decimal quantity in megapascals."
    ] ;
    sh:sparql [
        sh:message "Net web thickness must be at least 0.35 times net attached shell plate thickness times the square root of yield strength divided by 235." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:netWebThickness ?webQuantity ;
                      nltl:netAttachedShellPlateThickness ?plateQuantity ;
                      nltl:yieldStrength ?yieldQuantity .
                ?webQuantity qudt:numericValue ?webThickness ;
                             qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?plateQuantity qudt:numericValue ?plateThickness ;
                               qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?yieldQuantity qudt:numericValue ?yieldStrength ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                FILTER (
                    ?webThickness >= "0"^^xsd:decimal &&
                    ?plateThickness >= "0"^^xsd:decimal &&
                    ?yieldStrength >= "0"^^xsd:decimal &&
                    (?webThickness * ?webThickness) <
                    (
                        "0.1225"^^xsd:decimal *
                        (?plateThickness * ?plateThickness) *
                        (?yieldStrength / "235"^^xsd:decimal)
                    )
                )
            }
        """
    ] .

gen:I2_041_MillimetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MilliM ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:I2_041_MegapascalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
