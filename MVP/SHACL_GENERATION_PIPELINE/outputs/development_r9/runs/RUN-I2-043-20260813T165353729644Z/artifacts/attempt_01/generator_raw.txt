<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2_043_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each structural member subject to the flange-outstand calculation requires a decimal flange outstand in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember .
                FILTER NOT EXISTS {
                    ?member nltl:flangeOutstand ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:MilliM .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each structural member subject to the flange-outstand calculation requires a decimal net flange thickness in millimetres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember .
                FILTER NOT EXISTS {
                    ?member nltl:netFlangeThickness ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:MilliM .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each structural member subject to the flange-outstand calculation requires a decimal yield strength in megapascals." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember .
                FILTER NOT EXISTS {
                    ?member nltl:yieldStrength ?quantity .
                    ?quantity qudt:numericValue ?value ;
                              qudt:unit unit:MegaPA .
                    FILTER(datatype(?value) = xsd:decimal)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Flange outstand divided by net flange thickness must not exceed 155 divided by the square root of yield strength; thickness and yield strength must be positive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT $this
            WHERE {
                $this nltl:hasComponent ?member .
                ?member a nltl:structuralMember ;
                        nltl:flangeOutstand ?outstandQuantity ;
                        nltl:netFlangeThickness ?thicknessQuantity ;
                        nltl:yieldStrength ?yieldQuantity .
                ?outstandQuantity
                    qudt:numericValue ?outstand ;
                    qudt:unit unit:MilliM .
                ?thicknessQuantity
                    qudt:numericValue ?thickness ;
                    qudt:unit unit:MilliM .
                ?yieldQuantity
                    qudt:numericValue ?yieldStrength ;
                    qudt:unit unit:MegaPA .
                FILTER(
                    datatype(?outstand) = xsd:decimal &&
                    datatype(?thickness) = xsd:decimal &&
                    datatype(?yieldStrength) = xsd:decimal
                )
                FILTER(
                    ?thickness <= 0 ||
                    ?yieldStrength <= 0 ||
                    (
                        ?outstand > 0 &&
                        (?outstand * ?outstand * ?yieldStrength) >
                        (24025 * ?thickness * ?thickness)
                    )
                )
            }
        """
    ] .
<END_SHACL>
