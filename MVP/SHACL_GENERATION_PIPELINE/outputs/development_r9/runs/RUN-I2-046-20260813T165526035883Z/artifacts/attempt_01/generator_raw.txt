<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:I2-046Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasInternalIceStrengthenedStructure ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:internalIceStrengthenedStructure
    ] ;
    sh:sparql [
        sh:message "Each related internal ice-strengthened plating structure must have a corrosion/abrasion allowance of at least 1.0 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasInternalIceStrengthenedStructure ?structure .
                ?structure rdf:type nltl:plating .
                FILTER NOT EXISTS {
                    ?structure nltl:corrosionAbrasionAllowance ?allowance .
                    ?allowance qudt:numericValue ?value ;
                               qudt:unit unit:MilliM .
                    FILTER (
                        datatype(?value) = xsd:decimal &&
                        ?value >= 1.0
                    )
                }
            }
        """
    ] .
<END_SHACL>
