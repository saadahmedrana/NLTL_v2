<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:SteelGradeRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        )
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeRequirementCase ;
        sh:node gen:SteelGradeRequirementCaseShape
    ] ;
    sh:sparql [
        sh:message "A Polar Class ship requires a steel-grade requirement case containing the Table 8 lookup inputs and result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:polarClass ?polarClass .
                FILTER (
                    ?polarClass IN (
                        nltl:polarClassPc1,
                        nltl:polarClassPc2,
                        nltl:polarClassPc3,
                        nltl:polarClassPc4,
                        nltl:polarClassPc5,
                        nltl:polarClassPc6,
                        nltl:polarClassPc7
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:hasSteelGradeRequirementCase ?case .
                    ?case a nltl:steelGradeRequirementCase .
                }
            }
        """
    ] .

gen:SteelGradeRequirementCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:minCount 1 ;
        sh:nodeKind sh:BlankNodeOrIRI ;
        sh:node gen:AsBuiltPlateThicknessShape ;
        sh:message "Each steel-grade requirement case requires an as-built plate thickness in millimetres."
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:message "Each steel-grade requirement case requires material class one, two, or three."
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:message "Each steel-grade requirement case requires a normal-strength or high-tensile steel category."
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:message "Each steel-grade requirement case requires the hull structural steel grade selected from Table 8."
    ] .

gen:AsBuiltPlateThicknessShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "As-built plate thickness requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:hasValue unit:MilliM ;
        sh:message "As-built plate thickness must be expressed in millimetres."
    ] .
<END_SHACL>
