<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:UnitlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DegreeQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:MeganewtonQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DesignVerticalIceForceAtBowShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:crushingFailureClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:flexuralFailureClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:longitudinalStrengthClassFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:hullFormCoefficientKh ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceStrengthCoefficientKI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:UnitlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DegreeQuantityShape
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineDraughtDUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateOne ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeganewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBowCandidateTwo ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeganewtonQuantityShape
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:MeganewtonQuantityShape
    ] ;
    sh:sparql [
        sh:message "The bow ice-force candidates must satisfy their formulas, and the design force must equal their minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:crushingFailureClassFactor ?cfcNode ;
                      nltl:flexuralFailureClassFactor ?cffNode ;
                      nltl:longitudinalStrengthClassFactor ?cflNode ;
                      nltl:hullFormCoefficientKh ?khNode ;
                      nltl:iceStrengthCoefficientKI ?kiNode ;
                      nltl:stemAngle ?gammaNode ;
                      nltl:upperIceWaterlineDraughtDUI ?duiNode ;
                      nltl:designVerticalIceForceAtBowCandidateOne ?fib1Node ;
                      nltl:designVerticalIceForceAtBowCandidateTwo ?fib2Node ;
                      nltl:designVerticalIceForceAtBow ?fibNode .

                ?cfcNode qudt:numericValue ?cfc .
                ?cffNode qudt:numericValue ?cff .
                ?cflNode qudt:numericValue ?cfl .
                ?khNode qudt:numericValue ?kh .
                ?kiNode qudt:numericValue ?ki .
                ?gammaNode qudt:numericValue ?gammaDegrees .
                ?duiNode qudt:numericValue ?dui .
                ?fib1Node qudt:numericValue ?fib1 .
                ?fib2Node qudt:numericValue ?fib2 .
                ?fibNode qudt:numericValue ?fib .

                BIND(
                    math:sin(
                        xsd:double(?gammaDegrees) *
                        0.017453292519943295
                    )
                    AS ?stemSine
                )
                BIND(
                    0.534 *
                    math:pow(xsd:double(?ki), 0.15) *
                    math:pow(?stemSine, 0.2) *
                    math:pow(xsd:double(?dui) * xsd:double(?kh), 0.5) *
                    xsd:double(?cfl)
                    AS ?expectedFib1
                )
                BIND(
                    1.20 * xsd:double(?cff)
                    AS ?expectedFib2
                )
                BIND(
                    IF(
                        xsd:double(?fib1) <= xsd:double(?fib2),
                        xsd:double(?fib1),
                        xsd:double(?fib2)
                    )
                    AS ?expectedFib
                )
                BIND(
                    IF(
                        ABS(?expectedFib1) * 0.000001 > 0.000000001,
                        ABS(?expectedFib1) * 0.000001,
                        0.000000001
                    )
                    AS ?fib1Tolerance
                )
                BIND(
                    IF(
                        ABS(?expectedFib2) * 0.000001 > 0.000000001,
                        ABS(?expectedFib2) * 0.000001,
                        0.000000001
                    )
                    AS ?fib2Tolerance
                )
                BIND(
                    IF(
                        ABS(?expectedFib) * 0.000001 > 0.000000001,
                        ABS(?expectedFib) * 0.000001,
                        0.000000001
                    )
                    AS ?fibTolerance
                )

                FILTER(
                    xsd:double(?ki) < 0.0 ||
                    ?stemSine < 0.0 ||
                    (xsd:double(?dui) * xsd:double(?kh)) < 0.0 ||
                    !BOUND(?expectedFib1) ||
                    !BOUND(?expectedFib2) ||
                    !BOUND(?expectedFib) ||
                    ABS(xsd:double(?fib1) - ?expectedFib1) > ?fib1Tolerance ||
                    ABS(xsd:double(?fib2) - ?expectedFib2) > ?fib2Tolerance ||
                    ABS(xsd:double(?fib) - ?expectedFib) > ?fibTolerance
                )
            }
        """
    ] .
<END_SHACL>
