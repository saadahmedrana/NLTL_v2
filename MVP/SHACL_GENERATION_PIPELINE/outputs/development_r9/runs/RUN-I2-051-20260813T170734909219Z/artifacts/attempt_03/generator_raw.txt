@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:I2_051_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shearForceDirection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:positiveShearForce ;
        sh:message "The shear-force direction must be positive shear force."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceShearForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The design vertical ice shear force must equal Cf multiplied by FIB, converted from MN to kN, within relative tolerance 1e-6 or absolute tolerance 1e-9 kN near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:longitudinalDistributionFactor ?cfQuantity ;
                      nltl:designVerticalIceForceAtBow ?fibQuantity ;
                      nltl:designVerticalIceShearForce ?fiQuantity .
                ?cfQuantity qudt:numericValue ?cf ;
                            qudt:unit unit:UNITLESS .
                ?fibQuantity qudt:numericValue ?fib ;
                             qudt:unit unit:MegaN .
                ?fiQuantity qudt:numericValue ?fi ;
                            qudt:unit unit:KiloN .
                BIND((?cf * ?fib * "1000"^^xsd:decimal) AS ?expected)
                BIND(
                    IF(
                        ABS(?expected) * "0.000001"^^xsd:decimal >
                            "0.000000001"^^xsd:decimal,
                        ABS(?expected) * "0.000001"^^xsd:decimal,
                        "0.000000001"^^xsd:decimal
                    )
                    AS ?tolerance
                )
                FILTER(ABS(?fi - ?expected) > ?tolerance)
            }
        """
    ] .

gen:I2_051_AftRegionShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                      nltl:shearForceDirection nltl:positiveShearForce ;
                      nltl:upperIceWaterlineLengthLUI ?luiQuantity ;
                      nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity .
                ?luiQuantity qudt:numericValue ?lui ;
                             qudt:unit unit:M .
                ?positionQuantity qudt:numericValue ?position ;
                                  qudt:unit unit:M .
                FILTER(
                    ?position >= "0"^^xsd:decimal &&
                    ?position <= ("0.6"^^xsd:decimal * ?lui)
                )
            }
        """
    ] ;
    sh:property [
        sh:path nltl:longitudinalDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "0.0"^^xsd:decimal ;
                sh:maxInclusive "0.0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Cf must equal 0.0 from the aft end through 0.6 LUI from aft."
    ] .
