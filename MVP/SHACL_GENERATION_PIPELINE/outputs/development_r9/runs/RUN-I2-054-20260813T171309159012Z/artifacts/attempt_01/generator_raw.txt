<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designVerticalIceBendingMoment ;
        sh:minCount 1 ;
        sh:node gen:BendingMomentValueShape ;
        sh:message "A design vertical ice bending moment in MN m is required."
    ] ;
    sh:property [
        sh:path nltl:designVerticalIceForceAtBow ;
        sh:minCount 1 ;
        sh:node gen:IceForceValueShape ;
        sh:message "The design vertical ice force at bow in MN is required for the calculation."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineLengthLUI ;
        sh:minCount 1 ;
        sh:node gen:LengthValueShape ;
        sh:message "The upper ice waterline length in metres is required for the calculation."
    ] ;
    sh:property [
        sh:path nltl:stemAngle ;
        sh:minCount 1 ;
        sh:node gen:AngleValueShape ;
        sh:message "The stem angle in degrees is required for the calculation."
    ] .

gen:CalculationCaseShape
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:PositionValueShape ;
        sh:message "Each calculation case requires one longitudinal position from aft in metres."
    ] ;
    sh:property [
        sh:path nltl:bendingMomentDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gen:DistributionFactorValueShape ;
        sh:message "Each calculation case requires one dimensionless bending moment distribution factor between 0 and 1."
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:class nltl:interpolationPoint ;
        sh:message "The calculation case requires an interpolation point relationship."
    ] ;
    sh:sparql [
        sh:message "At the aft end, where longitudinal position from aft is zero, the bending moment distribution factor must be 0.0." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullGirderLongitudinalPositionFromAft ?positionQuantity ;
                      nltl:bendingMomentDistributionFactor ?factorQuantity .
                ?positionQuantity qudt:numericValue ?position .
                ?factorQuantity qudt:numericValue ?factor .
                FILTER (
                    ?position = "0"^^xsd:decimal &&
                    ?factor != "0"^^xsd:decimal
                )
            }
        """
    ] .

gen:BendingMomentValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN-M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:IceForceValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:LengthValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:AngleValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PositionValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DistributionFactorValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "0"^^xsd:decimal ;
        sh:maxInclusive "1"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
