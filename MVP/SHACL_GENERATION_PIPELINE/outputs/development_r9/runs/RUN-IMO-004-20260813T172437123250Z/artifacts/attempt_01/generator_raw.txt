<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO004Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:class nltl:iceOriginValue ;
        sh:message "Ice origin is required to determine whether the first-year ice definition applies."
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Winter growth count is required as an integer."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue unit:M
            ]
        ] ;
        sh:message "Ice thickness is required as a decimal quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:class nltl:iceTypeValue
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Sea ice of not more than one winter growth and thickness from 0.3 m to 2.0 m must be classified as first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
          nltl:winterGrowthCount ?growthCount ;
          nltl:iceThickness ?thicknessQuantity .
    ?thicknessQuantity qudt:numericValue ?thickness ;
                       qudt:unit unit:M .
    FILTER (
        ?growthCount <= "1"^^xsd:integer &&
        ?thickness >= "0.3"^^xsd:decimal &&
        ?thickness <= "2.0"^^xsd:decimal
    )
    FILTER NOT EXISTS {
        $this nltl:iceTypeClassification nltl:firstYearIce .
    }
}
"""
    ] .
<END_SHACL>
