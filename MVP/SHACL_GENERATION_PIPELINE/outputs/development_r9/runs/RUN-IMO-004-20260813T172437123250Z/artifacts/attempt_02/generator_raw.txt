<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO004Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one ice origin type is required to evaluate the classification."
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Exactly one integer winter growth count is required."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal ;
                sh:message "Ice thickness requires exactly one decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M ;
                sh:message "Ice thickness must be expressed in metres."
            ]
        ] ;
        sh:message "Exactly one ice thickness quantity is required."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Sea ice of not more than one winter growth and thickness from 0.3 m to 2.0 m must have exactly one classification: first-year ice." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:iceOriginType nltl:seaIceOrigin ;
          nltl:winterGrowthCount ?growth ;
          nltl:iceThickness ?quantity .
    ?quantity qudt:numericValue ?thickness ;
              qudt:unit unit:M .
    FILTER (
        ?growth <= "1"^^xsd:integer &&
        ?thickness >= "0.3"^^xsd:decimal &&
        ?thickness <= "2.0"^^xsd:decimal
    )
    OPTIONAL {
        $this nltl:iceTypeClassification ?classification .
    }
}
GROUP BY $this
HAVING (
    COUNT(?classification) != 1 ||
    SUM(IF(?classification = nltl:firstYearIce, 1, 0)) != 1
)
"""
    ] .
<END_SHACL>
