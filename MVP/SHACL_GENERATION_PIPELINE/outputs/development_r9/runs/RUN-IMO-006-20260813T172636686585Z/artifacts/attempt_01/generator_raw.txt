<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gs:MediumFirstYearIceThicknessShape
    a sh:NodeShape ;
    sh:targetClass <https://w3id.org/nltl-benchmark/vocab#ship> ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A ship classified for medium first-year ice must record ice thickness from 0.7 m through 1.2 m as a decimal QUDT quantity." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceTypeClassification nltl:mediumFirstYearIce .
    OPTIONAL {
        $this nltl:iceThickness ?quantity .
        OPTIONAL {
            ?quantity a qudt:QuantityValue .
            BIND(true AS ?isQuantityValue)
        }
        OPTIONAL {
            ?quantity qudt:numericValue ?value .
        }
        OPTIONAL {
            ?quantity qudt:unit unit:M .
            BIND(true AS ?hasMetreUnit)
        }
    }
    FILTER (
        !BOUND(?quantity) ||
        !BOUND(?isQuantityValue) ||
        !BOUND(?value) ||
        DATATYPE(?value) != xsd:decimal ||
        ?value < "0.7"^^xsd:decimal ||
        ?value > "1.2"^^xsd:decimal ||
        !BOUND(?hasMetreUnit)
    )
}
"""
    ] .
<END_SHACL>
