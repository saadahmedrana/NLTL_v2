<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:IMO008ThinFirstYearIceShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceTypeClassification ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:message "Ice type classification is required to determine whether the thin first-year ice thickness range applies."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Thin first-year ice thickness must be recorded in metres as an xsd:decimal from 0.3 through 0.7 inclusive." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceTypeClassification nltl:thinFirstYearIce .
    FILTER NOT EXISTS {
        $this nltl:iceThickness ?quantity .
        ?quantity qudt:numericValue ?value ;
                  qudt:unit unit:M .
        FILTER(
            DATATYPE(?value) = xsd:decimal &&
            ?value >= "0.3"^^xsd:decimal &&
            ?value <= "0.7"^^xsd:decimal
        )
    }
}
"""
    ] .
<END_SHACL>
