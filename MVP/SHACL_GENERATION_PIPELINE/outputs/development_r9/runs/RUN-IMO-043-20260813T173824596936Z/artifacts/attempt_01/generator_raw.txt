<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:ShipHazardProtectionShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasMachineryInstallation ;
        sh:minCount 1 ;
        sh:class nltl:machineryInstallationItem ;
        sh:node gen:MachineryInstallationHazardProtectionShape ;
        sh:message "Each ship must identify its machinery installations and their hazard-protection information."
    ] .

gen:MachineryInstallationHazardProtectionShape
    a sh:NodeShape ;
    sh:class nltl:machineryInstallationItem ;
    sh:property [
        sh:path nltl:hasAssociatedEquipment ;
        sh:minCount 1 ;
        sh:class nltl:associatedEquipmentItem ;
        sh:message "Each machinery installation must identify its associated equipment."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:iceAccretionHazard ;
        sh:message "Protection coverage must include the ice accretion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowAccumulationHazard ;
        sh:message "Protection coverage must include the snow accumulation hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIceIngestionHazard ;
        sh:message "Protection coverage must include the seawater ice ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:liquidFreezingOrViscosityIncreaseHazard ;
        sh:message "Protection coverage must include the liquid freezing or viscosity increase hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:seawaterIntakeTemperatureHazard ;
        sh:message "Protection coverage must include the seawater intake temperature hazard."
    ] ;
    sh:property [
        sh:path nltl:hasApplicableHazard ;
        sh:hasValue nltl:snowIngestionHazard ;
        sh:message "Protection coverage must include the snow ingestion hazard."
    ] ;
    sh:property [
        sh:path nltl:hasHazardProtectionEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "Each machinery installation must provide hazard-protection evidence."
    ] .
<END_SHACL>
