<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:CelsiusQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG_C ;
        sh:minCount 1
    ] .

gen:IMO047Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:class nltl:essentialEngine ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:combustionAirTemperatureControlPresent ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:combustionAirTemperature ;
                sh:node gen:CelsiusQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:manufacturerMinAirTemperature ;
                sh:node gen:CelsiusQuantityShape ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:manufacturerMaxAirTemperature ;
                sh:node gen:CelsiusQuantityShape ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "Ships with essential engines must provide combustion-air temperature control and manufacturer temperature criteria in degrees Celsius." ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion air temperature is outside the recorded manufacturer temperature criteria." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:hasEssentialEngine ?engine ;
          nltl:combustionAirTemperature ?temperatureQuantity .
    ?temperatureQuantity qudt:numericValue ?temperature .
    FILTER NOT EXISTS {
        $this nltl:manufacturerMinAirTemperature ?minimumQuantity ;
              nltl:manufacturerMaxAirTemperature ?maximumQuantity .
        ?minimumQuantity qudt:numericValue ?minimum .
        ?maximumQuantity qudt:numericValue ?maximum .
        FILTER (
            xsd:decimal(?temperature) >= xsd:decimal(?minimum) &&
            xsd:decimal(?temperature) <= xsd:decimal(?maximum)
        )
    }
}
"""
    ] .
<END_SHACL>
