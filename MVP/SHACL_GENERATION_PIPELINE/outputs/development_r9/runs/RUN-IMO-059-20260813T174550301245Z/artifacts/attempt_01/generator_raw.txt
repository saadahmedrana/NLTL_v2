<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

g:FirefighterOutfitStorageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasFirefighterOutfit ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:firefighterOutfitItem ;
        sh:node g:FirefighterOutfitItemShape ;
        sh:message "Each linked firefighter outfit must be stored in a warm location."
    ] .

g:FirefighterOutfitItemShape
    a sh:NodeShape ;
    sh:class nltl:equipmentItem ;
    sh:property [
        sh:path nltl:hasStorageLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:physicalStorageLocation ;
        sh:minCount 1 ;
        sh:node g:WarmStorageLocationShape ;
        sh:message "A firefighter outfit must have a physical storage location classified as warm."
    ] .

g:WarmStorageLocationShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:storageTemperatureClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:storageTemperatureClassValue ;
        sh:hasValue nltl:warmStorageLocation ;
        sh:message "The storage location must have the warm storage temperature class."
    ] .
<END_SHACL>
