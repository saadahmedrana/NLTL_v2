<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:PassengerShipPersonalSurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:passengerShip ;
    sh:property [
        sh:path nltl:sizeCompatibilityStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:in ( true ) ;
        sh:message "The assigned personal survival equipment must be recorded as properly sized."
    ] ;
    sh:property [
        sh:path nltl:hasPersonOnBoardMember ;
        sh:class nltl:personOnBoardMember ;
        sh:node shape:PersonOnBoardEquipmentAssignmentShape ;
        sh:message "Each person on board must have a conforming personal survival equipment assignment."
    ] .

shape:PersonOnBoardEquipmentAssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasPersonalSurvivalEquipmentAssignment ;
        sh:class nltl:personalSurvivalEquipmentAssignment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node shape:PersonalSurvivalEquipmentAssignmentShape ;
        sh:message "Each person on board must have one personal survival equipment assignment."
    ] .

shape:PersonalSurvivalEquipmentAssignmentShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:assignedPersonalSurvivalEquipment ;
        sh:class nltl:personalSurvivalEquipment ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [ sh:class nltl:immersionSuitEquipment ]
            [ sh:class nltl:thermalProtectiveAidEquipment ]
        ) ;
        sh:message "The assignment must provide one immersion suit or one thermal protective aid."
    ] .
<END_SHACL>
