<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix shape: <urn:nltl:generated-shape:> .

shape:LifeboatTypeRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Every lifeboat must have a type classification that is partially enclosed or totally enclosed." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    ?lifeboat a nltl:lifeboat .
    FILTER (
        NOT EXISTS {
            ?lifeboat nltl:lifeboatTypeClassification ?lifeboatType .
        }
        ||
        EXISTS {
            ?lifeboat nltl:lifeboatTypeClassification ?lifeboatType .
            FILTER (
                ?lifeboatType NOT IN (
                    nltl:partiallyEnclosedLifeboat,
                    nltl:totallyEnclosedLifeboat
                )
            )
        }
    )
}
"""
    ] .
<END_SHACL>
