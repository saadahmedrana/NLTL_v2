<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ShipIceStrengtheningShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minCount 1 ;
        sh:datatype xsd:integer ;
        sh:message "Propeller count is required to determine whether the strengthening rule applies."
    ] ;
    sh:or (
        gen:ApplicableTwinOrTripleScrewBranch
        gen:NonApplicableLowerPropellerCountBranch
        gen:NonApplicableHigherPropellerCountBranch
    ) .

gen:ApplicableTwinOrTripleScrewBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:in ( 2 3 )
    ] ;
    sh:property [
        sh:path nltl:shellIceStrengtheningExtent ;
        sh:hasValue "tank top"^^xsd:string ;
        sh:message "Shell ice strengthening shall extend to the tank top."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningExtent ;
        sh:hasValue "tank top"^^xsd:string ;
        sh:message "Framing ice strengthening shall extend to the tank top."
    ] ;
    sh:property [
        sh:path nltl:hasSidePropellerStrengtheningEnvelope ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:sidePropellerStrengtheningEnvelope ;
        sh:node gen:SidePropellerStrengtheningEnvelopeShape ;
        sh:message "Twin and triple screw ships require a side-propeller strengthening envelope."
    ] .

gen:NonApplicableLowerPropellerCountBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:maxInclusive 1
    ] .

gen:NonApplicableHigherPropellerCountBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerCount ;
        sh:minInclusive 4
    ] .

gen:SidePropellerStrengtheningEnvelopeShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:strengtheningForwardExtent ;
        sh:minCount 1 ;
        sh:node gen:OnePointFiveMetreExtentShape ;
        sh:message "The strengthening envelope shall extend 1.5 metres forward of the side propeller."
    ] ;
    sh:property [
        sh:path nltl:strengtheningAftExtent ;
        sh:minCount 1 ;
        sh:node gen:OnePointFiveMetreExtentShape ;
        sh:message "The strengthening envelope shall extend 1.5 metres aft of the side propeller."
    ] .

gen:OnePointFiveMetreExtentShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:minInclusive "1.5"^^xsd:decimal ;
        sh:maxInclusive "1.5"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M
    ] .
<END_SHACL>
