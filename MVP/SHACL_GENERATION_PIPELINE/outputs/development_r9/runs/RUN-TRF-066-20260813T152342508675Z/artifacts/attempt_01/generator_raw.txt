<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF066ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:minCount 1 ;
        sh:message "Ice class is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIaSuper
                )
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:in ( true false ) ;
                sh:message "Ice-knife practicability must be recorded for ice class IA or IA Super."
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:hasRudderProtectionArrangement ;
                        sh:qualifiedValueShape [
                            sh:node gen:TRF066EquivalentProtectionShape
                        ] ;
                        sh:qualifiedMinCount 1 ;
                        sh:message "An equivalent rudder protection means is required."
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:iceKnifePracticable ;
                        sh:hasValue true
                    ] ;
                    sh:property [
                        sh:path nltl:lowerIceWaterline ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:message "The lower ice waterline is required for the ice-knife alternative."
                    ] ;
                    sh:property [
                        sh:path nltl:hasRudderProtectionArrangement ;
                        sh:qualifiedValueShape [
                            sh:node gen:TRF066IceKnifeShape
                        ] ;
                        sh:qualifiedMinCount 1 ;
                        sh:message "A practicable ice knife extending below the lower ice waterline is required."
                    ]
                ]
            )
        ]
    ) .

gen:TRF066EquivalentProtectionShape
    a sh:NodeShape ;
    sh:class nltl:equivalentRudderProtectionMeans .

gen:TRF066IceKnifeShape
    a sh:NodeShape ;
    sh:class nltl:iceKnife ;
    sh:property [
        sh:path nltl:extendsBelowLowerIceWaterline ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "The ice knife must extend below the lower ice waterline."
    ] .
<END_SHACL>
