<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF069Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "Ice class is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Propeller type is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Pitch type is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An applicable propulsion arrangement requires maximum ice torque in kN m as a decimal quantity value." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT DISTINCT $this
WHERE {
    $this nltl:iceClass ?iceClass ;
          nltl:propellerType ?propellerType ;
          nltl:pitchType ?pitchType .
    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa,
            nltl:iceClassIb,
            nltl:iceClassIc
        )
    )
    FILTER (?propellerType IN ("open"^^xsd:string, "ducted"^^xsd:string))
    FILTER (?pitchType IN ("controllable"^^xsd:string, "fixed"^^xsd:string))
    FILTER NOT EXISTS {
        $this nltl:maximumIceTorque ?torque .
        ?torque a qudt:QuantityValue ;
                qudt:numericValue ?value ;
                qudt:unit unit:KiloN-M .
        FILTER (DATATYPE(?value) = xsd:decimal)
    }
}
"""
    ] .
<END_SHACL>
