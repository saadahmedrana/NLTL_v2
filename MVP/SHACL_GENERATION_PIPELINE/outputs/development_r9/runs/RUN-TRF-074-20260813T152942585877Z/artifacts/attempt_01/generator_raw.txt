<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:MetreQuantityValueShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal ;
        sh:message "The quantity requires a decimal numeric value."
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:message "The quantity unit must include metre."
    ] .

g:PropellerSubmersionRequirementShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:minCount 1 ;
                sh:node g:MetreQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:minCount 1 ;
                sh:node g:MetreQuantityValueShape
            ] ;
            sh:sparql [
                sh:message "The reported ballast-condition propeller-top depth is less than hi." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
          nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .
    ?depthQuantity qudt:numericValue ?depth .
    ?requiredQuantity qudt:numericValue ?requiredDepth .
    FILTER (?depth < ?requiredDepth)
}
"""
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:ballastConditionPropellerTopDepth ;
                sh:minCount 1 ;
                sh:node g:MetreQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:requiredPropellerSubmersionDepthHi ;
                sh:minCount 1 ;
                sh:node g:MetreQuantityValueShape
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemDesignedForIceClass ;
                sh:hasValue nltl:iceClassIa ;
                sh:message "When the propeller-top depth is less than hi, the propulsion system must be designed according to ice class IA."
            ] ;
            sh:sparql [
                sh:message "The ice-class IA propulsion alternative applies only when the propeller-top depth is less than hi." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this nltl:ballastConditionPropellerTopDepth ?depthQuantity ;
          nltl:requiredPropellerSubmersionDepthHi ?requiredQuantity .
    ?depthQuantity qudt:numericValue ?depth .
    ?requiredQuantity qudt:numericValue ?requiredDepth .
    FILTER (?depth >= ?requiredDepth)
}
"""
            ]
        ]
    ) ;
    sh:message "The propeller-submersion requirement is not satisfied or lacks the information needed to establish the applicable alternative." .
<END_SHACL>
