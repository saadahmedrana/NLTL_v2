            sh:property [

            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:hasValue unit:REV-PER-MIN ;

            ]
        ] ;
        sh:message "The ship must record its propeller rotational speed at MCR in bollard condition."
