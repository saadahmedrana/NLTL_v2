<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:ApplicableNonHemisphericalImpactTarget
    a sh:SPARQLTarget ;
    sh:select """
        PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
        PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
        SELECT DISTINCT ?this
        WHERE {
            ?this a nltl:ship ;
                  nltl:nonHemisphericalImpactArea ?impactArea .
        }
    """ .

gen:NonHemisphericalImpactRadiusShape
    a sh:NodeShape ;
    sh:target gen:ApplicableNonHemisphericalImpactTarget ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactArea ;
        sh:datatype xsd:string
    ] ;
    sh:property [
        sh:path nltl:nonHemisphericalImpactContactArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M2
            ]
        ] ;
        sh:message "A non-hemispherical impact contact area in square metres is required."
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class qudt:QuantityValue ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue unit:M
            ]
        ] ;
        sh:message "An equivalent impact sphere radius in metres is required."
    ] .
<END_SHACL>
