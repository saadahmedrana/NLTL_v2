<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:KilowattQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:KiloW ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:NumericQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:TRF132EngineOutputShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one ice class is needed to determine the applicable minimum engine output."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gs:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaBasePowerP0 ;
        sh:node gs:KilowattQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaDisplacement ;
        sh:node gs:NumericQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF1 ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF2 ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF3 ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:engineOutputFormulaFactorF4 ;
        sh:node gs:NumericQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "740"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minInclusive "740"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Maximum continuous rating power must be no less than f1*f2*f3*(f4*Delta + P0)." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:maximumContinuousRatingPower ?mcrNode ;
                      nltl:engineOutputFormulaBasePowerP0 ?p0Node ;
                      nltl:engineOutputFormulaDisplacement ?deltaNode ;
                      nltl:engineOutputFormulaFactorF1 ?f1Node ;
                      nltl:engineOutputFormulaFactorF2 ?f2Node ;
                      nltl:engineOutputFormulaFactorF3 ?f3Node ;
                      nltl:engineOutputFormulaFactorF4 ?f4Node .
                ?mcrNode qudt:numericValue ?mcr .
                ?p0Node qudt:numericValue ?p0 .
                ?deltaNode qudt:numericValue ?delta .
                ?f1Node qudt:numericValue ?f1 .
                ?f2Node qudt:numericValue ?f2 .
                ?f3Node qudt:numericValue ?f3 .
                ?f4Node qudt:numericValue ?f4 .
                BIND((?f1 * ?f2 * ?f3 * ((?f4 * ?delta) + ?p0)) AS ?requiredPower)
                FILTER (?mcr < ?requiredPower)
            }
        """
    ] .
<END_SHACL>
