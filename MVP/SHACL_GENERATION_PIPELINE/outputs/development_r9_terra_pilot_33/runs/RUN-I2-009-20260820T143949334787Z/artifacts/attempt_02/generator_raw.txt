<urn:nltl:generated-shape:OtherIceStrengthenedAreaParametersShape>
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:hasHullAreaAssignment ?assignment .
                ?assignment nltl:assignedHullArea
                    nltl:otherIceStrengthenedArea .
            }
        """
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:MPaQuantityShape> ;
        sh:message "Average ice pressure is required for an other ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:MetreQuantityShape> ;
        sh:message "Non-bow load-patch length is required for an other ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:MetreQuantityShape> ;
        sh:message "Non-bow load-patch height is required for an other ice-strengthened area."
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:node <urn:nltl:generated-shape:UnitlessQuantityShape> ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:minInclusive 3.6 ;
                sh:maxInclusive 3.6
            ]
        ] ;
        sh:message "The load-patch aspect ratio for other ice-strengthened areas shall be 3.6."
    ] .

<urn:nltl:generated-shape:BowFormApplicabilityShape>
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT DISTINCT ?this
            WHERE {
                ?this a nltl:ship ;
                    nltl:bowFormApplicability ?applicability .
            }
        """
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bowFormApplicability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:buttockAngle ;
                sh:minCount 1 ;
                sh:node [
                    a sh:NodeShape ;
                    sh:node <urn:nltl:generated-shape:DegreeQuantityShape> ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minExclusive 0 ;
                        sh:maxExclusive 80
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:normalFrameAngle ;
                sh:minCount 1 ;
                sh:node [
                    a sh:NodeShape ;
                    sh:node <urn:nltl:generated-shape:DegreeQuantityShape> ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:minExclusive 10
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:bowFormApplicability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:buttockAngle ;
                        sh:minCount 1 ;
                        sh:node [
                            a sh:NodeShape ;
                            sh:node <urn:nltl:generated-shape:DegreeQuantityShape> ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:maxInclusive 0
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:buttockAngle ;
                        sh:minCount 1 ;

                            a sh:NodeShape ;
                            sh:node <urn:nltl:generated-shape:DegreeQuantityShape> ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:minInclusive 80
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:normalFrameAngle ;
                        sh:minCount 1 ;
                        sh:node [
                            a sh:NodeShape ;
                            sh:node <urn:nltl:generated-shape:DegreeQuantityShape> ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:maxInclusive 10
                            ]
                        ]
                    ]
                ]
            )
        ]
    ) ;
    sh:message "Bow-form applicability must be supported by unit-valid buttock-angle and normal-frame-angle evidence."
