<BEGIN_SHACL>
@prefix gs: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gs:I2_009Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:bowFormApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Bow-form applicability must be recorded as exactly one boolean value."
    ] ;
    sh:property [
        sh:path nltl:buttockAngle ;
        sh:node gs:DegreeAngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:normalFrameAngle ;
        sh:node gs:DegreeAngleQuantityShape
    ] ;
    sh:property [
        sh:path nltl:averageIcePressure ;
        sh:node gs:MegaPascalQuantityShape
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchHeight ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:nonBowLoadPatchLength ;
        sh:node gs:MetreQuantityShape
    ] ;
    sh:property [
        sh:path nltl:loadPatchAspectRatio ;
        sh:node gs:AspectRatioQuantityShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bowFormApplicability ;
                sh:hasValue true
            ] ;
            sh:sparql [
                sh:message "Bow-form applicability may be true only when buttock angle is greater than 0 deg and less than 80 deg, and normal frame angle is greater than 10 deg." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                      FILTER NOT EXISTS {
                        $this nltl:buttockAngle ?gamma ;
                              nltl:normalFrameAngle ?beta .
                        ?gamma qudt:unit unit:DEG ;
                               qudt:numericValue ?gammaValue .
                        ?beta qudt:unit unit:DEG ;
                              qudt:numericValue ?betaValue .
                        FILTER(
                          datatype(?gammaValue) = xsd:decimal &&
                          datatype(?betaValue) = xsd:decimal &&
                          ?gammaValue > 0 &&
                          ?gammaValue < 80 &&
                          ?betaValue > 10
                        )
                      }
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:bowFormApplicability ;
                sh:hasValue false
            ] ;
            sh:sparql [
                sh:message "Bow-form applicability may be false only when unit-valid angle evidence shows that at least one qualifying angle condition is not met." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX unit: <http://qudt.org/vocab/unit/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                      FILTER NOT EXISTS {
                        {
                          $this nltl:buttockAngle ?gamma .
                          ?gamma qudt:unit unit:DEG ;
                                 qudt:numericValue ?gammaValue .
                          FILTER(
                            datatype(?gammaValue) = xsd:decimal &&
                            (?gammaValue <= 0 || ?gammaValue >= 80)
                          )
                        }
                        UNION
                        {
                          $this nltl:normalFrameAngle ?beta .
                          ?beta qudt:unit unit:DEG ;
                                qudt:numericValue ?betaValue .
                          FILTER(
                            datatype(?betaValue) = xsd:decimal &&
                            ?betaValue <= 10
                          )
                        }
                      }
                    }
                """
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For an assigned other ice-strengthened area, average ice pressure, non-bow load-patch dimensions, and a 3.6 load-patch aspect ratio are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
              $this nltl:hasHullAreaAssignment ?assignment .
              ?assignment nltl:assignedHullArea nltl:otherIceStrengthenedArea .
              FILTER NOT EXISTS {
                $this nltl:averageIcePressure ?pressure ;
                      nltl:nonBowLoadPatchHeight ?height ;
                      nltl:nonBowLoadPatchLength ?length ;
                      nltl:loadPatchAspectRatio ?ratio .
              }
            }
        """
    ] .

gs:DegreeAngleQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:DEG ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:MegaPascalQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:MetreQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:AspectRatioQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:minInclusive 3.6 ;
        sh:maxInclusive 3.6
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
