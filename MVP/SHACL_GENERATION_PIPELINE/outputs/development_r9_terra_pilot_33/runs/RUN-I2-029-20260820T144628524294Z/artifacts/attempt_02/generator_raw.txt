gs:CalculationCaseShape a sh:NodeShape ;
    sh:targetObjectsOf nltl:hasCalculationCase ;
    sh:property [
        sh:path nltl:attachedShellPlatingIncludedInSectionModulus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:attachedShellPlatingExcludedFromShearArea ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:flangeMaterialIncludedInShearArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "Flange material may be included in shear area only for a calculation case linked to a structural member with a fitted flange." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:flangeMaterialIncludedInShearArea true .
                FILTER NOT EXISTS {
                    ?ship nltl:hasCalculationCase $this ;
                          nltl:hasStructuralMember ?member .
                    ?member nltl:flangeFitted true .
                }
            }
        """
    ] .
