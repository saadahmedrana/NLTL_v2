<urn:nltl:generated-shape:LocalFrameSectionCalculationCaseShape>
    a sh:NodeShape ;
    sh:node <urn:nltl:generated-shape:CalculationCaseReadinessShape> ;
    sh:class nltl:localFrameSectionCalculationCase ;
    sh:property [
        sh:path nltl:sectionCalculationCaseStructuralMember ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:LocalFrameMemberInputShape>
    ] ;
    sh:property [
        sh:path nltl:sectionCalculationCasePlating ;
        sh:minCount 1 ;
        sh:node <urn:nltl:generated-shape:LocalFramePlatingInputShape>
    ] ;
    sh:sparql [
        sh:message "Where the local-frame area exceeds the attached-plate-flange area, plastic neutral axis height and net effective plastic section modulus results are required." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:sectionCalculationCaseStructuralMember ?member .
                ?member nltl:netLocalFrameFlangeArea ?areaQuantity ;
                        nltl:netAttachedShellPlateThickness ?plateThicknessQuantity ;
                        nltl:frameSpacing ?spacingQuantity .
                ?areaQuantity qudt:numericValue ?area .
                ?plateThicknessQuantity qudt:numericValue ?plateThickness .
                ?spacingQuantity qudt:numericValue ?spacing .
                FILTER (100 * ?area > 1000 * ?plateThickness * ?spacing)
                FILTER NOT EXISTS {
                    $this nltl:plasticNeutralAxisHeight ?neutralAxisQuantity .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Where the local-frame area exceeds the attached-plate-flange area, plastic neutral axis height and net effective plastic section modulus results are required." ;
