<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:I2-060ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember
    ] ;
    sh:sparql [
        sh:message "A structural member identified as being in way of a cutout must identify a cutout." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout ?inWay .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER NOT EXISTS {
                    ?member nltl:hasCutout ?cutout .
                    ?cutout a nltl:cutout .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "A member in way of a cutout must state whether stiffening is necessary." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout ?inWay .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER NOT EXISTS {
                    ?member nltl:stiffeningNecessary ?necessary .
                    FILTER(datatype(?necessary) = xsd:boolean)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Stiffening necessity for a member in way of a cutout must be a boolean value." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout ?inWay ;
                        nltl:stiffeningNecessary ?necessary .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER(datatype(?necessary) != xsd:boolean)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Loads carried by a member in way of a cutout must not cause member instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout ?inWay .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER NOT EXISTS {
                    $this nltl:memberInstability ?instability .
                    FILTER(datatype(?instability) = xsd:boolean && ?instability = false)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Member instability must be recorded as false when a member is in way of a cutout." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:memberInstability ?instability .
                ?member nltl:memberInWayOfCutout ?inWay .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER(datatype(?instability) != xsd:boolean || ?instability != false)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Structural stiffening must be recorded when it is necessary for a member in way of a cutout." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout ?inWay ;
                        nltl:stiffeningNecessary ?necessary .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER(datatype(?necessary) = xsd:boolean && ?necessary = true)
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER(datatype(?stiffening) = xsd:string)
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Structural stiffening must be recorded as a string when it is necessary." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:structuralStiffening ?stiffening .
                ?member nltl:memberInWayOfCutout ?inWay ;
                        nltl:stiffeningNecessary ?necessary .
                FILTER(datatype(?inWay) = xsd:boolean && ?inWay = true)
                FILTER(datatype(?necessary) = xsd:boolean && ?necessary = true)
                FILTER(datatype(?stiffening) != xsd:string)
            }
        """
    ] .
<END_SHACL>
