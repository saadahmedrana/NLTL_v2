<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:generated-shape:> .

gen:IMO018ConditionalMethodologyApprovalShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceCapabilityMethodologyApplicable ;
        sh:datatype xsd:boolean ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:certificateMethodologyReferencePresent ;
        sh:datatype xsd:boolean
    ] ;
    sh:property [
        sh:path nltl:administrationAcceptanceStatus ;
        sh:datatype xsd:boolean
    ] ;
    sh:sparql [
        sh:message "When the ice capability methodology is applicable, the certificate methodology reference and Administration acceptance status must both be true." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>

            SELECT $this
            WHERE {
                $this nltl:iceCapabilityMethodologyApplicable true .
                FILTER NOT EXISTS {
                    $this nltl:certificateMethodologyReferencePresent true .
                    $this nltl:administrationAcceptanceStatus true .
                }
            }
        """
    ] .
<END_SHACL>
