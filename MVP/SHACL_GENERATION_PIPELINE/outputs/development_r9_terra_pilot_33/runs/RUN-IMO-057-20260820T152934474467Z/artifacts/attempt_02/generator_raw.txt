<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:FirePumpCompartmentTemperatureShape
    a sh:NodeShape ;
    sh:target [
        a sh:SPARQLTarget ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT ?this
            WHERE {
                ?ship a nltl:ship ;
                      nltl:hasComponent ?this .
                ?this a ?pumpType .
                FILTER(?pumpType IN (
                    nltl:firePump,
                    nltl:emergencyFirePump,
                    nltl:waterMistPump,
                    nltl:waterSprayPump
                ))
            }
        """
    ] ;
    sh:property [
        sh:path nltl:hasContainingCompartment ;
        sh:minCount 1 ;
        sh:message "Each applicable fire pump must be located in a compartment." ;
        sh:node [
            sh:class nltl:compartment ;
            sh:property [
                sh:path nltl:maintainedTemperature ;
                sh:minCount 1 ;
                sh:message "The containing compartment must have maintained temperature above 0 deg C." ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minExclusive "0"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:DEG_C ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] .
<END_SHACL>
