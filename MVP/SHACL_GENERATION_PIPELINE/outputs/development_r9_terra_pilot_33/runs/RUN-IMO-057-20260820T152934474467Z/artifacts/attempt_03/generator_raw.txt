<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:IMO057FirePumpCompartmentTemperatureShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each fire pump must be located in a compartment with at least one maintained temperature, and every asserted maintained temperature must be a positive decimal value in degrees Celsius." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?pump .
                ?pump rdf:type ?pumpType .
                FILTER (
                    ?pumpType IN (
                        nltl:firePump,
                        nltl:emergencyFirePump,
                        nltl:waterMistPump,
                        nltl:waterSprayPump
                    )
                )

                {
                    FILTER NOT EXISTS {
                        ?pump nltl:hasContainingCompartment ?compartment .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    FILTER NOT EXISTS {
                        ?compartment rdf:type nltl:compartment .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment rdf:type nltl:compartment .
                    FILTER NOT EXISTS {
                        ?compartment nltl:maintainedTemperature ?temperature .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    FILTER NOT EXISTS {
                        ?temperature rdf:type qudt:QuantityValue .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    FILTER NOT EXISTS {
                        ?temperature qudt:numericValue ?value .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature qudt:numericValue ?value .
                    FILTER (datatype(?value) != xsd:decimal)
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature qudt:numericValue ?value .
                    FILTER (datatype(?value) = xsd:decimal && ?value <= "0"^^xsd:decimal)
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature qudt:numericValue ?value1, ?value2 .
                    FILTER (
                        STR(?value1) != STR(?value2) ||
                        datatype(?value1) != datatype(?value2)
                    )
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    FILTER NOT EXISTS {
                        ?temperature qudt:unit unit:DEG_C .
                    }
                }
                UNION
                {
                    ?pump nltl:hasContainingCompartment ?compartment .
                    ?compartment nltl:maintainedTemperature ?temperature .
                    ?temperature qudt:unit ?temperatureUnit .
                    FILTER (?temperatureUnit != unit:DEG_C)
                }
            }
        """
    ] .
<END_SHACL>
