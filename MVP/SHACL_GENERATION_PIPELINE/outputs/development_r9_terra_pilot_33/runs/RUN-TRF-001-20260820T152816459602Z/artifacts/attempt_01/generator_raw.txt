<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF001IceClassRegulations2021Shape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:message "Construction contract date must be an xsd:date when provided."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021 ;
        sh:message "Section 1.8 of the 2021 Ice Class Regulations applies to every ship."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021 ;
        sh:message "Chapter 2 of the 2021 Ice Class Regulations applies to every ship."
    ] ;
    sh:sparql [
        sh:message "Ships contracted on or after 5 July 2021 must identify the 2021 Ice Class Regulations as applicable." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?contractDate .
                FILTER (
                    YEAR(?contractDate) > 2021 ||
                    (
                        YEAR(?contractDate) = 2021 &&
                        (
                            MONTH(?contractDate) > 7 ||
                            (MONTH(?contractDate) = 7 && DAY(?contractDate) >= 5)
                        )
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2021 .
                }
            }
        """
    ] .
<END_SHACL>
