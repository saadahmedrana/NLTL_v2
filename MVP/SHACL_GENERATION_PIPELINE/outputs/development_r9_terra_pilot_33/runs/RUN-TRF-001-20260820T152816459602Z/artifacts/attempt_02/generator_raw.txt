<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF001IceClassRegulationsShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:minCount 1 ;
        sh:datatype xsd:date ;
        sh:message "A construction contract date is required as an xsd:date."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassSection1Point8Provision2021
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulatoryProvision ;
        sh:hasValue nltl:iceClassDraughtChapter2Provision2021
    ] ;
    sh:sparql [
        sh:message "Ships contracted on or after 2021-07-05 must identify the 2021 Ice Class Regulations edition." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:constructionContractDate ?date .
                FILTER(datatype(?date) = xsd:date)
                FILTER(
                    YEAR(?date) > 2021 ||
                    (YEAR(?date) = 2021 && MONTH(?date) > 7) ||
                    (YEAR(?date) = 2021 && MONTH(?date) = 7 && DAY(?date) >= 5)
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                        nltl:iceClassRegulationEdition2021 .
                }
            }
        """
    ] .
<END_SHACL>
