<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:DimensionlessQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:PressureQuantityShape a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:MegaPA ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gen:DirectAnalysisCaseShape a sh:NodeShape ;
    sh:class nltl:directAnalysisCase ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape ;
        sh:message "Each direct analysis case must provide iceLoadAreaFactorCa."
    ] .

gen:TRF037Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:minCount 1 ;
        sh:class nltl:directAnalysisCase ;
        sh:node gen:DirectAnalysisCaseShape ;
        sh:message "A direct analysis case with iceLoadAreaFactorCa is required."
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientA ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientB ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputCoefficientK ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:shipSizeEngineOutputFactorCd ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:iceClassFactorCp ;
        sh:minCount 1 ;
        sh:node gen:DimensionlessQuantityShape
    ] ;
    sh:property [
        sh:path nltl:nominalIcePressureP0 ;
        sh:minCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:minCount 1 ;
        sh:node gen:PressureQuantityShape
    ] ;
    sh:sparql [
        sh:message "shipSizeEngineOutputFactorCd must equal min(1, (a*k+b)/1000) within a relative tolerance of 1e-6 and absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:shipSizeEngineOutputCoefficientA ?aQuantity ;
                      nltl:shipSizeEngineOutputCoefficientB ?bQuantity ;
                      nltl:shipSizeEngineOutputCoefficientK ?kQuantity ;
                      nltl:shipSizeEngineOutputFactorCd ?cdQuantity .
                ?aQuantity qudt:numericValue ?aValue .
                ?bQuantity qudt:numericValue ?bValue .
                ?kQuantity qudt:numericValue ?kValue .
                ?cdQuantity qudt:numericValue ?cdValue .
                BIND ((?aValue * ?kValue + ?bValue) / 1000 AS ?rawCd)
                BIND (IF(?rawCd > 1, 1, ?rawCd) AS ?expectedCd)
                FILTER (
                    ABS(?cdValue - ?expectedCd) >
                    ((0.000001 * ABS(?expectedCd)) + 0.000000001)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "icePressure must equal shipSizeEngineOutputFactorCd multiplied by iceClassFactorCp, iceLoadAreaFactorCa, and nominalIcePressureP0 within a relative tolerance of 1e-6 and absolute fallback of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:hasDirectAnalysisCase ?case ;
                      nltl:shipSizeEngineOutputFactorCd ?cdQuantity ;
                      nltl:iceClassFactorCp ?cpQuantity ;
                      nltl:nominalIcePressureP0 ?p0Quantity ;
                      nltl:icePressure ?pressureQuantity .
                ?case nltl:iceLoadAreaFactorCa ?caQuantity .
                ?cdQuantity qudt:numericValue ?cdValue .
                ?cpQuantity qudt:numericValue ?cpValue .
                ?caQuantity qudt:numericValue ?caValue .
                ?p0Quantity qudt:numericValue ?p0Value .
                ?pressureQuantity qudt:numericValue ?pressureValue .
                BIND ((?cdValue * ?cpValue * ?caValue * ?p0Value) AS ?expectedPressure)
                FILTER (
                    ABS(?pressureValue - ?expectedPressure) >
                    ((0.000001 * ABS(?expectedPressure)) + 0.000000001)
                )
            }
        """
    ] .
<END_SHACL>
