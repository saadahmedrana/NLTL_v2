            SELECT $this

                      nltl:spacing ?spacingQuantity .

            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?heightQuantity ;
                      nltl:spacing ?spacingQuantity ;
                      nltl:longitudinalPlatingFactorF2 ?factorQuantity .
                ?heightQuantity qudt:numericValue ?h .
                ?spacingQuantity qudt:numericValue ?s .
                ?factorQuantity qudt:numericValue ?reportedFactor .
                FILTER (?h != 0 && ?s != 0)
                BIND((?h / ?s) AS ?ratio)
                FILTER (?ratio <= 1.8)
                BIND(

                        ?ratio <= 1,
                        (0.4 / ?ratio) + 0.6,
                        1.4 - (0.4 * ?ratio)
                    )
                    AS ?expectedFactor
                )
                BIND(
                    IF(
                        (0.000001 * ABS(?expectedFactor)) > 0.000000000001,
                        0.000001 * ABS(?expectedFactor),
                        0.000000000001
                    )
                    AS ?tolerance
                )
                FILTER (ABS(?reportedFactor - ?expectedFactor) > ?tolerance)
            }
        """
    ] .
