<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF049FrameAttachmentShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;

    sh:sparql [
        sh:message "Within the ice-strengthened area, each supporting bulkhead or web frame must have an effectively confirmed attachment to the frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:withinIceStrengthenedArea true ;
                      (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:effectiveAttachmentConfirmed true .
                    ?support nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "A longitudinal frame must be attached by a bracket to each supporting bulkhead and supporting web frame." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:longitudinalFrame ;
                      (nltl:hasSupportingBulkhead|nltl:hasSupportingWebFrame) ?support .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment nltl:hasAttachedSupportingStructure ?support ;
                                nltl:hasConnectionBracket ?bracket .
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "A transverse frame terminating at a deck or ice stringer must have a bracket or similar attachment construction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this a nltl:transverseFrame ;
                      nltl:terminatesAtDeckOrIceStringer true .
                FILTER NOT EXISTS {
                    $this nltl:hasFrameAttachment ?attachment .
                    ?attachment ?attachmentProperty ?attachmentValue .
                    FILTER(
                        ?attachmentProperty = nltl:hasConnectionBracket ||
                        (
                            ?attachmentProperty = nltl:similarAttachmentConstructionPresent &&
                            ?attachmentValue = true
                        )
                    )
                }
            }
        """
    ] ;

    sh:sparql [
        sh:message "A frame passing through a supporting structure must have both web-plate sides connected." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:passesThroughSupportingStructure true .
                OPTIONAL {
                    $this nltl:webPlateConnectionSideCount ?sideCount .
                }
                FILTER(
                    !BOUND(?sideCount) ||
                    datatype(?sideCount) != xsd:integer ||
                    ?sideCount < 2
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "Each installed bracket must be at least as thick as the frame web and have an edge stiffened against buckling." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                FILTER NOT EXISTS {
                    $this nltl:frameWebThickness ?webThickness .
                    ?webThickness qudt:numericValue ?webValue ;
                                  qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    ?bracket nltl:bracketThickness ?bracketThickness ;
                             nltl:bracketEdgeStiffened true ;
                             nltl:bucklingStiffening true .
                    ?bracketThickness qudt:numericValue ?bracketValue ;
                                      qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                    FILTER(
                        datatype(?webValue) = xsd:decimal &&
                        datatype(?bracketValue) = xsd:decimal &&
                        ?bracketValue >= ?webValue
                    )
                }
            }
        """
    ] .
<END_SHACL>
