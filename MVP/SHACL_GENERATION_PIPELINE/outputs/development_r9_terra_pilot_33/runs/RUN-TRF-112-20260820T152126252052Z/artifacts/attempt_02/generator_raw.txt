<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:generated-shape:TRF-112-ShipBladeFailureReadinessShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Each blade-failure propulsion load case must provide valid blade-failure ultimate-load and combined-load inputs in kN." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .

                FILTER (
                    !EXISTS {
                        ?loadCase a nltl:loadCase ;
                            nltl:bladeFailureUltimateLoad ?bladeLoad ;
                            nltl:combinedAxialBendingTorsionLoad ?combinedLoad .

                        ?bladeLoad a qudt:QuantityValue ;
                            qudt:numericValue ?bladeValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?bladeValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?bladeLoad qudt:numericValue ?otherBladeValue .
                            FILTER (datatype(?otherBladeValue) != xsd:decimal)
                        }
                        FILTER NOT EXISTS {
                            ?bladeLoad qudt:unit ?otherBladeUnit .
                            FILTER (?otherBladeUnit != <http://qudt.org/vocab/unit/KiloN>)
                        }

                        ?combinedLoad a qudt:QuantityValue ;
                            qudt:numericValue ?combinedValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloN> .
                        FILTER (datatype(?combinedValue) = xsd:decimal)
                        FILTER NOT EXISTS {
                            ?combinedLoad qudt:numericValue ?otherCombinedValue .
                            FILTER (datatype(?otherCombinedValue) != xsd:decimal)
                        }
                        FILTER NOT EXISTS {
                            ?combinedLoad qudt:unit ?otherCombinedUnit .
                            FILTER (?otherCombinedUnit != <http://qudt.org/vocab/unit/KiloN>)
                        }
                    }
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "Each linked propeller shaft line component must provide bending and torsional yield safety factors of at least 1.0 for the blade-failure load case." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasPropulsionLoadCase ?loadCase ;
                      nltl:hasPropellerShaftLineComponent ?component .
                ?loadCase nltl:shaftLineLoadCaseType nltl:bladeFailureLoadCase .

                FILTER (
                    !EXISTS {
                        ?component a nltl:propellerShaftLineComponent ;
                            nltl:bendingYieldSafetyFactor ?bendingFactor ;
                            nltl:torsionalYieldSafetyFactor ?torsionalFactor .

                        ?bendingFactor a qudt:QuantityValue ;
                            qudt:numericValue ?bendingValue ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?bendingValue) = xsd:decimal &&
                            ?bendingValue >= "1.0"^^xsd:decimal
                        )
                        FILTER NOT EXISTS {
                            ?bendingFactor qudt:numericValue ?otherBendingValue .
                            FILTER (
                                datatype(?otherBendingValue) != xsd:decimal ||
                                ?otherBendingValue < "1.0"^^xsd:decimal
                            )
                        }
                        FILTER NOT EXISTS {
                            ?bendingFactor qudt:unit ?otherBendingUnit .
                            FILTER (?otherBendingUnit != <http://qudt.org/vocab/unit/UNITLESS>)
                        }

                        ?torsionalFactor a qudt:QuantityValue ;
                            qudt:numericValue ?torsionalValue ;
                            qudt:unit <http://qudt.org/vocab/unit/UNITLESS> .
                        FILTER (
                            datatype(?torsionalValue) = xsd:decimal &&
                            ?torsionalValue >= "1.0"^^xsd:decimal
                        )
                        FILTER NOT EXISTS {
                            ?torsionalFactor qudt:numericValue ?otherTorsionalValue .
                            FILTER (
                                datatype(?otherTorsionalValue) != xsd:decimal ||
                                ?otherTorsionalValue < "1.0"^^xsd:decimal
                            )
                        }
                        FILTER NOT EXISTS {
                            ?torsionalFactor qudt:unit ?otherTorsionalUnit .
                            FILTER (?otherTorsionalUnit != <http://qudt.org/vocab/unit/UNITLESS>)
                        }
                    }
                )
            }
        """
    ] .
<END_SHACL>
