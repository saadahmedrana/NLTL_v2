<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:generated-shape:> .

gs:TRF114ThrusterIceImpactShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasThrusterIceImpactLoadCase ;
        sh:class nltl:thrusterIceImpactLoadCase ;
        sh:node gs:TRF114ImpactLoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:equivalentImpactSphereRadius ;
        sh:node gs:TRF114LengthQuantityShape
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasThrusterIceImpactLoadCase ;
                sh:maxCount 0
            ]
        ]
        [
            sh:property [
                sh:path nltl:hasThrusterIceImpactLoadCase ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path nltl:iceOperatingSpeed ;
                sh:minCount 1 ;
                sh:node gs:TRF114SpeedQuantityShape ;
                sh:message "A ship with thruster ice impact load cases must provide its ice operating speed."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "A non-hemispherical contact geometry requires an equivalent impact sphere radius." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            SELECT $this
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:contactGeometryClassification nltl:nonHemisphericalContactGeometry .
                FILTER NOT EXISTS {
                    $this nltl:equivalentImpactSphereRadius ?radius .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "Thruster resistance capacity must be at least the thruster ice impact demand." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this ?case ?body ?capacityValue ?demandValue
            WHERE {
                $this nltl:hasThrusterIceImpactLoadCase ?case .
                ?case nltl:thrusterImpactCaseAppliesToBody ?body ;
                      nltl:thrusterIceImpactDemand ?demandNode .
                ?body nltl:thrusterResistanceCapacity ?capacityNode .
                ?demandNode qudt:numericValue ?demandValue .
                ?capacityNode qudt:numericValue ?capacityValue .
                FILTER(xsd:decimal(?capacityValue) < xsd:decimal(?demandValue))
            }
        """
    ] .

gs:TRF114ImpactLoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:thrusterIceImpactLoadCase ;
    sh:property [
        sh:path nltl:designIceBlockTableReference ;
        sh:minCount 1 ;
        sh:hasValue nltl:traficomTable6Dash3Reference ;
        sh:message "The impact load case must reference TRAFICOM Table 6-3 for the design ice block."
    ] ;
    sh:property [
        sh:path nltl:impactLoadCaseTableReference ;
        sh:minCount 1 ;
        sh:hasValue nltl:traficomTable6Dash16Reference ;
        sh:message "The impact load case must reference TRAFICOM Table 6-16."
    ] ;
    sh:property [
        sh:path nltl:thrusterImpactCaseAppliesToBody ;
        sh:minCount 1 ;
        sh:class nltl:thrusterBody ;
        sh:node gs:TRF114ThrusterBodyShape ;
        sh:message "The impact load case must apply to a thruster body."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoad ;
        sh:minCount 1 ;
        sh:node gs:TRF114ForceQuantityShape ;
        sh:message "The impact load case must provide the thruster ice impact load."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactDemand ;
        sh:minCount 1 ;
        sh:node gs:TRF114ForceQuantityShape ;
        sh:message "The impact load case must provide the thruster ice impact demand."
    ] ;
    sh:property [
        sh:path nltl:thrusterIceImpactLoadedAreaEvidence ;
        sh:minCount 1 ;
        sh:class nltl:evidenceArtifact ;
        sh:message "The impact load case must provide loaded-area evidence."
    ] ;
    sh:property [
        sh:path nltl:contactGeometryClassification ;
        sh:minCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:hemisphericalContactGeometry
            ]
        ]
        [
            sh:property [
                sh:path nltl:contactGeometryClassification ;
                sh:hasValue nltl:nonHemisphericalContactGeometry
            ] ;
            sh:property [
                sh:path nltl:contactGeometryCorrespondenceEvidence ;
                sh:minCount 1 ;
                sh:class nltl:evidenceArtifact ;
                sh:message "Non-hemispherical contact geometry requires correspondence evidence."
            ]
        ]
    ) .

gs:TRF114ThrusterBodyShape
    a sh:NodeShape ;
    sh:class nltl:thrusterBody ;
    sh:property [
        sh:path nltl:thrusterResistanceCapacity ;
        sh:minCount 1 ;
        sh:node gs:TRF114ForceQuantityShape ;
        sh:message "The thruster body must provide its resistance capacity."
    ] .

gs:TRF114ForceQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KiloN
    ] .

gs:TRF114LengthQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:M
    ] .

gs:TRF114SpeedQuantityShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:minCount 1 ;
        sh:datatype xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue unit:KN
    ] .
<END_SHACL>
