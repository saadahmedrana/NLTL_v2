<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:FirePumpCompartmentTemperatureShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each fire pump must be located in a compartment maintained above 0 degrees Celsius." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX unit: <http://qudt.org/vocab/unit/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT $this ?pump
WHERE {
    $this nltl:hasComponent ?pump .
    ?pump rdf:type ?pumpType .
    FILTER (
        ?pumpType IN (
            nltl:firePump,
            nltl:emergencyFirePump,
            nltl:waterMistPump,
            nltl:waterSprayPump
        )
    )
    FILTER (
        NOT EXISTS {
            ?pump nltl:hasContainingCompartment ?assignedCompartment .
        }
        ||
        EXISTS {
            ?pump nltl:hasContainingCompartment ?compartment .
            FILTER NOT EXISTS {
                ?compartment rdf:type nltl:compartment ;
                    nltl:maintainedTemperature ?temperature .
                ?temperature rdf:type qudt:QuantityValue ;
                    qudt:numericValue ?numericValue ;
                    qudt:unit unit:DEG_C .
                FILTER (
                    DATATYPE(?numericValue) = xsd:decimal &&
                    ?numericValue > "0"^^xsd:decimal
                )
            }
        }
    )
}
"""
    ] .
<END_SHACL>
