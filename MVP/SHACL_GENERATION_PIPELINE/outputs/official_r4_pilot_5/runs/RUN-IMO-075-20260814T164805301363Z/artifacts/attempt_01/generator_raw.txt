<BEGIN_SHACL>
@prefix gen: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO075SurvivalEquipmentShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The survival-equipment applicability status must be recorded as one boolean value."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:personsOnBoard ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Persons on board must be recorded when survival equipment is required."
            ] ;
            sh:property [
                sh:path nltl:availableSurvivalEquipmentCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Available survival-equipment capacity must be recorded when survival equipment is required."
            ] ;
            sh:property [
                sh:path nltl:storageAccessibilityStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Required survival equipment must be stored in an accessible location."
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Available survival-equipment capacity must be at least 110% of persons on board, rounded up." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:personalOrGroupSurvivalEquipmentRequired true ;
                      nltl:personsOnBoard ?persons ;
                      nltl:availableSurvivalEquipmentCapacity ?capacity .
                FILTER (
                    xsd:decimal(?capacity) <
                    CEIL(xsd:decimal(?persons) * 1.10)
                )
            }
        """
    ] .
<END_SHACL>
