<BEGIN_SHACL>
@prefix g: <urn:nltl:generated-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:SurvivalEquipmentCapacityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Exactly one boolean applicability value is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:personalOrGroupSurvivalEquipmentRequired ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:personsOnBoard ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Exactly one integer persons-on-board count is required when survival equipment is required."
            ] ;
            sh:property [
                sh:path nltl:availableSurvivalEquipmentCapacity ;
                sh:datatype xsd:integer ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:message "Exactly one integer survival-equipment capacity is required when survival equipment is required."
            ] ;
            sh:property [
                sh:path nltl:storageAccessibilityStatus ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "Storage must be recorded as accessible when survival equipment is required."
            ] ;
            sh:sparql [
                sh:message "Available survival-equipment capacity must be at least ceil(persons on board multiplied by 1.10)." ;
                sh:select """
                    SELECT $this
                    WHERE {
                        $this <https://w3id.org/nltl-benchmark/vocab#personsOnBoard> ?persons ;
                              <https://w3id.org/nltl-benchmark/vocab#availableSurvivalEquipmentCapacity> ?capacity .
                        FILTER (?capacity < CEIL(?persons * 1.10))
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
