<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_001_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        nltl:I2_001_Post2021Case
        nltl:I2_001_Pre2021Case
    ) ;
    sh:message "The ship's construction contract date must determine UR I2 Rev.4 applicability." .

nltl:I2_001_Post2021Case
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:minInclusive "2021-01-01"^^xsd:date ;
        sh:maxCount 1 ;
        sh:message "A ship contracted for construction on or after 1 January 2021 must have a valid construction contract date."
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue true ;
        sh:message "UR I2 Rev.4 must be applicable to ships contracted for construction on or after 1 January 2021."
    ] .

nltl:I2_001_Pre2021Case
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxExclusive "2021-01-01"^^xsd:date ;
        sh:maxCount 1 ;
        sh:message "A ship outside the UR I2 Rev.4 applicability date must have a valid construction contract date."
    ] ;
    sh:property [
        sh:path nltl:urI2RevisionApplicability ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:hasValue false ;
        sh:message "UR I2 Rev.4 must not be applicable to ships contracted for construction before 1 January 2021."
    ] .
<END_SHACL>
