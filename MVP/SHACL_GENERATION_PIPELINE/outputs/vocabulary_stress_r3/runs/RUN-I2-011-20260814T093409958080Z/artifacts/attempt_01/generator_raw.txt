<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_011_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:bulbousBowPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc1
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc2
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc3
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc4
            ]
        ]
        [
            sh:property [
                sh:path nltl:polarClass ;
                sh:hasValue nltl:polarClassPc5
            ]
        ]
        [
            sh:property [
                sh:path nltl:bulbousBowPresent ;
                sh:hasValue false
            ]
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:polarClass ;
                        sh:in (
                            nltl:polarClassPc6
                            nltl:polarClassPc7
                        )
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bulbousBowPresent ;
                        sh:hasValue true
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ;
                        sh:minCount 1 ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/MegaN> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:sparql [
                        sh:message "For an applicable PC6 or PC7 ship with a bulbous bow, the design force from I2.3.2.1 (iv) must not be less than the force from I2.3.2.1 (iii)." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

                            SELECT $this
                            WHERE {
                                $this nltl:bowDesignForceMethodI2Point3Point2Point1PartIii ?iiiNode ;
                                      nltl:bowDesignForceMethodI2Point3Point2Point1PartIv ?ivNode .
                                ?iiiNode <http://qudt.org/schema/qudt/numericValue> ?iii .
                                ?ivNode <http://qudt.org/schema/qudt/numericValue> ?iv .
                                FILTER (
                                    xsd:decimal(?iv) < xsd:decimal(?iii)
                                )
                            }
                        """
                    ]
                ]
            )
        )
    ) .

</END_SHACL>
