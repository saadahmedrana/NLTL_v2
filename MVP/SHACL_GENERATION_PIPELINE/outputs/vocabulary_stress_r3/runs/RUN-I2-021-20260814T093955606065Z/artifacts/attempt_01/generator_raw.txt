<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_021_ShellPlateThicknessShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:thickness ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship must provide the required shell plate thickness as a QUDT quantity in metres."
    ] ;
    sh:property [
        sh:path nltl:hasPlating ;
        sh:class nltl:plating ;
        sh:node [
            sh:property [
                sh:path nltl:iceLoadRequiredNetPlateThickness ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:corrosionAbrasionAllowance ;
                sh:minCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Shell plate thickness must equal the ice-load-required net plate thickness plus the corrosion-abrasion allowance, converted from millimetres to metres; comparison uses a 1e-6 relative tolerance with a 1e-6 absolute tolerance near zero." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:thickness ?thicknessQuantity ;
                      nltl:hasPlating ?plating .
                ?thicknessQuantity qudt:numericValue ?actualThickness .
                ?plating nltl:iceLoadRequiredNetPlateThickness ?netQuantity ;
                         nltl:corrosionAbrasionAllowance ?allowanceQuantity .
                ?netQuantity qudt:numericValue ?netThickness .
                ?allowanceQuantity qudt:numericValue ?allowance .
                BIND ((?netThickness + ?allowance) / 1000 AS ?expectedThickness)
                FILTER (
                    ABS(?actualThickness - ?expectedThickness) >
                    IF(
                        ABS(?expectedThickness) < 0.001,
                        0.000001,
                        0.000001 * ABS(?expectedThickness)
                    )
                )
            }
        """
    ] .
<END_SHACL>
