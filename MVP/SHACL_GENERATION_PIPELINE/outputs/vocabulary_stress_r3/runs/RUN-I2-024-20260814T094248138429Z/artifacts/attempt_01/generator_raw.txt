<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:I2-024-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:framingAngle ;
        sh:datatype qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:I2-024-FramingAngleShape> ;
        sh:message "A ship must provide one framing angle quantity in degrees."
    ] ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:class nltl:calculationCase ;
        sh:node <urn:nltl:stress-shape:I2-024-CalculationCaseShape>
    ] ;
    sh:property [
        sh:path nltl:hasObliquePlatingInterpolationCase ;
        sh:class nltl:obliquePlatingInterpolationCase ;
        sh:node <urn:nltl:stress-shape:I2-024-ObliqueCaseShape>
    ] ;
    sh:sparql [
        sh:message "For a framing angle strictly between 20 and 70 degrees, an oblique plating interpolation case must be provided." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:framingAngle ?angleQuantity .
                ?angleQuantity qudt:numericValue ?angle .
                FILTER (?angle > 20.0 && ?angle < 70.0)
                FILTER NOT EXISTS {
                    $this nltl:hasObliquePlatingInterpolationCase ?case .
                }
            }
        """
    ] .

<urn:nltl:stress-shape:I2-024-FramingAngleShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/DEG> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:stress-shape:I2-024-CalculationCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must identify one lower interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must identify one upper interpolation endpoint."
    ] ;
    sh:property [
        sh:path nltl:longitudinalFramingNetPlateThickness ;
        sh:node <urn:nltl:stress-shape:I2-024-ThicknessShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must provide one longitudinal framing net plate thickness."
    ] ;
    sh:property [
        sh:path nltl:transverseFramingNetPlateThickness ;
        sh:node <urn:nltl:stress-shape:I2-024-ThicknessShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each calculation case must provide one transverse framing net plate thickness."
    ] .

<urn:nltl:stress-shape:I2-024-ObliqueCaseShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolatedNetPlateThickness ;
        sh:node <urn:nltl:stress-shape:I2-024-ThicknessShape> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each oblique plating interpolation case must provide one interpolated net plate thickness."
    ] .

<urn:nltl:stress-shape:I2-024-ThicknessShape>
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
