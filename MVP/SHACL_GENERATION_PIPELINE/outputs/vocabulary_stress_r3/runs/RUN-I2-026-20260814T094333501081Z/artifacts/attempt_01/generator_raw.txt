<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:I2_026_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:class nltl:structuralMember ;
        sh:node nltl:I2_026_StructuralMemberShape ;
        sh:message "Each linked structural member must provide support-condition information and support data."
    ] .

nltl:I2_026_StructuralMemberShape
    a sh:NodeShape ;
    sh:class nltl:structuralMember ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:class nltl:memberSupport ;
        sh:node nltl:I2_026_MemberSupportShape ;
        sh:minCount 1 ;
        sh:message "A structural member must identify the support information needed to establish its support fixity."
    ] ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:class nltl:frameSupportConditionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A structural member must have exactly one frame support condition."
    ] ;
    sh:or (
        nltl:I2_026_FixedSupportCaseShape
        nltl:I2_026_SimpleSupportCaseShape
    ) ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member support demonstrating significant rotational restraint must have significant rotational restraint evidence." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasMemberSupport ?support .
                ?support nltl:demonstratedRotationalRestraint true .
                FILTER NOT EXISTS {
                    $this nltl:significantRotationalRestraintEvidence ?evidence .
                }
            }
        """ ;
    ] .

nltl:I2_026_MemberSupportShape
    a sh:NodeShape ;
    sh:class nltl:memberSupport ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1 ;
    ] .

nltl:I2_026_EvidenceShape
    a sh:NodeShape ;
    sh:class nltl:evidenceArtifact .

nltl:I2_026_FixedSupportCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:fixedFrameSupport ;
    ] ;
    sh:or (
        nltl:I2_026_ContinuousSupportShape
        nltl:I2_026_BracketSupportShape
        nltl:I2_026_DemonstratedRestraintSupportShape
    ) .

nltl:I2_026_ContinuousSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:continuousThroughSupport ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A fixed support case must demonstrate continuity through the support."
    ] .

nltl:I2_026_BracketSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node nltl:I2_026_BracketMemberSupportShape ;
        sh:message "A fixed support case must identify a support with a connection bracket."
    ] .

nltl:I2_026_BracketMemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:connectionBracketPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
    ] .

nltl:I2_026_DemonstratedRestraintSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasMemberSupport ;
        sh:minCount 1 ;
        sh:node nltl:I2_026_RestraintMemberSupportShape ;
    ] .

nltl:I2_026_RestraintMemberSupportShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:demonstratedRotationalRestraint ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
    ] .

nltl:I2_026_SimpleSupportCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:frameSupportCondition ;
        sh:hasValue nltl:simpleFrameSupport ;
    ] .

nltl:I2_026_EvidencePropertyShape
    a sh:PropertyShape ;
    sh:path nltl:significantRotationalRestraintEvidence ;
    sh:class nltl:evidenceArtifact ;
    sh:node nltl:I2_026_EvidenceShape .
</END_SHACL>
