stress:CalculationCaseShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:plasticNeutralAxisHeight ;
        sh:node stress:MillimetreQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:interpolationLowerEndpoint ;
        sh:class nltl:interpolationPoint
    ] ;
    sh:property [
        sh:path nltl:interpolationUpperEndpoint ;
        sh:class nltl:interpolationPoint
    ] ;
    sh:sparql [
        sh:message "The plastic neutral axis height and section modulus must satisfy the I2.5.8-I2.5.9 calculation, or the oblique-framing result must be obtained by linear interpolation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega .
                ?member nltl:netLocalFrameFlangeArea ?aQ ;
                        nltl:webHeight ?hQ ;
                        nltl:netWebThickness ?twQ ;
                        nltl:netAttachedShellPlateThickness ?tpQ ;
                        nltl:frameSpacing ?sQ ;
                        nltl:localFrameFlangeCentreHeight ?hfcQ ;
                        nltl:webAngleToShellPlate ?phiQ ;
                        nltl:webToFlangeCentreDistance ?bwQ ;
                        nltl:netEffectivePlasticSectionModulus ?zpQ .
                ?aQ qudt:numericValue ?a .
                ?hQ qudt:numericValue ?h .
                ?twQ qudt:numericValue ?tw .
                ?tpQ qudt:numericValue ?tp .
                ?sQ qudt:numericValue ?s .
                ?hfcQ qudt:numericValue ?hfc .
                ?phiQ qudt:numericValue ?phi .
                ?bwQ qudt:numericValue ?bw .
                ?zpQ qudt:numericValue ?reportedZp .
                $this nltl:plasticNeutralAxisHeight ?znaQ .
                ?znaQ qudt:numericValue ?reportedZna .
                BIND ((100 * ?a + ?h * ?tw - 1000 * ?tp * ?s) / (2 * ?tw) AS ?zna)
                BIND (?phi * 3.141592653589793 / 180 AS ?phiRad)
                BIND (
                    ?tp * ?s * (?zna + ?tp / 2) * math:sin(?phiRad) +
                    (((?h - ?zna) * (?h - ?zna) + ?zna * ?zna) * ?tw * math:sin(?phiRad)) / 2000 +
                    ?a * ((?hfc - ?zna) * math:sin(?phiRad) - ?bw * math:cos(?phiRad)) / 10
                    AS ?zp
                )
                FILTER (
                    (?omega <= 20 || ?omega >= 70) &&
                    (
                        ABS(?reportedZna - ?zna) >
                        (0.000001 * ABS(?zna) + 0.000001) ||
                        ABS(?reportedZp - ?zp) >
                        (0.000001 * ABS(?zp) + 0.000001)
                    )
                )
            }
        """
    ] .

stress:InterpolationReadinessShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        sh:message "Oblique framing requires linked lower and upper interpolation points with dimensionless coordinates and results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasCalculationCase ?case ;
                      nltl:hasStructuralMember ?member .
                ?member nltl:hasLocalFrameSectionCalculationCase ?localCase .
                ?localCase nltl:sectionCalculationCasePlating ?plating .
                ?plating nltl:framingAngleOmega ?omegaQ .
                ?omegaQ qudt:numericValue ?omega .
                FILTER (?omega > 20 && ?omega < 70)
                FILTER (
                    NOT EXISTS {
                        ?case nltl:interpolationLowerEndpoint ?lower ;
                              nltl:interpolationUpperEndpoint ?upper .
                        ?lower a nltl:interpolationPoint ;
                               nltl:interpolationPointCoordinate ?lc ;
                               nltl:interpolationPointResult ?lr .
                        ?upper a nltl:interpolationPoint ;
                               nltl:interpolationPointCoordinate ?uc ;
                               nltl:interpolationPointResult ?ur .
                        ?lc qudt:numericValue ?lcv .
                        ?uc qudt:numericValue ?ucv .
                        ?lr qudt:numericValue ?lrv .
                        ?ur qudt:numericValue ?urv .
                        FILTER (?lcv < ?ucv)
                    }
                )
            }
        """
    ] .
