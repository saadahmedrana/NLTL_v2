<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix ss: <urn:nltl:stress-shape:> .

ss:I2_031_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:patchLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide the patch load as a single kilonewton quantity."
    ] ;
    sh:property [
        sh:path nltl:plasticStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide the plastic strength as a single kilonewton quantity."
    ] ;
    sh:property [
        sh:path nltl:frameProfileType ;
        sh:in ( nltl:flatBarSection nltl:profileSection ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame profile type shall be a permitted profile section value."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node ss:I2_031_StructuralMemberShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The combined shear and bending demand shall not exceed the plastic strength of the member." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>

            SELECT $this
            WHERE {
                $this nltl:plasticStrength ?strengthQuantity .
                ?strengthQuantity qudt:numericValue ?strength .
                $this nltl:hasStructuralMember ?member .
                ?member nltl:hasStructuralMemberLoadCase ?loadCase .
                ?loadCase nltl:combinedShearAndBendingDemand ?demandQuantity .
                ?demandQuantity qudt:numericValue ?demand .
                FILTER ( ?demand > ?strength )
            }
        """
    ] .

ss:I2_031_StructuralMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:structuralLocation ;
        sh:in ( nltl:bottomStructureLocation nltl:sideStructureLocation ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each linked structural member shall identify a permitted bottom or side structure location."
    ] ;
    sh:property [
        sh:path nltl:midspanPlasticCollapseLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each linked structural member shall provide its midspan plastic collapse load as a single kilonewton quantity."
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMemberLoadCase ;
        sh:node ss:I2_031_LoadCaseShape
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For a bottom structural member, the patch load orientation shall be parallel with the frame direction." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:structuralLocation nltl:bottomStructureLocation .
                $this nltl:hasStructuralMemberLoadCase ?loadCase .
                FILTER NOT EXISTS {
                    ?loadCase nltl:loadPatchOrientation
                        nltl:loadPatchHeightParallelToFrame .
                }
            }
        """
    ] .

ss:I2_031_LoadCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combinedShearAndBendingDemand ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Each linked load case shall provide the combined shear and bending demand as a decimal quantity."
    ] ;
    sh:property [
        sh:path nltl:loadPatchOrientation ;
        sh:in ( nltl:loadPatchHeightParallelToFrame )
    ] .

</END_SHACL>
