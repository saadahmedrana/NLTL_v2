<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix stress: <urn:nltl:stress-shape:> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

stress:I2_040_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;

    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "yieldStrength must be a positive quantity in MPa."
    ] ;

    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node stress:I2_040_StructuralMemberShape
    ] ;

    sh:sparql [
        sh:message "For each structural member, h_w/t_wn must not exceed 282/sqrt(sigma_y) for flat-bar sections or 805/sqrt(sigma_y) for bulb, tee, and angle sections." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member ;
                      nltl:yieldStrength ?yieldQuantity .

                ?member nltl:structuralSectionType ?section ;
                        nltl:webHeight ?heightQuantity ;
                        nltl:netWebThickness ?thicknessQuantity .

                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit unit:MegaPA .
                ?heightQuantity qudt:numericValue ?height ;
                                qudt:unit unit:MilliM .
                ?thicknessQuantity qudt:numericValue ?thickness ;
                                    qudt:unit unit:MilliM .

                FILTER (?section IN (
                    nltl:flatBarStructuralSection,
                    nltl:bulbSection,
                    nltl:teeSection,
                    nltl:angleSection
                ))
                BIND(
                    IF(
                        ?section = nltl:flatBarStructuralSection,
                        "282"^^xsd:decimal,
                        "805"^^xsd:decimal
                    ) AS ?limit
                )
                FILTER (
                    ((?height / ?thickness) * (?height / ?thickness))
                    >
                    ((?limit * ?limit) / ?yield)
                )
            }
        """
    ] .

stress:I2_040_StructuralMemberShape
    a sh:NodeShape ;

    sh:property [
        sh:path nltl:structuralSectionType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:flatBarStructuralSection
            nltl:bulbSection
            nltl:teeSection
            nltl:angleSection
        ) ;
        sh:message "structuralSectionType must be flat-bar, bulb, tee, or angle section."
    ] ;

    sh:property [
        sh:path nltl:webHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "webHeight must be a positive quantity in millimetres."
    ] ;

    sh:property [
        sh:path nltl:netWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minExclusive "0"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MilliM ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "netWebThickness must be a positive quantity in millimetres."
    ] .
<END_SHACL>
