<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix ss: <urn:nltl:stress-shape:> .

ss:I2_041_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MegaPA ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            sh:class nltl:structuralMember ;
            sh:property [
                sh:path nltl:netAttachedShellPlateThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:netWebThickness ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:MilliM ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each structural member net web thickness must satisfy t_wn >= 0.35*t_pn*sqrt(sigma_y/235), within a relative tolerance of 1e-6 and an absolute squared-value fallback of 1e-12." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:yieldStrength ?yieldQuantity ;
                      nltl:hasStructuralMember ?member .
                ?yieldQuantity qudt:numericValue ?yield ;
                               qudt:unit <http://qudt.org/vocab/unit/MegaPA> .
                ?member nltl:netAttachedShellPlateThickness ?plateQuantity ;
                        nltl:netWebThickness ?webQuantity .
                ?plateQuantity qudt:numericValue ?plate ;
                              qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?webQuantity qudt:numericValue ?web ;
                            qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                BIND (0.35 * 0.35 * ?plate * ?plate * (?yield / 235.0) AS ?expectedSquared)
                FILTER (
                    ?web < 0 ||
                    (?web * ?web + (0.000001 * ?expectedSquared + 0.000000000001) < ?expectedSquared)
                )
            }
        """
    ] .
<END_SHACL>
