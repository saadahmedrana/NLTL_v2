<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:I2_048_ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarClassValue ;
        sh:in (
            nltl:polarClassPc1
            nltl:polarClassPc2
            nltl:polarClassPc3
            nltl:polarClassPc4
            nltl:polarClassPc5
            nltl:polarClassPc6
            nltl:polarClassPc7
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must have one applicable Polar Class value." ;
    ] ;
    sh:property [
        sh:path nltl:materialClass ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The ship must specify its structural material class for the Table 8 selection." ;
    ] ;
    sh:property [
        sh:path nltl:asBuiltThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/M> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "The ship must provide one as-built thickness quantity in metres." ;
    ] ;
    sh:property [
        sh:path nltl:hasSteelGradeRequirementCase ;
        sh:node nltl:stress_shape_I2_048_SteelGradeCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide a steel-grade requirement case." ;
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node nltl:stress_shape_I2_048_TableLookupCaseShape ;
        sh:minCount 1 ;
        sh:message "The ship must provide the Table 8 lookup case." ;
    ] .

nltl:stress_shape_I2_048_SteelGradeCaseShape
    a sh:NodeShape ;
    sh:class nltl:steelGradeRequirementCase ;
    sh:property [
        sh:path nltl:actualHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:requiredHullStructuralSteelGrade ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelGradeValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:asBuiltPlateThickness ;
        sh:nodeKind sh:IRI ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:property [
            sh:path qudt:numericValue ;
            sh:datatype xsd:decimal ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:property [
            sh:path qudt:unit ;
            sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
            sh:minCount 1 ;
            sh:maxCount 1
        ] ;
        sh:message "The steel-grade requirement case must provide one as-built plate thickness quantity in millimetres." ;
    ] ;
    sh:property [
        sh:path nltl:steelMaterialClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelMaterialClassValue ;
        sh:in (
            nltl:steelMaterialClassOne
            nltl:steelMaterialClassTwo
            nltl:steelMaterialClassThree
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:property [
        sh:path nltl:steelStrengthCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:steelStrengthCategoryValue ;
        sh:in (
            nltl:normalStrengthSteelCategory
            nltl:highTensileSteelCategory
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] ;
    sh:sparql [
        sh:message "The actual hull structural steel grade must not be lower than the Table 8 required steel grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:actualHullStructuralSteelGrade ?actual ;
                      nltl:requiredHullStructuralSteelGrade ?required .
                ?actual nltl:steelGradeOrderRank ?actualRank .
                ?required nltl:steelGradeOrderRank ?requiredRank .
                FILTER (
                    !sameTerm(?actual, ?required)
                    && xsd:integer(?actualRank) < xsd:integer(?requiredRank)
                )
            }
        """
    ] .

nltl:stress_shape_I2_048_TableLookupCaseShape
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:hasValue nltl:iacsUrI2Table8 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The lookup case must cite IACS UR I2 Table 8." ;
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The Table 8 lookup case must provide selection evidence." ;
    ] .
<END_SHACL>
