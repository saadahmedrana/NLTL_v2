nltl:StressI2_054_CalculationCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hullGirderLongitudinalPositionFromAft ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:bendingMomentDistributionFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:hasInterpolationPoint ;
        sh:minCount 1 ;
        sh:node nltl:StressI2_054_InterpolationPointShape
    ] ;
    sh:sparql [
        sh:message "The calculation case must use the prescribed C_m distribution and reproduce the ship design vertical ice bending moment within tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                ?ship nltl:hasCalculationCase $this ;
                      nltl:upperIceWaterlineLengthLUI ?luiNode ;
                      nltl:stemAngle ?angleNode ;
                      nltl:designVerticalIceForceAtBow ?forceNode ;
                      nltl:designVerticalIceBendingMoment ?resultNode .
                $this nltl:hullGirderLongitudinalPositionFromAft ?positionNode ;
                      nltl:bendingMomentDistributionFactor ?cmNode .
                ?luiNode qudt:numericValue ?lui .
                ?angleNode qudt:numericValue ?angle .
                ?forceNode qudt:numericValue ?force .
                ?resultNode qudt:numericValue ?observed .
                ?positionNode qudt:numericValue ?position .
                ?cmNode qudt:numericValue ?cm .
                FILTER (
                    datatype(?lui) = xsd:decimal &&
                    datatype(?angle) = xsd:decimal &&
                    datatype(?force) = xsd:decimal &&
                    datatype(?observed) = xsd:decimal &&
                    datatype(?position) = xsd:decimal &&
                    datatype(?cm) = xsd:decimal
                )
                FILTER (?lui > 0)
                BIND (?position / ?lui AS ?x)
                BIND (
                    IF(
                        ?x <= 0.5,
                        2.0 * ?x,
                        IF(
                            ?x <= 0.7,
                            1.0,
                            IF(
                                ?x <= 0.95,
                                1.0 - 0.7 * ((?x - 0.7) / 0.25),
                                0.3 * (1.0 - ((?x - 0.95) / 0.05))
                            )
                        )
                    ) AS ?expectedCm
                )
                BIND (math:sin(?angle * 3.141592653589793 / 180.0) AS ?sine)
                BIND (
                    0.1 * ?expectedCm * ?lui *
                    math:pow(?sine, -0.2) * ?force
                    AS ?expectedMoment
                )
                FILTER (
                    ?x < 0.0 || ?x > 1.0 ||
                    ABS(?cm - ?expectedCm) >
                        MAX(0.000001 * ABS(?expectedCm), 0.000000001) ||
                    ABS(?observed - ?expectedMoment) >
                        MAX(0.000001 * ABS(?expectedMoment), 0.000000001)
                )
            }
        """
    ] .

nltl:StressI2_054_InterpolationPointShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:interpolationPointCoordinate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:interpolationPointResult ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Each interpolation point result must follow the prescribed C_m values and linear interpolation." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:interpolationPointCoordinate ?coordinateNode ;
                      nltl:interpolationPointResult ?resultNode .
                ?coordinateNode qudt:numericValue ?x .
                ?resultNode qudt:numericValue ?observed .
                FILTER (
                    datatype(?x) = xsd:decimal &&
                    datatype(?observed) = xsd:decimal
                )
                BIND (
                    IF(
                        ?x <= 0.5,
                        2.0 * ?x,
                        IF(
                            ?x <= 0.7,
                            1.0,
                            IF(
                                ?x <= 0.95,
                                1.0 - 0.7 * ((?x - 0.7) / 0.25),
                                0.3 * (1.0 - ((?x - 0.95) / 0.05))
                            )
                        )
                    ) AS ?expected
                )
                FILTER (
                    ?x < 0.0 || ?x > 1.0 ||
                    ABS(?observed - ?expected) >
                        MAX(0.000001 * ABS(?expected), 0.000000001)
                )
            }
        """
    ] .
