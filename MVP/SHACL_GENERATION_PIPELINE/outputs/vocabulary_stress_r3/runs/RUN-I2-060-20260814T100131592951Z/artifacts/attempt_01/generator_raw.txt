<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:StructuralCutoutStabilityShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasStructuralMember ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path nltl:memberInWayOfCutout ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:property [
                sh:path nltl:stiffeningNecessary ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:memberInWayOfCutout ;
                        sh:hasValue true ;
                    ] ;
                    sh:property [
                        sh:path nltl:hasCutout ;
                        sh:minCount 1 ;
                    ] ;
                ]
                [
                    sh:property [
                        sh:path nltl:memberInWayOfCutout ;
                        sh:hasValue false ;
                    ]
                ]
            ) ;
        ] ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "A member carrying loads in way of a cut-out must not cause instability." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true .
                FILTER NOT EXISTS {
                    $this nltl:memberInstability ?instability .
                    FILTER ( ?instability = false )
                }
            }
        """ ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Where stiffening is necessary for a member in way of a cut-out, structural stiffening must be recorded." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasStructuralMember ?member .
                ?member nltl:memberInWayOfCutout true .
                ?member nltl:stiffeningNecessary true .
                FILTER NOT EXISTS {
                    $this nltl:structuralStiffening ?stiffening .
                    FILTER (
                        datatype(?stiffening) = xsd:string &&
                        STRLEN(STR(?stiffening)) > 0
                    )
                }
            }
        """ ;
    ] .
<END_SHACL>
