<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasCalculationCase ;
        sh:node <urn:nltl:stress-shape:CalculationCaseShape>
    ] ;
    sh:sparql [
        sh:message "For a nonlinear calculation case, fracture, major buckling, yielding, and deformation assessment inputs must be supplied with their prescribed units." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER EXISTS {
                    $this nltl:hasCalculationCase ?case .
                    ?case nltl:calculationMethod
                          nltl:nonlinearCalculationMethodValue .
                }
                FILTER (
                    NOT EXISTS {
                        $this nltl:fractureMargin ?fracture .
                        ?fracture a qudt:QuantityValue ;
                                  qudt:numericValue ?fractureValue ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?fractureValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:majorBucklingMargin ?majorBuckling .
                        ?majorBuckling a qudt:QuantityValue ;
                                       qudt:numericValue ?majorBucklingValue ;
                                       qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?majorBucklingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:yieldingMargin ?yielding .
                        ?yielding a qudt:QuantityValue ;
                                  qudt:numericValue ?yieldingValue ;
                                  qudt:unit qudt:UNITLESS .
                        FILTER (datatype(?yieldingValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:permanentLateralDeformation ?lateral .
                        ?lateral a qudt:QuantityValue ;
                                 qudt:numericValue ?lateralValue ;
                                 qudt:unit qudt:MilliM .
                        FILTER (datatype(?lateralValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:permanentOutOfPlaneDeformation ?outOfPlane .
                        ?outOfPlane a qudt:QuantityValue ;
                                    qudt:numericValue ?outOfPlaneValue ;
                                    qudt:unit qudt:MilliM .
                        FILTER (datatype(?outOfPlaneValue) = xsd:decimal)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:relevantStructuralDimension ?dimension .
                        ?dimension a qudt:QuantityValue ;
                                   qudt:numericValue ?dimensionValue ;
                                   qudt:unit qudt:MilliM .
                        FILTER (datatype(?dimensionValue) = xsd:decimal)
                    }
                )
            }
        """
    ] .

<urn:nltl:stress-shape:CalculationCaseShape>
    a sh:NodeShape ;
    sh:targetClass nltl:calculationCase ;
    sh:property [
        sh:path nltl:calculationMethod ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:nonlinearCalculationMethodValue
            nltl:directCalculationMethodValue
            nltl:prescribedAnalyticalProcedure
        )
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:nonlinearCalculationMethodValue
            ] ;
            sh:property [
                sh:path nltl:minorPermanentDeformationAssessmentSatisfied ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:nonlinearAnalysisAcceptanceEvidence ;
                sh:class nltl:evidenceArtifact ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:directCalculationMethodValue
            ]
        ]
        [
            sh:property [
                sh:path nltl:calculationMethod ;
                sh:hasValue nltl:prescribedAnalyticalProcedure
            ]
        ]
    ] .
<END_SHACL>
