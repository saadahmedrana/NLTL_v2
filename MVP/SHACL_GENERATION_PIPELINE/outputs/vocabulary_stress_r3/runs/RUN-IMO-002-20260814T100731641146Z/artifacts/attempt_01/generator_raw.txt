<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO_002_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceCondition ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "A ship must provide its design ice condition to determine applicability of the category rule."
    ] ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:polarShipCategoryValue ;
        sh:minCount 1 ;
        sh:message "A ship must provide its declared Polar Code category to determine applicability of the category rule."
    ] ;
    sh:sparql [
        sh:message "A ship that is not category A and is designed for at least thin first-year ice, including possible old-ice inclusions, must have ship category B." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:designIceCondition ?ice ;
                      nltl:shipCategory ?category .
                FILTER (datatype(?ice) = xsd:string)
                FILTER (regex(
                    str(?ice),
                    "thin[ -]+first[ -]+year[ -]+ice",
                    "i"
                ))
                FILTER (?category != nltl:polarShipCategoryA)
                FILTER NOT EXISTS {
                    $this nltl:shipCategory nltl:polarShipCategoryB .
                }
            }
        """
    ] .
<END_SHACL>
