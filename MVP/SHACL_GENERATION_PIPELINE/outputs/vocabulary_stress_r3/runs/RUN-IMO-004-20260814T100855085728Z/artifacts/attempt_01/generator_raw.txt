<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-004-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceOriginType ;
        sh:node [
            sh:in ( nltl:seaIceOrigin )
        ] ;
        sh:message "An ice origin value, when provided, must be sea ice."
    ] ;
    sh:property [
        sh:path nltl:winterGrowthCount ;
        sh:datatype xsd:integer ;
        sh:message "Winter growth count must be an integer."
    ] ;
    sh:property [
        sh:path nltl:iceThickness ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:maxCount 1 ;
                sh:message "Ice thickness must have a single decimal numeric value."
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:maxCount 1 ;
                sh:message "Ice thickness must use metres."
            ]
        ] ;
        sh:message "Ice thickness must be a quantity expressed in metres."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "When the ice is sea ice with no more than one winter growth and thickness from 0.3 m through 2.0 m, its ice type must be first-year ice." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceOriginType nltl:seaIceOrigin ;
                       nltl:winterGrowthCount ?growth ;
                       nltl:iceThickness ?thickness .
                ?thickness qudt:numericValue ?measuredThickness ;
                           qudt:unit unit:M .
                FILTER (datatype(?growth) = xsd:integer)
                FILTER (datatype(?measuredThickness) = xsd:decimal)
                FILTER (?growth <= 1)
                FILTER (?measuredThickness >= 0.3)
                FILTER (?measuredThickness <= 2.0)
                FILTER NOT EXISTS {
                    $this nltl:iceTypeClassification nltl:firstYearIce .
                }
            }
        """
    ] .
<END_SHACL>
