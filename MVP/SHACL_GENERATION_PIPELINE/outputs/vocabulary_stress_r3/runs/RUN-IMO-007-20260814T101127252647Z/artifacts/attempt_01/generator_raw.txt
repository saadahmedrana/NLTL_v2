<BEGIN_SHACL>
@prefix g: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:IMO_007_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceConditionClassification ;
        sh:class nltl:iceConditionValue ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A single ice condition classification is required to determine applicability."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For open water, sea ice concentration must be a decimal quantity in UNITLESS units less than 1, and ice of land origin must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceConditionClassification nltl:openWaterIceCondition .
                FILTER (
                    NOT EXISTS {
                        $this nltl:seaIceConcentration ?validConcentration .
                        ?validConcentration qudt:numericValue ?validNumber ;
                            qudt:unit qudt:UNITLESS .
                        FILTER (
                            datatype(?validNumber) = xsd:decimal &&
                            ?validNumber < 1
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:seaIceConcentration ?concentration .
                        OPTIONAL { ?concentration qudt:numericValue ?number . }
                        OPTIONAL { ?concentration qudt:unit ?unit . }
                        FILTER (
                            !BOUND(?number) ||
                            datatype(?number) != xsd:decimal ||
                            !BOUND(?unit) ||
                            ?unit != qudt:UNITLESS ||
                            !(?number < 1)
                        )
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:iceOfLandOriginPresent ?validLandOrigin .
                        FILTER (
                            datatype(?validLandOrigin) = xsd:boolean &&
                            ?validLandOrigin = false
                        )
                    }
                    ||
                    EXISTS {
                        $this nltl:iceOfLandOriginPresent ?landOrigin .
                        FILTER (
                            datatype(?landOrigin) != xsd:boolean ||
                            ?landOrigin != false
                        )
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
