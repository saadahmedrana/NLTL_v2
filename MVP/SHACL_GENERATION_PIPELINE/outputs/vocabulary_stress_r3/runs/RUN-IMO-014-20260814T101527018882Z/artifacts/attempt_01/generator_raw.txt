<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gen: <urn:nltl:stress-shape:> .

gen:IMO014Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:polarCodeApplies ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each ship must provide one boolean Polar Code applicability assessment."
    ] ;
    sh:or (
        gen:IMO014ApplicableShape
        gen:IMO014NotApplicableShape
    ) .

gen:IMO014ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:polarShipCertificatePresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A ship to which the Polar Code applies must have a Polar Ship Certificate."
    ] ;
    sh:property [
        sh:path nltl:polarShipCertificateStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A Polar Ship Certificate for an applicable ship must have valid status."
    ] .

gen:IMO014NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:polarCodeApplies ;
        sh:datatype xsd:boolean ;
        sh:hasValue false
    ] .
<END_SHACL>
