<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:stress-shape:> .

gs:IMO016ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPolarShipCertificate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node gs:IMO016CertificateShape ;
        sh:message "The ship shall have one Polar Ship Certificate conforming to the Appendix I model."
    ] .

gs:IMO016CertificateShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:certificateFormModel ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue nltl:appendixIPolarShipCertificateModel ;
        sh:message "The Polar Ship Certificate shall use the Appendix I model."
    ] ;
    sh:property [
        sh:path nltl:certificateLanguage ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The Polar Ship Certificate shall specify its language."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:in ( "English" "French" "Spanish" )
            ]
        ]
        [
            sh:property [
                sh:path nltl:certificateLanguage ;
                sh:not [
                    sh:in ( "English" "French" "Spanish" )
                ]
            ] ;
            sh:property [
                sh:path nltl:approvedTranslationPresent ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "A certificate in a language other than English, French, or Spanish shall include an approved translation."
            ]
        ]
    ) .
<END_SHACL>
