<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix stress: <urn:nltl:stress-shape:> .

stress:IMO_041_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:openingActuationType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:message "Opening actuation type is required to determine whether the hydraulic-fluid protection condition applies."
    ] ;
    sh:sparql [
        sh:message "Hydraulically operated openings require hydraulic-fluid freezing or excessive-viscosity control." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:openingActuationType ?actuation .
                FILTER (STR(?actuation) = "hydraulic")
                FILTER NOT EXISTS {
                    $this nltl:hydraulicFluidFreezingOrViscosityControlPresent ?present .
                    FILTER (
                        datatype(?present) = xsd:boolean &&
                        ?present = true
                    )
                }
            }
        """
    ] .
<END_SHACL>
