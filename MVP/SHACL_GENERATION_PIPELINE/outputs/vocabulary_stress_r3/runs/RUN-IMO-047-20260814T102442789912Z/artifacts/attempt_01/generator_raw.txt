<BEGIN_SHACL>
@prefix s: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

s:IMO047ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:hasEssentialEngine ;
                sh:maxCount 0
            ]
        ]
        s:IMO047RequirementsShape
    ) .

s:IMO047RequirementsShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:combustionAirTemperatureControlPresent ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Combustion-air temperature control shall be present for ships with an essential engine."
    ] ;
    sh:property [
        sh:path nltl:combustionAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node s:IMO047TemperatureQuantityShape ;
        sh:message "Combustion-air temperature shall be provided in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMinAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node s:IMO047TemperatureQuantityShape ;
        sh:message "The manufacturer's minimum combustion-air temperature criterion shall be provided in degrees Celsius."
    ] ;
    sh:property [
        sh:path nltl:manufacturerMaxAirTemperature ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node s:IMO047TemperatureQuantityShape ;
        sh:message "The manufacturer's maximum combustion-air temperature criterion shall be provided in degrees Celsius."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Combustion-air temperature shall be within the manufacturer's specified minimum and maximum criteria." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:combustionAirTemperature ?temperature .
                ?temperature qudt:numericValue ?temperatureValue ;
                             qudt:unit qudt:DEG_C .

                $this nltl:manufacturerMinAirTemperature ?minimum .
                ?minimum qudt:numericValue ?minimumValue ;
                         qudt:unit qudt:DEG_C .

                $this nltl:manufacturerMaxAirTemperature ?maximum .
                ?maximum qudt:numericValue ?maximumValue ;
                         qudt:unit qudt:DEG_C .

                FILTER (
                    ?temperatureValue < ?minimumValue ||
                    ?temperatureValue > ?maximumValue
                )
            }
        """
    ] .

s:IMO047TemperatureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue qudt:DEG_C
    ] .
<END_SHACL>
