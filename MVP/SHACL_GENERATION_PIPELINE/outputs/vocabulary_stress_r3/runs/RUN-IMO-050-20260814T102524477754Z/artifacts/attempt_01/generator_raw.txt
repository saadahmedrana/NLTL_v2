<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:IMO_050_Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:fireSafetyComponentExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:message "Each ship must provide the fire-safety component exposure status."
    ] ;
    sh:property [
        sh:path nltl:iceAndSnowProtectionPresent ;
        sh:datatype xsd:boolean ;
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "An exposed fire-safety component must have ice and snow protection." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:fireSafetyComponentExposed ?exposed .
                FILTER ( ?exposed = true )
                FILTER NOT EXISTS {
                    $this nltl:iceAndSnowProtectionPresent ?protected .
                    FILTER ( ?protected = true )
                }
            }
        """
    ] .
<END_SHACL>
