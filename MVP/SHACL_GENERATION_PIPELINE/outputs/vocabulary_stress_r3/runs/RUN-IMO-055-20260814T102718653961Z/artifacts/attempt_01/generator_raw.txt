<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-055-ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:valveExposed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Valve exposure status is required and must be a single boolean value."
    ] ;
    sh:property [
        sh:path nltl:valveType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Valve type is required and must be a single string value."
    ] ;
    sh:property [
        sh:path nltl:iceAccretionProtectionPresent ;
        sh:datatype xsd:boolean ;
        sh:message "Ice accretion protection status must be boolean when provided."
    ] ;
    sh:property [
        sh:path nltl:continuouslyAccessible ;
        sh:datatype xsd:boolean ;
        sh:message "Continuous accessibility status must be boolean when provided."
    ] ;
    sh:sparql [
        sh:message "Exposed isolating, pressure, or vacuum valves shall have ice accretion protection and remain continuously accessible." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            SELECT $this
            WHERE {
                $this nltl:valveExposed ?exposed ;
                      nltl:valveType ?type .
                FILTER (
                    ?exposed = true &&
                    ?type IN ("isolating", "pressure", "vacuum")
                )
                FILTER (
                    NOT EXISTS {
                        $this nltl:iceAccretionProtectionPresent ?protected .
                        FILTER (?protected = true)
                    }
                    ||
                    NOT EXISTS {
                        $this nltl:continuouslyAccessible ?accessible .
                        FILTER (?accessible = true)
                    }
                )
            }
        """
    ] .
<END_SHACL>
