<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltl:IMO073_ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLifeboat ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboat ;
        sh:node nltl:IMO073_LifeboatShape
    ] .

nltl:IMO073_LifeboatShape a sh:NodeShape ;
    sh:property [
        sh:path nltl:lifeboatTypeClassification ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:lifeboatTypeValue ;
        sh:minCount 1 ;
        sh:in (
            nltl:partiallyEnclosedLifeboat
            nltl:totallyEnclosedLifeboat
        ) ;
        sh:message "Each lifeboat must have a type classified as partially enclosed or totally enclosed."
    ] .
<END_SHACL>
