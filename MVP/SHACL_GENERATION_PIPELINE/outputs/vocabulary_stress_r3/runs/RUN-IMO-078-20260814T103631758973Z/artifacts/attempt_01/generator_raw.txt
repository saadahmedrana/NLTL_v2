<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix ss: <urn:nltl:stress-shape:> .

ss:IMO_078_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasPassenger ;
        sh:node ss:IMO_078_PassengerShape
    ] ;
    sh:property [
        sh:path nltl:hasCrewMemberInventory ;
        sh:node ss:IMO_078_CrewMemberShape
    ] .

ss:IMO_078_PassengerShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalSurvivalEquipmentInstructionCompleted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each linked passenger must have completed personal survival equipment instruction."
    ] .

ss:IMO_078_CrewMemberShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:personalAndGroupSurvivalEquipmentTrainingCompleted ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "Each linked crew member must have completed personal and group survival equipment training."
    ] .
<END_SHACL>
