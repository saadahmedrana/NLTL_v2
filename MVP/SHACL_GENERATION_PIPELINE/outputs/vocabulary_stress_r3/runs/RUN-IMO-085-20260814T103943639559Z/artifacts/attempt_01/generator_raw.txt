<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix stress: <urn:nltl:stress-shape:> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

stress:IMO_085_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:message "A valid ship category is required to determine applicability."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction date is required to determine applicability."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in ( nltl:polarShipCategoryC )
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:maxExclusive "2017-01-01"^^xsd:date
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:bridgeWingsEnclosed ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:shipCategory ;
                sh:in (
                    nltl:polarShipCategoryA
                    nltl:polarShipCategoryB
                )
            ] ;
            sh:property [
                sh:path nltl:constructionDate ;
                sh:minInclusive "2017-01-01"^^xsd:date
            ] ;
            sh:property [
                sh:path nltl:bridgeWingProtectionDesignStatus ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:message "For a Category A or B ship constructed on or after 1 January 2017, bridge wings shall be enclosed or have approved protection design status." .
<END_SHACL>
