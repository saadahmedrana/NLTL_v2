@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:stress-shape:> .

gs:IMO089ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        gs:IMO089NonEscortBranch
        gs:IMO089EscortBranch
    ) .

gs:IMO089NonEscortBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:icebreakerEscortOperation ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:IMO089EscortBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:icebreakerEscortOperation ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:manuallyInitiatedFlashingRedStoppedLightPresent ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "An icebreaker-escort operation requires a manually initiated flashing red stopped light."
    ] ;
    sh:property [
        sh:path nltl:lightVisibleFromAstern ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The stopped light must be visible from astern."
    ] ;
    sh:property [
        sh:path nltl:visibilityRange ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "2"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:MI_N ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The stopped light visibility range must be at least two nautical miles."
    ] ;
    sh:property [
        sh:path nltl:colregSternLightArcComplianceStatus ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The stopped light visibility arcs must comply with the applicable stern-light specifications."
    ] .
<END_SHACL>
