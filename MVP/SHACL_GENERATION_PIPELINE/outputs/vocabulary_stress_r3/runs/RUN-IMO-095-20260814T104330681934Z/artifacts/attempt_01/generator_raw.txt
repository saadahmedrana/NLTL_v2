<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-095-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:searchAndRescueCommunicationCapability ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ship shall declare search and rescue communication capability as true."
    ] ;
    sh:property [
        sh:path nltl:rescueCoordinationCentreVoiceOrData ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:message "The ship shall declare voice or data communications with relevant rescue coordination centres."
    ] ;
    sh:property [
        sh:path nltl:supportedAircraftVoiceFrequency ;
        sh:minCount 2 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue qudt:MegaHZ ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide supported aircraft voice frequencies of 121.5 MHz and 123.1 MHz."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Supported aircraft voice frequencies shall include both 121.5 MHz and 123.1 MHz." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                FILTER NOT EXISTS {
                    $this nltl:supportedAircraftVoiceFrequency ?frequency1215 .
                    ?frequency1215 qudt:numericValue ?value1215 .
                    ?frequency1215 qudt:unit qudt:MegaHZ .
                    FILTER (
                        ?value1215 >= 121.5
                        && ?value1215 <= 121.5
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:supportedAircraftVoiceFrequency ?frequency1231 .
                    ?frequency1231 qudt:numericValue ?value1231 .
                    ?frequency1231 qudt:unit qudt:MegaHZ .
                    FILTER (
                        ?value1231 >= 123.1
                        && ?value1231 <= 123.1
                    )
                }
            }
        """
    ] .
<END_SHACL>
