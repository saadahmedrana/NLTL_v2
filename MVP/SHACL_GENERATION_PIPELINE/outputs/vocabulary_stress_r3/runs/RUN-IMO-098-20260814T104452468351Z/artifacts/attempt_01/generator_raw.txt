<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix ss: <urn:nltl:stress-shape:> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

ss:IMO098Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        ss:IMO098ApplicableBranch
        ss:IMO098NonApplicableBranch
    ) .

ss:IMO098ApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "The low-air-temperature applicability value must be true or false and must be present exactly once."
    ] ;
    sh:property [
        sh:path nltl:hasOtherSurvivalCraft ;
        sh:node ss:IMO098OtherSurvivalCraftShape
    ] .

ss:IMO098NonApplicableBranch
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipOperatesInLowAirTemperature ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false ;
        sh:message "The low-air-temperature applicability value must be true or false and must be present exactly once."
    ] .

ss:IMO098OtherSurvivalCraftShape
    a sh:NodeShape ;
    sh:class nltl:otherSurvivalCraft ;
    sh:property [
        sh:path nltl:hasLocationSignalDevice ;
        sh:node [
            sh:class nltl:locationSignalDevice
        ] ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one location-signal device."
    ] ;
    sh:property [
        sh:path nltl:hasOnSceneCommunicationDevice ;
        sh:node [
            sh:class nltl:onSceneCommunicationDevice
        ] ;
        sh:minCount 1 ;
        sh:message "Each other survival craft must carry at least one on-scene transmit/receive communication device."
    ] .
<END_SHACL>
