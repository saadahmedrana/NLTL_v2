<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix s: <urn:nltl:stress-shape:> .

s:IMO105Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "Arctic_waters"^^xsd:string
            ] ;
            sh:property [
                sh:path nltl:oilOrOilyMixtureDischargeToSea ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ]
        ]
        [
            sh:property [
                sh:path nltl:operatingArea ;
                sh:datatype xsd:string ;
                sh:minCount 1
            ] ;
            sh:sparql [
                sh:message "Operating area must not be Arctic_waters for the non-applicable branch." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:operatingArea ?area .
                        FILTER (?area = "Arctic_waters"^^xsd:string)
                    }
                """
            ]
        ]
    ) .
<END_SHACL>
