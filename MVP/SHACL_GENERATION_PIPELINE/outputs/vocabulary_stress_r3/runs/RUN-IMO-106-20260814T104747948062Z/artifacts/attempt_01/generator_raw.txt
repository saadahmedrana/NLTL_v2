<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-106-ship>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For clean or segregated ballast discharge, the Polar Code oil-discharge paragraph 1.1.1 applicability value must be false." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:dischargedMaterialClassification ?material .
                FILTER (
                    ?material IN (
                        nltl:cleanBallastMaterial,
                        nltl:segregatedBallastMaterial
                    )
                )
                FILTER NOT EXISTS {
                    $this nltl:polarCodeOilDischargeParagraph1Point1Point1Applicable false
                }
            }
        """ ;
    ] .
<END_SHACL>
