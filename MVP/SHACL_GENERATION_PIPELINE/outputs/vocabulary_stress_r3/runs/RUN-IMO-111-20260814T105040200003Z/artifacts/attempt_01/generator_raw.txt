<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-111> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:shipCategory ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:polarShipCategoryA
            nltl:polarShipCategoryB
            nltl:polarShipCategoryC
        ) ;
        sh:minCount 1 ;
        sh:message "shipCategory must be Category A, B, or C."
    ] ;
    sh:property [
        sh:path nltl:constructionDate ;
        sh:datatype xsd:date ;
        sh:message "constructionDate must be an xsd:date when provided."
    ] ;
    sh:property [
        sh:path nltl:tankType ;
        sh:datatype xsd:string ;
        sh:message "tankType must be an xsd:string when provided."
    ] ;
    sh:property [
        sh:path nltl:individualTankCapacity ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3
            ]
        ] ;
        sh:message "individualTankCapacity must be a volume quantity expressed in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:tankOuterShellSeparation ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M
            ]
        ] ;
        sh:message "tankOuterShellSeparation must be a length quantity expressed in metres."
    ] ;
    sh:sparql [
        sh:message "For a Category A or B ship constructed on or after 1 January 2017, each oil residue or oily bilge water holding tank with capacity greater than 30 m3 must have an outer-shell separation of at least 0.76 m; required applicability or measurement information is missing or non-conforming." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipCategory ?category .
                FILTER (
                    ?category = nltl:polarShipCategoryA ||
                    ?category = nltl:polarShipCategoryB
                )

                OPTIONAL {
                    $this nltl:constructionDate ?constructionDate .
                }
                FILTER (
                    !BOUND(?constructionDate) ||
                    ?constructionDate >= "2017-01-01"^^xsd:date
                )

                OPTIONAL {
                    $this nltl:tankType ?tankType .
                }
                FILTER (
                    !BOUND(?tankType) ||
                    STR(?tankType) IN (
                        "oil_residue_tank",
                        "oily_bilge_water_holding_tank"
                    )
                )

                OPTIONAL {
                    $this nltl:individualTankCapacity ?capacityNode .
                    ?capacityNode qudt:numericValue ?capacityValue ;
                                  qudt:unit unit:M3 .
                }

                OPTIONAL {
                    $this nltl:tankOuterShellSeparation ?separationNode .
                    ?separationNode qudt:numericValue ?separationValue ;
                                    qudt:unit unit:M .
                }

                FILTER (
                    !BOUND(?constructionDate) ||
                    !BOUND(?tankType) ||
                    !BOUND(?capacityValue) ||
                    (
                        ?capacityValue > "30"^^xsd:decimal &&
                        (
                            !BOUND(?separationValue) ||
                            ?separationValue < "0.76"^^xsd:decimal
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
