<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:IMO-120> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:operatingArea ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:foodWasteDischargeToSea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "For a ship operating in Arctic_waters and discharging food waste to sea, the required ice-distance records and the minimum 12 nautical mile distance must be present." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:operatingArea ?area ;
                      nltl:foodWasteDischargeToSea ?discharge .
                FILTER (
                    STR(?area) = "Arctic_waters" &&
                    ?discharge = true
                )
                FILTER NOT EXISTS {
                    $this nltl:distanceToNearestLandIceShelfOrFastIce ?nearest .
                    ?nearest qudt:numericValue ?nearestValue ;
                             qudt:unit unit:MI_N .
                    FILTER (
                        datatype(?nearestValue) = xsd:decimal &&
                        ?nearestValue >= "12"^^xsd:decimal
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:nearbyIceConcentration ?concentration .
                    ?concentration qudt:numericValue ?concentrationValue ;
                                   qudt:unit unit:UNITLESS .
                    FILTER (
                        datatype(?concentrationValue) = xsd:decimal &&
                        ?concentrationValue > "0.1"^^xsd:decimal
                    )
                }
                FILTER NOT EXISTS {
                    $this nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ?areaDistance .
                    ?areaDistance qudt:numericValue ?areaDistanceValue ;
                                  qudt:unit unit:MI_N .
                    FILTER datatype(?areaDistanceValue) = xsd:decimal
                }
            }
        """
    ] .
<END_SHACL>
