<BEGIN_SHACL>
@prefix gen: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:IMO124Shape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:operatingArea ;
                        sh:datatype xsd:string ;
                        sh:hasValue "Antarctic_area" ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:antarcticArea ;
                        sh:datatype xsd:string ;
                        sh:hasValue "Antarctic_area" ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:distanceToNearestFastIce ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minInclusive "12"^^xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:MI_N ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:nearbyIceConcentration ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minExclusive "0.1"^^xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:UNITLESS ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:distanceFromAreaWithIceConcentrationAboveOneTenth ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:MI_N ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:foodWasteDischargeOntoIce ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue false ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            )
        ]
        [
            sh:and (
                [
                    sh:property [
                        sh:path nltl:operatingArea ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:antarcticArea ;
                        sh:datatype xsd:string ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:marpolAnnexVRegulation6Point1DischargeOccurs ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
                [
                    sh:sparql [
                        sh:message "The applicable Antarctic discharge branch must be selected explicitly." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                $this nltl:operatingArea ?area .
                                $this nltl:antarcticArea ?antarctic .
                                $this nltl:marpolAnnexVRegulation6Point1DischargeOccurs ?discharge .
                                FILTER (
                                    STR(?area) != "Antarctic_area" ||
                                    STR(?antarctic) != "Antarctic_area" ||
                                    ?discharge = false
                                )
                            }
                        """
                    ]
                ]
            )
        ]
    ) .
<END_SHACL>
