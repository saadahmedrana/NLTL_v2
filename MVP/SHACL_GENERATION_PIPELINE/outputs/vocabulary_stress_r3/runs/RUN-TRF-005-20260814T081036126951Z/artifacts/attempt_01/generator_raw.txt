<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF-005-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:constructionContractDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction contract date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A construction stage date is required to determine applicability of the 2002 Ice Class Regulations."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ships with a construction stage on or after 1 September 2003 and a construction contract before 1 January 2010 shall identify the 2002 Ice Class Regulations." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:constructionStageDate ?stageDate ;
                      nltl:constructionContractDate ?contractDate .
                FILTER (
                    ?stageDate >= "2003-09-01"^^xsd:date &&
                    ?contractDate < "2010-01-01"^^xsd:date
                )
                FILTER NOT EXISTS {
                    $this nltl:applicableIceClassRegulationEdition
                          nltl:iceClassRegulationEdition2002 .
                }
            }
        """
    ] .
<END_SHACL>
