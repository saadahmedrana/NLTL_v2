<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF-006> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        <urn:nltl:stress-shape:TRF-006-applicable>
        <urn:nltl:stress-shape:TRF-006-not-applicable>
    ) ;
    sh:message "The applicable ice-class regulation must be established from the ship's construction-stage date." .

<urn:nltl:stress-shape:TRF-006-applicable> a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:minInclusive "1986-11-01"^^xsd:date ;
        sh:maxExclusive "2003-09-01"^^xsd:date ;
        sh:message "For ships constructed from 1 November 1986 and before 1 September 2003, the construction-stage date must be within the applicability period."
    ] ;
    sh:property [
        sh:path nltl:applicableIceClassRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:in ( nltl:iceClassRuleEdition1985 ) ;
        sh:minCount 1 ;
        sh:message "The 1985 Ice Class Rules must apply to the ship."
    ] ;
    sh:property [
        sh:path nltl:engineOutputRegulationEdition ;
        sh:nodeKind sh:IRI ;
        sh:in (
            nltl:iceClassRuleEdition1985
            nltl:iceClassRegulationEdition2008
        ) ;
        sh:minCount 1 ;
        sh:message "The engine-output regulation edition must be the 1985 Rules or the permitted 2008 Ice Class Regulations."
    ] .

<urn:nltl:stress-shape:TRF-006-not-applicable> a sh:NodeShape ;
    sh:or (
        <urn:nltl:stress-shape:TRF-006-before-applicability>
        <urn:nltl:stress-shape:TRF-006-after-applicability>
    ) .

<urn:nltl:stress-shape:TRF-006-before-applicability> a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:maxExclusive "1986-11-01"^^xsd:date
    ] .

<urn:nltl:stress-shape:TRF-006-after-applicability> a sh:NodeShape ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:minInclusive "2003-09-01"^^xsd:date
    ] .
<END_SHACL>
