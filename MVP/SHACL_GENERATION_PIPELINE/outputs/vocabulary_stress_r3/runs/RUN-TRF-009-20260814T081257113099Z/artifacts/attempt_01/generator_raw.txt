<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix stress: <urn:nltl:stress-shape:> .

stress:TRF009ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:message "A ship must provide its ice class to determine applicability of this requirement."
    ] ;
    sh:property [
        sh:path nltl:constructionStageDate ;
        sh:datatype xsd:date ;
        sh:minCount 1 ;
        sh:message "A ship must provide its construction stage date to determine applicability of this requirement."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For an IA Super or IA ship whose construction stage began before 1 September 2003, the ship must identify an applicable 2017 Ice Class Regulations compliance method and have an assessment date no later than 1 January of the year twenty years after delivery." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX sh: <http://www.w3.org/ns/shacl#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT $this
WHERE {
    $this nltl:iceClass ?iceClass .
    $this nltl:constructionStageDate ?constructionStageDate .

    FILTER (
        ?iceClass IN (
            nltl:iceClassIaSuper,
            nltl:iceClassIa
        )
    )

    FILTER (
        YEAR(?constructionStageDate) < 2003
        ||
        (
            YEAR(?constructionStageDate) = 2003
            &&
            (
                MONTH(?constructionStageDate) < 9
                ||
                (
                    MONTH(?constructionStageDate) = 9
                    &&
                    DAY(?constructionStageDate) < 1
                )
            )
        )
    )

    FILTER NOT EXISTS {
        $this nltl:deliveryDate ?deliveryDate .
        $this nltl:assessmentDate ?assessmentDate .
        $this nltl:engineOutputComplianceMethod ?complianceMethod .

        FILTER (
            ?complianceMethod IN (
                nltl:traficom2017Section3Point2Point2Method,
                nltl:traficom2017Section3Point2Point4Method
            )
        )

        FILTER (
            YEAR(?assessmentDate) < YEAR(?deliveryDate) + 20
            ||
            (
                YEAR(?assessmentDate) = YEAR(?deliveryDate) + 20
                &&
                MONTH(?assessmentDate) = 1
                &&
                DAY(?assessmentDate) <= 1
            )
        )
    }
}
""" ;
    ] .
<END_SHACL>
