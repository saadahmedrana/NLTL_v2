<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:QuantityValueShape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:IceWaterlineShape
    a sh:NodeShape ;
    sh:class nltl:iceWaterline ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF013NotApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:TRF013ApplicableShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:shipIceStrengthened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasLowerIceWaterline ;
        sh:node nltl:IceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:hasUpperIceWaterline ;
        sh:node nltl:IceWaterlineShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAtAftPerpendicular ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:draughtAtForePerpendicular ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingAftDraught ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:property [
        sh:path nltl:operatingForeDraught ;
        sh:node nltl:QuantityValueShape ;
        sh:minCount 1
    ] ;
    sh:sparql [
        sh:message "Ice-class and operating draughts at the fore and aft perpendiculars shall be between the lower and upper ice waterlines." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasLowerIceWaterline ?lower ;
                      nltl:hasUpperIceWaterline ?upper ;
                      nltl:draughtAtAftPerpendicular ?iceAft ;
                      nltl:draughtAtForePerpendicular ?iceFore ;
                      nltl:operatingAftDraught ?operatingAft ;
                      nltl:operatingForeDraught ?operatingFore .

                ?lower qudt:numericValue ?lowerValue ;
                       qudt:unit unit:M .
                ?upper qudt:numericValue ?upperValue ;
                       qudt:unit unit:M .
                ?iceAft qudt:numericValue ?iceAftValue ;
                        qudt:unit unit:M .
                ?iceFore qudt:numericValue ?iceForeValue ;
                         qudt:unit unit:M .
                ?operatingAft qudt:numericValue ?operatingAftValue ;
                              qudt:unit unit:M .
                ?operatingFore qudt:numericValue ?operatingForeValue ;
                               qudt:unit unit:M .

                FILTER (
                    ?lowerValue > ?upperValue ||
                    ?iceAftValue < ?lowerValue ||
                    ?iceAftValue > ?upperValue ||
                    ?iceForeValue < ?lowerValue ||
                    ?iceForeValue > ?upperValue ||
                    ?operatingAftValue < ?lowerValue ||
                    ?operatingAftValue > ?upperValue ||
                    ?operatingForeValue < ?lowerValue ||
                    ?operatingForeValue > ?upperValue
                )
            }
        """
    ] .

nltl:TRF013Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [ sh:node nltl:TRF013NotApplicableShape ]
        [ sh:node nltl:TRF013ApplicableShape ]
    ) .
<END_SHACL>
