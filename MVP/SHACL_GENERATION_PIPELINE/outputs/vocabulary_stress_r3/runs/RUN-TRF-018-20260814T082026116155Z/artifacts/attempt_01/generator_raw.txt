<BEGIN_SHACL>
@prefix gen: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF018Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "iceClass must identify one permitted ice-class value."
    ] ;
    sh:property [
        sh:path nltl:calculatedRequiredPower ;
        sh:node gen:TRF018PowerQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "calculatedRequiredPower must be provided as one power quantity in kW."
    ] ;
    sh:property [
        sh:path nltl:maximumContinuousRatingPower ;
        sh:node gen:TRF018PowerQuantity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "maximumContinuousRatingPower must be provided as one power quantity in kW."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIb ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIc ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIi ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIii ;
                sh:minCount 1
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For ice class IA, IB, or IC, maximumContinuousRatingPower must be at least calculatedRequiredPower and 1000 kW; for IA Super it must be at least calculatedRequiredPower and 2800 kW." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceClass ?ice ;
                      nltl:calculatedRequiredPower ?calculatedNode ;
                      nltl:maximumContinuousRatingPower ?ratingNode .
                ?calculatedNode qudt:numericValue ?calculated .
                ?ratingNode qudt:numericValue ?rating .
                FILTER (
                    (
                        ?ice = nltl:iceClassIaSuper &&
                        (?rating < ?calculated || ?rating < "2800"^^xsd:decimal)
                    )
                    ||
                    (
                        ?ice IN (
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ) &&
                        (?rating < ?calculated || ?rating < "1000"^^xsd:decimal)
                    )
                )
            }
        """
    ] .

gen:TRF018PowerQuantity
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The power quantity must use the kW unit."
    ] .
<END_SHACL>
