<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix s: <urn:nltl:stress-shape:> .

s:TRF020Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:maximumContinuousRatingPower ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue qudt:KiloW ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerCount ;
                sh:datatype xsd:integer ;
                sh:minInclusive 1 ;
                sh:maxInclusive 3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerDiameter ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/M> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propellerPitchControlType ;
                sh:in ( nltl:controllablePitch nltl:fixedPitch ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:propulsionSystemType ;
                sh:in (
                    nltl:conventionalPropulsionSystem
                    nltl:electricPropulsionMachinery
                    nltl:hydraulicPropulsionMachinery
                ) ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:engineOutputCoefficientKe ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtUpperIceWaterline ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:brashIceChannelResistanceAtLowerIceWaterline ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtUpperIceWaterline ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:minimumRequiredPowerAtLowerIceWaterline ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue <http://qudt.org/vocab/unit/KiloW> ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ] ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:hasUpperIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:property [
                        sh:path nltl:upperIceWaterlineLength ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:upperIceWaterlineBreadth ;
                        sh:node [
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ] ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasLowerIceWaterline ;
                sh:class nltl:iceWaterline ;
                sh:nodeKind sh:IRI ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "For the applicable construction date and ice class, the table-selected Ke and both minimum-power results must satisfy formula 3.1, and maximum continuous rating power must be at least the greater result." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?class ;
                              nltl:propellerCount ?count ;
                              nltl:propellerPitchControlType ?pitch ;
                              nltl:propulsionSystemType ?propulsion ;
                              nltl:engineOutputCoefficientKe/qudt:numericValue ?ke ;
                              nltl:propellerDiameter/qudt:numericValue ?diameter ;
                              nltl:brashIceChannelResistanceAtUpperIceWaterline/qudt:numericValue ?rUpper ;
                              nltl:brashIceChannelResistanceAtLowerIceWaterline/qudt:numericValue ?rLower ;
                              nltl:maximumContinuousRatingPower/qudt:numericValue ?mcr ;
                              nltl:minimumRequiredPowerAtUpperIceWaterline/qudt:numericValue ?pUpper ;
                              nltl:minimumRequiredPowerAtLowerIceWaterline/qudt:numericValue ?pLower ;
                              nltl:hasUpperIceWaterline ?upper ;
                              nltl:hasLowerIceWaterline ?lower .
                        ?upper nltl:upperIceWaterlineLength/qudt:numericValue ?length ;
                               nltl:upperIceWaterlineBreadth/qudt:numericValue ?breadth .
                        FILTER (
                            YEAR(?date) > 2003 ||
                            (YEAR(?date) = 2003 && MONTH(?date) > 9) ||
                            (YEAR(?date) = 2003 && MONTH(?date) = 9 && DAY(?date) >= 1)
                        )
                        FILTER (?class IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ))
                        BIND(
                            IF(
                                ?count = 1,
                                IF(
                                    ?pitch = nltl:fixedPitch,
                                    2.26,
                                    2.03
                                ),
                                IF(
                                    ?count = 2,
                                    IF(
                                        ?pitch = nltl:fixedPitch,
                                        1.60,
                                        1.44
                                    ),
                                    IF(
                                        ?pitch = nltl:fixedPitch,
                                        1.31,
                                        1.18
                                    )
                                )
                            ) AS ?expectedKe
                        )
                        BIND(?expectedKe * ((?rUpper / 1000) ^ 1.5) / ?diameter AS ?expectedUpper)
                        BIND(?expectedKe * ((?rLower / 1000) ^ 1.5) / ?diameter AS ?expectedLower)
                        FILTER (
                            ?ke < ?expectedKe || ?ke > ?expectedKe ||
                            ?pUpper < (?expectedUpper - (ABS(?expectedUpper) * 0.000001 + 0.000000001)) ||
                            ?pUpper > (?expectedUpper + (ABS(?expectedUpper) * 0.000001 + 0.000000001)) ||
                            ?pLower < (?expectedLower - (ABS(?expectedLower) * 0.000001 + 0.000000001)) ||
                            ?pLower > (?expectedLower + (ABS(?expectedLower) * 0.000001 + 0.000000001)) ||
                            ?mcr < ?pUpper ||
                            ?mcr < ?pLower
                        )
                    }
                """
            ]
        ]
        [
            sh:property [
                sh:path nltl:constructionStageDate ;
                sh:datatype xsd:date ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:sparql [
                sh:message "The ship is not in the construction-date and ice-class applicability branch for this requirement." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:constructionStageDate ?date ;
                              nltl:iceClass ?class .
                        FILTER (
                            YEAR(?date) > 2003 ||
                            (YEAR(?date) = 2003 && MONTH(?date) > 9) ||
                            (YEAR(?date) = 2003 && MONTH(?date) = 9 && DAY(?date) >= 1)
                        )
                        FILTER (?class IN (
                            nltl:iceClassIaSuper,
                            nltl:iceClassIa,
                            nltl:iceClassIb,
                            nltl:iceClassIc
                        ))
                    }
                """
            ]
        ]
    ) .
</END_SHACL>
