    sh:sparql [
        sh:message "flareAngle must equal atan(tan(bowRakeAtQuarterBreadth) / sin(waterlineAngleAtQuarterBreadth)) after degree/radian conversion; the waterline-angle sine must be non-singular." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            PREFIX math: <http://www.w3.org/2005/xpath-functions/math#>
            SELECT $this
            WHERE {
                $this nltl:bowRakeAtQuarterBreadth ?phiNode ;
                      nltl:waterlineAngleAtQuarterBreadth ?alphaNode ;
                      nltl:flareAngle ?psiNode .
                ?phiNode qudt:numericValue ?phi .
                ?alphaNode qudt:numericValue ?alpha .
                ?psiNode qudt:numericValue ?psi .
                BIND(3.141592653589793 AS ?pi)
                BIND(?phi * ?pi / 180.0 AS ?phiRad)
                BIND(?alpha * ?pi / 180.0 AS ?alphaRad)
                BIND(math:sin(?alphaRad) AS ?sinAlpha)
                BIND(math:atan(math:tan(?phiRad) / ?sinAlpha) * 180.0 / ?pi AS ?expectedPsi)
                FILTER (
                    ABS(?sinAlpha) <= 0.000000000001 ||
                    ABS(?psi - ?expectedPsi) >
                    (0.000001 * ABS(?expectedPsi) + 0.000000001)
                )
            }
        """
    ] ;

    sh:sparql [
        sh:message "clampedHullGeometryTerm must equal the value of ((shipLength * iceClassDraught) / shipBreadth squared) cubed, clamped inclusively to [5,20]." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:shipLength ?lengthNode ;
                      nltl:iceClassDraught ?draughtNode ;
                      nltl:shipBreadth ?breadthNode ;
                      nltl:clampedHullGeometryTerm ?clampedNode .
                ?lengthNode qudt:numericValue ?length .
                ?draughtNode qudt:numericValue ?draught .
                ?breadthNode qudt:numericValue ?breadth .
                ?clampedNode qudt:numericValue ?clamped .
                BIND(((?length * ?draught) / (?breadth * ?breadth)) *
                     ((?length * ?draught) / (?breadth * ?breadth)) *
                     ((?length * ?draught) / (?breadth * ?breadth)) AS ?raw)
                BIND(IF(?raw < 5.0, 5.0, IF(?raw > 20.0, 20.0, ?raw)) AS ?expected)
                FILTER(
                    ABS(?clamped - ?expected) >
                    (0.000001 * ABS(?expected) + 0.000000001)
                )
            }
        """
    ] .
