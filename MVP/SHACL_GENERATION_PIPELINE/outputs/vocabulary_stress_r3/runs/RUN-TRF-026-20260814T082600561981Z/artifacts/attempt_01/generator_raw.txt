                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>

                              nltl:iceClassDraught ?t ;
                              nltl:coefficientF1 ?f1 ;
                              nltl:coefficientF2 ?f2 ;

                              nltl:coefficientF4 ?f4 ;

                              nltl:brashIceResistanceCoefficientC2 ?c2 .

                        ?l qudt:numericValue ?lv .
                        ?t qudt:numericValue ?tv .
                        ?f1 qudt:numericValue ?f1v .
                        ?f2 qudt:numericValue ?f2v .
                        ?f3 qudt:numericValue ?f3v .
                        ?f4 qudt:numericValue ?f4v .
                        ?g1 qudt:numericValue ?g1v .
                        ?g2 qudt:numericValue ?g2v .
                        ?g3 qudt:numericValue ?g3v .
                        ?c1 qudt:numericValue ?c1v .
                        ?c2 qudt:numericValue ?c2v .

                        )

                            + ?g3v * (1 + 1.2 * ?tv / ?bv)
                              * ?bv * ?bv / math:sqrt(?lv)
                            AS ?expectedC2
                        )
                        FILTER (
                            ABS(?c1v - ?expectedC1) >
                            ABS(?expectedC1) * 0.000001 + 0.000000001
                            ||
                            ABS(?c2v - ?expectedC2) >
                            ABS(?expectedC2) * 0.000001 + 0.000000001
                        )
