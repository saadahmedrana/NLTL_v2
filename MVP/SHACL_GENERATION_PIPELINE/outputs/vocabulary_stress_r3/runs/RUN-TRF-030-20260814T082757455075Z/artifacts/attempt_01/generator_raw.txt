<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix gs: <urn:nltl:stress-shape:> .

gs:PressureQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:LengthQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:DimensionlessQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:VerticalCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:verticalLoadPosition ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A vertical direct analysis case must specify its vertical load position."
    ] ;
    sh:property [
        sh:path nltl:verticalLoadPositionType ;
        sh:in (
            nltl:upperIceWaterlineCenteredPosition
            nltl:halfLoadHeightBelowLowerIceWaterlinePosition
            nltl:intermediateVerticalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A vertical direct analysis case must specify an allowed vertical load position type."
    ] ;
    sh:property [
        sh:path nltl:upperIceWaterlineReferencePosition ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:lowerIceWaterlineReferencePosition ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

gs:HorizontalCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:horizontalLoadPosition ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A horizontal direct analysis case must specify its horizontal load position."
    ] ;
    sh:property [
        sh:path nltl:horizontalLoadPositionType ;
        sh:in (
            nltl:midSpanOrSpacingCenteredPosition
            nltl:otherHorizontalPosition
        ) ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A horizontal direct analysis case must specify an allowed horizontal load position type."
    ] .

gs:DirectAnalysisCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:capacityMinimizingLoadPositionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must confirm the capacity-minimizing load position."
    ] ;
    sh:property [
        sh:path nltl:combinedBendingAndShearEvaluated ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Each direct analysis case must evaluate combined bending and shear."
    ] ;
    sh:or (
        gs:VerticalCaseShape
        gs:HorizontalCaseShape
    ) .

gs:LoadLengthDeterminedCaseShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadPatchLength ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A case must specify load patch length when load length is not determined from the arrangement."
    ] ;
    sh:property [
        sh:path nltl:iceLoadAreaFactorCa ;
        sh:node gs:DimensionlessQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A case must specify the corresponding ice load area factor ca when load length is not determined from the arrangement."
    ] .

gs:DirectAnalysisCaseWithoutArrangementShape
    a sh:NodeShape ;
    sh:and (
        gs:DirectAnalysisCaseShape
        gs:LoadLengthDeterminedCaseShape
    ) .

gs:ArrangementDeterminationBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:node gs:DirectAnalysisCaseShape ;
        sh:minCount 2
    ] .

gs:NoArrangementDeterminationBranchShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:loadLengthDeterminedFromArrangement ;
        sh:datatype xsd:boolean ;
        sh:hasValue false ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:node gs:DirectAnalysisCaseWithoutArrangementShape ;
        sh:minCount 2
    ] .

gs:TRF030ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:icePressure ;
        sh:node gs:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:appliedIcePressure ;
        sh:node gs:PressureQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:span ;
        sh:node gs:LengthQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:node gs:DirectAnalysisCaseShape ;
        sh:minCount 2
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape gs:VerticalCaseShape ;
        sh:qualifiedMinCount 2
    ] ;
    sh:property [
        sh:path nltl:hasDirectAnalysisCase ;
        sh:qualifiedValueShape gs:HorizontalCaseShape ;
        sh:qualifiedMinCount 2
    ] ;
    sh:or (
        gs:ArrangementDeterminationBranchShape
        gs:NoArrangementDeterminationBranchShape
    ) ;
    sh:sparql [
        sh:message "appliedIcePressure must equal 1.8 times icePressure within a relative tolerance of 1e-6 and an absolute tolerance of 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:icePressure ?iceQuantity ;
                      nltl:appliedIcePressure ?appliedQuantity .
                ?iceQuantity qudt:numericValue ?iceValue .
                ?appliedQuantity qudt:numericValue ?appliedValue .
                BIND((1.8 * xsd:decimal(?iceValue)) AS ?expected)
                BIND(IF(ABS(?expected) * 1e-6 > 1e-9,
                        ABS(?expected) * 1e-6,
                        1e-9) AS ?tolerance)
                FILTER(
                    ?appliedValue < (?expected - ?tolerance) ||
                    ?appliedValue > (?expected + ?tolerance)
                )
            }
        """
    ] .
<END_SHACL>
