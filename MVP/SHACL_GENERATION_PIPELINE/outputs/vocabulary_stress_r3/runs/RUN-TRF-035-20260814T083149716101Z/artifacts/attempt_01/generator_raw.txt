<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF035ShipShape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:ruleLength ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The ship shall provide its rule length as a metre-valued quantity."
    ] ;
    sh:property [
        sh:path nltl:classificationSocietyRuleLength ;
        sh:minCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The classification society rule length shall be provided as a metre-valued quantity."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The ship's rule length shall equal the rule length used by the classification society." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT ?this
            WHERE {
                ?this nltl:ruleLength ?ruleQuantity .
                ?this nltl:classificationSocietyRuleLength ?societyQuantity .
                ?ruleQuantity qudt:numericValue ?ruleValue .
                ?societyQuantity qudt:numericValue ?societyValue .
                FILTER (
                    datatype(?ruleValue) = xsd:decimal &&
                    datatype(?societyValue) = xsd:decimal &&
                    ?ruleValue != ?societyValue
                )
            }
        """
    ] .
<END_SHACL>
