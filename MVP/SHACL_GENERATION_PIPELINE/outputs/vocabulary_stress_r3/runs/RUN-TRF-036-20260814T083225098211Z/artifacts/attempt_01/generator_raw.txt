<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF036IceClassDesignParameterSetShape>
    a sh:NodeShape ;
    sh:targetClass nltl:iceClassDesignParameterSet ;
    sh:property [
        sh:path nltl:levelIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path [
            sh:inversePath nltl:hasIceClassDesignParameterSet
        ] ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:ship ;
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Level ice thickness and design ice load height must satisfy the Table 4-1 values for the linked ship ice class." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                ?ship nltl:hasIceClassDesignParameterSet $this ;
                      nltl:iceClass ?class .
                $this nltl:levelIceThickness ?levelQuantity ;
                      nltl:designIceLoadHeight ?heightQuantity .
                ?levelQuantity qudt:numericValue ?level .
                ?heightQuantity qudt:numericValue ?height .
                FILTER (
                    !(
                        (
                            ?class = nltl:iceClassIaSuper &&
                            ?level <= 1.0 &&
                            ?height >= 0.35 &&
                            ?height <= 0.35
                        ) ||
                        (
                            ?class = nltl:iceClassIa &&
                            ?level <= 0.8 &&
                            ?height >= 0.30 &&
                            ?height <= 0.30
                        ) ||
                        (
                            ?class = nltl:iceClassIb &&
                            ?level <= 0.6 &&
                            ?height >= 0.25 &&
                            ?height <= 0.25
                        ) ||
                        (
                            ?class = nltl:iceClassIc &&
                            ?level <= 0.4 &&
                            ?height >= 0.22 &&
                            ?height <= 0.22
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
