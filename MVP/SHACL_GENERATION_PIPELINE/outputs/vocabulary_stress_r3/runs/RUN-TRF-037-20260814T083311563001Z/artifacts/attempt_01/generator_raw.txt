        sh:message "The capped ship-size engine output factor and design ice pressure must match their specified formulas within numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:shipSizeEngineOutputCoefficientA ?aNode ;
                      nltl:shipSizeEngineOutputCoefficientB ?bNode ;
                      nltl:shipSizeEngineOutputCoefficientK ?kNode ;
                      nltl:shipSizeEngineOutputFactorCd ?cdNode ;
                      nltl:iceClassFactorCp ?cpNode ;
                      nltl:nominalIcePressureP0 ?p0Node ;
                      nltl:icePressure ?pNode ;
                      nltl:hasDirectAnalysisCase ?case .

                ?aNode qudt:numericValue ?a .
                ?bNode qudt:numericValue ?b .
                ?kNode qudt:numericValue ?k .
                ?cdNode qudt:numericValue ?cd .
                ?cpNode qudt:numericValue ?cp .
                ?p0Node qudt:numericValue ?p0 .
                ?pNode qudt:numericValue ?p .
                ?case nltl:iceLoadAreaFactorCa ?caNode .
                ?caNode qudt:numericValue ?ca .

                BIND (((?a * ?k) + ?b) / 1000 AS ?rawCd)
                BIND (IF(?rawCd > 1, 1, ?rawCd) AS ?expectedCd)
                BIND (?expectedCd * ?cp * ?ca * ?p0 AS ?expectedPressure)

                BIND (
                    ((IF(?expectedCd < 0, -?expectedCd, ?expectedCd) * 0.000001)
                     + 0.000000001)
                    AS ?cdTolerance
                )
                BIND (
                    ((IF(?expectedPressure < 0, -?expectedPressure, ?expectedPressure) * 0.000001)
                     + 0.000000001)
                    AS ?pressureTolerance
                )

                FILTER (
                    (?cd - ?expectedCd > ?cdTolerance) ||
                    (?expectedCd - ?cd > ?cdTolerance) ||
                    (?p - ?expectedPressure > ?pressureTolerance) ||
                    (?expectedPressure - ?p > ?pressureTolerance)
                )
            }
        """
    ] .
