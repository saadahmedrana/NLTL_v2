<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF_042_PlatingShape>
    a sh:NodeShape ;
    sh:targetClass nltl:plating ;
    sh:property [
        sh:path nltl:designIceLoadHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Design ice load height must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:spacing ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Spacing must be a single metre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:longitudinalPlatingFactorF2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Longitudinal plating factor f2 must be a single dimensionless quantity." ;
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Yield strength must be a single MPa quantity." ;
    ] ;
    sh:property [
        sh:path nltl:hullStructuralSteelGrade ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:class nltl:steelGradeValue ;
        sh:message "Hull structural steel grade must identify a steel grade value." ;
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Corrosion and abrasion addition must be a single millimetre quantity." ;
    ] ;
    sh:property [
        sh:path nltl:specialSurfaceCoatingMaintained ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:materialApprovalStatus ;
        sh:class nltl:evidenceState ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        sh:message "f2 must match the applicable piecewise formula using h divided by s within the stated numerical tolerance." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceLoadHeight ?hNode ;
                      nltl:spacing ?sNode ;
                      nltl:longitudinalPlatingFactorF2 ?fNode .
                ?hNode qudt:numericValue ?h .
                ?sNode qudt:numericValue ?s .
                ?fNode qudt:numericValue ?reported .
                BIND (?h / ?s AS ?ratio)
                BIND (
                    IF (?ratio <= 1.0,
                        0.4 / ?ratio + 0.6,
                        1.4 - 0.4 * ?ratio
                    ) AS ?expected
                )
                BIND (
                    IF (1e-6 * ABS(?expected) > 1e-12,
                        1e-6 * ABS(?expected),
                        1e-12
                    ) AS ?tolerance
                )
                FILTER (?ratio <= 1.8)
                FILTER (ABS(?reported - ?expected) > ?tolerance)
            }
        """
    ] ;
    sh:sparql [
        sh:message "Yield strength must be 235 MPa for normal-strength steel, at least 315 MPa for high-strength steel, or an explicitly supplied accepted value for another steel grade." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hullStructuralSteelGrade ?grade ;
                      nltl:yieldStrength ?yieldNode .
                ?yieldNode qudt:numericValue ?yield .
                FILTER (
                    (?grade = nltl:normalStrengthHullStructuralSteel &&
                     (?yield < 235.0 || ?yield > 235.0))
                    ||
                    (?grade = nltl:highStrengthHullStructuralSteel &&
                     ?yield < 315.0)
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "A corrosion and abrasion addition below 2 mm requires a maintained special surface coating and approved material approval status; values above 2 mm are not permitted." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:corrosionAbrasionAddition ?tcNode .
                ?tcNode qudt:numericValue ?tc .
                FILTER (
                    ?tc > 2.0
                    ||
                    (
                        ?tc < 2.0
                        &&
                        (
                            NOT EXISTS {
                                $this nltl:specialSurfaceCoatingMaintained true
                            }
                            ||
                            NOT EXISTS {
                                $this nltl:materialApprovalStatus nltl:evidenceStateApproved
                            }
                        )
                    )
                )
            }
        """
    ] .
<END_SHACL>
