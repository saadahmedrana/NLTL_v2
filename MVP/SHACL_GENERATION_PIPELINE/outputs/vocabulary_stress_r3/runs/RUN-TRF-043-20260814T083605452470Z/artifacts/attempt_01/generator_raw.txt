<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix ss: <urn:nltl:stress-shape:> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

ss:TRF043FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:datatype nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIaSuper
            nltl:iceClassIa
            nltl:iceClassIb
            nltl:iceClassIc
        ) ;
        sh:message "The frame must specify one of the Table 4-6 ice classes."
    ] ;
    sh:property [
        sh:path nltl:hullRegion ;
        sh:datatype nltl:hullRegionValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:bowRegion
            nltl:midbodyRegion
            nltl:sternRegion
        ) ;
        sh:message "The frame must specify a Table 4-6 hull region."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningAboveUpperIceWaterline ;
        sh:node ss:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must provide its vertical framing extent above the upper ice waterline."
    ] ;
    sh:property [
        sh:path nltl:framingIceStrengtheningBelowLowerIceWaterline ;
        sh:node ss:MetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must provide its vertical framing extent below the lower ice waterline."
    ] ;
    sh:property [
        sh:path nltl:upperBowIceBeltRequired ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must specify whether an upper bow ice belt is required."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:reachesUpperBowIceBeltTop ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true ;
                sh:message "When an upper bow ice belt is required, the framing must reach its top."
            ]
        ]
        [
            sh:property [
                sh:path nltl:upperBowIceBeltRequired ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:property [
        sh:path nltl:extensionBeyondAdjacentDeckOrTankBoundary ;
        sh:node ss:MillimetreQuantityShape ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame must provide its extension beyond the adjacent boundary."
    ] ;
    sh:property [
        sh:path nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:reachesTankTopOrBelowFloorTop ;
        sh:datatype xsd:boolean ;
        sh:maxCount 1
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing extent above the upper ice waterline is below the Table 4-6 minimum." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:framingIceStrengtheningAboveUpperIceWaterline ?quantity .
                ?quantity qudt:numericValue ?value .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper && ?value < 1.2) ||
                    (?iceClass = nltl:iceClassIa && ?value < 1.0) ||
                    (?iceClass = nltl:iceClassIb && ?value < 1.0) ||
                    (?iceClass = nltl:iceClassIc && ?value < 1.0)
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The framing extent below the lower ice waterline is below the Table 4-6 minimum for the specified class and region." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass ?iceClass ;
                      nltl:hullRegion ?region ;
                      nltl:framingIceStrengtheningBelowLowerIceWaterline ?quantity .
                ?quantity qudt:numericValue ?value .
                FILTER (
                    (?iceClass = nltl:iceClassIaSuper && ?region = nltl:midbodyRegion && ?value < 2.0) ||
                    (?iceClass = nltl:iceClassIaSuper && ?region = nltl:sternRegion && ?value < 1.6) ||
                    (?iceClass = nltl:iceClassIa && ?region = nltl:bowRegion && ?value < 1.6) ||
                    (?iceClass = nltl:iceClassIa && ?region = nltl:midbodyRegion && ?value < 1.3) ||
                    (?iceClass = nltl:iceClassIa && ?region = nltl:sternRegion && ?value < 1.0) ||
                    (?iceClass = nltl:iceClassIb && ?region = nltl:bowRegion && ?value < 1.6) ||
                    (?iceClass = nltl:iceClassIb && ?region = nltl:midbodyRegion && ?value < 1.3) ||
                    (?iceClass = nltl:iceClassIb && ?region = nltl:sternRegion && ?value < 1.0) ||
                    (?iceClass = nltl:iceClassIc && ?region = nltl:bowRegion && ?value < 1.6) ||
                    (?iceClass = nltl:iceClassIc && ?region = nltl:midbodyRegion && ?value < 1.3) ||
                    (?iceClass = nltl:iceClassIc && ?region = nltl:sternRegion && ?value < 1.0)
                )
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "For IA Super bow framing, the ice strengthening must reach the tank top or below the top of floors." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:iceClass nltl:iceClassIaSuper ;
                      nltl:hullRegion nltl:bowRegion .
                FILTER NOT EXISTS {
                    $this nltl:reachesTankTopOrBelowFloorTop true .
                }
            }
        """
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Termination at the adjacent boundary is permitted only when the extension is no more than 250 mm." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:extensionBeyondAdjacentDeckOrTankBoundary ?quantity ;
                      nltl:iceStrengtheningTerminationAtAdjacentBoundaryPermitted true .
                ?quantity qudt:numericValue ?value .
                FILTER (?value > 250.0)
            }
        """
    ] .

ss:MetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/M> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

ss:MillimetreQuantityShape
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .
<END_SHACL>
