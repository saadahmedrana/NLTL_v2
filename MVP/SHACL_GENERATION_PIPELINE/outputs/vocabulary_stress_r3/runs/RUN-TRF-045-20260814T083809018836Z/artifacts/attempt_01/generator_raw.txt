<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF045FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:ordinaryFrameScantlingsUsed ;
        sh:datatype xsd:boolean ;
        sh:message "ordinaryFrameScantlingsUsed must be a boolean when provided."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Ordinary frame scantlings may be used only when less than 15% of the frame span is situated within the ice-strengthening zone; both spans must be decimal quantities in metres." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:ordinaryFrameScantlingsUsed true .
                FILTER NOT EXISTS {
                    $this nltl:span ?spanQuantity .
                    ?spanQuantity qudt:numericValue ?spanValue ;
                                   qudt:unit unit:M .
                    $this nltl:frameSpanWithinIceStrengtheningZone ?iceQuantity .
                    ?iceQuantity qudt:numericValue ?iceValue ;
                                  qudt:unit unit:M .
                    FILTER (
                        datatype(?spanValue) = xsd:decimal &&
                        datatype(?iceValue) = xsd:decimal &&
                        ?iceValue < (0.15 * ?spanValue)
                    )
                }
            }
        """
    ] .
</END_SHACL>
