<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

<urn:nltl:stress-shape:TRF-046-FrameShape>
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A frame must identify its strengthened part."
    ] ;
    sh:property [
        sh:path nltl:hasUpperEnd ;
        sh:minCount 1 ;
        sh:node <urn:nltl:stress-shape:TRF-046-FrameEndShape> ;
        sh:message "The frame must identify an upper end whose strengthened part is attached to an eligible supporting structure."
    ] .

<urn:nltl:stress-shape:TRF-046-FrameEndShape>
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:node [
            sh:class nltl:supportingStructure ;
            sh:or (
                [ sh:class nltl:deckStructure ]
                [ sh:class nltl:tankBoundaryPlating ]
                [ sh:class nltl:iceStringer ]
            )
        ] ;
        sh:message "The upper end must be attached to a deck, tank boundary plating, or ice stringer."
    ] .
<END_SHACL>
