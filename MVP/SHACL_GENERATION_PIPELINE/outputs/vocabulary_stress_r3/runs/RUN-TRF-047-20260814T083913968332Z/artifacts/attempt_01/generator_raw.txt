<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

nltl:TRF047FrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:hasStrengthenedPart ;
        sh:class nltl:frameStrengthenedPart ;
        sh:minCount 1 ;
        sh:message "A frame shall identify its strengthened part."
    ] .

nltl:TRF047MainFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:mainFrame ;
    sh:property [
        sh:path nltl:mainFrameBelowIceBeltStrengthened ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true ;
        sh:message "A main frame below the ice belt shall be ice strengthened."
    ] ;
    sh:property [
        sh:path ( nltl:hasLowerEnd nltl:hasAttachedSupportingStructure ) ;
        sh:minCount 1 ;
        sh:or (
            [ sh:class nltl:deckStructure ]
            [ sh:class nltl:tankBoundaryPlating ]
            [ sh:class nltl:tankTop ]
            [ sh:class nltl:iceStringer ]
        ) ;
        sh:message "The lower end of a strengthened main frame shall be attached to a qualifying deck, tank boundary plating, tank top, or ice stringer."
    ] .

nltl:TRF047IntermediateIceFrameShape
    a sh:NodeShape ;
    sh:targetClass nltl:intermediateIceFrame ;
    sh:or (
        [
            sh:property [
                sh:path ( nltl:hasLowerEnd nltl:hasAttachedSupportingStructure ) ;
                sh:minCount 1 ;
                sh:or (
                    [ sh:class nltl:deckStructure ]
                    [ sh:class nltl:tankBoundaryPlating ]
                    [ sh:class nltl:tankTop ]
                    [ sh:class nltl:iceStringer ]
                )
            ]
        ]
        [
            sh:property [
                sh:path nltl:supportingStructureAtOrBelowIceBeltLowerLimit ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:sameScantlingsAsMainFrame ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path ( nltl:hasLowerEnd nltl:hasTerminationBelowSupportingStructure ) ;
                sh:minCount 1 ;
                sh:or (
                    [ sh:class nltl:deckStructure ]
                    [ sh:class nltl:tankBoundaryPlating ]
                    [ sh:class nltl:tankTop ]
                    [ sh:class nltl:iceStringer ]
                )
            ] ;
            sh:property [
                sh:path ( nltl:hasLowerEnd nltl:hasHorizontalConnectionMember ) ;
                sh:minCount 1 ;
                sh:class nltl:horizontalConnectionMember
            ] ;
            sh:property [
                sh:path ( nltl:hasLowerEnd nltl:hasHorizontalConnectionMember nltl:connectsToAdjacentMainFrame ) ;
                sh:minCount 1 ;
                sh:class nltl:mainFrame
            ] ;
            sh:sparql [
                sh:message "The permitted horizontal connection shall use the same lower-end relationship for the qualifying supporting structure and the member connected to an adjacent main frame." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
                    SELECT $this
                    WHERE {
                        FILTER NOT EXISTS {
                            $this nltl:hasLowerEnd ?end .
                            ?end nltl:hasTerminationBelowSupportingStructure ?supporting .
                            ?end nltl:hasHorizontalConnectionMember ?member .
                            ?member nltl:connectsToAdjacentMainFrame ?adjacent .
                            FILTER EXISTS {
                                ?supporting rdf:type ?supportingType .
                                FILTER (
                                    ?supportingType IN (
                                        nltl:deckStructure,
                                        nltl:tankBoundaryPlating,
                                        nltl:tankTop,
                                        nltl:iceStringer
                                    )
                                )
                            }
                            ?adjacent rdf:type nltl:mainFrame .
                        }
                    }
                """
            ]
        )
    ) .
<END_SHACL>
