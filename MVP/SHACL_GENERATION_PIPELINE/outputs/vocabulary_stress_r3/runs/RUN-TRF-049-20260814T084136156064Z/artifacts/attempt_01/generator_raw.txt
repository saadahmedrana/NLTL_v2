<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:QuantityValueShape_TRF049
    a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/MilliM> ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

nltl:BracketShape_TRF049
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:bracketThickness ;
        sh:node nltl:QuantityValueShape_TRF049 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "A connection bracket must have one thickness quantity in millimetres."
    ] ;
    sh:property [
        sh:path nltl:bracketEdgeStiffened ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The bracket edge must be stiffened against buckling."
    ] ;
    sh:property [
        sh:path nltl:bucklingStiffening ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Buckling stiffening must be provided for the connection bracket."
    ] .

nltl:AttachmentShape_TRF049
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The frame attachment must be confirmed effective."
    ] ;
    sh:property [
        sh:path nltl:hasAttachedSupportingStructure ;
        sh:minCount 1 ;
        sh:message "The frame attachment must identify its supporting structure."
    ] .

nltl:BracketedAttachmentShape_TRF049
    a sh:NodeShape ;
    sh:node nltl:AttachmentShape_TRF049 ;
    sh:property [
        sh:path nltl:hasConnectionBracket ;
        sh:node nltl:BracketShape_TRF049 ;
        sh:minCount 1 ;
        sh:message "The required frame attachment must be made by a connection bracket."
    ] .

nltl:AlternativeAttachmentShape_TRF049
    a sh:NodeShape ;
    sh:node nltl:AttachmentShape_TRF049 ;
    sh:or (
        [
            sh:path nltl:hasConnectionBracket ;
            sh:node nltl:BracketShape_TRF049 ;
            sh:minCount 1
        ]
        [
            sh:path nltl:similarAttachmentConstructionPresent ;
            sh:datatype xsd:boolean ;
            sh:hasValue true ;
            sh:minCount 1 ;
            sh:maxCount 1
        ]
    ) ;
    sh:message "A terminated transversal frame must have a bracket or similar construction."
.

nltl:LongitudinalFrameCaseShape_TRF049
    a sh:NodeShape ;
    sh:class nltl:longitudinalFrame ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node nltl:BracketedAttachmentShape_TRF049 ;
        sh:minCount 1 ;
        sh:message "A longitudinal frame must be attached by brackets."
    ] .

nltl:TransverseFrameCaseShape_TRF049
    a sh:NodeShape ;
    sh:class nltl:transverseFrame ;
    sh:property [
        sh:path nltl:terminatesAtDeckOrIceStringer ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether the transversal frame terminates at a deck or ice stringer must be recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue true
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:node nltl:AlternativeAttachmentShape_TRF049 ;
                sh:minCount 1
            ]
        ]
        [
            sh:property [
                sh:path nltl:terminatesAtDeckOrIceStringer ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasFrameAttachment ;
                sh:node nltl:AttachmentShape_TRF049 ;
                sh:minCount 1
            ]
        ]
    ) .

nltl:GenericFrameCaseShape_TRF049
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasFrameAttachment ;
        sh:node nltl:AttachmentShape_TRF049 ;
        sh:minCount 1
    ] .

nltl:InIceFrameShape_TRF049
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:effectiveAttachmentConfirmed ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Every frame in the ice-strengthened area must have effective attachment confirmed."
    ] ;
    sh:or (
        nltl:LongitudinalFrameCaseShape_TRF049
        nltl:TransverseFrameCaseShape_TRF049
        nltl:GenericFrameCaseShape_TRF049
    ) .

nltl:FrameShape_TRF049
    a sh:NodeShape ;
    sh:targetClass nltl:frame ;
    sh:property [
        sh:path nltl:withinIceStrengthenedArea ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether the frame is within the ice-strengthened area must be recorded."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue true
            ] ;
            sh:node nltl:InIceFrameShape_TRF049
        ]
        [
            sh:property [
                sh:path nltl:withinIceStrengthenedArea ;
                sh:hasValue false
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "Each attached supporting structure must identify the same frame attachment through hasSupportingStructureAttachment." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasAttachedSupportingStructure ?supportingStructure .
                FILTER NOT EXISTS {
                    ?supportingStructure nltl:hasSupportingStructureAttachment ?attachment .
                }
            }
        """
    ] ;
    sh:sparql [
        sh:message "The connection bracket thickness must be at least the frame web thickness." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:frameWebThickness ?frameThickness ;
                      nltl:hasFrameAttachment ?attachment .
                ?attachment nltl:hasConnectionBracket ?bracket .
                ?bracket nltl:bracketThickness ?bracketThickness .
                ?frameThickness qudt:numericValue ?frameValue ;
                                qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                ?bracketThickness qudt:numericValue ?bracketValue ;
                                   qudt:unit <http://qudt.org/vocab/unit/MilliM> .
                FILTER (?bracketValue < ?frameValue)
            }
        """
    ] .
<END_SHACL>
