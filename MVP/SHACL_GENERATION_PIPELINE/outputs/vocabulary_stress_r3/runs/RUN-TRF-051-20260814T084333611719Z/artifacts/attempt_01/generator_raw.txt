<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF051_HullStructureShape
    a sh:NodeShape ;
    sh:targetClass nltl:hullStructure ;
    sh:property [
        sh:path nltl:frameWebThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:adjacentFrameHeight ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:yieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MegaPA>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:netShellPlatingThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:corrosionAbrasionAddition ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/MilliM>
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:inLieuOfFrame ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:inLieuOfFrame ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:inLieuOfFrame ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:frameProfileType ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:flatBarSection
                    nltl:profileSection
                )
            ]
        ]
    ) ;
    sh:sparql [
        sh:message "For a structure used in lieu of a frame, frame web or plate thickness must meet the maximum of the calculated instability thickness, half the net shell-plating thickness after corrosion or abrasion deduction, and 9 mm; a relative tolerance of 1e-6 is applied to the calculated decimal result." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:inLieuOfFrame true ;
                      nltl:frameWebThickness ?webNode ;
                      nltl:adjacentFrameHeight ?heightNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:netShellPlatingThickness ?thicknessNode ;
                      nltl:corrosionAbrasionAddition ?additionNode .
                ?webNode qudt:numericValue ?web .
                ?heightNode qudt:numericValue ?height .
                ?yieldNode qudt:numericValue ?yield .
                ?thicknessNode qudt:numericValue ?thickness .
                ?additionNode qudt:numericValue ?addition .
                BIND(805.0 AS ?c)
                FILTER (
                    IF(
                        ?height > 0.0,
                        (((?c * ?web / ?height) * (?c * ?web / ?height))
                          < (?yield * 0.999999 * 0.999999)),
                        true
                    )
                    || ?web < (((?thickness - ?addition) / 2.0) * 0.999999)
                    || ?web < 9.0
                )
            }
        """
    ] ;
    sh:sparql [
        sh:message "For a frame profile, frame web thickness must meet the maximum of the profile-dependent calculated instability thickness, half the net shell-plating thickness after corrosion or abrasion deduction, and 9 mm; a relative tolerance of 1e-6 is applied to calculated decimal results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX sh: <http://www.w3.org/ns/shacl#>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:inLieuOfFrame false ;
                      nltl:frameProfileType ?profile ;
                      nltl:frameWebThickness ?webNode ;
                      nltl:adjacentFrameHeight ?heightNode ;
                      nltl:yieldStrength ?yieldNode ;
                      nltl:netShellPlatingThickness ?thicknessNode ;
                      nltl:corrosionAbrasionAddition ?additionNode .
                ?webNode qudt:numericValue ?web .
                ?heightNode qudt:numericValue ?height .
                ?yieldNode qudt:numericValue ?yield .
                ?thicknessNode qudt:numericValue ?thickness .
                ?additionNode qudt:numericValue ?addition .
                BIND(
                    IF(?profile = nltl:flatBarSection, 282.0, 805.0)
                    AS ?c
                )
                FILTER (
                    IF(
                        ?height > 0.0,
                        (((?c * ?web / ?height) * (?c * ?web / ?height))
                          < (?yield * 0.999999 * 0.999999)),
                        true
                    )
                    || ?web < (((?thickness - ?addition) / 2.0) * 0.999999)
                    || ?web < 9.0
                )
            }
        """
    ] .
<END_SHACL>
