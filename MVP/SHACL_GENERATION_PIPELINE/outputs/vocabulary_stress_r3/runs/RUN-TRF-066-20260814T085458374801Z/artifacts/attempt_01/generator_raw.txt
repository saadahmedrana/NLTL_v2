<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF066ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                ) ;
                sh:message "For a non-applicable ice class, iceClass must identify a permitted non-IA class."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIa ;
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "For ice class IA, iceKnifePracticable must be true or the applicable alternative branch must be satisfied."
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:class nltl:iceKnife ;
                    sh:property [
                        sh:path nltl:extendsBelowLowerIceWaterline ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                        sh:message "The ice knife must extend below the lower ice waterline."
                    ]
                ] ;
                sh:message "An applicable IA ship must have a rudder protection arrangement with an ice knife extending below the lower ice waterline."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIaSuper ;
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue true ;
                sh:message "For ice class IA Super, iceKnifePracticable must be true or the applicable alternative branch must be satisfied."
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:class nltl:iceKnife ;
                    sh:property [
                        sh:path nltl:extendsBelowLowerIceWaterline ;
                        sh:datatype xsd:boolean ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:hasValue true ;
                        sh:message "The ice knife must extend below the lower ice waterline."
                    ]
                ] ;
                sh:message "An applicable IA Super ship must have a rudder protection arrangement with an ice knife extending below the lower ice waterline."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIa ;
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:class nltl:equivalentRudderProtectionMeans ;
                ] ;
                sh:message "When an ice knife is not practicable for ice class IA, an equivalent rudder protection means is required."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue nltl:iceClassIaSuper ;
            ] ;
            sh:property [
                sh:path nltl:iceKnifePracticable ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:datatype xsd:boolean ;
                sh:hasValue false ;
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:rudderProtectionArrangement ;
                    sh:class nltl:equivalentRudderProtectionMeans ;
                ] ;
                sh:message "When an ice knife is not practicable for ice class IA Super, an equivalent rudder protection means is required."
            ]
        ]
    ) .
<END_SHACL>
