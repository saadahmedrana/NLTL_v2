<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix nltlgen: <urn:nltl:stress-shape:> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

nltlgen:RudderLoadAbsorbingArrangementShape
    a sh:NodeShape ;
    sh:class nltl:rudderProtectionArrangement ;
    sh:property [
        sh:path rdf:type ;
        sh:hasValue nltl:rudderLoadAbsorbingArrangement ;
        sh:message "The rudder protection arrangement shall be a load-absorbing arrangement."
    ] .

nltlgen:TRF067ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:message "The ship's ice class is required to determine applicability of the rudder load-absorbing arrangement requirement."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIa
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:nodeKind sh:IRI ;
                sh:node nltlgen:RudderLoadAbsorbingArrangementShape ;
                sh:minCount 1 ;
                sh:message "For ice class IA, a rudder load-absorbing protection arrangement shall be installed."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:hasRudderProtectionArrangement ;
                sh:nodeKind sh:IRI ;
                sh:node nltlgen:RudderLoadAbsorbingArrangementShape ;
                sh:minCount 1 ;
                sh:message "For ice class IA Super, a rudder load-absorbing protection arrangement shall be installed."
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ]
        ]
    ) .
<END_SHACL>
