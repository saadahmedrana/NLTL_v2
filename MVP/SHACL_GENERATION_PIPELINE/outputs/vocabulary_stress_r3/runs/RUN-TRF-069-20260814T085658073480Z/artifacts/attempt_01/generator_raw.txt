<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

nltl:TRF069ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:pitchType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIaSuper
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                )
            ] ;
            sh:property [
                sh:path nltl:propellerType ;
                sh:in ( "open" "ducted" )
            ] ;
            sh:property [
                sh:path nltl:pitchType ;
                sh:in ( "controllable" "fixed" )
            ] ;
            sh:property [
                sh:path nltl:maximumIceTorque ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN-M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:hasDesignCondition ;
                sh:minCount 1 ;
                sh:node [
                    sh:class nltl:designCondition ;
                    sh:property [
                        sh:path nltl:normalOperationalConditions ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:normalServiceLifePropellerIceLoadCondition ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:fixedPitchPropellerReversalLoadIncluded ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:stoppedPropellerDraggingOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:radialIceEntryOutsideLoadModel ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1
                    ]
                ]
            ]
        ]
        [
            sh:sparql [
                sh:message "The propulsion-machinery rule is non-applicable only when at least one applicability input is outside its permitted alternatives." ;
                sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
    $this ?p ?v .
    FILTER (
        ?p IN (
            nltl:iceClass,
            nltl:propellerType,
            nltl:pitchType
        )
        &&
        (
            (?p = nltl:iceClass &&
             ?v NOT IN (
                 nltl:iceClassIaSuper,
                 nltl:iceClassIa,
                 nltl:iceClassIb,
                 nltl:iceClassIc
             ))
            ||
            (?p = nltl:propellerType &&
             ?v NOT IN ("open", "ducted"))
            ||
            (?p = nltl:pitchType &&
             ?v NOT IN ("controllable", "fixed"))
        )
    )
}
"""
            ]
        ]
    ) .
<END_SHACL>
