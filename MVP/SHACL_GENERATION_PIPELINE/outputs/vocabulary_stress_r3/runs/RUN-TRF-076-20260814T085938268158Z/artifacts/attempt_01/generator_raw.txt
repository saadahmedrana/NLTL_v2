<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix nltlsh: <urn:nltl:stress-shape:> .

nltlsh:PropellerBladeLoadCaseNumber1Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 1 ;
        sh:maxInclusive 1 ;
        sh:message "The propeller blade load case number must be 1."
    ] .

nltlsh:PropellerBladeLoadCaseNumber2Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 2 ;
        sh:maxInclusive 2 ;
        sh:message "The propeller blade load case number must be 2."
    ] .

nltlsh:PropellerBladeLoadCaseNumber3Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 3 ;
        sh:maxInclusive 3 ;
        sh:message "The propeller blade load case number must be 3."
    ] .

nltlsh:PropellerBladeLoadCaseNumber4Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 4 ;
        sh:maxInclusive 4 ;
        sh:message "The propeller blade load case number must be 4."
    ] .

nltlsh:PropellerBladeLoadCaseNumber5Shape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:propellerBladeLoadCaseNumber ;
        sh:datatype xsd:integer ;
        sh:minCount 1 ;
        sh:minInclusive 5 ;
        sh:maxInclusive 5 ;
        sh:message "The propeller blade load case number must be 5."
    ] .

nltlsh:RequiredPropellerBladeLoadCasesShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node nltlsh:PropellerBladeLoadCaseNumber1Shape ;
        sh:message "Propeller blade load case 1 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node nltlsh:PropellerBladeLoadCaseNumber2Shape ;
        sh:message "Propeller blade load case 2 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node nltlsh:PropellerBladeLoadCaseNumber3Shape ;
        sh:message "Propeller blade load case 3 must be covered."
    ] ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node nltlsh:PropellerBladeLoadCaseNumber4Shape ;
        sh:message "Propeller blade load case 4 must be covered."
    ] .

nltlsh:RequiredPropellerBladeLoadCasesIncluding5Shape
    a sh:NodeShape ;
    sh:node nltlsh:RequiredPropellerBladeLoadCasesShape ;
    sh:property [
        sh:path nltl:hasPropellerBladeLoadCase ;
        sh:class nltl:propellerBladeLoadCase ;
        sh:minCount 1 ;
        sh:node nltlsh:PropellerBladeLoadCaseNumber5Shape ;
        sh:message "Load case 5 must be covered for a reversing FP propeller."
    ] .

nltlsh:PropellerBladeLoadCaseCoverageShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "CP"
            ] ;
            sh:node nltlsh:RequiredPropellerBladeLoadCasesShape
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "FP"
            ] ;
            sh:property [
                sh:path nltl:propellerReversingCapability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue true
            ] ;
            sh:node nltlsh:RequiredPropellerBladeLoadCasesIncluding5Shape
        ]
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:datatype xsd:string ;
                sh:minCount 1 ;
                sh:hasValue "FP"
            ] ;
            sh:property [
                sh:path nltl:propellerReversingCapability ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:hasValue false
            ] ;
            sh:node nltlsh:RequiredPropellerBladeLoadCasesShape
        ]
    ) ;
    sh:message "The ship must identify a CP or FP propeller and cover load cases 1-4; a reversing FP propeller must also cover load case 5."
.
<END_SHACL>
