    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;

        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue <http://qudt.org/vocab/unit/UNITLESS> ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

<urn:nltl:stress-shape:TableLookupCaseShape>
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableReferenceValue ;
        sh:in (
            nltl:traficomTable6Dash6
            nltl:traficomTable6Dash7
        ) ;

        sh:maxCount 2
    ] ;
    sh:property [

        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:QuantityValueShape>
    ] ;

        sh:maxCount 1
    ] .

<urn:nltl:stress-shape:TRF078ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:classLoadCycleCount ;
        sh:datatype xsd:integer
    ] ;
    sh:property [
        sh:path nltl:propellerLocation ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:propellerLocationValue ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:nodeKind sh:IRI ;
        sh:class nltl:tableLookupCase ;
        sh:minCount 1 ;
        sh:node <urn:nltl:stress-shape:TableLookupCaseShape>
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK1 ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK3 ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:QuantityValueShape>
    ] ;
    sh:property [
        sh:path nltl:propellerBladeLoadSpectrumCountFactor ;
        sh:nodeKind sh:IRI ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node <urn:nltl:stress-shape:QuantityValueShape>
    ] ;
    sh:sparql [
        sh:message "The propeller blade load spectrum count factor must equal k1 times k2 times k3 times the selected ice-class cycle count times the selected propeller-location factor; comparison uses relative tolerance 1e-6 plus absolute tolerance 1e-9." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:iceLoadCycleCoefficientK1 ?k1Node ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Node ;
                      nltl:iceLoadCycleCoefficientK3 ?k3Node ;
                      nltl:propellerBladeLoadSpectrumCountFactor ?resultNode ;
                      nltl:hasTableLookupCase ?case .

                ?k1Node qudt:numericValue ?k1 .
                ?k2Node qudt:numericValue ?k2 .
                ?k3Node qudt:numericValue ?k3 .
                ?resultNode qudt:numericValue ?observed .
                ?case nltl:selectedIceClassCycleCount ?nClassNode ;
                      nltl:selectedPropellerLocationFactor ?locationFactorNode .
                ?nClassNode qudt:numericValue ?nClass .
                ?locationFactorNode qudt:numericValue ?locationFactor .

                BIND((?k1 * ?k2 * ?k3 * ?nClass * ?locationFactor) AS ?expected)
                FILTER(
                    ABS(?observed - ?expected) >
                    ((0.000001 * ABS(?expected)) + 0.000000001)
                )
            }
        """
    ] .
</urn:nltl:stress-shape:TRF078ShipShape>
