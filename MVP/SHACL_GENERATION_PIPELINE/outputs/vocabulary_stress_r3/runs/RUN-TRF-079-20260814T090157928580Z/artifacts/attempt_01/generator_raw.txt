<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix stress: <urn:nltl:stress-shape:> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

stress:TRF_079_ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:designIceThickness ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerShaftCentrelineDepth ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:propellerImmersionFunction ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:iceLoadCycleCoefficientK2 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "Propeller immersion function and ice load cycle coefficient k2 must satisfy the specified formulae, with a relative tolerance of 1e-6 and an absolute fallback of 1e-12 for derived decimal results." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:designIceThickness ?iceQuantity ;
                      nltl:propellerDiameter ?diameterQuantity ;
                      nltl:propellerShaftCentrelineDepth ?depthQuantity ;
                      nltl:propellerImmersionFunction ?functionQuantity ;
                      nltl:iceLoadCycleCoefficientK2 ?k2Quantity .

                ?iceQuantity qudt:numericValue ?ice .
                ?diameterQuantity qudt:numericValue ?diameter .
                ?depthQuantity qudt:numericValue ?depth .
                ?functionQuantity qudt:numericValue ?observedF .
                ?k2Quantity qudt:numericValue ?observedK2 .

                BIND(
                    IF(
                        ?diameter = 0,
                        0,
                        (?depth - ?ice) / (?diameter / 2) - 1
                    )
                    AS ?expectedF
                )

                BIND(
                    IF(
                        ?expectedF < 0,
                        0.8 - ?expectedF,
                        IF(
                            ?expectedF <= 1,
                            0.8 - 0.4 * ?expectedF,
                            IF(
                                ?expectedF <= 2.5,
                                0.6 - 0.2 * ?expectedF,
                                0.1
                            )
                        )
                    )
                    AS ?expectedK2
                )

                FILTER(
                    ?diameter = 0
                    ||
                    ABS(?observedF - ?expectedF) >
                    (1e-6 * IF(ABS(?expectedF) > 0, ABS(?expectedF), 1e-12))
                    ||
                    ABS(?observedK2 - ?expectedK2) >
                    (1e-6 * IF(ABS(?expectedK2) > 0, ABS(?expectedK2), 1e-12))
                )
            }
        """
    ] .
<END_SHACL>
