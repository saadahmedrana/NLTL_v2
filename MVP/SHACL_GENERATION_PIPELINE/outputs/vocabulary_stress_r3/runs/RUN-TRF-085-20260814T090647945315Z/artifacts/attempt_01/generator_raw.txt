@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<urn:nltl:stress-shape:TRF-085-ShipShape>
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/M> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "A propeller diameter in metres is required."
    ] ;
    sh:property [
        sh:path nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue <http://qudt.org/vocab/unit/REV-PER-MIN> ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The propeller speed at MCR in bollard condition is required in revolutions per minute."
    ] ;
    sh:property [
        sh:path nltl:bollardPropellerSpeedKnown ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "Whether the bollard propeller speed is known must be recorded."
    ] ;
    sh:property [
        sh:path nltl:hasTableLookupCase ;
        sh:node <urn:nltl:stress-shape:TRF-085-TableLookupCaseShape>
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue true
            ]
        ]
        [
            sh:property [
                sh:path nltl:bollardPropellerSpeedKnown ;
                sh:datatype xsd:boolean ;
                sh:hasValue false
            ] ;
            sh:property [
                sh:path nltl:hasTableLookupCase ;
                sh:minCount 1 ;
                sh:node <urn:nltl:stress-shape:TRF-085-TableLookupCaseShape>
            ]
        ]
    ) ;
    sh:message "If the bollard propeller speed is not known, a Table 6-9 lookup case is required." .

<urn:nltl:stress-shape:TRF-085-TableLookupCaseShape>
    a sh:NodeShape ;
    sh:class nltl:tableLookupCase ;
    sh:property [
        sh:path nltl:tableLookupApplied ;
        sh:datatype xsd:boolean ;
        sh:hasValue true ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The fallback case must record that a table lookup was applied."
    ] ;
    sh:property [
        sh:path nltl:tableReference ;
        sh:hasValue nltl:traficomTable6Dash9 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:message "The fallback case must reference TrafiCOM Table 6-9."
    ] ;
    sh:property [
        sh:path nltl:lookupInputQuantity ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:lookupResultQuantity ;
        sh:class qudt:QuantityValue ;
        sh:minCount 1 ;
        sh:node [
            a sh:NodeShape ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:lookupSelectionEvidence ;
        sh:class nltl:evidenceArtifact ;
        sh:minCount 1 ;
        sh:message "The table selection must have supporting evidence."
    ] ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "The Table 6-9 lookup input must equal the propeller diameter and its result must equal the recorded bollard propeller speed." ;
        sh:select """
PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
PREFIX qudt: <http://qudt.org/schema/qudt/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
SELECT $this
WHERE {
  ?ship nltl:hasTableLookupCase $this ;
        nltl:propellerDiameter ?diameter ;
        nltl:propellerRotationalSpeedAtMaximumContinuousRatingBollard ?speed .
  $this nltl:lookupInputQuantity ?input ;
        nltl:lookupResultQuantity ?result .
  ?input qudt:numericValue ?inputValue .
  ?diameter qudt:numericValue ?diameterValue .
  ?result qudt:numericValue ?resultValue .
  ?speed qudt:numericValue ?speedValue .
  FILTER (?inputValue != ?diameterValue || ?resultValue != ?speedValue)
}
"""
    ] .
<END_SHACL>
