<BEGIN_SHACL>
@prefix gen: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF086Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:propellerType ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:propellerType ;
                sh:hasValue "CP" ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:or (
                [
                    sh:property [
                        sh:path nltl:pitchValueKnown ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue true ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrBollard ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ]
                ]
                [
                    sh:property [
                        sh:path nltl:pitchValueKnown ;
                        sh:datatype xsd:boolean ;
                        sh:hasValue false ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrBollard ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:property [
                        sh:path nltl:propellerPitchAtMcrFreeRunning ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:node [
                            sh:class qudt:QuantityValue ;
                            sh:property [
                                sh:path qudt:numericValue ;
                                sh:datatype xsd:decimal ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ] ;
                            sh:property [
                                sh:path qudt:unit ;
                                sh:hasValue unit:M ;
                                sh:minCount 1 ;
                                sh:maxCount 1
                            ]
                        ]
                    ] ;
                    sh:sparql [
                        sh:message "For a CP propeller with an unknown bollard pitch, the MCR bollard pitch must equal 0.7 times the MCR free-running pitch within a relative tolerance of 1e-6." ;
                        sh:select """
                            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                            PREFIX qudt: <http://qudt.org/schema/qudt/>
                            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                            SELECT $this
                            WHERE {
                                $this nltl:propellerPitchAtMcrBollard ?bollard .
                                ?bollard qudt:numericValue ?bollardValue .
                                $this nltl:propellerPitchAtMcrFreeRunning ?freeRunning .
                                ?freeRunning qudt:numericValue ?freeRunningValue .
                                BIND(0.7 * ?freeRunningValue AS ?expectedValue)
                                FILTER(
                                    ABS(?bollardValue - ?expectedValue)
                                    >
                                    IF(
                                        ABS(?expectedValue) * 0.000001 > 0.000000001,
                                        ABS(?expectedValue) * 0.000001,
                                        0.000000001
                                    )
                                )
                            }
                        """
                    ]
                ]
            )
        ]
        [
            sh:sparql [
                sh:message "The propeller type must identify a non-CP propeller or the CP-propeller pitch condition must be satisfied." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:propellerType ?propellerType .
                        FILTER(?propellerType != "CP")
                    }
                """
            ]
        ]
    ) .
</BEGIN_SHACL>
