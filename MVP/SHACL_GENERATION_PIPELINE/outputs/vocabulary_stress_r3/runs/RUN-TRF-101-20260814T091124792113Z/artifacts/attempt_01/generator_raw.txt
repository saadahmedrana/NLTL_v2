<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix stress: <urn:nltl:stress-shape:> .

stress:QuantityValueM a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:M ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue qudt:KiloN ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] .

stress:QuantityValueUnitless a sh:NodeShape ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:property [

        sh:hasValue qudt:UNITLESS ;

stress:LoadCaseShape a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:bladeFailureLoadApplicationRadiusRatio ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueUnitless ;
        sh:sparql [
            sh:message "The blade failure load application radius ratio must be exactly 0.8." ;
            sh:select """
                PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                PREFIX qudt: <http://qudt.org/schema/qudt/>
                PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                SELECT $this
                WHERE {
                    $this nltl:bladeFailureLoadApplicationRadiusRatio ?q .
                    ?q qudt:numericValue ?value .
                    FILTER (?value != 0.8)
                }
            """
        ]
    ] ;
    sh:property [
        sh:path nltl:bladeFailureLoadWeakestDirectionConfirmed ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] .

stress:ApprovedAnalysisBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:approvedNonlinearBladeStressAnalysis ;
        sh:class nltl:approvalRecord ;
        sh:minCount 1 ;
        sh:message "An approved nonlinear blade stress analysis is required when the formula alternative is not used."
    ] .

stress:FormulaBranch a sh:NodeShape ;
    sh:property [
        sh:path nltl:bladeChordLengthAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:bladeMaximumThicknessAtWeakestSection ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:bladeCylindricalRootRadius ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:bladeMinimumYieldStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueMegaPA
    ] ;
    sh:property [
        sh:path nltl:bladeUltimateTensileStrength ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueMegaPA
    ] ;
    sh:property [
        sh:path nltl:bladeReferenceStrength1 ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueMegaPA
    ] ;
    sh:property [
        sh:path nltl:bladeFailureUltimateLoad ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueKiloN
    ] ;
    sh:sparql [
        sh:message "The reported blade reference strength and ultimate load must satisfy the specified formulas within a relative tolerance of 1e-6 plus an absolute tolerance of 1e-9; the load formula denominator must be positive." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeChordLengthAtWeakestSection ?cNode ;
                      nltl:bladeMaximumThicknessAtWeakestSection ?tNode ;
                      nltl:bladeCylindricalRootRadius ?rNode ;
                      nltl:propellerDiameter ?dNode ;
                      nltl:bladeMinimumYieldStrength ?yNode ;
                      nltl:bladeUltimateTensileStrength ?uNode ;
                      nltl:bladeReferenceStrength1 ?refNode ;
                      nltl:bladeFailureUltimateLoad ?loadNode .

                ?cNode qudt:numericValue ?c .
                ?tNode qudt:numericValue ?t .
                ?rNode qudt:numericValue ?r .
                ?dNode qudt:numericValue ?d .
                ?yNode qudt:numericValue ?y .
                ?uNode qudt:numericValue ?u .
                ?refNode qudt:numericValue ?ref .
                ?loadNode qudt:numericValue ?load .

                BIND ((0.6 * ?y) + (0.4 * ?u) AS ?expectedRef)
                BIND (0.8 * ?d - 2 * ?r AS ?denominator)
                BIND ((300 * ?c * ?t * ?t * ?expectedRef) / ?denominator AS ?expectedLoad)

                FILTER (
                    ?denominator <= 0 ||
                    ABS(?ref - ?expectedRef) > (ABS(?expectedRef) * 0.000001 + 0.000000001) ||
                    ABS(?load - ?expectedLoad) > (ABS(?expectedLoad) * 0.000001 + 0.000000001)
                )
            }
        """
    ] .

stress:TRF101ShipShape a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:hasLoadCase ;
        sh:minCount 1 ;
        sh:node stress:LoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:bladeTipOffset ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:propellerDiameter ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node stress:QuantityValueM
    ] ;
    sh:property [
        sh:path nltl:bladeFailureStatus ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        stress:ApprovedAnalysisBranch
        stress:FormulaBranch
    ) ;
    sh:sparql [
        sh:message "Blade failure status must be true exactly when blade tip offset exceeds 10 percent of propeller diameter." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
            SELECT $this
            WHERE {
                $this nltl:bladeTipOffset ?offsetNode ;
                      nltl:propellerDiameter ?diameterNode ;
                      nltl:bladeFailureStatus ?status .

                ?offsetNode qudt:numericValue ?offset .
                ?diameterNode qudt:numericValue ?diameter .

                FILTER (
                    (?offset > (0.10 * ?diameter) && ?status != true) ||
                    (?offset <= (0.10 * ?diameter) && ?status != false)
                )
            }
        """
    ] .
<END_SHACL>
