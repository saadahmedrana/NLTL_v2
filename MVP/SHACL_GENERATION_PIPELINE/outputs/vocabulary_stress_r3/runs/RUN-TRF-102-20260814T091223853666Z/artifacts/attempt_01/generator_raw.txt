<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .
@prefix stress: <urn:nltl:stress-shape:> .

stress:TRF102Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:or (
        [
            sh:property [
                sh:path nltl:approvedSpindleTorqueStressAnalysis ;
                sh:class nltl:approvalRecord ;
                sh:minCount 1 ;
                sh:message "An approved spindle torque stress analysis is required when the direct-analysis alternative is used."
            ]
        ]
        [
            sh:property [
                sh:path nltl:bladeExpandedAreaRatio ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:leadingEdgeChordPortionAtZeroPointEightRadius ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:trailingEdgeChordPortionAtZeroPointEightRadius ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:extremeIceForce ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:bladeFailureSpindleTorqueFactor ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1 ;
                        sh:minInclusive "0.3"^^xsd:decimal
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:UNITLESS ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:property [
                sh:path nltl:bladeFailureSpindleTorque ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:node [
                    sh:class qudt:QuantityValue ;
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:datatype xsd:decimal ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ] ;
                    sh:property [
                        sh:path qudt:unit ;
                        sh:hasValue unit:KiloN-M ;
                        sh:minCount 1 ;
                        sh:maxCount 1
                    ]
                ]
            ] ;
            sh:sparql [
                sh:message "Blade failure spindle torque must equal max(C_LE0.8, 0.8*C_TE0.8) multiplied by C_spex and F_ex, within a relative tolerance of 1e-6 and an absolute fallback of 1e-9." ;
                sh:select """
                    PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
                    PREFIX qudt: <http://qudt.org/schema/qudt/>
                    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
                    SELECT $this
                    WHERE {
                        $this nltl:leadingEdgeChordPortionAtZeroPointEightRadius ?leNode ;
                              nltl:trailingEdgeChordPortionAtZeroPointEightRadius ?teNode ;
                              nltl:bladeFailureSpindleTorqueFactor ?factorNode ;
                              nltl:extremeIceForce ?forceNode ;
                              nltl:bladeFailureSpindleTorque ?torqueNode .
                        ?leNode qudt:numericValue ?le .
                        ?teNode qudt:numericValue ?te .
                        ?factorNode qudt:numericValue ?factor .
                        ?forceNode qudt:numericValue ?force .
                        ?torqueNode qudt:numericValue ?torque .
                        BIND(IF(?le > (0.8 * ?te), ?le, (0.8 * ?te)) AS ?lever)
                        BIND(?lever * ?factor * ?force AS ?expected)
                        FILTER(ABS(?torque - ?expected) > ((ABS(?expected) * 0.000001) + 0.000000001))
                    }
                """
            ]
        ]
    ] .
<END_SHACL>
