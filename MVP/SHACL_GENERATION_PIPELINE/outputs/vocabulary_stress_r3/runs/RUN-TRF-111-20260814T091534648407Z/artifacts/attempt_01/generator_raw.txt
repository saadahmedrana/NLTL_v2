<BEGIN_SHACL>
@prefix g: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

g:TRF111ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:extremeIceForce ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloN ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "Extreme ice force must be supplied as a single force quantity in kilonewtons."
    ] ;
    sh:property [
        sh:path nltl:hasPropulsionLoadCase ;
        sh:node g:TRF111LoadCaseShape
    ] ;
    sh:property [
        sh:path nltl:hasPropellerShaftLineComponent ;
        sh:node g:TRF111ComponentShape
    ] .

g:TRF111LoadCaseShape
    a sh:NodeShape ;
    sh:class nltl:loadCase ;
    sh:property [
        sh:path nltl:shaftLineLoadCaseType ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:or (
            [
                sh:hasValue nltl:extremeOperationalLoadCase ;
                sh:property [
                    sh:path nltl:componentYieldSafetyFactor ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node g:TRF111YieldFactorAtLeast13Shape
                ]
            ]
            [
                sh:hasValue nltl:bladeFailureLoadCase ;
                sh:property [
                    sh:path nltl:componentYieldSafetyFactor ;
                    sh:minCount 1 ;
                    sh:maxCount 1 ;
                    sh:node g:TRF111YieldFactorAtLeast10Shape
                ]
            ]
            [
                sh:hasValue nltl:fatigueLoadCase
            ]
        ) ;
        sh:message "Each propulsion load case must identify an allowed shaft-line load-case type and apply its required safety-factor rule."
    ] .

g:TRF111YieldFactorAtLeast13Shape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.3"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:message "The extreme-operational-load yielding safety factor must be at least 1.3."
    .

g:TRF111YieldFactorAtLeast10Shape
    a sh:NodeShape ;
    sh:class qudt:QuantityValue ;
    sh:property [
        sh:path qudt:numericValue ;
        sh:datatype xsd:decimal ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:minInclusive "1.0"^^xsd:decimal
    ] ;
    sh:property [
        sh:path qudt:unit ;
        sh:hasValue unit:UNITLESS ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:message "The blade-failure-load yielding safety factor must be at least 1.0."
    .

g:TRF111ComponentShape
    a sh:NodeShape ;
    sh:class nltl:shipComponent ;
    sh:property [
        sh:path nltl:fatigueSafetyFactor ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:minInclusive "1.5"^^xsd:decimal
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:UNITLESS ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ] ;
        sh:message "The fatigue-load safety factor for each linked shaft-line component must be at least 1.5."
    ] .
<END_SHACL>
