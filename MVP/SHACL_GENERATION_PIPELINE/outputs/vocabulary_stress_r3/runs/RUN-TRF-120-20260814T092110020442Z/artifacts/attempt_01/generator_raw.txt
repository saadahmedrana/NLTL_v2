<BEGIN_SHACL>
@prefix gen: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix sh: <http://www.w3.org/ns/shacl#> .

gen:TRF120ShipShape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:sparql [
        a sh:SPARQLConstraint ;
        sh:message "Each represented thruster natural-frequency calculation case must declare a longitudinal or transverse direction and use damping, water added mass, and ship attachment stiffness inputs owned by the appropriate nodes." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX unit: <http://qudt.org/vocab/unit/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:hasComponent ?body .
                ?body rdf:type nltl:thrusterBody .
                ?body nltl:hasNaturalFrequencyCalculationCase ?case .
                ?case rdf:type nltl:calculationCase .

                FILTER (
                    NOT EXISTS {
                        ?case nltl:vibrationDirection ?direction .
                    }
                    ||
                    EXISTS {
                        ?case nltl:vibrationDirection ?direction .
                        FILTER (
                            ?direction NOT IN (
                                nltl:longitudinalDirection,
                                nltl:transverseDirection
                            )
                        )
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:damping ?damping .
                        FILTER (datatype(?damping) = xsd:string)
                    }
                    ||
                    EXISTS {
                        $this nltl:damping ?damping .
                        FILTER (datatype(?damping) != xsd:string)
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:waterAddedMass ?waterMass .
                        ?waterMass qudt:numericValue ?waterValue ;
                                   qudt:unit unit:KiloGM .
                        FILTER (datatype(?waterValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:waterAddedMass ?waterMass .
                        {
                            ?waterMass qudt:numericValue ?waterValue .
                            FILTER (datatype(?waterValue) != xsd:decimal)
                        }
                        UNION
                        {
                            ?waterMass qudt:unit ?waterUnit .
                            FILTER (?waterUnit != unit:KiloGM)
                        }
                    }
                )

                FILTER (
                    NOT EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency .
                        ?frequency qudt:numericValue ?frequencyValue ;
                                   qudt:unit unit:HZ .
                        FILTER (datatype(?frequencyValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        $this nltl:thrusterNaturalFrequency ?frequency .
                        {
                            ?frequency qudt:numericValue ?frequencyValue .
                            FILTER (datatype(?frequencyValue) != xsd:decimal)
                        }
                        UNION
                        {
                            ?frequency qudt:unit ?frequencyUnit .
                            FILTER (?frequencyUnit != unit:HZ)
                        }
                    }
                )

                FILTER (
                    NOT EXISTS {
                        ?body nltl:shipAttachmentStiffness ?stiffness .
                        ?stiffness qudt:numericValue ?stiffnessValue ;
                                   qudt:unit unit:KiloN-PER-M .
                        FILTER (datatype(?stiffnessValue) = xsd:decimal)
                    }
                    ||
                    EXISTS {
                        ?body nltl:shipAttachmentStiffness ?stiffness .
                        {
                            ?stiffness qudt:numericValue ?stiffnessValue .
                            FILTER (datatype(?stiffnessValue) != xsd:decimal)
                        }
                        UNION
                        {
                            ?stiffness qudt:unit ?stiffnessUnit .
                            FILTER (?stiffnessUnit != unit:KiloN-PER-M)
                        }
                    }
                )
            }
        """ ;
    ] .
<END_SHACL>
