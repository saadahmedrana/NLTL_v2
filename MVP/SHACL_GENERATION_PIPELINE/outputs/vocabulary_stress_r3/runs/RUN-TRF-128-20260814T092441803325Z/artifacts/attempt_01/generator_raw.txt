<BEGIN_SHACL>
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

<nltl:TRF128Shape> a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:iceClass ;
        sh:datatype nltl:iceClassValue ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:in (
            nltl:iceClassIa
            nltl:iceClassIaSuper
            nltl:iceClassIb
            nltl:iceClassIc
            nltl:iceClassIi
            nltl:iceClassIii
        ) ;
        sh:message "A permitted ice class is required."
    ] ;
    sh:property [
        sh:path nltl:airReceiverCapacity ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/M3>
            ]
        ] ;
        sh:message "Air receiver capacity must be provided as a normal-volume quantity in cubic metres."
    ] ;
    sh:property [
        sh:path nltl:compressorChargeTime ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:minCount 1 ;
                sh:maxCount 1 ;
                sh:hasValue <http://qudt.org/vocab/unit/HR>
            ]
        ] ;
        sh:message "Compressor charge time must be provided as a time quantity in hours."
    ] ;
    sh:or (
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue true ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "0.5"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:hasValue nltl:iceClassIaSuper
            ] ;
            sh:property [
                sh:path nltl:propulsionEngineReversalRequiredForAstern ;
                sh:hasValue false ;
                sh:datatype xsd:boolean ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "1.0"^^xsd:decimal
                    ]
                ]
            ]
        ]
        [
            sh:property [
                sh:path nltl:iceClass ;
                sh:in (
                    nltl:iceClassIa
                    nltl:iceClassIb
                    nltl:iceClassIc
                    nltl:iceClassIi
                    nltl:iceClassIii
                )
            ] ;
            sh:property [
                sh:path nltl:compressorChargeTime ;
                sh:node [
                    sh:property [
                        sh:path qudt:numericValue ;
                        sh:maxInclusive "1.0"^^xsd:decimal
                    ]
                ]
            ]
        ]
    ) ;
    sh:message "Compressor charge time must not exceed 0.5 hours for an IA Super ship requiring propulsion-engine reversal for astern, and must not exceed 1 hour otherwise." .
<END_SHACL>
