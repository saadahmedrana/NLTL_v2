<BEGIN_SHACL>
@prefix gen: <urn:nltl:stress-shape:> .
@prefix nltl: <https://w3id.org/nltl-benchmark/vocab#> .
@prefix qudt: <http://qudt.org/schema/qudt/> .
@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

gen:TRF_130_Shape
    a sh:NodeShape ;
    sh:targetClass nltl:ship ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1
    ] ;
    sh:or (
        gen:TRF_130_NormalArrangementShape
        gen:TRF_130_AlternativeArrangementShape
    ) .

gen:TRF_130_NormalArrangementShape
    a sh:NodeShape ;
    sh:property [
        sh:path nltl:coolingWaterChestVolumeAndHeightRequirementsCannotBeMet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue false
    ] ;
    sh:property [
        sh:path nltl:seaInletLocation ;
        sh:datatype xsd:string ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue "near centreline and well aft if possible"
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:iceAccumulationSpaceAboveInlet ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:fullCapacityDischargePipeConnected ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:hasValue true
    ] ;
    sh:property [
        sh:path nltl:recommendedChestVolume ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M3 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:totalRelevantEnginePower ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:KiloW ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:strainerOpenArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:property [
        sh:path nltl:inletPipeArea ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;
            sh:property [
                sh:path qudt:numericValue ;
                sh:datatype xsd:decimal ;
                sh:minCount 1 ;
                sh:maxCount 1
            ] ;
            sh:property [
                sh:path qudt:unit ;
                sh:hasValue unit:M2 ;
                sh:minCount 1 ;
                sh:maxCount 1
            ]
        ]
    ] ;
    sh:sparql [
        sh:message "The recommended chest volume must be approximately total relevant engine power divided by 750, and the strainer open area must be at least four times the inlet pipe area." ;
        sh:select """
            PREFIX nltl: <https://w3id.org/nltl-benchmark/vocab#>
            PREFIX qudt: <http://qudt.org/schema/qudt/>
            PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

            SELECT $this
            WHERE {
                $this nltl:recommendedChestVolume ?recommendedNode ;
                      nltl:totalRelevantEnginePower ?powerNode ;
                      nltl:strainerOpenArea ?strainerNode ;
                      nltl:inletPipeArea ?inletNode .

                ?recommendedNode qudt:numericValue ?recommendedValue ;
                                 qudt:unit <http://qudt.org/vocab/unit/M3> .
                ?powerNode qudt:numericValue ?powerValue ;
                            qudt:unit <http://qudt.org/vocab/unit/KiloW> .
                ?strainerNode qudt:numericValue ?strainerValue ;
                               qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?inletNode qudt:numericValue ?inletValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M2> .

                FILTER (
                    ABS(?recommendedValue - (?powerValue / 750))
                        > (0.000001 * ABS(?powerValue / 750) + 0.000000001)
                    ||
                    ?strainerValue < (4 * ?inletValue)
                )
            }
        """
    ] .

    sh:property [

        sh:maxCount 1 ;
        sh:hasValue "near centreline and well aft if possible"
    ] ;
    sh:property [
        sh:path nltl:nearCentrelineAndWellAftIfPossible ;
        sh:datatype xsd:boolean ;
        sh:minCount 1 ;
        sh:maxCount 1 ;

    sh:property [

    sh:property [
        sh:path nltl:strainerOpenArea ;

        sh:maxCount 1 ;
        sh:node [
            sh:class qudt:QuantityValue ;

                $this nltl:strainerOpenArea ?strainerNode ;
                      nltl:inletPipeArea ?inletNode .

                               qudt:unit <http://qudt.org/vocab/unit/M2> .
                ?inletNode qudt:numericValue ?inletValue ;
                            qudt:unit <http://qudt.org/vocab/unit/M2> .

                FILTER (?strainerValue < (4 * ?inletValue))
            }
        """
    ] .
